228{
229 if(a.pinfo.size() == b.pinfo.size()) {
230 for(int i = 0; i < a.pinfo.size(); ++i) {
231 if(!a.pinfo.at(i).isApprox(b.pinfo.at(i))) {
232 return false;
233 }
234 }
235 } else {
236 return false;
237 }
238
239 return (a.type == b.type &&
240 a.id == b.id &&
241 a.np == b.np &&
242 a.ntri == b.ntri &&

244 a.rr.isApprox(b.rr, 0.0001f) &&

252 a.nearest.isApprox(b.nearest) &&
253 a.nearest_dist.isApprox(b.nearest_dist, 0.0001) &&
254 a.patch_inds.isApprox(b.patch_inds) &&
255 //a.dist_limit == b.dist_limit && //TODO: We still not sure if dist_limit can also be a matrix. This needs to be debugged
256 a.dist.isApprox(b.dist, 0.0001) &&
257 a.tri_cent.isApprox(b.tri_cent, 0.0001) &&
258 a.tri_nn.isApprox(b.tri_nn, 0.0001) &&
259 a.tri_area.isApprox(b.tri_area, 0.0001) &&
260 a.use_tri_cent.isApprox(b.use_tri_cent, 0.0001) &&
261 a.use_tri_nn.isApprox(b.use_tri_nn, 0.0001) &&
262 a.use_tri_area.isApprox(b.use_tri_area, 0.0001) &&
263 [&a, &b]() {
264 if (a.neighbor_tri.size() != b.neighbor_tri.size()) return false;
265 for (size_t i = 0; i < a.neighbor_tri.size(); ++i) {
266 if (a.neighbor_tri[i].size() != b.neighbor_tri[i].size()) return false;
267 if (a.neighbor_tri[i].size() > 0 && !(a.neighbor_tri[i].array() == b.neighbor_tri[i].array()).all()) return false;
268 }
269 return true;

272 if (a.neighbor_vert.size() != b.neighbor_vert.size()) return false;

276 }
