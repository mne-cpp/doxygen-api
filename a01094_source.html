78int uniqueChannel(const QList<FiffChInfo>& chs, const QString& name)
79{
80 int found = -1;
81 for (int i = 0; i < chs.size(); ++i) {
82 if (chs[i].ch_name == name) {
83 if (found >= 0) {
84 qWarning("Ambiguous channel %s", name.toUtf8().constData());
85 return -1;
86 }
87 found = i;
88 }
89 }
90 if (found < 0)
91 qWarning("Channel %s is not available in data", name.toUtf8().constData());
92 return found;

97bool FiffCtfComp::calibrate(const QList<FiffChInfo>& chs)
98{
99 rowcals = MatrixXd::Ones(data->data.rows(), 1);
100 colcals = MatrixXd::Ones(data->data.cols(), 1);
101 if (save_calibrated)
102 return true;
103 for (Index col = 0; col < data->data.cols(); ++col) {
104 const int p = uniqueChannel(chs, data->col_names.value(col));
105 if (p < 0)
106 return false;
107 colcals(col) = 1.0 / (static_cast<double>(chs[p].range) * chs[p].cal);
108 }
109 for (Index row = 0; row < data->data.rows(); ++row) {
110 const int p = uniqueChannel(chs, data->row_names.value(row));
111 if (p < 0)
112 return false;
113 rowcals(row) = static_cast<double>(chs[p].range) * chs[p].cal;
114 }
115 data->data = rowcals.asDiagonal() * data->data * colcals.asDiagonal();
116 return true;
117}
