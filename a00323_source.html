186template<typename T>
187inline bool CircularBuffer<T>::push(const T* pArray, unsigned int size)
188{
189 if(m_bPause) {
190 return false;
191 }
192
193 if(m_pFreeElements->tryAcquire(size, m_iTimeout)) {
194 for(unsigned int i = 0; i < size; ++i) {
195 m_pBuffer[mapIndex(m_iCurrentWriteIndex)] = pArray[i];
196 }
197 const QSemaphoreReleaser releaser(m_pUsedElements, size);
198 } else {
199 return false;
200 }
201
202 return true;
203}
