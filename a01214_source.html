137std::unique_ptr<MneMshDisplaySurfaceSet> MneMshDisplaySurfaceSet::load(const QString &subject_id, const QString &surf, const QString &subjects_dir)
138 /*
139 * Load new display surface data
140 */
141{
142 QString pathLh = QString("%1/%2/surf/%3.%4").arg(subjects_dir).arg(subject_id).arg("lh").arg(surf);
143 QString pathLhCurv = QString("%1/%2/surf/%3.%4").arg(subjects_dir).arg(subject_id).arg("lh").arg("curv");
144 QString pathRh = QString("%1/%2/surf/%3.%4").arg(subjects_dir).arg(subject_id).arg("rh").arg(surf);
145 QString pathRhCurv = QString("%1/%2/surf/%3.%4").arg(subjects_dir).arg(subject_id).arg("rh").arg("curv");
146
147 printf("Loading surface %s ...\n", pathLh.toUtf8().constData());
148 std::unique_ptr<MneSourceSpaceOld> left(MneSurfaceOrVolume::load_surface(pathLh, pathLhCurv));
149 if (!left) {
150 left.reset(MneSurfaceOrVolume::load_surface(pathLh, QString()));
151 if (!left)
152 return nullptr;
153 MneSurfaceOrVolume::add_uniform_curv(reinterpret_cast<MneSurfaceOld&>(*left));
154 }
155
156 printf("Loading surface %s ...\n", pathRh.toUtf8().constData());
157 std::unique_ptr<MneSourceSpaceOld> right(MneSurfaceOrVolume::load_surface(pathRh, pathRhCurv));
158 if (!right) {
159 right.reset(MneSurfaceOrVolume::load_surface(pathRh, QString()));
160 if (!right)
161 return nullptr;
162 MneSurfaceOrVolume::add_uniform_curv(reinterpret_cast<MneSurfaceOld&>(*right));
163 }
164
165 auto result = std::make_unique<MneMshDisplaySurfaceSet>(2);
166
167 result->surfs[0] = std::make_unique<MneMshDisplaySurface>();
168 result->surfs[1] = std::make_unique<MneMshDisplaySurface>();
169
170 result->active[0] = true;
171 result->active[1] = false;
172 result->drawable[0] = true;
173 result->drawable[1] = true;
174
175 auto* pThis = result->surfs[0].get();
176 pThis->filename = pathLh;
177 pThis->s.reset(reinterpret_cast<MneSurfaceOld*>(left.release()));
178 pThis->s->id = SURF_LEFT_HEMI;
179 pThis->subj = subject_id;
180 pThis->surf_name = surf;
181
182 decide_surface_extent(*pThis,"Left hemisphere");
183 decide_curv_display(surf, *pThis);
184 setup_curvature_colors(*pThis);
185
186 pThis = result->surfs[1].get();
187 pThis->filename = pathRh;
188 pThis->s.reset(reinterpret_cast<MneSurfaceOld*>(right.release()));
189 pThis->s->id = SURF_RIGHT_HEMI;
190 pThis->subj = subject_id;
191 pThis->surf_name = surf;
192
193 decide_surface_extent(*pThis,"Right hemisphere");
194 decide_curv_display(surf, *pThis);
195 setup_curvature_colors(*pThis);
196
197 result->apply_left_right_eyes();
198 result->setup_current_lights();
199
200 return result;
201}

205void MneMshDisplaySurfaceSet::decide_surface_extent(MneMshDisplaySurface& surf,
206 const QString& tag)
207
208{
209 auto* s = reinterpret_cast<MneSourceSpaceOld*>(surf.s.get());
210
211 Eigen::Vector3f minv = s->rr.row(0).transpose();
212 Eigen::Vector3f maxv = minv;
213
214 for (int k = 1; k < s->np; k++) {
215 Eigen::Vector3f r = s->rr.row(k).transpose();
216 minv = minv.cwiseMin(r);
217 maxv = maxv.cwiseMax(r);
218 }
219
220#ifdef DEBUG
221 printf("%s:\n",tag.toUtf8().constData());
222 printf("\tx = %f ... %f mm\n",1000*minv[0],1000*maxv[0]);
223 printf("\ty = %f ... %f mm\n",1000*minv[1],1000*maxv[1]);
224 printf("\tz = %f ... %f mm\n",1000*minv[2],1000*maxv[2]);
225#endif
226
227 surf.fov = std::max(minv.cwiseAbs().maxCoeff(), maxv.cwiseAbs().maxCoeff());
228
229 surf.minv = minv;
230 surf.maxv = maxv;
231 surf.fov_scale = 1.1f;
232}

249int MneMshDisplaySurfaceSet::add_bem_surface(const QString& filepath,
250 int kind,
251 const QString& bemname,
252 int full_geom,
253 int check)
254{
255 printf("Loading BEM surface %s (id = %d) from %s ...\n",
256 bemname.toUtf8().constData(), kind, filepath.toUtf8().constData());
257
258 std::unique_ptr<MneSurfaceOld> surf(MneSurfaceOld::read_bem_surface2(filepath,kind,full_geom,nullptr));
259 if (!surf)
260 return -1;
261
262 if (check) {
263 MneSurfaceOld::compute_surface_cm(*surf);
264 double sum = MneSurfaceOld::sum_solids(Eigen::Map<const Eigen::Vector3f>(surf->cm), *surf) / (4*M_PI);
265 if (std::fabs(sum - 1.0) > 1e-4) {
266 printf( "%s surface is not closed "
267 "(sum of solid angles = %g * 4*PI).",
268 bemname.toUtf8().constData(), sum);
269 return -1;
270 }
271 }
272
273 auto newSurf = std::make_unique<MneMshDisplaySurface>();
274 newSurf->filename = filepath;
275 //newSurf->time_loaded = time(nullptr); //Comment out due to unknown timestamp function ToDo
276 newSurf->s.reset(surf.release());
277 newSurf->s->id = kind;
278 newSurf->surf_name = bemname;
279
280 newSurf->curvature_color_mode = SHOW_CURVATURE_NONE;
281 newSurf->overlay_color_mode = SHOW_OVERLAY_HEAT;
282
283 decide_surface_extent(*newSurf, bemname);
284 add_replace_surface(std::move(newSurf), true, true);
285 apply_left_eyes();
286 setup_current_lights();
287
288 return 0;
289}

293void MneMshDisplaySurfaceSet::add_replace_surface(std::unique_ptr<MneMshDisplaySurface> newSurf,
294 bool replace,
295 bool drawable)
296{
297 if (replace) {
298 for (int k = 0; k < nsurf; k++) {
299 auto& surf = surfs[k];
300 if (surf->s->id == newSurf->s->id) {
301 newSurf->transparent = surf->transparent;
302 newSurf->show_aux_data = surf->show_aux_data;
303 surfs[k] = std::move(newSurf);
304 if (!drawable) {
305 active[k] = false;
306 this->drawable[k] = false;
307 }
308 return;
309 }
310 }
311 }
312 if (newSurf) { /* New surface */
313 surfs.push_back(std::move(newSurf));
314 patches.push_back(nullptr);
315 patch_rot.push_back(0.0f);
316 active.conservativeResize(nsurf+1);
317 this->drawable.conservativeResize(nsurf+1);
318 active[nsurf] = drawable;
319 this->drawable[nsurf] = drawable;
320 nsurf++;
321 }
322}

326void MneMshDisplaySurfaceSet::setup_curvature_colors(MneMshDisplaySurface& surf)
327{
328 if (!surf.s)
329 return;
330
331 auto* s = reinterpret_cast<MneSourceSpaceOld*>(surf.s.get());
332
333 const int ncolor = surf.nvertex_colors;
334 const int totalSize = ncolor * s->np;
335
336 if (surf.vertex_colors.size() == 0)
337 surf.vertex_colors.resize(totalSize);
338
339 float curv_sum = 0.0f;
340 if (surf.curvature_color_mode == SHOW_CURVATURE_OVERLAY) {
341 for (int k = 0; k < s->np; k++) {
342 const int base = k * ncolor;
343 curv_sum += std::fabs(s->curv[k]);
344 const float c = (s->curv[k] > 0) ? POS_CURV_COLOR : NEG_CURV_COLOR;
345 for (int j = 0; j < 3; j++)
346 surf.vertex_colors[base + j] = c;
347 if (ncolor == 4)
348 surf.vertex_colors[base + 3] = 1.0f;
349 }
350 }
351 else {
352 for (int k = 0; k < s->np; k++) {
353 const int base = k * ncolor;
354 curv_sum += std::fabs(s->curv[k]);
355 for (int j = 0; j < 3; j++)
356 surf.vertex_colors[base + j] = EVEN_CURV_COLOR;
357 if (ncolor == 4)
358 surf.vertex_colors[base + 3] = 1.0f;
359 }
360 }
361#ifdef DEBUG
362 printf("Average curvature : %f\n",curv_sum/s->np);
363#endif
364}

368void MneMshDisplaySurfaceSet::apply_left_right_eyes()
369{
370 MneMshEyes* eyes = nullptr;
371
372 if (neyes == 0 || current_eyes < 0 || current_eyes > neyes-1) {
373 eyes = &default_eyes;
374 } else {
375 eyes = all_eyes+current_eyes;
376 }
377
378 for (int k = 0; k < nsurf; k++) {
379 auto* surf = surfs[k].get();
380 switch(surf->s->id) {
381 case SURF_LEFT_HEMI :
382 case SURF_LEFT_MORPH_HEMI :
383 surf->eye = Eigen::Vector3f::Map(eyes->left);
384 surf->up = Eigen::Vector3f::Map(eyes->left_up);
385 break;
386 case SURF_RIGHT_HEMI :
387 case SURF_RIGHT_MORPH_HEMI :
388 surf->eye = Eigen::Vector3f::Map(eyes->right);
389 surf->up = Eigen::Vector3f::Map(eyes->right_up);
390 break;
391 default :
392 surf->eye = Eigen::Vector3f::Map(eyes->left);
393 surf->up = Eigen::Vector3f::Map(eyes->left_up);
394 break;
395 }
396 }
397}

401void MneMshDisplaySurfaceSet::apply_left_eyes()
402{
403 for (int k = 0; k < nsurf; k++) {
404 if (neyes == 0 || current_eyes < 0 || current_eyes > neyes-1) {
405 surfs[k]->eye = Eigen::Vector3f::Map(default_eyes.left);
406 surfs[k]->up = Eigen::Vector3f::Map(default_eyes.left_up);
407 }
408 else {
409 surfs[k]->eye = Eigen::Vector3f::Map(all_eyes[current_eyes].left);
410 surfs[k]->up = Eigen::Vector3f::Map(all_eyes[current_eyes].left_up);
411 }
