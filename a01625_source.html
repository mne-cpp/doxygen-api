92MnaProject MnaIO::readJson(const QString& path)
93{
94 QFile file(path);
95 if(!file.open(QIODevice::ReadOnly)) {
96 qWarning() << "[MnaIO::readJson] Cannot open file:" << path;
97 return MnaProject();
98 }
99
100 const QByteArray data = file.readAll();
101 file.close();
102
103 QJsonParseError parseError;
104 const QJsonDocument doc = QJsonDocument::fromJson(data, &parseError);
105 if(parseError.error != QJsonParseError::NoError) {
106 qWarning() << "[MnaIO::readJson] Parse error:" << parseError.errorString();
107 return MnaProject();
108 }
109
110 const QJsonObject root = doc.object();
111
112 // Version check: reject if major version differs
113 const QString fileVersion = root[QLatin1String("mna_version")].toString();
114 const int fileMajor = fileVersion.section(QLatin1Char('.'), 0, 0).toInt();
115 const int currentMajor = QString::fromLatin1(MnaProject::CURRENT_SCHEMA_VERSION).section(QLatin1Char('.'), 0, 0).toInt();
116 if(fileMajor != currentMajor) {
117 qWarning() << "[MnaIO::readJson] Incompatible major version:" << fileVersion
118 << "(expected major" << currentMajor << ")";
119 return MnaProject();
120 }
121
122 return MnaProject::fromJson(root);
123}

127bool MnaIO::writeJson(const MnaProject& project, const QString& path)
128{
129 QFile file(path);
130 if(!file.open(QIODevice::WriteOnly | QIODevice::Truncate)) {
131 qWarning() << "[MnaIO::writeJson] Cannot open file for writing:" << path;
132 return false;
133 }
134
135 const QJsonDocument doc(project.toJson());
136 const QByteArray data = doc.toJson(QJsonDocument::Indented);
137 const qint64 written = file.write(data);
138 file.close();
139
140 return (written == data.size());
141}

145MnaProject MnaIO::readCbor(const QString& path)
146{
147 QFile file(path);
148 if(!file.open(QIODevice::ReadOnly)) {
149 qWarning() << "[MnaIO::readCbor] Cannot open file:" << path;
150 return MnaProject();
151 }
152
153 // Verify magic header
154 const QByteArray magic = file.read(MNX_MAGIC.size());
155 if(magic != MNX_MAGIC) {
156 qWarning() << "[MnaIO::readCbor] Invalid magic header in:" << path;
157 file.close();
158 return MnaProject();
159 }
160
161 const QByteArray cborData = file.readAll();
162 file.close();
163
164 const QCborValue cborVal = QCborValue::fromCbor(cborData);
165 if(!cborVal.isMap()) {
166 qWarning() << "[MnaIO::readCbor] Root CBOR value is not a map";
167 return MnaProject();
168 }
169
170 const QCborMap root = cborVal.toMap();
171
172 // Version check: reject if major version differs
173 const QString fileVersion = root[QLatin1String("mna_version")].toString();
174 const int fileMajor = fileVersion.section(QLatin1Char('.'), 0, 0).toInt();
175 const int currentMajor = QString::fromLatin1(MnaProject::CURRENT_SCHEMA_VERSION).section(QLatin1Char('.'), 0, 0).toInt();
176 if(fileMajor != currentMajor) {
177 qWarning() << "[MnaIO::readCbor] Incompatible major version:" << fileVersion
178 << "(expected major" << currentMajor << ")";
179 return MnaProject();
180 }
181
182 return MnaProject::fromCbor(root);
183}

187bool MnaIO::writeCbor(const MnaProject& project, const QString& path)
188{
189 QFile file(path);
190 if(!file.open(QIODevice::WriteOnly | QIODevice::Truncate)) {
191 qWarning() << "[MnaIO::writeCbor] Cannot open file for writing:" << path;
192 return false;
193 }
194
195 // Write magic header
196 if(file.write(MNX_MAGIC) != MNX_MAGIC.size()) {
197 qWarning() << "[MnaIO::writeCbor] Failed to write magic header";
198 file.close();
199 return false;
200 }
201
202 const QCborValue cborVal(project.toCbor());
203 const QByteArray cborData = cborVal.toCbor();
204 const qint64 written = file.write(cborData);
205 file.close();
206
207 return (written == cborData.size());
208}
