59TFplot::TFplot(const MatrixXd& tf_matrix,
60 qreal sample_rate,
61 qreal lower_frq,
62 qreal upper_frq,
63 ColorMaps cmap,
64 qreal tmin)
65{
66 if (tf_matrix.rows() == 0 || tf_matrix.cols() == 0 || sample_rate <= 0) {
67 return;
68 }
69
70 // Row k is bin k of an FFT over all N columns
71 const qreal binWidth = sample_rate / tf_matrix.cols();
72 const Index last = tf_matrix.rows() - 1;
73 const Index first = std::clamp<Index>(static_cast<Index>(std::ceil(std::min(lower_frq, upper_frq) / binWidth)), 0, last);
74 const Index top = std::clamp<Index>(static_cast<Index>(std::floor(std::max(lower_frq, upper_frq) / binWidth)), first, last);
75
76 // Highest frequency in the first row, drawn at the top
77 MatrixXd power = tf_matrix.middleRows(first, top - first + 1).colwise().reverse().cwiseAbs();
78 const double maxPower = power.maxCoeff();
79 if (maxPower > 0.0) {
80 power /= maxPower;
81 }
82
83 setColorMap(colorMapName(cmap));
84 setColorRange(0.0, 1.0);
85 updateData(power);
86 setExtent(tmin, tmin + (tf_matrix.cols() - 1) / sample_rate, first * binWidth, top * binWidth);
87 setXLabel(QStringLiteral("Time [s]"));
88 setYLabel(QStringLiteral("Frequency [Hz]"));
89 setColorbarLabel(QStringLiteral("Power (normalized)"));
90}
