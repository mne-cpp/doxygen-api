46VectorXi rows_where(const std::vector<FiffChInfo>& chs, bool (*pick)(const FiffChInfo&))
47{
48 std::vector<int> rows;
49 rows.reserve(chs.size());
50 for (int r = 0; r < std::ssize(chs); ++r) {
51 if (pick(chs[static_cast<std::size_t>(r)])) {
52 rows.push_back(r);
53 }
54 }
55 return Map<const VectorXi>(rows.data(), static_cast<Index>(rows.size()));
56}

64void RtRawDisplay::set_info(std::shared_ptr<const FiffInfo> info)
65{
66 m_info = std::move(info);
67 const int channels = m_info ? static_cast<int>(m_info->chs.size()) : 0;
68
69 m_correction.reset(channels);
70 m_sphara = SparseMatrix<double>(channels, channels);
71 m_sphara.setIdentity();
72 m_overlap = MatrixXd::Zero(channels, m_maxFilterLength);
73
74 if (!m_info) {
75 m_filteredRows.clear();
76 return;
77 }
78
79 update_compensator(0);
80 set_filter_channel_type(m_filterChannelType);
81
82 m_indicesVVGrad = rows_where(m_info->chs, [](const FiffChInfo& ch) { return ch.chpos.coil_type == FIFFV_COIL_VV_PLANAR_T1; });
83 m_indicesVVMag = rows_where(m_info->chs, [](const FiffChInfo& ch) { return ch.chpos.coil_type == FIFFV_COIL_VV_MAG_T3; });
84 m_indicesBabyMEGInner = rows_where(m_info->chs, [](const FiffChInfo& ch) { return ch.chpos.coil_type == FIFFV_COIL_BABY_MAG; });
85 m_indicesEEG = rows_where(m_info->chs, [](const FiffChInfo& ch) { return ch.kind == FIFFV_EEG_CH; });
86}

122void RtRawDisplay::update_sphara_options(const std::string& system, int first, int second)
123{
124 if (!m_info) {
125 return;
126 }
127
128 const auto channels = static_cast<Index>(m_info->chs.size());
129 MatrixXd multFirst = MatrixXd::Identity(channels, channels);
130 MatrixXd multSecond = MatrixXd::Identity(channels, channels);
131
132 if (system == "VectorView" && m_spharaVVGrad.size() != 0 && m_spharaVVMag.size() != 0) {
133 multFirst = makeSpharaProjector(m_spharaVVGrad, m_indicesVVGrad, m_info->nchan, first, 1);
134 multSecond = makeSpharaProjector(m_spharaVVMag, m_indicesVVMag, m_info->nchan, second, 0);
135 }
136
137 if (system == "BabyMEG" && m_spharaBabyMEGInner.size() != 0) {
138 multFirst = makeSpharaProjector(m_spharaBabyMEGInner, m_indicesBabyMEGInner, m_info->nchan, first, 0);
139 }
140
141 if (system == "EEG" && m_spharaEEG.size() != 0) {
142 multFirst = makeSpharaProjector(m_spharaEEG, m_indicesEEG, m_info->nchan, first, 0);
143 }
144
145 const SparseMatrix<double> sparseFirst = multFirst.sparseView();
146 const SparseMatrix<double> sparseSecond = multSecond.sparseView();
147 m_sphara = sparseFirst * sparseSecond;
148}

159void RtRawDisplay::set_filters(std::vector<FilterKernel> kernels)
160{
161 m_kernels = std::move(kernels);
162
163 m_maxFilterLength = 1;
164 for (const FilterKernel& kernel : m_kernels) {
165 m_maxFilterLength = std::max(m_maxFilterLength, kernel.filter_order());

174void RtRawDisplay::set_filter_channel_type(const std::string& type)
175{
176 m_filterChannelType = type;
177 m_filteredRows.assign(m_info ? m_info->chs.size() : 0, false);
178 if (!m_info) {
179 return;
180 }
181
182 for (std::size_t r = 0; r < m_info->chs.size(); ++r) {
183 const FiffChInfo& ch = m_info->chs[r];
184 const bool filterable = ch.kind == FIFFV_MEG_CH || ch.kind == FIFFV_EEG_CH || ch.kind == FIFFV_EOG_CH ||
185 ch.kind == FIFFV_ECG_CH || ch.kind == FIFFV_EMG_CH;
186 m_filteredRows[r] = filterable && (type == "All" || ch.ch_name.find(type) != std::string::npos);
187 }
188}

220std::optional<RtRawDisplay::Filtered> RtRawDisplay::filter(const MatrixXd& corrected)
221{
222 if (corrected.cols() < m_maxFilterLength) {
223 return std::nullopt;
224 }
225
226 Filtered result;
227 result.data = corrected;
228 result.zero_columns = m_drawFilterFront ? 0 : m_maxFilterLength;
229 for (int r = 0; r < corrected.rows(); ++r) {
230 const bool filtered = r < std::ssize(m_filteredRows) && m_filteredRows[static_cast<std::size_t>(r)];
231 (filtered ? result.filtered_rows : result.passthrough_rows).push_back(r);
232 }
233
234 // Every row filters with its own copy of the kernels, since the FFT filter is not const.
235 std::vector<RowVectorXd> rows(result.filtered_rows.size());
236 ThreadPool::global().parallel_for(0, rows.size(), [&](std::size_t i) {
237 rows[i] = corrected.row(result.filtered_rows[i]);
238 std::vector<FilterKernel> kernels = m_kernels;
239 for (FilterKernel& kernel : kernels) {
240 // FFT convolution with the overhead kept, so the overlap carries into the next block.
241 kernel.apply_fft_filter(rows[i], true);
242 }
243 });
244
245 const Index cols = corrected.cols();
246 if (m_overlap.rows() < corrected.rows()) {
247 m_overlap.conservativeResize(corrected.rows(), m_maxFilterLength);
248 }
249
250 for (std::size_t i = 0; i < rows.size(); ++i) {
251 const int row = result.filtered_rows[i];
252 RowVectorXd& data = rows[i];
253
254 if (m_drawFilterFront) {
255 data.head(m_maxFilterLength) += m_overlap.row(row);
256 } else {
257 data.head(m_maxFilterLength).setZero();
258 }
259
260 result.data.row(row) = data.head(cols);
261 m_overlap.row(row) = data.tail(m_maxFilterLength);
262 }
263
264 m_drawFilterFront = true;
265 return result;
266}
