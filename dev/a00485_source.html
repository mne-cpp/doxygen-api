66std::vector<int> fiff_get_matrix_dims(const FiffTag& tag)
67/*
68 * Interpret dimensions from matrix data (dense and sparse)
69 */
70{
71 int ndim;
72 int* dims;
73 const std::size_t tsize = tag.size();
74 /*
75 * Initial checks
76 */
77 if (tag.data() == nullptr) {
78 mne::Log::critical("fiff_get_matrix_dims: no data available!");
79 return {};
80 }
81 if (fiff_type_fundamental(tag.type) != FIFFTS_FS_MATRIX) {
82 mne::Log::critical("fiff_get_matrix_dims: tag does not contain a matrix!");
83 return {};
84 }
85 if (tsize < sizeof(fiff_int_t)) {
86 mne::Log::critical("fiff_get_matrix_dims: too small matrix data!");
87 return {};
88 }
89 /*
90 * Get the number of dimensions and check
91 */
92 ndim = *((fiff_int_t*)((fiff_byte_t*)(tag.data()) + tag.size() - sizeof(fiff_int_t)));
93 if (ndim <= 0 || ndim > FIFFC_MATRIX_MAX_DIM) {
94 mne::Log::critical("fiff_get_matrix_dims: unreasonable # of dimensions!");
95 return {};
96 }
97 if (fiff_type_matrix_coding(tag.type) == FIFFTS_MC_DENSE) {
98 if (tsize < (ndim + 1) * sizeof(fiff_int_t)) {
99 mne::Log::critical("fiff_get_matrix_dims: too small matrix data!");
100 return {};
101 }
102 std::vector<int> res(ndim + 1);
103 res[0] = ndim;
104 dims = ((fiff_int_t*)((fiff_byte_t*)(tag.data()) + tag.size())) - ndim - 1;
105 for (int k = 0; k < ndim; k++)
106 res[k + 1] = dims[k];
107 return res;
108 } else if (fiff_type_matrix_coding(tag.type) == FIFFTS_MC_CCS ||
109 fiff_type_matrix_coding(tag.type) == FIFFTS_MC_RCS) {
110 if (tsize < (ndim + 2) * sizeof(fiff_int_t)) {
111 mne::Log::critical("fiff_get_matrix_sparse_dims: too small matrix data!");
112 return {};
113 }
114 std::vector<int> res(ndim + 2);
115 res[0] = ndim;
116 dims = ((fiff_int_t*)((fiff_byte_t*)(tag.data()) + tag.size())) - ndim - 1;
117 for (int k = 0; k < ndim; k++)
118 res[k + 1] = dims[k];
119 res[ndim + 1] = dims[-1];
120 return res;
121 } else {
122 mne::Log::critical("fiff_get_matrix_dims: unknown matrix coding.");
123 return {};
124 }
125}

154std::unique_ptr<FiffSparseMatrix> FiffSparseMatrix::fiff_get_float_sparse_matrix(const FiffTag& tag)
155{
156 int m, n, nz;
157 int cod;
158 std::size_t correct_size;
159
160 if (fiff_type_fundamental(tag.type) != FIFFT_MATRIX ||
161 fiff_type_base(tag.type) != FIFFT_FLOAT ||
162 (fiff_type_matrix_coding(tag.type) != FIFFTS_MC_CCS &&
163 fiff_type_matrix_coding(tag.type) != FIFFTS_MC_RCS)) {
164 mne::Log::warning("[FiffSparseMatrix::fiff_get_float_sparse_matrix] wrong data type!");
165 return nullptr;
166 }
167
168 auto dims = fiff_get_matrix_sparse_dims(tag);
169 if (dims.empty())
170 return nullptr;
171
172 if (dims[0] != 2) {
173 mne::Log::warning("[FiffSparseMatrix::fiff_get_float_sparse_matrix] wrong # of dimensions!");
174 return nullptr;
175 }
176
177 m = dims[1];
178 n = dims[2];
179 nz = dims[3];
180
181 cod = fiff_type_matrix_coding(tag.type);
182 if (cod == FIFFTS_MC_CCS)
183 correct_size = nz * (sizeof(fiff_float_t) + sizeof(fiff_int_t)) +
184 (n + 1 + dims[0] + 2) * (sizeof(fiff_int_t));
185 else if (cod == FIFFTS_MC_RCS)
186 correct_size = nz * (sizeof(fiff_float_t) + sizeof(fiff_int_t)) +
187 (m + 1 + dims[0] + 2) * (sizeof(fiff_int_t));
188 else {
189 mne::Log::warning("[FiffSparseMatrix::fiff_get_float_sparse_matrix] Incomprehensible sparse matrix coding");
190 return nullptr;
191 }
192 if (tag.size() != correct_size) {
193 mne::Log::warning("[FiffSparseMatrix::fiff_get_float_sparse_matrix] wrong data size!");
194 return nullptr;
195 }
196 /*
197 * Parse tag data into triplets and build Eigen sparse matrix directly
198 */
199 const float* src_data = reinterpret_cast<const float*>(tag.data());
200 const int* src_inds = reinterpret_cast<const int*>(src_data + nz);
201 const int* src_ptrs = src_inds + nz;
202
203 // A corrupt file must not index outside the matrix: the pointers run from 0 to nz without going
204 // back, and every index lies within the other dimension
205 const int nOuter = (cod == FIFFTS_MC_RCS) ? m : n;
206 const int nInner = (cod == FIFFTS_MC_RCS) ? n : m;
207 bool valid = m >= 0 && n >= 0 && nz >= 0 && src_ptrs[0] == 0 && src_ptrs[nOuter] == nz;
208 for (int k = 0; valid && k < nOuter; ++k)
209 valid = src_ptrs[k] <= src_ptrs[k + 1];
210 for (int j = 0; valid && j < nz; ++j)
211 valid = src_inds[j] >= 0 && src_inds[j] < nInner;
212 if (!valid) {
213 mne::Log::warning("[FiffSparseMatrix::fiff_get_float_sparse_matrix] index or pointer outside the matrix!");
214 return nullptr;
215 }
216
217 using T = Eigen::Triplet<float>;
218 std::vector<T> triplets;
219 triplets.reserve(nz);
220
221 if (cod == FIFFTS_MC_RCS) {
222 for (int row = 0; row < m; ++row) {
223 for (int j = src_ptrs[row]; j < src_ptrs[row + 1]; ++j) {
224 triplets.push_back(T(row, src_inds[j], src_data[j]));
225 }
226 }
227 } else { // CCS
228 for (int col = 0; col < n; ++col) {
229 for (int j = src_ptrs[col]; j < src_ptrs[col + 1]; ++j) {
230 triplets.push_back(T(src_inds[j], col, src_data[j]));
231 }
232 }
233 }
234
235 Eigen::SparseMatrix<float> eigenMat(m, n);
236 eigenMat.setFromTriplets(triplets.begin(), triplets.end());
237 eigenMat.makeCompressed();
238
239 auto res = std::make_unique<FiffSparseMatrix>(std::move(eigenMat), cod);
240 return res;
241}

245std::unique_ptr<FiffSparseMatrix> FiffSparseMatrix::create_sparse_rcs(int nrow, int ncol, int* nnz, int** colindex, float** vals)
246{
247 int j, k, totalNz;
248
249 for (j = 0, totalNz = 0; j < nrow; j++)
250 totalNz = totalNz + nnz[j];
251
252 if (totalNz <= 0) {
253 mne::Log::warning("[FiffSparseMatrix::create_sparse_rcs] No nonzero elements specified.");
254 return nullptr;
255 }
256
257 using T = Eigen::Triplet<float>;
258 std::vector<T> triplets;
259 triplets.reserve(totalNz);
260
261 for (j = 0; j < nrow; j++) {
262 for (k = 0; k < nnz[j]; k++) {
263 int col = colindex[j][k];
264 if (col < 0 || col >= ncol) {
265 mne::Log::warning("[FiffSparseMatrix::create_sparse_rcs] Column index out of range");
266 return nullptr;
267 }
268 float val = vals ? vals[j][k] : 0.0f;
269 triplets.push_back(T(j, col, val));
270 }
271 }
272
273 Eigen::SparseMatrix<float> eigenMat(nrow, ncol);
274 eigenMat.setFromTriplets(triplets.begin(), triplets.end());
275 eigenMat.makeCompressed();
276
277 return std::make_unique<FiffSparseMatrix>(std::move(eigenMat), FIFFTS_MC_RCS);
278}

282std::unique_ptr<FiffSparseMatrix> FiffSparseMatrix::mne_add_upper_triangle_rcs()
283/*
284 * Fill in upper triangle with the lower triangle values
285 */
286{
287 int nRows = rows();
288 int nCols = cols();
289
290 if (nRows != nCols) {
291 mne::Log::warning("[FiffSparseMatrix::mne_add_upper_triangle_rcs] input must be square");
292 return nullptr;
293 }
294
295 // Build full (lower + upper) by adding transpose
296 Eigen::SparseMatrix<float> full = m_eigen + Eigen::SparseMatrix<float>(m_eigen.transpose());
297
298 // The diagonal was counted twice — fix by subtracting the diagonal once
299 for (int k = 0; k < full.outerSize(); ++k) {
300 for (Eigen::SparseMatrix<float>::InnerIterator it(full, k); it; ++it) {
301 if (it.row() == it.col()) {
302 // Original diagonal value is in m_eigen; full has 2x, so set back to 1x
303 it.valueRef() = 0.5f * it.value();
304 }
305 }
306 }
307
308 full.makeCompressed();
309 return std::make_unique<FiffSparseMatrix>(std::move(full), FIFFTS_MC_RCS);
310}

342std::unique_ptr<FiffSparseMatrix> FiffSparseMatrix::pickLowerTriangleRcs() const
343{
344 int nRows = rows();
345 int nCols = cols();
346
347 if (nRows != nCols) {
348 mne::Log::warning("[FiffSparseMatrix::pickLowerTriangleRcs] input must be square");
349 return nullptr;
350 }
351
352 using T = Eigen::Triplet<float>;
353 std::vector<T> triplets;
354 triplets.reserve(m_eigen.nonZeros());
355
356 for (int k = 0; k < m_eigen.outerSize(); ++k) {
357 for (Eigen::SparseMatrix<float>::InnerIterator it(m_eigen, k); it; ++it) {
358 if (it.row() >= it.col()) { // lower triangle including diagonal
359 triplets.push_back(T(it.row(), it.col(), it.value()));
360 }
361 }
362 }
363
364 Eigen::SparseMatrix<float> lower(nRows, nCols);
365 lower.setFromTriplets(triplets.begin(), triplets.end());
366 lower.makeCompressed();
367
368 return std::make_unique<FiffSparseMatrix>(std::move(lower), FIFFTS_MC_RCS);
369}
