45RtCov::RtCov(const FiffInfo& info, double halfLife)
46: m_info(info)
47{
48 for (int i = 0; i < std::ssize(m_info.chs); ++i) {
49 if (m_info.chs[i].kind == FIFFV_MEG_CH || m_info.chs[i].kind == FIFFV_EEG_CH) {
50 m_picks.push_back(i);
51 }
52 }
53 if (m_picks.empty()) {
54 m_picks.resize(m_info.chs.size());
55 std::iota(m_picks.begin(), m_picks.end(), 0);
56 }
57 if (std::isfinite(halfLife)) {
58 if (halfLife > 0.0 && m_info.sfreq > 0.0) {
59 m_decay = std::exp2(-1.0 / (halfLife * m_info.sfreq));
60 } else {
61 Log::warning("[RtCov] A half-life needs a positive half-life and sampling rate; every sample counts the same.");
62 }
63 }
64}

68void RtCov::update(const MatrixXd& data)
69{
70 const Index n = data.cols();
71 const auto nchan = std::ssize(m_info.chs);
72 const auto used = std::ssize(m_picks);
73 if (n == 0) {
74 return;
75 }
76 if (used == 0 || (data.rows() != nchan && data.rows() != used)) {
77 Log::warning("[RtCov::update] Ignoring a block of {} channels; expected {} or {}.", data.rows(), nchan, used);
78 return;
79 }
80 MatrixXd block = data.rows() == used ? data : MatrixXd(data(m_picks, placeholders::all));
81
82 // The newest sample has weight 1, each older one m_decay times the next
83 VectorXd weights = VectorXd::Ones(n);
84 for (Index t = n - 2; t >= 0 && m_decay != 1.0; --t) {
85 weights(t) = weights(t + 1) * m_decay;
86 }
87 const double blockWeight = weights.sum();
88 const VectorXd blockMean = block * weights / blockWeight;
89 block.colwise() -= blockMean;
90 if (m_decay != 1.0) {
91 block *= weights.cwiseSqrt().asDiagonal();
92 }
93
94 if (m_weight == 0.0) {
95 m_mean = blockMean;
96 m_scatter = MatrixXd::Zero(used, used);
97 m_scatter.selfadjointView<Lower>().rankUpdate(block);
98 m_weight = blockWeight;
99 m_weightSq = weights.squaredNorm();
100 return;
101 }
102 // Age the running statistics by the block, then merge the block's centred statistics into them
103 const double aging = std::pow(m_decay, static_cast<double>(n));
104 m_weight *= aging;
105 m_weightSq *= aging * aging;
106 m_scatter *= aging;
107 const VectorXd delta = blockMean - m_mean;
108 const double total = m_weight + blockWeight;
109 m_scatter.selfadjointView<Lower>().rankUpdate(block);
110 m_scatter.selfadjointView<Lower>().rankUpdate(delta, m_weight * blockWeight / total);
111 m_mean += delta * (blockWeight / total);
112 m_weight = total;
113 m_weightSq += weights.squaredNorm();
114}

118std::optional<FiffCov> RtCov::covariance() const
119{
120 // The reliability-weights correction of numpy.cov; zero for a single sample
121 const double dof = m_weight > 0.0 ? m_weight - m_weightSq / m_weight : 0.0;
122 if (dof <= 0.0) {
123 return std::nullopt;
124 }
125 FiffCov cov;
126 cov.data = MatrixXd(m_scatter.selfadjointView<Lower>()) / dof;
127 cov.kind = FIFFV_MNE_NOISE_COV;
128 cov.diag = false;
129 cov.dim = static_cast<int>(cov.data.rows());
130 cov.names.reserve(m_picks.size());
131 for (const int pick : m_picks) {
132 cov.names.push_back(m_info.ch_names[static_cast<std::size_t>(pick)]);
133 }
134 cov.projs = m_info.projs;
135 cov.bads = m_info.bads;
136 cov.nfree = static_cast<int>(std::lround(m_weight * m_weight / m_weightSq));
137 return cov.regularize(m_info, 0.05, 0.05, 0.1, true, {});
138}
