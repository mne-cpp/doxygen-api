92void ButterflyPlot::draw(Canvas& canvas, const Font& font) const
93{
94 const int w = static_cast<int>(canvas.width());
95 const int h = static_cast<int>(canvas.height());
96 canvas.save();
97 canvas.set_font(font);
98
99 // Background, its outline just outside the figure
100 canvas.set_pen(Color::black());
101 canvas.set_brush(Brush::solid(m_background));
102 canvas.draw_rect({-1, -1, static_cast<double>(w + 2), static_cast<double>(h + 2)});
103 canvas.set_brush(Brush{});
104 canvas.set_antialiasing(true);
105 if (m_samples == 0) {
106 canvas.restore();
107 return;
108 }
109 const float fDx = static_cast<float>(w) / static_cast<float>(m_samples);
110 const Pen dashedRed{{255, 0, 0}, 1, LineStyle::Dash};
111
112 // Baseline interval
113 {
114 canvas.save();
115 canvas.set_pen(dashedRed);
116 canvas.set_brush(Brush::solid({255, 0, 0}));
117 canvas.set_opacity(0.1);
118 const float fromSample = m_baselineFrom * m_sfreq + static_cast<float>(m_preStimSamples);
119 const float toSample = m_baselineTo * m_sfreq + static_cast<float>(m_preStimSamples);
120 canvas.draw_rect({static_cast<double>(static_cast<int>(fDx * fromSample)), 0,
121 static_cast<double>(static_cast<int>(fDx * (toSample - fromSample))), static_cast<double>(h)});
122 canvas.restore();
123 }
124
125 // Stimulus
126 {
127 canvas.save();
128 canvas.set_pen(dashedRed);
129 const float posX = fDx * static_cast<float>(m_preStimSamples);
130 const double x = static_cast<int>(posX);
131 canvas.draw_line({x, 1}, {x, static_cast<double>(h)});
132 canvas.draw_text(Point{posX + 5.0f, static_cast<double>(h - 1 - 5)}, "0ms / Stimulus");
133 canvas.restore();
134 }
135
136 // A dashed line every 50 ms
137 if (m_timeSpacers) {
138 canvas.save();
139 canvas.set_pen(Pen{{0, 0, 0, 128}, 1, LineStyle::Dash});
140 const double yEnd = h - 1;
141 const float timeDistanceMSec = 50.0f;
142 const float timeDistanceSamples = (timeDistanceMSec / 1000.0f) * m_sfreq;
143 const auto spacer = [&](float sampleCounter, const std::string& label) {
144 const float x = fDx * sampleCounter;
145 const double xi = static_cast<int>(x);
146 canvas.draw_line({xi, 0}, {xi, static_cast<double>(static_cast<int>(yEnd))});
147 canvas.draw_text(Point{x + 5.0f, yEnd - 5}, label);
148 };
149 float sampleCounter = static_cast<float>(m_preStimSamples);
150 for (int counter = 1; sampleCounter - timeDistanceSamples > 0; ++counter) {
151 sampleCounter -= timeDistanceSamples;
152 spacer(sampleCounter, std::format("-{}ms", timeDistanceMSec * static_cast<float>(counter)));
153 }
154 sampleCounter = static_cast<float>(m_preStimSamples);
155 for (int counter = 1; sampleCounter + timeDistanceSamples < static_cast<float>(m_samples); ++counter) {
156 sampleCounter += timeDistanceSamples;
157 spacer(sampleCounter, std::format("{}ms", timeDistanceMSec * static_cast<float>(counter)));
158 }
159 canvas.restore();
160 }
161
162 // Zero line
163 canvas.save();
164 canvas.set_pen(Pen{Color::black(), 1, LineStyle::Dash});
165 // The zero line sits on a whole pixel
166 const int zeroY = h / 2;
167 canvas.draw_line({0, static_cast<double>(zeroY)}, {static_cast<double>(w), static_cast<double>(zeroY)});
168 canvas.restore();
169
170 canvas.translate(0, zeroY);
171 for (const Channel& channel : m_channels) {
172 canvas.save();
173 drawChannel(canvas, channel, w, h);
174 canvas.restore();
175 }
176 canvas.restore();
177}

181void ButterflyPlot::drawChannel(Canvas& canvas, const Channel& channel, int width, int height) const
182{
183 canvas.set_opacity(channel.bad ? 0.20 : 0.75);
184 const auto fMaxValue = static_cast<float>(channel.max_value);
185 const float fScaleY = static_cast<float>(height) / (2 * fMaxValue);
186 const float fWinMaxVal = (static_cast<float>(height) - 2) / 2.0f;
187 const float fDx = static_cast<float>(width - 2) / (static_cast<float>(m_samples) - 1.0f);
188
189 for (const Trace& trace : channel.traces) {
190 Color color = trace.color;
191 if (m_frozen) {
192 color.a = 128;
193 }
194 canvas.set_pen(Pen{color, 1});
195
196 const int cols = static_cast<int>(trace.samples.size());
197 const int dsFactor = std::max(1, cols / width);
198 std::vector<Point> path{{1, 0}};
199 if (cols > 0) {
200 const float fValue = static_cast<float>(trace.samples[0]) * fScaleY;
201 path[0] = {1, -static_cast<double>(fValue)};
202 }
203 // At most `width` segments, as many as the widget has pixels
204 for (int i = 1; i < cols && static_cast<int>(path.size()) <= width; i += dsFactor) {
205 float fValue = static_cast<float>(trace.samples[static_cast<std::size_t>(i)]) * fScaleY;
206 // Cut at the edges of the figure
207 fValue = fValue > fWinMaxVal ? fWinMaxVal : fValue < -fWinMaxVal ? -fWinMaxVal
208 : fValue;
209 path.push_back({1.0 + static_cast<double>(fDx * static_cast<float>(i)), -static_cast<double>(fValue)});
210 }
211 canvas.draw_polyline(path);
212 }
213}
