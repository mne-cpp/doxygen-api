95struct RegionMT
96{
97 Eigen::MatrixXd matRoiMT;
98 Eigen::MatrixXd matRoiMTOrig;
99 std::int32_t nClusters;
100 Eigen::VectorXi idcs;
101 std::int32_t iLabelIdxIn;
102 std::string sDistMeasure;
103
108 RegionMTOut cluster() const
109 {
110 const std::string distMeasure = sDistMeasure.empty() ? std::string("cityblock") : sDistMeasure;
111
112 RegionMTOut out;
113 mne::KMeans kMeans(distMeasure, "sample", 5);
114 kMeans.calculate(matRoiMT, nClusters,
115 out.roiIdx, out.ctrs, out.sumd, out.D);
116 out.iLabelIdxOut = iLabelIdxIn;
117 return out;
118 }
119};
