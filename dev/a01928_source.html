113void SceneViewport::zoomBy(double factor, const QPointF& anchor)
114{
115 if (!(factor > 0.0)) {
116 return;
117 }
118 const viz::Point fixed = mapToScene(anchor);
119 m_zoom *= factor;
120 m_offset = anchor - QPointF(fixed.x, fixed.y) * m_zoom;
121 m_fitted = false;
122 update();
123}

150mne::viz::DisplayList SceneViewport::figure(const QSize& size) const
151{
152 viz::Canvas canvas(size.width(), size.height(), toSceneColor(m_background));
153 const viz::Rect bounds = sceneBounds();
154 if (!bounds.empty()) {
155 const double zoom = std::min((size.width() - 2 * kFitMargin) / bounds.width,
156 (size.height() - 2 * kFitMargin) / bounds.height);
157 canvas.translate(size.width() / 2.0 - bounds.center().x * zoom, size.height() / 2.0 - bounds.center().y * zoom);
158 canvas.scale(zoom, zoom);
159 drawScene(canvas);
160 }
161 return canvas.take();
162}

243void SceneViewport::mousePressEvent(QMouseEvent* event)
244{
245 if (event->button() == Qt::RightButton) {
246 m_drag = Drag::Pan;
247 m_lastPosition = event->position();
248 } else if (event->button() == Qt::LeftButton) {
249 if (scenePressed(mapToScene(event->position()), event->modifiers())) {
250 m_drag = Drag::Item;
251 } else if (bandSelectionEnabled()) {
252 m_drag = Drag::Band;
253 m_bandStart = m_bandEnd = event->position();
254 m_bandModifiers = event->modifiers();
255 }
256 }
257 event->accept();
258}

276void SceneViewport::mouseReleaseEvent(QMouseEvent* event)
277{
278 const Drag drag = std::exchange(m_drag, Drag::None);
279 if (drag == Drag::Item) {
280 sceneReleased(mapToScene(event->position()));
281 } else if (drag == Drag::Band) {
282 m_bandEnd = event->position();
283 const viz::Point a = mapToScene(m_bandStart);
284 const viz::Point b = mapToScene(m_bandEnd);
285 sceneBandSelected(viz::Rect{std::min(a.x, b.x), std::min(a.y, b.y), std::abs(b.x - a.x), std::abs(b.y - a.y)},
286 m_bandModifiers);
287 update();
288 }
289 event->accept();
290}

304bool SceneViewport::event(QEvent* event)
305{
306 if (event->type() == QEvent::Gesture) {
307 auto* gestures = static_cast<QGestureEvent*>(event);
308 if (auto* pinch = static_cast<QPinchGesture*>(gestures->gesture(Qt::PinchGesture))) {
309 zoomBy(pinch->scaleFactor(), mapFromGlobal(pinch->centerPoint()));
310 }
311 if (auto* pan = static_cast<QPanGesture*>(gestures->gesture(Qt::PanGesture))) {
312 panBy(-pan->delta());
313 }
314 return true;
315 }
316 return QWidget::event(event);
317}

321void SceneViewport::applyFit()
322{
323 const viz::Rect bounds = sceneBounds();
324 if (bounds.empty() || width() <= 2 * kFitMargin || height() <= 2 * kFitMargin) {
325 return;
326 }
327 m_zoom = std::min((width() - 2 * kFitMargin) / bounds.width, (height() - 2 * kFitMargin) / bounds.height);
328 m_offset = QPointF(width() / 2.0 - bounds.center().x * m_zoom, height() / 2.0 - bounds.center().y * m_zoom);
329}
