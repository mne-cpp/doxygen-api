42// Lane of each type in butterfly mode, -1 if the type has none
43int lane_of(const std::vector<TraceSource::Group>& groups, const std::string& type)
44{
45 for (int lane = 0; lane < std::ssize(groups); ++lane) {
46 if (groups[lane].type_label == type) {
47 return lane;
48 }
49 }
50 return -1;

53TraceUniforms uniforms_for(const std::optional<TraceSource::Channel>& channel, bool hidden, float yCenter, float yRange,
54 const TraceView& view, const TraceStyle& style)
55{
56 const bool bad = channel && channel->bad;
57 const Color color = hidden ? style.background : (bad ? TraceStyle::bad_color() : channel->color);
58 TraceUniforms u;
59 u.color[0] = static_cast<float>(color.r) / 255.f;
60 u.color[1] = static_cast<float>(color.g) / 255.f;
61 u.color[2] = static_cast<float>(color.b) / 255.f;
62 u.color[3] = static_cast<float>(color.a) / 255.f;
63 u.scroll_sample = static_cast<float>(view.scroll_sample());
64 u.samples_per_pixel = static_cast<float>(view.samples_per_pixel());
65 u.view_width = static_cast<float>(view.view_width());
66 u.view_height = static_cast<float>(view.view_height());
67 u.channel_y_center = yCenter;
68 u.channel_y_range = hidden ? 0.f : yRange;
69 u.amplitude_max = style.z_score ? 4.f : (channel ? channel->amplitude_max : 0.f);
70 u.show_clipping = (style.show_clipping && !bad && !style.z_score) ? 1.f : 0.f;
71 return u;
72}

102bool TraceGeometry::needs_stream(const TraceView& view) const
103{
104 const double visible = view.view_width() * view.samples_per_pixel();
105 // Refetch once half of the prefetch on a side is used up, unless the window already ends with the data there
106 const double margin = 0.5 * m_prefetchFactor * visible;
107 const double scroll = view.scroll_sample();
108 const bool nearFirst = !m_windowAtDataFirst && scroll < m_windowFirst + margin;
109 const bool nearLast = !m_windowAtDataLast && scroll + visible > m_windowLast - margin;
110 return m_dirty || nearFirst || nearLast;
111}

115bool TraceGeometry::stream(const TraceView& view, const TraceSource& source, bool z_score)
116{
117 const double spp = view.samples_per_pixel();
118 const double visible = view.view_width() * spp;
119 const double scroll = view.scroll_sample();
120 const int dataFirst = source.first_sample();
121 const int first = std::max(static_cast<int>(scroll - m_prefetchFactor * visible), dataFirst);
122 const int last = std::min(static_cast<int>(scroll + (1.0 + m_prefetchFactor) * visible), dataFirst + source.sample_count());
123 m_dirty = false;
124 if (first >= last) {
125 return false;
126 }
127 m_windowFirst = first;
128 m_windowLast = last;
129 m_windowAtDataFirst = first == dataFirst;
130 m_windowAtDataLast = last == dataFirst + source.sample_count();
131
132 // Decimated to the view's scale over the whole window
133 const int pixelWidth = static_cast<int>((last - first) / spp);
134 const int count = source.channel_count();
135 m_channels.assign(static_cast<std::size_t>(std::max(count, 0)), TraceVertices{});
136 for (int index = 0; index < count; ++index) {
137 TraceVertices vertices = source.vertices(index, first, last, pixelWidth);
138 if (z_score) {
139 ::mne::viz::z_score(vertices.xy);
140 }
141 m_channels[static_cast<std::size_t>(index)] = std::move(vertices);
142 }
143 return true;
144}

157std::vector<TraceDraw> TraceGeometry::draws(const TraceView& view, const TraceSource& source, const TraceStyle& style) const
158{
159 std::vector<TraceDraw> result;
160 const int total = source.channel_count();
161 const auto firstSampleOf = [this](int index) {
162 return index < std::ssize(m_channels) ? static_cast<float>(m_channels[static_cast<std::size_t>(index)].first_sample) : 0.f;
163 };
164
165 if (view.butterfly_mode()) {
166 // Every channel, in the lane of its type
167 const std::vector<TraceSource::Group> groups = source.groups();
168 const int lanes = static_cast<int>(groups.size());
169 if (lanes <= 0) {
170 return result;
171 }
172 const float laneRange = 2.f / static_cast<float>(lanes);
173 const int count = std::min(total, kMaxTraces);
174 for (int index = 0; index < count; ++index) {
175 const std::optional<TraceSource::Channel> channel = source.channel(index);
176 const int lane = channel ? lane_of(groups, channel->type_label) : -1;
177 const bool hidden = !channel || (style.hide_bad_channels && channel->bad) || lane < 0;
178 const float yCenter = lane >= 0 ? 1.f - laneRange * (static_cast<float>(lane) + 0.5f) : 0.f;
179 TraceDraw draw{index, uniforms_for(channel, hidden, yCenter, laneRange, view, style)};
180 draw.uniforms.first_sample = firstSampleOf(index);
181 result.push_back(draw);
182 }
183 return result;
184 }
185
186 // The visible rows, top (NDC +1) to bottom (NDC -1)
187 const int first = std::clamp(view.first_visible_channel(), 0, std::max(total, 0));
188 const int rows = std::min({view.visible_channel_count(), total - first, kMaxTraces});
189 if (rows <= 0) {
190 return result;
191 }
192 const float laneRange = 2.f / static_cast<float>(rows);
193 for (int row = 0; row < rows; ++row) {
194 const int index = first + row;
195 const std::optional<TraceSource::Channel> channel = source.channel(index);
196 const bool hidden = !channel || (style.hide_bad_channels && channel->bad);
197 TraceDraw draw{index, uniforms_for(channel, hidden, 1.f - laneRange * (static_cast<float>(row) + 0.5f), laneRange, view, style)};
198 draw.uniforms.first_sample = firstSampleOf(index);
199 result.push_back(draw);
200 }
201 return result;
202}

208std::vector<float> mne::viz::decimate_trace(std::span<const float> samples, int pixel_width, Detrend detrend)
209{
210 const int nSamples = static_cast<int>(samples.size());
211 if (pixel_width <= 0 || nSamples <= 0) {
212 return {};
213 }
214
215 // Trend over the window: its mean, or the least-squares line over t = 0 .. n - 1
216 float dcOffset = 0.f;
217 float slope = 0.f;
218 float intercept = 0.f;
219 if (detrend == Detrend::Mean) {
220 double sum = 0.0;
221 for (const float value : samples) {
222 sum += value;
223 }
224 dcOffset = static_cast<float>(sum / nSamples);
225 } else if (detrend == Detrend::Linear) {
226 double sumX = 0.0;
227 double sumY = 0.0;
228 double sumXX = 0.0;
229 double sumXY = 0.0;
230 for (int i = 0; i < nSamples; ++i) {
231 const double x = static_cast<double>(i);
232 const double y = static_cast<double>(samples[static_cast<std::size_t>(i)]);
233 sumX += x;
234 sumY += y;
235 sumXX += x * x;
236 sumXY += x * y;
237 }
238 const double denom = nSamples * sumXX - sumX * sumX;
239 if (std::abs(denom) > 1e-30) {
240 slope = static_cast<float>((nSamples * sumXY - sumX * sumY) / denom);
241 intercept = static_cast<float>((sumY - slope * sumX) / nSamples);
242 }
243 }
244 const auto trendAt = [&](float t) {
245 return detrend == Detrend::Mean ? dcOffset : detrend == Detrend::Linear ? slope * t + intercept
246 : 0.f;
247 };
248
249 std::vector<float> result;
250 if (nSamples <= pixel_width * 2) {
251 // One vertex per sample
252 result.reserve(static_cast<std::size_t>(nSamples) * 2);
253 for (int i = 0; i < nSamples; ++i) {
254 result.push_back(static_cast<float>(i));
255 result.push_back(samples[static_cast<std::size_t>(i)] - trendAt(static_cast<float>(i)));
256 }
257 return result;
258 }
259
260 // Min/max envelope: (x, max), (x, min) per pixel bin; as a line strip a spike per column, diagonals between
261 result.reserve(static_cast<std::size_t>(pixel_width) * 4);
262 const float spp = static_cast<float>(nSamples) / static_cast<float>(pixel_width);
263 for (int px = 0; px < pixel_width; ++px) {
264 int begin = static_cast<int>(static_cast<float>(px) * spp);
265 const int end = std::min(static_cast<int>(static_cast<float>(px + 1) * spp), nSamples);
266 if (begin >= end) {
267 begin = std::max(end - 1, 0);
268 }
269 float minV = samples[static_cast<std::size_t>(begin)];
270 float maxV = minV;
271 for (int s = begin + 1; s < end; ++s) {
272 minV = std::min(minV, samples[static_cast<std::size_t>(s)]);
273 maxV = std::max(maxV, samples[static_cast<std::size_t>(s)]);
274 }
275 const float trend = trendAt(static_cast<float>(begin) + static_cast<float>(end - begin) * 0.5f);
276 const float x = static_cast<float>(px) * spp;
277 result.push_back(x);
278 result.push_back(maxV - trend);
279 result.push_back(x);
280 result.push_back(minV - trend);
281 }
282 return result;
283}

287void mne::viz::z_score(std::vector<float>& xy)
288{
289 const int n = static_cast<int>(xy.size()) / 2;
290 if (n <= 1) {
291 return;
292 }
293 double sum = 0.0;
294 double sumSq = 0.0;
295 for (int v = 0; v < n; ++v) {
296 const double a = static_cast<double>(xy[static_cast<std::size_t>(v) * 2 + 1]);
297 sum += a;
298 sumSq += a * a;
299 }
300 const float mean = static_cast<float>(sum / n);
301 const double var = sumSq / n - static_cast<double>(mean) * mean;
302 const float sd = var > 0.0 ? static_cast<float>(std::sqrt(var)) : 1.f;
303 for (int v = 0; v < n; ++v) {
304 float& y = xy[static_cast<std::size_t>(v) * 2 + 1];
305 y = (y - mean) / sd;
306 }
307}

313{
314 std::vector<int> indices;
315 if (filter.empty()) {
316 indices.reserve(static_cast<std::size_t>(std::max(channel_count, 0)));
317 for (int channel = 0; channel < channel_count; ++channel) {
318 indices.push_back(channel);
319 }
320 } else {

322 }
323 if (!hide_bad) {

332void mne::viz::draw_trace_figure(Canvas& canvas, const TextMetrics& metrics, const Font& font, const TraceView& view,
333 const TraceSource& source, const TraceMarkers& markers, const TraceStyle& style)
334{
335 const int pw = static_cast<int>(canvas.width());
336 const int ph = static_cast<int>(canvas.height());
337 if (pw <= 0 || ph <= 0 || view.samples_per_pixel() <= 0.0) {
338 return;
339 }
340
341 // Same rows and time span as on screen at any output width
342 const float spp = static_cast<float>(view.samples_per_pixel()) * static_cast<float>(std::max(view.view_width(), 1)) / static_cast<float>(pw);
343 const float scrollSample = static_cast<float>(view.scroll_sample());
344 const float sfreq = static_cast<float>(view.sfreq());
345 const int firstFileSample = view.first_file_sample();
346
347 // (logical channel, lane) of every trace, laid out like TraceGeometry::draws()
348 std::vector<std::pair<int, int>> traces;
349 int laneCount = 0;
350 const int total = source.channel_count();
351 if (view.butterfly_mode()) {
352 const std::vector<TraceSource::Group> groups = source.groups();
353 laneCount = static_cast<int>(groups.size());
354 for (int lane = 0; lane < laneCount; ++lane) {
355 for (int index = 0; index < total; ++index) {
356 const std::optional<TraceSource::Channel> channel = source.channel(index);
357 if (channel && channel->type_label == groups[lane].type_label && !(style.hide_bad_channels && channel->bad)) {
358 traces.emplace_back(index, lane);
359 }
360 }
361 }
362 } else {
363 const int first = std::clamp(view.first_visible_channel(), 0, std::max(total, 0));
364 laneCount = std::min(view.visible_channel_count(), total - first);
365 for (int lane = 0; lane < laneCount; ++lane) {
366 traces.emplace_back(first + lane, lane);
367 }
368 }
369 if (laneCount <= 0) {
370 return;
371 }
372
373 canvas.save();
374 canvas.set_antialiasing(false);
375 canvas.set_brush(Brush{});
376
377 const float laneH = static_cast<float>(ph) / static_cast<float>(laneCount);
378 const int firstSample = static_cast<int>(scrollSample);
379 const int lastSample = firstSample + static_cast<int>(static_cast<float>(pw) * spp) + 1;
380
381 // Alternating per-second bands: odd seconds darkened by 8 %, as in overlay.frag
382 if (style.grid && sfreq > 0.f) {
383 const float firstBound = std::floor((scrollSample - static_cast<float>(firstFileSample)) / sfreq) * sfreq + static_cast<float>(firstFileSample);
384 bool oddBand = (static_cast<long long>((firstBound - static_cast<float>(firstFileSample)) / sfreq) & 1) != 0;
385 const Color& bg = style.background;
386 const Color altColor{static_cast<std::uint8_t>(std::lround(bg.r * 0.92)), static_cast<std::uint8_t>(std::lround(bg.g * 0.92)),
387 static_cast<std::uint8_t>(std::lround(bg.b * 0.92)), bg.a};
388 for (int band = 0;; ++band, oddBand = !oddBand) {
389 const float s = firstBound + static_cast<float>(band) * sfreq;
390 if (s >= static_cast<float>(lastSample)) {
391 break;
392 }
393 if (!oddBand) {
394 continue;
395 }
396 const float xStart = std::clamp((s - scrollSample) / spp, 0.f, static_cast<float>(pw));
397 const float xEnd = std::clamp((s - scrollSample) / spp + sfreq / spp, 0.f, static_cast<float>(pw));
398 if (xEnd > xStart) {
399 canvas.fill_rect({xStart, 0.0, static_cast<double>(xEnd - xStart), static_cast<double>(ph)}, Brush::solid(altColor));
400 }
401 }
402 }
403
404 // Grid: lane separators, ±45 % guides, centre lines and time ticks at least 80 px apart
405 if (style.grid) {
406 for (int i = 0; i < laneCount; ++i) {
407 const float yMid = (static_cast<float>(i) + 0.5f) * laneH;
408 const float yTop = static_cast<float>(i) * laneH;
409 if (i > 0) {
410 canvas.set_pen(Pen{{205, 205, 215}, 1.0});
411 canvas.draw_line({0.0, yTop}, {static_cast<double>(pw), yTop});
412 }
413 canvas.set_pen(Pen{{228, 228, 235}, 1.0, LineStyle::Dot});
414 canvas.draw_line({0.0, yMid - laneH * 0.44f}, {static_cast<double>(pw), yMid - laneH * 0.44f});
415 canvas.draw_line({0.0, yMid + laneH * 0.44f}, {static_cast<double>(pw), yMid + laneH * 0.44f});
416 canvas.set_pen(Pen{{210, 210, 218}, 1.0});
417 canvas.draw_line({0.0, yMid}, {static_cast<double>(pw), yMid});
418 }
419 if (sfreq > 0.f) {
420 static constexpr std::array<float, 10> kNiceIntervals = {0.05f, 0.1f, 0.2f, 0.5f, 1.f, 2.f, 5.f, 10.f, 30.f, 60.f};
421 const float pxPerSecond = sfreq / spp;
422 float tickIntervalS = kNiceIntervals[0];
423 for (const float interval : kNiceIntervals) {
424 tickIntervalS = interval;
425 if (interval * pxPerSecond >= 80.f) {
426 break;
427 }
428 }
429 const float tickSamples = tickIntervalS * sfreq;
430 const float origin = static_cast<float>(firstFileSample);
431 const float firstTick = std::ceil((scrollSample - origin) / tickSamples) * tickSamples + origin;
432 canvas.set_pen(Pen{{205, 205, 210}, 1.0});
433 for (int tick = 0;; ++tick) {
434 const float s = firstTick + static_cast<float>(tick) * tickSamples;
435 if (s >= static_cast<float>(lastSample)) {
436 break;
437 }
438 const double x = (s - scrollSample) / spp;
439 canvas.draw_line({x, 0.0}, {x, static_cast<double>(ph)});
440 }
441 }
442 }
443
444 markers.draw_annotations(canvas, metrics, font, view.annotations(), scrollSample, 1.0 / spp);
445
446 // Traces, with the mapping of channeldata.vert / .frag
447 for (const auto& [index, lane] : traces) {
448 const std::optional<TraceSource::Channel> channel = source.channel(index);
449 if (!channel) {
450 continue;
451 }
452 const TraceVertices vertices = source.vertices(index, firstSample, lastSample, pw);
453 const int n = vertices.vertex_count();
454 if (n < 2) {
455 continue;
456 }
457
458 // Z-score: (y - mean) / std, with ±4 std filling the row like the GPU path
459 float offset = 0.f;
460 float amplitudeMax = channel->amplitude_max;
461 if (style.z_score) {
462 double sum = 0.0;
463 double sumSq = 0.0;
464 for (int v = 0; v < n; ++v) {
465 const double amp = static_cast<double>(vertices.xy[static_cast<std::size_t>(v) * 2 + 1]);
466 sum += amp;
467 sumSq += amp * amp;
468 }
469 const double mean = sum / n;
470 const double var = sumSq / n - mean * mean;
471 offset = static_cast<float>(mean);
472 amplitudeMax = 4.f * (var > 0.0 ? static_cast<float>(std::sqrt(var)) : 1.f);
473 }
474
475 const Pen normalPen{channel->bad ? TraceStyle::bad_color() : channel->color, 1.2};
476 const Pen clipPen{{255, 0, 0}, 1.6};
477 const bool doClip = style.show_clipping && !channel->bad && !style.z_score;
478 const float yMid = (static_cast<float>(lane) + 0.5f) * laneH;
479
480 std::vector<Point> segment;
481 segment.reserve(static_cast<std::size_t>(n));
482 bool prevClipped = false;
483 for (int v = 0; v < n; ++v) {
484 const float y = vertices.xy[static_cast<std::size_t>(v) * 2 + 1];
485 const float norm = amplitudeMax > 0.f ? std::clamp((y - offset) / amplitudeMax, -2.f, 2.f) : 0.f;
486 const Point pt{(static_cast<float>(vertices.first_sample) + vertices.xy[static_cast<std::size_t>(v) * 2] - scrollSample) / spp,
487 yMid - norm * 0.45f * laneH};
488 const bool clipped = doClip && std::abs(norm) >= 0.95f;
489 if (v > 0 && clipped != prevClipped) {
490 segment.push_back(pt);
491 canvas.set_pen(prevClipped ? clipPen : normalPen);
492 canvas.draw_polyline(segment);
493 segment.clear();
494 }
495 segment.push_back(pt);
496 prevClipped = clipped;
497 }
498 if (segment.size() > 1) {
499 canvas.set_pen(prevClipped ? clipPen : normalPen);
500 canvas.draw_polyline(segment);
501 }
502 }
503
504 markers.draw_lines(canvas, scrollSample, 1.0 / spp);
505 canvas.restore();
506}
