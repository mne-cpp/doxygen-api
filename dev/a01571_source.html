49void EpochExtractor::applyBaseline(MatrixXd& matEpoch, int iBase0, int iBase1)
50{
51 if (iBase0 > iBase1 || iBase0 < 0 || iBase1 >= matEpoch.cols()) return;
52 const int nBaseSamp = iBase1 - iBase0 + 1;
53 // Per-channel baseline mean
54 VectorXd baseline = matEpoch.block(0, iBase0, matEpoch.rows(), nBaseSamp).rowwise().mean();
55 matEpoch.colwise() -= baseline;
56}

62QVector<MNEEpochData> EpochExtractor::extract(const MatrixXd& matData,
63 const QVector<int>& eventSamples,
64 double dSFreq,
65 const Params& params,
66 const QVector<int>& eventCodes)
67{
68 QVector<MNEEpochData> epochs;
69
70 if (matData.size() == 0 || eventSamples.isEmpty()) return epochs;
71 if (dSFreq <= 0.0) {
72 qWarning() << "EpochExtractor::extract: invalid sampling frequency.";
73 return epochs;
74 }
75
76 const int nSamp = static_cast<int>(matData.cols());
77 const int nCh = static_cast<int>(matData.rows());
78
79 // Convert time to sample offsets
80 const int iOffset0 = static_cast<int>(std::round(params.dTmin * dSFreq));
81 const int iOffset1 = static_cast<int>(std::round(params.dTmax * dSFreq));
82 const int epochLen = iOffset1 - iOffset0 + 1;
83
84 if (epochLen <= 0) {
85 qWarning() << "EpochExtractor::extract: tmax must be > tmin.";
86 return epochs;
87 }
88
89 // Baseline window sample indices within the epoch (0-based)
90 const int iBase0 = static_cast<int>(std::round((params.dBaseMin - params.dTmin) * dSFreq));
91 const int iBase1 = static_cast<int>(std::round((params.dBaseMax - params.dTmin) * dSFreq));
92
93 const bool bHaveCodes = (eventCodes.size() == eventSamples.size());
94
95 for (int ev = 0; ev < eventSamples.size(); ++ev) {
96 const int evSamp = eventSamples[ev];
97 const int s0 = evSamp + iOffset0;
98 const int s1 = evSamp + iOffset1;
99
100 // Skip if epoch extends outside the recording
101 if (s0 < 0 || s1 >= nSamp) continue;
102
103 MNEEpochData epoch;
104 epoch.epoch = matData.block(0, s0, nCh, epochLen);
105 epoch.tmin = static_cast<float>(params.dTmin);
106 epoch.tmax = static_cast<float>(params.dTmax);
107 epoch.event = bHaveCodes ? eventCodes[ev] : 1;
108 epoch.bReject = false;
109
110 // Baseline correction
111 if (params.bApplyBaseline) {
112 applyBaseline(epoch.epoch, iBase0, iBase1);
113 }
114
115 // Amplitude rejection: peak-to-peak per channel
116 if (params.dThreshold > 0.0) {
117 for (int ch = 0; ch < nCh; ++ch) {
118 const RowVectorXd row = epoch.epoch.row(ch);
119 double ptp = row.maxCoeff() - row.minCoeff();
120 if (ptp > params.dThreshold) {
121 epoch.bReject = true;
122 break;
123 }
124 }
125 }
126
127 epochs.append(epoch);
128 }
129
130 return epochs;
131}

135MatrixXd EpochExtractor::average(const QVector<MNEEpochData>& epochs)
136{
137 MatrixXd result;
138 int nGood = 0;
139
140 for (const MNEEpochData& ep : epochs) {
141 if (ep.bReject) continue;
142 if (result.size() == 0) {
143 result = ep.epoch;
144 } else {
145 if (ep.epoch.rows() != result.rows() || ep.epoch.cols() != result.cols()) {
146 qWarning() << "EpochExtractor::average: epoch dimension mismatch, skipping.";
147 continue;
148 }
149 result += ep.epoch;
150 }
151 ++nGood;
152 }
153
154 if (nGood > 1) result /= static_cast<double>(nGood);
155 return result;
156}
