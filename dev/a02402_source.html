60template<typename T>
61bool take_number(std::string_view& text, T& value)
62{
63 text = text.substr(std::min(text.find_first_not_of(whitespace), text.size()));
64 const char* first = text.data();
65 const char* end = first + text.size();
66 const auto [ptr, ec] = std::from_chars(first, end, value);
67 if (ec != std::errc{} || (ptr != end && whitespace.find(*ptr) == std::string_view::npos)) {
68 return false;
69 }
70 text.remove_prefix(static_cast<std::size_t>(ptr - first));
71 return true;
72}

76std::string utf8_from_latin1(std::string_view bytes)
77{
78 std::string text;
79 text.reserve(bytes.size());
80 for (const char c : bytes) {
81 const auto u = static_cast<unsigned char>(c);
82 if (u < 0x80) {
83 text += c;
84 } else {
85 text += static_cast<char>(0xC0 | (u >> 6));
86 text += static_cast<char>(0x80 | (u & 0x3F));
87 }
88 }
89 return text;
90}

96std::string latin1_from_utf8(std::string_view text)
97{
98 std::string bytes;
99 bytes.reserve(text.size());
100 for (std::size_t i = 0; i < text.size();) {
101 const auto u = static_cast<unsigned char>(text[i]);
102 const std::size_t length = u < 0x80 ? 1 : u >= 0xF0 ? 4
103 : u >= 0xE0 ? 3
104 : 2;
105 if (length == 1) {
106 bytes += text[i];
107 } else if (length == 2 && u <= 0xC3 && i + 1 < text.size()) {
108 bytes += static_cast<char>(((u & 0x1F) << 6) | (static_cast<unsigned char>(text[i + 1]) & 0x3F));
109 } else {
110 bytes += '?';
111 }
112 i += length;
113 }
114 return bytes;
115}

123std::optional<MNEEventList> MNEEventList::read_fif(const std::filesystem::path& path, int offset)
124{
125 FiffStream stream(path);
126 if (!stream.open()) {
127 return std::nullopt;
128 }
129 const std::vector<std::shared_ptr<FiffDirNode>> blocks = stream.dirtree()->dir_tree_find(FIFFB_MNE_EVENTS);
130 const auto tag = blocks.empty() ? std::nullopt : blocks.front()->find_tag(stream, FIFF_MNE_EVENT_LIST);
131 if (!tag) {
132 mne::Log::warning("MNEEventList::read_fif - No event data in {}", mne::utf8_from_path(path));
133 return std::nullopt;
134 }
135 // Written as signed or unsigned integers; both have the same bit pattern.
136 const auto* values = reinterpret_cast<const std::int32_t*>(tag->data());
137 const int nevent = static_cast<int>(tag->size() / (3 * sizeof(std::int32_t)));
138 MNEEventList list;
139 list.events.resize(nevent);
140 for (int k = 0; k < nevent; ++k) {
141 list.events[k].sample = values[3 * k] - offset;
142 list.events[k].from = static_cast<unsigned int>(values[3 * k + 1]);
143 list.events[k].to = static_cast<unsigned int>(values[3 * k + 2]);
144 }
145 // The comments are one NUL-separated string, one entry per event.
146 if (const auto commentTag = blocks.front()->find_tag(stream, FIFF_MNE_EVENT_COMMENTS)) {
147 const std::string_view comments(reinterpret_cast<const char*>(commentTag->data()), commentTag->size());
148 int k = 0;
149 for (auto&& comment : comments | std::views::split('\0')) {
150 if (k >= nevent) {
151 break;
152 }
153 list.events[k++].comment = utf8_from_latin1(std::string_view(comment.begin(), comment.end()));
154 }
155 }
156 return list;
157}

161std::optional<MNEEventList> MNEEventList::read_text(const std::filesystem::path& path, int offset, int old_offset, float sfreq)
162{
163 std::ifstream in(path);
164 if (!in) {
165 mne::Log::warning("MNEEventList::read_text - Cannot open {}", mne::utf8_from_path(path));
166 return std::nullopt;
167 }
168 MNEEventList list;
169 bool first = true;
170 std::string line;
171 while (std::getline(in, line)) {
172 std::string_view rest = trimmed(line);
173 if (rest.empty() || rest.front() == '#') {
174 continue;
175 }
176 int sample = 0;
177 float time = 0.0f;
178 unsigned int from = 0;
179 unsigned int to = 0;
180 if (!take_number(rest, sample) || !take_number(rest, time) || !take_number(rest, from) || !take_number(rest, to)) {
181 break;
182 }
183 if (sample < 0) {
184 sample = static_cast<int>(time * sfreq);
185 }
186 if (first) {
187 first = false;
188 if (from == 0 && to == 0) {
189 if (offset != sample) {
190 mne::Log::warning("MNEEventList::read_text - First sample of the data ({}) and of {} ({}) differ", offset, mne::utf8_from_path(path), sample);
191 }
192 continue;
193 }
194 offset -= old_offset;
195 }
196 list.events.push_back(MNEEvent{sample - offset, from, to, false, false, std::string(trimmed(rest))});
197 }
198 if (list.events.empty()) {
199 mne::Log::warning("MNEEventList::read_text - No events read from {}", mne::utf8_from_path(path));
200 return std::nullopt;
201 }
202 return list;
203}

207bool MNEEventList::write_fif(const std::filesystem::path& path, int offset) const
208{
209 auto stream = FiffStream::start_file(path);
210 if (!stream) {
211 return false;
212 }
213 std::vector<std::int32_t> values;
214 values.reserve(3 * events.size());
215 std::string comments;
216 bool anyComment = false;
217 for (const MNEEvent& event : events) {
218 values.push_back(event.sample + offset);
219 values.push_back(static_cast<std::int32_t>(event.from));
220 values.push_back(static_cast<std::int32_t>(event.to));
221 comments += latin1_from_utf8(event.comment);
222 comments += '\0';
223 anyComment = anyComment || !event.comment.empty();
224 }
225 stream->start_block(FIFFB_MNE_EVENTS);
226 // Unsigned like MNE-C; trigger values are bit masks.
227 FiffTag list;
228 list.kind = FIFF_MNE_EVENT_LIST;
229 list.type = FIFFT_UINT;
230 list.next = FIFFV_NEXT_SEQ;
231 list.assign(std::as_bytes(std::span(values)));
232 stream->write_tag(list);
233 if (anyComment) {
234 // Latin-1 bytes, one NUL-terminated comment per event, as MNE-C writes and mne-python reads them.
235 FiffTag tag;
236 tag.kind = FIFF_MNE_EVENT_COMMENTS;
237 tag.type = FIFFT_BYTE;
238 tag.next = FIFFV_NEXT_SEQ;
239 tag.assign(std::as_bytes(std::span(comments)));
240 stream->write_tag(tag);
241 }
242 stream->end_block(FIFFB_MNE_EVENTS);
243 stream->end_file();
244 return true;
245}
