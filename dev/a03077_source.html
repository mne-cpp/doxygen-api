75std::shared_ptr<FiffDirNode> MNERawInfo::find_meas(const std::shared_ptr<FiffDirNode>& node)
76{
77 std::shared_ptr<FiffDirNode> block = node;
78 while (block && block->type != FIFFB_MEAS) {
79 block = block->parent.lock();
80 }
81 return block;

86std::shared_ptr<FiffDirNode> MNERawInfo::find_meas_info(const std::shared_ptr<FiffDirNode>& node)
87{
88 const std::shared_ptr<FiffDirNode> meas = find_meas(node);
89 if (!meas) {
90 return {};
91 }
92 const auto info = std::ranges::find(meas->children, FIFFB_MEAS_INFO, &FiffDirNode::type);
93 return info != meas->children.end() ? *info : std::shared_ptr<FiffDirNode>{};
94}

99{
100 std::shared_ptr<FiffDirNode> raw;
101 std::vector<std::shared_ptr<FiffDirNode>> temp;
102 temp = node->dir_tree_find(FIFFB_RAW_DATA);
103 if (temp.size() == 0) {
104 temp = node->dir_tree_find(FIFFB_CONTINUOUS_DATA);
105 if (temp.size() > 0)
106 raw = temp[0];
107 } else
108 raw = temp[0];
109 return raw;

143int MNERawInfo::get_meas_info(FiffStream& stream,
144 std::shared_ptr<FiffDirNode>& node,
145 std::unique_ptr<FiffId>& id,
146 int* nchan,
147 float* sfreq,
148 float* highpass,
149 float* lowpass,
150 std::vector<FiffChInfo>& chp,
151 FiffCoordTrans& trans,
152 FiffTime& start_time)
153{
154 std::optional<FiffTag> t_pTag;
155 std::vector<FiffChInfo> ch;
156 FiffChInfo this_ch;
157 FiffCoordTrans t;
158 int j, k;
159 int to_find = 4;
160 std::vector<std::shared_ptr<FiffDirNode>> hpi;
161 std::shared_ptr<FiffDirNode> meas;
162 std::shared_ptr<FiffDirNode> meas_info;
163 fiff_int_t kind, pos;
164 bool found_meas_date = false;
165
166 trans = FiffCoordTrans();
167 id.reset();
168 start_time = FiffTime();
169 /*
170 * Find desired parents
171 */
172 if (meas = find_meas(node); !meas) {
173 mne::Log::critical("Meas. block not found!");
174 return (-1);
175 }
176
177 if (meas_info = find_meas_info(node); !meas_info) {
178 mne::Log::critical("Meas. info not found!");
179 return (-1);
180 }
181 /*
182 * Is there a block id is in the FIFFB_MEAS node?
183 */
184 if (!meas->id.empty()) {
185 id = std::make_unique<FiffId>();
186 id->version = meas->id.version;
187 id->machid[0] = meas->id.machid[0];
188 id->machid[1] = meas->id.machid[1];
189 id->time = meas->id.time;
190 }
191 /*
192 * Others from FIFFB_MEAS_INFO
193 */
194 *lowpass = -1;
195 *highpass = -1;
196 for (k = 0; k < meas_info->nent(); k++) {
197 kind = meas_info->dir[k].kind;
198 pos = meas_info->dir[k].pos;
199 switch (kind) {
200 case FIFF_NCHAN:
201 if (t_pTag = stream.read_tag(pos); !t_pTag)
202 return (-1);
203 *nchan = *t_pTag->to_int();
204
205 for (j = 0; j < *nchan; j++) {
206 ch.push_back(FiffChInfo());
207 ch[j].scanNo = -1;
208 }
209 to_find = to_find + *nchan - 1;
210 break;
211
212 case FIFF_SFREQ:
213 if (t_pTag = stream.read_tag(pos); !t_pTag)
214 return (-1);
215 *sfreq = *t_pTag->to_float();
216 to_find--;
217 break;
218
219 case FIFF_LOWPASS:
220 if (t_pTag = stream.read_tag(pos); !t_pTag)
221 return (-1);
222 *lowpass = *t_pTag->to_float();
223 to_find--;
224 break;
225
226 case FIFF_HIGHPASS:
227 if (t_pTag = stream.read_tag(pos); !t_pTag)
228 return (-1);
229 *highpass = *t_pTag->to_float();
230 to_find--;
231 break;
232
233 case FIFF_CH_INFO:
234 if (t_pTag = stream.read_tag(pos); !t_pTag)
235 return (-1);
236
237 this_ch = t_pTag->to_ch_info();
238 if (this_ch.scanNo <= 0 || this_ch.scanNo > *nchan) {
239 mne::Log::critical("FIFF_CH_INFO : scan # out of range!");
240 return (-1);
241 } else
242 ch[this_ch.scanNo - 1] = this_ch;
243 to_find--;
244 break;
245
246 case FIFF_MEAS_DATE:
247 if (t_pTag = stream.read_tag(pos); !t_pTag)
248 return (-1);
249 {
250 const FiffTime* pTime = reinterpret_cast<const FiffTime*>(t_pTag->data());
251 start_time = FiffTime(pTime->secs, pTime->usecs);
252 found_meas_date = true;
253 }
254 break;
255
256 case FIFF_COORD_TRANS:
257 if (t_pTag = stream.read_tag(pos); !t_pTag)
258 return (-1);
259 t = FiffCoordTrans::from_tag(*t_pTag);
260 /*
261 * Require this particular transform!
262 */
263 if (t.from == FIFFV_COORD_DEVICE && t.to == FIFFV_COORD_HEAD) {
264 trans = t;
265 break;
266 }
267 }
268 }
269 /*
270 * Search for the coordinate transformation from
271 * HPI_RESULT block if it was not previously found
272 */
273 hpi = meas_info->dir_tree_find(FIFFB_HPI_RESULT);
274
275 if (hpi.size() > 0 && trans.empty())
276 for (k = 0; k < hpi[0]->nent(); k++)
277 if (hpi[0]->dir[k].kind == FIFF_COORD_TRANS) {
278 if (t_pTag = stream.read_tag(hpi[0]->dir[k].pos); !t_pTag)
279 return (-1);
280 t = FiffCoordTrans::from_tag(*t_pTag);
281 if (t.from == FIFFV_COORD_DEVICE && t.to == FIFFV_COORD_HEAD) {
282 trans = t;
283 break;
284 }
285 }
286 if (to_find < 3) {
287 if (*lowpass < 0) {
288 *lowpass = *sfreq / 2.0;
289 to_find--;
290 }
291 if (*highpass < 0) {
292 *highpass = 0.0;
293 to_find--;
294 }
295 }
296 if (to_find != 0) {
297 mne::Log::critical("Not all essential tags were found!");
298 return (-1);
299 }
300 chp = ch;
301 /*
302 * Resolve the measurement start time using the best available source
303 */
304 if (!found_meas_date) {
305 if (id)
306 start_time = id->time;
307 else
308 start_time = FiffTime();
309 }
310 return (0);
311}

315int MNERawInfo::load(const std::filesystem::path& name, int allow_maxshield, std::unique_ptr<MNERawInfo>& infop)
316{
317 FiffStream stream(name);
318
319 int res = FIFF_FAIL;
320 std::vector<FiffChInfo> chs; /* Channel info */
321 FiffCoordTrans trans; /* The coordinate transformation */
322 std::unique_ptr<FiffId> id; /* Measurement id */
323 std::unique_ptr<MNERawInfo> info;
324 int nchan = 0; /* Number of channels */
325 float sfreq = 0.0; /* Sampling frequency */
326 float highpass; /* Highpass filter frequency */
327 float lowpass; /* Lowpass filter frequency */
328 std::shared_ptr<FiffDirNode> raw;
329 FiffTime start_time;
330 int k;
331 int maxshield_data = false;
332 /*
333 * Open file
334 */
335 if (!stream.open()) {
336 stream.close();
337 return FIFF_FAIL;
338 }
339 raw = find_raw(stream.dirtree());
340 if (raw->empty()) {
341 if (allow_maxshield) {
342 raw = find_maxshield(stream.dirtree());
343 if (raw->empty()) {
344 mne::Log::critical("No raw data in this file.");
345 stream.close();
346 return FIFF_FAIL;
347 }
348 maxshield_data = true;
349 } else {
350 mne::Log::critical("No raw data in this file.");
351 stream.close();
352 return FIFF_FAIL;
353 }
354 }
355 /*
356 * Get the essential measurement information
357 */
358 if (get_meas_info(stream,
359 raw,
360 id,
361 &nchan,
362 &sfreq,
363 &highpass,
364 &lowpass,
365 chs,
366 trans,
367 start_time) < 0) {
368 stream.close();
369 return FIFF_FAIL;
370 }
371 /*
372 * Ready to put everything together
373 */
374 info = std::make_unique<MNERawInfo>();
375 info->filename = utf8_from_path(name);
376 info->nchan = nchan;
377 info->chInfo = chs;
378 info->coord_frame = FIFFV_COORD_DEVICE;
379 info->trans = std::make_unique<FiffCoordTrans>(trans);
380 info->sfreq = sfreq;
381 info->lowpass = lowpass;
382 info->highpass = highpass;
383 info->maxshield_data = maxshield_data;
384 if (id) {
385 info->id = std::make_unique<FiffId>(*id);
386 }
387 /*
388 * Getting starting time from measurement ID is not too accurate...
389 */
390 info->start_time = start_time;
391 info->buf_size = 0;
392 for (k = 0; k < raw->nent(); k++) {
393 if (raw->dir[k].kind == FIFF_DATA_BUFFER) {
394 if (raw->dir[k].type == FIFFT_DAU_PACK16 || raw->dir[k].type == FIFFT_SHORT)
395 info->buf_size = static_cast<int>(raw->dir[k].size / (nchan * sizeof(fiff_short_t)));
396 else if (raw->dir[k].type == FIFFT_FLOAT)
397 info->buf_size = static_cast<int>(raw->dir[k].size / (nchan * sizeof(fiff_float_t)));
398 else if (raw->dir[k].type == FIFFT_INT)
399 info->buf_size = static_cast<int>(raw->dir[k].size / (nchan * sizeof(fiff_int_t)));
400 else if (raw->dir[k].type == FIFFT_DOUBLE)
401 info->buf_size = static_cast<int>(raw->dir[k].size / (nchan * sizeof(double)));
402 else {
403 mne::Log::critical("We are not prepared to handle raw data type: {}", raw->dir[k].type);
404 stream.close();
405 return FIFF_FAIL;
406 }
407 break;
408 }
409 }
410 if (info->buf_size <= 0) {
411 mne::Log::critical("No raw data buffers available.");
412 stream.close();
413 return FIFF_FAIL;
414 }
415 info->rawDir = raw->dir;
416 infop = std::move(info);
417 res = FIFF_OK;
418
419 stream.close();
420 return res;
421}
