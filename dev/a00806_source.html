49// Each pair of a trial is taken out of the matching running sum, as long as both have the same length
50template<typename T>
51void subtractPairs(std::vector<std::pair<int, T>>& sums, const std::vector<std::pair<int, T>>& trial, int rows)
52{
53 if (sums.size() != trial.size())
54 return;
55 for (int i = 0; i < rows && i < std::ssize(sums); ++i)
56 sums[i].second -= trial.at(i).second;
57}
58
59// Removes the influence of one trial from the sums over all trials
60void subtractFromSums(ConnectivitySettings::IntermediateSumData& sums, const ConnectivitySettings::IntermediateTrialData& trial)
61{
62 const int rows = static_cast<int>(trial.matData.rows());
63 subtractPairs(sums.vecPairCsdSum, trial.vecPairCsd, rows);
64 subtractPairs(sums.vecPairCsdNormalizedSum, trial.vecPairCsdNormalized, rows);
65 subtractPairs(sums.vecPairCsdImagSignSum, trial.vecPairCsdImagSign, rows);
66 subtractPairs(sums.vecPairCsdImagAbsSum, trial.vecPairCsdImagAbs, rows);
67 subtractPairs(sums.vecPairCsdImagSqrdSum, trial.vecPairCsdImagSqrd, rows);
68
69 if (sums.matPsdSum.rows() == trial.matPsd.rows() && sums.matPsdSum.cols() == trial.matPsd.cols())
70 sums.matPsdSum -= trial.matPsd;
71}

172void ConnectivitySettings::removeFirst(int iAmount)
173{
174 if (m_trialData.empty()) {
175 mne::Log::debug("ConnectivitySettings::removeFirst - No elements to delete. Returning.");
176 return;
177 }
178
179 if (std::ssize(m_trialData) < iAmount) {
180 mne::Log::debug("ConnectivitySettings::removeFirst - Not enough elements stored in list in order to delete them. Returning.");
181 return;
182 }
183
184 for (int j = 0; j < iAmount; ++j) {
185 subtractFromSums(m_intermediateSumData, m_trialData[j]);
186 }
187 m_trialData.erase(m_trialData.begin(), m_trialData.begin() + iAmount);
188}
189
190//*******************************************************************************************************
191
192void ConnectivitySettings::removeLast(int iAmount)
193{
194 if (m_trialData.empty()) {
195 mne::Log::debug("ConnectivitySettings::removeLast - No elements to delete. Returning.");
196 return;
197 }
198
199 if (std::ssize(m_trialData) < iAmount) {
200 mne::Log::debug("ConnectivitySettings::removeLast - Not enough elements stored in list in order to delete them. Returning.");
201 return;
202 }
203
204 for (int j = 0; j < iAmount; ++j) {
205 subtractFromSums(m_intermediateSumData, m_trialData.back());
206 m_trialData.pop_back();
207 }
208}

339void ConnectivitySettings::setNodePositions(const MNEForwardSolution& forwardSolution, const FsSurfaceSet& surfSet)
340{
341 //Generate node vertices
342 MatrixX3f matNodeVertLeft, matNodeVertRight;
343
344 if (forwardSolution.isClustered()) {
345 auto* lhHemi = forwardSolution.src.hemisphereAt(0);
346 auto* rhHemi = forwardSolution.src.hemisphereAt(1);
347 matNodeVertLeft.resize(lhHemi->cluster_info.centroidVertno.size(), 3);
348 for (int j = 0; j < matNodeVertLeft.rows(); ++j) {
349 matNodeVertLeft.row(j) = surfSet[0].rr().row(lhHemi->cluster_info.centroidVertno.at(j)) - surfSet[0].offset().transpose();
350 }
351
352 matNodeVertRight.resize(rhHemi->cluster_info.centroidVertno.size(), 3);
353 for (int j = 0; j < matNodeVertRight.rows(); ++j) {
354 matNodeVertRight.row(j) = surfSet[1].rr().row(rhHemi->cluster_info.centroidVertno.at(j)) - surfSet[1].offset().transpose();
355 }
356 } else {
357 matNodeVertLeft.resize(forwardSolution.src[0].vertno.rows(), 3);
358 for (int j = 0; j < matNodeVertLeft.rows(); ++j) {
359 matNodeVertLeft.row(j) = surfSet[0].rr().row(forwardSolution.src[0].vertno(j)) - surfSet[0].offset().transpose();
360 }
361
362 matNodeVertRight.resize(forwardSolution.src[1].vertno.rows(), 3);
363 for (int j = 0; j < matNodeVertRight.rows(); ++j) {
364 matNodeVertRight.row(j) = surfSet[1].rr().row(forwardSolution.src[1].vertno(j)) - surfSet[1].offset().transpose();
365 }
366 }
367
368 m_matNodePositions.resize(matNodeVertLeft.rows() + matNodeVertRight.rows(), 3);
369 m_matNodePositions << matNodeVertLeft, matNodeVertRight;
370}
