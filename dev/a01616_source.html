282template<typename T>
283Eigen::VectorXi Linalg::sort(Eigen::Matrix<T, Eigen::Dynamic, 1>& v,
284 bool desc)
285{
286 std::vector<std::pair<int, T>> t_vecIdxValue;
287 Eigen::VectorXi idx(v.size());
288
289 if (v.size() > 0) {
290 for (std::int32_t i = 0; i < v.size(); ++i)
291 t_vecIdxValue.push_back(std::pair<int, T>(i, v[i]));
292
293 if (desc)
294 std::sort(t_vecIdxValue.begin(), t_vecIdxValue.end(), Linalg::compareIdxValuePairBiggerThan<T>);
295 else
296 std::sort(t_vecIdxValue.begin(), t_vecIdxValue.end(), Linalg::compareIdxValuePairSmallerThan<T>);
297
298 for (std::int32_t i = 0; i < v.size(); ++i) {
299 idx[i] = t_vecIdxValue[i].first;
300 v[i] = t_vecIdxValue[i].second;
301 }
302 }
303
304 return idx;
305}

309template<typename T>
310Eigen::VectorXi Linalg::sort(Eigen::Matrix<T, Eigen::Dynamic, 1>& v_prime,
311 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& mat,
312 bool desc)
313{
314 Eigen::VectorXi idx = Linalg::sort<T>(v_prime, desc);
315
316 if (v_prime.size() > 0) {
317 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic> newMat(mat.rows(), mat.cols());
318 for (std::int32_t i = 0; i < idx.size(); ++i)
319 newMat.col(i) = mat.col(idx[i]);
320 mat = newMat;
321 }
322
323 return idx;
324}

328template<typename T>
329std::vector<Eigen::Triplet<T>> Linalg::sortrows(const std::vector<Eigen::Triplet<T>>& A,
330 std::int32_t column)
331{
332 std::vector<Eigen::Triplet<T>> p_ASorted;
333
334 for (std::uint32_t i = 0; i < A.size(); ++i)
335 p_ASorted.push_back(A[i]);
336
337 if (column == 0)
338 std::sort(p_ASorted.begin(), p_ASorted.end(), Linalg::compareTripletFirstEntry<T>);
339 if (column == 1)
340 std::sort(p_ASorted.begin(), p_ASorted.end(), Linalg::compareTripletSecondEntry<T>);
341
342 return p_ASorted;
343}

383template<typename T>
384Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic> Linalg::pinv(const Eigen::Matrix<T,
385 Eigen::Dynamic,
386 Eigen::Dynamic>& a)
387{
388 double epsilon = std::numeric_limits<double>::epsilon();
389 Eigen::JacobiSVD<Eigen::MatrixXd> svd(a, Eigen::ComputeThinU | Eigen::ComputeThinV);
390 double tolerance = epsilon * static_cast<double>(std::max(a.cols(), a.rows())) * svd.singularValues().array().abs()(0);
391 return svd.matrixV() * (svd.singularValues().array().abs() > tolerance).select(svd.singularValues().array().inverse(), 0).matrix().asDiagonal() * svd.matrixU().adjoint();
392}
