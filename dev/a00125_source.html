64Network WeightedPhaseLagIndex::calculate(ConnectivitySettings& connectivitySettings)
65{
66 Network finalNetwork("WPLI");
67
68 if (connectivitySettings.empty()) {
69 Log::warning("WeightedPhaseLagIndex::calculate - Input data is empty");
70 return finalNetwork;
71 }
72
73 if (!connectivitySettings.keeps_trial_data()) {
74 connectivitySettings.clearIntermediateData();
75 }
76
77 finalNetwork.set_sampling_frequency(connectivitySettings.getSamplingFrequency());
78
79 //Create nodes
80 int rows = connectivitySettings.at(0).matData.rows();
81 add_nodes(finalNetwork, rows, connectivitySettings.getNodePositions());
82
83 // Check that iNfft >= signal length
84 int iSignalLength = connectivitySettings.at(0).matData.cols();
85 int iNfft = connectivitySettings.getFFTSize();
86
87 // Generate tapers
88 std::pair<MatrixXd, VectorXd> tapers = Spectral::generateTapers(iSignalLength, connectivitySettings.getWindowType());
89
90 // Initialize
91 int iNRows = connectivitySettings.at(0).matData.rows();
92 int iNFreqs = int(floor(iNfft / 2.0)) + 1;
93
94 const int binStart = connectivitySettings.frequency_bins().first;
95 const int binAmount = connectivitySettings.frequency_bins().second;
96
97 // Pass information about the FFT length. Use iNFreqs because we only use the half spectrum
98 finalNetwork.set_fft_size(iNFreqs);
99 finalNetwork.set_used_freq_bins(binAmount);
100
101 std::mutex mutex;
102
103 std::function<void(ConnectivitySettings::IntermediateTrialData&)> computeLambda = [&](ConnectivitySettings::IntermediateTrialData& inputData) {
104 compute(inputData,
105 connectivitySettings.getIntermediateSumData().vecPairCsdSum,
106 connectivitySettings.getIntermediateSumData().vecPairCsdImagAbsSum,
107 mutex,
108 iNRows,
109 iNFreqs,
110 iNfft,
111 tapers,
112 binStart,
113 binAmount,
114 connectivitySettings.keeps_trial_data());
115 };
116 // Compute WPLI in parallel for all trials
117 auto& items = connectivitySettings.getTrialData();
118 ThreadPool::global().parallel_for(0, items.size(), [&](std::size_t k) { computeLambda(items[k]); });
119 // Compute WPLI
120 computeWPLI(connectivitySettings,
121 finalNetwork);
122 return finalNetwork;
123}

127void WeightedPhaseLagIndex::compute(ConnectivitySettings::IntermediateTrialData& inputData,
128 std::vector<std::pair<int, MatrixXcd>>& vecPairCsdSum,
129 std::vector<std::pair<int, MatrixXd>>& vecPairCsdImagAbsSum,
130 std::mutex& mutex,
131 int iNRows,
132 int iNFreqs,
133 int iNfft,
134 const std::pair<MatrixXd, VectorXd>& tapers,
135 int binStart,
136 int binAmount,
137 bool keepTrialData)
138{
139 if (std::ssize(inputData.vecPairCsd) == iNRows &&
140 std::ssize(inputData.vecPairCsdImagAbs) == iNRows) {
141 return;
142 }
143
144 int i;
145
146 // Calculate tapered spectra if not available already
147 // This code was copied and changed modified Utils/Spectra since we do not want to call the function due to time loss.
148 if (std::ssize(inputData.vecTapSpectra) != iNRows) {
149 inputData.vecTapSpectra.clear();
150
151 RowVectorXd vecInputFFT, rowData;
152 RowVectorXcd vecTmpFreq;
153
154 MatrixXcd matTapSpectrum(tapers.first.rows(), iNFreqs);
155
156 FFT<double> fft;
157 fft.SetFlag(fft.HalfSpectrum);
158
159 for (i = 0; i < iNRows; ++i) {
160 // Substract mean
161 rowData.array() = inputData.matData.row(i).array() - inputData.matData.row(i).mean();
162
163 // Calculate tapered spectra if not available already
164 for (int j = 0; j < tapers.first.rows(); j++) {
165 // Zero padd if necessary. The zero padding in Eigen's FFT is only working for column vectors.
166 if (rowData.cols() < iNfft) {
167 vecInputFFT.setZero(iNfft);
168 vecInputFFT.block(0, 0, 1, rowData.cols()) = rowData.cwiseProduct(tapers.first.row(j));
169 ;
170 } else {
171 vecInputFFT = rowData.cwiseProduct(tapers.first.row(j));
172 }
173
174 // FFT for freq domain returning the half spectrum and multiply taper weights
175 fft.fwd(vecTmpFreq, vecInputFFT, iNfft);
176 matTapSpectrum.row(j) = vecTmpFreq * tapers.second(j);
177 }
178
179 inputData.vecTapSpectra.push_back(matTapSpectrum);
180 }
181 }
182
183 // Compute CSD
184 if (inputData.vecPairCsd.empty()) {
185 double denomCSD = sqrt(tapers.second.cwiseAbs2().sum()) * sqrt(tapers.second.cwiseAbs2().sum()) / 2.0;
186 bool bNfftEven = false;
187 if (iNfft % 2 == 0) {
188 bNfftEven = true;
189 }
190
191 MatrixXcd matCsd = MatrixXcd(iNRows, binAmount);
192
193 for (i = 0; i < iNRows; ++i) {
194 for (int j = i; j < iNRows; ++j) {
195 // Compute CSD (average over tapers if necessary)
196 matCsd.row(j) = inputData.vecTapSpectra.at(i).block(0, binStart, inputData.vecTapSpectra.at(i).rows(), binAmount).cwiseProduct(inputData.vecTapSpectra.at(j).block(0, binStart, inputData.vecTapSpectra.at(j).rows(), binAmount).conjugate()).colwise().sum() / denomCSD;
197
198 // Divide first and last element by 2 due to half spectrum
199 if (binStart == 0) {
200 matCsd.row(j)(0) /= 2.0;
201 }
202
203 if (bNfftEven && binStart + binAmount >= iNFreqs) {
204 matCsd.row(j).tail(1) /= 2.0;
205 }
206 }
207
208 inputData.vecPairCsd.push_back(std::pair<int, MatrixXcd>(i, matCsd));
209 inputData.vecPairCsdImagAbs.push_back(std::pair<int, MatrixXd>(i, matCsd.imag().cwiseAbs()));
210 }
211 {
212 const std::lock_guard lock(mutex);
213
214 if (vecPairCsdSum.empty()) {
215 vecPairCsdSum = inputData.vecPairCsd;
216 vecPairCsdImagAbsSum = inputData.vecPairCsdImagAbs;
217 } else {
218 for (int j = 0; j < std::ssize(vecPairCsdSum); ++j) {
219 vecPairCsdSum[j].second += inputData.vecPairCsd.at(j).second;
220 vecPairCsdImagAbsSum[j].second += inputData.vecPairCsdImagAbs.at(j).second;
221 }
222 }
223 }
224 } else {
225 if (inputData.vecPairCsdImagAbs.empty()) {
226 inputData.vecPairCsdImagAbs.clear();
227 for (i = 0; i < std::ssize(inputData.vecPairCsd); ++i) {
228 inputData.vecPairCsdImagAbs.push_back(std::pair<int, MatrixXd>(i, inputData.vecPairCsd.at(i).second.imag().cwiseAbs()));
229 }
230
231 {
232 const std::lock_guard lock(mutex);
233
234 if (vecPairCsdImagAbsSum.empty()) {
235 vecPairCsdImagAbsSum = inputData.vecPairCsdImagAbs;
236 } else {
237 for (int j = 0; j < std::ssize(vecPairCsdImagAbsSum); ++j) {
238 vecPairCsdImagAbsSum[j].second += inputData.vecPairCsdImagAbs.at(j).second;
239 }
240 }
241 }
242 }
243 }
244
245 //Do not store data to save memory
246 if (!keepTrialData) {
247 inputData.vecPairCsd.clear();
248 inputData.vecPairCsdImagAbs.clear();
249 inputData.vecTapSpectra.clear();
250 }
251}

255void WeightedPhaseLagIndex::computeWPLI(ConnectivitySettings& connectivitySettings,
256 Network& finalNetwork)
257{
258 // Compute final WPLI and create Network
259 MatrixXd matDenom, matNom;
260 MatrixXd matWeight;
261 int j;
262
263 for (int i = 0; i < std::ssize(connectivitySettings.getIntermediateSumData().vecPairCsdSum); ++i) {
264 matDenom = connectivitySettings.getIntermediateSumData().vecPairCsdImagAbsSum.at(i).second;
265 matDenom = (matDenom.array() == 0.).select(INFINITY, matDenom);
266
267 matNom = connectivitySettings.getIntermediateSumData().vecPairCsdSum.at(i).second.imag().cwiseAbs().cwiseQuotient(matDenom);
268
269 for (j = i; j < matNom.rows(); ++j) {
270 matWeight = matNom.row(j).transpose();
271
272 finalNetwork.add_edge(NetworkEdge(i, j, std::move(matWeight)));
273 }
274 }
