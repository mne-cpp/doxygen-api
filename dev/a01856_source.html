144void SpharaPanel::update_system_fields()
145{
146 const bool updating = std::exchange(m_updating, true);
147 SettingsForm& f = form();
148 const std::string current = system();
149 f.set_value(kHasSecond, current != "EEG");
150 if (current == "VectorView") {
151 f.set_label(kFirst, "Mag");
152 f.set_range(kFirst, 1, 102);
153 f.set_label(kSecond, "Grad");
154 f.set_range(kSecond, 1, 102);
155 } else if (current == "BabyMEG") {
156 f.set_label(kFirst, "Inner layer");
157 f.set_range(kFirst, 1, 270);
158 f.set_label(kSecond, "Outer layer");
159 f.set_range(kSecond, 1, 105);
160 } else if (current == "EEG") {
161 f.set_label(kFirst, "EEG");
162 f.set_range(kFirst, 1, 256);
163 }
164 m_updating = updating;
