81bool PolhemusCoregistration::captureCurrentPenPositionAsFiducial(FiducialId id)
82{
83 if (!m_havePenPos) {
84 return false;
85 }
86
87 static const char* labels[] = { nullptr, "LPA", "NAS", "RPA" };
88 const int ident = static_cast<int>(id);
89
90 // Store pen fiducial BEFORE append (which emits pointsChanged)
91 m_penFid[ident] = m_penPosition;
92 m_hasPenFid[ident] = true;
93
94 qInfo().nospace() << "Fiducial captured: " << labels[ident]
95 << " (" << m_penPosition.x()*1000.f << ", "
96 << m_penPosition.y()*1000.f << ", "
97 << m_penPosition.z()*1000.f << ") mm";
98
99 // Log distances to previously captured fiducials
100 for (int j = 1; j <= 3; ++j) {
101 if (j != ident && m_hasPenFid[j]) {
102 float dist = (m_penFid[ident] - m_penFid[j]).length() * 1000.0f;
103 qInfo().nospace() << " " << labels[ident] << " ↔ " << labels[j]
104 << ": " << dist << " mm"
105 << (dist > 200.0f ? " *** WARNING: > 200 mm!" : "");
106 }
107 }
108
109 m_pPoints->removeFiducial(id);
110
111 DigitizedPoint dp;
112 dp.kind = PointKind::Fiducial;
113 dp.label = QString::fromLatin1(labels[ident]);
114 dp.identNumber = ident;
115 dp.position = m_penPosition;
116 m_pPoints->append(dp);
117 return true;
118}

195bool PolhemusCoregistration::computeRegistration()
196{
197 if (!hasAllPenFiducials()) {
198 return false;
199 }
200
201 // Pen fiducial positions in Polhemus world frame (metres)
202 const QVector3D pNas = m_penFid[static_cast<int>(FiducialId::NAS)];
203 const QVector3D pLpa = m_penFid[static_cast<int>(FiducialId::LPA)];
204 const QVector3D pRpa = m_penFid[static_cast<int>(FiducialId::RPA)];
205
206 // Spread check: reject degenerate input
207 const float dNL = (pNas - pLpa).length() * 1000.0f;
208 const float dNR = (pNas - pRpa).length() * 1000.0f;
209 const float dLR = (pLpa - pRpa).length() * 1000.0f;
210 const float minSpread = std::min({dNL, dNR, dLR});
211 if (minSpread < 20.0f) {
212 qWarning() << "Registration failed: pen fiducials too close"
213 << "(min spread" << minSpread << "mm, need > 20 mm)";
214 m_registrationValid = false;
215 emit registrationChanged();
216 return false;
217 }
218
219 // --- Paired path: SVD-based rigid registration (Kabsch / Procrustes) ---
220 //
221 // Finds the optimal rotation R and translation t that map pen
222 // fiducials to model fiducials: model_pos ≈ R * pen_pos + t
223 //
224 // The Kabsch algorithm guarantees det(R) = +1 (proper rotation,
225 // no reflection) regardless of how the two coordinate systems are
226 // oriented. This avoids the left-right / up-down inversion bugs
227 // that plagued the old buildFrame approach, where an asymmetric
228 // vertex correction could flip ey or ez in one frame but not the
229 // other.
230 if (hasAllModelFiducials()) {
231 const QVector3D mNas = m_modelFid[static_cast<int>(FiducialId::NAS)];
232 const QVector3D mLpa = m_modelFid[static_cast<int>(FiducialId::LPA)];
233 const QVector3D mRpa = m_modelFid[static_cast<int>(FiducialId::RPA)];
234
235 // Debug: raw fiducial positions
236 qInfo().nospace()
237 << "Registration — pen fiducial positions (Polhemus world, mm):"
238 << "\n NAS: (" << pNas.x()*1000.f << ", " << pNas.y()*1000.f << ", " << pNas.z()*1000.f << ")"
239 << "\n LPA: (" << pLpa.x()*1000.f << ", " << pLpa.y()*1000.f << ", " << pLpa.z()*1000.f << ")"
240 << "\n RPA: (" << pRpa.x()*1000.f << ", " << pRpa.y()*1000.f << ", " << pRpa.z()*1000.f << ")";
241 qInfo().nospace()
242 << "Registration — model fiducial positions (MRI/surface-RAS, mm):"
243 << "\n NAS: (" << mNas.x()*1000.f << ", " << mNas.y()*1000.f << ", " << mNas.z()*1000.f << ")"
244 << "\n LPA: (" << mLpa.x()*1000.f << ", " << mLpa.y()*1000.f << ", " << mLpa.z()*1000.f << ")"
245 << "\n RPA: (" << mRpa.x()*1000.f << ", " << mRpa.y()*1000.f << ", " << mRpa.z()*1000.f << ")";
246 qInfo().nospace()
247 << "Registration — axis mirror: mirrorX=" << m_mirrorX << " mirrorY=" << m_mirrorY;
248
249 // Compare pen vs model fiducial distances — should be similar for the same head
250 const float mNL = (mNas - mLpa).length() * 1000.0f;
251 const float mNR = (mNas - mRpa).length() * 1000.0f;
252 const float mLR = (mLpa - mRpa).length() * 1000.0f;
253 qInfo().nospace()
254 << "Registration — fiducial distances (pen / model):"
255 << "\n NAS ↔ LPA: " << dNL << " / " << mNL << " mm (ratio " << dNL/mNL << ")"
256 << "\n NAS ↔ RPA: " << dNR << " / " << mNR << " mm (ratio " << dNR/mNR << ")"
257 << "\n LPA ↔ RPA: " << dLR << " / " << mLR << " mm (ratio " << dLR/mLR << ")";
258
259 const float maxRatio = std::max({dNL/mNL, dNR/mNR, dLR/mLR});
260 const float minRatio = std::min({dNL/mNL, dNR/mNR, dLR/mLR});
261 if (maxRatio / minRatio > 2.0f) {
262 qWarning() << "Registration: shape mismatch — pen fiducial triangle"
263 << "has very different proportions from model."
264 << "Check fiducial placement!";
265 }
266
267 // --- Kabsch algorithm (SVD least-squares rigid alignment) ---
268
269 // 1. Centroids
270 const Eigen::Vector3d penC(
271 (pNas.x() + pLpa.x() + pRpa.x()) / 3.0,
272 (pNas.y() + pLpa.y() + pRpa.y()) / 3.0,
273 (pNas.z() + pLpa.z() + pRpa.z()) / 3.0);
274 const Eigen::Vector3d modC(
275 (mNas.x() + mLpa.x() + mRpa.x()) / 3.0,
276 (mNas.y() + mLpa.y() + mRpa.y()) / 3.0,
277 (mNas.z() + mLpa.z() + mRpa.z()) / 3.0);
278
279 // 2. Centered point matrices (3×3, each column = one centered point)
280 Eigen::Matrix3d P, Q;
281 P.col(0) = Eigen::Vector3d(pNas.x(), pNas.y(), pNas.z()) - penC;
282 P.col(1) = Eigen::Vector3d(pLpa.x(), pLpa.y(), pLpa.z()) - penC;
283 P.col(2) = Eigen::Vector3d(pRpa.x(), pRpa.y(), pRpa.z()) - penC;
284
285 Q.col(0) = Eigen::Vector3d(mNas.x(), mNas.y(), mNas.z()) - modC;
286 Q.col(1) = Eigen::Vector3d(mLpa.x(), mLpa.y(), mLpa.z()) - modC;
287 Q.col(2) = Eigen::Vector3d(mRpa.x(), mRpa.y(), mRpa.z()) - modC;
288
289 // 3. Cross-covariance matrix H = P * Qᵀ
290 const Eigen::Matrix3d H = P * Q.transpose();
291
292 // 4. SVD of H
293 Eigen::JacobiSVD<Eigen::Matrix3d> svd(H, Eigen::ComputeFullU | Eigen::ComputeFullV);
294 const Eigen::Matrix3d U = svd.matrixU();
295 const Eigen::Matrix3d V = svd.matrixV();
296
297 // 5. Optimal rotation — ensure proper rotation (det = +1)
298 const double d = (V * U.transpose()).determinant();
299 Eigen::Matrix3d D = Eigen::Matrix3d::Identity();
300 D(2, 2) = (d > 0.0) ? 1.0 : -1.0;
301 Eigen::Matrix3d R = V * D * U.transpose();
302
303 // 6. Translation
304 Eigen::Vector3d t = modC - R * penC;
305
306 const Eigen::Vector3d sv = svd.singularValues();
307 qInfo().nospace()
308 << "Registration — Kabsch SVD:"
309 << "\n det(V*Uᵀ) = " << d << (d < 0 ? " (det correction applied)" : " (no det correction)")
310 << "\n det(R) = " << R.determinant()
311 << "\n singular values: (" << sv(0) << ", " << sv(1) << ", " << sv(2) << ")";
312
313 // 6b. Vertex disambiguation for coplanar degeneracy.
314 //
315 // With only 3 coplanar fiducials the SVD's 3rd singular value
316 // is ~0, leaving the out-of-plane rotation direction ambiguous.
317 // The det(V*Uᵀ) sign heuristic picks one direction but may
318 // choose wrong, inverting superior ↔ inferior.
319 //
320 // Fix: if a pen vertex (CZ) and model vertex are available,
321 // test both candidate rotations (D₃₃ = +1 and D₃₃ = -1) and
322 // keep whichever maps pen CZ closer to model CZ.
323 if (m_hasPenVertex && m_hasModelVertex) {
324 const Eigen::Vector3d penCZ(m_penVertex.x(), m_penVertex.y(), m_penVertex.z());
325 const Eigen::Vector3d modCZ(m_modelVertex.x(), m_modelVertex.y(), m_modelVertex.z());
326
327 const Eigen::Vector3d mappedCZ = R * penCZ + t;
328 const double errCurrent = (mappedCZ - modCZ).norm();
329
330 // Try the alternative sign
331 Eigen::Matrix3d D_alt = D;
332 D_alt(2, 2) = -D(2, 2);
333 const Eigen::Matrix3d R_alt = V * D_alt * U.transpose();
334 const Eigen::Vector3d t_alt = modC - R_alt * penC;
335 const Eigen::Vector3d mappedCZ_alt = R_alt * penCZ + t_alt;
336 const double errAlt = (mappedCZ_alt - modCZ).norm();
337
338 qInfo().nospace()
339 << "Registration — vertex disambiguation:"
340 << "\n CZ (pen): (" << m_penVertex.x()*1000.f << ", " << m_penVertex.y()*1000.f << ", " << m_penVertex.z()*1000.f << ") mm"
341 << "\n CZ (model): (" << m_modelVertex.x()*1000.f << ", " << m_modelVertex.y()*1000.f << ", " << m_modelVertex.z()*1000.f << ") mm"
342 << "\n D₃₃=" << D(2,2) << " → mapped CZ err: " << errCurrent*1000.0 << " mm"
343 << "\n D₃₃=" << D_alt(2,2) << " → mapped CZ err: " << errAlt*1000.0 << " mm";
344
345 if (errAlt < errCurrent) {
346 R = R_alt;
347 t = t_alt;
348 D = D_alt;
349 qInfo().nospace()
350 << "Registration — vertex correction APPLIED: flipped out-of-plane direction"
351 << " (det(R) now = " << R.determinant() << ")";
352 } else {
353 qInfo() << "Registration — vertex check passed, no flip needed.";
354 }
355 } else {
356 qInfo() << "Registration — no vertex available for out-of-plane disambiguation."
357 << "Capture CZ (pen vertex) and load BEM to enable this check.";
358 }
359
360 // 7. Build QMatrix4x4 worldToModel = [R | t]
361 m_worldToModel.setToIdentity();
362 for (int r = 0; r < 3; ++r) {
363 for (int c = 0; c < 3; ++c) {
364 m_worldToModel(r, c) = static_cast<float>(R(r, c));
365 }
366 m_worldToModel(r, 3) = static_cast<float>(t(r));
367 }
368
369 // Residuals
370 auto residual = [&](const char* label, const QVector3D& pw, const QVector3D& pm) {
371 QVector3D mapped = m_worldToModel.map(pw);
372 float err = (mapped - pm).length() * 1000.0f;
373 qInfo().nospace() << " " << label
374 << ": pen(" << pw.x()*1000.f << ", " << pw.y()*1000.f << ", " << pw.z()*1000.f << ")"
375 << " → mapped(" << mapped.x()*1000.f << ", " << mapped.y()*1000.f << ", " << mapped.z()*1000.f << ")"
376 << " model(" << pm.x()*1000.f << ", " << pm.y()*1000.f << ", " << pm.z()*1000.f << ")"
377 << " err=" << err << " mm";
378 return err;
379 };
380 const float rNas = residual("NAS", pNas, mNas);
381 const float rLpa = residual("LPA", pLpa, mLpa);
382 const float rRpa = residual("RPA", pRpa, mRpa);
383
384 m_headToWorld.setToIdentity();
385 m_headToDevice = m_deviceToWorld.inverted() * m_headToWorld;
386 m_registrationValid = true;
387 qInfo() << "Registration succeeded (analytical paired).";
388 emit registrationChanged();
389 return true;
390 }
391
392 // --- Fallback: head-frame-only registration ---
393 const QMatrix4x4 headFrame = buildHeadFrame();
394 const QVector3D ez(headFrame(0, 2), headFrame(1, 2), headFrame(2, 2));
395 if (ez.length() < 1e-6f) {
396 return false;
397 }
398
399 m_headToWorld = headFrame;
400 m_headToDevice = m_deviceToWorld.inverted() * m_headToWorld;
401 m_worldToModel.setToIdentity();
402 m_registrationValid = true;
403 qInfo() << "Registration succeeded (head-frame fallback).";
404 emit registrationChanged();
405 return true;
406}

410void PolhemusCoregistration::onPointReceived(int station,
411 const QVector3D& position,
412 const QQuaternion& orientation)
413{
414 // Apply axis mirroring to compensate for transmitter placement
415 const QVector3D pos(m_mirrorX ? -position.x() : position.x(),
416 m_mirrorY ? -position.y() : position.y(),
417 position.z());
418
419 if (station == m_trackerStation) {
420 m_deviceToWorld = buildDevicePose(pos, orientation);
421 emit devicePoseChanged(m_deviceToWorld);
422 } else if (station == m_penStation) {
423 // Gimbal-lock guard: for ZYX Euler, sin(el) = 2*(w*y - x*z).
424 // When |el| > 80° the Euler→quaternion conversion is unreliable,
425 // so freeze the pen pose at its last good value.
426 const float sinEl = 2.0f * (orientation.scalar() * orientation.y()
427 - orientation.x() * orientation.z());
428 constexpr float kGimbalSinEl = 0.9848f; // sin(80°)
429 const bool gimbalLock = (std::abs(sinEl) > kGimbalSinEl);
430
431 if (!gimbalLock) {
432 const QVector3D tipAdj = m_tipOffsetEnabled
433 ? orientation.rotatedVector(m_penTipOffset) : QVector3D();
434 m_penPosition = pos + tipAdj;
435 m_penOrientation = orientation;
436 m_havePenPos = true;
437 }
438
439 // Collect raw (un-offset) samples during pivot calibration.
440 // Only keep samples with sufficient angular change from the last
441 // accepted sample to avoid redundant near-identical rows in the SVD.
442 // Also reject position jumps and gimbal-lock orientations.
443 if (m_pivotState == PivotState::Collecting) {
444 constexpr float kMinAngleDeg = 3.0f;
445 constexpr float kMaxPosJumpM = 0.05f; // 5 cm
446 constexpr float kGimbalSinEl2 = 0.9848f; // sin(80°) — reject |el|>80°
447
448 // Gimbal-lock check: for ZYX Euler, sin(el) = 2*(w*y - x*z)
449 const float sinEl2 = 2.0f * (orientation.scalar() * orientation.y()
450 - orientation.x() * orientation.z());
451 if (std::abs(sinEl2) > kGimbalSinEl2) {
452 // Near gimbal lock — skip this sample silently
453 } else {
454 bool accept = m_pivotOrientations.empty();
455 if (!accept) {
456 const QQuaternion& prev = m_pivotOrientations.back();
457 float dot = std::abs(QQuaternion::dotProduct(prev, orientation));
458 float angleDeg = 2.0f * std::acos(std::min(dot, 1.0f)) * (180.0f / 3.14159265f);
459 float posDelta = (pos - m_pivotPositions.back()).length();
460 accept = (angleDeg >= kMinAngleDeg) && (posDelta < kMaxPosJumpM);
461 }
462 if (accept) {
463 m_pivotPositions.push_back(pos);
464 m_pivotOrientations.push_back(orientation);
465
466 // Compute angular span for live feedback
467 float spanDeg = 0.0f;
468 if (m_pivotOrientations.size() > 1) {
469 float minDot = 1.0f;
470 const auto& first = m_pivotOrientations.front();
471 for (size_t k = 1; k < m_pivotOrientations.size(); ++k) {
472 float d = std::abs(QQuaternion::dotProduct(first, m_pivotOrientations[k]));
473 if (d < minDot) minDot = d;
474 }
475 spanDeg = 2.0f * std::acos(std::min(minDot, 1.0f)) * (180.0f / 3.14159265f);
476 }
477 emit pivotSampleCollected(static_cast<int>(m_pivotPositions.size()), spanDeg);
478 }
479 }
480 }
481
482 emit penPoseChanged(m_penPosition, m_penOrientation);
483 }
484}

486void PolhemusCoregistration::onPenButtonPressedFromConn(int station,
487 const QVector3D& position,
488 const QQuaternion& orientation)
489{
490 if (station == m_penStation) {
491 // Apply axis mirroring to compensate for transmitter placement
492 const QVector3D pos(m_mirrorX ? -position.x() : position.x(),
493 m_mirrorY ? -position.y() : position.y(),
494 position.z());
495
496 const QVector3D tipAdj = m_tipOffsetEnabled
497 ? orientation.rotatedVector(m_penTipOffset) : QVector3D();
498 m_penPosition = pos + tipAdj;
499 m_penOrientation = orientation;
500 m_havePenPos = true;
501
502 // Pivot calibration state machine
503 if (m_pivotState == PivotState::WaitingForStart) {
504 m_pivotState = PivotState::Collecting;
505 m_pivotPositions.clear();
506 m_pivotOrientations.clear();
507 qInfo() << "Pivot calibration: collecting — pivot the pen around its tip";
508 emit pivotStateChanged(m_pivotState);
509 return;
510 }
511 if (m_pivotState == PivotState::Collecting) {
512 solvePivotCalibration();
513 return;
514 }
515
516 emit penButtonPressed(m_penPosition, orientation);
517 }

522QMatrix4x4 PolhemusCoregistration::buildDevicePose(const QVector3D& trackerPos,
523 const QQuaternion& trackerOri) const
524{
525 QMatrix4x4 trackerToWorld;

538QMatrix4x4 PolhemusCoregistration::buildHeadFrame() const
539{
540 const QVector3D nas = m_pPoints->fiducial(FiducialId::NAS);
541 const QVector3D lpa = m_pPoints->fiducial(FiducialId::LPA);
542 const QVector3D rpa = m_pPoints->fiducial(FiducialId::RPA);
543
544 const QVector3D origin = (lpa + rpa) * 0.5f;
545 const QVector3D ex = (nas - origin).normalized();
546 const QVector3D eyApprox = (lpa - origin).normalized();
547 // Gram-Schmidt: orthogonalize ey against ex, preserving LPA direction
548 const QVector3D ey = (eyApprox - QVector3D::dotProduct(eyApprox, ex) * ex).normalized();
549 const QVector3D ez = QVector3D::crossProduct(ex, ey).normalized();
550
551 QMatrix4x4 frame;
552 frame.setToIdentity();
553 frame(0, 0) = ex.x(); frame(0, 1) = ey.x(); frame(0, 2) = ez.x(); frame(0, 3) = origin.x();
554 frame(1, 0) = ex.y(); frame(1, 1) = ey.y(); frame(1, 2) = ez.y(); frame(1, 3) = origin.y();
555 frame(2, 0) = ex.z(); frame(2, 1) = ey.z(); frame(2, 2) = ez.z(); frame(2, 3) = origin.z();
556 frame(3, 0) = 0.0f; frame(3, 1) = 0.0f; frame(3, 2) = 0.0f; frame(3, 3) = 1.0f;
557 return frame;
558}

582bool PolhemusCoregistration::solvePivotCalibration()
583{
584 const int N = static_cast<int>(m_pivotPositions.size());
585 if (N < 10) {
586 qWarning() << "Pivot calibration: only" << N << "samples, need at least 10";
587 m_pivotState = PivotState::Idle;
588 emit pivotStateChanged(m_pivotState);
589 return false;
590 }
591
592 // Build the linear system A * x = b (double precision)
593 // where x = [offset(3); tipPos(3)]
594 // For each sample i: R_i * offset - I * tipPos = -p_i
595 Eigen::MatrixXd A(3 * N, 6);
596 Eigen::VectorXd b(3 * N);
597
598 for (int i = 0; i < N; ++i) {
599 const QVector3D& p = m_pivotPositions[static_cast<size_t>(i)];
600 const QQuaternion& q = m_pivotOrientations[static_cast<size_t>(i)];
601
602 // Extract 3x3 rotation matrix directly from quaternion
603 const QMatrix3x3 rm = q.toRotationMatrix();
604
605 const int row = 3 * i;
606 for (int r = 0; r < 3; ++r) {
607 for (int c = 0; c < 3; ++c) {
608 A(row + r, c) = static_cast<double>(rm(r, c)); // R_i
609 A(row + r, 3 + c) = (r == c) ? -1.0 : 0.0; // -I
610 }
611 }
612 b(row + 0) = -static_cast<double>(p.x());
613 b(row + 1) = -static_cast<double>(p.y());
614 b(row + 2) = -static_cast<double>(p.z());
615 }
616
617 // Solve via SVD least squares
618 Eigen::JacobiSVD<Eigen::MatrixXd> svd(A, Eigen::ComputeThinU | Eigen::ComputeThinV);
619
620 // Check condition number for rank deficiency (insufficient angular diversity)
621 const auto& sv = svd.singularValues();
622 double cond = sv(0) / sv(sv.size() - 1);
623 if (cond > 1e6) {
624 qWarning() << "Pivot calibration: ill-conditioned (cond =" << cond
625 << "). Pivot the pen through a wider range of angles.";
626 m_pivotState = PivotState::Idle;
627 emit pivotStateChanged(m_pivotState);
628 return false;
629 }
630
631 Eigen::VectorXd x = svd.solve(b);
632
633 QVector3D offset(static_cast<float>(x(0)),
634 static_cast<float>(x(1)),
635 static_cast<float>(x(2)));
636 QVector3D tipPos(static_cast<float>(x(3)),
637 static_cast<float>(x(4)),
638 static_cast<float>(x(5)));
639
640 // Compute RMS residual
641 double sumSq = 0.0;
642 for (int i = 0; i < N; ++i) {
643 const QVector3D& p = m_pivotPositions[static_cast<size_t>(i)];
644 const QQuaternion& q = m_pivotOrientations[static_cast<size_t>(i)];
645 QVector3D computed = p + q.rotatedVector(offset);
646 float err = (computed - tipPos).length();
647 sumSq += static_cast<double>(err) * static_cast<double>(err);
648 }
649 float rms = static_cast<float>(std::sqrt(sumSq / static_cast<double>(N))) * 1000.0f;
650 m_pivotResidualMm = rms;
651
652 qInfo() << "Pivot calibration:" << N << "samples, cond =" << cond
653 << ", offset =" << offset.x() * 1000.0f << offset.y() * 1000.0f
654 << offset.z() * 1000.0f << "mm, RMS residual =" << rms << "mm";
655
656 m_penTipOffset = offset;
657 m_pivotState = PivotState::Done;
658 emit pivotStateChanged(m_pivotState);
659 emit pivotCalibrationDone(offset, rms);
660 return true;
661}

667bool PolhemusCoregistration::captureOpticalCalibSample()
668{
669 if (!m_havePenPos) {
670 qWarning() << "Optical calibration: no pen position available";
671 return false;
672 }
673
674 const QMatrix4x4& dev = m_deviceToWorld;
675 if (dev.isIdentity()) {
676 qWarning() << "Optical calibration: no tracker data available";
677 return false;
678 }
679
680 // Extract tracker position and orientation from deviceToWorld.
681 // deviceToWorld = trackerToWorld * offset, but we want the raw tracker
682 // pose (before offset). Reconstruct from the current raw tracker data
683 // by removing the offset: trackerToWorld = deviceToWorld * offset^-1
684 QMatrix4x4 offsetInv;
685 offsetInv.setToIdentity();
686 offsetInv.translate(m_offsetTranslation);
687 offsetInv.rotate(m_offsetRotation);
688 offsetInv = offsetInv.inverted();
689
690 const QMatrix4x4 trackerToWorld = dev * offsetInv;
691 const QVector3D trackerPos(trackerToWorld(0, 3), trackerToWorld(1, 3), trackerToWorld(2, 3));
692 const QQuaternion trackerOri = QQuaternion::fromRotationMatrix(trackerToWorld.toGenericMatrix<3, 3>());
693
694 OpticalCalibSample sample;
695 sample.trackerPos = trackerPos;
696 sample.trackerOri = trackerOri;
697 sample.focusPoint = m_penPosition;
698 m_opticalCalibSamples.push_back(sample);
699
700 const int n = static_cast<int>(m_opticalCalibSamples.size());
701 const QVector3D localFocus = trackerOri.inverted().rotatedVector(m_penPosition - trackerPos);
702 const float distToTracker = localFocus.length() * 1000.0f;
703
704 qInfo().nospace()
705 << "Optical calibration sample " << n << ":"
706 << "\n tracker pos: (" << trackerPos.x()*1000.f << ", " << trackerPos.y()*1000.f << ", " << trackerPos.z()*1000.f << ") mm"
707 << "\n focus point: (" << m_penPosition.x()*1000.f << ", " << m_penPosition.y()*1000.f << ", " << m_penPosition.z()*1000.f << ") mm"
708 << "\n focus (local): (" << localFocus.x()*1000.f << ", " << localFocus.y()*1000.f << ", " << localFocus.z()*1000.f << ") mm"
709 << "\n distance tracker\u2194focus: " << distToTracker << " mm";
710
711 // Log convergence angle between tracker→focus directions for successive samples.
712 // As the OPMI moves farther away, this angle shrinks (lines become parallel);
713 // at close range the angle is large because the ~20 cm tracker offset dominates.
714 if (n >= 2) {
715 const auto& prev = m_opticalCalibSamples[static_cast<size_t>(n - 2)];
716 const QVector3D prevLocal = prev.trackerOri.inverted().rotatedVector(prev.focusPoint - prev.trackerPos);
717 const QVector3D curDir = localFocus.normalized();
718 const QVector3D prevDir = prevLocal.normalized();
719 const float dot = std::clamp(QVector3D::dotProduct(curDir, prevDir), -1.0f, 1.0f);
720 const float angleDeg = std::acos(dot) * (180.0f / 3.14159265f);
721 const float prevDist = prevLocal.length() * 1000.0f;
722 qInfo().nospace()
723 << " convergence angle (sample " << n-1 << "\u2194" << n << "): " << angleDeg << "\u00b0"
724 << " (distances: " << prevDist << " / " << distToTracker << " mm)";
725 }

738
739bool PolhemusCoregistration::solveOpticalCalibration()
740{
741 const int N = static_cast<int>(m_opticalCalibSamples.size());
742 if (N < 2) {
743 qWarning() << "Optical calibration: need at least 2 samples, have" << N;
744 return false;
745 }
746
747 // Step 1: Transform all focus points into the tracker's local frame
748 std::vector<Eigen::Vector3d> localPoints(static_cast<size_t>(N));
749 for (int i = 0; i < N; ++i) {
750 const auto& s = m_opticalCalibSamples[static_cast<size_t>(i)];
751 const QVector3D local = s.trackerOri.inverted().rotatedVector(s.focusPoint - s.trackerPos);
752 localPoints[static_cast<size_t>(i)] = Eigen::Vector3d(local.x(), local.y(), local.z());
753 }
754
755 // Step 2: Compute centroid
756 Eigen::Vector3d centroid = Eigen::Vector3d::Zero();
757 for (const auto& p : localPoints) centroid += p;
758 centroid /= static_cast<double>(N);
759
760 // Step 3: PCA — fit line via SVD of the centered point matrix
761 Eigen::MatrixXd centered(3, N);
762 for (int i = 0; i < N; ++i) {
763 centered.col(i) = localPoints[static_cast<size_t>(i)] - centroid;
764 }
765
766 Eigen::JacobiSVD<Eigen::MatrixXd> svd(centered, Eigen::ComputeThinU);
767 const Eigen::Vector3d axis = svd.matrixU().col(0); // principal direction
768
769 // Ensure the axis points away from the tracker origin (toward the focus points)
770 Eigen::Vector3d axisDir = axis;
771 if (centroid.dot(axisDir) < 0.0) {
772 axisDir = -axisDir;
773 }
774
775 // Step 4: Find the point on the fitted line closest to the tracker origin (0,0,0)
776 // Line: P = centroid + t * axisDir
777 // Closest point to origin: t = -dot(centroid, axisDir)
778 const double t0 = -centroid.dot(axisDir);
779 const Eigen::Vector3d opticalCenter = centroid + t0 * axisDir;
780
781 // Step 5: Compute RMS residual (perpendicular distance from each point to the line)
782 double sumSq = 0.0;
783 for (const auto& p : localPoints) {
784 const Eigen::Vector3d diff = p - centroid;
785 const double along = diff.dot(axisDir);
786 const double perpSq = (diff - along * axisDir).squaredNorm();
787 sumSq += perpSq;
788 }
789 const double rms = std::sqrt(sumSq / static_cast<double>(N)) * 1000.0; // mm
790
791 // Step 6: Check collinearity — with only 2 points the line is exact.
792 // With 3+ points, warn if residual is large.
793 const auto& sv = svd.singularValues();
794 const double linearityRatio = (sv.size() > 1 && sv(0) > 1e-12) ? sv(1) / sv(0) : 0.0;
795
796 // Compute depth spread: range of focal distances along the fitted axis.
797 // The tracker→focus line and the optical axis converge at the focus point
798 // but diverge at the tracker end due to the ~20 cm offset. At large
799 // focal distances the lines are nearly parallel and the angular difference
800 // between them is small; at close range the angle is large (approaching
801 // 90° when focus point is at the optical center). Good calibration
802 // requires samples at varied focal distances so the PCA can reliably
803 // separate the axis direction from noise.
804 double minDepth = 1e30, maxDepth = -1e30;
805 for (const auto& p : localPoints) {
806 const double depth = (p - opticalCenter).dot(axisDir);
807 minDepth = std::min(minDepth, depth);
808 maxDepth = std::max(maxDepth, depth);
809 }
810 const double depthSpreadMm = (maxDepth - minDepth) * 1000.0;
811
812 m_opticalAxisLocal = QVector3D(static_cast<float>(axisDir.x()),
813 static_cast<float>(axisDir.y()),
814 static_cast<float>(axisDir.z()));
815 m_opticalCenterLocal = QVector3D(static_cast<float>(opticalCenter.x()),
816 static_cast<float>(opticalCenter.y()),
817 static_cast<float>(opticalCenter.z()));
818 m_opticalCalibResidualMm = static_cast<float>(rms);
819 m_opticalCalibDepthSpreadMm = static_cast<float>(depthSpreadMm);
820 m_opticalCalibValid = true;
821
822 const float distMm = m_opticalCenterLocal.length() * 1000.0f;
823
824 qInfo().nospace()
825 << "Optical calibration solved (" << N << " samples):"
826 << "\n axis (local): (" << m_opticalAxisLocal.x() << ", " << m_opticalAxisLocal.y() << ", " << m_opticalAxisLocal.z() << ")"
827 << "\n center (local): (" << m_opticalCenterLocal.x()*1000.f << ", " << m_opticalCenterLocal.y()*1000.f << ", " << m_opticalCenterLocal.z()*1000.f << ") mm"
828 << "\n tracker\u2194center: " << distMm << " mm"
829 << "\n RMS residual: " << rms << " mm"
830 << "\n linearity: " << (1.0 - linearityRatio) * 100.0 << "% (sv ratio " << linearityRatio << ")"
831 << "\n depth spread: " << depthSpreadMm << " mm (range " << minDepth*1000.0 << " .. " << maxDepth*1000.0 << " mm)";
832
833 // Per-sample: report convergence angle between tracker→focus direction
834 // and fitted optical axis. This angle shrinks as the OPMI moves farther
835 // from the head (lines become parallel) and grows at close range where
836 // the tracker offset dominates.
837 qInfo() << " Per-sample convergence angles (tracker\u2192focus vs optical axis):";
838 for (int i = 0; i < N; ++i) {
839 const Eigen::Vector3d& p = localPoints[static_cast<size_t>(i)];
840 const Eigen::Vector3d trkToFocus = p.normalized();
841 const double cosAngle = std::clamp(trkToFocus.dot(axisDir), -1.0, 1.0);
842 const double angleDeg = std::acos(cosAngle) * (180.0 / 3.14159265358979);
843 const double depth = (p - opticalCenter).dot(axisDir);
844 const double perpDist = (p - centroid - (p - centroid).dot(axisDir) * axisDir).norm() * 1000.0;
845 qInfo().nospace()
846 << " sample " << (i + 1) << ": depth=" << depth*1000.0
847 << " mm, convergence angle=" << angleDeg << "\u00b0"
848 << ", perp err=" << perpDist << " mm";
849 }
850
851 if (depthSpreadMm < 50.0) {
852 qWarning() << "Optical calibration: depth spread is only" << depthSpreadMm
853 << "mm. For a reliable axis direction, move the OPMI to vary the"
854 << "focal distance by at least 50 mm between samples.";
855 }
856
857 if (rms > 10.0) {
858 qWarning() << "Optical calibration: RMS residual" << rms
859 << "mm is large. Check that the stylus accurately marks the microscope focus point.";
860 }
861
862 // Plausibility check: the perpendicular distance from the tracker sensor
863 // to the optical axis should be approximately 270 mm for the Kinevo.
864 // The tracker is mounted on the microscope body, offset from the objective
865 // by a roughly fixed distance. Flag anything clearly outside [100, 500] mm.
866 constexpr float kExpectedOffsetMm = 270.0f;
867 constexpr float kMinPlausibleMm = 100.0f;
868 constexpr float kMaxPlausibleMm = 500.0f;
869 if (distMm < kMinPlausibleMm || distMm > kMaxPlausibleMm) {
870 qWarning().nospace()
871 << "Optical calibration: tracker\u2194axis distance " << distMm
872 << " mm is outside the plausible range [" << kMinPlausibleMm
873 << ", " << kMaxPlausibleMm << "] mm (expected ~" << kExpectedOffsetMm
874 << " mm for Kinevo). Calibration data may be unreliable.";
875 }
876
877 emit opticalCalibrationChanged();
878 return true;
879}

881bool PolhemusCoregistration::opticalRayInWorld(QVector3D& origin, QVector3D& direction) const
882{
883 if (!m_opticalCalibValid) return false;
884
885 const QMatrix4x4& dev = m_deviceToWorld;
886 if (dev.isIdentity()) return false;
887
888 // Recover raw tracker pose (before device offset)
889 QMatrix4x4 offsetMat;
890 offsetMat.setToIdentity();
891 offsetMat.translate(m_offsetTranslation);
892 offsetMat.rotate(m_offsetRotation);
893

896 const QQuaternion trackerOri = QQuaternion::fromRotationMatrix(trackerToWorld.toGenericMatrix<3, 3>());
897
898 origin = trackerPos + trackerOri.rotatedVector(m_opticalCenterLocal);
899 direction = trackerOri.rotatedVector(m_opticalAxisLocal).normalized();
