53void VectorPlot::add_data(const std::vector<double>& values)
54{
55 if (values.empty()) {
56 return;
57 }
58 const auto [lo, hi] = std::minmax_element(values.begin(), values.end());
59 m_maxX = std::max(m_maxX, static_cast<double>(values.size() - 1));
60 m_minY = std::min(m_minY, *lo);
61 m_maxY = std::max(m_maxY, *hi);
62 m_data.push_back(values);
63}

111void VectorPlot::draw(Canvas& canvas, const TextMetrics& metrics) const
112{
113 if (m_data.empty()) {
114 return;
115 }
116 const int widgetW = static_cast<int>(canvas.width());
117 const int widgetH = static_cast<int>(canvas.height());
118 const int border = static_cast<int>(kBorderLeftRight);
119 const int borderTB = static_cast<int>(kBorderTopBottom);
120 const double left = border;
121 const double top = borderTB;
122 const double plotW = widgetW - 2 * border;
123 const double plotH = widgetH - 2 * borderTB;
124
125 canvas.save();
126 canvas.fill_rect({left, top, plotW, plotH}, Brush::solid(Color::white()));
127 canvas.set_pen(Color::black());
128 canvas.draw_rect({left - 1, top - 1, plotW + 1, plotH + 1});
129
130 // Data: x from 0 to the last index over the width, y upwards over 90 % of the height
131 canvas.set_pen(Pen{{0, 0, 255}, 1});
132 const double minX = 0.0;
133 const double rangeX = m_maxX > minX ? m_maxX - minX : 1.0;
134 const double rangeY = m_maxY > m_minY ? m_maxY - m_minY : 1.0;
135 const double scaleX = plotW / rangeX;
136 const double scaleY = 0.9 * plotH / rangeY;
137 const double bottom = top + 0.95 * plotH;
138 for (const std::vector<double>& values : m_data) {
139 for (std::size_t k = 1; k < values.size(); ++k) {
140 canvas.draw_line({left + (k - 1 - minX) * scaleX, bottom - (values[k - 1] - m_minY) * scaleY},
141 {left + (k - minX) * scaleX, bottom - (values[k] - m_minY) * scaleY});
142 }
143 }
144
145 // Ticks
146 const Font axes = axes_font();
147 canvas.set_pen(Color::black());
148 canvas.set_font(axes);
149 const double line = metrics.height(axes);
150 const double frameBottom = top + plotH;
151 for (double v : nice_ticks(minX, minX + rangeX, 8)) {
152 const double px = left + (v - minX) * scaleX;
153 canvas.draw_line({px, frameBottom}, {px, frameBottom + 4});
154 canvas.draw_text(Rect{px - 50, frameBottom + 5, 100, line}, Align::HCenter | Align::Top, tick_label(v));
155 }
156 const double padY = 0.05 * plotH / scaleY;
157 for (double v : nice_ticks(m_minY - padY, m_minY + rangeY + padY, 6)) {
158 const double py = bottom - (v - m_minY) * scaleY;
159 canvas.draw_line({left - 5, py}, {left - 1, py});
160 canvas.draw_text(Rect{0, py - line, left - 8, 2 * line}, Align::Right | Align::VCenter, tick_label(v));
161 }
162
163 // Title and axis labels in boxes as wide as the frame and 100 pixels high
164 const int labelWidth = widgetW - 2 * border;
165 const int labelHeight = 100;
166 if (!m_title.empty()) {
167 canvas.save();
168 canvas.set_font(title_font());
169 canvas.translate((widgetW - labelWidth) / 2, (widgetH - static_cast<int>(plotH)) / 2 - borderTB * 1.5);
170 canvas.draw_text(Rect{0, 0, static_cast<double>(labelWidth), labelHeight}, Align::Center, m_title);
171 canvas.restore();
172 }
173 if (!m_xLabel.empty()) {
174 canvas.save();
175 // Centred 12 px above the bottom edge, below the tick labels
176 canvas.translate((widgetW - labelWidth) / 2, widgetH - 12 - labelHeight / 2);
177 canvas.draw_text(Rect{0, 0, static_cast<double>(labelWidth), labelHeight}, Align::Center, m_xLabel);
178 canvas.restore();
179 }
180 if (!m_yLabel.empty()) {
181 canvas.save();
182 canvas.rotate(270);
183 // Centred 16 px from the left edge, left of the tick labels
184 canvas.translate(-(widgetH + labelWidth) / 2, 16 - labelHeight / 2);
185 canvas.draw_text(Rect{0, 0, static_cast<double>(labelWidth), labelHeight}, Align::Center, m_yLabel);
186 canvas.restore();
187 }
188 canvas.restore();
189}
