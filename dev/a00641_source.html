64mne::Expected<std::filesystem::path> FiffFileSharer::copy_realtime_file(const std::filesystem::path& source)
65{
66 std::error_code ec;
67 std::filesystem::create_directories(m_directory, ec);
68 std::filesystem::path target = fileAt(m_next);
69 // Complete the copy under another name and rename it, so a consumer never sees a half-written file
70 std::filesystem::path part = target;
71 part += ".part";
72 if (ec || !std::filesystem::copy_file(source, part, std::filesystem::copy_options::overwrite_existing, ec)) {
73 return fail(ErrorCode::Io, "cannot copy {} to {}: {}", mne::utf8_from_path(source), mne::utf8_from_path(part), ec.message());
74 }
75 // The copy is cut mid-recording: close its raw data blocks so it reads as a complete file
76 if (auto bytes = mne::FileByteStream::open(part, mne::FileByteStream::Mode::ReadWrite)) {
77 bytes->seek(bytes->size());
78 FiffStream(std::move(bytes)).finish_writing_raw();
79 }
80 std::filesystem::rename(part, target, ec);
81 if (ec) {
82 std::filesystem::remove(part, ec);
83 return fail(ErrorCode::Io, "cannot share {}", mne::utf8_from_path(target));
84 }
85 ++m_next;
86 return target;
87}

103bool FiffFileSharer::clear()
104{
105 std::error_code ec;
106 std::filesystem::create_directories(m_directory, ec);
107 for (const std::filesystem::directory_entry& entry : std::filesystem::directory_iterator(m_directory, ec)) {
108 if (entry.is_regular_file(ec)) {
109 std::filesystem::remove(entry.path(), ec);
110 }
111 }
112 m_next = 0;
113 return std::filesystem::is_directory(m_directory, ec) && std::filesystem::is_empty(m_directory, ec);
