81bool RtInvOp::update(const FiffCov& noiseCov)
82{
83 const InvEstimateMethod method = m_settings.method;
84 if (method != InvEstimateMethod::MNE && method != InvEstimateMethod::dSPM && method != InvEstimateMethod::sLORETA) {
85 Log::warning("[RtInvOp::update] Only MNE, dSPM and sLORETA are supported.");
86 return false;
87 }
88
89 // The channels MNEForwardSolution::prepare_forward selects
90 StringList names;
91 for (const FiffChInfo& ch : m_info.chs) {
92 if (!has_channel(m_info.bads, ch.ch_name) && !has_channel(noiseCov.bads, ch.ch_name) && has_channel(noiseCov.names, ch.ch_name) && has_channel(m_forward.info.ch_names, ch.ch_name)) {
93 names.push_back(ch.ch_name);
94 }
95 }
96 if (names.empty()) {
97 Log::warning("[RtInvOp::update] No channel is in the measurement info, the covariance and the forward solution.");
98 return false;
99 }
100 if (names != m_names) {
101 prepare_channels(names);
102 }
103
104 FiffCov cov = noiseCov.prepare_noise_cov(m_info, m_names);
105 VectorXd scale = VectorXd::Zero(cov.eig.size());
106 int rank = 0;
107 for (Index i = 0; i < scale.size(); ++i) {
108 if (cov.eig[i] > 0) {
109 scale[i] = 1.0 / std::sqrt(cov.eig[i]);
110 ++rank;
111 }
112 }
113 if (rank == 0) {
114 Log::warning("[RtInvOp::update] The noise covariance has no positive eigenvalue.");
115 return false;
116 }
117 // Rows of eigvec are the eigenvectors
118 const MatrixXd whitener = scale.asDiagonal() * cov.eigvec;
119 MatrixXd proj;
120 FiffProj::make_projector(m_info.projs, cov.names, proj);
121 if (proj.rows() != whitener.cols()) {
122 proj = MatrixXd::Identity(whitener.cols(), whitener.cols());
123 }
124
125 // Source prior scaled so that the trace of the whitened G·R·Gᵀ is the rank
126 const MatrixXd whitenedGainCov = whitener * m_gainCov * whitener.transpose();
127 const double s = rank / whitenedGainCov.trace();
128 MatrixXd c = s * whitenedGainCov;
129 c.diagonal().array() += m_settings.lambda2;
130 const SelfAdjointEigenSolver<MatrixXd> eig(c);
131 const MatrixXd& e = eig.eigenvectors();
132 const VectorXd& d = eig.eigenvalues();
133
134 const MatrixXd ew = e.transpose() * whitener;
135 m_kernel = s * ew.transpose() * d.cwiseInverse().asDiagonal() * ew * proj;
136
137 m_noiseNorm.resize(0);
138 if (method != InvEstimateMethod::MNE) {
139 const VectorXd f = method == InvEstimateMethod::dSPM ? VectorXd(d.cwiseInverse()) : VectorXd(d.cwiseSqrt().cwiseInverse() / std::sqrt(m_settings.lambda2));
140 VectorXd norms = ((f.asDiagonal() * ew) * m_gain).colwise().norm().transpose();
141 norms = s * norms.cwiseProduct(m_sourcePrior);
142 if (!m_fixed) {
143 norms = Linalg::combine_xyz(norms.transpose()).cwiseSqrt();
144 }
145 m_noiseNorm = norms.cwiseInverse();
146 }
147 m_scale = s;
148 m_noiseCov = std::move(cov);
149 return true;
150}

161MatrixXd RtInvOp::apply(const MatrixXd& data, int nave) const
162{
163 if (!ready() || data.rows() != std::ssize(m_info.chs)) {
164 return {};
165 }
166 MatrixXd sol = m_outputLeads.transpose() * (m_kernel * data(m_picks, placeholders::all));
167 if (!m_fixed && !m_settings.pick_normal) {
168 // Magnitude of the three components of each source
169 MatrixXd magnitude = sol.reshaped(3, sol.size() / 3).colwise().norm().reshaped(sol.rows() / 3, sol.cols());
170 sol = std::move(magnitude);
171 }
172 if (m_noiseNorm.size() > 0) {
173 // Averaging divides the noise variance by nave
174 sol = (std::sqrt(static_cast<double>(nave)) * m_noiseNorm).asDiagonal() * sol;
175 }
176 return sol;
177}

181void RtInvOp::prepare_channels(const StringList& names)
182{
183 m_names = names;
184 const auto n = std::ssize(names);
185 m_picks.resize(static_cast<std::size_t>(n));
186 VectorXi infoIdx(n);
187 std::vector<int> fwdIdx(static_cast<std::size_t>(n));
188 for (Index i = 0; i < n; ++i) {
189 const std::string& name = names[static_cast<std::size_t>(i)];
190 m_picks[static_cast<std::size_t>(i)] = infoIdx[i] = channel_index(m_info.ch_names, name);
191 fwdIdx[static_cast<std::size_t>(i)] = channel_index(m_forward.info.ch_names, name);
192 }
193 m_gain = m_forward.sol.data(fwdIdx, placeholders::all);
194
195 if (m_settings.depth > 0) {
196 m_sourcePrior = MNEForwardSolution::compute_depth_prior(m_gain, m_info.pick_info(infoIdx.transpose()), m_fixed, m_settings.depth, 10.0, MatrixXd(), true).data.col(0);
197 } else {
198 m_sourcePrior = VectorXd::Ones(m_gain.cols());
199 }
200 if (!m_fixed) {
201 m_sourcePrior = m_sourcePrior.cwiseProduct(m_orient);
202 }
203
204 const MatrixXd weighted = m_gain * m_sourcePrior.cwiseSqrt().asDiagonal();
205 m_gainCov = MatrixXd::Zero(n, n);
206 m_gainCov.selfadjointView<Lower>().rankUpdate(weighted);
207 m_gainCov = m_gainCov.selfadjointView<Lower>();
208
209 // Only the normal component of each free-orientation source if that is all that is estimated
210 const Index step = !m_fixed && m_settings.pick_normal ? 3 : 1;
211 const Index components = m_gain.cols() / step;
212 m_outputLeads.resize(n, components);
213 for (Index k = 0; k < components; ++k) {
214 const Index component = k * step + step - 1;
215 m_outputLeads.col(k) = m_gain.col(component) * m_sourcePrior[component];
216 }
217}
