117void ChannelDataModel::setData(const MatrixXd &data, int firstSample)
118{
119 if (data.rows() == 0 || data.cols() == 0)
120 return;
121
122 QWriteLocker lk(&m_lock);
123 int nCh = static_cast<int>(data.rows());
124 m_channelData.resize(nCh);
125 for (int ch = 0; ch < nCh; ++ch) {
126 m_channelData[ch].resize(static_cast<int>(data.cols()));
127 for (int s = 0; s < static_cast<int>(data.cols()); ++s)
128 m_channelData[ch][s] = static_cast<float>(data(ch, s));
129 }
130 m_firstSample = firstSample;
131 lk.unlock();
132
133 emit dataChanged();
134}

138void ChannelDataModel::appendData(const MatrixXd &data)
139{
140 if (data.rows() == 0 || data.cols() == 0)
141 return;
142
143 QWriteLocker lk(&m_lock);
144 int nCh = static_cast<int>(data.rows());
145 int nNew = static_cast<int>(data.cols());
146
147 if (m_channelData.size() != nCh)
148 m_channelData.resize(nCh);
149
150 for (int ch = 0; ch < nCh; ++ch) {
151 auto &buf = m_channelData[ch];
152 int oldSize = buf.size();
153 buf.resize(oldSize + nNew);
154 for (int s = 0; s < nNew; ++s)
155 buf[oldSize + s] = static_cast<float>(data(ch, s));
156
157 // Drop oldest samples when capacity exceeded
158 if (m_maxStoredSamples > 0 && buf.size() > m_maxStoredSamples) {
159 int drop = buf.size() - m_maxStoredSamples;
160 buf.remove(0, drop);
161 if (ch == 0)
162 m_firstSample += drop;
163 }
164 }
165 lk.unlock();
166
167 emit dataChanged();
168}

329float ChannelDataModel::channelRms(int channelIdx, int first, int last) const
330{
331 QReadLocker lk(&m_lock);
332 if (channelIdx < 0 || channelIdx >= m_channelData.size())
333 return 0.f;
334 const QVector<float> &src = m_channelData[channelIdx];
335 int bufFirst = qBound(0, first - m_firstSample, src.size());
336 int bufLast = qBound(0, last - m_firstSample, src.size());
337 if (bufLast <= bufFirst)
338 return 0.f;
339 // Cap at 1000 samples: use the last kMax samples of the window for speed
340 constexpr int kMax = 1000;
341 if (bufLast - bufFirst > kMax)
342 bufFirst = bufLast - kMax;
343 double sum = 0.0;
344 for (int i = bufFirst; i < bufLast; ++i)
345 sum += static_cast<double>(src[i]) * src[i];
346 return static_cast<float>(qSqrt(sum / (bufLast - bufFirst)));
347}

365QVector<float> ChannelDataModel::decimatedVertices(int channelIdx,
366 int firstSample,
367 int lastSample,
368 int pixelWidth,
369 int &vboFirstSample) const
370{
371 QReadLocker lk(&m_lock);
372
373 vboFirstSample = firstSample; // may be updated below after clamping
374
375 if (channelIdx < 0 || channelIdx >= m_channelData.size()
376 || pixelWidth <= 0 || firstSample >= lastSample) {
377 return {};
378 }
379
380 const QVector<float> &src = m_channelData[channelIdx];
381 // Map absolute sample indices to buffer indices
382 int bufFirst = firstSample - m_firstSample;
383 int bufLast = lastSample - m_firstSample;
384
385 // Clamp to available data
386 bufFirst = qBound(0, bufFirst, src.size());
387 bufLast = qBound(0, bufLast, src.size());
388
389 // Update to the actual absolute sample index at vertex 0 after clamping.
390 // When firstSample < m_firstSample (tile extends before buffer start),
391 // bufFirst is clamped to 0 so vboFirstSample must reflect m_firstSample,
392 // not the original (possibly negative) firstSample.
393 vboFirstSample = m_firstSample + bufFirst;
394
395 if (bufLast <= bufFirst)
396 return {};
397
398 int nSamples = bufLast - bufFirst;
399
400 // ── DC removal: compute mean over the window ────────────────────────
401 float dcOffset = 0.f;
402 if (m_removeDC) {
403 double sum = 0.0;
404 for (int i = bufFirst; i < bufLast; ++i)
405 sum += src[i];
406 dcOffset = static_cast<float>(sum / nSamples);
407 }
408
409 if (nSamples <= pixelWidth * 2) {
410 // ── Raw path: one vertex per sample ────────────────────────────
411 QVector<float> result;
412 result.reserve(nSamples * 2);
413 for (int i = 0; i < nSamples; ++i) {
414 result.append(static_cast<float>(i));
415 result.append(src[bufFirst + i] - dcOffset);
416 }
417 return result;
418 }
419
420 // ── Decimation path: min/max envelope, 2 vertices per pixel ─────────
421 // Interleaved as (x_at_col, max), (x_at_col, min) per column.
422 // Connected as LINE_STRIP this draws: vertical spike at each column,
423 // diagonals between columns — the classic oscilloscope envelope look.
424
425 QVector<float> result;
426 result.reserve(pixelWidth * 4); // 2 vertices × 2 floats
427
428 float spp = static_cast<float>(nSamples) / pixelWidth; // samples per pixel
429
430 for (int px = 0; px < pixelWidth; ++px) {
431 int sBegin = bufFirst + static_cast<int>(px * spp);
432 int sEnd = bufFirst + static_cast<int>((px + 1) * spp);
433 sEnd = qMin(sEnd, bufLast);
434 if (sBegin >= sEnd) sBegin = qMax(sEnd - 1, bufFirst);
435
436 float minV = src[sBegin];
437 float maxV = src[sBegin];
438 for (int s = sBegin + 1; s < sEnd; ++s) {
439 if (src[s] < minV) minV = src[s];
440 if (src[s] > maxV) maxV = src[s];
441 }
442 minV -= dcOffset;
443 maxV -= dcOffset;
444
445 float xOffset = px * spp;
446
447 // Emit max first, then min: draws ascending spike first
448 result.append(xOffset); result.append(maxV);
449 result.append(xOffset); result.append(minV);
450 }
451
452 return result;
453}

459void ChannelDataModel::rebuildDisplayInfo()
460{
461 QWriteLocker lk(&m_lock);
462 const int realChannelCount = m_pFiffInfo ? m_pFiffInfo->nchan : 0;
463 const int nCh = qMax(m_channelData.size(), realChannelCount + m_virtualDisplayInfo.size());
464 m_displayInfo.resize(nCh);
465 for (int ch = 0; ch < nCh; ++ch) {
466 if (ch >= realChannelCount && ch - realChannelCount < m_virtualDisplayInfo.size()) {
467 ChannelDisplayInfo info = m_virtualDisplayInfo.at(ch - realChannelCount);
468 if (info.name.isEmpty())
469 info.name = QString("Virtual %1").arg(ch - realChannelCount + 1);
470 if (info.typeLabel.isEmpty())
471 info.typeLabel = QStringLiteral("MISC");
472 if (!info.color.isValid())
473 info.color = m_signalColor;
474 if (info.amplitudeMax <= 0.f)
475 info.amplitudeMax = kScaleFallback;
476 m_displayInfo[ch] = info;
477 m_displayInfo[ch].isVirtualChannel = true;
478 continue;
479 }
480
481 m_displayInfo[ch].amplitudeMax = amplitudeMaxForChannel(ch);
482 m_displayInfo[ch].color = colorForChannel(ch);
483 if (m_pFiffInfo && ch < realChannelCount)
484 m_displayInfo[ch].name = m_pFiffInfo->ch_names[ch];
485 else
486 m_displayInfo[ch].name = QString("CH %1").arg(ch + 1);
487 m_displayInfo[ch].typeLabel = typeLabelForChannel(ch);
488 m_displayInfo[ch].bad = (m_pFiffInfo && ch < realChannelCount)
489 ? m_pFiffInfo->bads.contains(m_displayInfo[ch].name)
490 : false;
491 m_displayInfo[ch].isVirtualChannel = false;
492 }
493}

497float ChannelDataModel::amplitudeMaxForChannel(int ch) const
498{
499 if (!m_pFiffInfo || ch >= m_pFiffInfo->nchan)
500 return kScaleFallback;
501
502 const auto &info = m_pFiffInfo->chs[ch];
503 qint32 kind = info.kind;
504
505 // MEG: distinguish gradiometer vs. magnetometer by unit
506 if (kind == FIFFV_MEG_CH) {
507 if (info.unit == FIFF_UNIT_T_M)
508 return m_scaleMap.value(kMEGGradKind,
509 m_scaleMap.value(FIFFV_MEG_CH, kScaleMEGGrad));
510 else
511 return m_scaleMap.value(kMEGMagKind,
512 m_scaleMap.value(FIFFV_MEG_CH, kScaleMEGMag));
513 }
514 if (m_scaleMap.contains(kind))
515 return m_scaleMap.value(kind);
516 return kScaleFallback;
517}

521QColor ChannelDataModel::colorForChannel(int ch) const
522{
523 if (!m_pFiffInfo || ch >= m_pFiffInfo->nchan)
524 return m_signalColor;
525
526 // Dark colours chosen to be readable on a light (near-white) background
527 switch (m_pFiffInfo->chs[ch].kind) {
528 case FIFFV_MEG_CH: return QColor(20, 90, 180); // dark blue for MEG
529 case FIFFV_EEG_CH: return QColor(170, 55, 10); // dark orange for EEG
530 case FIFFV_EOG_CH: return QColor(130, 0, 130); // dark purple for EOG
531 case FIFFV_ECG_CH: return QColor(190, 15, 45); // dark crimson for ECG
532 case FIFFV_EMG_CH: return QColor(20, 110, 20); // dark green for EMG
533 case FIFFV_STIM_CH: return QColor(180, 100, 0); // dark amber for STIM
534 default: return m_signalColor;
535 }
