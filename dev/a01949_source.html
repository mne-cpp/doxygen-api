64QList<QSet<int>> FsLabelUtils::buildAdjacency(const MatrixX3i& tris, int nVerts)
65{
66 QList<QSet<int>> adj(nVerts);
67
68 for (int t = 0; t < tris.rows(); ++t) {
69 int v0 = tris(t, 0);
70 int v1 = tris(t, 1);
71 int v2 = tris(t, 2);
72
73 if (v0 >= 0 && v0 < nVerts && v1 >= 0 && v1 < nVerts && v2 >= 0 && v2 < nVerts) {
74 adj[v0].insert(v1);
75 adj[v0].insert(v2);
76 adj[v1].insert(v0);
77 adj[v1].insert(v2);
78 adj[v2].insert(v0);
79 adj[v2].insert(v1);
80 }
81 }
82
83 return adj;
84}

88FsLabel FsLabelUtils::growLabel(const FsLabel& label,
89 const FsSurface& surface,
90 int nSteps)
91{
92 if (label.isEmpty() || surface.isEmpty() || nSteps <= 0)
93 return label;
94
95 const int nVerts = static_cast<int>(surface.rr().rows());
96 QList<QSet<int>> adj = buildAdjacency(surface.tris(), nVerts);
97
98 // Start with seed vertices
99 QSet<int> current;
100 for (int i = 0; i < label.vertices.size(); ++i)
101 current.insert(label.vertices[i]);
102
103 QSet<int> allVerts = current;
104
105 // BFS expansion
106 for (int step = 0; step < nSteps; ++step) {
107 QSet<int> frontier;
108 for (int v : current) {
109 if (v >= 0 && v < nVerts) {
110 for (int neighbor : adj[v]) {
111 if (!allVerts.contains(neighbor))
112 frontier.insert(neighbor);
113 }
114 }
115 }
116 allVerts.unite(frontier);
117 current = frontier;
118
119 if (frontier.isEmpty())
120 break;
121 }
122
123 // Build result label
124 QList<int> sortedVerts(allVerts.begin(), allVerts.end());
125 std::sort(sortedVerts.begin(), sortedVerts.end());
126
127 FsLabel result;
128 result.hemi = label.hemi;
129 result.name = label.name + "_grown";
130 result.vertices.resize(sortedVerts.size());
131 result.pos.resize(sortedVerts.size(), 3);
132 result.values = VectorXd::Ones(sortedVerts.size());
133
134 for (int i = 0; i < sortedVerts.size(); ++i) {
135 result.vertices[i] = sortedVerts[i];
136 if (sortedVerts[i] < nVerts)
137 result.pos.row(i) = surface.rr().row(sortedVerts[i]);
138 }
139
140 return result;
141}

145QList<FsLabel> FsLabelUtils::splitLabel(const FsLabel& label,
146 const FsSurface& surface)
147{
148 QList<FsLabel> components;
149
150 if (label.isEmpty())
151 return components;
152
153 // If surface is empty, treat each vertex as its own component
154 if (surface.isEmpty()) {
155 for (int i = 0; i < label.vertices.size(); ++i) {
156 FsLabel comp;
157 comp.hemi = label.hemi;
158 comp.name = QString("%1_part%2").arg(label.name).arg(i);
159 comp.vertices.resize(1);
160 comp.vertices[0] = label.vertices[i];
161 comp.pos = label.pos.row(i);
162 comp.values = VectorXd::Ones(1);
163 if (i < label.values.size())
164 comp.values[0] = label.values[i];
165 components.append(comp);
166 }
167 return components;
168 }
169
170 const int nVerts = static_cast<int>(surface.rr().rows());
171 QList<QSet<int>> adj = buildAdjacency(surface.tris(), nVerts);
172
173 // Build set of label vertices for fast lookup
174 QSet<int> labelSet;
175 for (int i = 0; i < label.vertices.size(); ++i)
176 labelSet.insert(label.vertices[i]);
177
178 QSet<int> visited;
179 int compIdx = 0;
180
181 for (int i = 0; i < label.vertices.size(); ++i) {
182 int seed = label.vertices[i];
183 if (visited.contains(seed))
184 continue;
185
186 // BFS from seed, restricted to label vertices
187 QQueue<int> queue;
188 queue.enqueue(seed);
189 visited.insert(seed);
190 QList<int> component;
191
192 while (!queue.isEmpty()) {
193 int v = queue.dequeue();
194 component.append(v);
195
196 if (v >= 0 && v < nVerts) {
197 for (int neighbor : adj[v]) {
198 if (labelSet.contains(neighbor) && !visited.contains(neighbor)) {
199 visited.insert(neighbor);
200 queue.enqueue(neighbor);
201 }
202 }
203 }
204 }
205
206 std::sort(component.begin(), component.end());
207
208 FsLabel comp;
209 comp.hemi = label.hemi;
210 comp.name = QString("%1_part%2").arg(label.name).arg(compIdx++);
211 comp.vertices.resize(component.size());
212 comp.pos.resize(component.size(), 3);
213 comp.values = VectorXd::Ones(component.size());
214
215 for (int j = 0; j < component.size(); ++j) {
216 comp.vertices[j] = component[j];
217 if (component[j] < nVerts)
218 comp.pos.row(j) = surface.rr().row(component[j]);
219 }
220
221 components.append(comp);
222 }
223
224 return components;
225}

229QList<FsLabel> FsLabelUtils::stcToLabel(const MatrixXd& stcData,
230 const VectorXi& vertices,
231 const FsSurface& surface,
232 double dThreshold,
233 int iHemi)
234{
235 QList<FsLabel> labels;
236
237 if (stcData.size() == 0 || vertices.size() == 0 || surface.isEmpty())
238 return labels;
239
240 // Find vertices above threshold (max absolute value across time)
241 VectorXd maxAbs = stcData.cwiseAbs().rowwise().maxCoeff();
242
243 QSet<int> aboveThresh;
244 for (int i = 0; i < vertices.size(); ++i) {
245 if (maxAbs[i] > dThreshold)
246 aboveThresh.insert(vertices[i]);
247 }
248
249 if (aboveThresh.isEmpty())
250 return labels;
251
252 // Build a label from above-threshold vertices and split into components
253 FsLabel fullLabel;
254 fullLabel.hemi = iHemi;
255 fullLabel.name = "stc_label";
256
257 QList<int> sortedVerts(aboveThresh.begin(), aboveThresh.end());
258 std::sort(sortedVerts.begin(), sortedVerts.end());
259
260 const int nSurfVerts = static_cast<int>(surface.rr().rows());
261 fullLabel.vertices.resize(sortedVerts.size());
262 fullLabel.pos.resize(sortedVerts.size(), 3);
263 fullLabel.values.resize(sortedVerts.size());
264
265 for (int i = 0; i < sortedVerts.size(); ++i) {
266 fullLabel.vertices[i] = sortedVerts[i];
267 if (sortedVerts[i] < nSurfVerts)
268 fullLabel.pos.row(i) = surface.rr().row(sortedVerts[i]);
269
270 // Find the vertex's row in the STC
271 for (int j = 0; j < vertices.size(); ++j) {
272 if (vertices[j] == sortedVerts[i]) {
273 fullLabel.values[i] = maxAbs[j];
274 break;
275 }
276 }
277 }
278
279 // Split into connected components
280 labels = splitLabel(fullLabel, surface);
281
282 return labels;
283}

287MatrixXd FsLabelUtils::labelsToStc(const QList<FsLabel>& labels,
288 const VectorXi& stcVertices,
289 int nTimes)
290{
291 const int nVerts = static_cast<int>(stcVertices.size());
292 MatrixXd mask = MatrixXd::Zero(nVerts, nTimes);
293
294 // Build vertex-to-row map
295 QMap<int, int> vertToRow;
296 for (int i = 0; i < nVerts; ++i)
297 vertToRow.insert(stcVertices[i], i);
298
299 for (const auto& label : labels) {
300 for (int i = 0; i < label.vertices.size(); ++i) {
301 auto it = vertToRow.find(label.vertices[i]);
302 if (it != vertToRow.end()) {
303 mask.row(it.value()).setOnes();
304 }
305 }
306 }
307
308 return mask;
309}
