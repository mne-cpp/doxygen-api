139scene::AverageLayout AverageLayoutWidget::averageLayout() const
140{
141 std::vector<scene::AverageTile> tiles;
142 tiles.reserve(m_channels.size());
143 for (std::size_t i = 0; i < m_channels.size(); ++i) {
144 const Channel& channel = m_channels[i];
145 scene::AverageTile tile;
146 tile.box = scene::AverageLayout::box_at({channel.position.x(), channel.position.y()});
147 tile.name = channel.name;
148 tile.name_size = static_cast<int>(i) == m_pressedTile ? kPressedNameSize : kNameSize;
149 tile.bad = channel.bad;
150 tile.max_value = getScalingValue(m_scaleMap, channel.kind, channel.unit);
151 tile.stim_sample = channel.firstSample;
152 tile.columns = channel.averages.empty() ? 0 : channel.averages.front().samples;
153 for (const Average& average : channel.averages) {
154 // Averages missing from the activation map are drawn
155 if (const auto active = m_activation.find(average.comment); active != m_activation.end() && !active->second) {
156 continue;
157 }
158 scene::AverageTile::Trace trace;
159 const auto color = m_colors.find(average.comment);
160 trace.color = toSceneColor(color != m_colors.end() ? color->second : m_signalColor);
161 if (average.data) {
162 trace.samples.reserve(static_cast<std::size_t>(average.samples));
163 for (int s = 0; s < average.samples; ++s) {
164 trace.samples.push_back(average.data[static_cast<std::ptrdiff_t>(s) * average.stride]);
165 }
166 }
167 tile.traces.push_back(std::move(trace));
168 }
169 tiles.push_back(std::move(tile));
170 }
171 scene::AverageLayout layout;
172 layout.set_tiles(std::move(tiles));
173 return layout;
174}
