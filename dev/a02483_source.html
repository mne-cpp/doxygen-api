42GpuPipeline surface_pipeline(const char* shader)
43{
44 GpuPipeline pipeline;
45 pipeline.shader = shader;
46 pipeline.bindings = {{6 * sizeof(float), false}, {4, false}, {4, false}};
47 pipeline.attributes = {{0, 0, GpuVertexFormat::Float3, 0},
48 {1, 0, GpuVertexFormat::Float3, 3 * sizeof(float)},
49 {2, 1, GpuVertexFormat::UNormByte4, 0},
50 {3, 2, GpuVertexFormat::UNormByte4, 0}};
51 pipeline.depth_test = true;
52 pipeline.depth_write = true;
53 return pipeline;
54}

56GpuPipeline instanced_pipeline(const char* shader)
57{
58 GpuPipeline pipeline;
59 pipeline.shader = shader;
60 pipeline.bindings = {{6 * sizeof(float), false}, {21 * sizeof(float), true}};
61 pipeline.attributes = {{0, 0, GpuVertexFormat::Float3, 0}, {1, 0, GpuVertexFormat::Float3, 3 * sizeof(float)}};
62 for (int column = 0; column < 5; ++column) {
63 pipeline.attributes.push_back({2 + column, 1, GpuVertexFormat::Float4, static_cast<std::uint32_t>(4 * column * sizeof(float))});
64 }
65 pipeline.attributes.push_back({7, 1, GpuVertexFormat::Float, 20 * sizeof(float)});
66 pipeline.depth_test = true;
67 pipeline.depth_write = true;
68 return pipeline;
69}

77BrainFrameBuilder::BrainFrameBuilder()
78{
79 const auto index = [](BrainShader shader) {
80 return static_cast<std::size_t>(shader);
81 };
82 m_pipelines[index(BrainShader::Standard)] = m_resources.create_pipeline(surface_pipeline("standard"));
83 m_pipelines[index(BrainShader::Anatomical)] = m_resources.create_pipeline(surface_pipeline("anatomical"));
84 m_pipelines[index(BrainShader::ShowNormals)] = m_resources.create_pipeline(surface_pipeline("shownormals"));
85
86 // Holographic shells add their glow, back faces first; x-ray ones also show through other surfaces
87 GpuPipeline holographic = surface_pipeline("holographic");
88 holographic.blend = GpuBlend::Additive;
89 holographic.depth_write = false;
90 holographic.cull = GpuCull::Back;
91 GpuPipeline xray = holographic;
92 xray.depth_test = false;
93 m_pipelines[index(BrainShader::Holographic)] = m_resources.create_pipeline(holographic);
94 m_pipelines[index(BrainShader::XRay)] = m_resources.create_pipeline(xray);
95 holographic.cull = GpuCull::Front;
96 xray.cull = GpuCull::Front;
97 m_backFacePipelines[0] = m_resources.create_pipeline(std::move(holographic));
98 m_backFacePipelines[1] = m_resources.create_pipeline(std::move(xray));
99
100 GpuPipeline dipole = instanced_pipeline("dipole");
101 dipole.blend = GpuBlend::Alpha;
102 dipole.depth_write = false;
103 m_pipelines[index(BrainShader::Dipole)] = m_resources.create_pipeline(std::move(dipole));
104 m_pipelines[index(BrainShader::Electrode)] = m_resources.create_pipeline(instanced_pipeline("electrode"));
105
106 GpuPipeline overlay;
107 overlay.shader = "screen_overlay";
108 overlay.bindings = {{4 * sizeof(float), false}};
109 overlay.attributes = {{0, 0, GpuVertexFormat::Float2, 0}, {1, 0, GpuVertexFormat::Float2, 2 * sizeof(float)}};
110 overlay.topology = GpuTopology::TriangleStrip;
111 overlay.blend = GpuBlend::Alpha;
112 m_overlayPipeline = m_resources.create_pipeline(std::move(overlay));
113 m_quad = m_resources.create_buffer(GpuBufferUsage::Vertex);
114 m_resources.set_buffer_values(m_quad, std::span<const float>(kQuad));
115}

133GpuHandle BrainFrameBuilder::stream_buffer(Mesh& mesh, int slot, GpuBufferUsage usage, const GpuStreamView& stream)
134{
135 const auto at = static_cast<std::size_t>(slot);
136 if (stream.bytes.empty()) {
137 return 0;
138 }
139 if (mesh.buffers[at] == 0) {
140 mesh.buffers[at] = m_resources.create_buffer(usage);
141 }
142 if (!mesh.uploaded[at] || mesh.generations[at] != stream.generation) {
143 m_resources.set_buffer_data(mesh.buffers[at], stream.bytes);
144 mesh.generations[at] = stream.generation;
145 mesh.uploaded[at] = true;
146 }
147 return mesh.buffers[at];
148}

152bool BrainFrameBuilder::draw_surface(std::uint64_t key, const BrainSurfaceStreams& streams, BrainShader shader, const BrainSurfaceUniforms& uniforms,
153 const GpuRect& viewport)
154{
155 if (shader == BrainShader::Dipole || shader == BrainShader::Electrode || streams.index_count == 0 || m_frame.passes.empty()) {
156 return false;
157 }
158 Mesh& mesh = m_meshes[key];
159 mesh.last_frame = m_frameIndex;
160 const GpuHandle geometry = stream_buffer(mesh, 0, GpuBufferUsage::Vertex, streams.geometry);
161 const GpuHandle colors = stream_buffer(mesh, 1, GpuBufferUsage::Vertex, streams.colors);
162 const GpuHandle annotations = stream_buffer(mesh, 2, GpuBufferUsage::Vertex, streams.annotations);
163 const GpuHandle indices = stream_buffer(mesh, 3, GpuBufferUsage::Index32, streams.indices);
164 if (geometry == 0 || colors == 0 || annotations == 0 || indices == 0) {
165 return false;
166 }
167
168 GpuDraw draw;
169 draw.vertex_buffers = {geometry, colors, annotations};
170 draw.index_buffer = indices;
171 draw.count = streams.index_count;
172 draw.set_uniforms(0, uniforms);
173 draw.viewport = viewport;
174 draw.scissor = viewport;
175 std::vector<GpuDraw>& draws = m_frame.passes.back().draws;
176 if (shader == BrainShader::Holographic || shader == BrainShader::XRay) {
177 GpuDraw back = draw;
178 back.pipeline = m_backFacePipelines[shader == BrainShader::Holographic ? 0 : 1];
179 draws.push_back(std::move(back));
180 }
181 draw.pipeline = m_pipelines[static_cast<std::size_t>(shader)];
182 draws.push_back(std::move(draw));
183 return true;
184}

188bool BrainFrameBuilder::draw_instanced(std::uint64_t key, const BrainInstancedStreams& streams, BrainShader shader, const BrainInstanceUniforms& uniforms,
189 const GpuRect& viewport)
190{
191 if ((shader != BrainShader::Dipole && shader != BrainShader::Electrode) || streams.index_count == 0 || streams.instance_count == 0 || m_frame.passes.empty()) {
192 return false;
193 }
194 Mesh& mesh = m_meshes[key];
195 mesh.last_frame = m_frameIndex;
196 const GpuHandle vertices = stream_buffer(mesh, 0, GpuBufferUsage::Vertex, streams.vertices);
197 const GpuHandle instances = stream_buffer(mesh, 1, GpuBufferUsage::Vertex, streams.instances);
198 const GpuHandle indices = stream_buffer(mesh, 2, GpuBufferUsage::Index32, streams.indices);
199 if (vertices == 0 || instances == 0 || indices == 0) {
200 return false;
201 }
202 GpuDraw draw;
203 draw.pipeline = m_pipelines[static_cast<std::size_t>(shader)];
204 draw.vertex_buffers = {vertices, instances};
205 draw.index_buffer = indices;
206 draw.count = streams.index_count;
207 draw.instance_count = streams.instance_count;
208 draw.set_uniforms(0, uniforms);
209 draw.viewport = viewport;
210 draw.scissor = viewport;
211 m_frame.passes.back().draws.push_back(std::move(draw));
212 return true;
213}

241GpuFrame BrainFrameBuilder::finish()
242{
243 for (auto it = m_meshes.begin(); it != m_meshes.end();) {
244 if (m_frameIndex - it->second.last_frame > kUnusedFrames) {
245 for (const GpuHandle buffer : it->second.buffers) {
246 m_resources.release(buffer);
247 }
248 it = m_meshes.erase(it);
249 } else {
250 ++it;
251 }
252 }
253 return std::exchange(m_frame, GpuFrame{});
254}
