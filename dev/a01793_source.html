58void SourceEstimateManager::cancelLoading()
59{
60 if (m_stcWorker)
61 m_stcWorker->requestCancel();
62 if (m_loadingThread.joinable())
63 m_loadingThread.join();
64 // The worker is deleted here, once its thread has stopped, never by the thread itself:
65 // onStcLoadingFinished still reads it after the thread may have finished. A finished signal it
66 // already queued is dropped by the generation check in load().
67 m_stcWorker.reset();
68 ++m_loadGeneration;
69 m_isLoading = false;
70}

74bool SourceEstimateManager::load(const std::filesystem::path& lhPath, const std::filesystem::path& rhPath,
75 const std::map<std::string, std::shared_ptr<BrainSurface>>& surfaces,
76 const std::string& activeSurfaceType)
77{
78 if (m_isLoading) {
79 mne::Log::warning("SourceEstimateManager: STC loading already in progress");
80 return false;
81 }
82
83 // Find surfaces for the active surface type
84 BrainSurface* lhSurface = nullptr;
85 BrainSurface* rhSurface = nullptr;
86
87 const std::string lhKey = "lh_" + activeSurfaceType;
88 const std::string rhKey = "rh_" + activeSurfaceType;
89
90 if (const auto it = surfaces.find(lhKey); it != surfaces.end())
91 lhSurface = it->second.get();
92 if (const auto it = surfaces.find(rhKey); it != surfaces.end())
93 rhSurface = it->second.get();
94
95 // Fallback: search for any lh_*/rh_* brain surface
96 if (!lhSurface || !rhSurface) {
97 for (const auto& [key, surface] : surfaces) {
98 if (surface && surface->tissueType() == BrainSurface::TissueBrain) {
99 if (!lhSurface && key.starts_with("lh_")) {
100 lhSurface = surface.get();
101 mne::Log::debug("SourceEstimateManager: Using fallback LH surface: {}", key);
102 } else if (!rhSurface && key.starts_with("rh_")) {
103 rhSurface = surface.get();
104 mne::Log::debug("SourceEstimateManager: Using fallback RH surface: {}", key);
105 }
106 }
107 }
108 }
109
110 if (!lhSurface && !rhSurface) {
111 std::string keys;
112 for (const auto& [key, surface] : surfaces)
113 keys += (keys.empty() ? "" : ", ") + key;
114 mne::Log::warning("SourceEstimateManager: No surfaces available for STC loading. Active surface type: {} Available keys: {}", activeSurfaceType, keys);
115 return false;
116 }
117
118 // Clean up any previous loading thread and worker
119 cancelLoading();
120
121 // Create overlay for results
122 m_overlay = std::make_unique<SourceEstimateOverlay>();
123
124 // The worker stays on the GUI thread; process() runs on the loading thread, so its signals arrive queued
125 m_stcWorker = std::make_unique<StcLoadingWorker>(lhPath, rhPath, lhSurface, rhSurface);
126
127 const std::uint64_t generation = m_loadGeneration;
128 connect(m_stcWorker.get(), &StcLoadingWorker::progress, this, &SourceEstimateManager::loadingProgress);
129 connect(m_stcWorker.get(), &StcLoadingWorker::finished, this, [this, generation](bool success) {
130 if (generation == m_loadGeneration)
131 onStcLoadingFinished(success);
132 });
133
134 m_isLoading = true;
135 m_loadingThread = std::jthread([worker = m_stcWorker.get()] { worker->process(); });
136
137 return true;
138}

149void SourceEstimateManager::onStcLoadingFinished(bool success)
150{
151 m_isLoading = false;
152
153 if (!success || !m_stcWorker) {
154 mne::Log::warning("SourceEstimateManager: Async STC loading failed");
155 m_overlay.reset();
156 return;
157 }
158
159 // Transfer data from worker to overlay
160 if (m_stcWorker->hasLh()) {
161 m_overlay->setStcData(m_stcWorker->stcLh(), 0);
162 if (m_stcWorker->interpolationMatLh())
163 m_overlay->setInterpolationMatrix(m_stcWorker->interpolationMatLh(), 0);
164 }
165
166 if (m_stcWorker->hasRh()) {
167 m_overlay->setStcData(m_stcWorker->stcRh(), 1);
168 if (m_stcWorker->interpolationMatRh())
169 m_overlay->setInterpolationMatrix(m_stcWorker->interpolationMatRh(), 1);
170 }
171
172 m_overlay->updateThresholdsFromData();
173 emit thresholdsUpdated(m_overlay->thresholdMin(),
174 m_overlay->thresholdMid(),
175 m_overlay->thresholdMax());
176
177 if (m_overlay->isLoaded()) {
178 emit loaded(m_overlay->numTimePoints());
179 } else {
180 m_overlay.reset();
181 }
182}

186void SourceEstimateManager::setTimePoint(int index,
187 const std::map<std::string, std::shared_ptr<BrainSurface>>& surfaces,
188 const SubView& singleView,
189 const std::vector<SubView>& subViews)
190{
191 if (!m_overlay || !m_overlay->isLoaded())
192 return;
193
194 m_currentTimePoint = std::clamp(index, 0, std::max(0, m_overlay->numTimePoints() - 1));
195
196 // Collect all distinct surface types used across single + multi views
197 std::set<std::string> activeTypes;
198 activeTypes.insert(singleView.surfaceType);
199 for (const SubView& subView : subViews)
200 activeTypes.insert(subView.surfaceType);
201
202 // Apply source estimate to surfaces matching ANY active type
203 for (const auto& [key, surface] : surfaces) {
204 for (const std::string& type : activeTypes) {
205 if (key.ends_with(type)) {
206 m_overlay->applyToSurface(surface.get(), m_currentTimePoint);
207 break;
208 }
209 }
210 }
211
212 emit timePointChanged(m_currentTimePoint, m_overlay->timeAtIndex(m_currentTimePoint));
213}

238int SourceEstimateManager::closestIndex(float timeSec) const
239{
240 if (!m_overlay || !m_overlay->isLoaded())
241 return -1;
242
243 const float t0 = m_overlay->tmin();
244 const float dt = m_overlay->tstep();
245 const int numPts = m_overlay->numTimePoints();
246 if (numPts <= 0 || dt <= 0.0f)
247 return -1;
248
249 const int idx = static_cast<int>(std::lround((timeSec - t0) / dt));
250 return std::clamp(idx, 0, numPts - 1);
251}

274void SourceEstimateManager::startStreaming([[maybe_unused]] const std::map<std::string, std::shared_ptr<BrainSurface>>& surfaces,
275 [[maybe_unused]] const SubView& singleView,
276 [[maybe_unused]] const std::vector<SubView>& subViews)
277{
278 if (m_isStreaming) {
279 mne::Log::debug("SourceEstimateManager: Real-time streaming already active");
280 return;
281 }
282
283 if (!m_overlay || !m_overlay->isLoaded()) {
284 mne::Log::warning("SourceEstimateManager: Cannot start streaming — no source estimate loaded");
285 return;
286 }
287
288 // Create controller on first use
289 if (!m_rtController) {
290 m_rtController = std::make_unique<RtSourceDataController>(this);
291 connect(m_rtController.get(), &RtSourceDataController::newSmoothedDataAvailable,
292 this, &SourceEstimateManager::realtimeColorsAvailable);
293 }
294
295 // Propagate interpolation matrices from the overlay
296 m_rtController->setInterpolationMatrixLeft(m_overlay->interpolationMatLh());
297 m_rtController->setInterpolationMatrixRight(m_overlay->interpolationMatRh());
298
299 // Propagate current visualization parameters
300 m_rtController->setColormapType(m_overlay->colormap());
301 m_rtController->setThresholds(m_overlay->thresholdMin(),
302 m_overlay->thresholdMid(),
303 m_overlay->thresholdMax());
304 m_rtController->setSFreq(1.0 / m_overlay->tstep());
305
306 // Feed all STC time-points into the queue
307 const int nTimePoints = m_overlay->numTimePoints();
308 mne::Log::debug("SourceEstimateManager: Feeding {} time points into real-time queue", nTimePoints);
309 m_rtController->clearData();
310
311 for (int t = 0; t < nTimePoints; ++t) {
312 Eigen::VectorXd col = m_overlay->sourceDataColumn(t);
313 if (col.size() > 0)
314 m_rtController->addData(col);
315 }
316
317 m_rtController->setStreamingState(true);
318 m_isStreaming = true;
319
320 mne::Log::debug("SourceEstimateManager: Real-time streaming started");
321}
