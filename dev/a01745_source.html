97QVariant RtFiffRawViewModel::data(const QModelIndex& index, int role) const
98{
99 if (role != Qt::DisplayRole && role != Qt::BackgroundRole) {
100 return QVariant();
101 }
102
103 if (role == Qt::BackgroundRole) {
104 return QVariant(QBrush(m_colBackground));
105 }
106
107 if (index.isValid()) {
108 qint32 row = detail::valueOr(m_qMapIdxRowSelection, index.row(), 0);
109
110 //******** first column (chname) ********
111 if (index.column() == 0 && role == Qt::DisplayRole)
112 return QString::fromStdString(m_pFiffInfo->ch_names[row]);
113
114 //******** second column (data plot) ********
115 if (index.column() == 1) {
116 QVariant v;
117 RowVectorPair rowVectorPair;
118
119 switch (role) {
120 case Qt::DisplayRole: {
121 // The frozen copy while frozen
122 const viz::SweepBuffer::Matrix& matData = m_sweep.layer(m_display.filtering());
123 rowVectorPair.first = matData.data() + row * matData.cols();
124 rowVectorPair.second = static_cast<int>(matData.cols());
125 v.setValue(rowVectorPair);
126
127 return v;
128 }
129 } // end role switch
130 } // end column check
131
132 //******** third column (bad channel) ********
133 if (index.column() == 2 && role == Qt::DisplayRole) {
134 return QVariant(std::ranges::contains(m_pFiffInfo->bads, m_pFiffInfo->ch_names[row]));
135 } // end column check
136
137 } // end index.valid() check
138
139 return QVariant();
140}

227void RtFiffRawViewModel::addData(std::span<const MatrixXd> data)
228{
229 MatrixXdR& matDataRaw = m_sweep.raw();
230 MatrixXdR& matDataFiltered = m_sweep.filtered();
231 const int iMaxCols = static_cast<int>(matDataRaw.cols());
232 if (iMaxCols == 0) {
233 return;
234 }
235
236 const bool doSphara = m_display.applies_sphara(matDataRaw.rows());
237 const bool doFilter = m_display.filtering();
238 const int iTriggerCh = m_display.trigger_channel();
239
240 //Copy new data into the ring buffer: stream sample s goes to column s mod iMaxCols
241 for (const MatrixXd& matBlock : data) {
242 if (matBlock.rows() != matDataRaw.rows()) {
243 mne::Log::debug("incoming data does not match internal data row size. Returning...");
244 return;
245 }
246
247 const int nCol = static_cast<int>(matBlock.cols());
248 const mne::RtRawDisplay::Corrected corrected = m_display.correct(matBlock);
249
250 m_sweep.write(nCol, [this] { startNewSweep(); }, [&](int iCol, int iOffset, int iLength) {
251 matDataRaw.middleCols(iCol, iLength) = corrected.data.middleCols(iOffset, iLength);
252 if (!doFilter) {
253 matDataFiltered.middleCols(iCol, iLength).setZero();
254 if (doSphara) {
255 matDataRaw.middleCols(iCol, iLength) = m_display.sphara_operator() * matDataRaw.middleCols(iCol, iLength);
256 }
257 }
258
259 for (const mne::TriggerFlank& trigger : corrected.triggers) {
260 if (trigger.sample >= iOffset && trigger.sample < iOffset + iLength) {
261 m_qMapDetectedTrigger[iTriggerCh].push_back({iCol + trigger.sample - iOffset, trigger.value});
262 }
263 } });
264
265 if (!corrected.triggers.empty()) {
266 m_iDetectedTriggers += static_cast<int>(corrected.triggers.size());
267 emit triggerDetected(m_iDetectedTriggers, m_qMapDetectedTrigger);
268 }
269
270 //Filter the whole block at once so the overlap-add stays continuous across the wrap
271 if (doFilter) {
272 const int iBlockStart = m_sweep.write_position() - nCol;
273 writeFiltered(corrected.data, iBlockStart);
274
275 //Perform SPHARA on filtered data after actual filtering - SPHARA should be applied on the best possible data
276 if (doSphara) {
277 viz::for_wrapped_span(iBlockStart - m_display.filter_delay(), nCol, iMaxCols, [&](int iCol, int, int iLength) {
278 matDataFiltered.middleCols(iCol, iLength) = m_display.sphara_operator() * matDataFiltered.middleCols(iCol, iLength);
279 });
280 }
281 }
282 }
283
284 //Update data content
285 QModelIndex topLeft = this->index(0, 1);
286 QModelIndex bottomRight = this->index(static_cast<int>(m_pFiffInfo->ch_names.size()) - 1, 1);
287 emit dataChanged(topLeft, bottomRight, {Qt::DisplayRole});
288}

492 mne::Log::debug("RawModel: {} marked as good.", chInfolist[ch.row()].ch_name);
493 }

495 emit dataChanged(ch, ch);

563void RtFiffRawViewModel::writeFiltered(const MatrixXd& data, int iDataIndex)
564{
565 const std::optional<mne::RtRawDisplay::Filtered> filtered = m_display.filter(data);
566 if (!filtered) {
567 return;
568 }
569
570 MatrixXdR& matDataFiltered = m_sweep.filtered();
571 const int iMaxCols = static_cast<int>(matDataFiltered.cols());
572 const int nCol = static_cast<int>(data.cols());
573
574 //Filter output sample k of this block belongs iFilterDelay samples before the block start
575 viz::for_wrapped_span(iDataIndex - m_display.filter_delay(), nCol, iMaxCols, [&](int iCol, int iOffset, int iLength) {
576 for (int iRow : filtered->filtered_rows) {
577 matDataFiltered.row(iRow).segment(iCol, iLength) = filtered->data.row(iRow).segment(iOffset, iLength);
578 }
579 });
580
581 //Fill filtered data with raw data if the channel was not filtered
582 viz::for_wrapped_span(iDataIndex, nCol, iMaxCols, [&](int iCol, int iOffset, int iLength) {
583 for (int iRow : filtered->passthrough_rows) {
584 matDataFiltered.row(iRow).segment(iCol, iLength) = data.row(iRow).segment(iOffset, iLength);
585 }
586 });
587}

603double RtFiffRawViewModel::getMaxValueFromRawViewModel(int row) const
604{
605 double dMaxValue;
606 qint32 kind = getKind(row);
607
608 switch (kind) {
609 case FIFFV_MEG_CH: {
610 dMaxValue = 1e-11f;
611 qint32 unit = getUnit(row);
612 if (unit == FIFF_UNIT_T_M) { //gradiometers
613 dMaxValue = 1e-10f;
614 if (getScaling().contains(FIFF_UNIT_T_M))
615 dMaxValue = getScaling().at(FIFF_UNIT_T_M);
616 } else if (unit == FIFF_UNIT_T) //magnetometers
617 {
618 dMaxValue = 1e-11f;
619 if (getScaling().contains(FIFF_UNIT_T))
620 dMaxValue = getScaling().at(FIFF_UNIT_T);
621 }
622 break;
623 }
624
625 case FIFFV_REF_MEG_CH: {
626 dMaxValue = 1e-11f;
627 if (getScaling().contains(FIFF_UNIT_T))
628 dMaxValue = getScaling().at(FIFF_UNIT_T);
629 break;
630 }
631 case FIFFV_EEG_CH: {
632 dMaxValue = 1e-4f;
633 if (getScaling().contains(FIFFV_EEG_CH))
634 dMaxValue = getScaling().at(FIFFV_EEG_CH);
635 break;
636 }
637 case FIFFV_EOG_CH: {
638 dMaxValue = 1e-3f;
639 if (getScaling().contains(FIFFV_EOG_CH))
640 dMaxValue = getScaling().at(FIFFV_EOG_CH);
641 break;
642 }
643 case FIFFV_STIM_CH: {
644 dMaxValue = 5;
645 if (getScaling().contains(FIFFV_STIM_CH))
646 dMaxValue = getScaling().at(FIFFV_STIM_CH);
647 break;
648 }
649 case FIFFV_MISC_CH: {
650 dMaxValue = 1e-3f;
651 if (getScaling().contains(FIFFV_MISC_CH))
652 dMaxValue = getScaling().at(FIFFV_MISC_CH);
653 break;
654 }
655 default:
656 dMaxValue = 1e-9f;
657 break;
658 }
659
660 return dMaxValue;
661}
