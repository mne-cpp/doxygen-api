49SpectrumPanel::SpectrumPanel()
50: Panel("SpectrumView", "Frequency Spectrum")
51, m_spectrum(std::make_shared<SpectrumFigure>())
52{
53 set_figure(m_spectrum);
54 for (const auto& [key, label] : {std::pair{kLower, "Lower Frequency"}, std::pair{kUpper, "Upper Frequency"}}) {
55 SettingField bound;
56 bound.key = key;
57 bound.kind = SettingField::Kind::Double;
58 bound.label = label;
59 bound.default_value = 0.0;
60 // Until init() knows the Nyquist frequency, bounds set before are kept
61 bound.maximum = 1e6;
62 bound.step = 0.001;
63 bound.decimals = 3;
64 bound.unit = " Hz";
65 bound.style = SettingField::Style::Slider;
66 bound.persistent = false;
67 form().add(std::move(bound));
68 }
69 m_listener = form().add_listener([this](const std::string&, const std::optional<SettingValue>& value) {
70 if (m_updating || !value) {
71 return;
72 }
73 // The lower bound stays below the upper one
74 if (form().get<double>(kLower) > form().get<double>(kUpper)) {
75 form().set_value(kLower, form().get<double>(kUpper));
76 return;
77 }
78 apply_bounds();
79 if (m_boundsCallback) {
80 m_boundsCallback(lower_bound(), upper_bound());
81 }
82 });
83}

94void SpectrumPanel::init(std::shared_ptr<FiffInfo> info, SpectrumTable::Scale scale)
95{
96 const double nyquist = info ? std::floor(info->sfreq / 2.0) : 0.0;
97 m_spectrum->set_info(std::move(info), scale);
98 m_boundsApplied = false;
99 const bool updating = std::exchange(m_updating, true);
100 form().set_range(kLower, 0.0, nyquist);
101 form().set_range(kUpper, 0.0, nyquist);
102 // Without bounds yet the whole spectrum shows
103 if (form().get<double>(kUpper) <= 0.0) {
104 form().set_value(kUpper, nyquist);
105 }
106 m_updating = updating;
107}
