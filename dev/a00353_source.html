39bool validScheme(std::string_view scheme) noexcept
40{
41 if (scheme.empty() || scheme[0] < 'a' || scheme[0] > 'z') {
42 return false;
43 }
44 for (const char c : scheme) {
45 const bool ok = (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '+' || c == '-' || c == '.' || c == '_';
46 if (!ok) {
47 return false;
48 }
49 }
50 return true;
51}

93AcqError AcqDeviceRegistry::registerScheme(AcqSchemeInfo info, Factory factory, Discoverer discoverer)
94{
95 if (!validScheme(info.scheme)) {
96 return {AcqErrorCode::InvalidParameter, "scheme '" + info.scheme + "' is not lower-case [a-z][a-z0-9+.-]*"};
97 }
98 if (!factory) {
99 return {AcqErrorCode::InvalidParameter, "scheme '" + info.scheme + "' has no factory"};
100 }
101 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
102 if (m_pImpl->entries.count(info.scheme) != 0) {
103 return {AcqErrorCode::InvalidParameter, "scheme '" + info.scheme + "' is already registered"};
104 }
105 std::string scheme = info.scheme;
106 m_pImpl->entries.emplace(std::move(scheme), Impl::Entry{std::move(info), std::move(factory), std::move(discoverer)});
107 return {};
108}

112AcqExpected<AcqDevice> AcqDeviceRegistry::open(std::string_view uri) const
113{
114 auto parsed = AcqUri::parse(uri);
115 if (!parsed) {
116 return parsed.error();
117 }
118 Factory factory;
119 AcqSchemeInfo info;
120 {
121 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
122 const auto it = m_pImpl->entries.find(parsed->scheme);
123 if (it == m_pImpl->entries.end()) {
124 std::string known;
125 for (const auto& [name, entry] : m_pImpl->entries) {
126 known.append(known.empty() ? "" : ", ").append(name);
127 }
128 return AcqError(AcqErrorCode::UnknownScheme, "'" + parsed->scheme + "://'; registered: " + known);
129 }
130 factory = it->second.factory;
131 info = it->second.info;
132 }
133 for (const auto& [key, value] : parsed->params) {
134 const bool known = std::any_of(info.parameters.begin(), info.parameters.end(),
135 [&key](const AcqParameterInfo& parameter) { return parameter.name == key; });
136 if (!known) {
137 std::string message = "unknown parameter '" + key + "' for " + info.scheme + "://; accepted:";
138 for (const AcqParameterInfo& parameter : info.parameters) {
139 message.append(" ").append(parameter.name);
140 }
141 return AcqError(AcqErrorCode::InvalidParameter, std::move(message));
142 }
143 }
144 auto backend = factory(*parsed);
145 if (!backend) {
146 return backend.error();
147 }
148 if (!*backend) {
149 return AcqError(AcqErrorCode::Internal, "the " + info.scheme + ":// factory returned no backend");
150 }
151 return AcqDevice(std::move(*backend));
152}

169std::vector<AcqDiscoveredDevice> AcqDeviceRegistry::discover(std::chrono::milliseconds timeout) const
170{
171 std::vector<Discoverer> discoverers;
172 {
173 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
174 for (const auto& [name, entry] : m_pImpl->entries) {
175 if (entry.discoverer) {
176 discoverers.push_back(entry.discoverer);
177 }
178 }
179 }
180 std::vector<std::vector<AcqDiscoveredDevice>> found(discoverers.size());
181 std::vector<std::thread> threads;
182 threads.reserve(discoverers.size());
183 for (std::size_t i = 0; i < discoverers.size(); ++i) {
184 threads.emplace_back([&found, &discoverers, i, timeout] { found[i] = discoverers[i](timeout); });
185 }
186 std::vector<AcqDiscoveredDevice> out;
187 for (std::size_t i = 0; i < threads.size(); ++i) {
188 threads[i].join();
189 out.insert(out.end(), found[i].begin(), found[i].end());
190 }
191 return out;
192}
