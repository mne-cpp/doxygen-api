145{
146 if (m_sSettingsPath.isEmpty()) {
147 return;
148 }
149
150 QSettings settings("MNECPP");
151
152 settings.beginGroup(m_sSettingsPath + QString("/AverageSelectionView/averageColorMap"));
153 for (const QString& key : settings.childKeys()) {

159 for (const QString& key : settings.childKeys()) {
160 m_qMapAverageActivation->insert_or_assign(key.toStdString(), settings.value(key).toBool());
161 }

191void AverageSelectionView::redrawGUI()
192{
193 if (m_qMapAverageColor->size() != m_qMapAverageActivation->size()) {
194 mne::Log::debug("AverageSelectionView::update - m_qMapAverageColor and m_qMapAverageActivation do not match in size. Returning.");
195 return;
196 }
197
198 //Delete all widgets in the averages layout
199 QGridLayout* topLayout = static_cast<QGridLayout*>(this->layout());
200 if (!topLayout) {
201 topLayout = new QGridLayout(this);
202 }
203
204 // The widgets go right away: the view is redrawn synchronously and looked up by name
205 detail::clearLayout(topLayout);
206
207 // Create new GUI elements
208 int count = 0;
209 for (const auto& [key, color] : *m_qMapAverageColor) {
210 if (count >= m_iMaxNumAverages) {
211 break;
212 }
213
214 const QString name = QString::fromStdString(key);
215
216 //Create average active checkbox
217 QPointer<QCheckBox> pCheckBox = new QCheckBox(name, this);
218 pCheckBox->setChecked(detail::valueOr(*m_qMapAverageActivation, key));
219 pCheckBox->setObjectName(name);
220 topLayout->addWidget(pCheckBox, count, 0);
221 connect(pCheckBox.data(), &QCheckBox::clicked,
222 this, &AverageSelectionView::onAverageSelectionColorChanged);
223
224 //Create average color pushbutton
225 QPointer<QPushButton> pButton = new QPushButton("Click to change", this);
226 pButton->setObjectName(name);
227 pButton->setStyleSheet(QString("background-color: rgb(%1, %2, %3);").arg(color.red()).arg(color.green()).arg(color.blue()));
228 topLayout->addWidget(pButton, count, 1);
229 connect(pButton.data(), &QPushButton::clicked,
230 this, &AverageSelectionView::onAverageSelectionColorChanged);
231
232 ++count;
233 }
234}

238void AverageSelectionView::onAverageSelectionColorChanged()
239{
240 //Change color for average
241 if (QPointer<QPushButton> button = qobject_cast<QPushButton*>(sender())) {
242 const std::string sObjectName = button->objectName().toStdString();
243
244 QColor color = QColorDialog::getColor(detail::valueOr(*m_qMapAverageColor, sObjectName), this, "Set average color");
245 // Cancelled: getColor() returns an invalid colour
246 if (!color.isValid()) {
247 return;
248 }
249
250 if (button) {
251 QPalette palette(QPalette::Button, color);
252 button->setPalette(palette);
253 button->update();
254
255 //Set color of button new new scene color
256 button->setStyleSheet(QString("background-color: rgb(%1, %2, %3);").arg(color.red()).arg(color.green()).arg(color.blue()));
257 }
258
259 m_qMapAverageColor->insert_or_assign(sObjectName, color);
260
261 emit newAverageColorMap(m_qMapAverageColor);
262 }
263
264 //Change color for average
265 if (QPointer<QCheckBox> checkBox = qobject_cast<QCheckBox*>(sender())) {
266 const std::string sObjectName = checkBox->objectName().toStdString();
267
268 m_qMapAverageActivation->insert_or_assign(sObjectName, checkBox->isChecked());
269
270 emit newAverageActivationMap(m_qMapAverageActivation);
271 }
272
273 saveSettings();
274}
