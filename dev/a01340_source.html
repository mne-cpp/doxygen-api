54class JsonCursor
55{
56public:
57 explicit JsonCursor(std::string_view text)
58 : m_text(text)
59 {
60 }
61
62 bool eat(char c)
63 {
64 skip_space();
65 if (m_at < m_text.size() && m_text[m_at] == c) {
66 ++m_at;
67 return true;
68 }
69 return false;
70 }
71
72 [[nodiscard]] char peek()
73 {
74 skip_space();
75 return m_at < m_text.size() ? m_text[m_at] : '\0';
76 }
77
78 std::optional<std::string> string()
79 {
80 if (!eat('"')) {
81 return std::nullopt;
82 }
83 std::string out;
84 while (m_at < m_text.size() && m_text[m_at] != '"') {
85 if (m_text[m_at] == '\\' && m_at + 1 < m_text.size()) {
86 ++m_at; // board names are ASCII; an escape stands for the character after it
87 }
88 out += m_text[m_at++];
89 }
90 if (m_at == m_text.size()) {
91 return std::nullopt;
92 }
93 ++m_at;
94 return out;
95 }
96
98 std::optional<int> integer()
99 {
100 skip_space();
101 int value = 0;
102 const char* begin = m_text.data() + m_at;
103 const auto [end, error] = std::from_chars(begin, m_text.data() + m_text.size(), value);
104 if (error != std::errc()) {
105 return std::nullopt;
106 }
107 m_at += static_cast<std::size_t>(end - begin);
108 return value;
109 }
110
112 bool skip_value()
113 {
114 const char c = peek();
115 if (c == '"') {
116 return string().has_value();
117 }
118 if (c == '[' || c == '{') {
119 const char close = c == '[' ? ']' : '}';
120 ++m_at;
121 if (eat(close)) {
122 return true;
123 }
124 do {
125 if (close == '}' && (!string() || !eat(':'))) {
126 return false;
127 }
128 if (!skip_value()) {
129 return false;
130 }
131 } while (eat(','));
132 return eat(close);
133 }
134 const std::size_t begin = m_at;
135 while (m_at < m_text.size() && std::string_view(",]} \t\r\n").find(m_text[m_at]) == std::string_view::npos) {
136 ++m_at;
137 }
138 return m_at > begin;
139 }
140
141private:
142 void skip_space()
143 {
144 while (m_at < m_text.size() && std::isspace(static_cast<unsigned char>(m_text[m_at])) != 0) {
145 ++m_at;
146 }
147 }
148
149 std::string_view m_text;
150 std::size_t m_at = 0;
151};

175AcqExpected<std::shared_ptr<const BrainFlowLibrary>> BrainFlowLibrary::load(const std::vector<std::string>& candidates)
176{
177 // One library per candidate list while any session uses it
178 static std::mutex mutex;
179 static std::map<std::vector<std::string>, std::weak_ptr<const BrainFlowLibrary>> loaded;
180 const std::lock_guard<std::mutex> lock(mutex);
181 if (auto existing = loaded[candidates].lock()) {
182 return existing;
183 }
184 auto shared = AcqSharedLibrary::load(candidates);
185 if (!shared) {
186 return AcqError(AcqErrorCode::BackendUnavailable,
187 "BrainFlow not found: " + shared.error().text() + "; pip install brainflow, set MNE_CPP_BRAINFLOW_LIB or pass library=");
188 }
189 auto library = std::make_shared<BrainFlowLibrary>();
190 library->m_library = std::move(*shared);
191 std::string missing;
192 const auto bind = [&](auto& function, const char* name) {
193 function = reinterpret_cast<std::remove_reference_t<decltype(function)>>(library->m_library.symbol(name));
194 if (function == nullptr) {
195 missing += (missing.empty() ? "" : ", ") + std::string(name);
196 }
197 };
198 bind(library->prepare_session, "prepare_session");
199 bind(library->start_stream, "start_stream");
200 bind(library->get_board_data_count, "get_board_data_count");
201 bind(library->get_board_data, "get_board_data");
202 bind(library->stop_stream, "stop_stream");
203 bind(library->release_session, "release_session");
204 bind(library->m_get_board_descr, "get_board_descr");
205 bind(library->m_get_version, "get_version_board_controller");
206 if (!missing.empty()) {
207 return AcqError(AcqErrorCode::BackendUnavailable, library->m_library.path() + " is not BrainFlow's BoardController: it lacks " + missing);
208 }
209 std::array<char, 64> version{};
210 int length = 0;
211 if (const int code = library->m_get_version(version.data(), &length, static_cast<int>(version.size())); code != 0) {
212 return brainflow_error(code, "get_version_board_controller");
213 }
214 library->m_version.assign(version.data(), static_cast<std::size_t>(std::clamp(length, 0, static_cast<int>(version.size()))));
215 if (major_of(library->m_version) != kSupportedMajor) {
216 return AcqError(AcqErrorCode::UnsupportedVersion, std::format("BrainFlow {} at {} is not supported; brainflow:// needs BrainFlow {}.x", library->m_version, library->m_library.path(), kSupportedMajor));
217 }
218 loaded[candidates] = library;
219 return std::shared_ptr<const BrainFlowLibrary>(std::move(library));
220}

238AcqExpected<int> BrainFlowLibrary::find_board(std::string_view name) const
239{
240 const std::string wanted = brainflow_board_key(name);
241 std::string known;
242 for (int board = kFirstBoard; board <= kLastBoard; ++board) {
243 auto description = this->description(board);
244 if (!description) {
245 continue;
246 }
247 if (brainflow_board_key(description->name) == wanted) {
248 return board;
249 }
250 known += (known.empty() ? "" : ", ") + brainflow_board_key(description->name);
251 }
252 return AcqError(AcqErrorCode::InvalidParameter, std::format("BrainFlow {} has no board '{}'; pass a board id or one of: {}", m_version, name, known));
253}

259AcqExpected<BrainFlowDescription> mne::detail::parse_brainflow_description(std::string_view json)
260{
261 const auto malformed = [&] {
262 return AcqError(AcqErrorCode::DeviceIo, "malformed BrainFlow board description: " + std::string(json.substr(0, 200)));
263 };
264 JsonCursor cursor(json);
265 BrainFlowDescription out;
266 if (!cursor.eat('{')) {
267 return malformed();
268 }
269 if (cursor.eat('}')) {
270 return out;
271 }
272 do {
273 auto key = cursor.string();
274 if (!key || !cursor.eat(':')) {
275 return malformed();
276 }
277 if (*key == "name" || *key == "eeg_names") {
278 auto text = cursor.string();
279 if (!text) {
280 return malformed();
281 }
282 if (*key == "name") {
283 out.name = std::move(*text);
284 } else {
285 for (std::size_t begin = 0; begin <= text->size();) {
286 const std::size_t end = std::min(text->find(',', begin), text->size());
287 out.eeg_names.push_back(text->substr(begin, end - begin));
288 begin = end + 1;
289 }
290 }
291 } else if (cursor.peek() == '[' && key->ends_with("_channels")) {
292 cursor.eat('[');
293 std::vector<int> rows;
294 if (!cursor.eat(']')) {
295 do {
296 const auto row = cursor.integer();
297 if (!row) {
298 return malformed();
299 }
300 rows.push_back(*row);
301 } while (cursor.eat(','));
302 if (!cursor.eat(']')) {
303 return malformed();
304 }
305 }
306 out.roles.emplace_back(std::move(*key), std::move(rows));
307 } else if (*key == "sampling_rate" || *key == "num_rows" || key->ends_with("_channel")) {
308 const auto value = cursor.integer();
309 if (!value) {
310 return malformed();
311 }
312 if (*key == "sampling_rate") {
313 out.sampling_rate = *value;
314 } else if (*key == "num_rows") {
315 out.rows = static_cast<std::size_t>(std::max(*value, 0));
316 } else {
317 out.roles.emplace_back(std::move(*key), std::vector<int>{*value});
318 }
319 } else if (!cursor.skip_value()) {
320 return malformed();
321 }
322 } while (cursor.eat(','));
323 if (!cursor.eat('}')) {
324 return malformed();
325 }
326 for (const auto& [role, rows] : out.roles) {
327 if (std::any_of(rows.begin(), rows.end(), [&](int row) { return row < 0 || static_cast<std::size_t>(row) >= out.rows; })) {
328 return AcqError(AcqErrorCode::DeviceIo, std::format("BrainFlow board {}: {} lies outside its {} rows", out.name, role, out.rows));
329 }
330 }
331 return out;
332}

336AcqError mne::detail::brainflow_error(int code, std::string_view call)
337{
338 // BrainFlow's exit codes
339 static constexpr std::array<std::string_view, 24> kNames = {
340 "STATUS_OK", "PORT_ALREADY_OPEN_ERROR", "UNABLE_TO_OPEN_PORT_ERROR",
341 "SET_PORT_ERROR", "BOARD_WRITE_ERROR", "INCOMMING_MSG_ERROR",
342 "INITIAL_MSG_ERROR", "BOARD_NOT_READY_ERROR", "STREAM_ALREADY_RUN_ERROR",
343 "INVALID_BUFFER_SIZE_ERROR", "STREAM_THREAD_ERROR", "STREAM_THREAD_IS_NOT_RUNNING",
344 "EMPTY_BUFFER_ERROR", "INVALID_ARGUMENTS_ERROR", "UNSUPPORTED_BOARD_ERROR",
345 "BOARD_NOT_CREATED_ERROR", "ANOTHER_BOARD_IS_CREATED_ERROR", "GENERAL_ERROR",
346 "SYNC_TIMEOUT_ERROR", "JSON_NOT_FOUND_ERROR", "NO_SUCH_DATA_IN_JSON_ERROR",
347 "CLASSIFIER_IS_NOT_PREPARED_ERROR", "ANOTHER_CLASSIFIER_IS_PREPARED_ERROR",
348 "UNSUPPORTED_CLASSIFIER_AND_METRIC_COMBINATION_ERROR"};
349 AcqErrorCode kind = AcqErrorCode::DeviceIo;
350 switch (code) {
351 case 2:
352 case 3:
353 kind = AcqErrorCode::DeviceNotFound;
354 break;
355 case 13:
356 case 14:
357 kind = AcqErrorCode::InvalidParameter;
358 break;
359 case 18:
360 kind = AcqErrorCode::Timeout;
361 break;
362 default:
363 break;
364 }
365 const std::string_view name = code >= 0 && static_cast<std::size_t>(code) < kNames.size() ? kNames[static_cast<std::size_t>(code)] : "unknown";
366 return {kind, std::format("BrainFlow {} failed: {} ({})", call, name, code)};
367}

371std::string mne::detail::brainflow_board_key(std::string_view name)
372{
373 std::string out;
374 for (const char c : name) {
375 if (std::isalnum(static_cast<unsigned char>(c)) != 0) {
376 out += static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
377 }
378 }
379 if (out.size() > 5 && out.ends_with("board")) {
380 out.resize(out.size() - 5);
381 }
382 return out;
