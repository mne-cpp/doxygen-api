48FiffInfo MaxwellMovementComp::transformFiffInfo(const FiffInfo& fiffInfo,
49 const HeadPosEntry& headPosRef,
50 const HeadPosEntry& headPosCurrent)
51{
52 FiffInfo info(fiffInfo);
53
54 // Compute relative rotation: from current → ref
55 // R_rel = R_ref * R_current^-1
56 Matrix3d rotRef = headPosRef.rotation.toRotationMatrix();
57 Matrix3d rotCur = headPosCurrent.rotation.toRotationMatrix();
58 Matrix3d rotRel = rotRef * rotCur.transpose();
59
60 // Translation: t_rel = t_ref - R_rel * t_current
61 Vector3d tRel = headPosRef.translation - rotRel * headPosCurrent.translation;
62
63 // Apply transform to MEG channel positions
64 for (int i = 0; i < std::ssize(info.chs); ++i) {
65 if (info.chs[i].kind == FIFFV_MEG_CH || info.chs[i].kind == FIFFV_REF_MEG_CH) {
66 // Transform position
67 Vector3f r0 = info.chs[i].chpos.r0;
68 Vector3d r0d = r0.cast<double>();
69 r0d = rotRel * r0d + tRel;
70 info.chs[i].chpos.r0 = r0d.cast<float>();
71
72 // Transform orientation vectors
73 Vector3f ex = info.chs[i].chpos.ex;
74 Vector3d exd = ex.cast<double>();
75 exd = rotRel * exd;
76 info.chs[i].chpos.ex = exd.cast<float>();
77
78 Vector3f ey = info.chs[i].chpos.ey;
79 Vector3d eyd = ey.cast<double>();
80 eyd = rotRel * eyd;
81 info.chs[i].chpos.ey = eyd.cast<float>();
82
83 Vector3f ez = info.chs[i].chpos.ez;
84 Vector3d ezd = ez.cast<double>();
85 ezd = rotRel * ezd;
86 info.chs[i].chpos.ez = ezd.cast<float>();
87 }
88 }
89
90 return info;
91}

95MatrixXd MaxwellMovementComp::apply(const MatrixXd& matData,
96 const FiffInfo& fiffInfo,
97 const std::vector<HeadPosEntry>& headPos,
98 double dSFreq,
99 const MaxwellMoveCompParams& params)
100{
101 if (headPos.empty()) {
102 Log::warning("[MaxwellMovementComp::apply] No head positions provided.");
103 return matData;
104 }
105
106 const int nSamples = matData.cols();
107
108 // Determine reference head position
109 HeadPosEntry refPos;
110 if (params.iRefIdx >= 0 && params.iRefIdx < std::ssize(headPos)) {
111 refPos = headPos[params.iRefIdx];
112 } else {
113 // Mean position
114 Vector3d meanTrans = Vector3d::Zero();
115 for (const auto& hp : headPos) {
116 meanTrans += hp.translation;
117 }
118 meanTrans /= headPos.size();
119 refPos.translation = meanTrans;
120 refPos.rotation = headPos[0].rotation; // Use first rotation as reference
121 }
122
123 // Compute reference SSS basis
124 SSSParams sssParams;
125 sssParams.iOrderIn = params.iOrderIn;
126 sssParams.iOrderOut = params.iOrderOut;
127 sssParams.origin = params.origin;
128 sssParams.dRegIn = params.dRegIn;
129
130 SSS::Basis basisRef = SSS::computeBasis(fiffInfo, sssParams);
131
132 MatrixXd result = matData;
133
134 // Process each time segment
135 for (int iSeg = 0; iSeg < std::ssize(headPos); ++iSeg) {
136 // Determine sample range for this segment
137 int iStart = static_cast<int>(headPos[iSeg].dTime * dSFreq);
138 int iEnd;
139 if (iSeg + 1 < std::ssize(headPos)) {
140 iEnd = static_cast<int>(headPos[iSeg + 1].dTime * dSFreq);
141 } else {
142 iEnd = nSamples;
143 }
144
145 iStart = std::clamp(iStart, 0, nSamples);
146 iEnd = std::clamp(iEnd, iStart, nSamples);
147 if (iStart >= iEnd)
148 continue;
149
150 int nSegSamples = iEnd - iStart;
151
152 // Transform FiffInfo to current head position
153 FiffInfo infoCurrentToRef = transformFiffInfo(fiffInfo, refPos, headPos[iSeg]);
154
155 // Compute SSS basis at current head position (as seen from reference)
156 SSS::Basis basisCurrent = SSS::computeBasis(infoCurrentToRef, sssParams);
157
158 // Extract segment data
159 MatrixXd segData = matData.middleCols(iStart, nSegSamples);
160
161 // Project into multipole space using current-position basis
162 // multipoles = pinv_current * data_meg
163 MatrixXd megData(basisCurrent.megChannelIdx.size(), nSegSamples);
164 for (int ch = 0; ch < std::ssize(basisCurrent.megChannelIdx); ++ch) {
165 megData.row(ch) = segData.row(basisCurrent.megChannelIdx[ch]);
166 }
167
168 MatrixXd multipoles = basisCurrent.matPinvAll * megData;
169
170 // Reconstruct at reference position using only internal multipoles
171 MatrixXd reconstructed = basisRef.matSin * multipoles.topRows(basisRef.iNin);
172
173 // Place reconstructed data back
174 for (int ch = 0; ch < std::ssize(basisRef.megChannelIdx); ++ch) {
175 result.row(basisRef.megChannelIdx[ch]).segment(iStart, nSegSamples) = reconstructed.row(ch);
176 }
177 }
178
179 return result;
180}

184Expected<std::vector<HeadPosEntry>> MaxwellMovementComp::read_head_pos(const std::filesystem::path& path)
185{
186 const Expected<std::string> text = read_text_file(path);
187 if (!text) {
188 return std::unexpected(text.error());
189 }
190 std::vector<HeadPosEntry> positions;
191 std::string_view rest = *text;
192 for (bool header = true; !rest.empty(); header = false) {
193 const std::size_t end = rest.find('\n');
194 const std::string_view line = rest.substr(0, end);
195 rest = end == std::string_view::npos ? std::string_view() : rest.substr(end + 1);
196 const std::vector<std::string_view> fields = split_whitespace(line);
197 if (header || fields.empty() || fields.front().starts_with('#')) {
198 continue;
199 }
200 std::array<double, 10> values{};
201 bool ok = fields.size() == values.size();
202 for (std::size_t k = 0; ok && k < values.size(); ++k) {
203 const std::optional<double> value = to_double(fields[k]);
204 ok = value.has_value();
205 values[k] = value.value_or(0.0);
206 }
207 if (!ok) {
208 return fail(ErrorCode::Format, "{} has no 10 head position values in line \"{}\"", utf8_from_path(path),
209 line);
210 }
211 HeadPosEntry& entry = positions.emplace_back();
212 entry.dTime = values[0];
213 const Vector3d q(values[1], values[2], values[3]);
214 entry.rotation = Quaterniond(std::sqrt(std::max(0.0, 1.0 - q.squaredNorm())), q.x(), q.y(), q.z());
215 entry.translation = Vector3d(values[4], values[5], values[6]);
216 entry.dGof = values[7];
217 entry.dError = values[8];
218 entry.dVelocity = values[9];
219 }
220 return positions;
221}

225Expected<void> MaxwellMovementComp::write_head_pos(const std::filesystem::path& path,
226 std::span<const HeadPosEntry> headPos)
227{
228 std::string text = " Time q1 q2 q3 q4 q5 q6 g-value error velocity\n";
229 for (const HeadPosEntry& entry : headPos) {
230 // The file holds the quaternion with a non-negative scalar part: q and -q are the same rotation
231 const Quaterniond q = entry.rotation.w() < 0.0 ? Quaterniond(-entry.rotation.coeffs()) : entry.rotation;
232 std::format_to(std::back_inserter(text), " {: 9.3f}", entry.dTime);
233 for (const double value : {q.x(), q.y(), q.z(), entry.translation.x(), entry.translation.y(),
234 entry.translation.z(), entry.dGof, entry.dError, entry.dVelocity}) {
235 std::format_to(std::back_inserter(text), " {: 8.5f}", value);
236 }
237 text += '\n';
238 }
239 return write_text_file(path, text);
240}
