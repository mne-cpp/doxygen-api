52std::unique_ptr<InfoVersion> makeVersion(AcqStreamInfo info, std::uint32_t version, std::uint64_t validFrom)
53{
54 auto v = std::make_unique<InfoVersion>();
55 info.version = version;
56 info.validFromSample = validFrom;
57 for (const AcqChannelInfo& ch : info.channels) {
58 v->scales.push_back(ch.scale);

69template<typename T>
70void initRecords(RecordRing<T>& ring, std::size_t capacity, std::size_t payloadBytes)
71{
72 const std::uint64_t cap = nextPow2(std::max<std::uint64_t>(capacity, 2));
73 ring.records.assign(cap, T{});
74 ring.payloadBytes = payloadBytes;
75 ring.payloads.assign(cap * payloadBytes, std::byte{0});
76 ring.mask = cap - 1;
77}

79template<typename T>
80void pushRecord(RecordRing<T>& ring, const T& record, std::span<const std::byte> payload, bool publish) noexcept
81{
82 const std::uint64_t index = ring.pushed;
83 ring.reserved.store(index + 1, std::memory_order_relaxed);
84 std::atomic_thread_fence(std::memory_order_seq_cst);
85 const std::uint64_t slot = index & ring.mask;
86 storeRecord(ring.records[slot], record);
87 if (ring.payloadBytes > 0) {
88 std::byte* dst = ring.payloads.data() + slot * ring.payloadBytes;
89 const std::size_t n = std::min(payload.size(), ring.payloadBytes);
90 if (n > 0) { // an empty span's data() may be null, which memcpy must never receive
91 std::memcpy(dst, payload.data(), n);
92 }
93 std::memset(dst + n, 0, ring.payloadBytes - n);
94 }
95 ring.pushed = index + 1;
96 if (publish) {
97 ring.published.store(ring.pushed, std::memory_order_release);
98 }
99}

132AcqExpected<std::shared_ptr<AcqStreamCore>> AcqStreamCore::create(AcqStreamInfo info, const AcqStreamOptions& options)
133{
134 const std::size_t bps = mne::bytesPerSample(info.format);
135 if (bps == 0 || (info.layout != AcqLayout::Interleaved && info.layout != AcqLayout::Planar)) {
136 return AcqErrorCode::UnsupportedFormat;
137 }
138 if (!(info.sfreq >= 0.0) || !std::isfinite(info.sfreq) || options.maxReaders == 0 || options.maxReaders > 4096 ||
139 options.rowAlignment == 0 || !std::has_single_bit(options.rowAlignment) || options.alignment < 64 ||
140 !std::has_single_bit(options.alignment)) {
141 return AcqErrorCode::InvalidParameter;
142 }
143
144 std::shared_ptr<AcqStreamCore> core(new AcqStreamCore());
145 core->direction = info.direction;
146 core->format = info.format;
147 core->layout = info.layout;
148 core->underrunFill = info.underrunFill;
149 core->channels = info.channels.size();
150 core->bytesPerSample = bps;
151 core->frameBytes = core->channels * bps;
152
153 if (options.external.data != nullptr) {
154 const AcqExternalBuffer& ext = options.external;
155 if (reinterpret_cast<std::uintptr_t>(ext.data) % options.alignment != 0 || core->channels == 0 ||
156 (ext.memory.kind != AcqMemoryKind::Host && ext.memory.device < 0)) {
157 return AcqErrorCode::InvalidBuffer;
158 }
159 std::uint64_t capacity = 0;
160 if (info.layout == AcqLayout::Interleaved) {
161 capacity = ext.bytes / core->frameBytes;
162 if (capacity * core->frameBytes != ext.bytes) {
163 return AcqErrorCode::InvalidBuffer;
164 }
165 } else {
166 core->rowStride = ext.channelStride != 0 ? ext.channelStride
167 : static_cast<std::ptrdiff_t>(ext.bytes / core->channels);
168 capacity = options.capacitySamples != 0 ? options.capacitySamples
169 : static_cast<std::uint64_t>(core->rowStride) / bps;
170 if (capacity * bps > static_cast<std::uint64_t>(core->rowStride) ||
171 static_cast<std::uint64_t>(core->rowStride) * core->channels > ext.bytes) {
172 return AcqErrorCode::InvalidBuffer;
173 }
174 }
175 if (capacity < 2 || !std::has_single_bit(capacity)) {
176 return AcqErrorCode::InvalidBuffer;
177 }
178 if (options.capacitySamples != 0 && options.capacitySamples != capacity) {
179 return AcqErrorCode::InvalidParameter;
180 }
181 core->capacity = capacity;
182 core->base = ext.data;
183 core->memory = ext.memory;
184 core->bufferBytes = ext.bytes;
185 core->releaseExternal = ext.release;
186 } else {
187 std::uint64_t capacity = 0;
188 if (options.capacitySamples != 0) {
189 capacity = nextPow2(std::max<std::uint64_t>(options.capacitySamples, 2));
190 } else {
191 capacity = nextPow2(std::max<std::uint64_t>(65536, static_cast<std::uint64_t>(std::ceil(2.0 * info.sfreq))));
192 while (capacity > 1024 && capacity * core->frameBytes > kDefaultMaxRingBytes) {
193 capacity /= 2;
194 }
195 }
196 core->capacity = capacity;
197 if (info.layout == AcqLayout::Planar) {
198 const std::uint64_t row = capacity * bps;
199 const std::uint64_t align = options.rowAlignment;
200 core->rowStride = static_cast<std::ptrdiff_t>((row + align - 1) / align * align);
201 core->bufferBytes = static_cast<std::size_t>(core->rowStride) * core->channels;
202 } else {
203 core->bufferBytes = static_cast<std::size_t>(capacity * core->frameBytes);
204 }
205 if (core->bufferBytes > 0) {
206 // Only interleaved rings mirror: a planar ring would need every channel row mirrored on its own.
207 core->ownedMemory = allocateRingMemory(core->bufferBytes, options.alignment, options.hugePages,
208 options.numaNode,
209 options.mirror && info.layout == AcqLayout::Interleaved);
210 if (core->ownedMemory.data == nullptr) {
211 return AcqErrorCode::OutOfMemory;
212 }
213 core->base = core->ownedMemory.data;
214 if (options.prefault) {
215 // Touch the mirror too, so its page-table entries exist before start().
216 std::memset(core->base, 0, core->ownedMemory.bytes * (core->ownedMemory.mirrored ? 2 : 1));
217 }
218 if (options.lockMemory) {
219 lockRingMemory(core->ownedMemory);
220 }
221 }
222 }
223 core->mask = core->capacity - 1;
224
225 initRecords(core->events, options.eventCapacity, 0);
226 if (core->direction == AcqDirection::Output) {
227 initRecords(core->outputCommits, kOutputCommitLog, 0);
228 }
229 if (info.segmentation.mode == AcqSegmentation::Mode::Triggered) {
230 const std::uint64_t perSegment = std::max<std::uint64_t>(info.segmentation.samplesPerSegment, 1);
231 const std::size_t count = options.segmentCapacity != 0
232 ? options.segmentCapacity
233 : static_cast<std::size_t>(std::max<std::uint64_t>(64, core->capacity / perSegment));
234 initRecords(core->segments, count, info.segmentation.headerBytes);
235 } else {
236 initRecords(core->segments, 2, 0);
237 }
238
239 AcqClockModel clock;
240 clock.sfreq = info.sfreq;
241 clock.output = info.direction == AcqDirection::Output;
242 core->clock.store(clock);
243
244 core->maxReaders = options.maxReaders;
245 core->slots = std::make_unique<ReaderSlot[]>(options.maxReaders);
246 core->versions[0] = makeVersion(std::move(info), 0, 0);
247 core->versionCount.store(1, std::memory_order_release);
248 return core;
249}

297void AcqStreamCore::fillRange(std::uint64_t from, std::uint64_t to) noexcept
298{
299 if (to <= from || base == nullptr) {
300 return;
301 }
302 if (to - from > capacity) {
303 from = to - capacity;
304 }
305 reserveUpTo(to);
306 if (!memory.hostAccessible()) {
307 return; // the device emits its fill itself
308 }
309 std::array<AcqWritePart, 2> parts;
310 const std::size_t n = buildParts(from, to - from, parts);
311 const bool hold = underrunFill == AcqUnderrunFill::HoldLast && from > 0;
312 std::array<std::byte, 16> zero{};
313 for (std::size_t p = 0; p < n; ++p) {
314 for (std::size_t c = 0; c < channels; ++c) {
315 const std::byte* src = zero.data();
316 if (hold) {
317 const std::uint64_t last = (from - 1) & mask;
318 src = layout == AcqLayout::Interleaved ? base + last * frameBytes + c * bytesPerSample
319 : base + static_cast<std::ptrdiff_t>(c) * rowStride +
320 static_cast<std::ptrdiff_t>(last * bytesPerSample);
321 }
322 for (std::size_t s = 0; s < parts[p].samples; ++s) {
323 std::byte* dst = parts[p].at(c, s);
324 if (dst != src) {
325 std::memcpy(dst, src, bytesPerSample);
326 }
327 }
328 }
329 }

335{
336 if (waiters.load(std::memory_order_seq_cst) == 0) {
337 return;
338 }

340 ReaderSlot& slot = slots[i];

342 slot.wake.release();
343 }

443AcqExpected<AcqReader> AcqStream::reader(AcqReader::ReadMode mode, const AcqReaderOptions& options) const
444{
445 if (!m_core) {
446 return AcqErrorCode::InvalidState;
447 }
448 AcqStreamCore& c = *m_core;
449 std::lock_guard<ControlMutex> lock(c.controlMutex);
450 if (options.gating && c.gatingSlot.load(std::memory_order_acquire) >= 0) {
451 return AcqErrorCode::InvalidState;
452 }
453 for (std::size_t i = 0; i < c.maxReaders; ++i) {
454 ReaderSlot& slot = c.slots[i];
455 if (slot.inUse.load(std::memory_order_acquire)) {
456 continue;
457 }
458 slot.mode = mode;
459 slot.options = options;
460 slot.gating = options.gating;
461 slot.holding = false;
462 slot.pendingLost = 0;
463 slot.clockSeq = SeqlockValue<AcqClockModel>::kSeqNone;
464 slot.timebaseSeq = SeqlockValue<AcqTimebase>::kSeqNone;
465 slot.waiting.store(false, std::memory_order_relaxed);
466 std::uint64_t start = c.committed.load(std::memory_order_acquire);
467 if (options.gating && c.direction == AcqDirection::Output) {
468 start = c.gatePosition.load(std::memory_order_acquire);
469 } else if (options.startAtOldest) {
470 const std::uint64_t r = c.reserved.load(std::memory_order_acquire);
471 start = r > c.capacity ? r - c.capacity : 0;
472 }
473 slot.position.store(start, std::memory_order_relaxed);
474 slot.eventCursor = c.events.oldest();
475 slot.segmentCursor = c.segments.oldest();
476 for (auto* counter : {&slot.delivered, &slot.lost, &slot.skipped, &slot.idle, &slot.overruns, &slot.views,
477 &slot.eventsLost, &slot.segmentsLost}) {
478 counter->store(0, std::memory_order_relaxed);
479 }
480 if (options.gating) {
481 slot.commitCursor = c.outputCommits.published.load(std::memory_order_acquire);
482 slot.dropEnd = 0;
483 c.commitsValidated.store(slot.commitCursor, std::memory_order_seq_cst);
484 c.gatePosition.store(start, std::memory_order_seq_cst);
485 c.deviceGone.store(false, std::memory_order_seq_cst);
486 c.gatingSlot.store(static_cast<int>(i), std::memory_order_seq_cst);
487 }
488 slot.inUse.store(true, std::memory_order_release);
489 return AcqReader(m_core, static_cast<std::uint32_t>(i));
490 }
491 return AcqErrorCode::TooManyReaders;
492}

589// Output: moves the write position past samples the device already emitted as fill and accounts the underrun.
590std::uint64_t catchUpWithDevice(AcqStreamCore& c) noexcept
591{
592 const std::uint64_t idle = c.idleTotal.load(std::memory_order_acquire);
593 const std::uint64_t underrun = idle - c.idleSeen;
594 c.idleSeen = idle;
595 if (underrun > 0) {
596 c.underrun.fetch_add(underrun, std::memory_order_relaxed);
597 c.underruns.fetch_add(1, std::memory_order_relaxed);
598 }
599 const std::uint64_t gate = c.gatePosition.load(std::memory_order_acquire);
600 if (gate > c.writePos) {
601 c.fillRange(c.writePos, gate);
602 c.writePos = gate;
603 c.committed.store(gate, std::memory_order_seq_cst);
604 c.notifyReaders();
605 }
606 return underrun;
607}

609AcqError waitForGate(AcqStreamCore& c, std::uint64_t end, std::chrono::nanoseconds timeout) noexcept
610{
611 auto fits = [&c, end] {
612 const std::uint64_t gate = c.gatePosition.load(std::memory_order_seq_cst);
613 const bool logSpace = c.outputCommits.pushed - c.commitsValidated.load(std::memory_order_seq_cst) <
614 c.outputCommits.capacity();
615 return logSpace && (gate >= end || end - gate <= c.capacity);
616 };
617 if (fits()) {
618 return {};
619 }
620 if (c.deviceGone.load(std::memory_order_acquire)) {
621 return AcqErrorCode::EndOfStream;
622 }
623 c.writerWaits.fetch_add(1, std::memory_order_relaxed);
624 const auto deadline = std::chrono::steady_clock::now() + timeout;
625 auto ready = [&] {
626 return fits() || c.deviceGone.load(std::memory_order_acquire);
627 };
628 if (!waitFor(ready, c.writerWait, std::chrono::microseconds(50), deadline, c.writerWaiting, c.writerWake,
629 nullptr)) {
630 return AcqErrorCode::Timeout;
631 }
632 return fits() ? AcqError() : AcqError(AcqErrorCode::EndOfStream);
633}

635// Input: waits while a gating reader has not consumed what reserving up to @p end would overwrite.
636AcqError waitForInputSpace(AcqStreamCore& c, std::uint64_t end, std::chrono::nanoseconds timeout) noexcept
637{
638 if (!c.gated()) {
639 return {};
640 }
641 const AcqError e = waitForGate(c, end, timeout);
642 return e.code() == AcqErrorCode::EndOfStream ? AcqError() : e; // the gating reader left: nothing to wait for

657AcqExpected<AcqWriteView> AcqWriter::beginWrite(std::size_t samples, std::chrono::nanoseconds timeout) noexcept
658{
659 if (!m_core) {
660 return AcqErrorCode::InvalidState;
661 }
662 AcqStreamCore& c = *m_core;
663 if (samples > c.capacity) {
664 return AcqErrorCode::InvalidParameter;
665 }
666 std::uint64_t underrun = 0;
667 if (c.direction == AcqDirection::Output) {
668 underrun = catchUpWithDevice(c);
669 if (AcqError e = waitForGate(c, c.writePos + samples, timeout); !e.ok()) {
670 return e;
671 }
672 } else if (AcqError e = waitForInputSpace(c, c.writePos + samples, timeout); !e.ok()) {
673 return e;
674 }
675 return makeWriteView(c, c.writePos, samples, underrun);
676}

679{

681 return AcqErrorCode::InvalidState;
682 }
683 AcqStreamCore& c = *m_core;
684 if (samples > c.capacity) {
685 return AcqErrorCode::InvalidParameter;

687 if (c.direction == AcqDirection::Output) {
688 return waitForGate(c, std::max(c.writePos, c.gatePosition.load(std::memory_order_acquire)) + samples,

694AcqExpected<AcqWriteView> AcqWriter::beginWriteAt(std::uint64_t firstSample, std::size_t samples,
695 std::chrono::nanoseconds timeout) noexcept
696{
697 if (!m_core) {
698 return AcqErrorCode::InvalidState;
699 }
700 AcqStreamCore& c = *m_core;
701 if (c.direction != AcqDirection::Output) {
702 return AcqErrorCode::WrongDirection;
703 }
704 const std::uint32_t alignment = c.timebase.load().sampleAlignment;
705 if (alignment > 1 && firstSample % alignment != 0) {
706 return AcqErrorCode::NotOnGrid;
707 }
708 if (firstSample < c.gatePosition.load(std::memory_order_acquire)) {
709 return AcqErrorCode::Late;
710 }
711 // The device may have played fill far past writePos; the range check refers to the caught-up position.
712 const std::uint64_t underrun = catchUpWithDevice(c);
713 if (firstSample < c.writePos || firstSample + samples - c.writePos > c.capacity) {
714 return AcqErrorCode::InvalidParameter;
715 }
716 if (AcqError e = waitForGate(c, firstSample + samples, timeout); !e.ok()) {
717 return e;
718 }
719 c.fillRange(c.writePos, firstSample);
720 return makeWriteView(c, firstSample, samples, underrun);
721}

723AcqError AcqWriter::grant(std::size_t samples) noexcept
724{
725 if (!m_core) {
726 return AcqErrorCode::InvalidState;
727 }
728 AcqStreamCore& c = *m_core;
729 if (c.direction != AcqDirection::Input) {
730 return AcqErrorCode::WrongDirection;
731 }
732 const std::uint64_t end = std::max(c.grantedEnd, c.writePos) + samples;
733 if (end - c.writePos > c.capacity) {
734 return AcqErrorCode::InvalidParameter;
735 }
736 if (AcqError e = waitForInputSpace(c, end, std::chrono::nanoseconds(0)); !e.ok()) {
737 return e;
738 }
739 c.grantedEnd = end;
740 c.reserveUpTo(end);
741 return {};
742}

744void AcqWriter::commit(std::size_t samples) noexcept
745{
746 if (!m_core) {
747 return;
748 }
749 AcqStreamCore& c = *m_core;
750 if (c.pendingCount == 0 && c.grantedEnd > c.writePos) {
751 c.pendingFirst = c.writePos;
752 c.pendingCount = c.grantedEnd - c.writePos;
753 }
754 const std::uint64_t n = std::min<std::uint64_t>(samples, c.pendingCount);
755 const std::uint64_t end = c.pendingFirst + n;
756 c.pendingCount = 0;
757 if (end <= c.writePos && n == 0) {
758 return;
759 }
760 c.writePos = std::max(c.writePos, end);
761 if (c.direction == AcqDirection::Output && n > 0) {
762 // Published before the samples, so the device sees the scheduled start of every block it can play.
763 pushRecord(c.outputCommits, OutputCommit{c.pendingFirst, end}, {}, true);
764 }
765 if (c.direction == AcqDirection::Input) {
766 c.events.published.store(c.events.pushed, std::memory_order_release);
767 c.segments.published.store(c.segments.pushed, std::memory_order_release);
768 }
769 c.committed.store(c.writePos, std::memory_order_seq_cst);
770 addOwned(c.commits, 1);
771 c.notifyReaders();
772}

774AcqError AcqWriter::pushEvent(const AcqEvent& event) noexcept
775{
776 if (!m_core) {
777 return AcqErrorCode::InvalidState;
778 }
779 AcqStreamCore& c = *m_core;
780 const bool output = c.direction == AcqDirection::Output;
781 if (output && event.sample < c.gatePosition.load(std::memory_order_acquire)) {
782 c.lateEvents.fetch_add(1, std::memory_order_relaxed);
783 return AcqErrorCode::Late;
784 }
785 if (c.events.pushed > 0 && event.sample < c.lastEventSample) {
786 return AcqErrorCode::InvalidParameter;
787 }
788 c.lastEventSample = event.sample;
789 pushRecord(c.events, event, {}, output);
790 return {};
791}

819AcqError AcqWriter::updateInfo(AcqStreamInfo info, std::uint64_t fromSample)
820{
821 if (!m_core) {
822 return AcqErrorCode::InvalidState;
823 }
824 AcqStreamCore& c = *m_core;
825 std::lock_guard<ControlMutex> lock(c.controlMutex);
826 const std::size_t count = c.versionCount.load(std::memory_order_acquire);
827 const AcqStreamInfo& current = *c.versions[count - 1]->info;
828 if (fromSample < c.writePos || fromSample < c.versions[count - 1]->validFrom ||
829 info.channels.size() != current.channels.size() || info.format != current.format ||
830 info.layout != current.layout || info.direction != current.direction) {
831 return AcqError(AcqErrorCode::InvalidParameter, "updateInfo may not change the ring geometry or the past");
