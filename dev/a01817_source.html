56DataLoader::SensorLoadResult DataLoader::loadSensors(const std::filesystem::path& fifPath,
57 const std::filesystem::path& megHelmetOverridePath)
58{
59 SensorLoadResult result;
60
61 std::error_code error;
62 if (!std::filesystem::exists(fifPath, error))
63 return result;
64
65 FiffInfo info;
66 FiffDigPointSet digSet;
67 FiffStream stream(fifPath);
68
69 if (stream.open()) {
70 std::shared_ptr<FiffDirNode> tree = stream.dirtree();
71 std::shared_ptr<FiffDirNode> nodeInfo;
72 if (stream.read_meas_info(tree, info, nodeInfo)) {
73 result.hasInfo = !info.empty();
74 }
75 }
76
77 if (result.hasInfo) {
78 result.info = info;
79
80 // Prepare Device->Head transformation
81 QMatrix4x4 devHeadQTrans;
82 bool hasDevHead = false;
83 if (!info.dev_head_t.empty() &&
84 info.dev_head_t.from == FIFFV_COORD_DEVICE &&
85 info.dev_head_t.to == FIFFV_COORD_HEAD &&
86 !info.dev_head_t.trans.isIdentity()) {
87 hasDevHead = true;
88 devHeadQTrans = mne::disp::surface_keys::toQMatrix4x4(info.dev_head_t.trans);
89 }
90
91 result.devHeadTrans = devHeadQTrans;
92 result.hasDevHead = hasDevHead;
93
94 for (const auto& ch : info.chs) {
95 if (ch.kind == FIFFV_MEG_CH) {
96 QVector3D pos(ch.chpos.r0(0), ch.chpos.r0(1), ch.chpos.r0(2));
97
98 if (hasDevHead) {
99 pos = devHeadQTrans.map(pos);
100 }
101
102 auto item = std::make_unique<SensorTreeItem>(QString::fromStdString(ch.ch_name), pos, QColor(100, 100, 100), 0.01f);
103
104 // Store coil orientation
105 QMatrix4x4 orient;
106 for (int r = 0; r < 3; ++r)
107 for (int c = 0; c < 3; ++c)
108 orient(r, c) = ch.coil_trans(r, c);
109 if (hasDevHead) {
110 QMatrix4x4 devHeadRot;
111 for (int r = 0; r < 3; ++r)
112 for (int c = 0; c < 3; ++c)
113 devHeadRot(r, c) = devHeadQTrans(r, c);
114 orient = devHeadRot * orient;
115 }
116 item->setOrientation(orient);
117
118 if (ch.unit == FIFF_UNIT_T_M) {
119 result.megGradItems.push_back(std::move(item));
120 } else {
121 result.megMagItems.push_back(std::move(item));
122 }
123 } else if (ch.kind == FIFFV_EEG_CH) {
124 QVector3D pos(ch.chpos.r0(0), ch.chpos.r0(1), ch.chpos.r0(2));
125 result.eegItems.push_back(std::make_unique<SensorTreeItem>(QString::fromStdString(ch.ch_name), pos, QColor(0, 200, 220), 0.002f));
126 }
127 }
128
129 // ── MEG helmet surface ──────────────────────────────────────
130 if (!result.megGradItems.empty() || !result.megMagItems.empty()) {
131 auto pickHelmetFile = [&info]() -> std::filesystem::path {
132 int coilType = -1;
133 int nMeg = 0;
134 for (const auto& ch : info.chs) {
135 if (ch.kind == FIFFV_MEG_CH) {
136 coilType = ch.chpos.coil_type & 0xFFFF;
137 ++nMeg;
138 }
139 }
140
141 std::string fileName = "306m.fif";
142 if (coilType == FIFFV_COIL_BABY_GRAD) {
143 fileName = "BabySQUID.fif";
144 } else if (coilType == FIFFV_COIL_NM_122) {
145 fileName = "122m.fif";
146 } else if (coilType == FIFFV_COIL_CTF_GRAD) {
147 fileName = "CTF_275.fif";
148 } else if (coilType == FIFFV_COIL_KIT_GRAD) {
149 fileName = "KIT.fif";
150 } else if (coilType == FIFFV_COIL_MAGNES_MAG || coilType == FIFFV_COIL_MAGNES_GRAD) {
151 fileName = (nMeg > 150) ? "Magnes_3600wh.fif" : "Magnes_2500wh.fif";
152 } else if (coilType / 1000 == 3) {
153 fileName = "306m.fif";
154 }
155
156 return mne::executable_dir() / ".." / "resources" / "general" / "sensorSurfaces" / fileName;
157 };
158
159 const auto fileExists = [](const std::filesystem::path& path) {
160 std::error_code existsError;
161 return !path.empty() && std::filesystem::exists(path, existsError);
162 };
163
164 std::filesystem::path helmetPath;
165 if (!megHelmetOverridePath.empty()) {
166 helmetPath = megHelmetOverridePath;
167 if (!fileExists(helmetPath)) {
168 mne::Log::warning("MEG helmet override file not found: {} - falling back to auto selection.", mne::utf8_from_path(helmetPath));
169 helmetPath.clear();
170 }
171 }
172
173 if (helmetPath.empty()) {
174 helmetPath = pickHelmetFile();
175#ifdef __EMSCRIPTEN__
176 // WASM: the executable directory may not resolve correctly.
177 // Try the preloaded virtual FS path directly.
178 if (!fileExists(helmetPath)) {
179 std::filesystem::path fn = helmetPath.filename();
180 if (fn.empty())
181 fn = "306m.fif";
182 helmetPath = std::filesystem::path("/resources/general/sensorSurfaces") / fn;
183 }
184#endif
185 }
186
187 if (!fileExists(helmetPath)) {
188 const std::filesystem::path fallback = mne::executable_dir() / ".." / "resources" / "general" / "sensorSurfaces" / "306m.fif";
189 if (fileExists(fallback)) {
190 helmetPath = fallback;
191 }
192 }
193
194#ifdef __EMSCRIPTEN__
195 // WASM fallback: sensor surfaces are preloaded into the
196 // Emscripten virtual FS at /resources/general/sensorSurfaces/.
197 // The executable directory may not resolve the /../ correctly.
198 if (!fileExists(helmetPath)) {
199 std::filesystem::path fn = helmetPath.filename();
200 if (fn.empty())
201 fn = "306m.fif";
202 const std::filesystem::path wasmFallback = std::filesystem::path("/resources/general/sensorSurfaces") / fn;
203 if (fileExists(wasmFallback)) {
204 helmetPath = wasmFallback;
205 }
206 }
207#endif
208
209 if (!fileExists(helmetPath)) {
210 mne::Log::warning("MEG helmet surface file not found. Checked: {}", mne::utf8_from_path(helmetPath));
211 } else {
212 MNEBem helmetBem(helmetPath);
213 if (helmetBem.size() > 0) {
214 MNEBemSurface helmetSurf = helmetBem[0];
215 if (helmetSurf.nn.rows() != helmetSurf.rr.rows()) {
216 helmetSurf.nn = mne::FsSurface::compute_normals(Eigen::MatrixX3f(helmetSurf.rr), Eigen::MatrixX3i(helmetSurf.itris));
217 }
218
219 if (hasDevHead) {
220 QMatrix3x3 normalMat = devHeadQTrans.normalMatrix();
221 for (int i = 0; i < helmetSurf.rr.rows(); ++i) {
222 QVector3D pos(helmetSurf.rr(i, 0), helmetSurf.rr(i, 1), helmetSurf.rr(i, 2));
223 pos = devHeadQTrans.map(pos);
224 helmetSurf.rr(i, 0) = pos.x();
225 helmetSurf.rr(i, 1) = pos.y();
226 helmetSurf.rr(i, 2) = pos.z();
227
228 QVector3D nn(helmetSurf.nn(i, 0), helmetSurf.nn(i, 1), helmetSurf.nn(i, 2));
229 const float* d = normalMat.constData();
230 float nx = d[0] * nn.x() + d[3] * nn.y() + d[6] * nn.z();
231 float ny = d[1] * nn.x() + d[4] * nn.y() + d[7] * nn.z();
232 float nz = d[2] * nn.x() + d[5] * nn.y() + d[8] * nn.z();
233 QVector3D n = QVector3D(nx, ny, nz).normalized();
234 helmetSurf.nn(i, 0) = n.x();
235 helmetSurf.nn(i, 1) = n.y();
236 helmetSurf.nn(i, 2) = n.z();
237 }
238 }
239
240 auto helmetSurface = std::make_shared<BrainSurface>();
241 helmetSurface->fromBemSurface(helmetSurf, QColor(0, 0, 77, 200));
242 helmetSurface->setVisible(true);
243 result.helmetSurface = helmetSurface;
244 } else {
245 mne::Log::warning("DataLoader::loadSensors: helmetBem[0] has 0 verts/tris!");
246 }
247 }
248 }
249 }
250
251 // ── Digitizer points ────────────────────────────────────────────
252 if (result.hasInfo && info.dig.size() > 0) {
253 result.digitizerPoints = info.dig;
254 result.hasDigitizer = true;
255 } else if (!result.hasInfo) {
256 if (auto dig = FiffDigPointSet::read(fifPath); dig && !dig->empty()) {
257 result.digitizerPoints = std::move(*dig);
258 result.hasDigitizer = true;
259 }
260 }
261
262 return result;
263}

267std::shared_ptr<BrainSurface> DataLoader::loadHelmetSurface(
268 const std::filesystem::path& helmetFilePath,
269 const QMatrix4x4& devHeadTrans,
270 bool applyTrans)
271{
272 std::error_code error;
273 if (!std::filesystem::exists(helmetFilePath, error)) {
274 mne::Log::warning("DataLoader::loadHelmetSurface: file not found: {}", mne::utf8_from_path(helmetFilePath));
275 return nullptr;
276 }
277
278 MNEBem helmetBem(helmetFilePath);
279 if (helmetBem.size() == 0) {
280 mne::Log::warning("DataLoader::loadHelmetSurface: no BEM surfaces in {}", mne::utf8_from_path(helmetFilePath));
281 return nullptr;
282 }
283
284 MNEBemSurface helmetSurf = helmetBem[0];
285
286 if (helmetSurf.nn.rows() != helmetSurf.rr.rows()) {
287 helmetSurf.nn = mne::FsSurface::compute_normals(Eigen::MatrixX3f(helmetSurf.rr), Eigen::MatrixX3i(helmetSurf.itris));
288 }
289
290 if (applyTrans) {
291 QMatrix3x3 normalMat = devHeadTrans.normalMatrix();
292 for (int i = 0; i < helmetSurf.rr.rows(); ++i) {
293 QVector3D pos(helmetSurf.rr(i, 0), helmetSurf.rr(i, 1), helmetSurf.rr(i, 2));
294 pos = devHeadTrans.map(pos);
295 helmetSurf.rr(i, 0) = pos.x();
296 helmetSurf.rr(i, 1) = pos.y();
297 helmetSurf.rr(i, 2) = pos.z();
298
299 QVector3D nn(helmetSurf.nn(i, 0), helmetSurf.nn(i, 1), helmetSurf.nn(i, 2));
300 const float* d = normalMat.constData();
301 float nx = d[0] * nn.x() + d[3] * nn.y() + d[6] * nn.z();
302 float ny = d[1] * nn.x() + d[4] * nn.y() + d[7] * nn.z();
303 float nz = d[2] * nn.x() + d[5] * nn.y() + d[8] * nn.z();
304 QVector3D n = QVector3D(nx, ny, nz).normalized();
305 helmetSurf.nn(i, 0) = n.x();
306 helmetSurf.nn(i, 1) = n.y();
307 helmetSurf.nn(i, 2) = n.z();
308 }
309 }
310
311 auto surface = std::make_shared<BrainSurface>();
312 surface->fromBemSurface(helmetSurf, QColor(0, 0, 77, 200));
313 surface->setVisible(true);
314
315 return surface;
316}

375std::vector<std::string> DataLoader::probeEvokedSets(const std::filesystem::path& evokedPath)
376{
377 std::vector<std::string> result;
378 std::error_code error;
379 if (!std::filesystem::exists(evokedPath, error)) {
380 return result;
381 }
382
383 const auto read = FiffEvokedSet::read(evokedPath);
384 if (!read) {
385 return result;
386 }
387 const FiffEvokedSet& evokedSet = *read;
388 for (int i = 0; i < std::ssize(evokedSet.evoked); ++i) {
389 const auto& ev = evokedSet.evoked.at(i);
390 result.push_back(std::format("{}: {} ({}, nave={})",
391 i,
392 ev.comment.empty() ? std::format("Set {}", i) : ev.comment,
393 ev.aspectKindToString(),
394 ev.nave));
395 }
396 return result;
397}
