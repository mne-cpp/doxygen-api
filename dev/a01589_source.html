70void MlCsp::fit(const QList<MatrixXd>& epochsClass1,
71 const QList<MatrixXd>& epochsClass2)
72{
73 if (epochsClass1.isEmpty() || epochsClass2.isEmpty()) {
74 qWarning() << "[MlCsp::fit] Empty epoch lists.";
75 return;
76 }
77
78 // Convert QList → std::vector for skigen
79 std::vector<MatrixXd> e1(epochsClass1.cbegin(), epochsClass1.cend());
80 std::vector<MatrixXd> e2(epochsClass2.cbegin(), epochsClass2.cend());
81
82 Skigen::CSP<double> csp(m_nComponents);
83 csp.fit(e1, e2);
84
85 m_filters = csp.filters();
86 m_patterns = csp.patterns();
87 m_eigenvalues = csp.eigenvalues();
88 m_bFitted = true;
89}

93MatrixXd MlCsp::transform(const QList<MatrixXd>& epochs) const
94{
95 if (!m_bFitted) {
96 qWarning() << "[MlCsp::transform] CSP not fitted.";
97 return MatrixXd();
98 }
99
100 // Delegate to skigen CSP (reuse fitted filters stored in m_filters)
101 // We reconstruct a fitted CSP instance by directly computing features
102 // using the stored filters, matching the skigen transform logic.
103 std::vector<MatrixXd> e(epochs.cbegin(), epochs.cend());
104
105 Skigen::CSP<double> csp(m_nComponents);
106 // Use the stored filters directly rather than re-fitting
107 const int nEpochs = static_cast<int>(e.size());
108 const int nComp = static_cast<int>(m_filters.rows());
109 MatrixXd features(nEpochs, nComp);
110
111 for (int ep = 0; ep < nEpochs; ++ep) {
112 MatrixXd filtered = m_filters * e[static_cast<size_t>(ep)];
113 for (int c = 0; c < nComp; ++c) {
114 double var = filtered.row(c).squaredNorm()
115 / static_cast<double>(filtered.cols());
116 features(ep, c) = std::log(var);
117 }
118 }
119
120 // Normalize: subtract per-epoch mean for scale invariance
121 for (int ep = 0; ep < nEpochs; ++ep) {
122 features.row(ep).array() -= features.row(ep).mean();
123 }
124
125 return features;
126}
