35double ViewAnimator::ease(Easing easing, double t)
36{
37 t = std::clamp(t, 0.0, 1.0);
38 if (easing == Easing::Linear) {
39 return t;
40 }
41 const double r = 1.0 - t;
42 return 1.0 - r * r * r;
43}

47double ViewAnimator::Animation::value_at(std::int64_t now_ms) const
48{
49 if (finished_at(now_ms)) {
50 return to;
51 }
52 const double t = static_cast<double>(now_ms - start_ms) / duration_ms;
53 return from + (to - from) * ease(easing, t);
54}

79bool ViewAnimator::tick(std::int64_t now_ms, TraceView& view)
80{
81 if (m_zoom) {
82 const Animation zoom = *m_zoom;
83 if (zoom.finished_at(now_ms)) {
84 m_zoom.reset();
85 }
86 view.set_samples_per_pixel(zoom.value_at(now_ms));
87 }
88 if (m_scroll) {
89 const Animation scroll = *m_scroll;
90 const bool finished = scroll.finished_at(now_ms);
91 if (finished) {
92 m_scroll.reset();
93 }
94 // Setting the value may report effects that start a new animation; it then stays
95 view.set_scroll_sample(scroll.value_at(now_ms));
96 if (finished && !scroll.inertial && !m_scroll) {
97 view.scroll_animation_finished();
98 }
99 }
100 return running();
101}
