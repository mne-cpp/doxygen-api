65public:
66 explicit NeuromagSimulator(const NeuromagSimulation& simulation)
67 : m_simulation(simulation)
68 {
69 for (std::size_t i = 0; i < simulation.meg; ++i) {
70 m_channels.push_back(channel(std::format("MEG {:04}", i + 1), fiff_kind::kMeg, kCoilMagnetometer, fiff_unit::kTesla, kMegCal));
71 }
72 for (std::size_t i = 0; i < simulation.eeg; ++i) {
73 m_channels.push_back(channel(std::format("EEG {:03}", i + 1), fiff_kind::kEeg, kCoilEeg, fiff_unit::kVolt, kEegCal));
74 }
75 m_channels.push_back(channel("STI 014", fiff_kind::kStim, 0, fiff_unit::kNone, 1.0F));
76 }

88 AcqError start() override
89 {
90 // The session restarts at sample 0, as a new measurement does
91 std::string uri = std::format("synthetic://?sfreq={}&stim=1&meg={}&eeg={}&block={}&seed={}&realtime={}", m_simulation.rate,
92 m_simulation.meg, m_simulation.eeg, m_simulation.buffer, m_simulation.seed, m_simulation.realtime ? 1 : 0);
93 if (m_simulation.duration > 0.0) {
94 uri += std::format("&duration={}", m_simulation.duration);
95 }
96 auto session = AcqDevice::open(uri);
97 if (!session) {
98 return session.error();
99 }
100 AcqReaderOptions gating;
101 gating.gating = true; // the session waits for this reader instead of overwriting unread samples
102 auto reader = session->stream().reader(AcqReader::ReadMode::Lossless, gating);
103 if (!reader) {
104 return reader.error();
105 }
106 if (AcqError e = session->start(); !e.ok()) {
107 return e;
108 }
109 m_session = std::move(*session);
110 m_reader = std::move(*reader);
111 m_pending = measurement_start();
112 m_next = 0;
113 return {};
114 }
115
116 void stop() override
117 {
118 m_reader = AcqReader();
119 m_session.close();
120 m_pending.clear();
121 }

123 AcqExpected<const NeuromagTag*> next(std::chrono::milliseconds timeout) override
124 {
125 if (m_next < m_pending.size()) {
126 return &m_pending[m_next++];
127 }
128 const std::size_t samples = m_simulation.buffer;
129 auto view = m_reader.acquire(samples, timeout, samples);
130 if (!view) {
131 return view.error();
132 }
133 if (view->sampleCount() < samples) {
134 // The server sends whole buffers only; the end of the session cuts the rest
135 static_cast<void>(m_reader.release(*view));
136 return AcqError(AcqErrorCode::EndOfStream);
137 }
138 // The session's channels are EEG, MEG, then the trigger code; the server's MEG, EEG, then STI 014
139 const std::size_t meg = m_simulation.meg;
140 const std::size_t eeg = m_simulation.eeg;
141 const std::size_t width = m_channels.size();
142 m_buffer.kind = fiff::kDataBuffer;
143 m_buffer.type = fiff::kTypeInt;
144 m_buffer.data.resize(samples * width * sizeof(std::int32_t));
145 auto* values = reinterpret_cast<std::int32_t*>(m_buffer.data.data());
146 std::size_t s = 0;
147 for (std::size_t p = 0; p < view->partCount(); ++p) {
148 const AcqBlockPart& part = view->part(p);
149 for (std::size_t i = 0; i < part.samples; ++i, ++s) {
150 std::int32_t* frame = values + s * width;
151 for (std::size_t c = 0; c < width; ++c) {
152 const std::size_t source = c < meg ? eeg + c : c < meg + eeg ? c - meg
153 : meg + eeg;
154 float v = 0.0F;
155 std::memcpy(&v, part.at(source, i), sizeof(v));
156 const double physical = c + 1 < width ? m_simulation.gain * v : v;
157 frame[c] = encode(physical / static_cast<double>(m_channels[c].cal));
158 }
159 }
160 }
161 if (AcqError e = m_reader.release(*view); !e.ok()) {
162 return e;
163 }
164 return &m_buffer;
165 }

180 static std::int32_t encode(double raw) noexcept
181 {
182 constexpr double kLow = std::numeric_limits<std::int32_t>::min();
183 constexpr double kHigh = std::numeric_limits<std::int32_t>::max();
184 return static_cast<std::int32_t>(std::clamp(std::nearbyint(raw), kLow, kHigh));

188 std::vector<NeuromagTag> measurement_start()
189 {
190 std::vector<NeuromagTag> tags;
191 tags.push_back(tag(fiff::kBlockStart, fiff::kTypeInt, bytes_of<std::int32_t>(fiff::kBlockMeasInfo)));
192 tags.push_back(tag(fiff::kNchan, fiff::kTypeInt, bytes_of<std::int32_t>(static_cast<std::int32_t>(m_channels.size()))));
193 tags.push_back(tag(fiff::kSfreq, fiff::kTypeFloat, bytes_of<float>(static_cast<float>(m_simulation.rate))));
194 tags.push_back(tag(fiff::kLowpass, fiff::kTypeFloat, bytes_of<float>(static_cast<float>(m_simulation.rate / 3.0))));
195 tags.push_back(tag(fiff::kHighpass, fiff::kTypeFloat, bytes_of<float>(0.1F)));
196 for (std::size_t c = 0; c < m_channels.size(); ++c) {
197 NeuromagChannel ch = m_channels[c];
198 ch.scan_no = static_cast<std::int32_t>(c + 1);
199 ch.log_no = static_cast<std::int32_t>(c + 1);
200 tags.push_back(tag(fiff::kChInfo, fiff::kTypeChInfo, ch.serialize()));
201 }
202 tags.push_back(tag(fiff::kBlockEnd, fiff::kTypeInt, bytes_of<std::int32_t>(fiff::kBlockMeasInfo)));
203 tags.push_back(tag(fiff::kBlockStart, fiff::kTypeInt, bytes_of<std::int32_t>(fiff::kBlockRawData)));
204 return tags;
205 }
