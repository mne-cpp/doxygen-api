48RtHpi::RtHpi(std::shared_ptr<FiffInfo> info, Settings settings)

66std::optional<HpiFitResult> RtHpi::update(const MatrixXd& data)
67{
68 if (data.rows() != m_ring.rows()) {
69 Log::warning("[RtHpi::update] Ignoring a block of {} channels; the measurement info has {}.", data.rows(), m_ring.rows());
70 return std::nullopt;
71 }
72 const Index window = m_ring.cols();
73 const Index n = data.cols();
74 const Index kept = std::min(n, window);
75 for (Index done = 0; done < kept;) {
76 const Index chunk = std::min(kept - done, window - m_next);
77 m_ring.middleCols(m_next, chunk) = data.middleCols(n - kept + done, chunk);
78 m_next = (m_next + chunk) % window;
79 done += chunk;
80 }
81 m_filled = std::min(window, m_filled + n);
82 m_sinceFit += n;
83 if (m_filled < window || m_sinceFit < m_settings.hop) {
84 return std::nullopt;
85 }
86 if (m_settings.coil_frequencies.size() < 3) {
87 Log::warning("[RtHpi::update] At least three coil frequencies are needed to fit the head position.");
88 return std::nullopt;
89 }
90 m_sinceFit = 0;
91
92 // The oldest sample sits at m_next
93 MatrixXd samples(m_ring.rows(), window);
94 samples.leftCols(window - m_next) = m_ring.rightCols(window - m_next);
95 samples.rightCols(m_next) = m_ring.leftCols(m_next);
96 m_updater.prepareDataAndProjectors(samples, m_projectors);
97
98 const InvHpiModelParameters model(m_settings.coil_frequencies, m_info->sfreq, m_info->linefreq, false);
99 HpiFitResult result = m_last;
100 m_fit.fit(m_updater.getProjectedData(), m_updater.getProjectors(), model, m_updater.getHpiDigitizer(), std::exchange(m_order, false), result);
101 if (!result.hpiFreqs.empty()) {
102 m_settings.coil_frequencies = result.hpiFreqs;
103 }
104 if (accepted(result)) {
105 m_last = result;
106 }
107 return result;
108}

112bool RtHpi::accepted(const HpiFitResult& fit) const
113{
114 if (fit.errorDistances.empty()) {
115 return false;
116 }
117 const double mean = std::accumulate(fit.errorDistances.begin(), fit.errorDistances.end(), 0.0) / static_cast<double>(fit.errorDistances.size());
118 return mean < m_settings.max_error;
119}
