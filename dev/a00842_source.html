65std::expected<TcpServer, std::error_code> TcpServer::listen(const std::string& address, std::uint16_t port)
66{
67 if (!started()) {
68 return std::unexpected(std::make_error_code(std::errc::not_supported));
69 }
70 sockaddr_in local{};
71 local.sin_family = AF_INET;
72 local.sin_port = htons(port);
73 if (::inet_pton(AF_INET, address.c_str(), &local.sin_addr) != 1) {
74 return std::unexpected(std::make_error_code(std::errc::invalid_argument));
75 }
76 const Native s = ::socket(AF_INET, SOCK_STREAM, 0);
77 if (s == kInvalid) {
78 return std::unexpected(last_error());
79 }
80 TcpServer server;
81 server.m_socket = static_cast<Handle>(s);
82#ifndef _WIN32
83 const int on = 1; // a restarted server gets its port back while old connections linger
84 ::setsockopt(s, SOL_SOCKET, SO_REUSEADDR, &on, sizeof(on));
85#endif
86 if (::bind(s, reinterpret_cast<const sockaddr*>(&local), sizeof(local)) != 0 || ::listen(s, SOMAXCONN) != 0 ||
87 !set_non_blocking(s)) {
88 return std::unexpected(last_error());
89 }
90 return server;
91}

107std::expected<TcpSocket, std::error_code> TcpServer::accept(std::chrono::milliseconds timeout)
108{
109 const auto s = static_cast<Native>(m_socket);
110 while (true) {
111 const int ready = wait_for(s, POLLIN, timeout);
112 if (ready == 0) {
113 return std::unexpected(std::make_error_code(std::errc::timed_out));
114 }
115 if (ready < 0) {
116 return std::unexpected(last_error());
117 }
118 const Native client = ::accept(s, nullptr, nullptr);
119 if (client == kInvalid) {
120 const std::error_code error = last_error();
121 if (would_block(error)) {
122 continue; // the client gave up between poll and accept
123 }
124 return std::unexpected(error);
125 }
126 TcpSocket socket;
127 socket.m_socket = static_cast<Handle>(client);
128 if (!set_non_blocking(client)) {
129 return std::unexpected(last_error());
130 }
131#ifdef SO_NOSIGPIPE
132 int no_signal = 1;
133 ::setsockopt(client, SOL_SOCKET, SO_NOSIGPIPE, &no_signal, sizeof(no_signal));
134#endif
135 int on = 1;
136 ::setsockopt(client, IPPROTO_TCP, TCP_NODELAY, reinterpret_cast<const char*>(&on), sizeof(on));
137 return socket;
138 }
