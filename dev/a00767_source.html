102int InvDipoleFitData::setup_forward_model(InvDipoleFitData* d, MNECTFCompDataSet* comp_data, FwdCoilSet* comp_coils)
103{
104 dipoleFitFuncsRec* f;
105 int fit_sphere_to_bem = true;
106
107 if (!d->bemname.empty()) {
108 /*
109 * Set up the boundary-element model
110 */
111 d->bemname = mne::utf8_from_path(FwdBemModel::fwd_bem_make_bem_sol_name(mne::path_from_utf8(d->bemname)));
112
113 mne::Log::info("\nSetting up the BEM model using {}...", d->bemname);
114 mne::Log::info("\nLoading surfaces...");
115 d->bem_model = FwdBemModel::fwd_bem_load_three_layer_surfaces(mne::path_from_utf8(d->bemname));
116 if (d->bem_model) {
117 mne::Log::info("Three-layer model surfaces loaded.");
118 } else {
119 d->bem_model = FwdBemModel::fwd_bem_load_homog_surface(mne::path_from_utf8(d->bemname));
120 if (!d->bem_model)
121 return FAIL;
122 mne::Log::info("Homogeneous model surface loaded.");
123 }
124 if (d->neeg > 0 && d->bem_model->nsurf == 1) {
125 mne::Log::critical("Cannot use a homogeneous model in EEG calculations.");
126 return FAIL;
127 }
128 mne::Log::info("\nLoading the solution matrix...");
129 if (d->bem_model->fwd_bem_load_recompute_solution(mne::path_from_utf8(d->bemname), FWD_BEM_UNKNOWN, false) == FAIL)
130 return FAIL;
131 mne::Log::info("Employing the head->MRI coordinate transform with the BEM model.");
132 assert(d->mri_head_t);
133 if (d->bem_model->fwd_bem_set_head_mri_t(*d->mri_head_t) == FAIL)
134 return FAIL;
135 mne::Log::info("BEM model {} is now set up", d->bem_model->sol_name);
136 /*
137 * Find the best-fitting sphere
138 */
139 if (fit_sphere_to_bem) {
140 MNESurface* inner_skull;
141 float simplex_size = 2e-2f;
142 float R;
143 VectorXf r0_vec;
144
145 if ((inner_skull = d->bem_model->fwd_bem_find_surface(FIFFV_BEM_SURF_ID_BRAIN)) == nullptr)
146 return FAIL;
147
148 if (!mne::Sphere::fit_sphere_to_points(inner_skull->rr, simplex_size, r0_vec, R))
149 return FAIL;
150 d->r0 = r0_vec.head<3>();
151
152 FiffCoordTrans::apply_trans(d->r0.data(), *d->mri_head_t, true);
153 mne::Log::info("Fitted sphere model origin : {:6.1f} {:6.1f} {:6.1f} mm rad = {:6.1f} mm.", 1000 * d->r0[0], 1000 * d->r0[1], 1000 * d->r0[2], 1000 * R);
154 }
155 d->bem_funcs = std::make_unique<dipoleFitFuncsRec>();
156 f = d->bem_funcs.get();
157 if (d->nmeg > 0) {
158 /*
159 * Use the new compensated field computation
160 * It works the same way independent of whether or not the compensation is in effect
161 */
162 auto comp = FwdCompData::fwd_make_comp_data(comp_data, d->meg_coils.get(), comp_coils,
163 FwdBemModel::fwd_bem_field, nullptr, nullptr, d->bem_model.get());
164 if (!comp)
165 return FAIL;
166 mne::Log::info("Compensation setup done.");
167
168 mne::Log::info("MEG solution matrix...");
169 if (d->bem_model->fwd_bem_specify_coils(d->meg_coils.get()) == FAIL)
170 return FAIL;
171 if (d->bem_model->fwd_bem_specify_coils(comp->comp_coils.get()) == FAIL)
172 return FAIL;
173 mne::Log::info("[done]");
174
175 f->meg_field = FwdCompData::fwd_comp_field;
176 f->meg_vec_field = nullptr;
177 f->meg_client = comp.get();
178 f->meg_comp = std::move(comp);
179 }
180 if (d->neeg > 0) {
181 mne::Log::info("\tEEG solution matrix...");
182 if (d->bem_model->fwd_bem_specify_els(d->eeg_els.get()) == FAIL)
183 return FAIL;
184 mne::Log::info("[done]");
185 f->eeg_pot = FwdBemModel::fwd_bem_pot_els;
186 f->eeg_vec_pot = nullptr;
187 f->eeg_client = d->bem_model.get();
188 }
189 }
190 if (d->neeg > 0 && !d->eeg_model) {
191 mne::Log::critical("EEG sphere model not defined.");
192 return FAIL;
193 }
194 d->sphere_funcs = std::make_unique<dipoleFitFuncsRec>();
195 f = d->sphere_funcs.get();
196 if (d->neeg > 0) {
197 d->eeg_model->r0 = d->r0;
198 f->eeg_pot = FwdEegSphereModel::fwd_eeg_spherepot_coil;
199 f->eeg_vec_pot = FwdEegSphereModel::fwd_eeg_spherepot_coil_vec;
200 f->eeg_client = d->eeg_model.get();
201 }
202 if (d->nmeg > 0) {
203 /*
204 * Use the new compensated field computation
205 * It works the same way independent of whether or not the compensation is in effect
206 */
207 auto comp = FwdCompData::fwd_make_comp_data(comp_data, d->meg_coils.get(), comp_coils,
208 FwdBemModel::fwd_sphere_field,
209 FwdBemModel::fwd_sphere_field_vec,
210 nullptr,
211 d->r0.data());
212 if (!comp)
213 return FAIL;
214 f->meg_field = FwdCompData::fwd_comp_field;
215 f->meg_vec_field = FwdCompData::fwd_comp_field_vec;
216 f->meg_client = comp.get();
217 f->meg_comp = std::move(comp);
218 }
219 mne::Log::info("Sphere model origin : {:6.1f} {:6.1f} {:6.1f} mm.", 1000 * d->r0[0], 1000 * d->r0[1], 1000 * d->r0[2]);
220 /*
221 * Finally add the magnetic dipole fitting functions (for special purposes)
222 */
223 d->mag_dipole_funcs = std::make_unique<dipoleFitFuncsRec>();
224 f = d->mag_dipole_funcs.get();
225 if (d->nmeg > 0) {
226 /*
227 * Use the new compensated field computation
228 * It works the same way independent of whether or not the compensation is in effect
229 */
230 auto comp = FwdCompData::fwd_make_comp_data(comp_data, d->meg_coils.get(), comp_coils,
231 FwdBemModel::fwd_mag_dipole_field,
232 FwdBemModel::fwd_mag_dipole_field_vec,
233 nullptr,
234 nullptr);
235 if (!comp)
236 return FAIL;
237 f->meg_field = FwdCompData::fwd_comp_field;
238 f->meg_vec_field = FwdCompData::fwd_comp_field_vec;
239 f->meg_client = comp.get();
240 f->meg_comp = std::move(comp);
241 }
242 f->eeg_pot = FwdBemModel::fwd_mag_dipole_field;
243 f->eeg_vec_pot = FwdBemModel::fwd_mag_dipole_field_vec;
244 /*
245 * Select the appropriate fitting function
246 */
247 d->funcs = !d->bemname.empty() ? d->bem_funcs.get() : d->sphere_funcs.get();
248
249 return OK;
250}

257std::unique_ptr<MNECovMatrix> InvDipoleFitData::ad_hoc_noise(FwdCoilSet* meg, FwdCoilSet* eeg, float grad_std, float mag_std, float eeg_std)
258{
259 int nchan;
260 Eigen::VectorXd stds;
261 StringList names, ch_names;
262 int k, n;
263
264 mne::Log::info("Using standard noise values (MEG grad : {:6.1f} fT/cm MEG mag : {:6.1f} fT EEG : {:6.1f} uV)\n", 1e13 * grad_std, 1e15 * mag_std, 1e6 * eeg_std);
265
266 nchan = 0;
267 if (meg)
268 nchan = nchan + meg->ncoil();
269 if (eeg)
270 nchan = nchan + eeg->ncoil();
271
272 stds.resize(nchan);
273
274 n = 0;
275 if (meg) {
276 for (k = 0; k < meg->ncoil(); k++, n++) {
277 if (meg->coils[k]->is_axial_coil()) {
278 stds[n] = static_cast<double>(mag_std) * mag_std;
279#ifdef TEST_REF
280 if (meg->coils[k]->type == FIFFV_COIL_CTF_REF_MAG ||
281 meg->coils[k]->type == FIFFV_COIL_CTF_REF_GRAD ||
282 meg->coils[k]->type == FIFFV_COIL_CTF_REF_GOFF)
283 stds[n] = 1e6 * stds[n];
284#endif
285 } else
286 stds[n] = static_cast<double>(grad_std) * grad_std;
287 ch_names.push_back(meg->coils[k]->chname);
288 }
289 }
290 if (eeg) {
291 for (k = 0; k < eeg->ncoil(); k++, n++) {
292 stds[n] = static_cast<double>(eeg_std) * eeg_std;
293 ch_names.push_back(eeg->coils[k]->chname);
294 }
295 }
296 names = ch_names;
297 return MNECovMatrix::create(FIFFV_MNE_NOISE_COV, nchan, names, Eigen::VectorXd(), stds);
298}

302int InvDipoleFitData::scale_noise_cov(InvDipoleFitData* f, int nave)
303{
304 float nave_ratio = static_cast<float>(f->nave) / static_cast<float>(nave);
305 int k;
306
307 if (!f->noise)
308 return OK;
309
310 if (f->noise->cov.size() > 0) {
311 mne::Log::info("Decomposing the sensor noise covariance matrix...");
312 if (f->noise->decompose_eigen() == FAIL)
313 return FAIL;
314
315 for (k = 0; k < f->noise->ncov * (f->noise->ncov + 1) / 2; k++)
316 f->noise->cov[k] = nave_ratio * f->noise->cov[k];
317 for (k = 0; k < f->noise->ncov; k++) {
318 f->noise->lambda[k] = nave_ratio * f->noise->lambda[k];
319 if (f->noise->lambda[k] < 0.0)
320 f->noise->lambda[k] = 0.0;
321 }
322 if (f->noise->add_inv() == FAIL)
323 return FAIL;
324 } else {
325 for (k = 0; k < f->noise->ncov; k++)
326 f->noise->cov_diag[k] = nave_ratio * f->noise->cov_diag[k];
327 mne::Log::info("Decomposition not needed for a diagonal noise covariance matrix.");
328 if (f->noise->add_inv() == FAIL)
329 return FAIL;
330 }
331 mne::Log::info("Effective nave is now {}", nave);
332 f->nave = nave;
333 return OK;
334}

338int InvDipoleFitData::scale_dipole_fit_noise_cov(InvDipoleFitData* f, int nave)
339{
340 float nave_ratio = static_cast<float>(f->nave) / static_cast<float>(nave);
341 int k;
342
343 if (!f->noise)
344 return OK;
345 if (f->fixed_noise)
346 return OK;
347
348 if (f->noise->cov.size() > 0) {
349 /*
350 * Do the decomposition and check that the matrix is positive definite
351 */
352 mne::Log::info("Decomposing the noise covariance...");
353 if (f->noise->cov.size() > 0) {
354 if (f->noise->decompose_eigen() == FAIL)
355 return FAIL;
356 for (k = 0; k < f->noise->ncov; k++) {
357 if (f->noise->lambda[k] < 0.0)
358 f->noise->lambda[k] = 0.0;
359 }
360 }
361 for (k = 0; k < f->noise->ncov * (f->noise->ncov + 1) / 2; k++)
362 f->noise->cov[k] = nave_ratio * f->noise->cov[k];
363 for (k = 0; k < f->noise->ncov; k++) {
364 f->noise->lambda[k] = nave_ratio * f->noise->lambda[k];
365 if (f->noise->lambda[k] < 0.0)
366 f->noise->lambda[k] = 0.0;
367 }
368 if (f->noise->add_inv() == FAIL)
369 return FAIL;
370 } else {
371 for (k = 0; k < f->noise->ncov; k++)
372 f->noise->cov_diag[k] = nave_ratio * f->noise->cov_diag[k];
373 mne::Log::info("Decomposition not needed for a diagonal noise covariance matrix.");
374 if (f->noise->add_inv() == FAIL)
375 return FAIL;
376 }
377 mne::Log::info("Effective nave is now {}", nave);
378 f->nave = nave;
379 return OK;
380}

393int InvDipoleFitData::select_dipole_fit_noise_cov(InvDipoleFitData* f,
394 MNEMeasData* meas,
395 int nave_in,
396 const int* sels)
397{
398 int nave, j, k;
399 float nonsel_w = 30;
400 int min_nchan = 20;
401
402 if (!f || !f->noise_orig)
403 return OK;
404 if (!meas)
405 nave = 1;
406 else {
407 if (nave_in < 0)
408 nave = meas->current->nave;
409 else
410 nave = nave_in;
411 }
412 /*
413 * Channel selection
414 */
415 if (meas && sels) {
416 std::vector<float> wVec(f->noise_orig->ncov);
417 float* w = wVec.data();
418 int nomit_meg, nomit_eeg, nmeg, neeg;
419
420 nmeg = neeg = 0;
421 nomit_meg = nomit_eeg = 0;
422 for (k = 0; k < f->noise_orig->ncov; k++) {
423 if (f->noise_orig->ch_class[k] == MNE_COV_CH_EEG)
424 neeg++;
425 else
426 nmeg++;
427 /* Check whether this channel is selected in the measurement */
428 bool selected = false;
429 for (int c = 0; c < meas->nchan; c++) {
430 if (mne::iequals(f->noise_orig->names[k], meas->chs[c].ch_name)) {
431 selected = sels[c] != 0;
432 break;
433 }
434 }
435 if (selected)
436 w[k] = 1.0;
437 else {
438 w[k] = nonsel_w;
439 if (f->noise_orig->ch_class[k] == MNE_COV_CH_EEG)
440 nomit_eeg++;
441 else
442 nomit_meg++;
443 }
444 }
445 f->noise.reset();
446 if (nmeg > 0 && nmeg - nomit_meg > 0 && nmeg - nomit_meg < min_nchan) {
447 mne::Log::critical("Too few MEG channels remaining");
448 return FAIL;
449 }
450 if (neeg > 0 && neeg - nomit_eeg > 0 && neeg - nomit_eeg < min_nchan) {
451 mne::Log::critical("Too few EEG channels remaining");
452 return FAIL;
453 }
454 f->noise = f->noise_orig->dup();
455 if (nomit_meg + nomit_eeg > 0) {
456 if (f->noise->cov.size() > 0) {
457 for (j = 0; j < f->noise->ncov; j++)
458 for (k = 0; k <= j; k++) {
459 f->noise->cov[MNECovMatrix::lt_packed_index(j, k)] *= static_cast<double>(w[j]) * w[k];
460 }
461 } else {
462 for (j = 0; j < f->noise->ncov; j++) {
463 f->noise->cov_diag[j] *= static_cast<double>(w[j]) * w[j];
464 }
465 }
466 }
467 } else {
468 if (f->noise && f->nave == nave)
469 return OK;
470 f->noise = f->noise_orig->dup();
471 }
472
473 return scale_dipole_fit_noise_cov(f, nave);
474}

619 return nullptr;
620 }
621 /*
622 * Forward model setup
623 */
624 res->bemname = bemname;
625 if (r0) {
626 res->r0 = *r0;
627 }
628 res->eeg_model.reset(eeg_model);
629 /*
630 * Compensation data
631 */
632 comp_data = MNECTFCompDataSet::read(mne::path_from_utf8(measname));
633 if (!comp_data)
634 return nullptr;
635 if (comp_data->ncomp > 0) { /* Compensation channel information may be needed */
636 std::vector<FiffChInfo> comp_chs;
637 int ncomp = 0;
638

641 FiffStream compStream(mne::path_from_utf8(measname));

652 if (compInfo.chs[k].kind == FIFFV_REF_MEG_CH) {
653 comp_chs.push_back(compInfo.chs[k]);
654 ncomp++;
655 }
656 }
657 }
658 if (ncomp > 0) {
659 comp_coils = templates->create_meg_coils(comp_chs,
660 ncomp,
661 FWD_COIL_ACCURACY_NORMAL,
662 *res->meg_head_t);
663 if (!comp_coils) {
664 return nullptr;
665 }
666 mne::Log::info("{} compensation channels in {}", comp_coils->ncoil(), measname);
667 }
668 } else { /* Get rid of the empty data set */
669 comp_data.reset();
670 }
671 /*
672 * Ready to set up the forward model
673 */
674 if (setup_forward_model(res.get(), comp_data.get(), comp_coils.get()) == FAIL)
675 return nullptr;
676 res->column_norm = COLUMN_NORM_LOC;
677 /*
678 * Projection data should go here
679 */
680 std::vector<std::filesystem::path> projPaths;
681 projPaths.reserve(projnames.size());
682 for (const std::string& projname : projnames)
683 projPaths.emplace_back(mne::path_from_utf8(projname));
684 auto projection = MNEProjOp::make_projection(projPaths, std::span(res->chs).first(static_cast<std::size_t>(res->nmeg) + static_cast<std::size_t>(res->neeg)));
685 if (!projection) {
686 mne::Log::critical("{}", projection.error().what());
687 return nullptr;
688 }
689 res->proj = std::move(*projection);
690 if (res->proj && !res->proj->items.empty()) {
691 mne::Log::info("Final projection operator is:");
692 {
693 std::ostringstream report;
694 res->proj->report(report, "\t");
695 mne::Log::info("{}", report.str());
696 }
697
698 res->proj->assign_channels(res->ch_names);
699 if (!res->proj->make_proj())
700 return nullptr;
701 } else
702 mne::Log::info("No projection will be applied to the data.");
703
704 /*
705 * Noise covariance
706 */
707 if (!noisename.empty()) {
708 if ((cov = MNECovMatrix::read(mne::path_from_utf8(noisename), FIFFV_MNE_SENSOR_COV)) == nullptr)
709 return nullptr;
710 mne::Log::info("Read a {} noise-covariance matrix from {}", cov->cov_diag.size() > 0 ? "diagonal" : "full", noisename);
711 } else {
712 if ((cov = ad_hoc_noise(res->meg_coils.get(), res->eeg_els.get(), grad_std, mag_std, eeg_std)) == nullptr)
713 return nullptr;
714 }
715 res->noise = cov->pick_chs_omit(res->ch_names, true, res->chs);
716 if (!res->noise) {
717 return nullptr;
718 }
719
720 mne::Log::info("Picked appropriate channels from the noise-covariance matrix.");
721 cov.reset();
722
723 /*
724 * Apply the projection operator to the noise-covariance matrix
725 */
726 if (res->proj && !res->proj->items.empty() && res->proj->nvec > 0) {
727 if (!res->proj->apply_cov(*res->noise))
728 return nullptr;
729 mne::Log::info("Projection applied to the covariance matrix.");
730 }
731
732 /*
733 * Force diagonal noise covariance?
734 */
735 if (diagnoise) {
736 res->noise->revert_to_diag();
737 mne::Log::info("Using only the main diagonal of the noise-covariance matrix.");
738 }
739
740 /*
741 * Regularize the possibly deficient noise-covariance matrix
742 */
743 if (res->noise->cov.size() > 0) {
744 Eigen::Vector3f regs;
745 int do_it;
746
747 regs[MNE_COV_CH_MEG_MAG] = mag_reg;
748 regs[MNE_COV_CH_MEG_GRAD] = grad_reg;
749 regs[MNE_COV_CH_EEG] = eeg_reg;
750 /*
751 * Classify the channels
752 */
753 if (res->noise->classify_channels(res->chs,
754 res->nmeg + res->neeg) == FAIL)
755 return nullptr;
756 /*
757 * Do we need to do anything?
758 */
759 do_it = 0;
760 for (int k = 0; k < res->noise->ncov; k++) {
761 if (res->noise->ch_class[k] != MNE_COV_CH_UNKNOWN &&
762 regs[res->noise->ch_class[k]] > 0.0)
763 do_it++;
764 }
765 /*
766 * Apply regularization if necessary
767 */
768 if (do_it > 0)
769 res->noise->regularize(regs);
770 else
771 mne::Log::info("No regularization applied to the noise-covariance matrix");
772 }
773
774 /*
775 * Do the decomposition and check that the matrix is positive definite
776 */
777 mne::Log::info("Decomposing the noise covariance...");
778 if (res->noise->cov.size() > 0) {
779 if (res->noise->decompose_eigen() == FAIL)
780 return nullptr;
781 mne::Log::info("Eigenvalue decomposition done.");
782 for (int k = 0; k < res->noise->ncov; k++) {
783 if (res->noise->lambda[k] < 0.0)
784 res->noise->lambda[k] = 0.0;
785 }
786 } else {
787 mne::Log::info("Decomposition not needed for a diagonal covariance matrix.");
788 if (res->noise->add_inv() == FAIL)
789 return nullptr;
790 }
791
792 badlist.clear();
793 return res.release();
794}

836std::unique_ptr<InvDipoleForward> dipole_forward(InvDipoleFitData* d,
837 float** rd,
838 int ndip,
839 std::unique_ptr<InvDipoleForward> old)
840{
841 std::unique_ptr<InvDipoleForward> res;
842 float S[3];
843 int k;
844 /*
845 * Allocate data if necessary
846 */
847 if (old && old->ndip == ndip && old->nch == d->nmeg + d->neeg) {
848 res = std::move(old);
849 } else {
850 res = std::make_unique<InvDipoleForward>();
851 int nch = d->nmeg + d->neeg;
852 int m = 3 * ndip;
853 res->fwd.resize(m, nch);
854 res->uu.resize(m, nch);
855 res->vv.resize(m, m);
856 res->sing.resize(m);
857 res->scales.resize(m);
858 res->rd.resize(ndip, 3);
859 res->nch = nch;
860 res->ndip = ndip;
861 }
862
863 for (k = 0; k < ndip; k++) {
864 res->rd.row(k) = Eigen::Map<const Eigen::Vector3f>(rd[k]).transpose();
865 /*
866 * Calculate the field of three orthogonal dipoles
867 */
868 Eigen::MatrixXf this_fwd(d->nmeg + d->neeg, 3);
869 Eigen::Map<const Eigen::Vector3f> rd_k(rd[k]);
870 if ((InvDipoleFitData::compute_dipole_field(*d, rd_k, true, this_fwd)) == FAIL)
871 return nullptr;
872 for (int p = 0; p < 3; p++)
873 res->fwd.row(3 * k + p) = this_fwd.col(p).transpose();
874 /*
875 * Choice of column normalization
876 * (componentwise normalization is not recommended)
877 */
878 if (d->column_norm == COLUMN_NORM_LOC || d->column_norm == COLUMN_NORM_COMP) {
879 for (int p = 0; p < 3; p++)
880 S[p] = res->fwd.row(3 * k + p).squaredNorm();
881 if (d->column_norm == COLUMN_NORM_COMP) {
882 for (int p = 0; p < 3; p++)
883 res->scales[3 * k + p] = std::sqrt(static_cast<double>(S[p]));
884 } else {
885 /*
886 * Divide by three or not?
887 */
888 res->scales[3 * k + 0] = res->scales[3 * k + 1] = res->scales[3 * k + 2] = std::sqrt(static_cast<double>(S[0] + S[1] + S[2])) / 3.0;
889 }
890 for (int p = 0; p < 3; p++) {
891 if (res->scales[3 * k + p] > 0.0) {
892 res->scales[3 * k + p] = 1.0 / res->scales[3 * k + p];
893 res->fwd.row(3 * k + p) *= res->scales[3 * k + p];
894 } else
895 res->scales[3 * k + p] = 1.0;
896 }
897 } else {
898 res->scales[3 * k] = 1.0;
899 res->scales[3 * k + 1] = 1.0;
900 res->scales[3 * k + 2] = 1.0;
901 }
902 }
903
904 /*
905 * SVD: A = U · Σ · V^T where A is m×n (3*ndip × nch)
906 * uu stores right singular vectors (V^T rows, length nch) for data-space projections
907 * vv stores left singular vectors (U^T rows, length m) for dipole-moment reconstruction
908 */
909 {
910 int m = 3 * ndip;
911 int n = d->nmeg + d->neeg;
912 int udim = std::min(m, n);
913 JacobiSVD<MatrixXf> svd(res->fwd, ComputeFullU | ComputeFullV);
914 res->sing = svd.singularValues();
915 res->uu = svd.matrixV().transpose().topRows(udim);
916 res->vv = svd.matrixU().transpose().topRows(udim);
917 }
918
919 return res;
920}

941static float fit_eval(const VectorXf& rd, const void* user)
942{
943 InvDipoleFitData* fit = const_cast<InvDipoleFitData*>(static_cast<const InvDipoleFitData*>(user));
944 InvDipoleForward* fwd;
945 FitDipUserRec* fuser = fit->user;
946 double Bm2, one;
947 int ncomp, c;
948
949 fuser->fwd = InvDipoleFitData::dipole_forward_one(fit, rd.head<3>(), std::move(fuser->fwd));
950 fwd = fuser->fwd.get();
951 ncomp = fwd->sing[2] / fwd->sing[0] > fuser->limit ? 3 : 2;
952 if (fuser->report_dim)
953 mne::Log::info("ncomp = {}", ncomp);
954
955 Eigen::Map<const VectorXf> Bmap(fuser->B, fwd->nch);
956 for (c = 0, Bm2 = 0.0; c < ncomp; c++) {
957 one = fwd->uu.row(c).dot(Bmap);
958 Bm2 = Bm2 + one * one;
959 }
960 return fuser->B2 - Bm2;
961}

966static int find_best_guess(const Eigen::Ref<const Eigen::VectorXf>& B,
967 int nch,
968 InvGuessData* guess,
969 float limit,
970 int& bestp,
971 float& goodp)
972{
973 int k, c;
974 double B2, Bm2, this_good, one;
975 int best = -1;
976 float good = 0.0;
977 InvDipoleForward* fwd;
978 int ncomp;
979
980 B2 = B.squaredNorm();
981 for (k = 0; k < guess->nguess; k++) {
982 fwd = guess->guess_fwd[k].get();
983 if (fwd->nch == nch) {
984 ncomp = fwd->sing[2] / fwd->sing[0] > limit ? 3 : 2;
985 for (c = 0, Bm2 = 0.0; c < ncomp; c++) {
986 one = fwd->uu.row(c).dot(B);
987 Bm2 = Bm2 + one * one;
988 }
989 this_good = 1.0 - (B2 - Bm2) / B2;
990 if (this_good > good) {
991 best = k;
992 good = this_good;
993 }
994 }
995 }
996 if (best < 0) {
997 mne::Log::warning("No reasonable initial guess found.");
998 return FAIL;
999 }
1000 bestp = best;
1001 goodp = good;
1002 return OK;

1008static MatrixXf make_initial_dipole_simplex(const Eigen::Vector3f& r0,
1009 float size)
1010{
1011 /*
1012 * For this definition of a regular tetrahedron, see
1013 *
1014 * http://mathworld.wolfram.com/Tetrahedron.html
1015 *
1016 */
1017 float x = std::sqrt(3.0) / 3.0f;
1018 float r = std::sqrt(6.0) / 12.0f;
1019 float R = 3 * r;
1020 float d = x / 2.0f;
1021 float rr[][3] = {{x, 0.0f, -r},
1022 {-d, 0.5f, -r},
1023 {-d, -0.5f, -r},
1024 {0.0f, 0.0f, R}};
1025
1026 MatrixXf simplex = MatrixXf::Zero(4, 3);
1027
1028 for (int j = 0; j < 4; j++) {
1029 simplex.row(j) = Eigen::Map<const Vector3f>(rr[j]).transpose() * size + r0.transpose();
1030 }
1031 return simplex;
1032}

1048static int fit_Q(InvDipoleFitData* fit,
1049 const Eigen::Ref<const Eigen::VectorXf>& B,
1050 const Eigen::Vector3f& rd,
1051 float limit,
1052 Eigen::Vector3f& Q,
1053 int& ncomp,
1054 float& res)
1055{
1056 int c;
1057 const auto fwd = InvDipoleFitData::dipole_forward_one(fit, rd, nullptr);
1058 float Bm2, one;
1059
1060 if (!fwd)
1061 return FAIL;
1062
1063 ncomp = fwd->sing[2] / fwd->sing[0] > limit ? 3 : 2;
1064
1065 Q.setZero();
1066 for (c = 0, Bm2 = 0.0; c < ncomp; c++) {
1067 one = fwd->uu.row(c).dot(B);
1068 Q += (one / fwd->sing[c]) * fwd->vv.row(c).head(3).transpose();
1069 Bm2 = Bm2 + one * one;
1070 }
1071 /*
1072 * Counteract the effect of column normalization
1073 */
1074 for (c = 0; c < 3; c++)
1075 Q[c] = fwd->scales[c] * Q[c];
1076 res = B.squaredNorm() - Bm2;
1077
1078 return OK;
1079}

1094bool InvDipoleFitData::fit_one(InvDipoleFitData* fit,
1095 InvGuessData* guess,
1096 float time,
1097 Eigen::Ref<Eigen::VectorXf> B,
1098 int verbose,
1099 InvEcd& res)
1100{
1101 VectorXf vals(4); /* Values at the vertices */
1102 float limit = 0.2f; /* (pseudo) radial component omission limit */
1103 float size = 1e-2f; /* Size of the initial simplex */
1104 float ftol[] = {1e-2f, 1e-2f}; /* Tolerances on the the two passes */
1105 float atol[] = {0.2e-3f, 0.2e-3f}; /* If dipole movement between two iterations is less than this,
1106 we consider to have converged */
1107 int ntol = 2;
1108 int max_eval = 1000; /* Limit for fit function evaluations */
1109 int report_interval = verbose ? 1 : -1; /* How often to report the intermediate result */
1110
1111 int best;
1112 float good, final_val;
1113 Eigen::Vector3f rd_final, Q;
1114 FitDipUserRec user;
1115 int k, neval, neval_tot, nchan, ncomp;
1116 int fit_fail;
1117 Vector3f rd_guess;
1118
1119 nchan = fit->nmeg + fit->neeg;
1120 user.fwd = nullptr;
1121
1122 if (fit->proj && !fit->proj->project_vector(B, true))
1123 return false;
1124
1125 if (fit->noise->whiten_vector(B, B, nchan) == FAIL)
1126 return false;
1127 /*
1128 * Get the initial guess
1129 */
1130 if (find_best_guess(B, nchan, guess, limit, best, good) < 0)
1131 return false;
1132
1133 user.limit = limit;
1134 user.B = B.data();
1135 user.B2 = B.squaredNorm();
1136 user.fwd = nullptr;
1137 user.report_dim = false;
1138 fit->user = &user;
1139
1140 rd_guess = guess->rr.row(best).transpose();
1141 rd_final = rd_guess;
1142
1143 neval_tot = 0;
1144 fit_fail = false;
1145 for (k = 0; k < ntol; k++) {
1146 /*
1147 * Do first pass with the sphere model
1148 */
1149 if (fit->fit_mag_dipoles)
1150 fit->funcs = fit->mag_dipole_funcs.get(); // the guesses were computed with these too
1151 else if (k == 0)
1152 fit->funcs = fit->sphere_funcs.get();
1153 else
1154 fit->funcs = !fit->bemname.empty() ? fit->bem_funcs.get() : fit->sphere_funcs.get();
1155
1156 MatrixXf simplexMat = make_initial_dipole_simplex(rd_guess, size);
1157 for (int p = 0; p < 4; p++)
1158 vals[p] = fit_eval(simplexMat.row(p), fit);
1159
1160 // Capture fit pointer in type-safe lambda — no void* needed
1161 auto cost = [fit](const VectorXf& x) -> float {
1162 return fit_eval(x, fit);
1163 };
1164
1165 if (!mne::SimplexAlgorithm::simplex_minimize<float>(
1166 simplexMat, /* The initial simplex */
1167 vals, /* Function values at the vertices */
1168 ftol[k], /* Relative convergence tolerance for the target function */
1169 atol[k], /* Absolute tolerance for the change in the parameters */
1170 cost, /* The cost function (captures fit data) */
1171 max_eval, /* Maximum number of function evaluations */
1172 neval, /* Number of function evaluations */
1173 report_interval, /* How often to report (-1 = no_reporting) */
1174 dipole_report_func)) {
1175 if (k == 0) {
1176 return false;
1177 } else {
1178 float rv = 2.0f * (vals.maxCoeff() - vals.minCoeff()) / (vals.maxCoeff() + vals.minCoeff());
1179 mne::Log::warning("Warning (t = {:8.1f} ms) : g = {:6.1f} % final val = {:7.3f} rtol = {:f}", 1000 * time, 100 * (1 - vals[0] / user.B2), vals[0], rv);
1180 fit_fail = true;
1181 }
1182 }
1183 rd_final = simplexMat.row(0).transpose();
1184 rd_guess = simplexMat.row(0).transpose();
1185
1186 neval_tot += neval;
1187 final_val = vals[0];
1188 }
1189 /*
1190 * Confidence limits should be computed here
1191 */
1192 /*
1193 * Compute the dipole moment at the final point
1194 */
1195 if (fit_Q(fit, B, rd_final, user.limit, Q, ncomp, final_val) == OK) {
1196 res.time = time;
1197 res.valid = true;
1198 res.rd = rd_final;
1199 res.Q = Q;
1200 res.good = 1.0 - final_val / user.B2;
1201 if (fit_fail)
1202 res.good = -res.good;
1203 res.khi2 = final_val;
1204 if (fit->proj)
1205 res.nfree = nchan - 3 - ncomp - fit->proj->nvec;
1206 else
1207 res.nfree = nchan - 3 - ncomp;
1208 res.neval = neval_tot;
1209 } else {
1210 return false;
1211 }
1212
1213 return true;
1214}

1223int InvDipoleFitData::compute_dipole_field(InvDipoleFitData& d, const Eigen::Vector3f& rd, int whiten, Eigen::Ref<Eigen::MatrixXf> fwd)
1224{
1225 static const Eigen::Vector3f Qx(1.0f, 0.0f, 0.0f);
1226 static const Eigen::Vector3f Qy(0.0f, 1.0f, 0.0f);
1227 static const Eigen::Vector3f Qz(0.0f, 0.0f, 1.0f);
1228 int nch = d.nmeg + d.neeg;
1229 int k;
1230 /*
1231 * Compute the fields
1232 */
1233 if (d.nmeg > 0) {
1234 int nmeg = d.meg_coils->ncoil();
1235 if (d.funcs->meg_vec_field) {
1236 /*
1237 * Use the vector field function: computes all three dipole
1238 * orientations at once. Output is 3 x ncoil, we need nch x 3.
1239 */
1240 Eigen::MatrixXf vec_meg(3, nmeg);
1241 if (d.funcs->meg_vec_field(rd, *d.meg_coils, vec_meg, d.funcs->meg_client) != OK)
1242 return FAIL;
1243 fwd.topRows(nmeg) = vec_meg.transpose();
1244 } else {
1245 auto fwd0 = fwd.col(0).head(nmeg);
1246 auto fwd1 = fwd.col(1).head(nmeg);
1247 auto fwd2 = fwd.col(2).head(nmeg);
1248 if (d.funcs->meg_field(rd, Qx, *d.meg_coils, fwd0, d.funcs->meg_client) != OK)
1249 return FAIL;
1250 if (d.funcs->meg_field(rd, Qy, *d.meg_coils, fwd1, d.funcs->meg_client) != OK)
1251 return FAIL;
1252 if (d.funcs->meg_field(rd, Qz, *d.meg_coils, fwd2, d.funcs->meg_client) != OK)
1253 return FAIL;
1254 }
1255 }
1256
1257 if (d.neeg > 0) {
1258 int neeg = d.eeg_els->ncoil();
1259 if (d.funcs->eeg_vec_pot) {
1260 /*
1261 * Use the vector potential function: computes all three dipole
1262 * orientations at once. Output is 3 x ncoil, we need nch x 3.
1263 */
1264 Eigen::MatrixXf vec_eeg(3, neeg);
1265 if (d.funcs->eeg_vec_pot(rd, *d.eeg_els, vec_eeg, d.funcs->eeg_client) != OK)
1266 return FAIL;
1267 fwd.block(d.nmeg, 0, neeg, 3) = vec_eeg.transpose();
1268 } else {
1269 auto fwd0 = fwd.col(0).segment(d.nmeg, neeg);
1270 auto fwd1 = fwd.col(1).segment(d.nmeg, neeg);
1271 auto fwd2 = fwd.col(2).segment(d.nmeg, neeg);
1272 if (d.funcs->eeg_pot(rd, Qx, *d.eeg_els, fwd0, d.funcs->eeg_client) != OK)
1273 return FAIL;
1274 if (d.funcs->eeg_pot(rd, Qy, *d.eeg_els, fwd1, d.funcs->eeg_client) != OK)
1275 return FAIL;
1276 if (d.funcs->eeg_pot(rd, Qz, *d.eeg_els, fwd2, d.funcs->eeg_client) != OK)
1277 return FAIL;
1278 }
1279 }
1280
1281 /*
1282 * Apply projection
1283 */
1284 for (k = 0; k < 3; k++)
1285 if (d.proj && !d.proj->project_vector(fwd.col(k), true))
1286 return FAIL;
1287
1288 /*
1289 * Whiten
1290 */
1291 if (d.noise && whiten) {
1292 for (k = 0; k < 3; k++) {
1293 auto col_k = fwd.col(k);
1294 if (d.noise->whiten_vector(col_k, col_k, nch) == FAIL)
1295 return FAIL;
1296 }
1297 }
1298
1299 return OK;
1300}
