84 AcqExpected<FtMessage> request(std::uint16_t command, std::span<const std::byte> payload, std::chrono::milliseconds timeout) override
85 {
86 if (!m_started) {
87 if (AcqError e = m_source->start(); !e.ok()) {
88 return e;
89 }
90 m_started = true;
91 }
92 switch (command) {
93 case ft::kGetHdr:
94 return header(timeout);
95 case ft::kWaitDat:
96 return wait(payload, timeout);
97 case ft::kGetDat:
98 return data(payload);
99 default:
100 return FtMessage{ft::kGetErr, {}}; // the relay puts no events
101 }
102 }

104private:
106 AcqExpected<bool> pump(std::chrono::milliseconds timeout)
107 {
108 if (m_ended) {
109 return AcqError(AcqErrorCode::EndOfStream);
110 }
111 auto tag = m_source->next(timeout);
112 if (!tag) {
113 if (tag.error() == AcqErrorCode::Timeout) {
114 return false;
115 }
116 m_ended = tag.error() == AcqErrorCode::EndOfStream;
117 return tag.error();
118 }
119 const NeuromagTag& t = **tag;
120 if (t.kind != fiff::kDataBuffer) {
121 if (m_header.empty()) {
122 m_tags.push_back(t);
123 if (t.kind == fiff::kBlockStart && t.as_int() == fiff::kBlockRawData) {
124 make_header();
125 }
126 }
127 return true;
128 }
129 // The relay applies the calibration and puts floats, sample after sample
130 const std::size_t width = m_scales.size();
131 const std::size_t samples = t.data.size() / (width * sizeof(std::int32_t));
132 for (std::size_t s = 0; s < samples; ++s) {
133 for (std::size_t c = 0; c < width; ++c) {
134 const auto raw = get<std::int32_t>(t.data, (s * width + c) * sizeof(std::int32_t));
135 m_samples.push_back(static_cast<float>(m_scales[c] * raw));
136 }
137 }
138 m_total += samples;
139 while (m_samples.size() > m_ring * width) {
140 m_samples.erase(m_samples.begin(), m_samples.begin() + static_cast<std::ptrdiff_t>(width));
141 }
142 return true;
143 }

145 void make_header()
146 {
147 const std::vector<NeuromagTag> meas_info = block_of(m_tags, fiff::kBlockMeasInfo);
148 std::vector<std::byte> names;
149 float sfreq = 0.0F;
150 for (const NeuromagTag& tag : meas_info) {
151 if (tag.kind == fiff::kSfreq) {
152 sfreq = tag.as_float().value_or(0.0F);
153 } else if (const auto ch = tag.kind == fiff::kChInfo ? NeuromagChannel::parse(tag.data) : std::nullopt) {
154 m_scales.push_back(static_cast<double>(ch->cal) * static_cast<double>(ch->range));
155 const auto* chars = reinterpret_cast<const std::byte*>(ch->name.c_str());
156 names.insert(names.end(), chars, chars + ch->name.size() + 1);
157 }
158 }
159 const std::vector<std::byte> fiff = tag_file(m_tags, false);
160 std::vector<std::byte> chunks;
161 using Chunk = std::pair<std::uint32_t, const std::vector<std::byte>*>;
162 for (const auto& [type, data] : {Chunk{ft::kChunkChannelNames, &names}, Chunk{ft::kChunkNeuromagHeader, &fiff}}) {
163 put(chunks, type);
164 put(chunks, static_cast<std::uint32_t>(data->size()));
165 chunks.insert(chunks.end(), data->begin(), data->end());
166 }
167 put(m_header, static_cast<std::uint32_t>(m_scales.size()));
168 m_header_counts = m_header.size();
169 put(m_header, std::uint32_t{0}); // nsamples, set on every answer
170 put(m_header, std::uint32_t{0}); // nevents
171 put(m_header, sfreq);
172 put(m_header, ft::kTypeFloat32);
173 put(m_header, static_cast<std::uint32_t>(chunks.size()));
174 m_header.insert(m_header.end(), chunks.begin(), chunks.end());
175 }

177 AcqExpected<FtMessage> header(std::chrono::milliseconds timeout)
178 {
179 const auto deadline = Clock::now() + timeout;
180 while (m_header.empty() && Clock::now() < deadline) {
181 if (auto pumped = pump(kSlice); !pumped) {
182 return pumped.error();
183 }
184 }
185 if (m_header.empty()) {
186 return FtMessage{ft::kGetErr, {}};
187 }
188 FtMessage answer{ft::kGetOk, m_header};
189 const auto total = static_cast<std::uint32_t>(m_total);
190 std::memcpy(answer.payload.data() + m_header_counts, &total, sizeof(total));
191 return answer;
192 }

194 AcqExpected<FtMessage> wait(std::span<const std::byte> payload, std::chrono::milliseconds timeout)
195 {
196 if (payload.size() < 12) {
197 return FtMessage{ft::kWaitErr, {}};
198 }
199 const auto threshold = get<std::uint32_t>(payload, 0);
200 const auto deadline = Clock::now() + std::min(timeout, std::chrono::milliseconds(get<std::uint32_t>(payload, 8)));
201 while ((m_header.empty() || m_total <= threshold) && Clock::now() < deadline) {
202 if (auto pumped = pump(kSlice); !pumped) {
203 if (pumped.error() != AcqErrorCode::EndOfStream || m_total <= threshold) {
204 return pumped.error();
205 }
206 break;
207 }
208 }
209 FtMessage answer{ft::kWaitOk, {}};
210 put(answer.payload, static_cast<std::uint32_t>(m_total));
211 put(answer.payload, std::uint32_t{0});
212 return answer;
213 }

215 AcqExpected<FtMessage> data(std::span<const std::byte> payload) const
216 {
217 const std::size_t width = m_scales.size();
218 const std::size_t held = width == 0 ? 0 : m_samples.size() / width;
219 if (payload.size() < 8 || width == 0) {
220 return FtMessage{ft::kGetErr, {}};
221 }
222 const std::size_t begin = get<std::uint32_t>(payload, 0);
223 const std::size_t end = get<std::uint32_t>(payload, 4);
224 const std::size_t first = m_total - held;
225 if (begin < first || end < begin || end >= m_total) {
226 return FtMessage{ft::kGetErr, {}};
227 }
228 const std::size_t count = end - begin + 1;
229 FtMessage answer{ft::kGetOk, {}};
230 put(answer.payload, static_cast<std::uint32_t>(width));
231 put(answer.payload, static_cast<std::uint32_t>(count));
232 put(answer.payload, ft::kTypeFloat32);
233 put(answer.payload, static_cast<std::uint32_t>(count * width * sizeof(float)));
234 const auto from = m_samples.begin() + static_cast<std::ptrdiff_t>((begin - first) * width);
235 for (auto it = from; it != from + static_cast<std::ptrdiff_t>(count * width); ++it) {
236 put(answer.payload, *it);
237 }
238 return answer;
239 }
