62bool SourceEstimateManager::load(const QString& lhPath, const QString& rhPath,
63 const QMap<QString, std::shared_ptr<BrainSurface>>& surfaces,
64 const QString& activeSurfaceType)
65{
66 if (m_isLoading) {
67 qWarning() << "SourceEstimateManager: STC loading already in progress";
68 return false;
69 }
70
71 // Find surfaces for the active surface type
72 BrainSurface* lhSurface = nullptr;
73 BrainSurface* rhSurface = nullptr;
74
75 const QString lhKey = "lh_" + activeSurfaceType;
76 const QString rhKey = "rh_" + activeSurfaceType;
77
78 if (surfaces.contains(lhKey))
79 lhSurface = surfaces[lhKey].get();
80 if (surfaces.contains(rhKey))
81 rhSurface = surfaces[rhKey].get();
82
83 // Fallback: search for any lh_*/rh_* brain surface
84 if (!lhSurface || !rhSurface) {
85 for (auto it = surfaces.begin(); it != surfaces.end(); ++it) {
86 if (it.value() && it.value()->tissueType() == BrainSurface::TissueBrain) {
87 if (!lhSurface && it.key().startsWith("lh_")) {
88 lhSurface = it.value().get();
89 qDebug() << "SourceEstimateManager: Using fallback LH surface:" << it.key();
90 } else if (!rhSurface && it.key().startsWith("rh_")) {
91 rhSurface = it.value().get();
92 qDebug() << "SourceEstimateManager: Using fallback RH surface:" << it.key();
93 }
94 }
95 }
96 }
97
98 if (!lhSurface && !rhSurface) {
99 qWarning() << "SourceEstimateManager: No surfaces available for STC loading."
100 << "Active surface type:" << activeSurfaceType
101 << "Available keys:" << surfaces.keys();
102 return false;
103 }
104
105 // Clean up any previous loading thread
106 if (m_loadingThread) {
107 m_loadingThread->quit();
108 m_loadingThread->wait();
109 delete m_loadingThread;
110 m_loadingThread = nullptr;
111 }
112
113 // Create overlay for results
114 m_overlay = std::make_unique<SourceEstimateOverlay>();
115
116 // Create worker and thread
117 m_loadingThread = new QThread(this);
118 m_stcWorker = new StcLoadingWorker(lhPath, rhPath, lhSurface, rhSurface);
119 m_stcWorker->moveToThread(m_loadingThread);
120
121 connect(m_loadingThread, &QThread::started, m_stcWorker, &StcLoadingWorker::process);
122 connect(m_stcWorker, &StcLoadingWorker::progress, this, &SourceEstimateManager::loadingProgress);
123 connect(m_stcWorker, &StcLoadingWorker::finished, this, &SourceEstimateManager::onStcLoadingFinished);
124 connect(m_stcWorker, &StcLoadingWorker::finished, m_loadingThread, &QThread::quit);
125 connect(m_loadingThread, &QThread::finished, m_stcWorker, &QObject::deleteLater);
126
127 m_isLoading = true;
128 m_loadingThread->start();
129
130 return true;
131}

142void SourceEstimateManager::onStcLoadingFinished(bool success)
143{
144 m_isLoading = false;
145
146 if (!success || !m_stcWorker) {
147 qWarning() << "SourceEstimateManager: Async STC loading failed";
148 m_overlay.reset();
149 return;
150 }
151
152 // Transfer data from worker to overlay
153 if (m_stcWorker->hasLh()) {
154 m_overlay->setStcData(m_stcWorker->stcLh(), 0);
155 if (m_stcWorker->interpolationMatLh())
156 m_overlay->setInterpolationMatrix(m_stcWorker->interpolationMatLh(), 0);
157 }
158
159 if (m_stcWorker->hasRh()) {
160 m_overlay->setStcData(m_stcWorker->stcRh(), 1);
161 if (m_stcWorker->interpolationMatRh())
162 m_overlay->setInterpolationMatrix(m_stcWorker->interpolationMatRh(), 1);
163 }
164
165 m_overlay->updateThresholdsFromData();
166 emit thresholdsUpdated(m_overlay->thresholdMin(),
167 m_overlay->thresholdMid(),
168 m_overlay->thresholdMax());
169
170 if (m_overlay->isLoaded()) {
171 emit loaded(m_overlay->numTimePoints());
172 } else {
173 m_overlay.reset();
174 }
175}

179void SourceEstimateManager::setTimePoint(int index,
180 const QMap<QString, std::shared_ptr<BrainSurface>>& surfaces,
181 const SubView& singleView,
182 const QVector<SubView>& subViews)
183{
184 if (!m_overlay || !m_overlay->isLoaded())
185 return;
186
187 m_currentTimePoint = qBound(0, index, m_overlay->numTimePoints() - 1);
188
189 // Collect all distinct surface types used across single + multi views
190 QSet<QString> activeTypes;
191 activeTypes.insert(singleView.surfaceType);
192 for (int i = 0; i < subViews.size(); ++i)
193 activeTypes.insert(subViews[i].surfaceType);
194
195 // Apply source estimate to surfaces matching ANY active type
196 for (auto it = surfaces.begin(); it != surfaces.end(); ++it) {
197 for (const QString& type : activeTypes) {
198 if (it.key().endsWith(type)) {
199 m_overlay->applyToSurface(it.value().get(), m_currentTimePoint);
200 break;
201 }
202 }
203 }
204
205 emit timePointChanged(m_currentTimePoint, m_overlay->timeAtIndex(m_currentTimePoint));
206}

231int SourceEstimateManager::closestIndex(float timeSec) const
232{
233 if (!m_overlay || !m_overlay->isLoaded())
234 return -1;
235
236 const float t0 = m_overlay->tmin();
237 const float dt = m_overlay->tstep();
238 const int numPts = m_overlay->numTimePoints();
239 if (numPts <= 0 || dt <= 0.0f)
240 return -1;
241
242 const int idx = qRound((timeSec - t0) / dt);
243 return qBound(0, idx, numPts - 1);
244}

267void SourceEstimateManager::startStreaming(const QMap<QString, std::shared_ptr<BrainSurface>>& surfaces,
268 const SubView& singleView,
269 const QVector<SubView>& subViews)
270{
271 Q_UNUSED(surfaces)
272 Q_UNUSED(singleView)
273 Q_UNUSED(subViews)
274
275 if (m_isStreaming) {
276 qDebug() << "SourceEstimateManager: Real-time streaming already active";
277 return;
278 }
279
280 if (!m_overlay || !m_overlay->isLoaded()) {
281 qWarning() << "SourceEstimateManager: Cannot start streaming — no source estimate loaded";
282 return;
283 }
284
285 // Create controller on first use
286 if (!m_rtController) {
287 m_rtController = std::make_unique<RtSourceDataController>(this);
288 connect(m_rtController.get(), &RtSourceDataController::newSmoothedDataAvailable,
289 this, &SourceEstimateManager::realtimeColorsAvailable);
290 }
291
292 // Propagate interpolation matrices from the overlay
293 m_rtController->setInterpolationMatrixLeft(m_overlay->interpolationMatLh());
294 m_rtController->setInterpolationMatrixRight(m_overlay->interpolationMatRh());
295
296 // Propagate current visualization parameters
297 m_rtController->setColormapType(m_overlay->colormap());
298 m_rtController->setThresholds(m_overlay->thresholdMin(),
299 m_overlay->thresholdMid(),
300 m_overlay->thresholdMax());
301 m_rtController->setSFreq(1.0 / m_overlay->tstep());
302
303 // Feed all STC time-points into the queue
304 const int nTimePoints = m_overlay->numTimePoints();
305 qDebug() << "SourceEstimateManager: Feeding" << nTimePoints << "time points into real-time queue";
306 m_rtController->clearData();
307
308 for (int t = 0; t < nTimePoints; ++t) {
309 Eigen::VectorXd col = m_overlay->sourceDataColumn(t);
310 if (col.size() > 0)
311 m_rtController->addData(col);
312 }
313
314 m_rtController->setStreamingState(true);
315 m_isStreaming = true;
316
317 qDebug() << "SourceEstimateManager: Real-time streaming started";
318}
