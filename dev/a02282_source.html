110void SceneViewport::setFigure(std::shared_ptr<viz::SceneFigure> figure)
111{
112 m_view.reset();
113 m_figure = std::move(figure);
114 if (m_figure) {
115 m_view = std::make_unique<viz::FigureView>(*m_figure, *m_theme);
116 m_view->set_background(toSceneColor(m_background));
117 m_view->set_font(toSceneFont(font()));
118 m_view->set_text_metrics(m_metrics.get());
119 m_view->attach(m_surface.get());
120 m_view->set_view_size(width(), height());
121 }
122 update();

317bool SceneViewport::event(QEvent* event)
318{
319 if (event->type() == QEvent::Gesture) {
320 auto* gestures = static_cast<QGestureEvent*>(event);
321 if (auto* pinch = static_cast<QPinchGesture*>(gestures->gesture(Qt::PinchGesture))) {
322 zoomBy(pinch->scaleFactor(), mapFromGlobal(pinch->centerPoint()));
323 }
324 if (auto* pan = static_cast<QPanGesture*>(gestures->gesture(Qt::PanGesture))) {
325 panBy(-pan->delta());
326 }
327 return true;
328 }
329 if (event->type() == QEvent::FontChange) {
330 fontChanged();
331 }
332 return QWidget::event(event);
333}
