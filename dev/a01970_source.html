45GpuPipeline surface_pipeline(const char* shader)
46{
47 GpuPipeline pipeline;
48 pipeline.shader = shader;
49 pipeline.bindings = {{6 * sizeof(float), false}, {4, false}, {4, false}};
50 pipeline.attributes = {{0, 0, GpuVertexFormat::Float3, 0},
51 {1, 0, GpuVertexFormat::Float3, 3 * sizeof(float)},
52 {2, 1, GpuVertexFormat::UNormByte4, 0},
53 {3, 2, GpuVertexFormat::UNormByte4, 0}};
54 pipeline.depth_test = true;
55 pipeline.depth_write = true;
56 return pipeline;
57}
58
59GpuPipeline instanced_pipeline(const char* shader)
60{
61 GpuPipeline pipeline;
62 pipeline.shader = shader;
63 pipeline.bindings = {{6 * sizeof(float), false}, {21 * sizeof(float), true}};
64 pipeline.attributes = {{0, 0, GpuVertexFormat::Float3, 0}, {1, 0, GpuVertexFormat::Float3, 3 * sizeof(float)}};
65 for (int column = 0; column < 5; ++column) {
66 pipeline.attributes.push_back({2 + column, 1, GpuVertexFormat::Float4, static_cast<std::uint32_t>(4 * column * sizeof(float))});
67 }
68 pipeline.attributes.push_back({7, 1, GpuVertexFormat::Float, 20 * sizeof(float)});
69 pipeline.depth_test = true;
70 pipeline.depth_write = true;
71 return pipeline;
72}

74GpuPipeline textured_quad_pipeline(const char* shader)
75{
76 GpuPipeline pipeline;
77 pipeline.shader = shader;
78 pipeline.bindings = {{5 * sizeof(float), false}};
79 pipeline.attributes = {{0, 0, GpuVertexFormat::Float3, 0}, {1, 0, GpuVertexFormat::Float2, 3 * sizeof(float)}};
80 pipeline.topology = GpuTopology::TriangleStrip;
81 pipeline.blend = GpuBlend::Alpha;
82 return pipeline;

94BrainFrameBuilder::BrainFrameBuilder()
95{
96 const auto index = [](BrainShader shader) {
97 return static_cast<std::size_t>(shader);
98 };
99 m_pipelines[index(BrainShader::Standard)] = m_resources.create_pipeline(surface_pipeline("standard"));
100 m_pipelines[index(BrainShader::Anatomical)] = m_resources.create_pipeline(surface_pipeline("anatomical"));
101 m_pipelines[index(BrainShader::ShowNormals)] = m_resources.create_pipeline(surface_pipeline("shownormals"));
102
103 // Holographic shells add their glow, back faces first; x-ray ones also show through other surfaces
104 GpuPipeline holographic = surface_pipeline("holographic");
105 holographic.blend = GpuBlend::Additive;
106 holographic.depth_write = false;
107 holographic.cull = GpuCull::Back;
108 GpuPipeline xray = holographic;
109 xray.depth_test = false;
110 m_pipelines[index(BrainShader::Holographic)] = m_resources.create_pipeline(holographic);
111 m_pipelines[index(BrainShader::XRay)] = m_resources.create_pipeline(xray);
112 holographic.cull = GpuCull::Front;
113 xray.cull = GpuCull::Front;
114 m_backFacePipelines[0] = m_resources.create_pipeline(std::move(holographic));
115 m_backFacePipelines[1] = m_resources.create_pipeline(std::move(xray));
116
117 GpuPipeline dipole = instanced_pipeline("dipole");
118 dipole.blend = GpuBlend::Alpha;
119 dipole.depth_write = false;
120 m_pipelines[index(BrainShader::Dipole)] = m_resources.create_pipeline(std::move(dipole));
121 m_pipelines[index(BrainShader::Electrode)] = m_resources.create_pipeline(instanced_pipeline("electrode"));
122
123 // MRI slices blend behind opaque geometry; the video disk lies on top of everything
124 GpuPipeline slice = textured_quad_pipeline("slice");
125 slice.depth_test = true;
126 m_slicePipeline = m_resources.create_pipeline(std::move(slice));
127 m_videoBillboardPipeline = m_resources.create_pipeline(textured_quad_pipeline("video_overlay"));
128
129 // The decal lies on its surface; the relief-following decal moves vertices inwards, so it is not depth tested
130 GpuPipeline decal = surface_pipeline("video_decal");
131 decal.blend = GpuBlend::Alpha;
132 decal.depth_write = false;
133 m_videoDecalPipeline = m_resources.create_pipeline(decal);
134 decal.shader = "video_decal_depth";
135 decal.depth_test = false;
136 m_videoDecalDepthPipeline = m_resources.create_pipeline(std::move(decal));
137
138 GpuPipeline overlay;
139 overlay.shader = "screen_overlay";
140 overlay.bindings = {{4 * sizeof(float), false}};
141 overlay.attributes = {{0, 0, GpuVertexFormat::Float2, 0}, {1, 0, GpuVertexFormat::Float2, 2 * sizeof(float)}};
142 overlay.topology = GpuTopology::TriangleStrip;
143 overlay.blend = GpuBlend::Alpha;
144 m_overlayPipeline = m_resources.create_pipeline(std::move(overlay));
145 m_quad = m_resources.create_buffer(GpuBufferUsage::Vertex);
146 m_resources.set_buffer_values(m_quad, std::span<const float>(kQuad));
147}

165GpuHandle BrainFrameBuilder::stream_buffer(Mesh& mesh, int slot, GpuBufferUsage usage, const GpuStreamView& stream)
166{
167 const auto at = static_cast<std::size_t>(slot);
168 if (stream.bytes.empty()) {
169 return 0;
170 }
171 if (mesh.buffers[at] == 0) {
172 mesh.buffers[at] = m_resources.create_buffer(usage);
173 }
174 if (!mesh.uploaded[at] || mesh.generations[at] != stream.generation) {
175 m_resources.set_buffer_data(mesh.buffers[at], stream.bytes);
176 mesh.generations[at] = stream.generation;
177 mesh.uploaded[at] = true;
178 }
179 return mesh.buffers[at];
180}

184bool BrainFrameBuilder::surface_buffers(std::uint64_t key, const BrainSurfaceStreams& streams, GpuDraw& draw)
185{
186 Mesh& mesh = m_meshes[key];
187 mesh.last_frame = m_frameIndex;
188 const GpuHandle geometry = stream_buffer(mesh, 0, GpuBufferUsage::Vertex, streams.geometry);
189 const GpuHandle colors = stream_buffer(mesh, 1, GpuBufferUsage::Vertex, streams.colors);
190 const GpuHandle annotations = stream_buffer(mesh, 2, GpuBufferUsage::Vertex, streams.annotations);
191 const GpuHandle indices = stream_buffer(mesh, 3, GpuBufferUsage::Index32, streams.indices);
192 if (geometry == 0 || colors == 0 || annotations == 0 || indices == 0) {
193 return false;
194 }
195 draw.vertex_buffers = {geometry, colors, annotations};
196 draw.index_buffer = indices;
197 draw.count = streams.index_count;
198 return true;
199}

203GpuHandle BrainFrameBuilder::texture(std::uint64_t key, const BrainTextureView& view, bool mipmaps)
204{
205 if (!view.image || view.image->width <= 0 || view.image->height <= 0) {
206 return 0;
207 }
208 Texture& texture = m_textures[key];
209 texture.last_frame = m_frameIndex;
210 if (texture.handle == 0) {
211 GpuTexture created;
212 created.content = *view.image;
213 created.mipmaps = mipmaps;
214 texture.handle = m_resources.create_texture(std::move(created));
215 texture.revision = view.revision;
216 } else if (texture.revision != view.revision) {
217 m_resources.set_texture(texture.handle, *view.image);
218 texture.revision = view.revision;
219 }
220 return texture.handle;
221}

225GpuHandle BrainFrameBuilder::quad_buffer(std::uint64_t key, std::span<const float, 20> quad)
226{
227 Mesh& mesh = m_meshes[key];
228 mesh.last_frame = m_frameIndex;
229 if (mesh.buffers[0] == 0) {
230 mesh.buffers[0] = m_resources.create_buffer(GpuBufferUsage::Vertex);
231 }
232 const GpuBuffer* current = m_resources.buffer(mesh.buffers[0]);
233 const std::span<const std::byte> bytes = std::as_bytes(quad);
234 if (!current || current->data.size() != bytes.size() || !std::equal(bytes.begin(), bytes.end(), current->data.begin())) {
235 m_resources.set_buffer_data(mesh.buffers[0], bytes);
236 }
237 return mesh.buffers[0];
238}

290bool BrainFrameBuilder::draw_video_decal(std::uint64_t key, const BrainSurfaceStreams& streams, std::uint64_t textureKey, const BrainTextureView& frame,
291 const BrainTextureView& depth, const BrainVideoDecalUniforms& uniforms, const GpuRect& viewport)
292{
293 if (streams.index_count == 0 || m_frame.passes.empty()) {
294 return false;
295 }
296 const GpuHandle video = texture(textureKey, frame, false);
297 if (video == 0) {
298 return false;
299 }
300 GpuDraw draw;
301 if (!surface_buffers(key, streams, draw)) {
302 return false;
303 }
304 // The depth texture of a key sits at its complement; the vertex shader reads a blurred mip level of it
305 const GpuHandle relief = texture(~textureKey, depth, true);
306 GpuUniformBlock block;
307 block.binding = 0;
308 const std::size_t size = relief != 0 ? sizeof(BrainVideoDecalUniforms) : kVideoDecalBlockSize;
309 block.data.resize(size);
310 std::memcpy(block.data.data(), &uniforms, size);
311 draw.uniforms.push_back(std::move(block));
312 draw.textures = {{1, video}};
313 if (relief != 0) {
314 draw.textures.push_back({2, relief});
315 }
316 draw.pipeline = relief != 0 ? m_videoDecalDepthPipeline : m_videoDecalPipeline;
317 draw.viewport = viewport;
318 draw.scissor = viewport;
319 m_frame.passes.back().draws.push_back(std::move(draw));
320 return true;
321}

325bool BrainFrameBuilder::draw_surface(std::uint64_t key, const BrainSurfaceStreams& streams, BrainShader shader, const BrainSurfaceUniforms& uniforms,
326 const GpuRect& viewport)
327{
328 if (shader == BrainShader::Dipole || shader == BrainShader::Electrode || streams.index_count == 0 || m_frame.passes.empty()) {
329 return false;
330 }
331 GpuDraw draw;
332 if (!surface_buffers(key, streams, draw)) {
333 return false;
334 }
335 draw.set_uniforms(0, uniforms);
336 draw.viewport = viewport;
337 draw.scissor = viewport;
338 std::vector<GpuDraw>& draws = m_frame.passes.back().draws;
339 if (shader == BrainShader::Holographic || shader == BrainShader::XRay) {
340 GpuDraw back = draw;
341 back.pipeline = m_backFacePipelines[shader == BrainShader::Holographic ? 0 : 1];
342 draws.push_back(std::move(back));
343 }
344 draw.pipeline = m_pipelines[static_cast<std::size_t>(shader)];
345 draws.push_back(std::move(draw));
346 return true;
347}

351bool BrainFrameBuilder::draw_instanced(std::uint64_t key, const BrainInstancedStreams& streams, BrainShader shader, const BrainInstanceUniforms& uniforms,
352 const GpuRect& viewport)
353{
354 if ((shader != BrainShader::Dipole && shader != BrainShader::Electrode) || streams.index_count == 0 || streams.instance_count == 0 || m_frame.passes.empty()) {
355 return false;
356 }
357 Mesh& mesh = m_meshes[key];
358 mesh.last_frame = m_frameIndex;
359 const GpuHandle vertices = stream_buffer(mesh, 0, GpuBufferUsage::Vertex, streams.vertices);
360 const GpuHandle instances = stream_buffer(mesh, 1, GpuBufferUsage::Vertex, streams.instances);
361 const GpuHandle indices = stream_buffer(mesh, 2, GpuBufferUsage::Index32, streams.indices);
362 if (vertices == 0 || instances == 0 || indices == 0) {
363 return false;
364 }
365 GpuDraw draw;
366 draw.pipeline = m_pipelines[static_cast<std::size_t>(shader)];
367 draw.vertex_buffers = {vertices, instances};
368 draw.index_buffer = indices;
369 draw.count = streams.index_count;
370 draw.instance_count = streams.instance_count;
371 draw.set_uniforms(0, uniforms);
372 draw.viewport = viewport;
373 draw.scissor = viewport;
374 m_frame.passes.back().draws.push_back(std::move(draw));
375 return true;

415GpuFrame BrainFrameBuilder::finish()
416{
417 for (auto it = m_meshes.begin(); it != m_meshes.end();) {
418 if (m_frameIndex - it->second.last_frame > kUnusedFrames) {
419 for (const GpuHandle buffer : it->second.buffers) {
420 m_resources.release(buffer);
421 }
422 it = m_meshes.erase(it);
423 } else {
424 ++it;
425 }
426 }
427 for (auto it = m_textures.begin(); it != m_textures.end();) {
428 if (m_frameIndex - it->second.last_frame > kUnusedFrames) {
429 m_resources.release(it->second.handle);
430 it = m_textures.erase(it);
431 } else {
432 ++it;
433 }
434 }
435 return std::exchange(m_frame, GpuFrame{});
436}

440void BrainFrameBuilder::release(std::uint64_t key)
441{
442 if (const auto found = m_meshes.find(key); found != m_meshes.end()) {
443 for (const GpuHandle buffer : found->second.buffers) {
444 m_resources.release(buffer);
445 }
446 m_meshes.erase(found);
447 }
448 for (const std::uint64_t textureKey : {key, ~key}) {
449 if (const auto found = m_textures.find(textureKey); found != m_textures.end()) {
450 m_resources.release(found->second.handle);
451 m_textures.erase(found);
452 }
453 }
