963 // Reads one value in the stream's byte order; a short read yields 0 and marks the stream failed
964 template<mne::Swappable T>
965 void read(T& value)
966 {
967 if (m_bytes->read(&value, sizeof(T)) != sizeof(T)) {
968 value = T{};
969 m_readFailed = true;
970 return;
971 }
972 if (m_littleEndian != (std::endian::native == std::endian::little))
973 value = mne::byte_swap(value);
974 }
975
976 template<mne::Swappable T>
977 void write(T value)
978 {
979 if (m_littleEndian != (std::endian::native == std::endian::little))
980 value = mne::byte_swap(value);
981 m_bytes->write(&value, sizeof(T));
982 }
