241static QImage tightlyPackedRgba(const QImage& source)
242{
243 if (source.isNull())
244 return {};
245
246 QImage rgba = source.convertToFormat(QImage::Format_RGBA8888);
247 const qsizetype tightStride = qsizetype(rgba.width()) * 4;
248 if (rgba.bytesPerLine() == tightStride)
249 return rgba.copy();
250
251 QImage packed(rgba.size(), QImage::Format_RGBA8888);
252 for (int y = 0; y < rgba.height(); ++y) {
253 memcpy(packed.scanLine(y), rgba.constScanLine(y), size_t(tightStride));
254 }
255 return packed;
256}

318int BrainRenderer::Impl::claimUniformSlot(const char* caller)
319{
320 if (insidePass(caller))
321 return -1;
322 const int offset = currentUniformOffset;
323 // Once a pass has begun the buffer can no longer grow this frame
324 const qint64 capacity = passBegunThisFrame ? qint64(uniformBuffer->size()) : qint64(kMaxUniformSlots) * uniformSlotStride;
325 if (offset + uniformSlotStride > capacity) {
326 if (!overflowReported) {
327 mne::Log::warning("BrainRenderer: uniform buffer full at {}; further draws this frame are skipped", caller);
328 overflowReported = true;
329 }
330 return -1;
331 }
332 currentUniformOffset += uniformSlotStride;
333 if (std::ssize(uniformShadow) < currentUniformOffset)
334 uniformShadow.resize(static_cast<std::size_t>(currentUniformOffset));
335 return offset;
336}

340void BrainRenderer::Impl::flushUniforms(QRhiResourceUpdateBatch* u)
341{
342 if (currentUniformOffset <= flushedUniformBytes)
343 return;
344 if (quint32(currentUniformOffset) > uniformBuffer->size()) {
345 int slotCount = int(uniformBuffer->size()) / uniformSlotStride;
346 while (slotCount * uniformSlotStride < currentUniformOffset)
347 slotCount *= 2;
348 uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, quint32(slotCount * uniformSlotStride)));
349 uniformBuffer->create();
350 srb->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage, uniformBuffer.get(), kUniformBlockSize)});
351 srb->create();
352 flushedUniformBytes = 0;
353 }
354 u->updateDynamicBuffer(uniformBuffer.get(), quint32(flushedUniformBytes), quint32(currentUniformOffset - flushedUniformBytes),
355 uniformShadow.data() + flushedUniformBytes);
356 flushedUniformBytes = currentUniformOffset;
357}

384QRhiBuffer* BrainRenderer::Impl::upload(GpuMesh& m, int slot, QRhiBuffer::UsageFlags usage, const GpuStream& stream, QRhiResourceUpdateBatch* u)
385{
386 auto& buffer = m.buffers[slot];
387 if (!stream.data || stream.bytes == 0) {
388 buffer.reset();
389 return nullptr;
390 }
391 // Static buffers are updated in place; only a size change reallocates
392 if (!buffer || buffer->size() != stream.bytes) {
393 buffer.reset(rhi->newBuffer(QRhiBuffer::Static, usage, stream.bytes));
394 buffer->create();
395 m.generations[slot] = std::numeric_limits<quint64>::max();
396 }
397 if (m.generations[slot] != stream.generation) {
398 u->uploadStaticBuffer(buffer.get(), 0, stream.bytes, stream.data);
399 m.generations[slot] = stream.generation;
400 }
401 return buffer.get();
402}

434void BrainRenderer::Impl::createResources(QRhi* newRhi, QRhiRenderPassDescriptor* rp, int sampleCount)
435{
436 rhi = newRhi;
437 uniformSlotStride = rhi->ubufAligned(kUniformBlockSize);
438
439 if (!uniformBuffer) {
440 uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, kInitialUniformSlots * uniformSlotStride));
441 uniformBuffer->create();
442 }
443
444 // Create SRB
445 if (!srb) {
446 srb.reset(rhi->newShaderResourceBindings());
447 srb->setBindings({// Use dynamic offset for the uniform buffer.
448 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage, uniformBuffer.get(), kUniformBlockSize)});
449 srb->create();
450 }
451
452
453 // List of modes to initialize
454 std::vector<ShaderMode> modes = {Standard, Holographic, Anatomical, Dipole, XRay, ShowNormals};
455
456 for (ShaderMode mode : modes) {
457 QString vert = (mode == Holographic || mode == XRay) ? ":/holographic.vert.qsb" : (mode == Anatomical) ? ":/anatomical.vert.qsb"
458 : (mode == Dipole) ? ":/dipole.vert.qsb"
459 : (mode == ShowNormals) ? ":/shownormals.vert.qsb"
460 : ":/standard.vert.qsb";
461
462 QString frag = (mode == Holographic || mode == XRay) ? ":/holographic.frag.qsb" : (mode == Anatomical) ? ":/anatomical.frag.qsb"
463 : (mode == Dipole) ? ":/dipole.frag.qsb"
464 : (mode == ShowNormals) ? ":/shownormals.frag.qsb"
465 : ":/standard.frag.qsb";
466
467 QShader vS = loadShader(vert);
468 QShader fS = loadShader(frag);
469
470 if (!vS.isValid() || !fS.isValid()) {
471 mne::Log::warning("BrainRenderer: Could not load shaders for mode {} {} {}", static_cast<int>(mode), vert.toStdString(), frag.toStdString());
472 continue;
473 }
474
475 // Setup Pipeline
476 auto pipeline = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
477
478 QRhiGraphicsPipeline::TargetBlend blend;
479 if (mode == Holographic || mode == XRay) {
480 blend.enable = true;
481 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
482 blend.dstColor = QRhiGraphicsPipeline::One;
483 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
484 blend.dstAlpha = QRhiGraphicsPipeline::One;
485 } else if (mode == Dipole) {
486 blend.enable = true;
487 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
488 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
489 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
490 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
491 }
492
493 auto setup = [&](QRhiGraphicsPipeline* p, QRhiGraphicsPipeline::CullMode cull) {
494 p->setShaderStages({{QRhiShaderStage::Vertex, vS}, {QRhiShaderStage::Fragment, fS}});
495
496 p->setVertexInputLayout(mode == Dipole ? instancedInputLayout() : surfaceInputLayout());
497 p->setShaderResourceBindings(srb.get());
498 p->setRenderPassDescriptor(rp);
499 p->setSampleCount(sampleCount);
500 p->setCullMode(cull);
501 if (mode == Holographic) {
502 p->setTargetBlends({blend});
503 p->setDepthTest(true);
504 p->setDepthWrite(false);
505 } else if (mode == XRay) {
506 p->setTargetBlends({blend});
507 p->setDepthTest(false); // Disable Depth Test to see through head
508 p->setDepthWrite(false);
509 } else if (mode == Dipole) {
510 p->setTargetBlends({blend});
511 p->setCullMode(QRhiGraphicsPipeline::None);
512 p->setDepthTest(true);
513 p->setDepthWrite(false);
514 } else {
515 p->setDepthTest(true);
516 p->setDepthWrite(true);
517 }
518 p->setFlags(QRhiGraphicsPipeline::UsesScissor);
519 p->create();
520 };
521
522 if (mode == Holographic || mode == XRay) { // Handle XRay back-faces same as Holographic
523 auto pipelineBack = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
524 setup(pipelineBack.get(), QRhiGraphicsPipeline::Front);
525 pipelinesBackColor[mode] = std::move(pipelineBack);
526 setup(pipeline.get(), QRhiGraphicsPipeline::Back); // Front faces
527 } else {
528 // Culling: None (Double-sided) to be safe for FreeSurfer meshes
529 setup(pipeline.get(), QRhiGraphicsPipeline::None);
530 }
531 pipelines[mode] = std::move(pipeline);
532 }
533
534 // Electrodes: opaque and depth-written, unlike the translucent dipole glyphs
535 const QShader evS = loadShader(QStringLiteral(":/electrode.vert.qsb"));
536 const QShader efS = loadShader(QStringLiteral(":/electrode.frag.qsb"));
537 electrodePipeline.reset();
538 if (evS.isValid() && efS.isValid()) {
539 electrodePipeline.reset(rhi->newGraphicsPipeline());
540 electrodePipeline->setShaderStages({{QRhiShaderStage::Vertex, evS}, {QRhiShaderStage::Fragment, efS}});
541 electrodePipeline->setVertexInputLayout(instancedInputLayout());
542 electrodePipeline->setShaderResourceBindings(srb.get());
543 electrodePipeline->setRenderPassDescriptor(rp);
544 electrodePipeline->setSampleCount(sampleCount);
545 electrodePipeline->setCullMode(QRhiGraphicsPipeline::None);
546 electrodePipeline->setDepthTest(true);
547 electrodePipeline->setDepthWrite(true);
548 electrodePipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);
549 electrodePipeline->create();
550 } else {
551 mne::Log::warning("BrainRenderer: Could not load the electrode shaders");
552 }
553
554 resourcesDirty = false;
555}

561void BrainRenderer::ensureRenderTargets(QRhi* rhi, QRhiTexture* colorTex, const QSize& pixelSize)
562{
563 // Rebuild when size changes OR when the backing texture changes
564 // (QRhiWidget may return a different colorTexture() each frame).
565 if (d->rtClear && d->rtSize == pixelSize && d->rtColorTex == colorTex)
566 return;
567
568 d->rtSize = pixelSize;
569 d->rtColorTex = colorTex;
570
571 // Shared depth-stencil buffer
572 d->dsBuffer.reset(rhi->newRenderBuffer(QRhiRenderBuffer::DepthStencil, pixelSize));
573 d->dsBuffer->create();
574
575 QRhiColorAttachment colorAtt(colorTex);
576 QRhiTextureRenderTargetDescription desc(colorAtt);
577 desc.setDepthStencilBuffer(d->dsBuffer.get());
578
579 // RT 1: Clearing (no preserve flags) — used for the first pass of each frame
580 d->rtClear.reset(rhi->newTextureRenderTarget(desc));
581 d->rpClear.reset(d->rtClear->newCompatibleRenderPassDescriptor());
582 d->rtClear->setRenderPassDescriptor(d->rpClear.get());
583 d->rtClear->create();
584
585 // RT 2: Preserving (load previous contents) — used for passes 2+
586 d->rtPreserve.reset(rhi->newTextureRenderTarget(desc,
587 QRhiTextureRenderTarget::PreserveColorContents | QRhiTextureRenderTarget::PreserveDepthStencilContents));
588 d->rpPreserve.reset(d->rtPreserve->newCompatibleRenderPassDescriptor());
589 d->rtPreserve->setRenderPassDescriptor(d->rpPreserve.get());
590 d->rtPreserve->create();
591}

615{

617 d->flushedUniformBytes = 0;
618 d->overflowReported = false;
619 d->passBegunThisFrame = false;

692void BrainRenderer::prepareScreenOverlay(QRhi* rhi, QRhiResourceUpdateBatch* u, const QImage& overlay)
693{
694 auto& o = d->screenOverlay;
695 o.active = false;
696 QRhiRenderTarget* rt = d->clearTarget();
697 if (overlay.isNull() || !rhi || !u || !rt || o.failed || d->insidePass("prepareScreenOverlay"))
698 return;
699
700 if (!o.pipeline) {
701 const QShader vShader = loadShader(QStringLiteral(":/screen_overlay.vert.qsb"));
702 const QShader fShader = loadShader(QStringLiteral(":/screen_overlay.frag.qsb"));
703 if (!vShader.isValid() || !fShader.isValid()) {
704 mne::Log::warning("BrainRenderer: could not load screen overlay shaders");
705 o.failed = true;
706 return;
707 }
708
709 // Full-target quad as a strip; image row 0 is the top of the target on every backend
710 const float top = rhi->isYUpInNDC() ? 1.0f : -1.0f;
711 const float quad[16] = {-1.0f, top, 0.0f, 0.0f, -1.0f, -top, 0.0f, 1.0f,
712 1.0f, top, 1.0f, 0.0f, 1.0f, -top, 1.0f, 1.0f};
713 o.vertexBuffer.reset(rhi->newBuffer(QRhiBuffer::Immutable, QRhiBuffer::VertexBuffer, sizeof(quad)));
714 o.vertexBuffer->create();
715 u->uploadStaticBuffer(o.vertexBuffer.get(), quad);
716
717 o.sampler.reset(rhi->newSampler(QRhiSampler::Nearest, QRhiSampler::Nearest, QRhiSampler::None,
718 QRhiSampler::ClampToEdge, QRhiSampler::ClampToEdge));
719 o.sampler->create();
720
721 o.texture.reset(rhi->newTexture(QRhiTexture::RGBA8, overlay.size()));
722 o.texture->create();
723 o.srb.reset(rhi->newShaderResourceBindings());
724 o.srb->setBindings({QRhiShaderResourceBinding::sampledTexture(0, QRhiShaderResourceBinding::FragmentStage,
725 o.texture.get(), o.sampler.get())});
726 o.srb->create();
727
728 QRhiVertexInputLayout layout;
729 layout.setBindings({{4 * sizeof(float)}});
730 layout.setAttributes({{0, 0, QRhiVertexInputAttribute::Float2, 0},
731 {0, 1, QRhiVertexInputAttribute::Float2, 2 * sizeof(float)}});
732
733 QRhiGraphicsPipeline::TargetBlend blend;
734 blend.enable = true;
735 blend.srcColor = QRhiGraphicsPipeline::One;
736 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
737 blend.srcAlpha = QRhiGraphicsPipeline::One;
738 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
739
740 o.pipeline.reset(rhi->newGraphicsPipeline());
741 o.pipeline->setShaderStages({{QRhiShaderStage::Vertex, vShader}, {QRhiShaderStage::Fragment, fShader}});
742 o.pipeline->setVertexInputLayout(layout);
743 o.pipeline->setTopology(QRhiGraphicsPipeline::TriangleStrip);
744 o.pipeline->setShaderResourceBindings(o.srb.get());
745 o.pipeline->setRenderPassDescriptor(rt->renderPassDescriptor());
746 o.pipeline->setSampleCount(rt->sampleCount());
747 o.pipeline->setTargetBlends({blend});
748 o.pipeline->setDepthTest(false);
749 o.pipeline->setDepthWrite(false);
750 o.pipeline->create();
751 o.uploadedKey = 0;
752 } else if (o.texture->pixelSize() != overlay.size()) {
753 o.texture->setPixelSize(overlay.size());
754 o.texture->create();
755 o.srb->create();
756 o.uploadedKey = 0;
757 }
758
759 if (o.uploadedKey != overlay.cacheKey()) {
760 // RGBA8 rows are always 4-byte aligned, so the converted image is tightly packed
761 u->uploadTexture(o.texture.get(), overlay.convertToFormat(QImage::Format_RGBA8888_Premultiplied));
762 o.uploadedKey = overlay.cacheKey();
763 }
764 o.active = true;
765}

1061
1062int BrainRenderer::prepareVideoOverlayDraw(QRhiResourceUpdateBatch* u,
1063 const SceneData& data,
1064 VideoOverlay* overlay,
1065 BrainSurface* surface)
1066{
1067 if (!overlay || !overlay->isEnabled() || !overlay->hasFrame())
1068 return -1;
1069
1070 auto& k = d->videoOverlay;
1071 if (!k.initialized || (surface && !k.surfacePipeline))
1072 return -1;
1073 if (d->insidePass("prepareVideoOverlayDraw"))
1074 return -1;
1075 if (surface && !d->uploadSurface(surface, u))
1076 return -1;
1077
1078 const int uniformOffset = k.currentUniformOffset;
1079 if (static_cast<quint32>(uniformOffset + kUniformBlockSize) > k.uniformBuffer->size())
1080 return -1;
1081 k.currentUniformOffset += k.uniformSlotStride;
1082
1083 if (!surface) {
1084 // ── Billboard the quad to face the camera ──────────────────────
1085 const QVector3D centre = overlay->focusPosition();
1086 const float sizeM = overlay->sizeMeters();
1087
1088 // Preserve the video frame's aspect ratio: sizeMeters controls the
1089 // width, and the height is derived from the frame dimensions.
1090 const QImage& frm = overlay->frame();
1091 const float aspect = (frm.height() > 0)
1092 ? static_cast<float>(frm.width()) / frm.height()
1093 : 1.0f;
1094 const float halfW = 0.5f * sizeM;
1095 const float halfH = (aspect > 0.0f) ? halfW / aspect : halfW;
1096
1097 QVector3D viewDir = centre - data.cameraPos;
1098 if (viewDir.lengthSquared() < 1e-12f)
1099 viewDir = QVector3D(0, 0, -1);
1100 viewDir.normalize();
1101
1102 // Use the upHint (tracker→objective axis) when available so that
1103 // the quad's long edge is perpendicular to the optical axis.
1104 const QVector3D hint = overlay->upHint();
1105 QVector3D up;
1106 if (hint.lengthSquared() > 1e-8f) {
1107 // Project the hint onto the plane perpendicular to viewDir
1108 up = (hint - QVector3D::dotProduct(hint, viewDir) * viewDir).normalized();
1109 if (up.lengthSquared() < 1e-8f)
1110 up = QVector3D(0.0f, 0.0f, 1.0f);
1111 } else {
1112 QVector3D worldUp(0.0f, 0.0f, 1.0f);
1113 if (std::abs(QVector3D::dotProduct(viewDir, worldUp)) > 0.95f)
1114 worldUp = QVector3D(0.0f, 1.0f, 0.0f);
1115 up = QVector3D::crossProduct(
1116 QVector3D::crossProduct(viewDir, worldUp), viewDir)
1117 .normalized();
1118 }
1119 QVector3D right = QVector3D::crossProduct(viewDir, up).normalized();
1120
1121 const QVector3D c00 = centre - right * halfW - up * halfH;
1122 const QVector3D c10 = centre + right * halfW - up * halfH;
1123 const QVector3D c01 = centre - right * halfW + up * halfH;
1124 const QVector3D c11 = centre + right * halfW + up * halfH;
1125
1126 const float verts[4 * 5] = {
1127 c00.x(), c00.y(), c00.z(), 0.0f, 1.0f, // image y is flipped vs uv
1128 c10.x(), c10.y(), c10.z(), 1.0f, 1.0f,
1129 c01.x(), c01.y(), c01.z(), 0.0f, 0.0f,
1130 c11.x(), c11.y(), c11.z(), 1.0f, 0.0f};
1131 // Each slot has its own quad: every pane faces its own camera
1132 static_assert(sizeof(verts) == kVideoQuadBytes);
1133 u->updateDynamicBuffer(k.vertexBuffer.get(), quint32(uniformOffset / k.uniformSlotStride) * kVideoQuadBytes, sizeof(verts), verts);
1134
1135 // ── Uniforms (mat4 mvp, vec4 borderColor, float opacity, 3 pad) ─
1136 struct
1137 {
1138 float mvp[16];
1139 float borderColor[4];
1140 float opacity;
1141 float pad0, pad1, pad2;
1142 } ub;
1143 memcpy(ub.mvp, data.mvp.constData(), 64);
1144 ub.borderColor[0] = 0.0f;
1145 ub.borderColor[1] = 0.85f;
1146 ub.borderColor[2] = 1.0f;
1147 ub.borderColor[3] = 1.0f;
1148 ub.opacity = overlay->opacity();
1149 ub.pad0 = ub.pad1 = ub.pad2 = 0.0f;
1150 u->updateDynamicBuffer(k.uniformBuffer.get(), uniformOffset, sizeof(ub), &ub);
1151 return uniformOffset;
1152 }
1153
1154 // Approximate the local scalp normal from the focus position vector.
1155 QVector3D localNormal = overlay->focusPosition();
1156 if (localNormal.lengthSquared() < 1e-12f) {
1157 localNormal = QVector3D(0.0f, 0.0f, 1.0f);
1158 }
1159 localNormal.normalize();
1160
1161 // Use the tracker-derived up hint when available — this keeps the
1162 // decal orientation locked to the physical microscope and avoids
1163 // the 180° flip singularity that occurs with a fixed reference axis.
1164 QVector3D referenceUp = overlay->upHint();
1165 if (referenceUp.lengthSquared() < 1e-6f) {
1166 // Fallback when no tracker up is available.
1167 referenceUp = QVector3D(0.0f, 0.0f, 1.0f);
1168 if (std::abs(QVector3D::dotProduct(localNormal, referenceUp)) > 0.95f) {
1169 referenceUp = QVector3D(1.0f, 0.0f, 0.0f);
1170 }
1171 }
1172 // Orthogonalise against the surface normal so the frame is tangent.
1173 referenceUp = (referenceUp - QVector3D::dotProduct(referenceUp, localNormal) * localNormal);
1174 if (referenceUp.lengthSquared() < 1e-12f) {
1175 referenceUp = QVector3D(0.0f, 0.0f, 1.0f);
1176 }
1177 referenceUp.normalize();
1178
1179 const QVector3D axisV = referenceUp;
1180 const QVector3D axisU = QVector3D::crossProduct(axisV, localNormal).normalized();
1181
1182 const bool useDepth = d->videoOverlayUsesDepth(overlay);

1269void BrainRenderer::prepareSlice(QRhi* rhi,
1270 QRhiResourceUpdateBatch* u,
1271 mne::viz::qt::SliceObject* slice,
1272 int slotIndex)
1273{

1275 return;
1276 if (slotIndex < 0 || slotIndex >= Impl::kMaxSliceSlots)
1277 return;
1278
1279 auto& k = d->sliceRes;
1280 auto& sliceSlot = k.sliceSlots[slotIndex];
1281
1282 if (!slice) {
1283 sliceSlot.visible = false;
1284 return;
1285 }
1286
1287 const QImage& img = slice->image();
1288 if (img.isNull()) {
1289 sliceSlot.visible = false;
1290 return;
1291 }
1292
1293 // ── Lazy one-time resource creation ─────────────────────────────
1294 if (!k.initialized) {
1295 const QShader vShader = loadShader(QStringLiteral(":/slice.vert.qsb"));
1296 const QShader fShader = loadShader(QStringLiteral(":/slice.frag.qsb"));
1297 if (!vShader.isValid() || !fShader.isValid()) {
1298 mne::Log::warning("BrainRenderer: could not load MRI slice shaders");
1299 return;
1300 }
1301
1302 k.uniformSlotStride = rhi->ubufAligned(kUniformBlockSize);
1303 k.uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
1304 QRhiBuffer::UniformBuffer,
1305 64 * k.uniformSlotStride));
1306 k.uniformBuffer->create();
1307
1308 // Index buffer (shared by all 3 slots): 2 triangles = 6 indices
1309 k.indexBuffer.reset(rhi->newBuffer(QRhiBuffer::Immutable,
1310 QRhiBuffer::IndexBuffer,
1311 6 * sizeof(quint32)));
1312 k.indexBuffer->create();
1313
1314 k.sampler.reset(rhi->newSampler(QRhiSampler::Linear, QRhiSampler::Linear,
1315 QRhiSampler::None,
1316 QRhiSampler::ClampToEdge,
1317 QRhiSampler::ClampToEdge));
1318 k.sampler->create();
1319
1320 // Per-slot resources: vertex buffer + texture + SRB
1321 for (int i = 0; i < Impl::kMaxSliceSlots; ++i) {
1322 constexpr int kVbSize = 4 * 5 * sizeof(float); // 4 verts × (3 pos + 2 uv)
1323 k.vertexBuffer[i].reset(rhi->newBuffer(QRhiBuffer::Dynamic,
1324 QRhiBuffer::VertexBuffer, kVbSize));
1325 k.vertexBuffer[i]->create();
1326
1327 // 1×1 placeholder texture — real size set on first data
1328 k.sliceSlots[i].texture.reset(rhi->newTexture(QRhiTexture::R8, QSize(1, 1)));
1329 k.sliceSlots[i].texture->create();
1330 k.sliceSlots[i].textureSize = QSize(1, 1);
1331
1332 k.srb[i].reset(rhi->newShaderResourceBindings());
1333 k.srb[i]->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
1334 QRhiShaderResourceBinding::VertexStage |
1335 QRhiShaderResourceBinding::FragmentStage,
1336 k.uniformBuffer.get(), kUniformBlockSize),
1337 QRhiShaderResourceBinding::sampledTexture(1,
1338 QRhiShaderResourceBinding::FragmentStage,
1339 k.sliceSlots[i].texture.get(), k.sampler.get())});
1340 k.srb[i]->create();
1341 }
1342
1343 // Pipeline: alpha-blended, depth-test on, depth-write off
1344 QRhiGraphicsPipeline::TargetBlend blend;
1345 blend.enable = true;
1346 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
1347 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
1348 blend.srcAlpha = QRhiGraphicsPipeline::One;
1349 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
1350
1351 k.pipeline.reset(rhi->newGraphicsPipeline());
1352 k.pipeline->setShaderStages({{QRhiShaderStage::Vertex, vShader},
1353 {QRhiShaderStage::Fragment, fShader}});
1354 QRhiVertexInputLayout il;
1355 il.setBindings({{5 * sizeof(float)}});
1356 il.setAttributes({{0, 0, QRhiVertexInputAttribute::Float3, 0},
1357 {0, 1, QRhiVertexInputAttribute::Float2, 3 * sizeof(float)}});
1358 k.pipeline->setVertexInputLayout(il);
1359 k.pipeline->setShaderResourceBindings(k.srb[0].get());
1360 k.pipeline->setRenderPassDescriptor(d->clearTarget()->renderPassDescriptor());
1361 k.pipeline->setSampleCount(d->clearTarget()->sampleCount());
1362 k.pipeline->setCullMode(QRhiGraphicsPipeline::None);
1363 k.pipeline->setTargetBlends({blend});
1364 k.pipeline->setDepthTest(true);
1365 k.pipeline->setDepthWrite(false);
1366 k.pipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);
1367 k.pipeline->create();
1368
1369 k.initialized = true;
1370 }
1371
1372 // ── Index buffer (one-shot upload) ──────────────────────────────
1373 if (!k.indexUploaded) {
1374 const quint32 idx[6] = {0, 1, 2, 2, 1, 3};
1375 u->uploadStaticBuffer(k.indexBuffer.get(), idx);
1376 k.indexUploaded = true;
1377 }
1378
1379 // ── Texture (re-create on size change, upload data) ─────────────
1380 // Convert to R8 (grayscale) for the slice shader
1381 QImage gray = img.convertToFormat(QImage::Format_Grayscale8);
1382 const QSize imgSize = gray.size();
1383
1384 if (imgSize != sliceSlot.textureSize) {
1385 sliceSlot.texture.reset(rhi->newTexture(QRhiTexture::R8, imgSize));
1386 sliceSlot.texture->create();
1387 sliceSlot.textureSize = imgSize;
1388 // Rebuild SRB to point at new texture
1389 k.srb[slotIndex]->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
1390 QRhiShaderResourceBinding::VertexStage |
1391 QRhiShaderResourceBinding::FragmentStage,
1392 k.uniformBuffer.get(), kUniformBlockSize),
1393 QRhiShaderResourceBinding::sampledTexture(1,
1394 QRhiShaderResourceBinding::FragmentStage,
1395 sliceSlot.texture.get(), k.sampler.get())});
1396 k.srb[slotIndex]->create();
1397 }
1398
1399 // Upload texture — R8 needs tightly-packed row data
1400 {
1401 const qsizetype tightStride = qsizetype(gray.width());
1402 std::vector<std::byte> texData;
1403 texData.resize(gray.height() * tightStride);
1404 for (int y = 0; y < gray.height(); ++y) {
1405 memcpy(texData.data() + y * tightStride,
1406 gray.constScanLine(y),
1407 size_t(tightStride));
1408 }
1409 QRhiTextureSubresourceUploadDescription sub(texData.data(), static_cast<uint32_t>(texData.size()));
1410 sub.setSourceSize(imgSize);
1411 QRhiTextureUploadDescription desc({0, 0, sub});
1412 u->uploadTexture(sliceSlot.texture.get(), desc);
1413 }
1414
1415 // ── Vertex data ─────────────────────────────────────────────────
1416 slice->generateQuadVertices(sliceSlot.vertices);
1417 u->updateDynamicBuffer(k.vertexBuffer[slotIndex].get(), 0,
1418 static_cast<quint32>(sliceSlot.vertices.size() * sizeof(float)),
1419 sliceSlot.vertices.data());
1420
1421 sliceSlot.opacity = slice->opacity();
1422 sliceSlot.windowCenter = slice->windowCenter();
1423 sliceSlot.windowWidth = slice->windowWidth();
1424 sliceSlot.visible = true;
1425}
