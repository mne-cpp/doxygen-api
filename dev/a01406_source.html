71 bool push(const T& record) noexcept
72 {
73 std::size_t position = m_enqueue.load(std::memory_order_relaxed);
74 Cell* cell = nullptr;
75 while (true) {
76 cell = &m_cells[position & (Capacity - 1)];
77 const std::size_t sequence = cell->sequence.load(std::memory_order_acquire);
78 const auto difference = static_cast<std::intptr_t>(sequence) - static_cast<std::intptr_t>(position);
79 if (difference == 0) {
80 if (m_enqueue.compare_exchange_weak(position, position + 1, std::memory_order_relaxed)) {
81 break;
82 }
