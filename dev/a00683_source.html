47bool RayPicker::unproject(const QPoint& screenPos,
48 const QRect& paneRect,
49 const QMatrix4x4& pvm,
50 QVector3D& rayOrigin,
51 QVector3D& rayDir)
52{
53 const std::optional<scene::Ray> ray = scene::ray_from_pixel(
54 {static_cast<double>(screenPos.x()), static_cast<double>(screenPos.y())},
55 {static_cast<double>(paneRect.x()), static_cast<double>(paneRect.y()), static_cast<double>(paneRect.width()),
56 static_cast<double>(paneRect.height())},
57 toEigen(pvm));
58 if (!ray) {
59 return false;
60 }
61 rayOrigin = toQVector3D(ray->origin);
62 rayDir = toQVector3D(ray->direction);
63 return true;
64}

68RayHit RayPicker::pick(const QVector3D& rayOrigin,
69 const QVector3D& rayDir,
70 const SubView& subView,
71 const std::map<std::string, std::shared_ptr<BrainSurface>>& surfaces,
72 const std::map<const QStandardItem*, std::shared_ptr<BrainSurface>>& itemSurfaceMap,
73 const std::map<const QStandardItem*, std::shared_ptr<DipoleObject>>& itemDipoleMap)
74{
75 RayHit result;
76 float closestDist = std::numeric_limits<float>::max();
77
78 // ── Test surfaces ──────────────────────────────────────────────────
79 for (const auto& [key, surf] : surfaces) {
80 if (!surf->isVisible())
81 continue;
82 if (!subView.shouldRenderSurface(key))
83 continue;
84 if (key.starts_with("srcsp_"))
85 continue; // skip source space for picking
86
87 const bool isSensor = key.starts_with("sens_");
88 const bool isBem = key.starts_with("bem_");
89 const bool isDig = key.starts_with("dig_");
90
91 // Brain surfaces: only pick if matching active surface type
92 if (!isSensor && !isBem && !isDig) {
93 if (!subView.matchesSurfaceType(key))
94 continue;
95 }
96
97 float dist = 0.0f;
98 int vertexIdx = -1;
99 if (surf->intersects(rayOrigin, rayDir, dist, vertexIdx)) {
100 if (dist < closestDist) {
101 closestDist = dist;
102 result.hit = true;
103 result.distance = dist;
104 result.hitPoint = rayOrigin + dist * rayDir;
105 result.vertexIndex = vertexIdx;
106 result.surfaceKey = key;
107 result.isDipole = false;
108 result.dipoleIndex = -1;
109
110 // Reverse lookup: find tree item for this surface
111 result.item = nullptr;
112 for (const auto& [item, itemSurface] : itemSurfaceMap) {
113 if (itemSurface == surf) {
114 result.item = const_cast<QStandardItem*>(item);
115 break;
116 }
117 }
118
119 // FsAnnotation info
120 const auto itemSurface = result.item ? itemSurfaceMap.find(result.item) : itemSurfaceMap.end();
121 if (itemSurface != itemSurfaceMap.end()) {
122 result.regionName = itemSurface->second->getAnnotationLabel(vertexIdx);
123 result.regionId = itemSurface->second->getAnnotationLabelId(vertexIdx);
124 } else {
125 result.regionName.clear();
126 result.regionId = -1;
127 }
128 }
129 }
130 }
131
132 // ── Test dipoles ───────────────────────────────────────────────────
133 for (const auto& [item, dipole] : itemDipoleMap) {
134 if (!subView.visibility.dipoles)
135 continue;
136 if (!dipole->isVisible())
137 continue;
138
139 float dist = 0.0f;
140 int dipIdx = dipole->intersect(rayOrigin, rayDir, dist);
141 if (dipIdx != -1 && dist < closestDist) {
142 closestDist = dist;
143 result.hit = true;
144 result.distance = dist;
145 result.hitPoint = rayOrigin + dist * rayDir;
146 result.item = const_cast<QStandardItem*>(item);
147 result.surfaceKey.clear();
148 result.vertexIndex = dipIdx;
149 result.isDipole = true;
150 result.dipoleIndex = dipIdx;
151 result.regionName.clear();
152 result.regionId = -1;
153 }
154 }
155
156 return result;
157}

161std::string RayPicker::buildLabel(const RayHit& result,
162 [[maybe_unused]] const std::map<const QStandardItem*, std::shared_ptr<BrainSurface>>& itemSurfaceMap,
163 [[maybe_unused]] const std::map<std::string, std::shared_ptr<BrainSurface>>& surfaces)
164{
165 if (!result.hit)
166 return {};
167
168 const std::string& key = result.surfaceKey;
169 const std::string itemText = result.item ? result.item->text().toStdString() : std::string();
170 const auto capitalized = [](std::string text) {
171 if (!text.empty())
172 text[0] = static_cast<char>(std::toupper(static_cast<unsigned char>(text[0])));
173 return text;
174 };
175
176 // ── Brain surface with annotation ──────────────────────────────────
177 if (!result.regionName.empty()) {
178 return result.displayLabel();
179 }
180
181 // ── Dipole ─────────────────────────────────────────────────────────
182 if (result.isDipole) {
183 return std::format("{} (Dipole {})", result.item ? itemText : std::string("Dipole"), result.dipoleIndex);
184 }
185
186 // ── Sensor/BEM/Digitizer/Helmet ────────────────────────────────────
187 if (key.starts_with("sens_surface_meg")) {
188 return "MEG Helmet";
189 }
190 if (key.starts_with("sens_meg_")) {
191 return "MEG: " + (result.item ? itemText : key);
192 }
193 if (key.starts_with("sens_eeg_")) {
194 return "EEG: " + (result.item ? itemText : key);
195 }
196 if (key.starts_with("dig_")) {
197 // Resolve individual point from batched mesh
198 std::string pointName;
199 if (result.item && result.vertexIndex >= 0) {
200 AbstractTreeItem* abs = dynamic_cast<AbstractTreeItem*>(result.item);
201 if (abs && abs->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::DigitizerItem)) {
202 auto* digItem = static_cast<DigitizerTreeItem*>(abs);
203 constexpr int vertsPerSphere = 42;
204 int ptIdx = result.vertexIndex / vertsPerSphere;
205 const std::vector<std::string>& names = digItem->pointNames();
206 if (ptIdx >= 0 && ptIdx < std::ssize(names))
207 pointName = names[static_cast<std::size_t>(ptIdx)];
208 }
209 }
210 const std::string category = capitalized(key.substr(4));
211 return pointName.empty()
212 ? std::format("Digitizer ({})", category)
213 : std::format("Digitizer: {} ({})", pointName, category);
214 }
215 if (key.starts_with("bem_")) {
216 std::string compartment = capitalized(key.substr(4));
217 std::ranges::replace(compartment, '_', ' ');
218 return "BEM: " + compartment;
219 }
220
221 // ── Hemisphere fallback ────────────────────────────────────────────
222 if (key.starts_with("lh_"))
223 return "Left Hemisphere";
224 if (key.starts_with("rh_"))
225 return "Right Hemisphere";
226
227 return {};
228}

232std::string RayHit::displayLabel() const
233{
234 // Delegated to the static builder in RayPicker; this method is a
235 // convenience wrapper when the caller doesn't have the surface maps.
236 if (!hit)
237 return {};
238
239 if (!regionName.empty()) {
240 if (surfaceKey.starts_with("lh"))
241 return std::format("Region: {} (lh)", regionName);
242 if (surfaceKey.starts_with("rh"))
243 return std::format("Region: {} (rh)", regionName);
244 return "Region: " + regionName;
245 }
246
247 if (isDipole) {
248 return std::format("Dipole {}", dipoleIndex);
249 }
250
251 return surfaceKey;
252}
