192void HpiPanel::changed(const std::string& key)
193{
194 if (key == kDigitizerFile) {
195 const std::string file = form().get<std::string>(kDigitizerFile);
196 if (!file.empty()) {
197 load_digitizers(path_from_utf8(file));
198 }
199 } else if (key == kCoilFreqs) {
200 set_coil_frequencies(parse_frequencies(form().get<std::string>(kCoilFreqs)));
201 report_frequencies();
202 } else if (key == kCoilPreset) {
203 const std::string name = form().get<std::string>(kCoilPreset);
204 for (const JsonValue& entry : m_presets[std::to_string(m_frequencies.size())].to_array()) {
205 const JsonObject& preset = entry.to_object();
206 if (preset["name"].to_std_string() == name) {
207 std::vector<int> coils;
208 const JsonArray& values = preset["coils"].to_array();
209 coils.reserve(values.size());
210 for (const JsonValue& coil : values) {
211 coils.push_back(coil.to_int());
212 }
213 set_coil_frequencies(coils);
214 report_frequencies();
215 break;
216 }
217 }
218 } else if (key == kOrder) {
219 if (m_callbacks.order_frequencies) {
220 m_callbacks.order_frequencies();
221 }
222 } else if (key == kSingleFit) {
223 if (m_callbacks.single_fit) {
224 m_callbacks.single_fit();
225 }
226 } else if (key == kSsp) {
227 if (m_callbacks.ssp_changed) {
228 m_callbacks.ssp_changed(use_ssp());
229 }
230 } else if (key == kComp) {
231 if (m_callbacks.compensators_changed) {
232 m_callbacks.compensators_changed(use_compensators());
233 }
234 } else if (key == kContinuous) {
235 if (m_callbacks.continuous_fitting_changed) {
236 m_callbacks.continuous_fitting_changed(continuous_fitting());
237 }
238 } else if (key == kWindow) {
239 if (m_callbacks.fitting_window_size_changed) {
240 m_callbacks.fitting_window_size_changed(fitting_window_size());
241 }
242 } else if (key == kMaxError) {
243 if (m_callbacks.allowed_error_changed) {
244 m_callbacks.allowed_error_changed(allowed_error());
245 }
246 } else if (key == kMoveThreshold) {
247 if (m_callbacks.allowed_movement_changed) {
248 m_callbacks.allowed_movement_changed(allowed_movement());
249 }
250 } else if (key == kRotThreshold) {
251 if (m_callbacks.allowed_rotation_changed) {
252 m_callbacks.allowed_rotation_changed(allowed_rotation());
253 }
254 }
255}

360void HpiPanel::set_digitizers(const mne::FiffDigPointSet& digitizers)
361{
362 int coils = 0;
363 int fiducials = 0;
364 int eeg = 0;
365 int additional = 0;
366 for (const FiffDigPoint& point : digitizers) {
367 switch (point.kind) {
368 case FIFFV_POINT_HPI:
369 ++coils;
370 break;
371 case FIFFV_POINT_CARDINAL:
372 ++fiducials;
373 break;
374 case FIFFV_POINT_EEG:
375 ++eeg;
376 break;
377 case FIFFV_POINT_EXTRA:
378 ++additional;
379 break;
380 default:
381 break;
382 }
383 }
384 set_info(kLoadedCoils, std::to_string(coils));
385 set_info(kLoadedFiducials, std::to_string(fiducials));
386 set_info(kLoadedEeg, std::to_string(eeg));
387 set_info(kLoadedAdditional, std::to_string(additional));
388 set_info(kDigitizerStatus, "loaded", SettingField::Tone::Positive);
389
390 // One coil per HPI point; new coils start without a frequency and the fit results start afresh
391 m_frequencies.resize(static_cast<std::size_t>(coils), -1);
392 m_errors.clear();
393 m_goodness.clear();
394 show_frequencies();
395 setup_presets();
396}

400bool HpiPanel::load_digitizers(const std::filesystem::path& file)
401{
402 const bool updating = std::exchange(m_updating, true);
403 form().set_value(kDigitizerFile, utf8_from_path(file));
404 m_updating = updating;
405
406 std::error_code error;
407 FiffDigPointSet digitizers;
408 if (std::filesystem::is_regular_file(file, error)) {
409 if (auto read = FiffDigPointSet::read(file)) {
410 digitizers = std::move(*read);
411 }
412 }
413 if (digitizers.empty()) {
414 set_info(kDigitizerStatus, "File could not be loaded!", SettingField::Tone::Negative);
415 return false;
416 }
417 set_digitizers(digitizers);
418 report_frequencies();
419 // Continuous fitting stops while the digitizers change
420 if (continuous_fitting()) {
421 const bool updatingContinuous = std::exchange(m_updating, true);
422 form().set_value(kContinuous, false);
423 m_updating = updatingContinuous;
424 if (m_callbacks.continuous_fitting_changed) {
425 m_callbacks.continuous_fitting_changed(false);
426 }
427 }
428 if (m_callbacks.digitizers_changed) {
429 m_callbacks.digitizers_changed(digitizers, file);
430 }
431 return true;
432}

436bool HpiPanel::load_coil_presets(const std::filesystem::path& file)
437{
438 const Expected<std::string> text = read_text_file(file);
439 if (!text) {
440 Log::warning("Cannot read HPI coil presets: {}", text.error().what());
441 return false;
442 }
443 JsonParseError error;
444 const JsonDocument presets = JsonDocument::from_json(*text, &error);
445 if (error.error != JsonParseError::NoError || !presets.is_object()) {
446 Log::warning("{} holds no JSON object of HPI coil presets", utf8_from_path(file));
447 return false;
448 }
449 m_presets = presets.object();
450 setup_presets();
451 return true;
452}

529std::vector<int> HpiPanel::parse_frequencies(const std::string& text)
530{
531 std::vector<int> frequencies;
532 std::size_t start = 0;
533 while (start < text.size()) {
534 std::size_t end = text.find_first_of(", ;\t", start);
535 if (end == std::string::npos) {
536 end = text.size();
537 }
538 if (end > start) {
539 const std::string_view token(text.data() + start, end - start);
540 int frequency = -1;
541 const auto [rest, error] = std::from_chars(token.data(), token.data() + token.size(), frequency);
542 frequencies.push_back(error == std::errc() && rest == token.data() + token.size() && frequency > 0 ? frequency : -1);
543 }
544 start = end + 1;
545 }
546 return frequencies;
547}
