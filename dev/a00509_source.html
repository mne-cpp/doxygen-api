168void LinePlot::addSeries(const QVector<double>& x,
169 const QVector<double>& y,
170 const QString& label,
171 const QColor& color)
172{
173 const int n = qMin(x.size(), y.size());
174 const bool empty = std::all_of(m_series.cbegin(), m_series.cend(), [](const Series& s) { return s.x.isEmpty(); });
175 for (int i = 0; i < n; ++i) {
176 const bool first = empty && i == 0;
177 m_dMinX = first ? x[i] : qMin(m_dMinX, x[i]);
178 m_dMaxX = first ? x[i] : qMax(m_dMaxX, x[i]);
179 m_dMinY = first ? y[i] : qMin(m_dMinY, y[i]);
180 m_dMaxY = first ? y[i] : qMax(m_dMaxY, y[i]);
181 }
182 m_series.append({x.mid(0, n), y.mid(0, n), label, color.isValid() ? color : kPalette[m_series.size() % kPalette.size()]});
183
184 setWindowTitle(m_sTitle);
185 QWidget::update();
186}

197void LinePlot::paintEvent(QPaintEvent* /*event*/)
198{
199 QPainter painter(this);
200 painter.setRenderHint(QPainter::Antialiasing);
201 painter.setRenderHint(QPainter::TextAntialiasing);
202
203 const QColor text(51, 51, 51);
204 const QColor grid(230, 230, 230);
205 const QFont tickFont = font();
206 const QFontMetrics tickMetrics(tickFont);
207
208 double minY = m_dMinY;
209 double maxY = m_dMaxY;
210 const double padY = (maxY - minY) * 0.05;
211 minY -= padY == 0.0 ? 1.0 : padY;
212 maxY += padY == 0.0 ? 1.0 : padY;
213 const double minX = m_dMinX;
214 const double maxX = m_dMaxX > m_dMinX ? m_dMaxX : m_dMinX + 1.0;
215 const QVector<double> yTicks = niceTicks(minY, maxY, 6);
216 const QVector<double> xTicks = niceTicks(minX, maxX, 8);
217
218 int tickWidth = 0;
219 for (double v : yTicks) {
220 tickWidth = qMax(tickWidth, tickMetrics.horizontalAdvance(tickLabel(v)));
221 }
222 const int line = tickMetrics.height();
223 const int mLeft = tickWidth + 14 + (m_sYLabel.isEmpty() ? 0 : line + 6);
224 const int mRight = 24;
225 const int mTop = m_sTitle.isEmpty() ? 16 : 2 * line + 8;
226 const int mBottom = line + 14 + (m_sXLabel.isEmpty() ? 0 : line + 6);
227 QRectF area(mLeft, mTop, width() - mLeft - mRight, height() - mTop - mBottom);
228 if (area.width() <= 0 || area.height() <= 0) {
229 return;
230 }
231
232 auto toPixelX = [&](double dx) {
233 return area.left() + (dx - minX) / (maxX - minX) * area.width();
234 };
235 auto toPixelY = [&](double dy) {
236 return area.bottom() - (dy - minY) / (maxY - minY) * area.height();
237 };
238
239 // Legend: in the corner of the axes that hides the least of the lines, else to the right of them
240 int labelWidth = 0;
241 int labelCount = 0;
242 for (const Series& series : m_series) {
243 if (!series.label.isEmpty()) {
244 labelWidth = qMax(labelWidth, tickMetrics.horizontalAdvance(series.label));
245 ++labelCount;
246 }
247 }
248 const QSizeF legendSize(labelWidth + 38, labelCount * (line + 2) + 10);
249 QRectF legend;
250 if (labelCount > 0) {
251 const QList<QRectF> corners = {
252 QRectF(QPointF(area.right() - legendSize.width() - 8, area.top() + 8), legendSize),
253 QRectF(QPointF(area.left() + 8, area.top() + 8), legendSize),
254 QRectF(QPointF(area.right() - legendSize.width() - 8, area.bottom() - legendSize.height() - 8), legendSize),
255 QRectF(QPointF(area.left() + 8, area.bottom() - legendSize.height() - 8), legendSize)};
256 double fewest = std::numeric_limits<double>::max();
257 for (const QRectF& corner : corners) {
258 const double hidden = hiddenLength(corner, toPixelX, toPixelY);
259 if (hidden < fewest) {
260 fewest = hidden;
261 legend = corner;
262 }
263 }
264 if (fewest > 0.0 && area.width() > 2 * legendSize.width()) {
265 area.setRight(area.right() - legendSize.width() - 12);
266 legend = QRectF(QPointF(area.right() + 12, area.top()), legendSize);
267 }
268 }
269
270 if (!m_sTitle.isEmpty()) {
271 QFont titleFont = font();
272 titleFont.setBold(true);
273 titleFont.setPointSizeF(titleFont.pointSizeF() * 1.25);
274 painter.setFont(titleFont);
275 painter.setPen(text);
276 painter.drawText(QRectF(0, 4, width(), mTop - 4), Qt::AlignCenter, m_sTitle);
277 }
278
279 painter.setFont(tickFont);
280 for (double v : yTicks) {
281 const double py = toPixelY(v);
282 painter.setPen(QPen(grid, 1));
283 painter.drawLine(QPointF(area.left(), py), QPointF(area.right(), py));
284 painter.setPen(text);
285 painter.drawText(QRectF(0, py - line, area.left() - 8, 2 * line), Qt::AlignRight | Qt::AlignVCenter, tickLabel(v));
286 }
287 for (double v : xTicks) {
288 const double px = toPixelX(v);
289 painter.setPen(QPen(grid, 1));
290 painter.drawLine(QPointF(px, area.top()), QPointF(px, area.bottom()));
291 painter.setPen(text);
292 painter.drawText(QRectF(px - 50, area.bottom() + 6, 100, line), Qt::AlignHCenter | Qt::AlignTop, tickLabel(v));
293 }
294 if (!m_sXLabel.isEmpty()) {
295 painter.drawText(QRectF(area.left(), height() - line - 6, area.width(), line), Qt::AlignCenter, m_sXLabel);
296 }
297 if (!m_sYLabel.isEmpty()) {
298 painter.save();
299 painter.translate(6, area.center().y());
300 painter.rotate(-90);
301 painter.drawText(QRectF(-area.height() / 2, 0, area.height(), line), Qt::AlignCenter, m_sYLabel);
302 painter.restore();
303 }
304
305 painter.save();
306 painter.setClipRect(area);
307 const qreal penWidth = m_series.size() > 20 ? 1.0 : 1.75;
308 for (const Series& series : m_series) {
309 if (series.x.isEmpty()) {
310 continue;
311 }
312 QPainterPath path;
313 path.moveTo(toPixelX(series.x[0]), toPixelY(series.y[0]));
314 for (int i = 1; i < series.x.size(); ++i) {
315 path.lineTo(toPixelX(series.x[i]), toPixelY(series.y[i]));
316 }
317 painter.setPen(QPen(series.color, penWidth, Qt::SolidLine, Qt::RoundCap, Qt::RoundJoin));
318 painter.setBrush(Qt::NoBrush);
319 painter.drawPath(path);
320 }
321 painter.restore();
322
323 painter.setPen(QPen(QColor(136, 136, 136), 1));
324 painter.setBrush(Qt::NoBrush);
325 painter.drawRect(area);
326
327 if (labelCount > 0) {
328 painter.setPen(QPen(QColor(204, 204, 204), 1));
329 painter.setBrush(QColor(255, 255, 255, 225));
330 painter.drawRoundedRect(legend, 4, 4);
331 int row = 0;
332 for (const Series& series : m_series) {
333 if (series.label.isEmpty()) {
334 continue;
335 }
336 const double y = legend.top() + 5 + row++ * (line + 2) + line / 2.0;
337 painter.setPen(QPen(series.color, 2.5, Qt::SolidLine, Qt::RoundCap));
338 painter.drawLine(QPointF(legend.left() + 8, y), QPointF(legend.left() + 26, y));
339 painter.setPen(text);
340 painter.drawText(QRectF(legend.left() + 32, y - line / 2.0, labelWidth + 4, line), Qt::AlignLeft | Qt::AlignVCenter, series.label);
341 }
342 }
343}

347double LinePlot::hiddenLength(const QRectF& rect,
348 const std::function<double(double)>& toPixelX,
349 const std::function<double(double)>& toPixelY) const
350{
351 double hidden = 0.0;
352 for (const Series& series : m_series) {
353 for (int i = 1; i < series.x.size(); ++i) {
354 const QPointF a(toPixelX(series.x[i - 1]), toPixelY(series.y[i - 1]));
355 const QPointF b(toPixelX(series.x[i]), toPixelY(series.y[i]));
356 const double length = QLineF(a, b).length();
357 const int steps = qMax(1, static_cast<int>(length / 2.0));
358 for (int s = 0; s <= steps; ++s) {
359 hidden += rect.contains(a + (b - a) * (static_cast<double>(s) / steps)) ? length / (steps + 1) : 0.0;
360 }
361 }
362 }
363 return hidden;
364}
