51void SensorLayoutFigure::set_layout(const std::map<std::string, Point>& layout, const std::vector<std::string>& bad)
52{
53 std::vector<SensorLayout::Sensor> sensors;
54 sensors.reserve(layout.size());
55 for (const auto& [name, position] : layout) {
56 SensorLayout::Sensor sensor;
57 sensor.name = name;
58 sensor.position = position;
59 sensor.kind = name.find("EEG") != std::string::npos ? FIFFV_EEG_CH : FIFFV_MEG_CH;
60 sensor.unit = FIFF_UNIT_T_M;
61 sensor.bad = std::ranges::contains(bad, name);
62 sensors.push_back(std::move(sensor));
63 }
64 // New sensors start unselected; a lost selection is reported
65 const bool hadSelection = !m_layout.selected().empty();
66 m_layout.set_sensors(std::move(sensors));
67 changed(Change::Bounds);
68 if (hadSelection) {
69 selection_changed();
70 }
71}

82void SensorLayoutFigure::set_selectable(const std::vector<std::string>& names)
83{
84 bool deselected = false;
85 for (SensorLayout::Sensor& sensor : m_layout.sensors()) {
86 sensor.visible = sensor.kind == m_kind;
87 sensor.selectable = sensor.visible && std::ranges::contains(names, sensor.name);
88 deselected |= sensor.selected && !sensor.selectable;
89 sensor.selected = sensor.selected && sensor.selectable;
90 }
91 changed(Change::Bounds);
92 if (deselected) {
93 selection_changed();
94 }
95}

170bool SensorLayoutFigure::press(Point point, Modifier modifiers)
171{
172 const bool extend = has_flag(modifiers, Modifier::Ctrl);
173 const bool hit = m_layout.sensor_at(point) >= 0;
174 // A press beside the sensors clears the selection before a band starts, unless Ctrl is held
175 if ((hit || !extend) && m_layout.click(point, extend)) {
176 changed();
177 selection_changed();
178 }
