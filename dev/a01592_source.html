60bool CommandLineParser::add_option(CommandLineOption option)
61{
62 if (option.names.empty())
63 return false;
64 for (const std::string& name : option.names) {
65 if (name.empty() || name.front() == '-' || find(name) != std::string::npos)
66 return false;
67 }
68 m_options.push_back(std::move(option));
69 m_values.emplace_back();
70 return true;

93bool CommandLineParser::parse(std::span<const std::string> args)
94{
95 for (auto& given : m_values)
96 given.clear();
97 m_positional.clear();
98 m_error.clear();
99
100 bool optionsEnded = false;
101 for (std::size_t i = 1; i < args.size(); ++i) {
102 const std::string& arg = args[i];
103 if (optionsEnded || arg.size() < 2 || arg.front() != '-') {
104 m_positional.push_back(arg);
105 continue;
106 }
107 if (arg == "--") {
108 optionsEnded = true;
109 continue;
110 }
111 const std::string_view body = std::string_view(arg).substr(arg.starts_with("--") ? 2 : 1);
112 const std::size_t equals = body.find('=');
113 const std::string_view name = body.substr(0, equals);
114 const std::size_t index = find(name);
115 if (index == std::string::npos) {
116 m_error = "Unknown option '" + std::string(name) + "'.";
117 return false;
118 }
119 const CommandLineOption& option = m_options[index];
120 if (option.valueName.empty()) {
121 if (equals != std::string_view::npos) {
122 m_error = "Unexpected value after '" + std::string(name) + "'.";
123 return false;
124 }
125 m_values[index].emplace_back();
126 } else if (equals != std::string_view::npos) {
127 m_values[index].emplace_back(body.substr(equals + 1));
128 } else if (i + 1 < args.size()) {
129 m_values[index].push_back(args[++i]);
130 } else {
131 m_error = "Missing value after '" + arg + "'.";
132 return false;
133 }
134 }
135 return true;
136}

148void CommandLineParser::process(int argc, const char* const* argv, std::string_view name, std::string_view version)
149{
150 if (!parse(argc, argv)) {
151 std::println(stderr, "{}", m_error);
152 std::exit(1);

154 if (find("help") != std::string::npos && is_set("help")) {

156 std::exit(0);
157 }
158 if (find("version") != std::string::npos && is_set("version")) {
159 std::println("{} {}", name, version);
160 std::exit(0);

199std::string CommandLineParser::help_text(std::string_view name) const
200{
201 std::string usage = "Usage: " + std::string(name);
202 if (!m_options.empty())
203 usage += " [options]";
204 for (const Positional& positional : m_positionalHelp)
205 usage += " " + positional.name;
206 std::string text = usage + "\n";
207 if (!m_description.empty())
208 text += m_description + "\n";
209
210 std::vector<std::pair<std::string, std::string>> rows;
211 for (const CommandLineOption& option : m_options) {
212 std::string spelled;
213 for (const std::string& optionName : option.names) {
214 if (!spelled.empty())
215 spelled += ", ";
216 spelled += (optionName.size() == 1 ? "-" : "--") + optionName;
217 }
218 if (!option.valueName.empty())
219 spelled += " <" + option.valueName + ">";
220 std::string description = option.description;
221 // A description that already names its default keeps its own wording
222 if (!option.defaultValue.empty() && description.find("default") == std::string::npos) {
223 const bool sentence = description.ends_with('.');
224 if (sentence)
225 description.pop_back();
226 description += " (default: " + option.defaultValue + ")" + (sentence ? "." : "");
227 }
228 rows.emplace_back(std::move(spelled), std::move(description));
229 }
230 std::size_t width = 0;
231 for (const auto& row : rows)
232 width = std::max(width, row.first.size());
233 for (const Positional& positional : m_positionalHelp)
234 width = std::max(width, positional.name.size());
235
236 const auto table = [&](std::string_view heading, const std::vector<std::pair<std::string, std::string>>& entries) {
237 if (entries.empty())
238 return;
239 text.append("\n").append(heading).append(":\n");
240 for (const auto& [left, right] : entries)
241 text.append(" ").append(left).append(width - left.size() + 2, ' ').append(right).append("\n");
242 };
243 table("Options", rows);
244 std::vector<std::pair<std::string, std::string>> arguments;
245 arguments.reserve(m_positionalHelp.size());
246 for (const Positional& positional : m_positionalHelp)
247 arguments.emplace_back(positional.name, positional.description);
248 table("Arguments", arguments);
249 return text;
250}
