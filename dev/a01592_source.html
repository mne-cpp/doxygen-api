138bool PolhemusCoregistration::computeRegistration()
139{
140 if (!m_pPoints->hasAllFiducials()) {
141 return false;
142 }
143
144 const QMatrix4x4 headFrame = buildHeadFrame();
145
146 const QVector3D ez(headFrame(0, 2), headFrame(1, 2), headFrame(2, 2));
147 if (ez.length() < 1e-6f) {
148 return false;
149 }
150
151 m_headToWorld = headFrame;
152 m_headToDevice = m_deviceToWorld.inverted() * m_headToWorld;
153 m_registrationValid = true;
154 emit registrationChanged();
155 return true;
156}

189QMatrix4x4 PolhemusCoregistration::buildDevicePose(const QVector3D& trackerPos,
190 const QQuaternion& trackerOri) const
191{
192 QMatrix4x4 trackerToWorld;
193 trackerToWorld.setToIdentity();
194 trackerToWorld.translate(trackerPos);
195 trackerToWorld.rotate(trackerOri);
196
197 QMatrix4x4 offset;
198 offset.setToIdentity();
199 offset.translate(m_offsetTranslation);
200 offset.rotate(m_offsetRotation);
201
202 return trackerToWorld * offset;
203}

205QMatrix4x4 PolhemusCoregistration::buildHeadFrame() const
206{
207 const QVector3D nas = m_pPoints->fiducial(FiducialId::NAS);
208 const QVector3D lpa = m_pPoints->fiducial(FiducialId::LPA);
209 const QVector3D rpa = m_pPoints->fiducial(FiducialId::RPA);
210
211 const QVector3D origin = (lpa + rpa) * 0.5f;
212 const QVector3D ex = (nas - origin).normalized();
213 const QVector3D eyApprox = (lpa - origin).normalized();
214 const QVector3D ez = QVector3D::crossProduct(ex, eyApprox).normalized();
215 const QVector3D ey = QVector3D::crossProduct(ez, ex).normalized();
216
217 QMatrix4x4 frame;
218 frame.setToIdentity();
219 frame(0, 0) = ex.x(); frame(0, 1) = ey.x(); frame(0, 2) = ez.x(); frame(0, 3) = origin.x();
220 frame(1, 0) = ex.y(); frame(1, 1) = ey.y(); frame(1, 2) = ez.y(); frame(1, 3) = origin.y();
221 frame(2, 0) = ex.z(); frame(2, 1) = ey.z(); frame(2, 2) = ez.z(); frame(2, 3) = origin.z();
222 frame(3, 0) = 0.0f; frame(3, 1) = 0.0f; frame(3, 2) = 0.0f; frame(3, 3) = 1.0f;
223 return frame;
224}
