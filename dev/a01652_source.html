70void Panel::load(const SettingsStore& store, const std::string& path)
71{
72 if (path.empty()) {
73 return;
74 }
75 // Listeners may change the fields while values are restored
76 const std::vector<SettingField> fields = m_form->fields();
77 for (const SettingField& field : fields) {
78 if (!field.persistent) {
79 continue;
80 }
81 // Stores may hand back numbers and flags as text, so they are read as the field's type
82 const std::string key = settings_key(path, field.key);
83 if (!store.contains(key)) {
84 continue;
85 }
86 switch (field.kind) {
87 case SettingField::Kind::Bool:
88 m_form->set_value(field.key, store.read_bool(key, m_form->get<bool>(field.key)));
89 break;
90 case SettingField::Kind::Int:
91 m_form->set_value(field.key, store.read_int(key, m_form->get<int>(field.key)));
92 break;
93 case SettingField::Kind::Double:
94 m_form->set_value(field.key, store.read_double(key, m_form->get<double>(field.key)));
95 break;
96 default:
97 m_form->set_value(field.key, store.read_string(key, m_form->get<std::string>(field.key)));
98 break;
99 }
100 }
101}
