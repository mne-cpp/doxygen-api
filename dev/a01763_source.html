159void TriggerDetectionPanel::init(const mne::FiffInfo& info)
160{
161 std::vector<SettingChoice> channels;
162 for (const auto& ch : info.chs) {
163 if (ch.kind == FIFFV_STIM_CH) {
164 channels.push_back({ch.ch_name, ch.ch_name});
165 }
166 }
167 {
168 const bool updating = std::exchange(m_updating, true);
169 if (m_storedChannel >= 0 && m_storedChannel < static_cast<int>(channels.size())) {
170 const std::string stored = channels[static_cast<std::size_t>(m_storedChannel)].value;
171 form().set_choices(kChannel, std::move(channels));
172 form().set_value(kChannel, stored);
173 } else {
174 form().set_choices(kChannel, std::move(channels));
175 }
176 m_updating = updating;
177 }
178 m_initialized = true;
179 show_type_color();
180 report_info();
181}

185void TriggerDetectionPanel::set_detected_triggers(int total, const std::map<int, std::vector<std::pair<int, double>>>& triggers)
186{
187 const bool updating = std::exchange(m_updating, true);
188 form().set_value(kDetected, std::to_string(total));
189 std::vector<SettingChoice> types = form().field(kColorType)->choices;
190 const std::size_t count = types.size();
191 for (const auto& [channel, detections] : triggers) {
192 for (const auto& [sample, value] : detections) {
193 const std::string type = type_text(value);
194 if (std::ranges::none_of(types, [&type](const SettingChoice& choice) { return choice.value == type; })) {
195 types.push_back({type, type});
196 }
197 }
198 }
199 if (types.size() != count) {
200 form().set_choices(kColorType, std::move(types));
201 }
202 m_updating = updating;
203 show_type_color();
204}

229void TriggerDetectionPanel::load(const SettingsStore& store, const std::string& path)
230{
231 if (path.empty()) {
232 return;
233 }
234 const bool updating = std::exchange(m_updating, true);
235 form().set_value(kActive, store.read_bool(settings_key(path, "Activated"), false));
236 m_storedChannel = store.read_int(settings_key(path, "ChannelIndex"), 0);
237 const std::vector<SettingChoice>& channels = form().field(kChannel)->choices;
238 if (m_storedChannel >= 0 && m_storedChannel < static_cast<int>(channels.size())) {
239 form().set_value(kChannel, channels[static_cast<std::size_t>(m_storedChannel)].value);
240 }
241 form().set_value(kMantissa, store.read_double(settings_key(path, "FirstThresholdValue"), 0.1));
242 form().set_value(kExponent, store.read_int(settings_key(path, "SecondThresholdValue"), -1));
243 std::istringstream types(store.read_string(settings_key(path, "ColorTypes"), {}));
244 for (std::string type; std::getline(types, type, ';');) {
245 if (!type.empty()) {
246 m_colors[type_value(type)] = store.read_string(settings_key(path, "Colors/" + type), kDefaultColor);
247 }
248 }
249 m_updating = updating;
250 if (m_initialized) {
251 show_type_color();
252 report_info();
253 }
254}

258void TriggerDetectionPanel::save(SettingsStore& store, const std::string& path) const
259{
260 if (path.empty()) {
261 return;
262 }
263 const std::vector<SettingChoice>& channels = form().field(kChannel)->choices;
264 const auto current = std::ranges::find(channels, selected_stim_channel(), &SettingChoice::value);
265 const int index = current == channels.end() ? -1 : static_cast<int>(current - channels.begin());
266 store.set_value(settings_key(path, "Activated"), active());
267 store.set_value(settings_key(path, "ChannelIndex"), index);
268 store.set_value(settings_key(path, "FirstThresholdValue"), form().get<double>(kMantissa));
269 store.set_value(settings_key(path, "SecondThresholdValue"), form().get<int>(kExponent));
270 std::string types;
271 for (const auto& [type, color] : m_colors) {
272 const std::string text = type_text(type);
273 store.set_value(settings_key(path, "Colors/" + text), color);
274 types += (types.empty() ? "" : ";") + text;
275 }
276 store.set_value(settings_key(path, "ColorTypes"), types);
277}

290void TriggerDetectionPanel::changed(const std::string& key)
291{
292 if (key == kActive || key == kChannel || key == kMantissa || key == kExponent) {
293 report_info();
294 } else if (key == kColorType) {
295 show_type_color();
296 } else if (key == kColor) {
297 m_colors[type_value(form().get<std::string>(kColorType))] = form().get<std::string>(kColor);
298 report_info();
299 } else if (key == kReset) {
300 const bool updating = std::exchange(m_updating, true);
301 form().set_value(kDetected, std::string("0"));
302 form().set_choices(kColorType, {});
303 m_updating = updating;
304 show_type_color();
305 if (m_callbacks.reset_trigger_counter) {
306 m_callbacks.reset_trigger_counter();
307 }
308 } else if (key == kDetect) {
309 if (!selected_stim_channel().empty() && m_callbacks.detect_triggers) {
310 m_callbacks.detect_triggers(selected_stim_channel(), threshold());
311 }
312 }
313}
