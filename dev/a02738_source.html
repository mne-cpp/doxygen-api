40RtSourceStream::RtSourceStream()
41{
42 // Create data worker
43 // streamData() runs on the streaming thread, which also calls the result callbacks.
44 m_pWorker = std::make_unique<RtSourceDataWorker>();
45
46 // Forward worker results to the callbacks of this stream
47 m_pWorker->setNewRtSmoothedDataCallback([this](const std::vector<uint32_t>& colorsLh, const std::vector<uint32_t>& colorsRh) {
48 if (m_newSmoothedDataAvailableCallback) {
49 m_newSmoothedDataAvailableCallback(colorsLh, colorsRh);
50 }
51 });
52 m_pWorker->setNewRtRawDataCallback([this](const Eigen::VectorXd& dataLh, const Eigen::VectorXd& dataRh) {
53 if (m_newRawDataAvailableCallback) {
54 m_newRawDataAvailableCallback(dataLh, dataRh);
55 }
56 });
57
58 // Streaming thread driving the cadence
59 m_pStreamTask = std::make_unique<PeriodicTask>([worker = m_pWorker.get()] { worker->streamData(); });
60
61 // Create interpolation matrix worker
62 // One pool thread runs the recomputations in request order.
63 m_pInterpWorker = std::make_unique<RtSourceInterpolationMatWorker>();
64 m_pInterpPool = std::make_unique<mne::ThreadPool>(1);
65
66 // Forward interpolation results: auto-apply to data worker + re-emit
67 m_pInterpWorker->setNewInterpolationMatrixLeftAvailableCallback([this](const std::shared_ptr<Eigen::SparseMatrix<float>>& interpMat) {
68 onNewInterpolationMatrixLeft(interpMat);
69 });
70 m_pInterpWorker->setNewInterpolationMatrixRightAvailableCallback([this](const std::shared_ptr<Eigen::SparseMatrix<float>>& interpMat) {
71 onNewInterpolationMatrixRight(interpMat);
72 });
73
74 mne::Log::debug("RtSourceStream: Initialized with {} ms interval", m_iTimeInterval);
75}

79RtSourceStream::~RtSourceStream()
80{
81 // Join the threads while the workers are still alive. Pending
82 // recomputations are skipped.
83 m_bShuttingDown.store(true, std::memory_order_release);
84 m_pStreamTask.reset();
85 m_pInterpPool.reset();
86}
