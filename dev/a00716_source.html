137QString PolhemusConnection::autoDetectPortName()
138{
139 constexpr quint16 kPolhemusVid = 0x0F44;
140 constexpr quint16 kFtdiVid = 0x0403;
141
142 QString ftdiCandidate;
143 for (const auto& info : QSerialPortInfo::availablePorts()) {
144 if (!info.hasVendorIdentifier()) {
145 continue;
146 }
147 const quint16 vid = info.vendorIdentifier();
148 if (vid == kPolhemusVid) {
149 return info.portName();
150 }
151 if (vid == kFtdiVid && ftdiCandidate.isEmpty()) {
152 ftdiCandidate = info.portName();
153 }
154 }
155 return ftdiCandidate;
156}

175void PolhemusConnection::onMockTick()
176{
177 const double phase = (2.0 * M_PI * (m_mockSampleIdx % kMockPointsPerLap))
178 / static_cast<double>(kMockPointsPerLap);
179 const double elevation = 0.3 * std::sin(phase * 3.0);
180 const QVector3D pos(
181 kMockSphereRadiusM * static_cast<float>(std::cos(phase)),
182 kMockSphereRadiusM * static_cast<float>(std::sin(phase)),
183 kMockSphereRadiusM * static_cast<float>(elevation));
184 ++m_mockSampleIdx;
185
186 // Station 1 = pen (stylus — matches default penStation=1)
187 const QVector3D penPos(
188 kMockSphereRadiusM * static_cast<float>(std::cos(phase + 0.3)),
189 kMockSphereRadiusM * static_cast<float>(std::sin(phase + 0.3)),
190 kMockSphereRadiusM * static_cast<float>(elevation) + 0.02f);
191 const QQuaternion penOri = QQuaternion::fromAxisAndAngle(
192 QVector3D(0, 0, 1), static_cast<float>(phase * 180.0 / M_PI));
193 emit pointReceived(/*station*/ 1, penPos, penOri);
194
195 // Station 2 = tracker (device pose — matches default trackerStation=2)
196 emit pointReceived(/*station*/ 2, pos, QQuaternion());
197}

203bool PolhemusConnection::openSerial(const QString& portName, const PolhemusSerialConfig& cfg)
204{
205 auto* port = new QSerialPort(this);
206 port->setPortName(portName);
207 port->setBaudRate(cfg.baudRate);
208 port->setDataBits(QSerialPort::Data8);
209 port->setParity(QSerialPort::NoParity);
210 port->setStopBits(QSerialPort::OneStop);
211 port->setFlowControl(QSerialPort::NoFlowControl);
212
213 if (!port->open(QIODevice::ReadWrite)) {
214 emit errorOccurred(QStringLiteral("Serial open failed (%1): %2")
215 .arg(portName, port->errorString()));
216 port->deleteLater();
217 return false;
218 }
219
220 m_pSerial = port;
221 m_parser.reset();
222 m_parser.setUnits(cfg.units);
223
224 connect(m_pSerial, &QSerialPort::readyRead, this, &PolhemusConnection::onSerialReadyRead);
225 connect(m_pSerial, &QSerialPort::errorOccurred, this, &PolhemusConnection::onSerialError);
226
227 // Ensure all four stations are enabled and output position + Euler angles.
228 // Fastrak command: "l<station>,1\r" — enable station output
229 // Fastrak command: "O<station>,2,4,1\r" — output position (2), Euler angles (4), CRLF (1)
230 // NOTE: item 11 (quaternion) is not supported by all Fastrak firmware versions.
231 // Harmless for stations without a sensor physically connected.
232 for (int st = 1; st <= 4; ++st) {
233 m_pSerial->write(QStringLiteral("l%1,1\r").arg(st).toLatin1());
234 m_pSerial->write(QStringLiteral("O%1,2,4,1\r").arg(st).toLatin1());
235 }
236 m_pSerial->flush();
237
238 // Set tracking hemisphere per station if configured (non-zero vector).
239 // Fastrak command: "H<station>,x,y,z\r" — prevents position jumps
240 // when a sensor crosses the default hemisphere boundary.
241 if (!cfg.hemisphere.isNull()) {
242 const QVector3D& h = cfg.hemisphere;
243 for (int st = 1; st <= 4; ++st) {
244 const QByteArray cmd = QStringLiteral("H%1,%2,%3,%4\r")
245 .arg(st)
246 .arg(h.x(), 0, 'f', 1).arg(h.y(), 0, 'f', 1).arg(h.z(), 0, 'f', 1)
247 .toLatin1();
248 m_pSerial->write(cmd);
249 }
250 m_pSerial->flush();
251 qInfo() << "Fastrak: hemisphere set to" << cfg.hemisphere;
252 }
253
254 if (!cfg.streamCommand.isEmpty()) {
255 m_pSerial->write(cfg.streamCommand);
256 }
257
258 m_backendName = QStringLiteral("Fastrak (%1 @ %2 baud)")
259 .arg(portName)
260 .arg(cfg.baudRate);
261 return true;
262}

306void PolhemusConnection::drainParser()
307{
308 FastrakSample s;
309 while (m_parser.nextSample(s)) {
310 m_lastSamples[s.station] = {s.position, s.orientation};
311 emit pointReceived(s.station, s.position, s.orientation);

318void PolhemusConnection::onStreamPauseTimeout()
319{

321

326 m_pSerial->write(QByteArrayLiteral("C\r"));
