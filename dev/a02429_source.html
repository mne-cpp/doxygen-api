95FiffAnnotation::Extras mne::extras_from_json(const mne::JsonObject& object)
96{
97 FiffAnnotation::Extras extras;
98 for (const auto& [key, value] : object) {
99 FiffAnnotation::Extra extra;
100 if (value.is_bool()) {
101 extra = value.to_bool();
102 } else if (value.is_integer()) {
103 extra = value.to_integer();
104 } else if (value.is_number()) {
105 const double number = value.to_double();
106 // JSON has one number type; integral values come back as int, as from Python's json
107 if (std::trunc(number) == number && std::abs(number) < 9.007199254740992e15) {
108 extra = static_cast<std::int64_t>(number);
109 } else {
110 extra = number;
111 }
112 } else if (value.is_string()) {
113 extra = value.to_std_string();
114 } else if (!value.is_null()) {
115 continue;
116 }
117 extras.insert_or_assign(key, std::move(extra));
118 }
119 return extras;
120}

182FiffAnnotations FiffAnnotations::crop(double tmin, double tmax) const
183{
184 FiffAnnotations result;
185 for (const FiffAnnotation& a : m_annotations) {
186 if (a.onset > tmax || a.onset + a.duration < tmin) {
187 continue;
188 }
189 FiffAnnotation clipped = a;
190 if (clipped.onset < tmin) {
191 clipped.duration -= tmin - clipped.onset;
192 clipped.onset = tmin;
193 }
194 clipped.duration = std::min(clipped.duration, tmax - clipped.onset);
195 result.m_annotations.push_back(std::move(clipped));
196 }
197 return result;
198}

202Expected<FiffAnnotations> FiffAnnotations::read_json(const std::filesystem::path& path)
203{
204 const Expected<std::string> bytes = mne::read_text_file(path);
205 if (!bytes) {
206 return std::unexpected(bytes.error());
207 }
208 mne::JsonParseError parseError;
209 const mne::JsonDocument doc = mne::JsonDocument::from_json(*bytes, &parseError);
210 if (parseError.error != mne::JsonParseError::NoError || !doc.is_object()) {
211 return fail(ErrorCode::Format, "{} is not an annotation JSON file: {}", mne::utf8_from_path(path), parseError.error_string());
212 }
213
214 FiffAnnotations annot;
215 for (const mne::JsonValue& entry : doc.object().value("annotations").to_array()) {
216 const mne::JsonObject& obj = entry.to_object();
217 annot.m_annotations.push_back({obj.value("onset").to_double(0.0),
218 obj.value("duration").to_double(0.0),
219 obj.value("description").to_std_string(),
220 names_from_json(obj.value("channel_names").to_array()),
221 obj.value("comment").to_std_string(),
222 extras_from_json(obj.value("extras").to_object())});
223 }
224 return annot;
225}

246Expected<FiffAnnotations> FiffAnnotations::read_csv(const std::filesystem::path& path)
247{
248 const Expected<std::string> bytes = mne::read_text_file(path);
249 if (!bytes) {
250 return std::unexpected(bytes.error());
251 }
252 FiffAnnotations annot;
253 const mne::StringList lines = mne::split(*bytes, '\n');
254 for (std::size_t i = 1; i < lines.size(); ++i) {
255 const std::string_view line = mne::trimmed(lines[i]);
256 // The description may hold commas, so only the first two separate fields
257 const std::size_t first = line.find(',');
258 const std::size_t second = first == std::string_view::npos ? first : line.find(',', first + 1);
259 if (second == std::string_view::npos) {
260 continue;
261 }
262 const std::optional<double> onset = mne::to_double(line.substr(0, first));
263 const std::optional<double> duration = mne::to_double(line.substr(first + 1, second - first - 1));
264 if (onset && duration) {
265 annot.append(*onset, *duration, std::string(line.substr(second + 1)));
266 }
267 }
268 return annot;
269}
