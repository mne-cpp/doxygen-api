126bool FsSurfaceSet::read(const std::filesystem::path& p_sLHFileName, const std::filesystem::path& p_sRHFileName, FsSurfaceSet& p_SurfaceSet)
127{
128 p_SurfaceSet.clear();
129
130 for (const std::filesystem::path* file : {&p_sLHFileName, &p_sRHFileName}) {
131 FsSurface t_Surface;
132 if (FsSurface::read(*file, t_Surface))
133 p_SurfaceSet.m_surfs.insert_or_assign(t_Surface.hemi(), std::move(t_Surface));
134 }
135
136 if (p_SurfaceSet.m_surfs.empty())
137 return false;
138
139 p_SurfaceSet.calcOffset();
140
141 return true;
142}

206void FsSurfaceSet::calcOffset()
207{
208 //
209 // Correct inflated offset
210 //
211 if (m_surfs.size() == 2 && m_surfs.begin()->second.surf() == "inflated") {
212 float xOffset = m_surfs.at(0).rr().col(0).maxCoeff() - m_surfs.at(1).rr().col(0).minCoeff();
213 Vector3f vecLhOffset, vecRhOffset;
214 vecLhOffset << (xOffset / 2.0f), 0, 0;
215 vecRhOffset << (-xOffset / 2.0f), 0, 0;
216 m_surfs.at(0).offset() = vecLhOffset;
217 m_surfs.at(1).offset() = vecRhOffset;
218 }
