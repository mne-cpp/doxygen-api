56std::vector<int> goodChannelsOfKind(const FiffInfo& info, int kind)
57{
58 std::vector<int> idx;
59 for (int i = 0; i < info.nchan; ++i) {
60 if (info.chs[i].kind == kind && !info.isBad(info.chs[i].ch_name))
61 idx.push_back(i);
62 }
63 return idx;
64}

66// Adapted from mne.preprocessing.ssp._compute_exg_proj (MNE-Python, BSD-3-Clause).
67std::vector<FiffProj> exgProjectors(const MatrixXd& data, const FiffInfo& info, double sf, const std::vector<int>& events,
68 const ArtifactDetectProjParams& p, const std::string& mode)
69{
70 std::vector<FiffProj> projs = info.projs;
71 if (events.empty()) {
72 Log::warning("[ArtifactDetect] No {} events found", mode);
73 return projs;
74 }
75
76 std::vector<int> picks;
77 for (int kind : {FIFFV_MEG_CH, FIFFV_EEG_CH, FIFFV_EOG_CH, FIFFV_ECG_CH}) {
78 std::ranges::copy(goodChannelsOfKind(info, kind), std::back_inserter(picks));
79 }
80 std::ranges::sort(picks);
81 RowVectorXi sel(picks.size());
82 MatrixXd picked(picks.size(), data.cols());
83 for (int c = 0; c < std::ssize(picks); ++c) {
84 sel(c) = picks[c];
85 picked.row(c) = data.row(picks[c]);
86 }
87 const FiffInfo pickedInfo = info.pick_info(sel);
88 const MatrixXd filtered = FirFilter::filterDataZeroDouble(picked, sf, p.dFilterLow, p.dFilterHigh, FirFilter::FirWindow::Hamming);
89
90 // Epochs(proj=True) applies every info projector before rejection.
91 std::vector<FiffProj> active = info.projs;
92 for (FiffProj& proj : active) {
93 proj.active = true;
94 }
95 MatrixXd projector;
96 FiffProj::make_projector(active, pickedInfo.ch_names, projector, pickedInfo.bads);
97
98 std::vector<double> limits(picks.size(), 0.0);
99 for (int c = 0; c < std::ssize(picks); ++c) {
100 const FiffChInfo& ch = pickedInfo.chs[c];
101 if (ch.kind == FIFFV_MEG_CH) {
102 limits[c] = ch.unit == FIFF_UNIT_T ? p.dMagReject : p.dGradReject;
103 } else if (ch.kind == FIFFV_EEG_CH) {
104 limits[c] = p.dEegReject;
105 } else if (ch.kind == FIFFV_EOG_CH) {
106 limits[c] = p.dEogReject;
107 }
108 }
109
110 const int first = static_cast<int>(std::round(p.dTmin * sf));
111 const int last = static_cast<int>(std::round(p.dTmax * sf));
112 MatrixXd sum = MatrixXd::Zero(picks.size(), last - first + 1);
113 int nave = 0;
114 for (int ev : events) {
115 if (ev + first < 0 || ev + last >= filtered.cols()) {
116 continue;
117 }
118 const MatrixXd epoch = projector * filtered.middleCols(ev + first, last - first + 1);
119 bool good = true;
120 for (int c = 0; c < std::ssize(picks) && good; ++c) {
121 good = limits[c] <= 0.0 || epoch.row(c).maxCoeff() - epoch.row(c).minCoeff() <= limits[c];
122 }
123 if (good) {
124 sum += epoch;
125 ++nave;
126 }
127 }
128 if (nave == 0) {
129 Log::warning("[ArtifactDetect] No good {} epochs found", mode);
130 return projs;
131 }
132
133 FiffEvoked evoked;
134 evoked.info = pickedInfo;
135 evoked.nave = nave;
136 evoked.data = sum / nave;
137 evoked.times = RowVectorXf::LinSpaced(last - first + 1, static_cast<float>(first / sf), static_cast<float>(last / sf));
138 std::vector<FiffProj> exg = FiffProj::compute_from_evoked(evoked, p.nGrad, p.nMag, p.nEeg);
139 for (FiffProj& proj : exg) {
140 proj.desc = mode + "-" + proj.desc;
141 }
142 std::ranges::move(exg, std::back_inserter(projs));
143 return projs;
144}

150std::vector<int> ArtifactDetect::qrsDetector(const RowVectorXd& filtered, double dSFreq, double dThreshold, double dTStart)
151{
152 // Adapted from mne.preprocessing.qrs_detector (MNE-Python, BSD-3-Clause), filtering done by the caller.
153 const int winSize = static_cast<int>(std::nearbyint(60.0 * dSFreq / 120.0));
154 const int init = static_cast<int>(dSFreq);
155 const int nStart = static_cast<int>(dSFreq * dTStart);
156 const RowVectorXd ecgAbs = filtered.tail(std::max<Index>(filtered.size() - nStart, 0)).cwiseAbs();
157 const int nPoints = static_cast<int>(ecgAbs.size());
158 if (nPoints < 3 * init || init < 1) {
159 Log::warning("ArtifactDetect::detectEcg: need at least 3 s of data after tstart.");
160 return {};
161 }
162 const double initMax = (ecgAbs.segment(0, init).maxCoeff() + ecgAbs.segment(init, init).maxCoeff() + ecgAbs.segment(2 * init, init).maxCoeff()) / 3.0;
163
164 std::vector<double> threshRuns;
165 if (dThreshold < 0.0) {
166 for (int k = 0; k < 16; ++k)
167 threshRuns.push_back(0.3 + 0.05 * k);
168 } else {
169 threshRuns.push_back(dThreshold);
170 }
171
172 std::vector<std::vector<int>> cleanEvents;
173 for (double threshValue : threshRuns) {
174 const double thresh1 = initMax * threshValue;
175 std::vector<int> times;
176 std::vector<double> rms;
177 for (int ii = 0; ii < nPoints - winSize;) {
178 const auto window = ecgAbs.segment(ii, winSize);
179 if (window(0) > thresh1) {
180 Index maxTime = 0;
181 window.maxCoeff(&maxTime);
182 times.push_back(ii + static_cast<int>(maxTime));
183 rms.push_back(std::sqrt(window.squaredNorm() / winSize));
184 ii += winSize;
185 } else {
186 ++ii;
187 }
188 }
189 // mne's crossing count per window is always < n_thresh (the window starts above threshold), so it is omitted
190 if (rms.empty())
191 continue;
192 double mean = 0.0;
193 for (double r : rms)
194 mean += r;
195 mean /= static_cast<double>(rms.size());
196 double var = 0.0;
197 for (double r : rms)
198 var += (r - mean) * (r - mean);
199 const double rmsThresh = mean + std::sqrt(var / static_cast<double>(rms.size())) * 2.5;
200 std::vector<int> events;
201 for (int k = 0; k < std::ssize(times); ++k) {
202 if (rms[static_cast<std::size_t>(k)] < rmsThresh)
203 events.push_back(times[k] + nStart);
204 }
205 if (!events.empty())
206 cleanEvents.push_back(events);
207 }
208 if (cleanEvents.empty())
209 return {};
210
211 // Pick the threshold whose heart rate is closest to the median plausible (40-160 bpm) one, else 80 bpm
212 std::vector<double> rates;
213 std::vector<double> plausible;
214 for (const std::vector<int>& events : std::as_const(cleanEvents)) {
215 rates.push_back(60.0 * events.size() / (filtered.size() / dSFreq));
216 if (rates.back() >= 40.0 && rates.back() <= 160.0)
217 plausible.push_back(rates.back());
218 }
219 double ideal = 80.0;
220 if (!plausible.empty()) {
221 std::sort(plausible.begin(), plausible.end());
222 const std::size_t m = plausible.size() / 2;
223 ideal = plausible.size() % 2 ? plausible[m] : 0.5 * (plausible[m - 1] + plausible[m]);
224 }
225 std::size_t best = 0;
226 for (std::size_t k = 1; k < rates.size(); ++k) {
227 if (std::abs(rates[k] - ideal) < std::abs(rates[best] - ideal))
228 best = k;
229 }
230 return cleanEvents[static_cast<int>(best)];
231}

237std::vector<int> ArtifactDetect::detectEcg(const MatrixXd& matData,
238 const FiffInfo& fiffInfo,
239 double dSFreq,
240 const EcgParams& params)
241{
242 // Adapted from mne.preprocessing.find_ecg_events / _make_ecg (MNE-Python, BSD-3-Clause).
243 if (matData.cols() == 0)
244 return {};
245
246 RowVectorXd ecg;
247 const std::vector<int> ecgIdx = goodChannelsOfKind(fiffInfo, FIFFV_ECG_CH);
248 if (!ecgIdx.empty()) {
249 ecg = matData.row(ecgIdx.front());
250 } else {
251 std::vector<int> mags, grads;
252 for (int i : goodChannelsOfKind(fiffInfo, FIFFV_MEG_CH))
253 (fiffInfo.chs[i].unit == FIFF_UNIT_T ? mags : grads).push_back(i);
254 const std::vector<int>& meg = mags.empty() ? grads : mags;
255 if (meg.empty()) {
256 Log::warning("ArtifactDetect::detectEcg: no ECG channel and no MEG channels to synthesise one from.");
257 return {};
258 }
259 ecg = RowVectorXd::Zero(matData.cols());
260 for (int i : meg)
261 ecg += matData.row(i);
262 ecg /= static_cast<double>(meg.size());
263 }
264
265 const RowVectorXd filtered = FirFilter::filterDataZeroDouble(ecg, dSFreq, params.dFilterLow, params.dFilterHigh);
266 return qrsDetector(filtered, dSFreq, params.dQrsThreshold, params.dTStart);
267}

271std::vector<int> ArtifactDetect::detectEog(const MatrixXd& matData,
272 const FiffInfo& fiffInfo,
273 double dSFreq,
274 const EogParams& params)
275{
276 // Adapted from mne.preprocessing._find_eog_events (MNE-Python, BSD-3-Clause).
277 const std::vector<int> eogIdx = goodChannelsOfKind(fiffInfo, FIFFV_EOG_CH);
278 if (eogIdx.empty() || matData.cols() == 0) {
279 Log::warning("ArtifactDetect::detectEog: No EOG channel found.");
280 return {};
281 }
282
283 // The channel with the most 2-45 Hz energy
284 MatrixXd eog(eogIdx.size(), matData.cols());
285 for (int k = 0; k < std::ssize(eogIdx); ++k)
286 eog.row(k) = matData.row(eogIdx[k]);
287 const MatrixXd broad = FirFilter::filterDataZeroDouble(eog, dSFreq, 2.0, std::min(45.0, dSFreq / 2.0 - 0.75));
288 Index best = 0;
289 broad.rowwise().squaredNorm().maxCoeff(&best);
290
291 const RowVectorXd filtered = FirFilter::filterDataZeroDouble(eog.row(best), dSFreq, params.dFilterLow, params.dFilterHigh);
292 const double mean = filtered.mean();
293 const int extrema = std::abs(filtered.maxCoeff() - mean) > std::abs(filtered.minCoeff() - mean) ? 1 : -1;
294 const int nStart = static_cast<int>(dSFreq * params.dTStart);
295
296 std::vector<int> events;
297 const VectorXd tail = filtered.tail(std::max<Index>(filtered.size() - nStart, 0)).transpose();
298 for (const auto& peak : peakFinderMne(tail, params.dThreshold, extrema))
299 events.push_back(peak.first + nStart);
300 return events;
301}
