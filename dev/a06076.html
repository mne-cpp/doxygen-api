Detailed Description

Covariance matrix storage.

Stores a noise or source covariance matrix in dense (packed lower-triangle), diagonal, or sparse form together with its eigendecomposition and associated metadata (projection operator, SSS info, channel classification, bad channels).

// Whiten one time point of the evoked response with the projected noise covariance
QFile aveFile(data + "/MEG/sample/sample_audvis-ave.fif");
FiffEvoked ave((data + "/MEG/sample/sample_audvis-ave.fif").toStdU16String(), 0, {}, false);
StringList megNames; // the 305 good MEG channels
for (const std::string& name : ave.info.ch_names) {
if (name.starts_with("MEG") && !std::ranges::contains(ave.info.bads, name)) {
megNames.push_back(name);
}
}
ave = ave.pick_channels(megNames);
const int nMeg = static_cast<int>(megNames.size());
std::unique_ptr<MNEProjOp> aveProj = MNEProjOp::make_projection({aveFile.fileName().toStdU16String()}, ave.info.chs).value();
aveProj->assign_channels(megNames);
VectorXf x = ave.data.col(200).cast<float>(); // t = 133 ms
const bool projected = aveProj->make_proj() && aveProj->project_vector(x, true);
auto noise = MNECovMatrix::read((data + "/MEG/sample/sample_audvis-cov.fif").toStdU16String(), FIFFV_MNE_NOISE_COV);
const int ncovFile = noise->ncov;
auto noiseMeg = noise->pick_chs_omit(megNames, false, ave.info.chs);
const bool covProjected = aveProj->apply_cov(*noiseMeg);
noiseMeg->classify_channels(ave.info.chs, nMeg);
noiseMeg->decompose_eigen(); // the 3 projected directions become zero eigenvalues
VectorXf whitened(nMeg);
noiseMeg->whiten_vector(x, whitened, nMeg);
