49PanelWidget::PanelWidget(std::shared_ptr<Panel> panel, const QString& settingsPath, QWidget* parent)
50: AbstractView(parent)
51, m_panel(std::move(panel))
52{
53 m_sSettingsPath = settingsPath;
54 setWindowTitle(QString::fromStdString(m_panel->title()));
55 if (m_panel->width() > 0) {
56 setFixedWidth(m_panel->width());
57 }
58 loadSettings();
59 auto* layout = new QVBoxLayout(this);
60 layout->setContentsMargins(0, 0, 0, 0);
61 m_formPanel = new SettingsPanel(m_panel->shared_form(), this);
62 layout->addWidget(m_formPanel);
63 if (m_panel->figure()) {
64 // The figure takes the room; the form keeps its height and shows only with fields
65 m_figureViewport = new SceneViewport(m_panel->figure(), this);
66 layout->addWidget(m_figureViewport, 1);
67 m_formPanel->setVisible(!m_panel->form().fields().empty());
68 m_structureListener = m_panel->form().add_structure_listener([this](const std::string&) {
69 m_formPanel->setVisible(!m_panel->form().fields().empty());
70 });
71 }
72 // The look of viz::FormView, so a panel looks alike in every backend
73 setTheme(viz::Theme::light());
74}
