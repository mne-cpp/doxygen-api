51// Polar angle and azimuth of a direction, as MNE-C's sphere_coord: on the z axis both are zero.
52Vector2d polar_azimuth(const Vector3d& r)
53{
54 const double rxy = std::hypot(r.x(), r.y());
55 if (rxy < 1e-6)
56 return Vector2d::Zero();
57 const double phi = std::atan2(r.y(), r.x());
58 return {std::acos(r.z() / r.norm()), phi < 0.0 ? phi + 2.0 * std::numbers::pi : phi};
59}

67std::optional<std::vector<Vector2d>> LayoutMaker::make_layout(std::span<const Vector3d> positions,
68 const LayoutOptions& options)
69{
70 if (positions.empty()) {
71 Log::warning("No positions to lay out.");
72 return std::nullopt;
73 }
74
75 Vector3d origin = Vector3d::Zero();
76 if (options.fit_sphere) {
77 MatrixX3f points(static_cast<Index>(positions.size()), 3);
78 for (Index k = 0; k < points.rows(); ++k)
79 points.row(k) = positions[static_cast<std::size_t>(k)].cast<float>().transpose();
80 const Sphere sphere = Sphere::fit_sphere_simplex(points, 0.05);
81 if (sphere.radius() > 0.0f) {
82 origin = sphere.center().cast<double>();
83 Log::info("Best-fitting sphere: origin ({}, {}, {}), radius {}", origin.x(), origin.y(), origin.z(), sphere.radius());
84 } else {
85 // As MNE-C mne_make_eeg_layout: a failed fit falls back to the origin
86 Log::warning("Sphere fitting failed, using the origin");
87 }
88 }
89
90 // Azimuthal equidistant projection: the polar angle becomes the distance from the centre
91 std::vector<Vector2d> centres;
92 centres.reserve(positions.size());
93 for (const Vector3d& position : positions) {
94 const Vector2d angles = polar_azimuth(position - origin);
95 const double distance = options.radius * 2.0 * angles.x() / std::numbers::pi;
96 centres.emplace_back(distance * std::cos(angles.y()), distance * std::sin(angles.y()));
97 }
98
99 Vector2d minimum = centres.front();
100 Vector2d maximum = minimum;
101 for (const Vector2d& centre : centres) {
102 minimum = minimum.cwiseMin(centre);
103 maximum = maximum.cwiseMax(centre);
104 }
105 if (minimum.x() == maximum.x() || minimum.y() == maximum.y()) {
106 Log::critical("Cannot make a layout. All positions are identical");
107 return std::nullopt;
108 }
109
110 const Vector2d halfBox(0.5 * options.box_width, 0.5 * options.box_height);
111 const Vector2d mirror(options.mirror_x ? -1.0 : 1.0, options.mirror_y ? -1.0 : 1.0);
112 for (Vector2d& centre : centres)
113 centre = (centre - halfBox).cwiseProduct(mirror);
114 return centres;
115}

119bool LayoutMaker::write_lout(const std::filesystem::path& path,
120 std::span<const Vector2d> corners,
121 const StringList& names,
122 const LayoutOptions& options)
123{
124 std::ofstream file(path, std::ios::binary);
125 if (!file) {
126 Log::critical("Cannot open the layout file {}", path.string());
127 return false;
128 }
129 file << "0.000000 0.000000 0.000000 0.000000\n";
130 for (std::size_t k = 0; k < corners.size(); ++k) {
131 file << std::format("{} {:.6f} {:.6f} {:.6f} {:.6f}", k + 1, corners[k].x(), corners[k].y(), options.box_width, options.box_height);
132 if (k < names.size())
133 file << ' ' << names[k];
134 file << '\n';
135 }
136 return static_cast<bool>(file);
137}
