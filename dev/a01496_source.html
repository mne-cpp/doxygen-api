138template<typename T>
139[[nodiscard]] AcqExpected<AcqEncodeResult> write(AcqWriter& writer,
140 const Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& in,
141 std::chrono::nanoseconds timeout,
142 AcqScaling scaling = AcqScaling::Physical)
143{
144 auto view = writer.beginWrite(static_cast<std::size_t>(in.cols()), timeout);
145 if (!view) {
146 return view.error();
147 }
148 AcqConvertOptions options;
149 options.scaling = scaling;
150 auto result = encode(ref(in), *view, options);
151 writer.commit(result ? view->sampleCount() : 0);
152 return result;
153}
