69// MNE-Python tells oxy- from deoxyhaemoglobin by the coil type and names the channels "S1_D1 hbo".
70int coilType(const AcqChannelInfo& channel, int kind)
71{
72 if (kind == FIFFV_EEG_CH) {
73 return FIFFV_COIL_EEG;
74 }
75 if (kind == FIFFV_FNIRS_CH) {
76 const auto endsWith = [&](const char* suffix) {
77 const std::size_t n = std::strlen(suffix);
78 return channel.name.size() >= n && channel.name.compare(channel.name.size() - n, n, suffix) == 0;
79 };
80 return endsWith("hbo") ? FIFFV_COIL_FNIRS_HBO
81 : endsWith("hbr") ? FIFFV_COIL_FNIRS_HBR
82 : FIFFV_COIL_FNIRS_CW_AMPLITUDE;
83 }
84 return FIFFV_COIL_NONE;
85}

87std::vector<std::byte> serialize(const FiffInfo& info)
88{
89 // The blocks of the source file are not copied: the payload must not depend on files elsewhere.
90 FiffInfo copy = info;
91 copy.filename.clear();
92 auto memory = std::make_unique<mne::MemoryByteStream>();
93 const mne::MemoryByteStream& written = *memory;
94 Eigen::RowVectorXd cals;
95 auto stream = FiffStream::start_writing_raw(std::move(memory), copy, cals, defaultMatrixXi, false);
96 stream->finish_writing_raw();
97 return written.data();
98}

100bool deserialize(const std::vector<std::byte>& data, FiffInfo& info)
101{
102 FiffStream stream(std::make_unique<mne::MemoryByteStream>(data));

105 }
106 std::shared_ptr<FiffDirNode> node;

116AcqStreamInfo mne::fromFiffInfo(const FiffInfo& info, AcqSampleFormat format)
117{
118 AcqStreamInfo stream;
119 stream.name = "raw";
120 stream.sourceId = info.filename;
121 stream.format = format;
122 stream.layout = AcqLayout::Interleaved;
123 stream.sfreq = static_cast<double>(info.sfreq);
124 stream.channels.reserve(static_cast<std::size_t>(info.chs.size()));
125 for (const FiffChInfo& ch : info.chs) {
126 AcqChannelInfo channel;
127 channel.name = ch.ch_name;
128 channel.kind = detail::acq_kind_of(ch.kind);
129 channel.unit = detail::acq_unit_of(ch.unit);
130 channel.unitExponent = ch.unit_mul;
131 channel.scale = static_cast<double>(ch.cal) * static_cast<double>(ch.range);
132 AcqFiffChannel fiff;
133 fiff.kind = ch.kind;
134 fiff.coilType = ch.chpos.coil_type;
135 fiff.logicalNo = ch.logNo;
136 fiff.scannerNo = ch.scanNo;
137 fiff.unit = ch.unit;
138 fiff.unitMul = ch.unit_mul;
139 fiff.coordFrame = ch.coord_frame;
140 fiff.cal = ch.cal;
141 fiff.range = ch.range;
142 for (std::size_t i = 0; i < 3; ++i) {
143 const auto e = static_cast<Eigen::Index>(i);
144 fiff.loc[i] = ch.chpos.r0[e];
145 fiff.loc[3 + i] = ch.chpos.ex[e];
146 fiff.loc[6 + i] = ch.chpos.ey[e];
147 fiff.loc[9 + i] = ch.chpos.ez[e];
148 }
149 channel.fiff = fiff;
150 stream.channels.push_back(std::move(channel));
151 }
152 AcqExtension extension;
153 extension.schema = kMeasInfoSchema;
154 extension.data = serialize(info);
155 stream.extensions.push_back(std::move(extension));
156 return stream;
157}

161AcqExpected<FiffInfo> mne::toFiffInfo(const AcqStreamInfo& info)
162{
163 if (isComplex(info.format) || info.segmentation.mode != AcqSegmentation::Mode::Continuous) {
164 return AcqError(AcqErrorCode::UnsupportedFormat, "FIFF raw data holds continuous real samples only");
165 }
166 FiffInfo fiff;
167 bool fromFiff = false;
168 for (const AcqExtension& extension : info.extensions) {
169 if (extension.schema == kMeasInfoSchema) {
170 if (!deserialize(extension.data, fiff)) {
171 return AcqError(AcqErrorCode::InvalidParameter, "corrupt fiff.MeasInfo/1 extension");
172 }
173 fromFiff = true;
174 }
175 }
176 if (!fromFiff) {
177 fiff.highpass = 0.0f;
178 fiff.lowpass = static_cast<float>(info.sfreq / 2.0);
179 fiff.proj_id = 0;
180 fiff.meas_date[0] = -1;
181 fiff.meas_date[1] = 0;
182 }
183 fiff.sfreq = static_cast<float>(info.sfreq);
184 fiff.chs.clear();
185 fiff.ch_names.clear();
186 int index = 0;
187 for (const AcqChannelInfo& channel : info.channels) {
188 FiffChInfo ch;
189 ch.ch_name = channel.name;
190 ch.scanNo = ++index;
191 if (channel.fiff) {
192 const AcqFiffChannel& f = *channel.fiff;
193 ch.kind = f.kind;
194 ch.chpos.coil_type = f.coilType;
195 ch.logNo = f.logicalNo;
196 ch.unit = f.unit;
197 ch.coord_frame = f.coordFrame;
198 for (std::size_t i = 0; i < 3; ++i) {
199 const auto e = static_cast<Eigen::Index>(i);
200 ch.chpos.r0[e] = f.loc[i];
201 ch.chpos.ex[e] = f.loc[3 + i];
202 ch.chpos.ey[e] = f.loc[6 + i];
203 ch.chpos.ez[e] = f.loc[9 + i];
204 }
205 const bool unchanged = channel.scale == static_cast<double>(f.cal) * static_cast<double>(f.range) &&
206 channel.unitExponent == f.unitMul;
207 ch.cal = unchanged ? f.cal : static_cast<float>(channel.scale * std::pow(10.0, channel.unitExponent));
208 ch.range = unchanged ? f.range : 1.0f;
209 ch.unit_mul = unchanged ? f.unitMul : FIFF_UNITM_NONE;
210 } else {
211 // MNE-Python applies cal * range but not unit_mul, so the power of ten goes into cal.
212 ch.kind = channel.unit == AcqUnit::Unknown ? FIFFV_MISC_CH : detail::fiff_kind_of(channel.kind);
213 ch.unit = detail::fiff_unit_of(channel.unit);
214 ch.unit_mul = FIFF_UNITM_NONE;
215 ch.logNo = index;
216 ch.chpos.coil_type = coilType(channel, ch.kind);
217 ch.coord_frame = ch.kind == FIFFV_EEG_CH ? FIFFV_COORD_HEAD : FIFFV_COORD_UNKNOWN;
218 ch.cal = static_cast<float>(channel.scale * std::pow(10.0, channel.unitExponent));
219 ch.range = 1.0f;
220 }
221 fiff.chs.push_back(ch);
222 fiff.ch_names.push_back(ch.ch_name);
223 }
224 fiff.nchan = static_cast<fiff_int_t>(fiff.chs.size());
225 return fiff;
226}
