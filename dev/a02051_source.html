355void AveragingSettingsView::loadSettings()
356{
357 if (m_sSettingsPath.isEmpty()) {
358 return;
359 }
360
361 const detail::ScopedFlag silent(m_bSilent);
362 // The stored baseline is checked against the stored epoch window, not against the current one
363 m_pForm->set_range("baselineFromSeconds", -10000, 10000);
364 m_pForm->set_range("baselineToSeconds", -10000, 10000);
365 m_pPanel->loadSettings(m_sSettingsPath + QString("/AveragingSettingsView"));
366
367 const int preStim = getPreStimMSeconds();
368 const int postStim = getPostStimMSeconds();
369 if (const int from = getBaselineFromSeconds(); from < -preStim || from > postStim) {
370 m_pForm->set_value("baselineFromSeconds", -preStim);
371 }
372 if (const int to = getBaselineToSeconds(); to > postStim || to < -preStim) {
373 m_pForm->set_value("baselineToSeconds", 0);
374 }
375
376 m_pForm->set_range("baselineFromSeconds", -preStim, postStim);
377 m_pForm->set_range("baselineToSeconds", -preStim, postStim);
378
379 QSettings settings("MNECPP");
380 m_sCurrentStimChan = settings.value(m_sSettingsPath + QString("/AveragingSettingsView/currentStimChannel"), "STI 014").toString().toStdString();
381}

404void AveragingSettingsView::onFormChanged(const std::string& key)
405{
406 if (m_bSilent) {
407 return;
408 }
409
410 // Fields that only report their value, and actions that report false like the clicked buttons did
411 for (const auto& [name, signal] : kBoolSignals) {
412 if (key == name) {
413 emit(this->*signal)(m_pForm->get<bool>(key));
414 if (m_pForm->value(key)) {
415 saveSettings();
416 }
417 return;
418 }
419 }
420
421 // The epoch window bounds the baseline, the baseline bounds itself
422 const int mSeconds = m_pForm->get<int>(key);
423 const auto range = [this](const std::string& field) {
424 const viz::SettingField* f = m_pForm->field(field);
425 return std::pair{static_cast<int>(f->minimum), static_cast<int>(f->maximum)};
426 };
427 if (key == "preStimSeconds") {
428 m_pForm->set_range("baselineToSeconds", -mSeconds, range("baselineToSeconds").second);
429 m_pForm->set_range("baselineFromSeconds", -mSeconds, range("baselineFromSeconds").second);
430 } else if (key == "postStimSeconds") {
431 m_pForm->set_range("baselineToSeconds", range("baselineToSeconds").first, mSeconds);
432 m_pForm->set_range("baselineFromSeconds", range("baselineFromSeconds").first, mSeconds);
433 } else if (key == "baselineFromSeconds") {
434 m_pForm->set_range("baselineToSeconds", mSeconds, range("baselineToSeconds").second);
435 } else if (key == "baselineToSeconds") {
436 m_pForm->set_range("baselineFromSeconds", range("baselineFromSeconds").first, mSeconds);
437 }
438
439 if (key == "currentStimChannel") {
440 m_sCurrentStimChan = m_pForm->get<std::string>(key);
441 emit changeStimChannel(m_sCurrentStimChan);
442 } else if (const auto it = std::ranges::find(kIntSignals, key, &IntSignal::first); it != kIntSignals.end()) {
443 emit(this->*it->second)(mSeconds);
444 } else {
445 return;
446 }
447
448 saveSettings();
449}
