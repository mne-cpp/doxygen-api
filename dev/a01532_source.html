53BabyMegInfo simulated_system(const BabyMegSimulation& simulation)
54{
55 BabyMegInfo info;
56 info.rate = simulation.rate;
57 info.samples = simulation.samples;
58 const std::size_t sensors = simulation.meg + simulation.ref;
59 for (std::size_t i = 0; i < sensors; ++i) {
60 const bool inner = i < simulation.meg;
61 const std::size_t n = inner ? simulation.meg : simulation.ref;
62 const std::size_t k = inner ? i : i - simulation.meg;
63 // A Fibonacci lattice on the upper half sphere, normals pointing outward
64 const double z = 1.0 - (static_cast<double>(k) + 0.5) / static_cast<double>(n);
65 const double r = std::sqrt(1.0 - z * z);
66 const double phi = static_cast<double>(k) * std::numbers::pi * (3.0 - std::sqrt(5.0));
67 const std::array<double, 3> normal = {r * std::cos(phi), r * std::sin(phi), z};
68 const double radius = inner ? 0.075 : 0.095;
69 BabyMegChannel& channel = info.channels.emplace_back();
70 channel.name = std::format("MEG{:03}", i + 1);
71 channel.coil_type = inner ? kCoilInner : kCoilOuter;
72 channel.cal = kMegCal;
73 channel.gain = kMegGain;
74 for (std::size_t c = 0; c < 3; ++c) {
75 channel.loc[c] = static_cast<float>(radius * normal[c]);
76 channel.loc[9 + c] = static_cast<float>(normal[c]);
77 }
78 channel.loc[3] = static_cast<float>(-std::sin(phi)); // ex, tangential
79 channel.loc[4] = static_cast<float>(std::cos(phi));
80 channel.loc[6] = static_cast<float>(normal[1] * channel.loc[5] - normal[2] * channel.loc[4]); // ey = ez x ex
81 channel.loc[7] = static_cast<float>(normal[2] * channel.loc[3] - normal[0] * channel.loc[5]);
82 channel.loc[8] = static_cast<float>(normal[0] * channel.loc[4] - normal[1] * channel.loc[3]);
83 }
84 for (std::size_t t = 0; t < kTriggerLines; ++t) {
85 BabyMegChannel& channel = info.channels.emplace_back();
86 channel.name = std::format("TRG{:03}", t + 1);
87 channel.coil_type = fiff_kind::kStim;
88 }
89 BabyMegChannel& digital = info.channels.emplace_back();
90 digital.name = "DTRG01";
91 digital.coil_type = fiff_kind::kStim;
92 return info;
93}

95void put_big(std::byte* out, float value)
96{
97 const auto bits = std::bit_cast<std::uint32_t>(value);
98 for (std::size_t i = 0; i < 4; ++i) {
99 out[i] = static_cast<std::byte>((bits >> (8 * (3 - i))) & 0xFFU);
100 }
101}

122 AcqError start() override
123 {
124 std::string uri = std::format("synthetic://?sfreq={}&meg={}&eeg=0&stim=1&block={}&seed={}&realtime={}", m_simulation.rate,
125 m_simulation.meg + m_simulation.ref, m_simulation.samples, m_simulation.seed, m_simulation.realtime ? 1 : 0);
126 if (m_simulation.duration > 0.0) {
127 uri += std::format("&duration={}", m_simulation.duration);
128 }
129 auto session = AcqDevice::open(uri);
130 if (!session) {
131 return session.error();
132 }
133 AcqReaderOptions gating;
134 gating.gating = true; // the session waits for this reader instead of overwriting unread samples
135 auto reader = session->stream().reader(AcqReader::ReadMode::Lossless, gating);
136 if (!reader) {
137 return reader.error();
138 }
139 if (AcqError e = session->start(); !e.ok()) {
140 return e;
141 }
142 m_session = std::move(*session);
143 m_reader = std::move(*reader);
144 return {};
145 }
146
147 AcqExpected<std::span<const std::byte>> next(std::chrono::milliseconds timeout) override
148 {
149 const std::size_t samples = m_simulation.samples;
150 const std::size_t sensors = m_simulation.meg + m_simulation.ref;
151 const std::size_t width = m_info.channels.size();
152 auto view = m_reader.acquire(samples, timeout, samples);
153 if (!view) {
154 return view.error();
155 }
156 if (view->sampleCount() < samples) {
157 static_cast<void>(m_reader.release(*view)); // whole packages only; the end of the session cuts the rest
158 return AcqError(AcqErrorCode::EndOfStream);
159 }
160 m_body.assign(1 + samples * width * sizeof(float), std::byte{0});
161 m_body[0] = std::byte{'4'};
162 std::byte* out = m_body.data() + 1;
163 for (std::size_t p = 0; p < view->partCount(); ++p) {
164 const AcqBlockPart& part = view->part(p);
165 for (std::size_t s = 0; s < part.samples; ++s) {
166 for (std::size_t c = 0; c < sensors; ++c) {
167 float tesla = 0.0F;
168 std::memcpy(&tesla, part.at(c, s), sizeof(tesla));
169 put_big(out, tesla * kMegGain / kMegCal);
170 out += sizeof(float);
171 }
172 float code = 0.0F;
173 std::memcpy(&code, part.at(sensors, s), sizeof(code));
174 const auto word = static_cast<std::uint32_t>(std::lround(code));
175 for (std::size_t t = 0; t < kTriggerLines; ++t) {
176 put_big(out, ((word >> t) & 1U) != 0 ? kTriggerHigh : 0.0F);
177 out += sizeof(float);
178 }
179 out += sizeof(float); // DTRG01 is the client's to fill
180 }
181 }
182 if (AcqError e = m_reader.release(*view); !e.ok()) {
183 return e;
184 }
185 return std::span<const std::byte>(m_body);
186 }
187
188 void stop() override
189 {
190 m_reader = AcqReader();
191 m_session.close();
192 }
193
194private:
195 BabyMegSimulation m_simulation;
196 BabyMegInfo m_info;
197 AcqDevice m_session;
198 AcqReader m_reader;
199 std::vector<std::byte> m_body;
200};
