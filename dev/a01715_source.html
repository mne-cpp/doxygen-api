141void MinimumNormPanel::set_trigger_types(const std::vector<std::string>& types)
142{
143 std::vector<SettingChoice> choices = form().field(kTriggerType)->choices;
144 const std::size_t count = choices.size();
145 for (const std::string& type : types) {
146 if (std::ranges::none_of(choices, [&type](const SettingChoice& choice) { return choice.value == type; })) {
147 choices.push_back({type, type});
148 }
149 }
150 if (choices.size() != count) {
151 form().set_choices(kTriggerType, std::move(choices));
152 }
153}
