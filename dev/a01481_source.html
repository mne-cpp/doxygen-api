82 [[nodiscard]] std::int64_t physicalNs(std::uint64_t sample) const noexcept
83 {
84 const double ds = static_cast<double>(sample) - static_cast<double>(anchorSample);
85 const auto t = anchorHostNs + static_cast<std::int64_t>(std::llround(ds * 1e9 / (sfreq * rateCorrection)));
86 return output ? t + pipelineDelayNs : t - pipelineDelayNs;
87 }

94 [[nodiscard]] std::uint64_t sampleAt(std::int64_t ns) const noexcept
95 {
96 const std::int64_t boundaryNs = output ? ns - pipelineDelayNs : ns + pipelineDelayNs;
97 const double s = static_cast<double>(anchorSample) +
98 static_cast<double>(boundaryNs - anchorHostNs) * sfreq * rateCorrection / 1e9;
99 return s <= 0.0 ? 0U : static_cast<std::uint64_t>(std::llround(s));
100 }
