168void CompensatorView::redrawGUI()
169{
170 if (m_pComps.empty()) {
171 return;
172 }
173
174 m_qListCompCheckBox.clear();
175 detail::resetLayout(this);
176
177 // Compensation Selection
178 auto topLayout = std::make_unique<QGridLayout>();
179
180 for (int i = 0; i < std::ssize(m_pComps); ++i) {
181 QString numStr;
182 QCheckBox* checkBox = new QCheckBox(numStr.setNum(m_pComps[i].kind), this);
183
184 m_qListCompCheckBox.append(checkBox);
185
186 connect(checkBox, &QCheckBox::toggled,
187 this, &CompensatorView::onCheckCompStatusChanged);
188
189 checkBox->setChecked(m_mapCompActive[m_pComps.at(i).kind]);
190
191 topLayout->addWidget(checkBox, i, 0);
192 }
193 setLayout(topLayout.release());
194}

198void CompensatorView::onCheckCompStatusChanged()
199{
200 if (QCheckBox* pCheckBox = qobject_cast<QCheckBox*>(sender())) {
201 bool currentState = false;
202 QString compName = pCheckBox->text();
203
204 for (int i = 0; i < m_qListCompCheckBox.size(); ++i) {
205 if (m_qListCompCheckBox[i]->text() != compName) {
206 m_qListCompCheckBox[i]->setChecked(false);
207 m_mapCompActive[compName.toInt()] = false;
208 } else {
209 currentState = m_qListCompCheckBox[i]->isChecked();
210 m_mapCompActive[compName.toInt()] = currentState;
211 }
212 }
213
214 if (currentState) {
215 emit compSelectionChanged(compName.toInt());
216 m_iLastTo = compName.toInt();
217 } else { //If none selected
218 emit compSelectionChanged(0);
219 m_iLastTo = 0;
220 }
221 }
222
223 saveSettings();
224}
