89std::string channelTypeName(const FiffChInfo& ch)
90{
91 // mne-python's _first_rule (by kind) refined by _second_rules (MEG by unit, EEG/fNIRS/eye tracking by coil type)
92 static const std::unordered_map<int, std::string> byKind{
93 {FIFFV_REF_MEG_CH, "ref_meg"}, {FIFFV_STIM_CH, "stim"}, {FIFFV_EOG_CH, "eog"}, {FIFFV_EMG_CH, "emg"}, {FIFFV_ECG_CH, "ecg"}, {FIFFV_RESP_CH, "resp"}, {FIFFV_MISC_CH, "misc"}, {FIFFV_EXCI_CH, "exci"}, {FIFFV_IAS_CH, "ias"}, {FIFFV_SYST_CH, "syst"}, {FIFFV_SEEG_CH, "seeg"}, {FIFFV_DBS_CH, "dbs"}, {FIFFV_BIO_CH, "bio"}, {FIFFV_DIPOLE_WAVE_CH, "dipole"}, {FIFFV_GOODNESS_FIT_CH, "gof"}, {FIFFV_ECOG_CH, "ecog"}, {FIFFV_TEMPERATURE_CH, "temperature"}, {FIFFV_GALVANIC_CH, "gsr"}};
94 static const std::unordered_map<int, std::string> eegByCoil{
95 {FIFFV_COIL_EEG, "eeg"}, {FIFFV_COIL_EEG_BIPOLAR, "eeg"}, {FIFFV_COIL_NONE, "eeg"}, {FIFFV_COIL_EEG_CSD, "csd"}};
96 static const std::unordered_map<int, std::string> fnirsByCoil{
97 {FIFFV_COIL_FNIRS_HBO, "hbo"}, {FIFFV_COIL_FNIRS_HBR, "hbr"}, {FIFFV_COIL_FNIRS_CW_AMPLITUDE, "fnirs_cw_amplitude"}, {FIFFV_COIL_FNIRS_FD_AC_AMPLITUDE, "fnirs_fd_ac_amplitude"}, {FIFFV_COIL_FNIRS_FD_PHASE, "fnirs_fd_phase"}, {FIFFV_COIL_FNIRS_OD, "fnirs_od"}, {FIFFV_COIL_FNIRS_TD_GATED_AMPLITUDE, "fnirs_td_gated_amplitude"}, {FIFFV_COIL_FNIRS_TD_MOMENTS_INTENSITY, "fnirs_td_moments_intensity"}, {FIFFV_COIL_FNIRS_TD_MOMENTS_MEAN, "fnirs_td_moments_mean"}, {FIFFV_COIL_FNIRS_TD_MOMENTS_VARIANCE, "fnirs_td_moments_variance"}};
98 static const std::unordered_map<int, std::string> eyetrackByCoil{{FIFFV_COIL_EYETRACK_POS, "eyegaze"}, {FIFFV_COIL_EYETRACK_PUPIL, "pupil"}};
99
100 const auto refine = [](const std::unordered_map<int, std::string>& rule, int key) {
101 const auto it = rule.find(key);
102 return it == rule.end() ? std::string() : it->second;
103 };
104 if (ch.kind == FIFFV_MEG_CH)
105 return refine({{FIFF_UNIT_T_M, "grad"}, {FIFF_UNIT_T, "mag"}}, ch.unit);
106 if (ch.kind == FIFFV_EEG_CH)
107 return refine(eegByCoil, ch.chpos.coil_type);
108 if (ch.kind == FIFFV_FNIRS_CH)
109 return refine(fnirsByCoil, ch.chpos.coil_type);
110 if (ch.kind == FIFFV_EYETRACK_CH)
111 return refine(eyetrackByCoil, ch.chpos.coil_type);
112 if (FIFFM_QUAT_CH(ch.kind))
113 return "chpi"; // channels relative to head position monitoring
114 return refine(byKind, ch.kind);
115}

145std::map<std::string, std::vector<int>> FiffInfoBase::channel_indices_by_type(const std::vector<int>& picks) const
146{
147 std::map<std::string, std::vector<int>> byType;
148 for (const char* type : {"grad", "mag", "ref_meg", "eeg", "csd", "seeg", "dbs", "ecog", "eog", "emg", "ecg", "resp", "bio",
149 "misc", "stim", "exci", "syst", "ias", "gof", "dipole", "chpi", "temperature", "gsr", "hbo", "hbr",
150 "fnirs_cw_amplitude", "fnirs_fd_ac_amplitude", "fnirs_fd_phase", "fnirs_od",
151 "fnirs_td_gated_amplitude", "fnirs_td_moments_intensity", "fnirs_td_moments_mean",
152 "fnirs_td_moments_variance", "eyegaze", "pupil"}) {
153 byType.try_emplace(type);
154 }
155 const std::size_t count = picks.empty() ? chs.size() : picks.size();
156 for (std::size_t k = 0; k < count; ++k) {
157 const int idx = picks.empty() ? static_cast<int>(k) : picks[k];
158 byType[channel_type(idx)].push_back(idx);
159 }
160 return byType;
161}

165RowVectorXi FiffInfoBase::pick_data_channels(const StringList& exclude) const
166{
167 static const std::vector<std::string_view> dataTypes{"mag", "grad", "ref_meg", "eeg", "csd", "seeg", "ecog", "dbs", "hbo", "hbr"};
168 std::vector<int> picks;
169 for (int i = 0; i < std::ssize(chs); ++i) {
170 const std::string type = channelTypeName(chs[i]);
171 const bool isData = std::find(dataTypes.begin(), dataTypes.end(), type) != dataTypes.end() || type.starts_with("fnirs_");
172 if (isData && std::find(exclude.begin(), exclude.end(), chs[i].ch_name) == exclude.end())
173 picks.push_back(i);
174 }
175 return Map<const RowVectorXi>(picks.data(), static_cast<Index>(picks.size()));

180RowVectorXi FiffInfoBase::pick_channels_regexp(const StringList& ch_names, const std::string& regexp)
181{
182 // Like re.match in Python: anchored at the start only
183 const std::regex re(regexp);
184 std::vector<int> sel;
185 for (int k = 0; k < static_cast<int>(ch_names.size()); ++k) {
186 if (std::regex_search(ch_names[static_cast<std::size_t>(k)], re, std::regex_constants::match_continuous))
187 sel.push_back(k);
188 }
189 return Eigen::Map<const RowVectorXi>(sel.data(), static_cast<Eigen::Index>(sel.size()));
190}

209RowVectorXi FiffInfoBase::pick_types(const std::string& meg, bool eeg, bool stim, const StringList& include, const StringList& exclude) const
210{
211 // A default-constructed info has nchan -1; count the channels actually present.
212 const std::int32_t nchPresent = static_cast<std::int32_t>(this->chs.size());
213 RowVectorXi pick = RowVectorXi::Zero(nchPresent);
214
215 fiff_int_t kind;
216 std::int32_t k;
217 for (k = 0; k < nchPresent; ++k) {
218 kind = this->chs[k].kind;
219
220 if ((kind == FIFFV_MEG_CH || kind == FIFFV_REF_MEG_CH)) {
221 if (meg == "all") {
222 pick(k) = 1;
223 } else if (meg == "grad" && this->chs[k].unit == FIFF_UNIT_T_M) {
224 pick(k) = 1;
225 } else if (meg == "mag" && this->chs[k].unit == FIFF_UNIT_T) {
226 pick(k) = 1;
227 }
228 } else if (kind == FIFFV_EEG_CH && eeg)
229 pick(k) = 1;
230 else if (kind == FIFFV_STIM_CH && stim)
231 pick(k) = 1;
232 }
233
234 // restrict channels to selection if provided
235 std::int32_t p = 0;
236 StringList myinclude;
237 for (k = 0; k < nchPresent; ++k) {
238 if (pick(0, k)) {
239 myinclude.push_back(this->ch_names[static_cast<std::size_t>(k)]);
240 ++p;
241 }
242 }
243 myinclude.insert(myinclude.end(), include.begin(), include.end());
244 p += static_cast<std::int32_t>(include.size());
245
246 RowVectorXi sel;
247 if (p != 0)
248 sel = FiffInfoBase::pick_channels(this->ch_names, myinclude, exclude);
249
250 return sel;
251}

262RowVectorXi FiffInfoBase::pick_channels(const StringList& ch_names, const StringList& include, const StringList& exclude)
263{
264 RowVectorXi sel = RowVectorXi::Zero(static_cast<Index>(ch_names.size()));
265 std::unordered_set<std::string> selected;
266 std::int32_t count = 0;
267 for (std::size_t k = 0; k < ch_names.size(); ++k) {
268 const std::string& name = ch_names[k];
269 if ((include.empty() || has_channel(include, name)) && !has_channel(exclude, name) && selected.insert(name).second) {
270 sel[count] = static_cast<int>(k);
271 ++count;
272 }
273 }
274 sel.conservativeResize(count);
275 return sel;
276}

280FiffInfoBase FiffInfoBase::pick_info(const RowVectorXi* sel) const
281{
282 FiffInfoBase res = *this; //new FiffInfo(this);
283 if (sel == nullptr)
284 return res;
285
286 //ToDo when pointer List do deletion
287 res.chs.clear();
288 res.ch_names.clear();
289
290 std::int32_t idx;
291 for (std::int32_t i = 0; i < sel->size(); ++i) {
292 idx = (*sel)(0, i);
293 res.chs.push_back(this->chs[idx]);
294 res.ch_names.push_back(this->ch_names[static_cast<std::size_t>(idx)]);
295 }
296 res.nchan = sel->size();
297 std::erase_if(res.bads, [&res](const std::string& bad) { return res.channelIndex(bad) < 0; });
298
299 return res;

343bool FiffInfoBase::readBadChannelsFromFile(const std::string& name, StringList& listOut)
344{
345 if (name.empty()) {
346 listOut.clear();
347 return true;
348 }
349 std::ifstream file(mne::path_from_utf8(name));
350 if (!file) {
351 mne::Log::critical("Cannot open bad channel file: {}", name);
352 return false;
353 }
354 StringList list;
355 std::string line;
356 while (std::getline(file, line)) {
357 const std::string_view trimmed = mne::trimmed(line);
358 if (trimmed.empty() || trimmed.front() == '#')
359 continue;
360 list.emplace_back(trimmed);
361 }
362 if (file.bad()) {
363 mne::Log::critical("Error reading bad channel file: {}", name);
364 return false;
365 }
366 listOut = std::move(list);
367 return true;
368}
