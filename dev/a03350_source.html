82class JsonStreamParser::DocumentBuilder final : public JsonHandler
83{
84public:
85 void on_start_object() override
86 {
87 m_frames.push_back({m_values.size(), m_keys.size()});
88 }
89
90 void on_end_object() override
91 {
92 const Frame frame = m_frames.back();
93 m_frames.pop_back();
94 JsonObject object;
95 object.reserve(m_values.size() - frame.firstValue);
96 for (std::size_t i = frame.firstValue, k = frame.firstKey; i < m_values.size(); ++i, ++k) {
97 JsonAccess::append_member(object, std::move(m_keys[k]), std::move(m_values[i]));
98 }
99 m_values.resize(frame.firstValue);
100 m_keys.resize(frame.firstKey);
101 JsonAccess::sort_members(object);
102 add(JsonValue(std::move(object)));
103 }
104
105 void on_start_array() override
106 {
107 m_frames.push_back({m_values.size(), m_keys.size()});
108 }
109
110 void on_end_array() override
111 {
112 const Frame frame = m_frames.back();
113 m_frames.pop_back();
114 JsonArray array;
115 array.reserve(m_values.size() - frame.firstValue);
116 for (std::size_t i = frame.firstValue; i < m_values.size(); ++i) {
117 array.append(std::move(m_values[i]));
118 }
119 m_values.resize(frame.firstValue);
120 add(JsonValue(std::move(array)));
121 }
122
123 void on_key(std::string_view key) override
124 {
125 m_keys.emplace_back(key);
126 }
127
128 void on_string(std::string_view value) override
129 {
130 add(value);
131 }
132
133 void on_integer(std::int64_t value) override
134 {
135 add(value);
136 }
137
138 void on_number(double value) override
139 {
140 add(value);
141 }
142
143 void on_bool(bool value) override
144 {
145 add(value);
146 }
147
148 void on_null() override
149 {
150 add(nullptr);
151 }

154 JsonDocument take_document()
155 {
156 JsonDocument document = m_root.is_array() ? JsonDocument(std::move(JsonAccess::array(m_root)))
157 : JsonDocument(std::move(JsonAccess::object(m_root)));
158 m_root = JsonValue();

160 }

165 m_frames.clear();

250bool is_json_number(std::string_view text, bool& integral) noexcept
251{
252 std::size_t i = 0;
253 const std::size_t n = text.size();
254 const auto digits = [&] {
255 const std::size_t start = i;
256 while (i < n && text[i] >= '0' && text[i] <= '9') {
257 ++i;
258 }
259 return i - start;
260 };
261 if (i < n && text[i] == '-') {
262 ++i;
263 }
264 if (i < n && text[i] == '0') {
265 ++i;
266 } else if (digits() == 0) {
267 return false;
268 }
269 integral = true;
270 if (i < n && text[i] == '.') {
271 ++i;
272 integral = false;
273 if (digits() == 0) {
274 return false;
275 }
276 }
277 if (i < n && (text[i] == 'e' || text[i] == 'E')) {
278 ++i;
279 integral = false;
280 if (i < n && (text[i] == '+' || text[i] == '-')) {
281 ++i;
282 }
283 if (digits() == 0) {
284 return false;
285 }
286 }
287 return i == n;
288}

297bool parse_double(std::string_view text, double& value) noexcept
298{
299#if defined(_LIBCPP_VERSION)
300 // Floating-point std::from_chars needs macOS 26 with libc++, so the C library parses in the "C" locale.
301 char buffer[128];
302 if (text.size() >= sizeof(buffer)) {
303 return false; // no double needs that many digits; reject rather than allocate
304 }
305 text.copy(buffer, text.size());
306 buffer[text.size()] = '\0';
307#ifdef _WIN32
308 static _locale_t c = _create_locale(LC_NUMERIC, "C");
309 value = _strtod_l(buffer, nullptr, c);
310#else
311 static locale_t c = newlocale(LC_NUMERIC_MASK, "C", nullptr);
312 value = strtod_l(buffer, nullptr, c);
313#endif
314#else
315 const auto result = std::from_chars(text.data(), text.data() + text.size(), value);
316 if (result.ec == std::errc::result_out_of_range) {
317 return false;
318 }
319#endif
320 if (std::isinf(value)) {
321 return false;
322 }
323 if (value == 0.0) {
324 const std::string_view mantissa = text.substr(0, text.find_first_of("eE"));
325 return mantissa.find_first_of("123456789") == std::string_view::npos;
326 }
327 return true;
328}

382bool JsonStreamParser::feed(std::string_view chunk)
383{
384 if (m_state == State::Error) {
385 return false;
386 }
387 const char* const begin = chunk.data();
388 const char* const end = begin + chunk.size();
389 const char* p = begin;
390 const auto offsetOf = [&](const char* at) {
391 return m_offset + static_cast<std::size_t>(at - begin);
392 };
393
394 // A string or number that began in an earlier chunk continues at the start of this one.
395 m_tokenStart = begin;
396
397 while (p < end) {
398 switch (m_state) {
399 case State::Root:
400 case State::Done:
401 p = skip_whitespace(p, end);
402 if (p == end) {
403 break;
404 }
405 if (m_state == State::Done) {
406 return fail(JsonParseError::GarbageAtEnd, offsetOf(p));
407 }
408 if (*p != '{' && *p != '[') {
409 return fail(JsonParseError::IllegalValue, offsetOf(p));
410 }
411 m_documentStart = offsetOf(p);
412 if (!open_container(*p == '{', offsetOf(p))) {
413 return false;
414 }
415 ++p;
416 break;
417
418 case State::Value:
419 case State::ValueOrEndArray:
420 p = skip_whitespace(p, end);
421 if (p == end) {
422 break;
423 }
424 switch (*p) {
425 case '{':
426 case '[':
427 if (!open_container(*p == '{', offsetOf(p))) {
428 return false;
429 }
430 ++p;
431 break;
432 case '"':
433 m_inKey = false;
434 m_tokenBuffered = false;
435 m_state = State::String;
436 m_tokenStart = ++p;
437 break;
438 case 't':
439 case 'f':
440 case 'n':
441 m_literal = *p == 't' ? "true" : *p == 'f' ? "false"
442 : "null";
443 m_literalPos = 0;
444 m_state = State::Literal;
445 break;
446 case ']':
447 if (m_state == State::ValueOrEndArray) {
448 close_container();
449 ++p;
450 break;
451 }
452 return fail(JsonParseError::MissingObject, offsetOf(p));
453 case '}':
454 return fail(JsonParseError::MissingObject, offsetOf(p));
455 default:
456 if (*p != '-' && (*p < '0' || *p > '9')) {
457 return fail(JsonParseError::IllegalValue, offsetOf(p));
458 }
459 m_tokenBuffered = false;
460 m_tokenStart = p;
461 m_state = State::Number;
462 break;
463 }
464 break;
465
466 case State::KeyOrEndObject:
467 case State::Key:
468 p = skip_whitespace(p, end);
469 if (p == end) {
470 break;
471 }
472 if (*p == '"') {
473 m_inKey = true;
474 m_tokenBuffered = false;
475 m_state = State::String;
476 m_tokenStart = ++p;
477 break;
478 }
479 if (*p == '}' && m_state == State::KeyOrEndObject) {
480 close_container();
481 ++p;
482 break;
483 }
484 return fail(m_state == State::Key ? JsonParseError::MissingObject : JsonParseError::UnterminatedObject, offsetOf(p));
485
486 case State::Colon:
487 p = skip_whitespace(p, end);
488 if (p == end) {
489 break;
490 }
491 if (*p != ':') {
492 return fail(JsonParseError::MissingNameSeparator, offsetOf(p));
493 }
494 m_state = State::Value;
495 ++p;
496 break;
497
498 case State::CommaOrEnd: {
499 p = skip_whitespace(p, end);
500 if (p == end) {
501 break;
502 }
503 const bool inObject = m_containers.back();
504 if (*p == ',') {
505 m_state = inObject ? State::Key : State::Value;
506 ++p;
507 } else if (*p == (inObject ? '}' : ']')) {
508 close_container();
509 ++p;
510 } else {
511 return fail(inObject ? JsonParseError::UnterminatedObject : JsonParseError::MissingValueSeparator, offsetOf(p));
512 }
513 break;
514 }
515
516 case State::String: {
517 const char* q = p;
518 for (; q < end; ++q) {
519 const unsigned char c = static_cast<unsigned char>(*q);
520 if (c < 0x80) {
521 if (m_utf8Pending != 0) {
522 return fail(JsonParseError::IllegalUTF8String, offsetOf(q));
523 }
524 if (c == '"' || c == '\\') {
525 break;
526 }
527 } else if (!accept_utf8(c)) {
528 return fail(JsonParseError::IllegalUTF8String, offsetOf(q));
529 }
530 }
531 if (q == end) {
532 p = end; // the rest of the string comes with the next chunk
533 break;
534 }
535 if (*q == '\\') {
536 if (!append_token(m_tokenStart, q)) {
537 return fail(JsonParseError::DocumentTooLarge, offsetOf(q));
538 }
539 m_state = State::Escape;
540 p = q + 1;
541 break;
542 }
543 std::string_view text;
544 if (!take_token(q, text)) {
545 return fail(JsonParseError::DocumentTooLarge, offsetOf(q));
546 }
547 p = q + 1;
548 if (m_inKey) {
549 m_handler->on_key(text);
550 m_state = State::Colon;
551 } else {
552 m_handler->on_string(text);
553 end_value();
554 }
555 m_token.clear();
556 m_tokenBuffered = false;
557 break;
558 }
559
560 case State::Escape: {
561 char decoded = 0;
562 switch (*p) {
563 case '"':
564 decoded = '"';
565 break;
566 case '\\':
567 decoded = '\\';
568 break;
569 case '/':
570 decoded = '/';
571 break;
572 case 'b':
573 decoded = '\b';
574 break;
575 case 'f':
576 decoded = '\f';
577 break;
578 case 'n':
579 decoded = '\n';
580 break;
581 case 'r':
582 decoded = '\r';
583 break;
584 case 't':
585 decoded = '\t';
586 break;
587 case 'u':
588 m_codeUnit = 0;
589 m_hexDigits = 0;
590 m_state = State::Unicode;
591 ++p;
592 continue;
593 default:
594 return fail(JsonParseError::IllegalEscapeSequence, offsetOf(p));
595 }
596 if (!append_token(&decoded, &decoded + 1)) {
597 return fail(JsonParseError::DocumentTooLarge, offsetOf(p));
598 }
599 m_state = State::String;
600 m_tokenStart = ++p;
601 break;
602 }
603
604 case State::Unicode: {
605 const int digit = hex_value(*p);
606 if (digit < 0) {
607 return fail(JsonParseError::IllegalEscapeSequence, offsetOf(p));
608 }
609 m_codeUnit = m_codeUnit * 16 + static_cast<std::uint32_t>(digit);
610 ++p;
611 if (++m_hexDigits == 4) {
612 if (!decode_code_unit()) {
613 return fail(JsonParseError::DocumentTooLarge, offsetOf(p));
614 }
615 m_tokenStart = p;
616 }
617 break;
618 }
619
620 case State::SurrogateBackslash:
621 if (*p == '\\') {
622 m_state = State::SurrogateU;
623 ++p;
624 break;
625 }
626 // A high surrogate without its low half becomes U+FFFD so that the string stays valid UTF-8.
627 m_highSurrogate = 0;
628 if (!append_code_point(0xFFFD)) {
629 return fail(JsonParseError::DocumentTooLarge, offsetOf(p));
630 }
631 m_state = State::String;
632 m_tokenStart = p;
633 break;
634
635 case State::SurrogateU:
636 if (*p == 'u') {
637 m_codeUnit = 0;
638 m_hexDigits = 0;
639 m_state = State::Unicode;
640 ++p;
641 break;
642 }
643 m_highSurrogate = 0;
644 if (!append_code_point(0xFFFD)) {
645 return fail(JsonParseError::DocumentTooLarge, offsetOf(p));
646 }
647 m_state = State::Escape; // the backslash already read starts another escape
648 break;
649
650 case State::Number: {
651 const char* q = p;
652 while (q < end && is_number_char(*q)) {
653 ++q;
654 }
655 if (q == end) {
656 p = end; // the number may go on in the next chunk
657 break;
658 }
659 std::string_view text;
660 if (!take_token(q, text)) {
661 return fail(JsonParseError::DocumentTooLarge, offsetOf(q));
662 }
663 if (!emit_number(text)) {
664 return fail(JsonParseError::IllegalNumber, offsetOf(q));
665 }
666 m_token.clear();
667 m_tokenBuffered = false;
668 p = q;
669 end_value();
670 break;
671 }
672
673 case State::Literal:
674 while (p < end && m_literalPos < m_literal.size()) {
675 if (*p != m_literal[m_literalPos]) {
676 return fail(JsonParseError::IllegalValue, offsetOf(p));
677 }
678 ++p;
679 ++m_literalPos;
680 }
681 if (m_literalPos == m_literal.size()) {
682 if (m_literal == "null") {
683 m_handler->on_null();
684 } else {
685 m_handler->on_bool(m_literal == "true");
686 }
687 end_value();
688 }
689 break;
690
691 case State::Error:
692 return false;
693 }
694 }
695
696 if (m_state == State::String || m_state == State::Number) {
697 if (!append_token(m_tokenStart, end)) {
698 return fail(JsonParseError::DocumentTooLarge, offsetOf(end));
699 }
700 m_tokenBuffered = true;
701 }
702 m_offset += chunk.size();
703 if (!is_idle() && m_offset - m_documentStart > m_options.maxDocumentSize) {
704 return fail(JsonParseError::DocumentTooLarge, m_offset);
705 }
706 return true;
707}

711bool JsonStreamParser::finish()
712{
713 switch (m_state) {
714 case State::Error:
715 return false;
716 case State::Root:
717 case State::Done:
718 return true;
719 case State::String:
720 case State::Escape:
721 case State::Unicode:
722 case State::SurrogateBackslash:
723 case State::SurrogateU:
724 return fail(JsonParseError::UnterminatedString, m_offset);
725 case State::Colon:
726 return fail(JsonParseError::MissingNameSeparator, m_offset);
727 case State::Literal:
728 return fail(JsonParseError::IllegalValue, m_offset);
729 default:
730 return fail(m_containers.back() ? JsonParseError::UnterminatedObject : JsonParseError::UnterminatedArray, m_offset);
731 }

834{
835 if (m_containers.size() >= m_options.maxDepth) {
836 return fail(JsonParseError::DeepNesting, offset);
837 }
838 if (!m_containers.empty() && offset - m_documentStart > m_options.maxDocumentSize) {
839 return fail(JsonParseError::DocumentTooLarge, offset);
840 }
841 m_containers.push_back(object);
842 if (object) {
843 m_handler->on_start_object();
844 m_state = State::KeyOrEndObject;
845 } else {
846 m_handler->on_start_array();
847 m_state = State::ValueOrEndArray;

913bool JsonStreamParser::emit_number(std::string_view text)
914{
915 bool integral = false;
916 if (!is_json_number(text, integral)) {
917 return false;
918 }
919 if (integral) {

938bool JsonStreamParser::append_code_point(std::uint32_t codePoint)
939{
940 char utf8[4];
941 std::size_t size = 0;
942 if (codePoint < 0x80) {
943 utf8[size++] = static_cast<char>(codePoint);
944 } else if (codePoint < 0x800) {
945 utf8[size++] = static_cast<char>(0xC0 | (codePoint >> 6));
946 utf8[size++] = static_cast<char>(0x80 | (codePoint & 0x3F));
947 } else if (codePoint < 0x10000) {
948 utf8[size++] = static_cast<char>(0xE0 | (codePoint >> 12));
949 utf8[size++] = static_cast<char>(0x80 | ((codePoint >> 6) & 0x3F));
950 utf8[size++] = static_cast<char>(0x80 | (codePoint & 0x3F));
951 } else {
952 utf8[size++] = static_cast<char>(0xF0 | (codePoint >> 18));
953 utf8[size++] = static_cast<char>(0x80 | ((codePoint >> 12) & 0x3F));
954 utf8[size++] = static_cast<char>(0x80 | ((codePoint >> 6) & 0x3F));
955 utf8[size++] = static_cast<char>(0x80 | (codePoint & 0x3F));
956 }
957 return append_token(utf8, utf8 + size);
958}

962bool JsonStreamParser::decode_code_unit()
963{
964 std::uint32_t unit = m_codeUnit;
965 if (m_highSurrogate != 0) {
966 const std::uint32_t high = std::exchange(m_highSurrogate, 0);
967 if (unit >= 0xDC00 && unit <= 0xDFFF) {
968 m_state = State::String;
969 return append_code_point(0x10000 + ((high - 0xD800) << 10) + (unit - 0xDC00));
970 }
971 if (!append_code_point(0xFFFD)) {
972 return false;
973 }
974 }
975 if (unit >= 0xD800 && unit <= 0xDBFF) {
976 m_highSurrogate = unit;
977 m_state = State::SurrogateBackslash;
978 return true;
979 }
980 if (unit >= 0xDC00 && unit <= 0xDFFF) {
981 unit = 0xFFFD; // a low surrogate on its own
982 }
983 m_state = State::String;
984 return append_code_point(unit);
985}

989bool JsonStreamParser::accept_utf8(unsigned char byte) noexcept
990{
991 if (m_utf8Pending != 0) {
992 if (byte < m_utf8Lower || byte > m_utf8Upper) {
993 return false;
994 }
995 --m_utf8Pending;
996 m_utf8Lower = 0x80;
997 m_utf8Upper = 0xBF;
998 return true;
999 }
1000 // Lead bytes per RFC 3629: no overlong forms, no surrogates, nothing beyond U+10FFFF.
1001 m_utf8Lower = 0x80;
1002 m_utf8Upper = 0xBF;
1003 if (byte >= 0xC2 && byte <= 0xDF) {
1004 m_utf8Pending = 1;
1005 } else if (byte == 0xE0) {
1006 m_utf8Pending = 2;
1007 m_utf8Lower = 0xA0;
1008 } else if (byte == 0xED) {
1009 m_utf8Pending = 2;
1010 m_utf8Upper = 0x9F;
1011 } else if (byte >= 0xE1 && byte <= 0xEF) {
1012 m_utf8Pending = 2;
1013 } else if (byte == 0xF0) {
1014 m_utf8Pending = 3;
1015 m_utf8Lower = 0x90;
1016 } else if (byte >= 0xF1 && byte <= 0xF3) {
1017 m_utf8Pending = 3;
1018 } else if (byte == 0xF4) {
1019 m_utf8Pending = 3;
1020 m_utf8Upper = 0x8F;
1021 } else {
1022 return false;
1023 }
1024 return true;
