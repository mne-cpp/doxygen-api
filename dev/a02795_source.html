109BrainSurfaceUniforms surface_uniforms(const PaneFrame& frame, const BrainSurface& surface, float overlayMode)
110{
111 BrainSurfaceUniforms uniforms;
112 copy_matrix(frame.mvp, uniforms.mvp);
113 copy_vector(frame.eye, uniforms.camera_pos);
114 copy_vector(frame.light, uniforms.light_dir);
115 // A highlighted region or vertex range replaces the whole-surface glow
116 const std::optional<std::uint32_t> label = surface.selectedLabel();
117 uniforms.is_selected = (surface.isSelected() && !label) ? 1.f : 0.f;
118 uniforms.tissue_type = static_cast<float>(surface.tissueType());
119 uniforms.lighting_enabled = frame.lighting ? 1.f : 0.f;

122 const std::uint32_t key = label.value_or(0u);
123 for (int c = 0; c < 4; ++c) {
124 uniforms.selected_label[c] = static_cast<float>((key >> (8 * c)) & 0xFFu) / 255.f;

139// The video disk faces the camera; its long edge is perpendicular to the up hint (the optical axis) when known
140std::array<float, 20> video_disk_quad(const BrainVideoLayer& video, const Eigen::Vector3f& eye)
141{
142 const float aspect = aspect_of(video.frame);
143 const float halfWidth = 0.5f * video.size;
144 const float halfHeight = aspect > 0.f ? halfWidth / aspect : halfWidth;
145
146 Eigen::Vector3f viewDir = video.focus - eye;
147 viewDir = viewDir.squaredNorm() < 1e-12f ? Eigen::Vector3f(0.f, 0.f, -1.f) : viewDir.normalized();
148 Eigen::Vector3f up;
149 if (video.up_hint.squaredNorm() > 1e-8f) {
150 up = video.up_hint - video.up_hint.dot(viewDir) * viewDir;
151 up = up.squaredNorm() < 1e-8f ? Eigen::Vector3f(0.f, 0.f, 1.f) : up.normalized();
152 } else {
153 Eigen::Vector3f worldUp(0.f, 0.f, 1.f);
154 if (std::abs(viewDir.dot(worldUp)) > 0.95f) {
155 worldUp = Eigen::Vector3f(0.f, 1.f, 0.f);
156 }
157 up = viewDir.cross(worldUp).cross(viewDir).normalized();
158 }
159 const Eigen::Vector3f right = viewDir.cross(up).normalized();
160
161 // Image rows run downwards, texture v upwards
162 const std::array<Eigen::Vector3f, 4> corners = {video.focus - right * halfWidth - up * halfHeight, video.focus + right * halfWidth - up * halfHeight,
163 video.focus - right * halfWidth + up * halfHeight, video.focus + right * halfWidth + up * halfHeight};
164 const std::array<float, 8> uv = {0.f, 1.f, 1.f, 1.f, 0.f, 0.f, 1.f, 0.f};
165 std::array<float, 20> quad{};
166 for (std::size_t c = 0; c < corners.size(); ++c) {
167 copy_vector(corners[c], quad.data() + 5 * c);
168 quad[5 * c + 3] = uv[2 * c];
169 quad[5 * c + 4] = uv[2 * c + 1];
170 }
171 return quad;
172}

174// The decal frame is tangent to the head at the focus; the up hint keeps it locked to the microscope
175BrainVideoDecalUniforms video_decal_uniforms(const BrainVideoLayer& video, const PaneFrame& frame, bool relief)
176{
177 Eigen::Vector3f normal = video.focus;
178 normal = normal.squaredNorm() < 1e-12f ? Eigen::Vector3f(0.f, 0.f, 1.f) : normal.normalized();
179 Eigen::Vector3f up = video.up_hint;
180 if (up.squaredNorm() < 1e-6f) {
181 up = std::abs(normal.z()) > 0.95f ? Eigen::Vector3f(1.f, 0.f, 0.f) : Eigen::Vector3f(0.f, 0.f, 1.f);
182 }
183 up -= up.dot(normal) * normal;
184 const Eigen::Vector3f axisV = up.squaredNorm() < 1e-12f ? Eigen::Vector3f(0.f, 0.f, 1.f) : up.normalized();
185 const Eigen::Vector3f axisU = axisV.cross(normal).normalized();
186
187 BrainVideoDecalUniforms uniforms;
188 copy_matrix(frame.mvp, uniforms.mvp);
189 copy_vector(video.focus, uniforms.focus_and_size);
190 uniforms.focus_and_size[3] = video.size;
191 copy_vector(axisU, uniforms.axis_u_and_opacity);
192 uniforms.axis_u_and_opacity[3] = video.opacity;
193 copy_vector(axisV, uniforms.axis_v_and_offset);
194 uniforms.axis_v_and_offset[3] = kVideoDecalOffset;
195 copy_vector(normal, uniforms.axis_n_and_depth);
196 uniforms.axis_n_and_depth[3] = std::max(0.015f, video.size * 0.45f);
197 copy_vector(frame.eye, uniforms.camera_pos_and_aspect);
198 uniforms.camera_pos_and_aspect[3] = aspect_of(video.frame);
199 uniforms.depth_params[0] = relief ? video.depth_scale : 0.f;
200 uniforms.depth_params[1] = relief ? static_cast<float>(video.depth_steps) : 0.f;
201 uniforms.depth_params[2] = relief ? 1.f : 0.f;
202 return uniforms;
203}

225void BrainFrameComposer::begin(const GpuFrameTarget& target, Color clear)
226{
227 // Renderables destroyed since the last frame release their buffers
228 for (auto it = m_keys.begin(); it != m_keys.end();) {
229 if (it->second.owner.expired()) {
230 m_builder.release(it->second.id);
231 it = m_keys.erase(it);
232 } else {
233 ++it;
234 }
235 }
236 m_clipCorrection = target.clip_correction;
237 m_builder.begin(target, clear);
238}

292void BrainFrameComposer::draw_pane(const BrainScene& scene, const BrainScenePane& pane, std::size_t index, std::vector<std::string>& drawn)
293{
294 const SubView& view = *pane.view;
295 const GpuRect& viewport = pane.viewport;
296 PaneFrame frame;
297 const Eigen::Matrix4f pvm = pane.camera.pvm();
298 frame.mvp = m_clipCorrection * pvm;
299 frame.eye = pane.camera.eye;
300 frame.light = pane.camera.eye.normalized();
301 frame.lighting = scene.lighting;
302
303 static const std::map<std::string, std::shared_ptr<BrainSurface>> noSurfaces;
304 const auto& surfaces = scene.surfaces ? *scene.surfaces : noSurfaces;
305
306 const auto drawSurface = [this, &frame, &viewport](const BrainSurface& surface, ShaderMode shader, float overlayMode) {
307 if (surface.isVisible()) {
308 m_builder.draw_surface(key_of(surface.renderKey()), surface_streams(surface), static_cast<BrainShader>(shader),
309 surface_uniforms(frame, surface, overlayMode), viewport);
310 }
311 };
312 const auto drawInstanced = [this, &frame, &viewport](const RenderKey& key, const InstancedStreams& streams, BrainShader shader) {
313 m_builder.draw_instanced(key_of(key), instanced_streams(streams), shader, instance_uniforms(frame), viewport);
314 };
315
316 // Brain surfaces of the pane's surface type, with the pane's overlay
317 const float paneOverlay = static_cast<float>(view.overlayMode);
318 for (const auto& [key, surface] : surfaces) {
319 if (surface && view.matchesSurfaceType(key) && view.shouldRenderSurface(key)) {
320 drawn.push_back(key);
321 drawSurface(*surface, view.brainShader, paneOverlay);
322 }
323 }

325 // The other surfaces: source spaces and digitizers are opaque except for the tracked markers; BEM and sensors
326 // are translucent
327 struct Translucent
328 {
329 const BrainSurface* surface;
330 ShaderMode shader;
331 float overlayMode;
332 float distance;
333 };
334 std::vector<Translucent> translucent;
335 const auto distanceTo = [&frame](const BrainSurface& surface) {
336 Eigen::Vector3f low;
337 Eigen::Vector3f high;
338 surface.boundingBox(low, high);
339 return (frame.eye - (low + high) * 0.5f).squaredNorm();
340 };
341 const float scientific = static_cast<float>(ModeScientific);
342 for (const auto& [key, surface] : surfaces) {
343 if (!surface || SubView::isBrainSurfaceKey(key) || !view.shouldRenderSurface(key) || !surface->isVisible()) {
344 continue;
345 }
346 if (key.starts_with("srcsp_") || key.starts_with("dig_")) {
347 if (key.starts_with("dig_live_t_") || key.starts_with("dig_ray_") || key.starts_with("dig_probe_")) {
348 translucent.push_back({surface.get(), Holographic, scientific, distanceTo(*surface)});
349 } else {
350 drawSurface(*surface, view.brainShader, scientific);
351 }
352 } else if (key.starts_with("sens_") || key.starts_with("bem_")) {
353 // A field map surface shows plain colours while its field map is hidden
354 float overlay = scientific;
355 if ((key == scene.meg_field_key && !view.visibility.megFieldMap) || (key == scene.eeg_field_key && !view.visibility.eegFieldMap)) {
356 overlay = static_cast<float>(ModeSurface);
357 }
358 translucent.push_back({surface.get(), key.starts_with("bem_") ? view.bemShader : Holographic, overlay, distanceTo(*surface)});
359 }
360 }
361
362 if (scene.electrodes && scene.electrodes->isVisible()) {
363 drawInstanced(scene.electrodes->bodyRenderKey(), scene.electrodes->bodyStreams(), BrainShader::Electrode);
364 drawInstanced(scene.electrodes->contactRenderKey(), scene.electrodes->contactStreams(), BrainShader::Electrode);
365 }
366
367 // Slices go after the opaque surfaces and before the translucent ones, which add their glow on top
368 if (view.visibility.mriSlices) {
369 for (std::size_t s = 0; s < scene.slices.size(); ++s) {
370 const BrainSliceLayer& slice = scene.slices[s];
371 BrainSliceUniforms uniforms;
372 copy_matrix(frame.mvp, uniforms.mvp);
373 uniforms.opacity = slice.opacity;
374 uniforms.window_center = slice.window_center;
375 uniforms.window_width = slice.window_width;
376 m_builder.draw_slice(kSliceKey + s, slice.image, slice.quad, uniforms, viewport);
377 }
378 }
379
380 // Translucent surfaces back to front; the video decal right after the head it lies on
381 std::stable_sort(translucent.begin(), translucent.end(), [](const Translucent& a, const Translucent& b) { return a.distance > b.distance; });
382 const BrainSurface* videoTarget = scene.video ? video_target(surfaces) : nullptr;
383 bool videoDrawn = !scene.video;
384 const auto drawVideo = [&]() {
385 const BrainVideoLayer& video = *scene.video;
386 if (videoTarget) {
387 const bool relief = video.depth.image != nullptr;
388 m_builder.draw_video_decal(key_of(videoTarget->renderKey()), surface_streams(*videoTarget), kVideoTextureKey, video.frame,
389 relief ? video.depth : BrainTextureView{}, video_decal_uniforms(video, frame, relief), viewport);
390 } else {
391 BrainVideoBillboardUniforms uniforms;
392 copy_matrix(frame.mvp, uniforms.mvp);
393 uniforms.opacity = video.opacity;
394 m_builder.draw_video_billboard(kVideoDiskKey + index, kVideoTextureKey, video.frame, video_disk_quad(video, frame.eye), uniforms, viewport);
395 }
396 videoDrawn = true;
397 };
398 for (const Translucent& item : translucent) {
399 drawSurface(*item.surface, item.shader, item.overlayMode);
400 if (!videoDrawn && item.surface == videoTarget) {
401 drawVideo();
402 }
403 }
404 if (!videoDrawn) {
405 drawVideo();
406 }
407
408 if (view.visibility.dipoles) {
409 for (const DipoleObject* dipoles : scene.dipoles) {
410 if (dipoles && dipoles->isVisible()) {
411 drawInstanced(dipoles->renderKey(), dipoles->streams(), BrainShader::Dipole);
412 }
413 }
414 }
415 if (view.visibility.network && scene.network && scene.network->isVisible() && scene.network->hasData()) {
416 drawInstanced(scene.network->nodeRenderKey(), scene.network->nodeStreams(), BrainShader::Dipole);
417 drawInstanced(scene.network->edgeRenderKey(), scene.network->edgeStreams(), BrainShader::Dipole);
418 }
419 // The head movement path is not connectivity data, so it does not follow the network visibility
420 if (scene.head_path && scene.head_path->isVisible() && scene.head_path->hasData()) {
421 drawInstanced(scene.head_path->renderKey(), scene.head_path->streams(), BrainShader::Dipole);
422 }
423
424 if (scene.pointer && scene.pointer->isVisible()) {
425 PaneFrame pointerFrame = frame;
426 pointerFrame.mvp = m_clipCorrection * pvm * translation_matrix(scene.pointer_position);
427 m_builder.draw_surface(key_of(scene.pointer->renderKey()), surface_streams(*scene.pointer), BrainShader::Holographic,
428 surface_uniforms(pointerFrame, *scene.pointer, 0.f), viewport);
429 }
