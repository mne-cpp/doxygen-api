192 template<Swappable T>
193 bool write_big_endian(const T* values, std::size_t count)
194 {
195 if constexpr (std::endian::native == std::endian::big) {
196 return write(values, count * sizeof(T)) == count * sizeof(T);
197 } else {
198 constexpr std::size_t kChunk = 4096;
199 T swapped[kChunk];
200 for (std::size_t done = 0; done < count;) {
201 const std::size_t n = count - done < kChunk ? count - done : kChunk;
202 for (std::size_t i = 0; i < n; ++i) {
203 swapped[i] = byte_swap(values[done + i]);
204 }
205 if (write(swapped, n * sizeof(T)) != n * sizeof(T)) {
206 return false;
207 }
208 done += n;
209 }
210 return true;
211 }
212 }
