83bool MNEInverseOperator::assemble_kernel(const FsLabel& label,
84 const std::string& method,
85 bool pick_normal,
86 MatrixXd& K,
87 SparseMatrix<double>& noise_norm,
88 std::vector<VectorXi>& vertno)
89{
90 MatrixXd t_eigen_leads = eigen_leads.data;
91 MatrixXd t_source_cov = source_cov.data;
92 if (method != "MNE")
93 noise_norm = noisenorm;
94
95 vertno = src.get_vertno();
96
97 typedef Eigen::Triplet<double> T;
98 std::vector<T> tripletList;
99
100 if (!label.empty()) {
101 VectorXi src_sel;
102 vertno = src.label_src_vertno_sel(label, src_sel);
103
104 if (method != "MNE" && noise_norm.rows() > 0) {
105 // noisenorm is diagonal, one entry per source.
106 tripletList.clear();
107 tripletList.reserve(src_sel.size());
108 for (std::int32_t i = 0; i < src_sel.size(); ++i)
109 tripletList.push_back(T(i, i, noise_norm.coeff(src_sel[i], src_sel[i])));
110
111 noise_norm = SparseMatrix<double>(src_sel.size(), src_sel.size());
112 noise_norm.setFromTriplets(tripletList.begin(), tripletList.end());
113 }
114
115 if (source_ori == FIFFV_MNE_FREE_ORI) {
116 VectorXi src_sel_new(src_sel.size() * 3);
117
118 for (std::int32_t i = 0; i < src_sel.size(); ++i) {
119 src_sel_new[i * 3] = src_sel[i] * 3;
120 src_sel_new[i * 3 + 1] = src_sel[i] * 3 + 1;
121 src_sel_new[i * 3 + 2] = src_sel[i] * 3 + 2;
122 }
123 src_sel = src_sel_new;
124 }
125
126 // src_sel is ascending with src_sel[i] >= i, so in-place compaction is safe.
127 for (std::int32_t i = 0; i < src_sel.size(); ++i) {
128 t_eigen_leads.row(i) = t_eigen_leads.row(src_sel[i]);
129 t_source_cov.row(i) = t_source_cov.row(src_sel[i]);
130 }
131 t_eigen_leads.conservativeResize(src_sel.size(), t_eigen_leads.cols());
132 t_source_cov.conservativeResize(src_sel.size(), t_source_cov.cols());
133 }
134
135 if (pick_normal) {
136 if (source_ori != FIFFV_MNE_FREE_ORI) {
137 Log::warning("Warning: Pick normal can only be used with a free orientation inverse operator.\n");
138 return false;
139 }
140
141 bool is_loose = (0 < orient_prior.data(0, 0)) && (orient_prior.data(0, 0) < 1);
142 if (!is_loose) {
143 Log::warning("The pick_normal parameter is only valid when working with loose orientations.\n");
144 return false;
145 }
146
147 // keep only the normal components
148 std::int32_t count = 0;
149 for (std::int32_t i = 2; i < t_eigen_leads.rows(); i += 3) {
150 t_eigen_leads.row(count) = t_eigen_leads.row(i);
151 ++count;
152 }
153 t_eigen_leads.conservativeResize(count, t_eigen_leads.cols());
154
155 count = 0;
156 for (std::int32_t i = 2; i < t_source_cov.rows(); i += 3) {
157 t_source_cov.row(count) = t_source_cov.row(i);
158 ++count;
159 }
160 t_source_cov.conservativeResize(count, t_source_cov.cols());
161 }
162
163 tripletList.clear();
164 tripletList.reserve(reginv.rows());
165 for (std::int32_t i = 0; i < reginv.rows(); ++i)
166 tripletList.push_back(T(i, i, reginv(i, 0)));
167 SparseMatrix<double> t_reginv(reginv.rows(), reginv.rows());
168 t_reginv.setFromTriplets(tripletList.begin(), tripletList.end());
169
170 MatrixXd trans = t_reginv * eigen_fields.data * whitener * proj;
171 //
172 // Transformation into current distributions by weighting the eigenleads
173 // with the weights computed above
174 //
175 if (eigen_leads_weighted) {
176 //
177 // R^0.5 has been already factored in
178 //
179 Log::info("(eigenleads already weighted)...\n");
180 K = t_eigen_leads * trans;
181 } else {
182 //
183 // R^0.5 has to factored in
184 //
185 Log::info("(eigenleads need to be weighted)...\n");
186
187 std::vector<T> tripletList2;
188 tripletList2.reserve(t_source_cov.rows());
189 for (std::int32_t i = 0; i < t_source_cov.rows(); ++i)
190 tripletList2.push_back(T(i, i, sqrt(t_source_cov(i, 0))));
191 SparseMatrix<double> t_sourceCov(t_source_cov.rows(), t_source_cov.rows());
192 t_sourceCov.setFromTriplets(tripletList2.begin(), tripletList2.end());
193
194 K = t_sourceCov * t_eigen_leads * trans;
195 }
196
197 if (method == "MNE")
198 noise_norm = SparseMatrix<double>();
199
200 //store assembled kernel
201 m_K = K;
202
203 return true;
204}

208bool MNEInverseOperator::check_ch_names(const FiffInfo& measInfo) const
209{
210 StringList inv_ch_names = this->eigen_fields.col_names;
211
212 bool t_bContains = true;
213 if (this->eigen_fields.col_names.size() != this->noise_cov.names.size())
214 t_bContains = false;
215 else {
216 for (std::int32_t i = 0; i < static_cast<int>(this->noise_cov.names.size()); ++i) {
217 if (inv_ch_names[i] != this->noise_cov.names[i]) {
218 t_bContains = false;
219 break;
220 }
221 }
222 }
223
224 if (!t_bContains) {
225 Log::critical("Channels in inverse operator eigen fields do not match noise covariance channels.");
226 return false;
227 }
228
229 StringList data_ch_names = measInfo.ch_names;
230
231 StringList missing_ch_names;
232 for (std::int32_t i = 0; i < static_cast<int>(inv_ch_names.size()); ++i)
233 if (!mne::has_channel(data_ch_names, inv_ch_names[i]))
234 missing_ch_names.push_back(inv_ch_names[i]);
235
236 std::int32_t n_missing = static_cast<int>(missing_ch_names.size());
237
238 if (n_missing > 0) {
239 Log::critical("{} channels in inverse operator are not present in the data ( {} )", n_missing, join(missing_ch_names, " "));
240 return false;
241 }
242
243 return true;
244}

248MatrixXd MNEInverseOperator::cluster_kernel(const FsAnnotationSet& p_AnnotationSet, std::int32_t p_iClusterSize, MatrixXd& p_D, const std::string& p_sMethod) const
249{
250 Log::info("Cluster kernel using {}.\n", p_sMethod);
251
252 MatrixXd p_outMT = m_K.transpose();
253
254 std::vector<MNEClusterInfo> t_qListMNEClusterInfo;
255 MNEClusterInfo t_MNEClusterInfo;
256 t_qListMNEClusterInfo.push_back(t_MNEClusterInfo);
257 t_qListMNEClusterInfo.push_back(t_MNEClusterInfo);
258
259 //
260 // Check consistency
261 //
262 if (isFixedOrient()) {
263 Log::critical("Error: Fixed orientation not implemented yet!\n");
264 return p_outMT;
265 }
266
267 //
268 // Assemble input data
269 //
270 std::int32_t offset;
271
272 MatrixXd t_MT_new;
273
274 for (std::int32_t h = 0; h < this->src.size(); ++h) {
275 offset = 0;
276
277 if (h > 0)
278 for (std::int32_t j = 0; j < h; ++j)
279 offset += this->src[j].nuse;
280
281 if (h == 0)
282 Log::info("Cluster Left Hemisphere\n");
283 else
284 Log::info("Cluster Right Hemisphere\n");
285
286 FsColortable t_CurrentColorTable = p_AnnotationSet[h].getColortable();
287 VectorXi label_ids = t_CurrentColorTable.getLabelIds();
288
289 // Get label ids for every vertex
290 VectorXi vertno_labeled = VectorXi::Zero(this->src[h].vertno.rows());
291
292 for (std::int32_t i = 0; i < vertno_labeled.rows(); ++i)
293 vertno_labeled[i] = p_AnnotationSet[h].getLabelIds()[this->src[h].vertno[i]];
294
295 // Work items of the parallel clustering
296 std::vector<RegionMT> m_qListRegionMTIn;
297
298 //
299 // Generate cluster input data
300 //
301 for (std::int32_t i = 0; i < label_ids.rows(); ++i) {
302 if (label_ids[i] != 0) {
303 const std::string& curr_name = t_CurrentColorTable.struct_names[static_cast<std::size_t>(i)]; //obj.label2AtlasName(label(i));
304 Log::info("\tCluster {} / {} {}...", i + 1, label_ids.rows(), curr_name);
305
306 //
307 // Get source space indeces
308 //
309 VectorXi idcs = VectorXi::Zero(vertno_labeled.rows());
310 std::int32_t c = 0;
311
312 //Select ROIs //change this use label info with a hash tabel
313 for (std::int32_t j = 0; j < vertno_labeled.rows(); ++j) {
314 if (vertno_labeled[j] == label_ids[i]) {
315 idcs[c] = j;
316 ++c;
317 }
318 }
319 idcs.conservativeResize(c);
320
321 //get selected MT
322 MatrixXd t_MT(p_outMT.rows(), idcs.rows() * 3);
323
324 for (std::int32_t j = 0; j < idcs.rows(); ++j)
325 t_MT.block(0, j * 3, t_MT.rows(), 3) = p_outMT.block(0, (idcs[j] + offset) * 3, t_MT.rows(), 3);
326
327 std::int32_t nSens = t_MT.rows();
328 std::int32_t nSources = t_MT.cols() / 3;
329
330 if (nSources > 0) {
331 RegionMT t_sensMT;
332
333 t_sensMT.idcs = idcs;
334 t_sensMT.iLabelIdxIn = i;
335 t_sensMT.nClusters = ceil(static_cast<double>(nSources) / static_cast<double>(p_iClusterSize));
336
337 t_sensMT.matRoiMTOrig = t_MT;
338
339 Log::info("{} Cluster(s)... ", t_sensMT.nClusters);
340
341 // Reshape Input data -> sources rows; sensors columns
342 t_sensMT.matRoiMT = MatrixXd(t_MT.cols() / 3, 3 * nSens);
343
344 for (std::int32_t j = 0; j < nSens; ++j)
345 for (std::int32_t k = 0; k < t_sensMT.matRoiMT.rows(); ++k)
346 t_sensMT.matRoiMT.block(k, j * 3, 1, 3) = t_MT.block(j, k * 3, 1, 3);
347
348 t_sensMT.sDistMeasure = p_sMethod;
349
350 m_qListRegionMTIn.push_back(t_sensMT);
351
352 Log::info("[added]\n");
353 } else {
354 Log::info("failed! FsLabel contains no sources.\n");
355 }
356 }
357 }
358
359 //
360 // Calculate clusters
361 //
362 Log::info("Clustering... ");
363 const std::vector<RegionMTOut> res = mne::ThreadPool::global().parallel_map(m_qListRegionMTIn, [](const RegionMT& region) { return region.cluster(); });
364
365 //
366 // Assign results
367 //
368 MatrixXd t_MT_partial;
369
370 std::int32_t nClusters;
371 std::int32_t nSens;
372 std::vector<RegionMT>::const_iterator itIn;
373 itIn = m_qListRegionMTIn.begin();
374 for (auto itOut = res.cbegin(); itOut != res.cend(); ++itOut) {
375 nClusters = itOut->ctrs.rows();
376 nSens = itOut->ctrs.cols() / 3;
377 t_MT_partial = MatrixXd::Zero(nSens, nClusters * 3);
378
379 //
380 // Assign the centroid for each cluster to the partial G
381 //
382 for (std::int32_t j = 0; j < nSens; ++j)
383 for (std::int32_t k = 0; k < nClusters; ++k)
384 t_MT_partial.block(j, k * 3, 1, 3) = itOut->ctrs.block(k, j * 3, 1, 3);
385
386 //
387 // Get cluster indices and their distances to the centroid
388 //
389 for (std::int32_t j = 0; j < nClusters; ++j) {
390 VectorXi clusterIdcs = VectorXi::Zero(itOut->roiIdx.rows());
391 VectorXd clusterDistance = VectorXd::Zero(itOut->roiIdx.rows());
392 std::int32_t nClusterIdcs = 0;
393 for (std::int32_t k = 0; k < itOut->roiIdx.rows(); ++k) {
394 if (itOut->roiIdx[k] == j) {
395 clusterIdcs[nClusterIdcs] = itIn->idcs[k];
396 clusterDistance[nClusterIdcs] = itOut->D(k, j);
397 ++nClusterIdcs;
398 }
399 }
400 clusterIdcs.conservativeResize(nClusterIdcs);
401 clusterDistance.conservativeResize(nClusterIdcs);
402
403 VectorXi clusterVertnos = VectorXi::Zero(clusterIdcs.size());
404 for (std::int32_t k = 0; k < clusterVertnos.size(); ++k)
405 clusterVertnos(k) = this->src[h].vertno[clusterIdcs(k)];
406
407 t_qListMNEClusterInfo[h].clusterVertnos.push_back(clusterVertnos);
408 }
409
410 //
411 // Assign partial G to new LeadField
412 //
413 if (t_MT_partial.rows() > 0 && t_MT_partial.cols() > 0) {
414 t_MT_new.conservativeResize(t_MT_partial.rows(), t_MT_new.cols() + t_MT_partial.cols());
415 t_MT_new.block(0, t_MT_new.cols() - t_MT_partial.cols(), t_MT_new.rows(), t_MT_partial.cols()) = t_MT_partial;
416 }
417
418 ++itIn;
419 }
420
421 Log::info("[done]\n");
422 }
423
424 //
425 // Cluster operator D (sources x clusters)
426 //
427 std::int32_t totalNumOfClust = 0;
428 for (std::int32_t h = 0; h < 2; ++h)
429 totalNumOfClust += static_cast<std::int32_t>(t_qListMNEClusterInfo[h].clusterVertnos.size());
430
431 if (isFixedOrient())
432 p_D = MatrixXd::Zero(p_outMT.cols(), totalNumOfClust);
433 else
434 p_D = MatrixXd::Zero(p_outMT.cols(), totalNumOfClust * 3);
435
436 std::vector<VectorXi> t_vertnos = src.get_vertno();
437
438 std::int32_t currentCluster = 0;
439 for (std::int32_t h = 0; h < 2; ++h) {
440 int hemiOffset = h == 0 ? 0 : t_vertnos[0].size();
441 for (std::int32_t i = 0; i < std::ssize(t_qListMNEClusterInfo[h].clusterVertnos); ++i) {
442 VectorXi idx_sel;
443 Linalg::intersect(t_vertnos[h], t_qListMNEClusterInfo[h].clusterVertnos[i], idx_sel);
444
445 idx_sel.array() += hemiOffset;
446
447 double selectWeight = 1.0 / idx_sel.size();
448 if (isFixedOrient()) {
449 for (std::int32_t j = 0; j < idx_sel.size(); ++j)
450 p_D.col(currentCluster)[idx_sel(j)] = selectWeight;
451 } else {
452 std::int32_t clustOffset = currentCluster * 3;
453 for (std::int32_t j = 0; j < idx_sel.size(); ++j) {
454 std::int32_t idx_sel_Offset = idx_sel(j) * 3;
455 //x
456 p_D(idx_sel_Offset, clustOffset) = selectWeight;
457 //y
458 p_D(idx_sel_Offset + 1, clustOffset + 1) = selectWeight;
459 //z
460 p_D(idx_sel_Offset + 2, clustOffset + 2) = selectWeight;
461 }
462 }
463 ++currentCluster;
464 }
465 }
466
467 //
468 // Put it all together
469 //
470 p_outMT = t_MT_new;
471
472 return p_outMT;
473}

477MNEInverseOperator MNEInverseOperator::make_inverse_operator(const FiffInfo& info,
478 MNEForwardSolution forward,
479 const FiffCov& p_noise_cov,
480 float loose,
481 float depth,
482 bool fixed,
483 bool limit_depth_chs)
484{
485 bool is_fixed_ori = forward.isFixedOrient();
486 MNEInverseOperator inv;
487
488 // Loose and fixed constraints are relative to the cortical normal, as in mne-python's _prepare_forward.
489 if (!is_fixed_ori && (fixed || loose < 1.0f)) {
490 forward.convert_to_surf_ori();
491 }
492
493 //Check parameters
494 if (fixed && loose > 0) {
495 Log::warning("Warning: When invoking make_inverse_operator with fixed = true, the loose parameter is ignored.\n");
496 loose = 0.0f;
497 }
498
499 if (is_fixed_ori && !fixed) {
500 Log::warning("Warning: Setting fixed parameter = true. Because the given forward operator has fixed orientation and can only be used to make a fixed-orientation inverse operator.\n");
501 fixed = true;
502 }
503
504 if (forward.source_ori == -1 && loose > 0) {
505 Log::critical("Forward solution is not oriented in surface coordinates. loose parameter should be 0 not {:f}.", loose);
506 return inv;
507 }
508
509 if (loose < 0 || loose > 1) {
510 Log::warning("Warning: Loose value should be in interval [0,1] not {:f}.\n", loose);
511 loose = loose > 1 ? 1 : 0;
512 Log::info("Setting loose to {:f}.\n", loose);
513 }
514
515 if (depth < 0 || depth > 1) {
516 Log::warning("Warning: Depth value should be in interval [0,1] not {:f}.\n", depth);
517 depth = depth > 1 ? 1 : 0;
518 Log::info("Setting depth to {:f}.\n", depth);
519 }
520
521 //
522 // 1. Read the bad channels
523 // 2. Read the necessary data from the forward solution matrix file
524 // 3. Load the projection data
525 // 4. Load the sensor noise covariance matrix and attach it to the forward
526 //
527 FiffInfo gain_info;
528 MatrixXd gain;
529 MatrixXd whitener;
530 std::int32_t n_nzero;
531 FiffCov p_outNoiseCov;
532 forward.prepare_forward(info, p_noise_cov, false, gain_info, gain, p_outNoiseCov, whitener, n_nzero);
533
534 //
535 // 5. Compose the depth weight matrix
536 //
537 FiffCov p_depth_prior;
538 MatrixXd patch_areas;
539 if (depth > 0) {
540 // TODO: load patch_areas from forward solution
541 p_depth_prior = MNEForwardSolution::compute_depth_prior(gain, gain_info, is_fixed_ori, depth, 10.0, patch_areas, limit_depth_chs);
542 } else {
543 p_depth_prior.data = MatrixXd::Ones(gain.cols(), 1);
544 p_depth_prior.kind = FIFFV_MNE_DEPTH_PRIOR_COV;
545 p_depth_prior.diag = true;
546 p_depth_prior.dim = gain.cols();
547 p_depth_prior.nfree = 1;
548 }
549
550 // Deal with fixed orientation forward / inverse
551 if (fixed) {
552 if (depth < 0 || depth > 1) {
553 // TODO: convert free-orientation depth prior to fixed-orientation
554 Log::info("\tPicking elements from free-orientation depth prior into fixed-orientation one.\n");
555 }
556 if (!is_fixed_ori) {
557 if (!forward.surf_ori) {
558 Log::warning("Warning: For a fixed-orientation inverse, the forward solution must be surface-oriented. Skipping fixed conversion.\n");
559 } else {
560 // Convert to the fixed orientation forward solution now
561 std::int32_t count = 0;
562 for (std::int32_t i = 2; i < p_depth_prior.data.rows(); i += 3) {
563 p_depth_prior.data.row(count) = p_depth_prior.data.row(i);
564 gain.col(count) = gain.col(i);
565 ++count;
566 }
567 p_depth_prior.data.conservativeResize(count, 1);
568 p_depth_prior.dim = count;
569 gain.conservativeResize(Eigen::NoChange, count);
570
571 forward.to_fixed_ori();
572 is_fixed_ori = forward.isFixedOrient();
573 }
574 }
575 }
576 Log::info("\tComputing inverse operator with {} channels.\n", gain_info.ch_names.size());
577
578 //
579 // 6. Compose the source covariance matrix
580 //
581 Log::info("\tCreating the source covariance matrix\n");
582 FiffCov p_source_cov = p_depth_prior;
583
584 // apply loose orientations
585 FiffCov p_orient_prior;
586 if (!is_fixed_ori) {
587 p_orient_prior = forward.compute_orient_prior(loose);
588 p_source_cov.data.array() *= p_orient_prior.data.array();
589 }
590
591 // 7. Apply fMRI weighting (not done)
592
593 //
594 // 8. Apply the linear projection to the forward solution
595 // 9. Apply whitening to the forward computation matrix
596 //
597 Log::info("\tWhitening the forward solution.\n");
598 gain = Linalg::parallel_product(whitener, gain);
599
600 // 10. Exclude the source space points within the labels (not done)
601
602 //
603 // 11. Do appropriate source weighting to the forward computation matrix
604 //
605
606 // Adjusting Source Covariance matrix to make trace of G*R*G' equal
607 // to number of sensors.
608 Log::info("\tAdjusting source covariance matrix.\n");
609 RowVectorXd source_std = p_source_cov.data.array().sqrt().transpose();
610
611 gain.array().rowwise() *= source_std.array();
612
613 double trace_GRGT = gain.squaredNorm();
614 double scaling_source_cov = static_cast<double>(n_nzero) / trace_GRGT;
615
616 p_source_cov.data.array() *= scaling_source_cov;
617
618 gain.array() *= sqrt(scaling_source_cov);
619
620 //
621 // 12. Decompose the combined matrix
622 //
623 Log::info("Computing SVD of whitened and weighted lead field matrix.\n");
624 // Parallel tall-skinny QR followed by divide and conquer; singular values come in descending order
625 Linalg::ThinSvd svd = Linalg::thin_svd(gain);
626 VectorXd p_sing = std::move(svd.singularValues);
627 FiffNamedMatrix p_eigen_fields(svd.U.cols(),
628 svd.U.rows(),
629 {},
630 gain_info.ch_names,
631 svd.U.transpose());
632
633 FiffNamedMatrix p_eigen_leads(svd.V.rows(),
634 svd.V.cols(),
635 {},
636 {},
637 svd.V);
638 Log::info("\tlargest singular value = {:f}\n", p_sing.maxCoeff());
639 Log::info("\tscaling factor to adjust the trace = {:f}\n", trace_GRGT);
640
641 std::int32_t p_nave = 1.0;
642
643 // Handle methods
644 bool has_meg = false;
645 bool has_eeg = false;
646
647 RowVectorXd ch_idx(info.chs.size());
648 std::int32_t count = 0;
649 for (std::int32_t i = 0; i < std::ssize(info.chs); ++i) {
650 if (mne::has_channel(gain_info.ch_names, info.chs[i].ch_name)) {
651 ch_idx[count] = i;
652 ++count;
653 }
654 }
655 ch_idx.conservativeResize(count);
656
657 for (std::int32_t i = 0; i < ch_idx.size(); ++i) {
658 const std::string ch_type = info.channel_type(ch_idx[i]);
659 if (ch_type == "eeg")
660 has_eeg = true;
661 if ((ch_type == "mag") || (ch_type == "grad"))
662 has_meg = true;
663 }
664
665 std::int32_t p_iMethods;
666
667 if (has_eeg && has_meg)

706MNEInverseOperator MNEInverseOperator::prepare_inverse_operator(std::int32_t nAve, float lambda2, bool dSPM, bool sLORETA) const
707{
708 if (nAve <= 0) {
709 Log::critical("The number of averages should be positive\n");
710 return MNEInverseOperator();
711 }
712 Log::info("Preparing the inverse operator for use...\n");
713 MNEInverseOperator inv(*this);
714 //
715 // Scale some of the stuff
716 //
717 float scale = static_cast<float>(inv.nave) / static_cast<float>(nAve);
718 inv.noise_cov.data *= scale;
719 inv.noise_cov.eig *= scale;
720 inv.source_cov.data *= scale;
721 //
722 if (inv.eigen_leads_weighted)
723 inv.eigen_leads.data *= sqrt(scale);
724 //
725 Log::info("\tScaled noise and source covariance from nave = {} to nave = {}\n", inv.nave, nAve);
726 inv.nave = nAve;
727 //
728 // Create the diagonal matrix for computing the regularized inverse
729 //
730 VectorXd tmp = inv.sing.cwiseProduct(inv.sing) + VectorXd::Constant(inv.sing.size(), lambda2);
731 inv.reginv = VectorXd(inv.sing.cwiseQuotient(tmp));
732 Log::info("\tCreated the regularized inverter\n");
733 //
734 // Create the projection operator
735 //
736
737 std::int32_t ncomp = FiffProj::make_projector(inv.projs, inv.noise_cov.names, inv.proj);
738 if (ncomp > 0)
739 Log::info("\tCreated an SSP operator (subspace dimension = {})\n", ncomp);
740
741 //
742 // Create the whitener
743 //
744 inv.whitener = MatrixXd::Zero(inv.noise_cov.dim, inv.noise_cov.dim);
745
746 std::int32_t nnzero, k;
747 if (inv.noise_cov.diag == 0) {
748 //
749 // Omit the zeroes due to projection. The eigenvalues are sorted per channel type, not
750 // globally, so the zeroes are not necessarily the first ncomp (mne-python compute_whitener).
751 //
752 nnzero = 0;
753
754 for (k = 0; k < inv.noise_cov.dim; ++k) {
755 if (inv.noise_cov.eig[k] > 0) {
756 inv.whitener(k, k) = 1.0 / sqrt(inv.noise_cov.eig[k]);
757 ++nnzero;
758 }
759 }
760 //
761 // Rows of eigvec are the eigenvectors
762 //
763 inv.whitener *= inv.noise_cov.eigvec;
764 Log::info("\tCreated the whitener using a full noise covariance matrix ({} small eigenvalues omitted)\n", inv.noise_cov.dim - nnzero);
765 } else {
766 //
767 // No need to omit the zeroes due to projection
768 //
769 for (k = 0; k < inv.noise_cov.dim; ++k)
770 inv.whitener(k, k) = 1.0 / sqrt(inv.noise_cov.data(k, 0));
771
772 Log::info("\tCreated the whitener using a diagonal noise covariance matrix ({} small eigenvalues discarded)\n", ncomp);
773 }
774 //
775 // Finally, compute the noise-normalization factors
776 //
777 if (dSPM || sLORETA) {
778 VectorXd noise_norm = VectorXd::Zero(inv.eigen_leads.nrow);
779 VectorXd noise_weight;
780 if (dSPM) {
781 Log::info("\tComputing noise-normalization factors (dSPM)...");
782 noise_weight = VectorXd(inv.reginv);
783 } else {
784 Log::info("\tComputing noise-normalization factors (sLORETA)...");
785 VectorXd sLoretaScale = (VectorXd::Constant(inv.sing.size(), 1) + inv.sing.cwiseProduct(inv.sing) / lambda2);
786 noise_weight = inv.reginv.cwiseProduct(sLoretaScale.cwiseSqrt());
787 }
788 VectorXd one;
789 if (inv.eigen_leads_weighted) {
790 for (k = 0; k < inv.eigen_leads.nrow; ++k) {
791 one = inv.eigen_leads.data.block(k, 0, 1, inv.eigen_leads.data.cols()).cwiseProduct(noise_weight);
792 noise_norm[k] = sqrt(one.dot(one));
793 }
794 } else {
795 double c;
796 for (k = 0; k < inv.eigen_leads.nrow; ++k) {
797 c = sqrt(inv.source_cov.data(k, 0));
798 one = c * (inv.eigen_leads.data.row(k).transpose()).cwiseProduct(noise_weight);
799 noise_norm[k] = sqrt(one.dot(one));
800 }
801 }
802
803 //
804 // Compute the final result
805 //
806 VectorXd noise_norm_new = noise_norm;
807 if (inv.source_ori == FIFFV_MNE_FREE_ORI) {
808 // For free orientations the variances at three consecutive entries
809 // must be squared and summed, yielding one factor per source location.
810 VectorXd t = Linalg::combine_xyz(noise_norm.transpose());
811 noise_norm_new = t.cwiseSqrt();
812 }
813 VectorXd vOnes = VectorXd::Ones(noise_norm_new.size());
814 VectorXd noiseNormInv = vOnes.cwiseQuotient(noise_norm_new.cwiseAbs());
815
816 typedef Eigen::Triplet<double> T;
817 std::vector<T> tripletList;
818 tripletList.reserve(noise_norm_new.size());
819 for (std::int32_t i = 0; i < noise_norm_new.size(); ++i)
820 tripletList.push_back(T(i, i, noiseNormInv[i]));
821
822 inv.noisenorm = SparseMatrix<double>(noise_norm_new.size(), noise_norm_new.size());
823 inv.noisenorm.setFromTriplets(tripletList.begin(), tripletList.end());
824
825 Log::info("[done]\n");
826 } else {
827 inv.noisenorm = SparseMatrix<double>();
828 }
829
830 return inv;
831}

835Expected<MNEInverseOperator> MNEInverseOperator::read(const std::filesystem::path& path)
836{
837 auto opened = FiffStream::open(path);
838 if (!opened) {
839 return std::unexpected(opened.error());
840 }
841 FiffStream& t_pStream = *opened;
842 const std::string name = utf8_from_path(path);
843 Log::info("Reading inverse operator decomposition from {}...", name);
844 //
845 // Find all inverse operators
846 //
847 std::vector<std::shared_ptr<FiffDirNode>> invs_list = t_pStream.dirtree()->dir_tree_find(FIFFB_MNE_INVERSE_SOLUTION);
848 if (invs_list.size() == 0) {
849 return fail(ErrorCode::NotFound, "no inverse solution in {}", name);
850 }
851 const std::shared_ptr<FiffDirNode>& invs = invs_list[0];
852 //
853 // Parent MRI data
854 //
855 std::vector<std::shared_ptr<FiffDirNode>> parent_mri = t_pStream.dirtree()->dir_tree_find(FIFFB_MNE_PARENT_MRI_FILE);
856 if (parent_mri.size() == 0) {
857 return fail(ErrorCode::Format, "no parent MRI information in {}", name);
858 }
859 Log::info("\tReading inverse operator info...");
860 //
861 // Methods and source orientations
862 //
863 MNEInverseOperator inv;
864 std::optional<FiffTag> t_pTag;
865 if (t_pTag = invs->find_tag(t_pStream, FIFF_MNE_INCLUDED_METHODS); !t_pTag) {
866 return fail(ErrorCode::Format, "no included methods in {}", name);
867 }
868
869 inv.methods = *t_pTag->to_int();
870 //
871 if (t_pTag = invs->find_tag(t_pStream, FIFF_MNE_SOURCE_ORIENTATION); !t_pTag) {
872 return fail(ErrorCode::Format, "no source orientation constraints in {}", name);
873 }
874 inv.source_ori = *t_pTag->to_int();
875 //
876 if (t_pTag = invs->find_tag(t_pStream, FIFF_MNE_SOURCE_SPACE_NPOINTS); !t_pTag) {
877 return fail(ErrorCode::Format, "no source count in {}", name);
878 }
879 inv.nsource = *t_pTag->to_int();
880 inv.nchan = 0;
881 //
882 // Coordinate frame
883 //
884 if (t_pTag = invs->find_tag(t_pStream, FIFF_MNE_COORD_FRAME); !t_pTag) {
885 return fail(ErrorCode::Format, "no coordinate frame in {}", name);
886 }
887 inv.coord_frame = *t_pTag->to_int();
888 //
889 // Units of the source estimates. Optional: files written before this was
890 // recorded simply do not carry it, so a missing tag is not an error.
891 //
892 if (t_pTag = invs->find_tag(t_pStream, FIFF_MNE_INVERSE_SOURCE_UNIT); t_pTag)
893 inv.units = *t_pTag->to_int();
894 else
895 inv.units = -1;
896 //
897 // The actual source orientation vectors
898 //
899 if (t_pTag = invs->find_tag(t_pStream, FIFF_MNE_INVERSE_SOURCE_ORIENTATIONS); !t_pTag) {
900 return fail(ErrorCode::Format, "no source orientation information in {}", name);
901 }
902
903 inv.source_nn = t_pTag->to_float_matrix();
904 inv.source_nn.transposeInPlace();
905
906 Log::info("[done]\n");
907 //
908 // The SVD decomposition...
909 //
910 Log::info("\tReading inverse operator decomposition...");
911 if (t_pTag = invs->find_tag(t_pStream, FIFF_MNE_INVERSE_SING); !t_pTag) {
912 return fail(ErrorCode::Format, "no singular values in {}", name);
913 }
914
915 inv.sing = Map<const VectorXf>(t_pTag->to_float(), t_pTag->size() / 4).cast<double>();
916 inv.nchan = inv.sing.rows();
917 //
918 // The eigenleads and eigenfields
919 //
920 inv.eigen_leads_weighted = false;
921 if (!t_pStream.read_named_matrix(invs, FIFF_MNE_INVERSE_LEADS, inv.eigen_leads)) {
922 inv.eigen_leads_weighted = true;
923 if (!t_pStream.read_named_matrix(invs, FIFF_MNE_INVERSE_LEADS_WEIGHTED, inv.eigen_leads)) {
924 return fail(ErrorCode::Format, "cannot read the eigenleads of {}", name);
925 }
926 }
927 //
928 // Having the eigenleads as columns is better for the inverse calculations
929 //
930 inv.eigen_leads.transpose_named_matrix();
931
932 if (!t_pStream.read_named_matrix(invs, FIFF_MNE_INVERSE_FIELDS, inv.eigen_fields)) {
933 return fail(ErrorCode::Format, "cannot read the eigenfields of {}", name);
934 }
935 Log::info("[done]\n");
936 //
937 // Read the covariance matrices
938 //
939 if (t_pStream.read_cov(invs, FIFFV_MNE_NOISE_COV, inv.noise_cov)) {
940 Log::info("\tNoise covariance matrix read.\n");
941 } else {
942 return fail(ErrorCode::Format, "cannot read the noise covariance of {}", name);
943 }
944
945 if (t_pStream.read_cov(invs, FIFFV_MNE_SOURCE_COV, inv.source_cov)) {
946 Log::info("\tSource covariance matrix read.\n");
947 } else {
948 return fail(ErrorCode::Format, "cannot read the source covariance of {}", name);
949 }
950 //
951 // Read the various priors
952 //
953 if (t_pStream.read_cov(invs, FIFFV_MNE_ORIENT_PRIOR_COV, inv.orient_prior)) {
954 Log::info("\tOrientation priors read.\n");
955 } else
956 inv.orient_prior.clear();
957
958 if (t_pStream.read_cov(invs, FIFFV_MNE_DEPTH_PRIOR_COV, inv.depth_prior)) {
959 Log::info("\tDepth priors read.\n");
960 } else {
961 inv.depth_prior.clear();
962 }
963 if (t_pStream.read_cov(invs, FIFFV_MNE_FMRI_PRIOR_COV, inv.fmri_prior)) {
964 Log::info("\tfMRI priors read.\n");
965 } else {
966 inv.fmri_prior.clear();
967 }
968 //
969 // Read the source spaces
970 //
971 auto sourceSpaces = MNESourceSpaces::read(t_pStream, false);
972 if (!sourceSpaces) {
973 return std::unexpected(sourceSpaces.error());
974 }
975 inv.src = std::move(*sourceSpaces);
976 for (std::int32_t k = 0; k < inv.src.size(); ++k)
977 inv.src[k].id = inv.src[k].find_source_space_hemi();
978 //
979 // Get the MRI <-> head coordinate transformation
980 //
981 FiffCoordTrans mri_head_t;
982 if (t_pTag = parent_mri[0]->find_tag(t_pStream, FIFF_COORD_TRANS); !t_pTag) {
983 return fail(ErrorCode::Format, "no MRI/head coordinate transformation in {}", name);
984 } else {
985 mri_head_t = t_pTag->to_coord_trans();
986 if (mri_head_t.from != FIFFV_COORD_MRI || mri_head_t.to != FIFFV_COORD_HEAD) {
987 mri_head_t.invert_transform();
988 if (mri_head_t.from != FIFFV_COORD_MRI || mri_head_t.to != FIFFV_COORD_HEAD) {
989 return fail(ErrorCode::Format, "no MRI/head coordinate transformation in {}", name);
990 }
991 }
992 }
993 inv.mri_head_t = mri_head_t;
994
995 //
996 // get parent MEG info
997 //
998 t_pStream.read_meas_info_base(t_pStream.dirtree(), inv.info);
999
1000 //
1001 // Transform the source spaces to the correct coordinate frame
1002 // if necessary
1003 //
1004 if (inv.coord_frame != FIFFV_COORD_MRI && inv.coord_frame != FIFFV_COORD_HEAD) {
1005 return fail(ErrorCode::Format, "{} holds an inverse solution in coordinate frame {}; only MRI and head coordinates are supported", name, inv.coord_frame);
1006 }
1007 //
1008 // Number of averages is initially one
1009 //
1010 inv.nave = 1;
1011 //
1012 // We also need the SSP operator
1013 //
1014 inv.projs = t_pStream.read_proj(t_pStream.dirtree());
1015
1016 // proj, whitener, reginv, and noisenorm are filled in by prepare_inverse_operator()
1017
1018 if (!inv.src.transform_source_space_to(inv.coord_frame, mri_head_t)) {
1019 return fail(ErrorCode::Format, "cannot transform the source spaces of {}", name);
1020 }
1021 Log::info("\tSource spaces transformed to the inverse solution coordinate frame\n");
1022 return inv;
1023}

1040void MNEInverseOperator::write_to_stream(FiffStream& stream) const
1041{
1042 stream.start_block(FIFFB_MNE_INVERSE_SOLUTION);
1043
1044 Log::info("\tWriting inverse operator info...\n");
1045
1046 stream.write_int(FIFF_MNE_INCLUDED_METHODS, &this->methods);
1047 stream.write_int(FIFF_MNE_SOURCE_ORIENTATION, &this->source_ori);
1048 stream.write_int(FIFF_MNE_SOURCE_SPACE_NPOINTS, &this->nsource);
1049 stream.write_int(FIFF_MNE_COORD_FRAME, &this->coord_frame);
1050
1051 // mne-python reads this to label the source estimates. It is optional
1052 // there, but omitting it leaves inv["units"] as None, so an operator that
1053 // travelled through MNE-CPP lost information the original file carried.
1054 if (this->units > 0)
1055 stream.write_int(FIFF_MNE_INVERSE_SOURCE_UNIT, &this->units);
1056 stream.write_float_matrix(FIFF_MNE_INVERSE_SOURCE_ORIENTATIONS, this->source_nn);
1057 VectorXf tmp_sing = this->sing.cast<float>();
1058 stream.write_float(FIFF_MNE_INVERSE_SING, tmp_sing.data(), tmp_sing.size());
1059
1060 //
1061 // The eigenleads and eigenfields
1062 //
1063 // The file stores the eigenleads as rows; transpose them in a single copy
1064 FiffNamedMatrix leads;
1065 leads.row_names = eigen_leads.col_names;
1066 leads.col_names = eigen_leads.row_names;
1067 leads.data = eigen_leads.data.transpose();
1068 leads.nrow = static_cast<fiff_int_t>(leads.data.rows());
1069 leads.ncol = static_cast<fiff_int_t>(leads.data.cols());
1070 stream.write_named_matrix(this->eigen_leads_weighted ? FIFF_MNE_INVERSE_LEADS_WEIGHTED : FIFF_MNE_INVERSE_LEADS, leads);
1071
1072 stream.write_named_matrix(FIFF_MNE_INVERSE_FIELDS, eigen_fields);
1073 Log::info("\t[done]\n");
1074 //
1075 // write the covariance matrices
1076 //
1077 Log::info("\tWriting noise covariance matrix.");
1078 stream.write_cov(noise_cov);
1079
1080 Log::info("\tWriting source covariance matrix.\n");
1081 stream.write_cov(source_cov);
1082 //
1083 // write the various priors
1084 //
1085 Log::info("\tWriting orientation priors.\n");
1086 if (!orient_prior.empty())
1087 stream.write_cov(orient_prior);
1088 if (!depth_prior.empty())
1089 stream.write_cov(depth_prior);
1090 if (!fmri_prior.empty())
1091 stream.write_cov(fmri_prior);
1092
1093 //
1094 // Parent MRI data
1095 //
1096 stream.start_block(FIFFB_MNE_PARENT_MRI_FILE);
1097 // write the MRI <-> head coordinate transformation
1098 stream.write_coord_trans(this->mri_head_t);
1099 stream.end_block(FIFFB_MNE_PARENT_MRI_FILE);
1100
1101 //
1102 // Parent MEG measurement info
1103 //
1104 stream.write_info_base(this->info);
1105
1106 //
1107 // Write the source spaces
1108 //
1109 if (!src.empty())
1110 this->src.write_to_stream(stream);
1111
1112 //
1113 // We also need the SSP operator
1114 //
1115 stream.write_proj(this->projs);
1116 //
1117 // Done!
1118 //
1119 stream.end_block(FIFFB_MNE_INVERSE_SOLUTION);
1120}
