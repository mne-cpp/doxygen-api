34RtSourceDataController::RtSourceDataController(QObject* parent)
35: QObject(parent)
36{
37 // Create data worker
38 m_pWorker = new DISP3DLIB::RtSourceDataWorker();
39
40 // Create and configure data thread
41 m_pWorkerThread = new QThread(this);
42 m_pWorker->moveToThread(m_pWorkerThread);
43
44 // Clean up worker when thread finishes
45 connect(m_pWorkerThread, &QThread::finished, m_pWorker, &QObject::deleteLater);
46
47 // Forward worker signals to controller signals
48 connect(m_pWorker, &DISP3DLIB::RtSourceDataWorker::newRtSmoothedData,
49 this, &RtSourceDataController::newSmoothedDataAvailable);
50 connect(m_pWorker, &DISP3DLIB::RtSourceDataWorker::newRtRawData,
51 this, &RtSourceDataController::newRawDataAvailable);
52
53 // Create timer for driving the streaming cadence
54 m_pTimer = new QTimer(this);
55 m_pTimer->setTimerType(Qt::PreciseTimer);
56 connect(m_pTimer, &QTimer::timeout, m_pWorker, &DISP3DLIB::RtSourceDataWorker::streamData);
57
58 // Start the data worker thread
59 m_pWorkerThread->start();
60
61 // Create interpolation matrix worker
62 m_pInterpWorker = new DISP3DLIB::RtSourceInterpolationMatWorker();
63 m_pInterpThread = new QThread(this);
64 m_pInterpWorker->moveToThread(m_pInterpThread);
65
66 connect(m_pInterpThread, &QThread::finished, m_pInterpWorker, &QObject::deleteLater);
67
68 // Forward interpolation results: auto-apply to data worker + re-emit
69 connect(m_pInterpWorker, &DISP3DLIB::RtSourceInterpolationMatWorker::newInterpolationMatrixLeftAvailable,
70 this, &RtSourceDataController::onNewInterpolationMatrixLeft);
71 connect(m_pInterpWorker, &DISP3DLIB::RtSourceInterpolationMatWorker::newInterpolationMatrixRightAvailable,
72 this, &RtSourceDataController::onNewInterpolationMatrixRight);
73
74 m_pInterpThread->start();
75
76 qDebug() << "RtSourceDataController: Initialized with" << m_iTimeInterval << "ms interval";
77}
