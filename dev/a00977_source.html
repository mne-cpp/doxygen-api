101void append_quoted(std::wstring& line, const std::wstring& arg)
102{
103 if (!arg.empty() && arg.find_first_of(L" \t\n\v\"") == std::wstring::npos) {
104 line += arg;
105 return;
106 }
107 line += L'"';
108 std::size_t backslashes = 0;
109 for (const wchar_t c : arg) {
110 if (c == L'\\') {
111 ++backslashes;
112 continue;
113 }
114 line.append(c == L'"' ? (2 * backslashes) + 1 : backslashes, L'\\');
115 backslashes = 0;
116 line += c;
117 }
118 line.append(2 * backslashes, L'\\');
119 line += L'"';
120}

123std::wstring environment_block(const std::vector<std::pair<std::string, std::string>>& overrides)
124{
125 if (overrides.empty())
126 return {};
127 std::vector<std::wstring> entries;
128 if (wchar_t* strings = GetEnvironmentStringsW()) {
129 for (const wchar_t* p = strings; *p != L'\0'; p += std::wcslen(p) + 1)
130 entries.emplace_back(p);
131 FreeEnvironmentStringsW(strings);
132 }
133 for (const auto& [key, value] : overrides) {
134 const std::wstring wkey = widen(key);
135 std::erase_if(entries, [&](const std::wstring& entry) {
136 return entry.size() > wkey.size() && entry[wkey.size()] == L'=' &&
137 CompareStringOrdinal(entry.data(), static_cast<int>(wkey.size()), wkey.data(), static_cast<int>(wkey.size()), TRUE) == CSTR_EQUAL;
138 });
139 entries.push_back(wkey + L'=' + widen(value));
140 }
141 std::wstring block;
142 for (const std::wstring& entry : entries) {
143 block += entry;
144 block += L'\0';
145 }
146 block += L'\0';
147 return block;
148}

151void drain(HANDLE& pipe, ProcessChannel channel, const ProcessOptions& options, ProcessResult& result, std::array<char, kChunk>& buffer)
152{
153 while (pipe != nullptr) {
154 DWORD available = 0;
155 if (!PeekNamedPipe(pipe, nullptr, 0, nullptr, &available, nullptr)) {
156 CloseHandle(pipe);
157 pipe = nullptr;
158 return;
159 }
160 if (available == 0)
161 return;
162 DWORD read = 0;
163 if (!ReadFile(pipe, buffer.data(), static_cast<DWORD>(std::min<std::size_t>(available, buffer.size())), &read, nullptr) || read == 0)
164 return;
165 deliver(options, result, channel, std::string_view(buffer.data(), read));
166 }
167}

210ProcessResult mne::run_process(const std::filesystem::path& program, std::span<const std::string> args, const ProcessOptions& options)
211{
212 ProcessResult result;
213 using Clock = std::chrono::steady_clock;
214 // No std::optional: GCC's -Wmaybe-uninitialized misreads one that is set on both branches
215 const bool hasDeadline = options.timeout.has_value();
216 const Clock::time_point deadline = Clock::now() + std::chrono::duration_cast<Clock::duration>(options.timeout.value_or(std::chrono::milliseconds(0)));
217 std::array<char, kChunk> buffer{};
218
219#if defined(__EMSCRIPTEN__)
220 static_cast<void>(program);
221 static_cast<void>(args);
222 static_cast<void>(hasDeadline);
223 static_cast<void>(deadline);
224 static_cast<void>(buffer);
225 result.error = "processes are not available in WebAssembly";
226 return result;
227
228#elif defined(_WIN32)
229 std::wstring line;
230 append_quoted(line, program.wstring());
231 for (const std::string& arg : args) {
232 line += L' ';
233 append_quoted(line, widen(arg));
234 }
235 std::wstring environment = environment_block(options.environment);
236
237 SECURITY_ATTRIBUTES inherit{sizeof(SECURITY_ATTRIBUTES), nullptr, TRUE};
238 HANDLE outRead = nullptr;
239 HANDLE outWrite = nullptr;
240 HANDLE errRead = nullptr;
241 HANDLE errWrite = nullptr;
242 HANDLE input = CreateFileW(L"NUL", GENERIC_READ, FILE_SHARE_READ | FILE_SHARE_WRITE, &inherit, OPEN_EXISTING, 0, nullptr);
243 if (!CreatePipe(&outRead, &outWrite, &inherit, 0) || !CreatePipe(&errRead, &errWrite, &inherit, 0)) {
244 result.error = std::system_category().message(static_cast<int>(GetLastError()));
245 for (HANDLE h : {input, outRead, outWrite, errRead, errWrite}) {
246 if (h != nullptr && h != INVALID_HANDLE_VALUE)
247 CloseHandle(h);
248 }
249 return result;
250 }
251 SetHandleInformation(outRead, HANDLE_FLAG_INHERIT, 0);
252 SetHandleInformation(errRead, HANDLE_FLAG_INHERIT, 0);
253
254 // Only these three handles reach the child, so concurrent runs cannot keep each other's pipes open
255 std::array<HANDLE, 3> inherited{input, outWrite, errWrite};
256 SIZE_T attributeSize = 0;
257 InitializeProcThreadAttributeList(nullptr, 1, 0, &attributeSize);
258 std::vector<std::byte> attributeStorage(attributeSize);
259 auto* attributes = reinterpret_cast<LPPROC_THREAD_ATTRIBUTE_LIST>(attributeStorage.data());
260 InitializeProcThreadAttributeList(attributes, 1, 0, &attributeSize);
261 UpdateProcThreadAttribute(attributes, 0, PROC_THREAD_ATTRIBUTE_HANDLE_LIST, inherited.data(), sizeof(inherited), nullptr, nullptr);
262
263 STARTUPINFOEXW startup{};
264 startup.StartupInfo.cb = sizeof(startup);
265 startup.StartupInfo.dwFlags = STARTF_USESTDHANDLES;
266 startup.StartupInfo.hStdInput = input;
267 startup.StartupInfo.hStdOutput = outWrite;
268 startup.StartupInfo.hStdError = errWrite;
269 startup.lpAttributeList = attributes;
270 PROCESS_INFORMATION info{};
271 const std::wstring workingDir = options.working_dir.wstring();
272 const BOOL created = CreateProcessW(nullptr, line.data(), nullptr, nullptr, TRUE, CREATE_NO_WINDOW | CREATE_UNICODE_ENVIRONMENT | EXTENDED_STARTUPINFO_PRESENT,
273 environment.empty() ? nullptr : environment.data(), workingDir.empty() ? nullptr : workingDir.c_str(),
274 &startup.StartupInfo, &info);
275 const DWORD createError = GetLastError();
276 DeleteProcThreadAttributeList(attributes);
277 CloseHandle(input);
278 CloseHandle(outWrite);
279 CloseHandle(errWrite);
280 if (!created) {
281 result.error = std::system_category().message(static_cast<int>(createError));
282 CloseHandle(outRead);
283 CloseHandle(errRead);
284 return result;
285 }
286 CloseHandle(info.hThread);
287 result.started = true;
288
289 for (;;) {
290 drain(outRead, ProcessChannel::StdOut, options, result, buffer);
291 drain(errRead, ProcessChannel::StdErr, options, result, buffer);
292 if (WaitForSingleObject(info.hProcess, static_cast<DWORD>(kPoll.count())) == WAIT_OBJECT_0)
293 break;
294 if (hasDeadline && Clock::now() >= deadline) {
295 TerminateProcess(info.hProcess, 1);
296 WaitForSingleObject(info.hProcess, INFINITE);
297 result.timed_out = true;
298 break;
299 }
300 }
301 drain(outRead, ProcessChannel::StdOut, options, result, buffer);
302 drain(errRead, ProcessChannel::StdErr, options, result, buffer);
303 for (HANDLE h : {outRead, errRead}) {
304 if (h != nullptr)
305 CloseHandle(h);
306 }
307 DWORD exitCode = 0;
308 GetExitCodeProcess(info.hProcess, &exitCode);
309 CloseHandle(info.hProcess);
310 if (!result.timed_out)
311 result.exit_code = static_cast<int>(exitCode);
312 return result;
313
314#else
315 std::array<int, 2> out{-1, -1};
316 std::array<int, 2> err{-1, -1};
317 if (!make_pipe(out) || !make_pipe(err)) {
318 result.error = std::generic_category().message(errno);
319 for (const int fd : {out[0], out[1], err[0], err[1]}) {
320 if (fd >= 0)
321 close(fd);
322 }
323 return result;
324 }
325
326 posix_spawn_file_actions_t actions;
327 posix_spawn_file_actions_init(&actions);
328 posix_spawn_file_actions_addopen(&actions, STDIN_FILENO, "/dev/null", O_RDONLY, 0);
329 posix_spawn_file_actions_adddup2(&actions, out[1], STDOUT_FILENO);
330 posix_spawn_file_actions_adddup2(&actions, err[1], STDERR_FILENO);
331 const std::string workingDir = options.working_dir.string();
332 if (!workingDir.empty())
333 add_chdir(actions, workingDir.c_str());
334
335 const std::string path = program.string();
336 std::vector<char*> argv;
337 argv.reserve(args.size() + 2);
338 argv.push_back(const_cast<char*>(path.c_str())); // NOLINT(cppcoreguidelines-pro-type-const-cast): posix_spawn takes char* const[]
339 for (const std::string& arg : args)
340 argv.push_back(const_cast<char*>(arg.c_str())); // NOLINT(cppcoreguidelines-pro-type-const-cast): posix_spawn takes char* const[]
341 argv.push_back(nullptr);
342
343 std::vector<std::string> environment;
344 std::vector<char*> envp;
345 char** env = environ;
346 if (!options.environment.empty()) {
347 for (char** p = environ; *p != nullptr; ++p) {
348 const std::string_view entry(*p);
349 const bool replaced = std::ranges::any_of(options.environment, [&](const auto& kv) {
350 return entry.size() > kv.first.size() && entry.starts_with(kv.first) && entry[kv.first.size()] == '=';
351 });
352 if (!replaced)
353 environment.emplace_back(entry);
354 }
355 for (const auto& [key, value] : options.environment)
356 environment.push_back(std::format("{}={}", key, value));
357 envp.reserve(environment.size() + 1);
358 for (std::string& entry : environment)
359 envp.push_back(entry.data());
360 envp.push_back(nullptr);
361 env = envp.data();
362 }
363
364 pid_t pid = 0;
365 const int spawned = posix_spawnp(&pid, path.c_str(), &actions, nullptr, argv.data(), env);
366 posix_spawn_file_actions_destroy(&actions);
367 close(out[1]);
368 close(err[1]);
369 if (spawned != 0) {
370 result.error = std::generic_category().message(spawned);
371 close(out[0]);
372 close(err[0]);
373 return result;
374 }
375 result.started = true;
376
377 std::array<int, 2> fds{out[0], err[0]};
378 int status = 0;
379 bool exited = false;
380 while (!exited) {
381 std::array<pollfd, 2> polled{pollfd{fds[0], POLLIN, 0}, pollfd{fds[1], POLLIN, 0}};
382 // Poll in short steps: a grandchild that inherited the pipes must not hold us after the child exits
383 if (poll(polled.data(), polled.size(), static_cast<int>(kPoll.count())) > 0) {
384 for (std::size_t i = 0; i < fds.size(); ++i) {
385 if (fds[i] >= 0 && (polled[i].revents & (POLLIN | POLLHUP | POLLERR)) != 0)
386 drain(fds[i], i == 0 ? ProcessChannel::StdOut : ProcessChannel::StdErr, options, result, buffer);
387 }
388 }
389 exited = waitpid(pid, &status, WNOHANG) == pid;
390 if (!exited && hasDeadline && Clock::now() >= deadline) {
391 kill(pid, SIGKILL);
392 while (waitpid(pid, &status, 0) < 0 && errno == EINTR) {
393 }
394 result.timed_out = true;
395 break;
396 }
397 }
398 for (int& fd : fds) {
399 if (fd < 0)
400 continue;
401 fcntl(fd, F_SETFL, fcntl(fd, F_GETFL) | O_NONBLOCK);
402 while (fd >= 0) {
403 const ssize_t n = read(fd, buffer.data(), buffer.size());
404 if (n <= 0) {
405 close(fd);
406 fd = -1;
407 break;
408 }
409 deliver(options, result, &fd == fds.data() ? ProcessChannel::StdOut : ProcessChannel::StdErr, std::string_view(buffer.data(), static_cast<std::size_t>(n)));
410 }
411 }
412 if (result.timed_out)
413 return result;
414 if (WIFEXITED(status))
415 result.exit_code = WEXITSTATUS(status);
416 else
417 result.crashed = WIFSIGNALED(status);
418 return result;
419#endif
420}

448std::string mne::os_pretty_name()
449{
450#if defined(_WIN32)
451 // RtlGetVersion reports the real version; GetVersionEx is capped by the application manifest
452 using RtlGetVersion = LONG(WINAPI*)(OSVERSIONINFOEXW*);
453 OSVERSIONINFOEXW version{};
454 version.dwOSVersionInfoSize = sizeof(version);
455 if (HMODULE ntdll = GetModuleHandleW(L"ntdll.dll")) {
456 if (auto get = reinterpret_cast<RtlGetVersion>(reinterpret_cast<void*>(GetProcAddress(ntdll, "RtlGetVersion"))))
457 get(&version);
458 }
459 const auto [major, minor, build] = std::tuple(version.dwMajorVersion, version.dwMinorVersion, version.dwBuildNumber);
460 if (version.wProductType != VER_NT_WORKSTATION && major == 10) {
461 // Server releases are named by year, told apart by their build, as QSysInfo does
462 const char* year = build >= 26100 ? "2025" : build >= 20348 ? "2022"
463 : build >= 17763 ? "2019"
464 : "2016";
465 return std::format("Windows Server {} ({}.{}.{})", year, major, minor, build);
466 }
467 const ULONG product = major == 10 && build >= 22000 ? 11 : major;
468 return std::format("Windows {} ({}.{}.{})", product, major, minor, build);
469#elif defined(__APPLE__)
470 std::array<char, 64> version{};
471 std::size_t size = version.size();
472 if (sysctlbyname("kern.osproductversion", version.data(), &size, nullptr, 0) != 0)
473 return "macOS";
474 return std::string("macOS ") + version.data();
475#elif defined(__EMSCRIPTEN__)
476 return "WebAssembly";
477#else
478 for (const char* file : {"/etc/os-release", "/usr/lib/os-release"}) {
479 std::ifstream in(file);
480 for (std::string line; std::getline(in, line);) {
481 if (!line.starts_with("PRETTY_NAME="))
482 continue;
483 std::string_view value = std::string_view(line).substr(12);
484 if (value.size() >= 2 && (value.front() == '"' || value.front() == '\'') && value.back() == value.front())
485 value = value.substr(1, value.size() - 2);
486 return std::string(value);
487 }
488 }
489 utsname system{};
490 if (uname(&system) != 0)
491 return {};
492 return std::string(system.sysname) + ' ' + system.release;
493#endif
494}

519std::vector<HardwareAddress> mne::hardware_addresses()
520{
521 std::vector<HardwareAddress> addresses;
522 [[maybe_unused]] const auto add = [&addresses](const unsigned char* bytes, std::size_t length) {
523 if (length != 6)
524 return;
525 HardwareAddress address{};
526 std::copy_n(bytes, address.size(), address.begin());
527 if (std::ranges::any_of(address, [](std::uint8_t b) { return b != 0; }))
528 addresses.push_back(address);
529 };
530#if defined(_WIN32)
531 ULONG size = 16 * 1024;
532 std::vector<unsigned char> buffer;
533 ULONG status = ERROR_BUFFER_OVERFLOW;
534 for (int attempt = 0; attempt < 3 && status == ERROR_BUFFER_OVERFLOW; ++attempt) {
535 buffer.resize(size);
536 status = GetAdaptersAddresses(AF_UNSPEC, GAA_FLAG_SKIP_ANYCAST | GAA_FLAG_SKIP_MULTICAST | GAA_FLAG_SKIP_DNS_SERVER, nullptr, reinterpret_cast<IP_ADAPTER_ADDRESSES*>(buffer.data()), &size);
537 }
538 if (status != NO_ERROR)
539 return addresses;
540 for (const IP_ADAPTER_ADDRESSES* adapter = reinterpret_cast<const IP_ADAPTER_ADDRESSES*>(buffer.data()); adapter; adapter = adapter->Next) {
541 if (adapter->OperStatus != IfOperStatusUp || adapter->IfType == IF_TYPE_SOFTWARE_LOOPBACK || adapter->IfType == IF_TYPE_PPP)
542 continue;
543 add(adapter->PhysicalAddress, adapter->PhysicalAddressLength);
544 }
545#elif defined(__APPLE__) || defined(__linux__)
546 ifaddrs* list = nullptr;
547 if (getifaddrs(&list) != 0)
548 return addresses;
549 for (const ifaddrs* entry = list; entry; entry = entry->ifa_next) {
550 const unsigned int flags = entry->ifa_flags;
551 if (!entry->ifa_addr || !(flags & IFF_UP) || !(flags & IFF_RUNNING) || (flags & IFF_LOOPBACK) || (flags & IFF_POINTOPOINT))
552 continue;
553#if defined(__APPLE__)
554 if (entry->ifa_addr->sa_family == AF_LINK) {
555 const auto* link = reinterpret_cast<const sockaddr_dl*>(entry->ifa_addr);
556 add(reinterpret_cast<const unsigned char*>(LLADDR(link)), link->sdl_alen);
557 }
558#else
559 if (entry->ifa_addr->sa_family == AF_PACKET) {
560 const auto* link = reinterpret_cast<const sockaddr_ll*>(entry->ifa_addr);
561 add(link->sll_addr, link->sll_halen);
562 }
563#endif
564 }
565 freeifaddrs(list);
566#endif
567 return addresses;
568}
