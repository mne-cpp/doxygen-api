40BidsCoordinateSystem BidsCoordinateSystem::read_json(const std::filesystem::path& path)
41{
42 BidsCoordinateSystem cs;
43 cs.transform = Eigen::Matrix4d::Identity();
44
45 const mne::Expected<std::string> text = mne::read_text_file(path);
46 if (!text) {
47 mne::Log::warning("[BidsCoordinateSystem::read_json] Cannot open {}", mne::utf8_from_path(path));
48 return cs;
49 }
50
51 mne::JsonParseError error;
52 const mne::JsonDocument doc = mne::JsonDocument::from_json(*text, &error);
53 if (error.error != mne::JsonParseError::NoError) {
54 mne::Log::warning("[BidsCoordinateSystem::read_json] Parse error in {} : {}", mne::utf8_from_path(path), error.error_string());
55 return cs;
56 }
57
58 const mne::JsonObject& json = doc.object();
59
60 // iEEG fields first, then EEG, then MEG
61 const auto field = [&json](std::string_view name) {
62 for (const char* prefix : {"iEEG", "EEG", "MEG"}) {
63 const std::string_view value = json.value(std::string(prefix).append(name)).to_string();
64 if (!value.empty())
65 return std::string(value);
66 }
67 return std::string();
68 };
69 cs.system = field("CoordinateSystem");
70 cs.units = field("CoordinateUnits");
71 cs.description = field("CoordinateSystemDescription");
72
73 cs.processingDescription = json.value("iEEGCoordinateProcessingDescription").to_string();
74 if (cs.processingDescription.empty())
75 cs.processingDescription = json.value("EEGCoordinateProcessingDescription").to_string();
76
77 cs.associatedImagePath = json.value("IntendedFor").to_string();
78
79 // Parse 4x4 transform if provided as "iEEGCoordinateProcessingTransform" or "Transform"
80 const std::string_view transformKey = json.contains("iEEGCoordinateProcessingTransform") ? "iEEGCoordinateProcessingTransform"
81 : json.contains("Transform") ? "Transform"
82 : "";
83 if (!transformKey.empty()) {
84 const mne::JsonArray& rows = json.value(transformKey).to_array();
85 if (rows.size() == 4) {
86 for (int r = 0; r < 4; ++r) {
87 const mne::JsonArray& cols = rows[static_cast<std::size_t>(r)].to_array();
88 if (cols.size() == 4) {
89 for (int c = 0; c < 4; ++c)
90 cs.transform(r, c) = cols[static_cast<std::size_t>(c)].to_double();
91 }
92 }
93 }
94 }
95
96 return cs;
97}

101bool BidsCoordinateSystem::write_json(const std::filesystem::path& path,
102 const BidsCoordinateSystem& cs,
103 std::string_view sModality)
104{
105 const std::string modality(sModality);
106 mne::JsonObject json{{modality + "CoordinateSystem", cs.system},
107 {modality + "CoordinateUnits", cs.units}};
108
109 if (!cs.description.empty())
110 json.insert(modality + "CoordinateSystemDescription", cs.description);
111 if (!cs.processingDescription.empty() && sModality == "iEEG")
112 json.insert("iEEGCoordinateProcessingDescription", cs.processingDescription);
113 if (!cs.associatedImagePath.empty())
114 json.insert("IntendedFor", cs.associatedImagePath);
115
116 // Serialize the 4x4 transform matrix
117 if (!cs.transform.isIdentity(1e-15) && sModality == "iEEG") {
118 mne::JsonArray rows;
119 rows.reserve(4);
120 for (int r = 0; r < 4; ++r) {
121 mne::JsonArray cols;
122 cols.reserve(4);
123 for (int c = 0; c < 4; ++c)
124 cols.push_back(cs.transform(r, c));
125 rows.push_back(std::move(cols));
126 }
127 json.insert("iEEGCoordinateProcessingTransform", std::move(rows));
128 }
129
130 if (!mne::write_text_file(path, mne::JsonDocument(std::move(json)).to_json())) {
131 mne::Log::warning("[BidsCoordinateSystem::write_json] Cannot open {} for writing", mne::utf8_from_path(path));
132 return false;
133 }
134 return true;
135}
