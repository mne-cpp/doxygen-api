41// Relative comparison with the tolerances of qFuzzyCompare (5 decimal digits for float, 12 for double)
42bool fuzzyEqual(float a, float b)
43{
44 return std::abs(a - b) * 100000.0f <= std::min(std::abs(a), std::abs(b));
45}
46
47bool fuzzyEqual(double a, double b)
48{
49 return std::abs(a - b) * 1000000000000.0 <= std::min(std::abs(a), std::abs(b));
50}

66void SceneState::addLayer(SceneLayer layer)
67{
68 if (layer.id.empty()) {
69 mne::Log::warning("SceneState::addLayer: refusing to add layer with empty id");
70 return;
71 }
72
73 const auto it = m_indexById.find(layer.id);
74 if (it != m_indexById.end()) {
75 m_layers[it->second] = std::move(layer);
76 } else {
77 m_indexById.insert_or_assign(layer.id, m_layers.size());
78 m_layers.push_back(std::move(layer));
79 }
80 rebuildOrder();
81 if (m_layersChangedCallback) {
82 m_layersChangedCallback();
83 }
84}

235void SceneState::setOverlayThresholds(float fmin, float fmid, float fmax)
236{
237 // Clamp to monotone fmin <= fmid <= fmax.
238 if (fmid < fmin) {
239 fmid = fmin;
240 }
241 if (fmax < fmid) {
242 fmax = fmid;
243 }
244 const bool same = fuzzyEqual(fmin + 1.0f, m_overlayFmin + 1.0f) && fuzzyEqual(fmid + 1.0f, m_overlayFmid + 1.0f) && fuzzyEqual(fmax + 1.0f, m_overlayFmax + 1.0f);
245 if (same) {
246 return;
247 }
248 m_overlayFmin = fmin;
249 m_overlayFmid = fmid;
250 m_overlayFmax = fmax;
251 if (m_overlayThresholdsChangedCallback) {
252 m_overlayThresholdsChangedCallback(fmin, fmid, fmax);
253 }

275void SceneState::worldBounds(Eigen::Vector3f& bbMin, Eigen::Vector3f& bbMax) const
276{
277 constexpr float fmax = std::numeric_limits<float>::max();
278 constexpr float flow = std::numeric_limits<float>::lowest();
279 bbMin = Eigen::Vector3f::Constant(fmax);
280 bbMax = Eigen::Vector3f::Constant(flow);
281
282 bool anyHit = false;
283 for (const auto& layer : m_layers) {
284 if (!layer.visible) {
285 continue;
286 }
287 const auto it = m_boundsFns.find(static_cast<int>(layer.kind));
288 if (it == m_boundsFns.end() || !it->second) {
289 continue;
290 }
291 Eigen::Vector3f mn = Eigen::Vector3f::Zero();
292 Eigen::Vector3f mx = Eigen::Vector3f::Zero();
293 if (!it->second(layer, mn, mx)) {
294 continue;
295 }
296 bbMin = bbMin.cwiseMin(mn);
297 bbMax = bbMax.cwiseMax(mx);
298 anyHit = true;
299 }
300
301 if (!anyHit) {
302 bbMin = Eigen::Vector3f::Constant(-1.0f);
303 bbMax = Eigen::Vector3f::Constant(1.0f);
304 }
305}

316void SceneState::rebuildOrder()
317{
318 // Stable sort by (kind, drawOrder); insertion order is preserved as tie-breaker.
319 std::stable_sort(m_layers.begin(), m_layers.end(),
320 [](const SceneLayer& a, const SceneLayer& b) {
321 if (a.kind != b.kind) {
322 return static_cast<int>(a.kind) < static_cast<int>(b.kind);
323 }
324 return a.drawOrder < b.drawOrder;
325 });
326 m_indexById.clear();
327 m_indexById.reserve(m_layers.size());
328 for (std::size_t i = 0; i < m_layers.size(); ++i) {
329 m_indexById.insert_or_assign(m_layers[i].id, i);
330 }
331}
