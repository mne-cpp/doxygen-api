49MatrixXd computePatterns(const MatrixXd& filters, const MatrixXd& dataCov)
50{
51 if (filters.size() == 0 || dataCov.size() == 0) {
52 return {};
53 }
54
55 MatrixXd gram = filters.transpose() * dataCov * filters;
56 CompleteOrthogonalDecomposition<MatrixXd> cod(gram);
57 return dataCov * filters * cod.pseudoInverse();
58}

66XdawnResult Xdawn::fit(const std::vector<MNEEpochData>& epochs,
67 int iTargetEvent,
68 int nComponents,
69 double dReg)
70{
71 XdawnResult result;
72 result.iTargetEvent = iTargetEvent;
73
74 if (epochs.empty()) {
75 Log::warning("Xdawn::fit: empty epoch list.");
76 return result;
77 }
78
79 std::vector<int> goodIdx;
80 goodIdx.reserve(epochs.size());
81 for (int i = 0; i < std::ssize(epochs); ++i) {
82 if (!epochs[i].bReject) {
83 goodIdx.push_back(i);
84 }
85 }
86
87 if (goodIdx.empty()) {
88 Log::warning("Xdawn::fit: no non-rejected epochs available.");
89 return result;
90 }
91
92 const int nCh = static_cast<int>(epochs[goodIdx[0]].epoch.rows());
93 const int nSamp = static_cast<int>(epochs[goodIdx[0]].epoch.cols());
94 if (nCh == 0 || nSamp == 0) {
95 Log::warning("Xdawn::fit: epoch matrices are empty.");
96 return result;
97 }
98
99 for (int idx : goodIdx) {
100 if (epochs[idx].epoch.rows() != nCh || epochs[idx].epoch.cols() != nSamp) {
101 Log::warning("Xdawn::fit: epoch dimension mismatch.");
102 return result;
103 }
104 }
105
106 nComponents = std::max(1, std::min(nComponents, nCh));
107
108 std::map<int, MatrixXd> classSums;
109 std::map<int, int> classCounts;
110 MatrixXd targetSum = MatrixXd::Zero(nCh, nSamp);
111 int nTarget = 0;
112
113 for (int idx : goodIdx) {
114 const MNEEpochData& ep = epochs[idx];
115 auto [sum, added] = classSums.try_emplace(ep.event, ep.epoch);
116 if (!added) {
117 sum->second += ep.epoch;
118 }
119 classCounts[ep.event] += 1;
120
121 if (ep.event == iTargetEvent) {
122 targetSum += ep.epoch;
123 ++nTarget;
124 }
125 }
126
127 if (nTarget == 0) {
128 Log::warning("Xdawn::fit: no target epochs found for event {}", iTargetEvent);
129 return result;
130 }
131
132 result.matTargetEvoked = targetSum / static_cast<double>(nTarget);
133
134 MatrixXd noiseCov = MatrixXd::Zero(nCh, nCh);
135 MatrixXd dataCov = MatrixXd::Zero(nCh, nCh);
136 long long nNoiseSamples = 0;
137 long long nDataSamples = 0;
138
139 // The sums become the class means
140 for (auto& [event, sum] : classSums) {
141 sum /= static_cast<double>(classCounts[event]);
142 }
143
144 for (int idx : goodIdx) {
145 const MNEEpochData& ep = epochs[idx];
146 const MatrixXd residual = ep.epoch - classSums.at(ep.event);
147
148 dataCov += ep.epoch * ep.epoch.transpose();
149 noiseCov += residual * residual.transpose();
150 nDataSamples += nSamp;
151 nNoiseSamples += nSamp;
152 }
153
154 if (nNoiseSamples <= 0 || nDataSamples <= 0) {
155 Log::warning("Xdawn::fit: failed to accumulate covariance samples.");
156 return result;
157 }
158
159 result.matSignalCov = result.matTargetEvoked * result.matTargetEvoked.transpose() / static_cast<double>(nSamp);
160 result.matNoiseCov = noiseCov / static_cast<double>(nNoiseSamples);
161 dataCov = dataCov / static_cast<double>(nDataSamples);
162
163 const double traceNoise = result.matNoiseCov.trace();
164 const double regValue = std::max(dReg, 0.0) * ((traceNoise > 0.0) ? traceNoise / static_cast<double>(nCh) : 1.0);
165 MatrixXd regNoiseCov = result.matNoiseCov;
166 regNoiseCov.diagonal().array() += regValue;
167
168 SelfAdjointEigenSolver<MatrixXd> noiseEig(regNoiseCov);
169 if (noiseEig.info() != Success) {
170 Log::warning("Xdawn::fit: noise covariance eigendecomposition failed.");
171 return result;
172 }
173
174 VectorXd noiseVals = noiseEig.eigenvalues().cwiseMax(1e-12);
175 MatrixXd noiseVecs = noiseEig.eigenvectors();
176 MatrixXd invSqrtNoise = noiseVecs * noiseVals.cwiseInverse().cwiseSqrt().asDiagonal() * noiseVecs.transpose();
177
178 MatrixXd whitenedSignal = invSqrtNoise * result.matSignalCov * invSqrtNoise;
179 SelfAdjointEigenSolver<MatrixXd> signalEig(whitenedSignal);
180 if (signalEig.info() != Success) {
181 Log::warning("Xdawn::fit: signal covariance eigendecomposition failed.");
182 return result;
183 }
184
185 result.matFilters.resize(nCh, nComponents);
186 const MatrixXd signalVecsAsc = signalEig.eigenvectors().rightCols(nComponents);
187 MatrixXd signalVecs(signalVecsAsc.rows(), signalVecsAsc.cols());
188 for (int i = 0; i < nComponents; ++i) {
189 signalVecs.col(i) = signalVecsAsc.col(nComponents - 1 - i);
190 }
191
192 result.matFilters = invSqrtNoise * signalVecs;
193
194 for (int col = 0; col < result.matFilters.cols(); ++col) {
195 const double noiseNorm = std::sqrt(result.matFilters.col(col).transpose() * regNoiseCov * result.matFilters.col(col));
196 if (noiseNorm > 1e-12) {
197 result.matFilters.col(col) /= noiseNorm;
198 }
199 }
200
201 result.matPatterns = computePatterns(result.matFilters, dataCov);
202 result.bValid = true;
203 return result;
204}

208MatrixXd Xdawn::apply(const MatrixXd& matEpoch, const XdawnResult& result)
209{
210 if (!result.bValid || result.matFilters.size() == 0) {
211 return {};
212 }
213
214 if (matEpoch.rows() != result.matFilters.rows()) {
215 Log::warning("Xdawn::apply: channel count mismatch.");
216 return {};
217 }
218
219 return result.matFilters.transpose() * matEpoch;
220}

224MatrixXd Xdawn::denoise(const MatrixXd& matEpoch, const XdawnResult& result, int nComponents)
225{
226 if (!result.bValid || result.matFilters.size() == 0 || result.matPatterns.size() == 0) {
227 return matEpoch;
228 }
229
230 if (matEpoch.rows() != result.matFilters.rows()) {
231 Log::warning("Xdawn::denoise: channel count mismatch.");
232 return {};
233 }
234
235 if (nComponents <= 0 || nComponents > result.matFilters.cols()) {
236 nComponents = result.matFilters.cols();
237 }
238
239 MatrixXd filters = result.matFilters.leftCols(nComponents);
240 MatrixXd patterns = result.matPatterns.leftCols(nComponents);
241
242 return patterns * (filters.transpose() * matEpoch);
243}
