583inline std::int32_t FiffTag::storage_size()
584{
585 // On-disk layout: kind, type, size, next (4 x fiff_int_t)
586 // The on-disk size field is the payload size, size()
587 return sizeof(FiffTag::kind) + sizeof(FiffTag::type) + sizeof(fiff_int_t) /* size field on disk */
588 + sizeof(FiffTag::next);
589}

719inline FiffCoordTrans FiffTag::to_coord_trans() const
720{
721 FiffCoordTrans p_FiffCoordTrans;
722 if (this->is_matrix() || this->data_type() != FIFFT_COORD_TRANS_STRUCT || this->size() < static_cast<std::size_t>(FiffCoordTrans::storage_size()))
723 return p_FiffCoordTrans;
724 else {
725 auto* t_pInt32 = reinterpret_cast<const std::int32_t*>(this->data());
726 p_FiffCoordTrans.from = t_pInt32[0];
727 p_FiffCoordTrans.to = t_pInt32[1];
728
729 p_FiffCoordTrans.trans.setIdentity(4, 4);
730 auto* t_pFloat = reinterpret_cast<const float*>(this->data());
731 int count = 0;
732 int r, c;
733 for (r = 0; r < 3; ++r) {
734 p_FiffCoordTrans.trans(r, 3) = t_pFloat[11 + r];
735 for (c = 0; c < 3; ++c) {
736 p_FiffCoordTrans.trans(r, c) = t_pFloat[2 + count];
737 ++count;
738 }
739 }
740
741 p_FiffCoordTrans.invtrans.setIdentity(4, 4);
742 count = 0;
743 for (r = 0; r < 3; ++r) {
744 p_FiffCoordTrans.invtrans(r, 3) = t_pFloat[23 + r];
745 for (c = 0; c < 3; ++c) {
746 p_FiffCoordTrans.invtrans(r, c) = t_pFloat[14 + count];
747 ++count;
748 }
749 }
750
751 return p_FiffCoordTrans;
752 }

759inline FiffChInfo FiffTag::to_ch_info() const
760{
761 FiffChInfo p_FiffChInfo;
762
763 if (this->is_matrix() || this->data_type() != FIFFT_CH_INFO_STRUCT || this->size() < static_cast<std::size_t>(FiffChInfo::storage_size()))
764 return p_FiffChInfo;
765 else {
766 auto* t_pInt32 = reinterpret_cast<const std::int32_t*>(this->data());
767 p_FiffChInfo.scanNo = t_pInt32[0];
768 p_FiffChInfo.logNo = t_pInt32[1];
769 p_FiffChInfo.kind = t_pInt32[2];
770 auto* t_pFloat = reinterpret_cast<const float*>(this->data());
771 p_FiffChInfo.range = t_pFloat[3];
772 p_FiffChInfo.cal = t_pFloat[4];
773
774 p_FiffChInfo.chpos.coil_type = t_pInt32[5];
775
776 //
777 // Read the coil coordinate system definition
778 //
779 std::int32_t r;
780 // r0
781 for (r = 0; r < 3; ++r)
782 p_FiffChInfo.chpos.r0[r] = t_pFloat[6 + r];
783 // ex
784 for (r = 0; r < 3; ++r)
785 p_FiffChInfo.chpos.ex[r] = t_pFloat[6 + 3 + r];
786 // ey
787 for (r = 0; r < 3; ++r)
788 p_FiffChInfo.chpos.ey[r] = t_pFloat[6 + 6 + r];
789 // ez
790 for (r = 0; r < 3; ++r)
791 p_FiffChInfo.chpos.ez[r] = t_pFloat[6 + 9 + r];
792
793 p_FiffChInfo.coord_frame = FIFFV_COORD_UNKNOWN;
794
795 //
796 // Convert loc into a more useful format
797 //
798 if (p_FiffChInfo.kind == FIFFV_MEG_CH || p_FiffChInfo.kind == FIFFV_REF_MEG_CH) {
799 p_FiffChInfo.coil_trans.setIdentity(4, 4);
800 // r0
801 for (r = 0; r < 3; ++r)
802 p_FiffChInfo.coil_trans(r, 3) = p_FiffChInfo.chpos.r0[r];
803 // ex
804 for (r = 0; r < 3; ++r)
805 p_FiffChInfo.coil_trans(r, 0) = p_FiffChInfo.chpos.ex[r];
806 // ey
807 for (r = 0; r < 3; ++r)
808 p_FiffChInfo.coil_trans(r, 1) = p_FiffChInfo.chpos.ey[r];
809 // ez
810 for (r = 0; r < 3; ++r)
811 p_FiffChInfo.coil_trans(r, 2) = p_FiffChInfo.chpos.ez[r];
812
813 p_FiffChInfo.coord_frame = FIFFV_COORD_DEVICE;
814 } else if (p_FiffChInfo.kind == FIFFV_EEG_CH) {
815 if (p_FiffChInfo.chpos.ex.norm() > 0) {
816 p_FiffChInfo.eeg_loc.block(0, 0, 3, 1) = p_FiffChInfo.chpos.r0.block(0, 0, 3, 1);
817 p_FiffChInfo.eeg_loc.block(0, 1, 3, 1) = p_FiffChInfo.chpos.ex.block(0, 0, 3, 1);
818 } else {
819 p_FiffChInfo.eeg_loc.block(0, 0, 3, 1) = p_FiffChInfo.chpos.r0.block(0, 0, 3, 1);
820 }
821 p_FiffChInfo.coord_frame = FIFFV_COORD_HEAD;
822 }
823 //
824 // Unit and exponent
825 //
826 p_FiffChInfo.unit = t_pInt32[18];
827 p_FiffChInfo.unit_mul = t_pInt32[19];
828
829 //
830 // Handle the channel name
831 //
832 const auto* name = reinterpret_cast<const char*>(data()) + 80;
833 p_FiffChInfo.ch_name.assign(name, std::find(name, name + 16, '\0'));
834
835 return p_FiffChInfo;
836 }
837}

841inline std::vector<FiffDirEntry> FiffTag::to_dir_entry() const
842{
843 std::vector<FiffDirEntry> entries;
844 if (this->is_matrix() || this->data_type() != FIFFT_DIR_ENTRY_STRUCT) {
845 return entries;
846 }
847 const auto* ints = reinterpret_cast<const std::int32_t*>(this->data());
848 const auto count = size() / static_cast<std::size_t>(FiffDirEntry::storage_size());
849 entries.reserve(count);
850 for (std::size_t k = 0; k < count; ++k) {
851 entries.push_back({ints[k * 4], ints[k * 4 + 1], ints[k * 4 + 2], ints[k * 4 + 3]});
852 }
853 return entries;
854}
