47in_addr default_interface() noexcept
48{
49 in_addr any{};
50 any.s_addr = htonl(INADDR_ANY);
51 const Native probe = ::socket(AF_INET, SOCK_DGRAM, 0);
52 if (probe == kInvalid) {
53 return any;
54 }
55 sockaddr_in to{};
56 to.sin_family = AF_INET;
57 to.sin_port = htons(53);
58 ::inet_pton(AF_INET, "8.8.8.8", &to.sin_addr);
59 sockaddr_in local{};
60 socklen_t length = sizeof(local);
61 // Connecting a UDP socket sends nothing; it only picks the route
62 const bool routed = ::connect(probe, reinterpret_cast<const sockaddr*>(&to), sizeof(to)) == 0 &&
63 ::getsockname(probe, reinterpret_cast<sockaddr*>(&local), &length) == 0;
64 close_native(probe);
65 return routed ? local.sin_addr : any;
66}

103std::expected<UdpSocket, std::error_code> UdpSocket::bind(const std::string& address, std::uint16_t port, bool shared)
104{
105 if (!started()) {
106 return std::unexpected(std::make_error_code(std::errc::not_supported));
107 }
108 sockaddr_in local{};
109 local.sin_family = AF_INET;
110 local.sin_port = htons(port);
111 if (::inet_pton(AF_INET, address.c_str(), &local.sin_addr) != 1) {
112 return std::unexpected(std::make_error_code(std::errc::invalid_argument));
113 }
114 const Native s = ::socket(AF_INET, SOCK_DGRAM, 0);
115 if (s == kInvalid) {
116 return std::unexpected(last_error());
117 }
118 UdpSocket socket;
119 socket.m_socket = static_cast<Handle>(s);
120 int size = 4 << 20; // a burst of datagrams must not overflow while the reader waits
121 ::setsockopt(s, SOL_SOCKET, SO_RCVBUF, reinterpret_cast<const char*>(&size), sizeof(size));
122 const int on = 1;
123 ::setsockopt(s, SOL_SOCKET, SO_BROADCAST, reinterpret_cast<const char*>(&on), sizeof(on));
124 ip_mreq membership{};
125 const bool group = is_multicast(local.sin_addr);
126 if (group || shared) {
127 ::setsockopt(s, SOL_SOCKET, SO_REUSEADDR, reinterpret_cast<const char*>(&on), sizeof(on));
128#ifdef SO_REUSEPORT
129 ::setsockopt(s, SOL_SOCKET, SO_REUSEPORT, &on, sizeof(on)); // BSD shares a wildcard-bound port only so
130#endif
131 }
132 if (group) {
133 membership.imr_multiaddr = local.sin_addr;
134 membership.imr_interface = default_interface();
135 local.sin_addr.s_addr = htonl(INADDR_ANY);
136 }
137 if (::bind(s, reinterpret_cast<const sockaddr*>(&local), sizeof(local)) != 0) {
138 return std::unexpected(last_error());
139 }
140 if (group) {
141 // Senders that leave the interface to the system use its multicast route, others the default route
142 // (BrainFlow's streamer the former, its receiver the latter); both differ behind a VPN
143 bool joined = false;
144 std::error_code error;
145 for (const std::uint32_t network_interface : {htonl(INADDR_ANY), membership.imr_interface.s_addr}) {
146 membership.imr_interface.s_addr = network_interface;
147 if (::setsockopt(s, IPPROTO_IP, IP_ADD_MEMBERSHIP, reinterpret_cast<const char*>(&membership), sizeof(membership)) == 0) {
148 joined = true;
149 } else {
150 error = last_error();
151 }
152 }
153 if (!joined) {
154 return std::unexpected(error);
155 }
156 }
157 return socket;
158}

174std::expected<std::size_t, std::error_code> UdpSocket::receive(std::span<std::byte> buffer, std::chrono::milliseconds timeout)
175{
176 const auto s = static_cast<Native>(m_socket);
177 const int ready = wait_for(s, POLLIN, timeout);
178 if (ready == 0) {
179 return std::unexpected(std::make_error_code(std::errc::timed_out));
180 }
181 if (ready < 0) {
182 return std::unexpected(last_error());
183 }
184#ifdef _WIN32
185 const int got = ::recv(s, reinterpret_cast<char*>(buffer.data()), static_cast<int>(buffer.size()), 0);
186 if (got < 0 && WSAGetLastError() == WSAEMSGSIZE) {
187 return buffer.size();
188 }
189#else
190 const auto got = ::recv(s, buffer.data(), buffer.size(), 0);
191#endif
192 if (got < 0) {
193 return std::unexpected(last_error());
194 }
195 return static_cast<std::size_t>(got);
196}

200std::error_code UdpSocket::send_to(std::span<const std::byte> datagram, const std::string& address, std::uint16_t port)
201{
202 sockaddr_in to{};
203 to.sin_family = AF_INET;
204 to.sin_port = htons(port);
205 if (::inet_pton(AF_INET, address.c_str(), &to.sin_addr) != 1) {
206 return std::make_error_code(std::errc::invalid_argument);
207 }
208 const auto s = static_cast<Native>(m_socket);
209 if (is_multicast(to.sin_addr) && !m_multicast_out) {
210 const in_addr out = default_interface();
211#ifdef _WIN32
212 const DWORD loop = 1;
213#else
214 const unsigned char loop = 1;
215#endif
216 ::setsockopt(s, IPPROTO_IP, IP_MULTICAST_IF, reinterpret_cast<const char*>(&out), sizeof(out));
217 ::setsockopt(s, IPPROTO_IP, IP_MULTICAST_LOOP, reinterpret_cast<const char*>(&loop), sizeof(loop));
218 m_multicast_out = true;
219 }
220#ifdef _WIN32
221 const int sent = ::sendto(s, reinterpret_cast<const char*>(datagram.data()), static_cast<int>(datagram.size()), 0,
222 reinterpret_cast<const sockaddr*>(&to), sizeof(to));
223#else
224 const auto sent = ::sendto(s, datagram.data(), datagram.size(), 0, reinterpret_cast<const sockaddr*>(&to), sizeof(to));
225#endif
226 if (sent < 0) {
227 return last_error();
228 }
229 return {};
230}

234std::error_code UdpSocket::join_group(const std::string& group, const std::string& interface_address)
235{
236 ip_mreq membership{};
237 membership.imr_interface.s_addr = htonl(INADDR_ANY);
238 if (::inet_pton(AF_INET, group.c_str(), &membership.imr_multiaddr) != 1 || !is_multicast(membership.imr_multiaddr) ||
239 (!interface_address.empty() && ::inet_pton(AF_INET, interface_address.c_str(), &membership.imr_interface) != 1)) {
240 return std::make_error_code(std::errc::invalid_argument);
241 }
242 if (::setsockopt(static_cast<Native>(m_socket), IPPROTO_IP, IP_ADD_MEMBERSHIP, reinterpret_cast<const char*>(&membership), sizeof(membership)) != 0) {
243 return last_error();
244 }
245 return {};
246}

272std::expected<UdpDatagram, std::error_code> UdpSocket::receive_from(std::span<std::byte> buffer, std::chrono::milliseconds timeout)
273{
274 const auto s = static_cast<Native>(m_socket);
275 const int ready = wait_for(s, POLLIN, timeout);
276 if (ready == 0) {
277 return std::unexpected(std::make_error_code(std::errc::timed_out));
278 }
279 if (ready < 0) {
280 return std::unexpected(last_error());
281 }
282 sockaddr_in from{};
283 socklen_t length = sizeof(from);
284 UdpDatagram datagram;
285#ifdef _WIN32
286 const int got = ::recvfrom(s, reinterpret_cast<char*>(buffer.data()), static_cast<int>(buffer.size()), 0, reinterpret_cast<sockaddr*>(&from), &length);
287 if (got < 0 && WSAGetLastError() == WSAEMSGSIZE) {
288 datagram.size = buffer.size();
289 } else
290#else
291 const auto got = ::recvfrom(s, buffer.data(), buffer.size(), 0, reinterpret_cast<sockaddr*>(&from), &length);
292#endif
293 {
294 if (got < 0) {
295 return std::unexpected(last_error());
296 }
297 datagram.size = static_cast<std::size_t>(got);
298 }
299 char text[INET_ADDRSTRLEN] = {};
300 if (::inet_ntop(AF_INET, &from.sin_addr, text, sizeof(text)) != nullptr) {
301 datagram.address = text;
302 }
303 datagram.port = ntohs(from.sin_port);
304 return datagram;
305}
