125bool FiffId::get_machid(int* fixed_id)
126{
127 fixed_id[0] = 0;
128 fixed_id[1] = 0;
129 // The first running interface that is neither loopback nor point-to-point, three bytes of its address per word
130 const std::vector<HardwareAddress> addresses = hardware_addresses();
131 if (addresses.empty())
132 return false;
133 const HardwareAddress& mac = addresses.front();
134 fixed_id[0] = (mac[0] << 16) | (mac[1] << 8) | mac[2];
135 fixed_id[1] = (mac[3] << 16) | (mac[4] << 8) | mac[5];
136 return true;
137}

150std::string FiffId::toMachidString() const
151{
152 // Two 32-bit words as 16 upper-case hex digits, colon after every pair: XX:XX:...:XX
153 const std::string hex = std::format("{:08X}{:08X}", static_cast<std::uint32_t>(machid[0]), static_cast<std::uint32_t>(machid[1]));
154 std::string out;
155 out.reserve(hex.size() * 3 / 2);
156 for (std::size_t i = 0; i < hex.size(); i += 2) {
157 if (i > 0) {
158 out += ':';
159 }
160 out.append(hex, i, 2);
161 }
162 return out;
163}

175std::string FiffId::to_string() const
176{
177 time_t secs = time.secs;
178
179 // localtime() returns a pointer to a shared static buffer, so two threads
180 // formatting an id at once can corrupt each other's result. Use the
181 // reentrant variant, which writes into a caller-supplied struct. MSVC
182 // spells it localtime_s and reverses the argument order.
183 struct tm ltime{};
184#ifdef _WIN32
185 const bool bTimeValid = (localtime_s(&ltime, &secs) == 0);
186#else
187 const bool bTimeValid = (localtime_r(&secs, &ltime) != nullptr);
188#endif
189
190 char timebuf[100];
191 if (!bTimeValid || strftime(timebuf, sizeof(timebuf), "%c", &ltime) == 0) {
192 timebuf[0] = '\0';
193 }
194
195 return std::format("{}.{} 0x{:x}{:x} {}", version >> 16, version & 0xFFFF, static_cast<std::uint32_t>(machid[0]),
196 static_cast<std::uint32_t>(machid[1]), timebuf);
197}
