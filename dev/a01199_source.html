61FiffInfo MaxwellMovementComp::transformFiffInfo(const FiffInfo& fiffInfo,
62 const HeadPosEntry& headPosRef,
63 const HeadPosEntry& headPosCurrent)
64{
65 FiffInfo info(fiffInfo);
66
67 // Compute relative rotation: from current → ref
68 // R_rel = R_ref * R_current^-1
69 Matrix3d rotRef = headPosRef.rotation.toRotationMatrix();
70 Matrix3d rotCur = headPosCurrent.rotation.toRotationMatrix();
71 Matrix3d rotRel = rotRef * rotCur.transpose();
72
73 // Translation: t_rel = t_ref - R_rel * t_current
74 Vector3d tRel = headPosRef.translation - rotRel * headPosCurrent.translation;
75
76 // Apply transform to MEG channel positions
77 for (int i = 0; i < info.chs.size(); ++i) {
78 if (info.chs[i].kind == FIFFV_MEG_CH || info.chs[i].kind == FIFFV_REF_MEG_CH) {
79 // Transform position
80 Vector3f r0 = info.chs[i].chpos.r0;
81 Vector3d r0d = r0.cast<double>();
82 r0d = rotRel * r0d + tRel;
83 info.chs[i].chpos.r0 = r0d.cast<float>();
84
85 // Transform orientation vectors
86 Vector3f ex = info.chs[i].chpos.ex;
87 Vector3d exd = ex.cast<double>();
88 exd = rotRel * exd;
89 info.chs[i].chpos.ex = exd.cast<float>();
90
91 Vector3f ey = info.chs[i].chpos.ey;
92 Vector3d eyd = ey.cast<double>();
93 eyd = rotRel * eyd;
94 info.chs[i].chpos.ey = eyd.cast<float>();
95
96 Vector3f ez = info.chs[i].chpos.ez;
97 Vector3d ezd = ez.cast<double>();
98 ezd = rotRel * ezd;
99 info.chs[i].chpos.ez = ezd.cast<float>();
100 }
101 }
102
103 return info;
104}

108MatrixXd MaxwellMovementComp::apply(const MatrixXd& matData,
109 const FiffInfo& fiffInfo,
110 const QList<HeadPosEntry>& headPos,
111 double dSFreq,
112 const MaxwellMoveCompParams& params)
113{
114 if (headPos.isEmpty()) {
115 qWarning() << "[MaxwellMovementComp::apply] No head positions provided.";
116 return matData;
117 }
118
119 const int nChannels = matData.rows();
120 const int nSamples = matData.cols();
121
122 // Determine reference head position
123 HeadPosEntry refPos;
124 if (params.iRefIdx >= 0 && params.iRefIdx < headPos.size()) {
125 refPos = headPos[params.iRefIdx];
126 } else {
127 // Mean position
128 Vector3d meanTrans = Vector3d::Zero();
129 for (const auto& hp : headPos) {
130 meanTrans += hp.translation;
131 }
132 meanTrans /= headPos.size();
133 refPos.translation = meanTrans;
134 refPos.rotation = headPos[0].rotation; // Use first rotation as reference
135 }
136
137 // Compute reference SSS basis
138 SSSParams sssParams;
139 sssParams.iOrderIn = params.iOrderIn;
140 sssParams.iOrderOut = params.iOrderOut;
141 sssParams.origin = params.origin;
142 sssParams.dRegIn = params.dRegIn;
143
144 SSS::Basis basisRef = SSS::computeBasis(fiffInfo, sssParams);
145
146 MatrixXd result = matData;
147
148 // Process each time segment
149 for (int iSeg = 0; iSeg < headPos.size(); ++iSeg) {
150 // Determine sample range for this segment
151 int iStart = static_cast<int>(headPos[iSeg].dTime * dSFreq);
152 int iEnd;
153 if (iSeg + 1 < headPos.size()) {
154 iEnd = static_cast<int>(headPos[iSeg + 1].dTime * dSFreq);
155 } else {
156 iEnd = nSamples;
157 }
158
159 iStart = qBound(0, iStart, nSamples);
160 iEnd = qBound(iStart, iEnd, nSamples);
161 if (iStart >= iEnd) continue;
162
163 int nSegSamples = iEnd - iStart;
164
165 // Transform FiffInfo to current head position
166 FiffInfo infoCurrentToRef = transformFiffInfo(fiffInfo, refPos, headPos[iSeg]);
167
168 // Compute SSS basis at current head position (as seen from reference)
169 SSS::Basis basisCurrent = SSS::computeBasis(infoCurrentToRef, sssParams);
170
171 // Extract segment data
172 MatrixXd segData = matData.middleCols(iStart, nSegSamples);
173
174 // Project into multipole space using current-position basis
175 // multipoles = pinv_current * data_meg
176 MatrixXd megData(basisCurrent.megChannelIdx.size(), nSegSamples);
177 for (int ch = 0; ch < basisCurrent.megChannelIdx.size(); ++ch) {
178 megData.row(ch) = segData.row(basisCurrent.megChannelIdx[ch]);
179 }
180
181 MatrixXd multipoles = basisCurrent.matPinvAll * megData;
182
183 // Reconstruct at reference position using only internal multipoles
184 MatrixXd reconstructed = basisRef.matSin * multipoles.topRows(basisRef.iNin);
185
186 // Place reconstructed data back
187 for (int ch = 0; ch < basisRef.megChannelIdx.size(); ++ch) {
188 result.row(basisRef.megChannelIdx[ch]).segment(iStart, nSegSamples) = reconstructed.row(ch);
189 }
190 }
191
192 return result;
193}

197QList<HeadPosEntry> MaxwellMovementComp::readHeadPos(const QString& sPath)
198{
199 QList<HeadPosEntry> positions;
200
201 QFile file(sPath);
202 if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
203 qWarning() << "[MaxwellMovementComp::readHeadPos] Cannot open file:" << sPath;
204 return positions;
205 }
206
207 QTextStream in(&file);
208 while (!in.atEnd()) {
209 QString line = in.readLine().trimmed();
210 if (line.isEmpty() || line.startsWith('#')) continue;
211
212 QStringList parts = line.split(QRegularExpression("\\s+"), Qt::SkipEmptyParts);
213 if (parts.size() < 7) continue;
214
215 bool ok = false;
216 HeadPosEntry entry;
217 entry.dTime = parts[0].toDouble(&ok); if (!ok) continue;
218
219 // Quaternion (scalar-last or scalar-first depends on convention)
220 double q1 = parts[1].toDouble(&ok); if (!ok) continue;
221 double q2 = parts[2].toDouble(&ok); if (!ok) continue;
222 double q3 = parts[3].toDouble(&ok); if (!ok) continue;
223
224 double tx = parts[4].toDouble(&ok); if (!ok) continue;
225 double ty = parts[5].toDouble(&ok); if (!ok) continue;
226 double tz = parts[6].toDouble(&ok); if (!ok) continue;
227
228 if (parts.size() >= 8) {
229 entry.dGof = parts[7].toDouble();
230 }
231
232 // MNE convention: quaternion stored as (q1, q2, q3) with q0 computed
233 double q0sq = 1.0 - q1*q1 - q2*q2 - q3*q3;
234 double q0 = (q0sq > 0.0) ? std::sqrt(q0sq) : 0.0;
235 entry.rotation = Quaterniond(q0, q1, q2, q3);
236 entry.translation = Vector3d(tx, ty, tz);
237
238 positions.append(entry);
239 }
240
241 file.close();
242 return positions;
243}

247bool MaxwellMovementComp::writeHeadPos(const QString& sPath,
248 const QList<HeadPosEntry>& headPos)
249{
250 QFile file(sPath);
251 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
252 qWarning() << "[MaxwellMovementComp::writeHeadPos] Cannot open file:" << sPath;
253 return false;
254 }
255
256 QTextStream out(&file);
257 out << "# Head position file\n";
258 out << "# time q1 q2 q3 tx ty tz gof\n";
259
260 for (const auto& entry : headPos) {
261 // Store quaternion as (q1, q2, q3), dropping q0 (recoverable)
262 out << QString::number(entry.dTime, 'f', 6) << " "
263 << QString::number(entry.rotation.x(), 'f', 6) << " "
264 << QString::number(entry.rotation.y(), 'f', 6) << " "
265 << QString::number(entry.rotation.z(), 'f', 6) << " "
266 << QString::number(entry.translation.x(), 'f', 6) << " "
267 << QString::number(entry.translation.y(), 'f', 6) << " "
268 << QString::number(entry.translation.z(), 'f', 6) << " "
269 << QString::number(entry.dGof, 'f', 6) << "\n";
270 }
271
272 file.close();
273 return true;
274}
