84 const auto option = [](const char* key, SettingField::Kind kind, const char* label, SettingValue value, const char* group,
85 const char* shownWith) {
86 SettingField field;
87 field.key = key;
88 field.kind = kind;
89 field.label = label;
90 field.default_value = std::move(value);
91 field.group = group;
92 field.visible_if = shownWith;
93 field.persistent = false;
94 return field;
95 };
96 f.add(option(kFullScreen, SettingField::Kind::Bool, "Show full screen", false, kViewGroup, kShowView));
97 f.add(option(kRotate, SettingField::Kind::Bool, "Automated rotation", false, kViewGroup, kShowView));
98 f.add(option(kCoordAxis, SettingField::Kind::Bool, "Show coord. axis", false, kViewGroup, kShowView));
99 SettingField viewMode = option(kViewMode, SettingField::Kind::Choice, "View:", std::string(kSingle), kViewGroup, kShowView);
100 viewMode.choices = {{kSingle, "Single View"}, {kMulti, "Multiview"}};
101 f.add(std::move(viewMode));
102 f.add(option(kSceneColor, SettingField::Kind::Color, "Change scene color:", std::string(kBlack), kViewGroup, kShowView));
103 f.add(option(kScreenshot, SettingField::Kind::Action, "Take screenshot", false, kViewGroup, kShowView));
104
105 f.add(option(kLightColor, SettingField::Kind::Color, "Change light color:", std::string(kWhite), kLightGroup, kShowLight));
106 SettingField intensity = option(kLightIntensity, SettingField::Kind::Double, "Change light intensity:", 0.0, kLightGroup, kShowLight);
107 intensity.minimum = 0.0;
108 intensity.maximum = 99.99;
109 intensity.step = 0.01;
110 intensity.decimals = 3;
111 f.add(std::move(intensity));
112
113 m_listener = f.add_listener([this](const std::string& key, const std::optional<SettingValue>& value) {
114 if (!m_updating && (value || key == kScreenshot)) {
115 changed(key);
116 }
117 });

266void Control3DPanel::changed(const std::string& key)
267{
268 if (key == kSceneColor && m_callbacks.scene_color_changed) {
269 m_callbacks.scene_color_changed(scene_color());
270 } else if (key == kFullScreen && m_callbacks.full_screen_changed) {
271 m_callbacks.full_screen_changed(full_screen());
272 } else if (key == kRotate && m_callbacks.rotation_changed) {
273 m_callbacks.rotation_changed(rotation());
274 } else if (key == kCoordAxis && m_callbacks.coord_axis_changed) {
275 m_callbacks.coord_axis_changed(coord_axis());
276 } else if (key == kLightColor && m_callbacks.light_color_changed) {
277 m_callbacks.light_color_changed(light_color());
278 } else if (key == kLightIntensity && m_callbacks.light_intensity_changed) {
279 m_callbacks.light_intensity_changed(light_intensity());
280 } else if (key == kScreenshot && m_callbacks.screenshot_wanted) {
281 m_callbacks.screenshot_wanted();
282 } else if (key == kViewMode) {
283 if (multi_view() && m_callbacks.multi_view_wanted) {
284 m_callbacks.multi_view_wanted();
285 } else if (!multi_view() && m_callbacks.single_view_wanted) {
286 m_callbacks.single_view_wanted();
287 }
288 }
