95AveragingPanel::AveragingPanel(const std::map<std::string, int>& stimChannels)
96: Panel("AveragingSettingsView", "Averaging Settings")
97, m_stimChannels(stimChannels)
98{
99 SettingsForm& f = form();

101 f.add_flag(kHasTrials);
102 constexpr const char* general = "General";
103 SettingField stimChannel;
104 stimChannel.key = kStimChannel;
105 stimChannel.kind = SettingField::Kind::Choice;
106 stimChannel.label = "Stimulus channel";
107 stimChannel.group = general;
108 stimChannel.visible_if = "!offline";
109 // The stored channel may be unknown until the stimulus channels arrive, so the panel keeps it aside
110 stimChannel.persistent = false;
111 f.add(std::move(stimChannel));
112 SettingField numAverages;
113 numAverages.key = kNumAverages;
114 numAverages.kind = SettingField::Kind::Int;
115 numAverages.label = "Number of trials";
116 numAverages.default_value = 10;
117 numAverages.minimum = 1;
118 numAverages.maximum = 1000;
119 numAverages.decimals = 0;
120 numAverages.group = general;
121 numAverages.visible_if = "!offline";
122 numAverages.commit = SettingField::Commit::WhenFinished;
123 f.add(std::move(numAverages));
124 f.add(milliseconds(kPreStim, "Pre-stimulus", 100, 0, general, "-"));
125 f.add(milliseconds(kPostStim, "Post-stimulus", 400, 10, general));
126 constexpr const char* baseline = "Baseline correction";
127 f.add_bool(kBaselineActive, "Activate", false, baseline);
128 f.add(milliseconds(kBaselineFrom, "Baseline min", 0, -kMostMilliseconds, baseline));
129 f.add(milliseconds(kBaselineTo, "Baseline max", 0, -kMostMilliseconds, baseline));
130 add_list(kTrials, {}, std::make_shared<ItemList>(), "Detected Trials");
131 SettingField trials = *f.field(kTrials);

133 f.remove(kTrials);
134 f.add(std::move(trials));
135 SettingField reset;
136 reset.key = kReset;
137 reset.kind = SettingField::Kind::Action;
138 reset.label = "Reset";
139 reset.visible_if = "!offline";
140 f.add(std::move(reset));
141 SettingField dropRejected;
142 dropRejected.key = kDropRejected;
143 dropRejected.kind = SettingField::Kind::Bool;
144 dropRejected.label = "Drop rejected";
145 dropRejected.default_value = false;
146 dropRejected.visible_if = kOffline;
147 dropRejected.persistent = false;
148 SettingField autoCompute = dropRejected;
149 autoCompute.key = kAutoCompute;
150 autoCompute.label = "Live average based on events";
151 f.add(std::move(dropRejected));
152 f.add(std::move(autoCompute));

161 if (!m_updating) {
162 changed(key);
163 }

165 const bool updating = std::exchange(m_updating, true);
166 f.set_range(kBaselineFrom, -pre_stim(), post_stim());
167 f.set_range(kBaselineTo, -pre_stim(), post_stim());
168 m_updating = updating;

170}

206void AveragingPanel::set_detected_epochs(const mne::FiffEvokedSet& evokedSet)
207{
208 form().set_value(kHasTrials, !evokedSet.evoked.empty());
209 if (evokedSet.evoked.empty()) {
210 return;
211 }
212 const std::size_t count = std::min<std::size_t>(evokedSet.evoked.size(), kMostShownAverages);
213 std::vector<std::string> items;
214 items.reserve(count);
215 for (std::size_t i = 0; i < count; ++i) {
216 items.push_back(evokedSet.evoked[i].comment + ": " + std::to_string(evokedSet.evoked[i].nave));
217 }
218 list(kTrials)->set_items(std::move(items));
219}

280void AveragingPanel::load(const SettingsStore& store, const std::string& path)
281{
282 if (path.empty()) {
283 return;
284 }
285 const bool updating = std::exchange(m_updating, true);
286 // The stored baseline is checked against the stored epoch window, not against the current one
287 form().set_range(kBaselineFrom, -kMostMilliseconds, kMostMilliseconds);
288 form().set_range(kBaselineTo, -kMostMilliseconds, kMostMilliseconds);
289 Panel::load(store, path);
290 const int preStim = pre_stim();
291 const int postStim = post_stim();
292 if (const int from = baseline_from(); from < -preStim || from > postStim) {
293 form().set_value(kBaselineFrom, -preStim);
294 }
295 if (const int to = baseline_to(); to > postStim || to < -preStim) {
296 form().set_value(kBaselineTo, 0);
297 }
298 form().set_range(kBaselineFrom, -preStim, postStim);
299 form().set_range(kBaselineTo, -preStim, postStim);
300 m_stimChannel = store.read_string(settings_key(path, kStimChannel), m_stimChannel);
301 m_updating = updating;
302 fill_stim_channels();
303}

325void AveragingPanel::changed(const std::string& key)
326{
327 if (key == kBaselineActive) {
328 report(m_callbacks.baseline_active_changed, baseline_correction());
329 } else if (key == kDropRejected) {
330 report(m_callbacks.drop_rejected_changed, form().get<bool>(kDropRejected));
331 } else if (key == kAutoCompute) {
332 report(m_callbacks.auto_compute_changed, auto_compute());
333 } else if (key == kReset) {
334 if (m_callbacks.reset) {
335 m_callbacks.reset();
336 }
337 } else if (key == kCompute) {
338 if (m_callbacks.compute) {
339 m_callbacks.compute();
340 }
341 } else if (key == kStimChannel) {
342 m_stimChannel = form().get<std::string>(kStimChannel);
343 if (m_callbacks.stim_channel_changed) {
344 m_callbacks.stim_channel_changed(m_stimChannel);
345 }
346 } else if (key == kNumAverages) {
347 report(m_callbacks.num_averages_changed, num_averages());
348 } else if (key == kPreStim || key == kPostStim || key == kBaselineFrom || key == kBaselineTo) {
349 // The epoch window bounds the baseline, the baseline bounds itself; clamped values report too
350 const SettingField* from = form().field(kBaselineFrom);
351 const SettingField* to = form().field(kBaselineTo);
352 const double fromMaximum = from->maximum;
353 const double toMinimum = to->minimum;
354 const double toMaximum = to->maximum;
355 const double fromMinimum = from->minimum;
356 if (key == kPreStim) {
357 form().set_range(kBaselineTo, -pre_stim(), toMaximum);
358 form().set_range(kBaselineFrom, -pre_stim(), fromMaximum);
359 } else if (key == kPostStim) {
360 form().set_range(kBaselineTo, toMinimum, post_stim());
361 form().set_range(kBaselineFrom, fromMinimum, post_stim());
362 } else if (key == kBaselineFrom) {
363 form().set_range(kBaselineTo, baseline_from(), toMaximum);
364 } else {
365 form().set_range(kBaselineFrom, fromMinimum, baseline_to());
366 }
367 const std::map<std::string, const std::function<void(int)>*> callbacks = {
368 {kPreStim, &m_callbacks.pre_stim_changed},
369 {kPostStim, &m_callbacks.post_stim_changed},
370 {kBaselineFrom, &m_callbacks.baseline_from_changed},
371 {kBaselineTo, &m_callbacks.baseline_to_changed}};
372 report(*callbacks.at(key), form().get<int>(key));
373 }
