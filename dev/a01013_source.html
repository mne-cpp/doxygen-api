52ModalitySelectionView::ModalitySelectionView(const std::vector<mne::FiffChInfo>& lChannelList,
53 const QString& sSettingsPath,
54 QWidget* parent,
55 Qt::WindowFlags f)
56: AbstractView(parent, f)
57{
58 m_sSettingsPath = sSettingsPath;
59 this->setWindowTitle("Modality Selection");
60 this->setMinimumWidth(330);
61 this->setMaximumWidth(330);
62
63 // Specify which channel types are needed
64 for (int i = 0; i < std::ssize(lChannelList); ++i) {
65 if (lChannelList.at(i).unit == FIFF_UNIT_T && !std::ranges::contains(m_lChannelTypeList, "MAG")) {
66 m_lChannelTypeList.push_back("MAG");
67 }
68 if (lChannelList.at(i).unit == FIFF_UNIT_T_M && !std::ranges::contains(m_lChannelTypeList, "GRAD")) {
69 m_lChannelTypeList.push_back("GRAD");
70 }
71 if (lChannelList.at(i).kind == FIFFV_EEG_CH && !std::ranges::contains(m_lChannelTypeList, "EEG")) {
72 m_lChannelTypeList.push_back("EEG");
73 }
74 if (lChannelList.at(i).kind == FIFFV_EOG_CH && !std::ranges::contains(m_lChannelTypeList, "EOG")) {
75 m_lChannelTypeList.push_back("EOG");
76 }
77 if (lChannelList.at(i).kind == FIFFV_STIM_CH && !std::ranges::contains(m_lChannelTypeList, "STIM")) {
78 m_lChannelTypeList.push_back("STIM");
79 }
80 if (lChannelList.at(i).kind == FIFFV_MISC_CH && !std::ranges::contains(m_lChannelTypeList, "MISC")) {
81 m_lChannelTypeList.push_back("MISC");
82 }
83 }

85 loadSettings();
86 redrawGUI();

197void ModalitySelectionView::redrawGUI()
198{
199 m_qListModalityCheckBox.clear();
200
201 //Delete all widgets in the averages layout
202 QGridLayout* topLayout = static_cast<QGridLayout*>(this->layout());
203 if (!topLayout) {
204 topLayout = new QGridLayout(this);
205 }
206
207 // The widgets go right away: the view is redrawn synchronously and looked up by name
208 detail::clearLayout(topLayout);
209
210 int count = 0;
211 for (const auto& [modality, checked] : m_modalityMap) {
212 const bool listed = std::ranges::any_of(m_lChannelTypeList, [&modality](const std::string& type) { return mne::iequals(type, modality); });
213 if (listed) {
214 QCheckBox* t_pCheckBoxModality = new QCheckBox(QString::fromStdString(modality), this);
215 t_pCheckBoxModality->setChecked(checked);
216 m_qListModalityCheckBox.push_back(t_pCheckBoxModality);
217 connect(t_pCheckBoxModality, &QCheckBox::checkStateChanged,
218 this, &ModalitySelectionView::onUpdateModalityCheckbox);
219 topLayout->addWidget(t_pCheckBoxModality, count, 0);
220 count++;
221 }
222 }
223}
