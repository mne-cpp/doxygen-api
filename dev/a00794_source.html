83QList<IcaLabelResult> MlIcaLabel::classify(const MatrixXd& matSources,
84 const MatrixXd& matEog,
85 const MatrixXd& matEcg,
86 double dSFreq,
87 double dEogThresh,
88 double dEcgThresh)
89{
90 QList<IcaLabelResult> results;
91
92 const int nComp = static_cast<int>(matSources.rows());
93 if (nComp == 0)
94 return results;
95
96 const double dMuscleThresh = 0.7;
97
98 for (int k = 0; k < nComp; ++k) {
99 IcaLabelResult res;
100 res.componentIndex = k;
101 res.label = IcaComponentLabel::Brain;
102 res.confidence = 0.0;
103
104 VectorXd source = matSources.row(k).transpose();
105
106 // Check EOG correlation
107 double eogCorr = 0.0;
108 if (matEog.rows() > 0 && matEog.cols() == matSources.cols())
109 eogCorr = maxAbsCorrelation(source, matEog);
110
111 // Check ECG correlation
112 double ecgCorr = 0.0;
113 if (matEcg.rows() > 0 && matEcg.cols() == matSources.cols())
114 ecgCorr = maxAbsCorrelation(source, matEcg);
115
116 // Check muscle score
117 double muscle = muscleScore(source, dSFreq);
118
119 // Classification logic: highest evidence wins
120 if (eogCorr >= dEogThresh && eogCorr >= ecgCorr && eogCorr >= muscle) {
121 res.label = IcaComponentLabel::Eog;
122 res.confidence = eogCorr;
123 } else if (ecgCorr >= dEcgThresh && ecgCorr >= eogCorr && ecgCorr >= muscle) {
124 res.label = IcaComponentLabel::Ecg;
125 res.confidence = ecgCorr;
126 } else if (muscle >= dMuscleThresh) {
127 res.label = IcaComponentLabel::Muscle;
128 res.confidence = muscle;
129 } else {
130 // Default to brain if no artifact criteria met
131 res.label = IcaComponentLabel::Brain;
132 res.confidence = 1.0 - std::max({eogCorr, ecgCorr, muscle});
133 }
134
135 results.append(res);
136 }
137
138 return results;
139}

158double MlIcaLabel::maxAbsCorrelation(const VectorXd& source, const MatrixXd& matRef)
159{
160 const int n = static_cast<int>(source.size());
161 if (n < 2)
162 return 0.0;
163
164 // Demean source
165 double srcMean = source.mean();
166 VectorXd srcCentered = source.array() - srcMean;
167 double srcStd = std::sqrt(srcCentered.squaredNorm() / static_cast<double>(n - 1));
168
169 if (srcStd < 1e-15)
170 return 0.0;
171
172 double maxCorr = 0.0;
173 for (int r = 0; r < matRef.rows(); ++r) {
174 VectorXd ref = matRef.row(r).transpose();
175 double refMean = ref.mean();
176 VectorXd refCentered = ref.array() - refMean;
177 double refStd = std::sqrt(refCentered.squaredNorm() / static_cast<double>(n - 1));
178
179 if (refStd < 1e-15)
180 continue;
181
182 double corr = std::abs(srcCentered.dot(refCentered) / (static_cast<double>(n - 1) * srcStd * refStd));
183 if (corr > maxCorr)
184 maxCorr = corr;
185 }
186
187 return maxCorr;
188}

192double MlIcaLabel::muscleScore(const VectorXd& source, double dSFreq)
193{
194 // Simple time-domain approximation of HF power ratio:
195 // Compute variance of first-differenced signal vs original.
196 // First-difference acts as a high-pass filter (accentuates high freq).
197 const int n = static_cast<int>(source.size());
198 if (n < 3 || dSFreq <= 0.0)
199 return 0.0;
200
201 double totalVar = 0.0;
202 double srcMean = source.mean();
203 for (int i = 0; i < n; ++i) {
204 double d = source[i] - srcMean;
205 totalVar += d * d;
206 }
207 totalVar /= static_cast<double>(n - 1);
208
209 if (totalVar < 1e-30)
210 return 0.0;
211
212 // Compute variance of second difference (approximates d²/dt²)
213 double hfVar = 0.0;
214 for (int i = 1; i < n - 1; ++i) {
215 double d2 = source[i + 1] - 2.0 * source[i] + source[i - 1];
216 hfVar += d2 * d2;
217 }
218 hfVar /= static_cast<double>(n - 2);
219
220 // Normalize: for white noise, hfVar/totalVar → 6.0 (analytical result for second diff of white noise)
221 // So ratio = hfVar / (6 * totalVar) gives ~1 for white noise, <1 for smooth signals
222 double ratio = hfVar / (6.0 * totalVar);
223
224 return std::min(ratio, 1.0);
225}
