154void ChannelLabelPanel::paintEvent(QPaintEvent *)
155{
156 QPainter p(this);
157 p.setRenderHint(QPainter::TextAntialiasing, true);
158
159 // Background — slightly more grey than the render surface
160 p.fillRect(rect(), QColor(242, 242, 242));
161
162 if (!m_model)
163 return;
164
165 int totalCh = m_channelIndices.isEmpty() ? m_model->channelCount()
166 : m_channelIndices.size();
167 int visibleCount = qMin(m_visibleChannelCount, totalCh - m_firstVisibleChannel);
168 if (visibleCount <= 0)
169 return;
170
171 const int pw = width();
172 const float fullLaneH = static_cast<float>(height()) / visibleCount;
173 // Collapsed height used for bad channels when m_hideBadChannels is true
174 constexpr float kCollapsedH = 6.f;
175
176 // Whether DC removal is active (shown as a pill in the status line)
177 const bool dcActive = m_model->removeDC();
178
179 // Font sizes proportional to lane height, clamped for readability
180 QFont nameFont = font();
181 nameFont.setPointSizeF(qBound(7.0, static_cast<double>(fullLaneH) * 0.22, 11.0));
182 nameFont.setBold(true);
183
184 QFont typeFont = nameFont;
185 typeFont.setPointSizeF(qBound(6.0, static_cast<double>(fullLaneH) * 0.15, 8.5));
186 typeFont.setBold(false);
187
188 auto resolveChannel = [this](int logicalIdx) -> int {
189 if (m_channelIndices.isEmpty()) return logicalIdx;
190 if (logicalIdx < 0 || logicalIdx >= m_channelIndices.size()) return -1;
191 return m_channelIndices[logicalIdx];
192 };
193
194 float yTop = 0.f;
195 for (int i = 0; i < visibleCount; ++i) {
196 int logIdx = m_firstVisibleChannel + i;
197 int ch = resolveChannel(logIdx);
198 if (ch < 0) { yTop += (m_hideBadChannels ? 6.f : (static_cast<float>(height()) / visibleCount)); continue; }
199 auto info = m_model->channelInfo(ch);
200
201 bool collapsed = m_hideBadChannels && info.bad;
202 float laneH = collapsed ? kCollapsedH : fullLaneH;
203 float yBot = yTop + laneH;
204
205 // Lane separator
206 p.setPen(QPen(QColor(200, 200, 200), 1));
207 if (i > 0)
208 p.drawLine(QPointF(0, yTop), QPointF(pw, yTop));
209
210 // Bad channel background (not when collapsed — it's just a thin strip then)
211 if (info.bad && !collapsed)
212 p.fillRect(QRectF(0, yTop, pw, laneH), QColor(255, 225, 225));
213
214 // Type-colour strip
215 QColor stripColor = (info.bad && collapsed) ? QColor(180, 180, 180) : info.color;
216 p.fillRect(QRectF(0, yTop, kStripWidth, laneH), stripColor);
217
218 if (!collapsed) {
219 // ── BAD badge (top-right corner) ─────────────────────────────────
220 if (info.bad) {
221 QRectF badRect(pw - kBadBadgeW - 2, yTop + 2, kBadBadgeW, kBadBadgeH);
222 p.fillRect(badRect, QColor(210, 30, 30));
223 QFont badFont = typeFont;
224 badFont.setPointSizeF(6.0);
225 badFont.setBold(true);
226 p.setFont(badFont);
227 p.setPen(Qt::white);
228 p.drawText(badRect, Qt::AlignCenter, QStringLiteral("BAD"));
229 }
230
231 // ── Channel name (bold, elided) ───────────────────────────────────
232 p.setFont(nameFont);
233 p.setPen(info.bad ? QColor(180, 20, 20) : QColor(25, 25, 25));
234 int nameBadgeGap = info.bad ? (kBadBadgeW + 4) : 4;
235 QRectF nameRect(kStripWidth + 4, yTop + 1,
236 pw - kStripWidth - nameBadgeGap - 4,
237 laneH * 0.48f);
238 QString elidedName = p.fontMetrics().elidedText(
239 info.name, Qt::ElideRight, static_cast<int>(nameRect.width()));
240 p.drawText(nameRect, Qt::AlignLeft | Qt::AlignVCenter, elidedName);
241
242 // ── Status line: type · bad · dc ─────────────────────────────────
243 // Build composite string, e.g. "MEG · bad · dc"
244 QString statusLine = info.typeLabel;
245 if (info.bad)
246 statusLine += QStringLiteral(" \u00B7 bad"); // · bad
247 if (dcActive)
248 statusLine += QStringLiteral(" \u00B7 dc"); // · dc
249
250 p.setFont(typeFont);
251 p.setPen(info.bad ? QColor(190, 60, 60) : QColor(110, 110, 120));
252 QRectF statusRect(kStripWidth + 4, yTop + laneH * 0.50f,
253 pw - kStripWidth - 8, laneH * 0.28f);
254 p.drawText(statusRect, Qt::AlignLeft | Qt::AlignVCenter, statusLine);
255
256 // ── RMS level bar ─────────────────────────────────────────────────
257 // Normalized signal level in [0, 1] relative to amplitudeMax
258 float rms = 0.f;
259 if (m_visSampleFirst < m_visSampleLast)
260 rms = m_model->channelRms(ch, m_visSampleFirst, m_visSampleLast);
261 float level = (info.amplitudeMax > 0.f)
262 ? qBound(0.f, rms / info.amplitudeMax, 1.f)
263 : 0.f;
264
265 const float barY = yTop + laneH * 0.80f;
266 const float barH = qMax(2.f, laneH * 0.14f);
267 const float barX0 = kStripWidth + 4.f;
268 const float barW = pw - kStripWidth - 8.f;
269
270 // Track
271 p.fillRect(QRectF(barX0, barY, barW, barH), QColor(215, 215, 215));
272
273 // Fill — colour green → amber → red by level
274 if (level > 0.f) {
275 QColor barColor;
276 if (level < 0.5f)
277 barColor = QColor(50, 180, 80);
278 else if (level < 0.85f)
279 barColor = QColor(220, 160, 30);
280 else
281 barColor = QColor(210, 50, 40);
282 if (info.bad)
283 barColor = barColor.darker(115);
284 p.fillRect(QRectF(barX0, barY, barW * level, barH), barColor);
285 }
286 }
287
288 yTop = yBot;
289 }
290}

306void ChannelLabelPanel::mouseMoveEvent(QMouseEvent *event)
307{
308 if (!m_dragging) {
309 event->ignore();
310 return;
311 }
312
313 const int totalCh = m_model ? (m_channelIndices.isEmpty() ? m_model->channelCount()
314 : m_channelIndices.size()) : 0;
315 const int maxFirst = qMax(0, totalCh - m_visibleChannelCount);
316 const float laneH = (m_visibleChannelCount > 0 && height() > 0)
317 ? static_cast<float>(height()) / m_visibleChannelCount
318 : 30.f;
319
320 int dy = event->position().toPoint().y() - m_dragStartY;
321 // Dragging DOWN means earlier channels (positive dy → lower first index)
322 int targetFirst = qBound(0,
323 m_dragStartFirst - static_cast<int>(dy / laneH),
324 maxFirst);
325
326 emit channelScrollRequested(targetFirst);
327 event->accept();
328}

342// Helper: return the row index (0-based within visibleCount) that contains pixel y,
343// accounting for collapsed bad-channel lanes.
344static int rowAtY(float y, int visibleCount, int firstCh,
345 ChannelDataModel *model, float fullLaneH, bool hideBad)
346{
347 constexpr float kCollapsedH = 6.f;
348 float top = 0.f;
349 for (int i = 0; i < visibleCount; ++i) {
350 int ch = firstCh + i;
351 auto info = model->channelInfo(ch);
352 float laneH = (hideBad && info.bad) ? kCollapsedH : fullLaneH;
353 if (y >= top && y < top + laneH)
354 return i;
355 top += laneH;
356 }
357 return -1;
358}

360bool ChannelLabelPanel::event(QEvent *e)
361{
362 if (e->type() == QEvent::ToolTip && m_model) {
363 auto *he = static_cast<QHelpEvent *>(e);
364 int totalCh = m_channelIndices.isEmpty() ? m_model->channelCount()
365 : m_channelIndices.size();
366 int visibleCount = qMin(m_visibleChannelCount, totalCh - m_firstVisibleChannel);
367 if (visibleCount > 0) {
368 float fullLaneH = static_cast<float>(height()) / visibleCount;
369 int i = rowAtY(static_cast<float>(he->pos().y()), visibleCount,
370 m_firstVisibleChannel, m_model, fullLaneH, m_hideBadChannels);
371 if (i >= 0) {
372 int logIdx = m_firstVisibleChannel + i;
373 int ch = m_channelIndices.isEmpty() ? logIdx
374 : (logIdx < m_channelIndices.size() ? m_channelIndices[logIdx] : -1);
375 if (ch < 0) return QWidget::event(e);
376 auto info = m_model->channelInfo(ch);
377 float rms = (m_visSampleFirst < m_visSampleLast)
378 ? m_model->channelRms(ch, m_visSampleFirst, m_visSampleLast)
379 : 0.f;
380 float level = (info.amplitudeMax > 0.f)
381 ? qBound(0.f, rms / info.amplitudeMax * 100.f, 100.f)
382 : 0.f;
383 QString tip = QStringLiteral("<b>%1</b><br>"
384 "Type: %2<br>"
385 "Scale: %3<br>"
386 "Level: %4 %<br>"
387 "Bad: %5")
388 .arg(info.name)
389 .arg(info.typeLabel)
390 .arg(info.amplitudeMax, 0, 'e', 2)
391 .arg(static_cast<int>(level))
392 .arg(info.bad ? QStringLiteral("yes") : QStringLiteral("no"));
393 QToolTip::showText(he->globalPos(), tip, this);
394 return true;
395 }
396 }
397 }
398 return QWidget::event(e);
399}
