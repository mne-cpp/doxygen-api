214int mne_ch_selection_assign_chs(mneChSelection sel,
215 MNERawData* data)
216/*
217 * Make the channel picking real easy
218 */
219{
220 int c,rc,d;
221 MNERawInfo* info;
222 int nch;
223 QString dash;
224
225 if (!sel || !data)
226 return 0;
227
228 info = data->info.get();
229 sel->chspick.clear();
230 sel->chspick_nospace.clear();
231 /*
232 * Expansion of possible regular expressions must be added eventually
233 */
234 sel->chspick = sel->chdef;
235 sel->chspick_nospace = sel->chdef;
236 omit_spaces(sel->chspick_nospace,sel->ndef);
237 sel->nchan = sel->ndef;
238
239 sel->pick.setConstant(sel->nchan, -1);
240 sel->pick_deriv.setConstant(sel->nchan, -1);
241 sel->ch_kind.setConstant(sel->nchan, -1);
242
243 for (c = 0; c < sel->nchan; c++) {
244 for (rc = 0; rc < info->nchan; rc++) {
245 if (QString::compare(sel->chspick[c],info->chInfo[rc].ch_name,Qt::CaseInsensitive) == 0 ||
246 QString::compare(sel->chspick_nospace[c],info->chInfo[rc].ch_name,Qt::CaseInsensitive) == 0) {
247 sel->pick[c] = rc;
248 sel->ch_kind[c] = info->chInfo[rc].kind;
249 break;
250 }
251 }
252 }
253 /*
254 * Maybe the derivations will help
255 */
256 sel->nderiv = 0;
257 if (data->deriv_matched) {
258 QStringList deriv_names = data->deriv_matched->deriv_data->rowlist;
259 int nderiv = data->deriv_matched->deriv_data->nrow;
260
261 for (c = 0; c < sel->nchan; c++) {
262 if (sel->pick[c] == -1) {
263 for (d = 0; d < nderiv; d++) {
264 if (QString::compare(sel->chspick[c],deriv_names[d],Qt::CaseInsensitive) == 0 &&
265 data->deriv_matched->valid.size() > 0 && data->deriv_matched->valid[d]) {
266 sel->pick_deriv[c] = d;
267 sel->ch_kind[c] = data->deriv_matched->chs[d].kind;
268 sel->nderiv++;
269 break;
270 }
271 }
272 }
273 }
274 }
275 /*
276 * Try simple channels again without the part after dashes
277 */
278 for (c = 0; c < sel->nchan; c++) {
279 if (sel->pick[c] == -1 && sel->pick_deriv[c] == -1) {
280 for (rc = 0; rc < info->nchan; rc++) {
281 dash = QString(info->chInfo[rc].ch_name).mid(QString(info->chInfo[rc].ch_name).indexOf("-")+1);// strchr(info->chInfo[rc].ch_name,'-');
282 if (!dash.isNull()) {
283// *dash = '\0';
284 if (QString::compare(sel->chspick[c],info->chInfo[rc].ch_name,Qt::CaseInsensitive) == 0 ||
285 QString::compare(sel->chspick_nospace[c],info->chInfo[rc].ch_name,Qt::CaseInsensitive) == 0) {
286// *dash = '-';
287 sel->pick[c] = rc;
288 sel->ch_kind[c] = info->chInfo[rc].kind;
289 break;
290 }
291// *dash = '-';
292 }
293 }
294 }
295 }
296 for (c = 0, nch = 0; c < sel->nchan; c++) {
297 if (sel->pick[c] >= 0)
298 nch++;
299 }
300 if (sel->nderiv > 0)
301 printf("Selection %c%s%c has %d matched derived channels.\n",'"',sel->name.toUtf8().constData(),'"',sel->nderiv);
302 return nch;

307int mne_get_values_from_data (float time, /* Interesting time point */
308 float integ, /* Time integration */
309 float **data, /* The data values (time by time) */
310 int nsamp, /* How many time points? */
311 int nch, /* How many channels */
312 float tmin, /* Time of first sample */
313 float sfreq, /* Sampling frequency */
314 int use_abs, /* Use absolute values */
315 float *value) /* The picked values */
316/*
317 * Pick a signal value using linear interpolation
318 */
319{
320 int n1,n2,k;
321 float s1,s2;
322 float f1,f2;
323 float sum;
324 float width;
325 int ch;
326
327 for (ch = 0; ch < nch; ch++) {
328 /*
329 * Find out the correct samples
330 */
331 if (std::fabs(sfreq*integ) < EPS_VALUES) { /* This is the single-sample case */
332 s1 = sfreq*(time - tmin);
333 n1 = floor(s1);
334 f1 = 1.0 + n1 - s1;
335 if (n1 < 0 || n1 > nsamp-1) {
336 printf("Sample value out of range %d (0..%d)",n1,nsamp-1);
337 return(-1);
338 }
339 /*
340 * Avoid rounding error
341 */
342 if (n1 == nsamp-1) {
343 if (std::fabs(f1-1.0) < 1e-3)
344 f1 = 1.0;
345 }
346 if (f1 < 1.0 && n1 > nsamp-2) {
347 printf("Sample value out of range %d (0..%d) %.4f",n1,nsamp-1,f1);
348 return(-1);
349 }
350 if (f1 < 1.0) {
351 if (use_abs)
352 sum = f1*std::fabs(data[n1][ch]) + (1.0-f1)*std::fabs(data[n1+1][ch]);
353 else
354 sum = f1*data[n1][ch] + (1.0-f1)*data[n1+1][ch];
355 }
356 else {
357 if (use_abs)
358 sum = std::fabs(data[n1][ch]);
359 else
360 sum = data[n1][ch];
361 }
362 }
363 else { /* Multiple samples */
364 s1 = sfreq*(time - 0.5*integ - tmin);
365 s2 = sfreq*(time + 0.5*integ - tmin);
366 n1 = ceil(s1); n2 = floor(s2);
367 if (n2 < n1) { /* We are within one sample interval */
368 n1 = floor(s1);
369 if (n1 < 0 || n1 > nsamp-2)
370 return (-1);
371 f1 = s1 - n1;
372 f2 = s2 - n1;
373 if (use_abs)
374 sum = 0.5*((f1+f2)*std::fabs(data[n1+1][ch]) + (2.0-f1-f2)*std::fabs(data[n1][ch]));
375 else
376 sum = 0.5*((f1+f2)*data[n1+1][ch] + (2.0-f1-f2)*data[n1][ch]);
377 }
378 else {
379 f1 = n1 - s1;
380 f2 = s2 - n2;
381 if (n1 < 0 || n1 > nsamp-1) {
382 printf("Sample value out of range %d (0..%d)",n1,nsamp-1);
383 return(-1);
384 }
385 if (n2 < 0 || n2 > nsamp-1) {
386 printf("Sample value out of range %d (0..%d)",n2,nsamp-1);
387 return(-1);
388 }
389 if (f1 != 0.0 && n1 < 1)
390 return(-1);
391 if (f2 != 0.0 && n2 > nsamp-2)
392 return(-1);
393 sum = 0.0;
394 width = 0.0;
395 if (n2 > n1) { /* Do the whole intervals */
396 if (use_abs) {
397 sum = 0.5 * std::fabs(data[n1][ch]);
398 for (k = n1+1; k < n2; k++)
399 sum = sum + std::fabs(data[k][ch]);
400 sum = sum + 0.5 * std::fabs(data[n2][ch]);
401 }
402 else {
403 sum = 0.5*data[n1][ch];
404 for (k = n1+1; k < n2; k++)
405 sum = sum + data[k][ch];
406 sum = sum + 0.5*data[n2][ch];
407 }
408 width = n2 - n1;
409 }
410 /*
411 * Add tails
412 */
413 if (use_abs) {
414 if (f1 != 0.0)
415 sum = sum + 0.5 * f1 * (f1 * std::fabs(data[n1-1][ch]) + (2.0 - f1) * std::fabs(data[n1][ch]));
416 if (f2 != 0.0)
417 sum = sum + 0.5 * f2 * (f2 * std::fabs(data[n2+1][ch]) + (2.0 - f2) * std::fabs(data[n2][ch]));
418 }
419 else {
420 if (f1 != 0.0)
421 sum = sum + 0.5*f1*(f1*data[n1-1][ch] + (2.0-f1)*data[n1][ch]);
422 if (f2 != 0.0)
423 sum = sum + 0.5*f2*(f2*data[n2+1][ch] + (2.0-f2)*data[n2][ch]);
424 }
425 width = width + f1 + f2;
426 sum = sum/width;
427 }
428 }
429 value[ch] = sum;
430 }
431 return (0);
432}

477 }
478 }
479 else { /* Multiple samples */

483 if (n2 < n1) { /* We are within one sample interval */
484 n1 = floor(s1);
485 if (n1 < 0 || n1 > nsamp-2)

492 sum = 0.5*((f1+f2)*data[ch][n1+1] + (2.0-f1-f2)*data[ch][n1]);
493 }
494 else {
495 f1 = n1 - s1;
496 f2 = s2 - n2;
497 if (n1 < 0 || n1 > nsamp-1 || n2 < 0 || n2 > nsamp-1)
498 return(-1);
499 if (f1 != 0.0 && n1 < 1)
500 return(-1);
501 if (f2 != 0.0 && n2 > nsamp-2)

510 sum = sum + 0.5 * std::fabs(data[ch][n2]);
511 }
512 else {
513 sum = 0.5*data[ch][n1];
514 for (k = n1+1; k < n2; k++)
515 sum = sum + data[ch][k];
516 sum = sum + 0.5*data[ch][n2];
517 }
518 width = n2 - n1;
519 }
520 /*
521 * Add tails
522 */

524 if (f1 != 0.0)

536 sum = sum/width;
537 }
538 }

540 }

727int DipoleFit::fit_dipoles_raw(const QString& dataname, MNERawData* raw, mneChSelection sel, DipoleFitData* fit, GuessData* guess, float tmin, float tmax, float tstep, float integ, int verbose, ECDSet& p_set)
728{
729 float *one = MALLOC(sel->nchan,float);
730 float sfreq = raw->info->sfreq;
731 float myinteg = integ > 0.0 ? 2*integ : 0.1;
732 int overlap = ceil(myinteg*sfreq);
733 int length = SEG_LEN*sfreq;
734 int step = length - overlap;
735 int stepo = step + overlap/2;
736 int start = raw->first_samp;
737 int s,picks;
738 float time,stime;
739 float **data = ALLOC_CMATRIX(sel->nchan,length);
740 ECD dip;
741 ECDSet set;
742 int report_interval = 10;
743
744 set.dataname = dataname;
745
746 /*
747 * Load the initial data segment
748 */
749 stime = start/sfreq;
750 if (raw->pick_data_filt(sel,start,length,data) == FAIL)
751 goto bad;
752 printf("Fitting...%c",verbose ? '\n' : '\0');
753 for (s = 0, time = tmin; time < tmax; s++, time = tmin + s*tstep) {
754 picks = time*sfreq - start;
755 if (picks > stepo) { /* Need a new data segment? */
756 start = start + step;
757 if (raw->pick_data_filt(sel,start,length,data) == FAIL)
758 goto bad;
759 picks = time*sfreq - start;
760 stime = start/sfreq;
761 }
762 /*
763 * Get the values
764 */
765 if (mne_get_values_from_data_ch (time,integ,data,length,sel->nchan,stime,sfreq,FALSE,one) == FAIL) {
766 printf("Cannot pick time: %8.3f s\n",time);
767 continue;
768 }
769 /*
770 * Fit
771 */
772 if (!DipoleFitData::fit_one(fit,guess,time,one,verbose,dip))
773 qWarning() << "Error";
774 else {
775 set.addEcd(dip);
776 if (verbose)
777 dip.print(stdout);
778 else {
779 if (set.size() % report_interval == 0)
780 printf("%d..",set.size());
781 }
782 }
783 }
784 if (!verbose)
785 printf("[done]\n");
786 FREE_CMATRIX(data);
787 FREE(one);
788 p_set = set;
789 return OK;
790
791bad : {
792 FREE_CMATRIX(data);
793 FREE(one);
794 return FAIL;
795 }
796}
