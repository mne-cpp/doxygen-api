89std::optional<VectorXd> PythonTestHelper::eval_vector(std::string_view code, std::chrono::milliseconds timeout) const
90{
91 const PythonRunnerResult result = eval(code, timeout);
92 if (!result.success) {
93 return std::nullopt;
94 }
95 // Values may be spread over lines of different length, so read them as one row
96 std::string flat = result.std_out;
97 std::ranges::replace(flat, '\n', ' ');
98 std::istringstream stream(flat);
99 RowVectorXd values;
100 if (!mne::read_matrix(values, stream) || values.size() == 0) {
101 return std::nullopt;
102 }
103 return values.transpose();
104}
