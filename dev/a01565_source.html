46class Encoder
47{
48public:
49 explicit Encoder(CborByteArray& out)
50 : m_out(out)
51 {
52 }
53
54 void head(unsigned major, std::uint64_t value)
55 {
56 const auto type = static_cast<std::byte>(major << 5);
57 if (value < 24) {
58 m_out.push_back(type | static_cast<std::byte>(value));
59 } else if (value <= 0xff) {
60 m_out.push_back(type | std::byte{24});
61 big_endian(value, 1);
62 } else if (value <= 0xffff) {
63 m_out.push_back(type | std::byte{25});
64 big_endian(value, 2);
65 } else if (value <= 0xffffffffULL) {
66 m_out.push_back(type | std::byte{26});
67 big_endian(value, 4);
68 } else {
69 m_out.push_back(type | std::byte{27});
70 big_endian(value, 8);
71 }
72 }
73
74 void text(std::string_view value)
75 {
76 head(3, value.size());
77 const auto* first = reinterpret_cast<const std::byte*>(value.data());
78 m_out.insert(m_out.end(), first, first + value.size());
79 }
80
81 void big_endian(std::uint64_t value, int bytes)
82 {
83 for (int shift = (bytes - 1) * 8; shift >= 0; shift -= 8) {
84 m_out.push_back(static_cast<std::byte>((value >> shift) & 0xff));
85 }
86 }
87
88 void value(const CborValue& v)
89 {
90 switch (v.type()) {
91 case CborValue::Integer: {
92 const std::int64_t i = v.to_integer();
93 if (i >= 0) {
94 head(0, static_cast<std::uint64_t>(i));
95 } else {
96 head(1, static_cast<std::uint64_t>(-1 - i));
97 }
98 break;
99 }
100 case CborValue::ByteArray: {
101 const CborByteArray& bytes = v.to_byte_array();
102 head(2, bytes.size());
103 m_out.insert(m_out.end(), bytes.begin(), bytes.end());
104 break;
105 }
106 case CborValue::String:
107 text(v.to_string());
108 break;
109 case CborValue::Array:
110 head(4, v.to_array().size());
111 for (const CborValue& element : v.to_array()) {
112 value(element);
113 }
114 break;
115 case CborValue::Map:
116 head(5, v.to_map().size());
117 for (const auto& [key, member] : v.to_map()) {
118 text(key);
119 value(member);
120 }
121 break;
122 case CborValue::Tag:
123 head(6, v.tag());
124 value(v.tagged_value());
125 break;
126 case CborValue::False:
127 m_out.push_back(std::byte{0xf4});
128 break;
129 case CborValue::True:
130 m_out.push_back(std::byte{0xf5});
131 break;
132 case CborValue::Null:
133 m_out.push_back(std::byte{0xf6});
134 break;
135 case CborValue::Undefined:
136 m_out.push_back(std::byte{0xf7});
137 break;
138 case CborValue::Double:
139 m_out.push_back(std::byte{0xfb});
140 big_endian(std::bit_cast<std::uint64_t>(v.to_double()), 8);
141 break;
142 }
143 }
144
145private:
146 CborByteArray& m_out;
147};

150class Decoder
151{
152public:
153 explicit Decoder(std::span<const std::byte> data)
154 : m_data(data)
155 {
156 }
157
158 CborParseError error;
159
160 std::size_t pos() const
161 {
162 return m_pos;
163 }
164
165 bool failed() const
166 {
167 return error.error != CborParseError::NoError;
168 }
169
170 std::size_t remaining() const
171 {
172 return m_data.size() - m_pos;
173 }
174
175 CborValue value(int depth)
176 {
177 const std::size_t start = m_pos;
178 if (remaining() == 0) {
179 return fail(CborParseError::EndOfFile, start);
180 }
181 const auto initial = std::to_integer<unsigned>(m_data[m_pos++]);
182 const unsigned major = initial >> 5;
183 const unsigned info = initial & 31;
184 if (info >= 28 && info <= 30) {
185 return fail(CborParseError::UnknownType, start);
186 }
187 if (info == 31 && (major <= 1 || major == 6)) {
188 return fail(CborParseError::IllegalNumber, start);
189 }
190 if (major == 7) {
191 return simple(info, start);
192 }
193 std::uint64_t argument = 0;
194 if (info != 31 && !read_argument(info, argument, start)) {
195 return {};
196 }
197 switch (major) {
198 case 0:
199 return argument <= static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) ? CborValue(static_cast<std::int64_t>(argument))
200 : CborValue(static_cast<double>(argument));
201 case 1:
202 return argument <= static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) ? CborValue(-1 - static_cast<std::int64_t>(argument))
203 : CborValue(-1.0 - static_cast<double>(argument));
204 case 2:
205 case 3:
206 return string(major, info == 31, argument, start);
207 case 4:
208 case 5:
209 case 6:
210 if (depth >= kMaxNesting) {
211 return fail(CborParseError::NestingTooDeep, start);
212 }
213 return major == 4 ? array(info == 31, argument, depth) : major == 5 ? map(info == 31, argument, depth)
214 : CborValue::tagged(argument, value(depth + 1));
215 default:
216 return {};
217 }
218 }
219
220private:
221 CborValue fail(CborParseError::Code code, std::size_t offset)
222 {
223 if (!failed()) {
224 error.error = code;
225 error.offset = offset;
226 }
227 return {};
228 }
229
230 bool read_argument(unsigned info, std::uint64_t& argument, std::size_t start)
231 {
232 if (info < 24) {
233 argument = info;
234 return true;
235 }
236 const std::size_t bytes = std::size_t{1} << (info - 24);
237 if (remaining() < bytes) {
238 fail(CborParseError::EndOfFile, start);
239 return false;
240 }
241 argument = 0;
242 for (std::size_t i = 0; i < bytes; ++i) {
243 argument = (argument << 8) | std::to_integer<std::uint64_t>(m_data[m_pos++]);
244 }
245 return true;
246 }
247
248 bool at_break()
249 {
250 if (remaining() > 0 && m_data[m_pos] == std::byte{0xff}) {
251 ++m_pos;
252 return true;
253 }
254 return false;
255 }
256
257 CborValue simple(unsigned info, std::size_t start)
258 {
259 switch (info) {
260 case 20:
261 return false;
262 case 21:
263 return true;
264 case 22:
265 return nullptr;
266 case 23:
267 return {};
268 case 24:
269 if (remaining() < 1) {
270 return fail(CborParseError::EndOfFile, start);
271 }
272 ++m_pos;
273 return fail(CborParseError::IllegalSimpleType, start);
274 case 25:
275 case 26:
276 case 27: {
277 std::uint64_t bits = 0;
278 if (!read_argument(info, bits, start)) {
279 return {};
280 }
281 if (info == 27) {
282 return std::bit_cast<double>(bits);
283 }
284 if (info == 26) {
285 return static_cast<double>(std::bit_cast<float>(static_cast<std::uint32_t>(bits)));
286 }
287 return half_to_double(static_cast<std::uint16_t>(bits));
288 }
289 case 31:
290 return fail(CborParseError::UnexpectedBreak, start);
291 default:
292 return fail(CborParseError::IllegalSimpleType, start);
293 }
294 }
295
296 static double half_to_double(std::uint16_t half)
297 {
298 const int exponent = (half >> 10) & 0x1f;
299 const int mantissa = half & 0x3ff;
300 double value = 0.0;
301 if (exponent == 0) {
302 value = std::ldexp(mantissa, -24);
303 } else if (exponent == 31) {
304 value = mantissa == 0 ? std::numeric_limits<double>::infinity() : std::numeric_limits<double>::quiet_NaN();
305 } else {
306 value = std::ldexp(mantissa + 1024, exponent - 25);
307 }
308 return (half & 0x8000) ? -value : value;
309 }
310
311 CborValue string(unsigned major, bool indefinite, std::uint64_t length, std::size_t start)
312 {
313 std::string bytes;
314 if (!indefinite) {
315 if (!append_chunk(bytes, length, start)) {
316 return {};
317 }
318 } else {
319 while (!at_break()) {
320 const std::size_t chunkStart = m_pos;
321 if (remaining() == 0) {
322 return fail(CborParseError::EndOfFile, chunkStart);
323 }
324 const auto initial = std::to_integer<unsigned>(m_data[m_pos++]);
325 const unsigned info = initial & 31;
326 if ((initial >> 5) != major || info == 31) {
327 return fail(CborParseError::IllegalType, chunkStart);
328 }
329 if (info >= 28) {
330 return fail(CborParseError::UnknownType, chunkStart);
331 }
332 std::uint64_t chunk = 0;
333 if (!read_argument(info, chunk, chunkStart) || !append_chunk(bytes, chunk, chunkStart)) {
334 return {};
335 }
336 }
337 }
338 if (major == 2) {
339 const auto* first = reinterpret_cast<const std::byte*>(bytes.data());
340 return CborByteArray(first, first + bytes.size());
341 }
342 if (!is_utf8(bytes)) {
343 return fail(CborParseError::InvalidUtf8String, start);
344 }
345 return bytes;
346 }
347
348 bool append_chunk(std::string& bytes, std::uint64_t length, std::size_t start)
349 {
350 if (length > remaining()) {
351 fail(CborParseError::DataTooLarge, start);
352 return false;
353 }
354 bytes.append(reinterpret_cast<const char*>(m_data.data() + m_pos), static_cast<std::size_t>(length));
355 m_pos += static_cast<std::size_t>(length);
356 return true;
357 }
358
359 CborValue array(bool indefinite, std::uint64_t count, int depth)
360 {
361 CborArray array;
362 if (indefinite) {
363 while (!failed() && !at_break()) {
364 array.append(value(depth + 1));
365 }
366 } else {
367 // Every element takes at least one byte
368 if (count > remaining()) {
369 return fail(CborParseError::DataTooLarge, m_pos);
370 }
371 array.reserve(static_cast<std::size_t>(count));
372 for (std::uint64_t i = 0; i < count && !failed(); ++i) {
373 array.append(value(depth + 1));
374 }
375 }
376 return failed() ? CborValue() : CborValue(std::move(array));
377 }
378
379 CborValue map(bool indefinite, std::uint64_t count, int depth)
380 {
381 CborMap map;
382 if (!indefinite) {
383 // Every key and every value takes at least one byte
384 if (count > remaining() / 2) {
385 return fail(CborParseError::DataTooLarge, m_pos);
386 }
387 map.reserve(static_cast<std::size_t>(count));
388 }
389 for (std::uint64_t i = 0; !failed() && (indefinite ? !at_break() : i < count); ++i) {
390 const std::size_t keyStart = m_pos;
391 const CborValue key = value(depth + 1);
392 std::string text;
393 if (key.is_string()) {
394 text = key.to_string();
395 } else if (key.is_integer()) {
396 text = std::to_string(key.to_integer());
397 } else if (!failed()) {
398 return fail(CborParseError::UnsupportedType, keyStart);
399 }
400 CborValue member = value(depth + 1);
401 if (!failed()) {
402 map.append(std::move(text), std::move(member));
403 }
404 }
405 return failed() ? CborValue() : CborValue(std::move(map));
406 }
407
408 std::span<const std::byte> m_data;
409 std::size_t m_pos = 0;
410};

418std::string_view CborParseError::error_string() const noexcept
419{
420 switch (error) {
421 case NoError:
422 return "No error";
423 case GarbageAtEnd:
424 return "Data found after the end of the stream";
425 case EndOfFile:
426 return "Unexpected end of input data";
427 case UnexpectedBreak:
428 return "Invalid CBOR stream: unexpected 'break' byte";
429 case UnknownType:
430 return "Invalid CBOR stream: unknown type";
431 case IllegalType:
432 return "Invalid CBOR stream: illegal type found";
433 case IllegalNumber:
434 return "Invalid CBOR stream: illegal number encoding (future extension)";
435 case IllegalSimpleType:
436 return "Invalid CBOR stream: illegal simple type";
437 case InvalidUtf8String:
438 return "Invalid CBOR stream: invalid UTF-8 text string";
439 case DataTooLarge:
440 return "Internal limitation: data set too large";
441 case NestingTooDeep:
442 return "Internal limitation: data nesting too deep";
443 case UnsupportedType:
444 return "Unsupported CBOR type: map key is neither text nor an integer";
445 }
446 return "Unknown error";
447}

872JsonValue CborValue::to_json_value() const
873{
874 switch (type()) {
875 case Integer:
876 return to_integer();
877 case ByteArray:
878 return to_base64(to_byte_array(), true);
879 case String:
880 return to_string();
881 case Array:
882 return to_array().to_json_array();
883 case Map:
884 return to_map().to_json_object();
885 case Tag: {
886 const CborValue& content = tagged_value();
887 if (content.is_byte_array() && (tag() == 22 || tag() == 23)) {
888 return tag() == 22 ? to_base64(content.to_byte_array()) : to_hex(content.to_byte_array());
889 }
890 return content.to_json_value();
891 }
892 case False:
893 return false;
894 case True:
895 return true;
896 case Null:
897 case Undefined:
898 return nullptr;
899 case Double:
900 return to_double();
901 }
902 return nullptr;
