116void FilterSettingsView::setSamplingRate(double dSFreq)
117{
118 //Update min max of spin boxes to nyquist
119 double nyquistFrequency = dSFreq/2;
120
121 m_pUi->m_pDoubleSpinBoxFrom->setMaximum(nyquistFrequency);
122 m_pUi->m_pDoubleSpinBoxTo->setMaximum(nyquistFrequency);
123
124 if(m_pUi->m_pDoubleSpinBoxFrom->value() > dSFreq/2) {
125 m_pUi->m_pDoubleSpinBoxFrom->setValue(dSFreq/2);
126 }
127
128 if(m_pUi->m_pDoubleSpinBoxTo->value() > dSFreq/2) {
129 m_pUi->m_pDoubleSpinBoxTo->setValue(dSFreq/2);
130 }
131
132 m_pFilterView->setSamplingRate(dSFreq);
133}
