50// Records of @p ring with a sample index in [from, to), starting at @p cursor. Counts as lost the records
51// overwritten before the scan and, if @p countSkipped, those skipped because their sample lies before @p from.
52template<typename T, typename SampleOf>
53RecordWindow selectRecords(const RecordRing<T>& ring, std::uint64_t cursor, std::uint64_t from, std::uint64_t to,
54 std::atomic<std::uint64_t>& lostCounter, bool countSkipped, SampleOf sampleOf) noexcept
55{
56 while (true) {
57 const std::uint64_t start = std::max(cursor, ring.oldest());
58 const std::uint64_t published = ring.published.load(std::memory_order_acquire);
59 RecordWindow window{start, start};
60 while (window.first < published && sampleOf(ring.records[window.first & ring.mask]) < from) {
61 ++window.first;
62 }
63 window.end = window.first;
64 while (window.end < published && sampleOf(ring.records[window.end & ring.mask]) < to) {
65 ++window.end;
66 }
67 // A slot the scan read may have been overwritten meanwhile, which can hide records; scan again.
68 std::atomic_thread_fence(std::memory_order_acquire);
69 if (ring.oldest() > start) {
70 continue;
71 }
72 const std::uint64_t lost = (start - cursor) + (countSkipped ? window.first - start : 0);
73 if (lost > 0) {
74 addOwned(lostCounter, lost);
75 }
76 return window;
77 }
78}

80template<typename T>
81AcqSpanPair<const T> recordSpans(const std::vector<T>& records, std::uint64_t mask, RecordWindow window) noexcept
82{
83 const std::uint64_t n = window.end - window.first;
84 if (n == 0) {
85 return {};
86 }
87 const std::uint64_t index = window.first & mask;
88 const std::uint64_t head = std::min<std::uint64_t>(n, mask + 1 - index);
89 return {std::span<const T>(records.data() + index, static_cast<std::size_t>(head)),
90 std::span<const T>(records.data(), static_cast<std::size_t>(n - head))};
91}

93AcqSpanPair<const std::byte> payloadSpans(const RecordRing<AcqSegmentHeader>& ring, RecordWindow window) noexcept
94{
95 const std::uint64_t n = window.end - window.first;
96 if (n == 0 || ring.payloadBytes == 0) {
97 return {};
98 }
99 const std::uint64_t index = window.first & ring.mask;
100 const std::uint64_t head = std::min<std::uint64_t>(n, ring.capacity() - index);
101 const std::size_t pb = ring.payloadBytes;
102 return {std::span<const std::byte>(ring.payloads.data() + index * pb, static_cast<std::size_t>(head) * pb),
103 std::span<const std::byte>(ring.payloads.data(), static_cast<std::size_t>(n - head) * pb)};
104}

106template<typename T>
107bool recordsOverwritten(const RecordRing<T>& ring, RecordWindow window) noexcept
108{
109 if (window.end == window.first) {
110 return false;
111 }
112 const std::uint64_t r = ring.reserved.load(std::memory_order_relaxed);
113 return r > ring.capacity() && r - ring.capacity() > window.first;
114}

120void validateCommits(AcqStreamCore& c, ReaderSlot& slot, std::uint64_t pos, std::uint64_t written) noexcept
121{
122 const std::uint64_t published = c.outputCommits.published.load(std::memory_order_acquire);
123 std::uint64_t i = slot.commitCursor;
124 for (; i < published; ++i) {
125 const OutputCommit& commit = c.outputCommits.records[i & c.outputCommits.mask];
126 if (commit.end > written) {
127 break;
128 }
129 if (commit.start < pos) {
130 slot.dropEnd = std::max(slot.dropEnd, commit.end);
131 c.lateCommits.fetch_add(1, std::memory_order_relaxed);
132 }
133 }
134 if (i != slot.commitCursor) {
135 slot.commitCursor = i;
136 c.commitsValidated.store(i, std::memory_order_seq_cst);
137 c.notifyWriter();
138 }
139}

211AcqExpected<AcqBlockView> AcqReader::acquire(std::size_t minSamples, std::chrono::nanoseconds timeout,
212 std::size_t maxSamples) noexcept
213{
214 if (!m_core) {
215 return AcqErrorCode::InvalidState;
216 }
217 AcqStreamCore& c = *m_core;
218 ReaderSlot& slot = c.slots[m_slot];
219 if (slot.holding) {
220 return AcqErrorCode::InvalidState;
221 }
222 if (minSamples > c.capacity) {
223 return AcqErrorCode::InvalidParameter;
224 }
225 const std::uint64_t maxCount = maxSamples == 0 ? c.capacity : std::min<std::uint64_t>(maxSamples, c.capacity);
226 const bool latest = slot.mode == ReadMode::Latest;
227 const AcqWait wait = slot.mode == ReadMode::LowLatency ? AcqWait::Spin : slot.options.wait;
228 // The clock is read only if the samples are not there yet; reading it costs as much as the rest of acquire().
229 std::optional<Clock::time_point> deadline;
230
231 std::uint64_t pos = 0;
232 std::uint64_t written = 0;
233 std::uint64_t oldest = 0;
234 while (true) {
235 pos = slot.position.load(std::memory_order_relaxed);
236 written = c.committed.load(std::memory_order_acquire);
237 const std::uint64_t r = c.reserved.load(std::memory_order_acquire);
238 oldest = r > c.capacity ? r - c.capacity : 0;
239 if (!latest && pos < oldest) {
240 const std::uint64_t lostNow = oldest - pos;
241 slot.pendingLost += lostNow;
242 addOwned(slot.lost, lostNow);
243 pos = oldest;
244 slot.position.store(pos, std::memory_order_release);
245 }
246 std::uint64_t available = written > pos ? written - pos : 0;
247 const bool finished = c.finished.load(std::memory_order_acquire);
248 if (finished) {
249 written = c.committed.load(std::memory_order_acquire);
250 available = written > pos ? written - pos : 0;
251 }
252 if (slot.gating && c.direction == AcqDirection::Output) {
253 validateCommits(c, slot, pos, written);
254 if (pos < slot.dropEnd) {
255 // Nothing playable until the device has played fill past the dropped commit.
256 if (minSamples != 0) {
257 return AcqErrorCode::Timeout;
258 }
259 written = pos;
260 break;
261 }
262 }
263 if (minSamples == 0 || available >= minSamples || (finished && available > 0)) {
264 break;
265 }
266 if (finished) {
267 return AcqErrorCode::EndOfStream;
268 }
269 if (!deadline) {
270 if (timeout <= std::chrono::nanoseconds(0)) {
271 return AcqErrorCode::Timeout;
272 }
273 deadline = Clock::now() + std::min(timeout, kMaxTimeout);
274 } else if (Clock::now() >= *deadline) {
275 return AcqErrorCode::Timeout;
276 }
277 const std::uint64_t from = pos;
278 auto ready = [&c, from, minSamples] {
279 const std::uint64_t w = c.committed.load(std::memory_order_seq_cst);
280 return (w > from && w - from >= minSamples) || c.finished.load(std::memory_order_seq_cst);
281 };
282 waitFor(ready, wait, slot.options.spinFor, *deadline, slot.waiting, slot.wake, &c.waiters);
283 }
284
285 const std::uint64_t available = written > pos ? written - pos : 0;
286 std::uint64_t start = pos;
287 std::uint64_t count = std::min(available, maxCount);
288 std::uint64_t skipped = 0;
289 if (latest) {
290 count = maxSamples != 0 ? std::min<std::uint64_t>(available, maxCount)
291 : (minSamples != 0 ? std::min<std::uint64_t>(available, minSamples) : available);
292 count = std::min(count, written > oldest ? written - oldest : 0);
293 start = written - count;
294 skipped = start - pos;
295 slot.position.store(start, std::memory_order_release);
296 }
297
298 const std::size_t versionIndex = c.versionAt(start);
299 if (versionIndex + 1 < c.versionCount.load(std::memory_order_acquire)) {
300 const std::uint64_t nextFrom = c.versions[versionIndex + 1]->validFrom;
301 if (nextFrom < start + count) {
302 count = nextFrom - start;
303 }
304 }
305
306 const RecordWindow events = selectRecords(c.events, slot.eventCursor, start, start + count, slot.eventsLost,
307 !latest, [](const AcqEvent& e) { return loadRecordKey(e); });
308 const RecordWindow segments = selectRecords(c.segments, slot.segmentCursor, start, start + count,
309 slot.segmentsLost, !latest,
310 [](const AcqSegmentHeader& h) { return loadRecordKey(h); });
311
312 c.clock.refresh(slot.clock, slot.clockSeq);
313 c.timebase.refresh(slot.timebase, slot.timebaseSeq);
314 AcqBlockView view;
315 ViewAccess::setSamples(view, c, start, count, *c.versions[versionIndex], slot.clock, slot.timebase);
316 ViewAccess::setRead(view, slot.pendingLost, skipped, recordSpans(c.events.records, c.events.mask, events),
317 recordSpans(c.segments.records, c.segments.mask, segments), payloadSpans(c.segments, segments),
318 c.segments.payloadBytes);
319 slot.ticket = (static_cast<std::uint64_t>(m_slot) << 48U) | (++slot.generation & 0xFFFFFFFFFFFFULL);
320 ViewAccess::setTicket(view, slot.ticket);
321 slot.holding = true;
322 slot.viewFirst = start;
323 slot.viewCount = count;
324 slot.eventFirst = events.first;
325 slot.eventEnd = events.end;
326 slot.segmentFirst = segments.first;
327 slot.segmentEnd = segments.end;
328 slot.pendingLost = 0;
329 addOwned(slot.views, 1);
330 if (skipped > 0) {
331 addOwned(slot.skipped, skipped);
332 }
333 return view;
334}

354AcqError mne::detail::releaseView(AcqStreamCore& c, std::uint32_t index, AcqBlockView& view) noexcept
355{
356 ReaderSlot& slot = c.slots[index];
357 if (!slot.holding || ViewAccess::ticket(view) != slot.ticket) {
358 return AcqErrorCode::InvalidState;
359 }
360 std::atomic_thread_fence(std::memory_order_acquire);
361 const std::uint64_t r = c.reserved.load(std::memory_order_relaxed);
362 const bool overwritten = (r > c.capacity && r - c.capacity > slot.viewFirst) ||
363 recordsOverwritten(c.events, {slot.eventFirst, slot.eventEnd}) ||
364 recordsOverwritten(c.segments, {slot.segmentFirst, slot.segmentEnd});
365 if (overwritten) {
366 addOwned(slot.lost, slot.viewCount);
367 addOwned(slot.eventsLost, slot.eventEnd - slot.eventFirst);
368 addOwned(slot.segmentsLost, slot.segmentEnd - slot.segmentFirst);
369 addOwned(slot.overruns, 1);
370 } else {
371 addOwned(slot.delivered, slot.viewCount);
372 }
373 slot.eventCursor = slot.eventEnd;
374 slot.segmentCursor = slot.segmentEnd;
375 const std::uint64_t end = slot.viewFirst + slot.viewCount;
376 slot.holding = false;
377 ViewAccess::clear(view);
378 if (slot.gating) {
379 slot.position.store(end, std::memory_order_release);
380 c.gatePosition.store(end, std::memory_order_seq_cst);
381 c.notifyWriter();
382 } else {
383 slot.position.store(end, std::memory_order_release);
384 }
385 return overwritten ? AcqError(AcqErrorCode::Overrun) : AcqError();
386}

432AcqError AcqReader::consumeIdle(std::size_t samples) noexcept
433{
434 if (!m_core) {
435 return AcqErrorCode::InvalidState;
436 }
437 AcqStreamCore& c = *m_core;
438 ReaderSlot& slot = c.slots[m_slot];
439 if (!slot.gating || c.direction != AcqDirection::Output) {
440 return AcqErrorCode::WrongDirection;
441 }
442 if (slot.holding) {
443 return AcqErrorCode::InvalidState;
444 }
445 const std::uint64_t end = slot.position.load(std::memory_order_relaxed) + samples;
446 slot.idle.fetch_add(samples, std::memory_order_relaxed);
447 c.idleTotal.fetch_add(samples, std::memory_order_acq_rel);
448 slot.position.store(end, std::memory_order_release);
449 c.gatePosition.store(end, std::memory_order_seq_cst);
450 c.notifyWriter();
451 return {};
452}

458template<typename T>
459AcqExpected<AcqReadResult> readInto(AcqReader& reader, const AcqStreamCore* core, const AcqMatrixRef<T>& out,
460 std::chrono::nanoseconds timeout, AcqScaling scaling) noexcept
461{
462 if (core == nullptr) {
463 return AcqErrorCode::InvalidState;
464 }
465 if (out.rows != core->channels) {
466 return AcqErrorCode::ShapeMismatch;
467 }
468 if (!core->memory.hostAccessible()) {
469 return AcqErrorCode::DeviceMemory;
470 }
471 auto view = reader.acquire(out.cols, timeout, out.cols);
472 if (!view) {
473 return view.error();
474 }
475 AcqConvertOptions options;
476 options.scaling = scaling;
477 const AcqError decoded = decode(*view, out, options);
478 const AcqReadResult result{view->firstSample(), view->sampleCount(), view->lost(), view->infoVersion()};
479 const AcqError released = reader.release(*view);
480 if (!decoded.ok()) {
481 return decoded;
482 }
483 if (!released.ok()) {
484 return released;
485 }
486 return result;
487}
