99inline std::string sensorTypeToObjectKey(std::string_view uiType)
100{
101 static const std::unordered_map<std::string_view, std::string_view> map = {
102 {"MEG", "sens_meg"},
103 {"MEG/Grad", "sens_meg_grad"},
104 {"MEG/Mag", "sens_meg_mag"},
105 {"MEG Helmet", "sens_meg_helmet"},
106 {"EEG", "sens_eeg"},
107 {"Digitizer", "dig"},
108 {"Digitizer/Cardinal", "dig_cardinal"},
109 {"Digitizer/HPI", "dig_hpi"},
110 {"Digitizer/EEG", "dig_eeg"},
111 {"Digitizer/Extra", "dig_extra"},
112 };
113 const auto it = map.find(uiType);
114 return it == map.end() ? std::string() : std::string(it->second);
115}
