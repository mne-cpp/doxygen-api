197bool FsAtlasLookup::load(const std::filesystem::path& parcellation)
198{
199 m_loaded = false;
200
201 const bool gzip = mne::to_lower(mne::utf8_from_path(parcellation.extension())) == ".mgz";
202 const auto bytes = mne::read_file_bytes(parcellation, gzip ? mne::Compression::Gzip : mne::Compression::None);
203 if (!bytes) {
204 mne::Log::warning("[FsAtlasLookup::load] {}", bytes.error().what());
205 return false;
206 }
207
208 // MGH header, all big-endian: version, width, height, depth, nframes, type, dof (int32), ras_good_flag
209 // (int16), voxel sizes, x/y/z direction cosines and centre RAS (float32); the voxels start at byte 284.
210 constexpr std::size_t kHeaderSize = 284;
211 if (bytes->size() < kHeaderSize) {
212 mne::Log::warning("[FsAtlasLookup::load] File too small: {}", mne::utf8_from_path(parcellation));
213 return false;
214 }
215 const std::byte* raw = bytes->data();
216 if (read_big_endian<std::int32_t>(raw) != 1) {
217 mne::Log::warning("[FsAtlasLookup::load] Unsupported MGH version: {}", read_big_endian<std::int32_t>(raw));
218 return false;
219 }
220 m_dimX = read_big_endian<std::int32_t>(raw + 4);
221 m_dimY = read_big_endian<std::int32_t>(raw + 8);
222 m_dimZ = read_big_endian<std::int32_t>(raw + 12);
223 const std::int32_t type = read_big_endian<std::int32_t>(raw + 20);
224 if (m_dimX <= 0 || m_dimY <= 0 || m_dimZ <= 0 || read_big_endian<std::int32_t>(raw + 16) <= 0) {
225 mne::Log::warning("[FsAtlasLookup::load] Invalid dimensions: {} {} {}", m_dimX, m_dimY, m_dimZ);
226 return false;
227 }
228
229 Matrix4f vox2ras = Matrix4f::Identity();
230 if (read_big_endian<std::int16_t>(raw + 28) > 0) {
231 const auto f = [raw](std::size_t offset) {
232 return read_big_endian<float>(raw + offset);
233 };
234 Matrix3f MdcD;
235 MdcD << f(42), f(54), f(66),
236 f(46), f(58), f(70),
237 f(50), f(62), f(74);
238 MdcD.col(0) *= f(30);
239 MdcD.col(1) *= f(34);
240 MdcD.col(2) *= f(38);
241 const Vector3f center(static_cast<float>(m_dimX) / 2.0f, static_cast<float>(m_dimY) / 2.0f, static_cast<float>(m_dimZ) / 2.0f);
242 vox2ras.block<3, 3>(0, 0) = MdcD;
243 vox2ras.block<3, 1>(0, 3) = Vector3f(f(78), f(82), f(86)) - MdcD * center;
244 }
245 m_ras2vox = vox2ras.inverse();
246
247 // Only the first frame holds labels
248 const std::size_t nVoxels = std::size_t(m_dimX) * std::size_t(m_dimY) * std::size_t(m_dimZ);
249 const std::size_t width = type == 0 ? 1 : type == 4 ? 2
250 : (type == 1 || type == 3) ? 4
251 : 0;
252 if (width == 0) {
253 mne::Log::warning("[FsAtlasLookup::load] Unsupported MGH data type: {}", type);
254 return false;
255 }
256 if ((bytes->size() - kHeaderSize) / width < nVoxels) {
257 mne::Log::warning("[FsAtlasLookup::load] Truncated voxel data in {}", mne::utf8_from_path(parcellation));
258 return false;
259 }
260 const std::byte* voxels = raw + kHeaderSize;
261 m_voxelData.resize(nVoxels);
262 for (std::size_t i = 0; i < nVoxels; ++i) {
263 switch (type) {
264 case 0:
265 m_voxelData[i] = std::to_integer<int>(voxels[i]);
266 break;
267 case 1:
268 m_voxelData[i] = read_big_endian<std::int32_t>(voxels + 4 * i);
269 break;
270 case 3:
271 m_voxelData[i] = static_cast<int>(std::lround(read_big_endian<float>(voxels + 4 * i)));
272 break;
273 default:
274 m_voxelData[i] = read_big_endian<std::int16_t>(voxels + 2 * i);
275 break;
276 }
277 }
278
279 m_loaded = true;
280 return true;
281}

285const std::string& FsAtlasLookup::label_at_ras(const Vector3f& ras) const
286{
287 if (!m_loaded) {
288 return unknown();
289 }
290 const Vector3i vox = ras_to_voxel(ras);
291 if (vox(0) < 0 || vox(0) >= m_dimX || vox(1) < 0 || vox(1) >= m_dimY || vox(2) < 0 || vox(2) >= m_dimZ) {
292 return unknown();
293 }
294 // MGH stores the voxels in column-major order: index = x + dimX * (y + dimY * z)
295 const std::size_t index = std::size_t(vox(0)) + std::size_t(m_dimX) * (std::size_t(vox(1)) + std::size_t(m_dimY) * std::size_t(vox(2)));
296 const auto& table = lookupTable();
297 const auto it = table.find(m_voxelData[index]);
298 return it == table.end() ? unknown() : it->second;
299}

322Vector3i FsAtlasLookup::ras_to_voxel(const Vector3f& ras) const
323{
324 const Vector4f vox = m_ras2vox * Vector4f(ras(0), ras(1), ras(2), 1.0f);
325 return Vector3i(static_cast<int>(std::round(vox(0))), static_cast<int>(std::round(vox(1))), static_cast<int>(std::round(vox(2))));
326}
