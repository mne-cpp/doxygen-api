75std::array<std::vector<int>, 3> pcaChannelGroups(const FiffInfo& info)
76{
77 std::array<std::vector<int>, 3> groups;
78 for (int k = 0; k < std::ssize(info.chs); ++k) {
79 if (has_channel(info.bads, info.ch_names[k])) {
80 continue;
81 }
82 if (info.chs[k].kind == FIFFV_MEG_CH) {
83 groups[info.chs[k].unit == FIFF_UNIT_T ? 1 : 0].push_back(k);
84 } else if (info.chs[k].kind == FIFFV_EEG_CH) {
85 groups[2].push_back(k);
86 }
87 }
88 return groups;
89}

95std::vector<FiffProj> pcaProjectors(const MatrixXd& data, const std::vector<int>& chIdx, const StringList& chNames,
96 int nVec, const StringList& descs)
97{
98 std::vector<FiffProj> projs;
99 if (nVec <= 0 || chIdx.empty()) {
100 return projs;
101 }
102 MatrixXd sub(chIdx.size(), data.cols());
103 for (int c = 0; c < std::ssize(chIdx); ++c) {
104 sub.row(c) = data.row(chIdx[c]);
105 }
106 const JacobiSVD<MatrixXd> svd(sub, ComputeThinU);
107 const int nComp = std::min(nVec, static_cast<int>(svd.matrixU().cols()));
108 for (int v = 0; v < nComp; ++v) {
109 FiffProj proj;
110 proj.kind = FIFFV_PROJ_ITEM_FIELD;
111 proj.active = false;
112 proj.desc = descs[static_cast<std::size_t>(v)];
113 proj.data.nrow = 1;
114 proj.data.ncol = static_cast<int>(chNames.size());
115 proj.data.col_names = chNames;
116 proj.data.data = MatrixXd::Zero(1, chNames.size());
117 for (int c = 0; c < std::ssize(chIdx); ++c) {
118 proj.data.data(0, chIdx[c]) = svd.matrixU()(c, v);
119 }
120 projs.push_back(proj);
121 }
122 return projs;
123}

177fiff_int_t FiffProj::make_projector(const std::vector<FiffProj>& projs, const StringList& ch_names, MatrixXd& proj, const StringList& bads, MatrixXd& U)
178{
179 fiff_int_t nchan = static_cast<int>(ch_names.size());
180 if (nchan == 0) {
181 throw std::invalid_argument("No channel names specified");
182 }
183
184 // if(proj)
185 // delete proj;
186 proj = MatrixXd::Identity(nchan, nchan);
187 fiff_int_t nproj = 0;
188 U = MatrixXd();
189
190 //
191 // Check trivial cases first
192 //
193 if (projs.size() == 0)
194 return 0;
195
196 fiff_int_t nvec = 0;
197 fiff_int_t k;
198 for (k = 0; k < std::ssize(projs); ++k) {
199 if (projs[k].active) {
200 ++nproj;
201 nvec += projs[k].data.nrow;
202 }
203 }
204
205 if (nproj == 0) {
206 Log::warning("FiffProj::make_projector - No projectors nproj=0\n");
207 return 0;
208 }
209
210 if (nvec <= 0) {
211 Log::warning("FiffProj::make_projector - No rows in projector matrices found nvec<=0\n");
212 return 0;
213 }
214
215 //
216 // Pick the appropriate entries
217 //
218 MatrixXd vecs = MatrixXd::Zero(nchan, nvec);
219 nvec = 0;
220 fiff_int_t nonzero = 0;
221 std::int32_t p, c, i, j, v;
222 double onesize;
223 bool isBad = false;
224 RowVectorXi sel(nchan);
225 RowVectorXi vecSel(nchan);
226 sel.setConstant(-1);
227 vecSel.setConstant(-1);
228 for (k = 0; k < std::ssize(projs); ++k) {
229 if (projs[k].active) {
230 FiffProj one = projs[k];
231
232 const std::set<std::string> unique(one.data.col_names.begin(), one.data.col_names.end());
233 if (one.data.col_names.size() != unique.size()) {
234 Log::warning("Channel name list in projection item {} contains duplicate items", k);
235 return 0;
236 }
237
238 //
239 // Get the two selection vectors to pick correct elements from
240 // the projection vectors omitting bad channels
241 //
242 sel.resize(nchan);
243 vecSel.resize(nchan);
244 sel.setConstant(-1);
245 vecSel.setConstant(-1);
246 p = 0;
247 for (c = 0; c < nchan; ++c) {
248 for (i = 0; i < static_cast<int>(one.data.col_names.size()); ++i) {
249 if (same_channel(ch_names[c], one.data.col_names[i])) {
250 isBad = false;
251 for (j = 0; j < static_cast<int>(bads.size()); ++j) {
252 if (same_channel(ch_names[c], bads[j])) {
253 isBad = true;
254 }
255 }
256
257 if (!isBad && sel[p] != c) {
258 sel[p] = c;
259 vecSel[p] = i;
260 ++p;
261 }
262 }
263 }
264 }
265 sel.conservativeResize(p);
266 vecSel.conservativeResize(p);
267 //
268 // If there is something to pick, pickit
269 //
270 if (sel.cols() > 0)
271 for (v = 0; v < one.data.nrow; ++v)
272 for (i = 0; i < p; ++i)
273 vecs(sel[i], nvec + v) = one.data.data(v, vecSel[i]);
274
275 //
276 // Rescale for more straightforward detection of small singular values
277 //
278 for (v = 0; v < one.data.nrow; ++v) {
279 onesize = sqrt((vecs.col(nvec + v).transpose() * vecs.col(nvec + v))(0, 0));
280 if (onesize > 0.0) {
281 vecs.col(nvec + v) = vecs.col(nvec + v) / onesize;
282 ++nonzero;
283 }
284 }
285 nvec += one.data.nrow;
286 }
287 }
288 //
289 // Check whether all of the vectors are exactly zero
290 //
291 if (nonzero == 0)
292 return 0;
293
294 //
295 // Reorthogonalize the vectors
296 //
297 JacobiSVD<MatrixXd> svd(vecs.block(0, 0, vecs.rows(), nvec), ComputeFullU);
298 //Sort singular values and singular vectors
299 VectorXd S = svd.singularValues();
300 MatrixXd t_U = svd.matrixU();
301 Linalg::sort<double>(S, t_U);
302
303 //
304 // Throw away the linearly dependent guys
305 //
306 nproj = 0;
307 for (k = 0; k < S.size(); ++k)
308 if (S[k] / S[0] > 1e-2)
309 ++nproj;
310
311 U = t_U.block(0, 0, t_U.rows(), nproj);
312
313 //
314 // Here is the celebrated result
315 //
316 proj -= U * U.transpose();
317
318 return nproj;
319}

323std::vector<FiffProj> FiffProj::compute_from_raw(const FiffRawData& raw,
324 const MatrixXi& events,
325 int eventCode,
326 float tmin,
327 float tmax,
328 int nGrad,
329 int nMag,
330 int nEeg,
331 const std::map<std::string, double>& mapReject)
332{
333 float sfreq = raw.info.sfreq;
334 int nchan = raw.info.nchan;
335
336 int minSamp = static_cast<int>(std::round(tmin * sfreq));
337 int maxSamp = static_cast<int>(std::round(tmax * sfreq));
338 int ns = maxSamp - minSamp + 1;
339
340 if (ns <= 0) {
341 Log::warning("[FiffProj::compute_from_raw] Invalid time window.");
342 return {};
343 }
344
345 // Collect matching epochs
346 std::vector<MatrixXd> epochs;
347 const auto reject = [&mapReject](const char* type) {
348 const auto it = mapReject.find(type);
349 return it == mapReject.end() ? 0.0 : it->second;
350 };
351 double gradReject = reject("grad");
352 double magReject = reject("mag");
353 double eegReject = reject("eeg");
354
355 for (int k = 0; k < events.rows(); ++k) {
356 if (events(k, 1) != 0 || events(k, 2) != eventCode)
357 continue;
358
359 int evSample = events(k, 0);
360 int epochStart = evSample + minSamp;
361 int epochEnd = evSample + maxSamp;
362
363 if (epochStart < raw.first_samp || epochEnd > raw.last_samp)
364 continue;
365
366 MatrixXd epochData, epochTimes;
367 if (!raw.read_raw_segment(epochData, epochTimes, epochStart, epochEnd))
368 continue;
369
370 // Simple peak-to-peak rejection
371 bool ok = true;
372 for (int c = 0; c < nchan && ok; ++c) {
373 if (has_channel(raw.info.bads, raw.info.ch_names[c]))
374 continue;
375 double pp = epochData.row(c).maxCoeff() - epochData.row(c).minCoeff();
376 if (raw.info.chs[c].kind == FIFFV_MEG_CH) {
377 if (raw.info.chs[c].unit == FIFF_UNIT_T && magReject > 0 && pp > magReject)
378 ok = false;
379 else if (raw.info.chs[c].unit != FIFF_UNIT_T && gradReject > 0 && pp > gradReject)
380 ok = false;
381 } else if (raw.info.chs[c].kind == FIFFV_EEG_CH && eegReject > 0 && pp > eegReject) {
382 ok = false;
383 }
384 }
385 if (!ok)
386 continue;
387
388 epochs.push_back(epochData);
389 }
390
391 if (epochs.empty()) {
392 Log::warning("[FiffProj::compute_from_raw] No valid epochs found for event {}", eventCode);
393 return {};
394 }
395
396 Log::info("[FiffProj::compute_from_raw] {} epochs collected for event {}", epochs.size(), eventCode);
397
398 // SVD of the uncentred data (MNE-C compute_cov_raw_epoch with remove_sample_mean = FALSE,
399 // mne.compute_proj_epochs); removing the mean first yields a different subspace.
400 MatrixXd data(nchan, epochs.size() * ns);
401 for (int e = 0; e < std::ssize(epochs); ++e) {
402 data.middleCols(e * ns, ns) = epochs[e];
403 }
404 const std::array<std::vector<int>, 3> groups = pcaChannelGroups(raw.info);
405 const int nVec[3] = {nGrad, nMag, nEeg};
406 const char* kinds[3] = {"grad", "mag", "eeg"};
407 std::vector<FiffProj> projs;
408 for (int g = 0; g < 3; ++g) {
409 StringList descs;
410 for (int v = 0; v < nVec[g]; ++v) {
411 descs.push_back(std::format("PCA-{}-v{}", kinds[g], v + 1));
412 }
413 std::ranges::move(pcaProjectors(data, groups[g], raw.info.ch_names, nVec[g], descs), std::back_inserter(projs));
414 }
415
416 return projs;
417}

421std::vector<FiffProj> FiffProj::compute_from_evoked(const FiffEvoked& evoked, int nGrad, int nMag, int nEeg)
422{
423 if (evoked.data.cols() == 0 || evoked.times.size() == 0) {
424 return {};
425 }
426 const std::string prefix = std::format("{:.3f}-{:.3f}", evoked.times(0), evoked.times(evoked.times.size() - 1));
427 const std::array<std::vector<int>, 3> groups = pcaChannelGroups(evoked.info);
428 const int nVec[3] = {nGrad, nMag, nEeg};
429 const char* kinds[3] = {"planar", "axial", "eeg"};
430 std::vector<FiffProj> projs;
431 for (int g = 0; g < 3; ++g) {
432 StringList descs;
433 for (int v = 0; v < nVec[g]; ++v) {
434 descs.push_back(std::format("{}-{}-PCA-{:02}", kinds[g], prefix, v + 1));
435 }
436 std::ranges::move(pcaProjectors(evoked.data, groups[g], evoked.info.ch_names, nVec[g], descs), std::back_inserter(projs));
437 }
438 return projs;
439}
