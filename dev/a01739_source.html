105std::vector<float> MriVolData::voxelDataAsFloat() const
106{
107 if (slices.empty() || width <= 0 || height <= 0 || depth <= 0)
108 return {};
109
110 const std::size_t nPixels = static_cast<std::size_t>(width) * static_cast<std::size_t>(height);
111 std::vector<float> result(nPixels * static_cast<std::size_t>(depth), 0.0f);
112 const std::size_t nSlices = std::min(static_cast<std::size_t>(depth), slices.size());
113 for (std::size_t k = 0; k < nSlices; ++k) {
114 const MriSlice& slice = slices[k];
115 float* out = result.data() + k * nPixels;
116 switch (slice.pixelFormat) {
117 case FIFFV_MRI_PIXEL_BYTE:
118 std::transform(slice.pixels.begin(), slice.pixels.begin() + std::ptrdiff_t(std::min(nPixels, slice.pixels.size())), out, [](unsigned char v) { return static_cast<float>(v); });
119 break;
120 case FIFFV_MRI_PIXEL_WORD:
121 std::transform(slice.pixelsWord.begin(), slice.pixelsWord.begin() + std::ptrdiff_t(std::min(nPixels, slice.pixelsWord.size())), out, [](unsigned short v) { return static_cast<float>(v); });
122 break;
123 case FIFFV_MRI_PIXEL_FLOAT:
124 std::copy_n(slice.pixelsFloat.begin(), std::min(nPixels, slice.pixelsFloat.size()), out);
125 break;
126 }
127 }
128 return result;
129}

133Matrix4f MriVolData::computeVox2Ras() const
134{
135 //
136 // Build voxel-to-surface-RAS transform (FreeSurfer convention).
137 //
138 // The direction cosine matrix Mdc stores the orientation of each voxel axis
139 // (columns are x, y, z directions). Scaling by voxel size gives the actual
140 // spacing matrix M:
141 //
142 // M = Mdc * diag(xsize, ysize, zsize)
143 //
144 // The origin P0 in RAS coordinates is:
145 //
146 // P0 = c_ras - M * (dim/2)
147 //
148 // The full 4x4 vox2ras matrix is:
149 //
150 // | M P0 | (in mm, then converted to meters for FIFF)
151 // | 0 1 |
152 //
153 // Reference: https://surfer.nmr.mgh.harvard.edu/fswiki/FsTutorial/MghFormat
154 //
155
156 // Construct M = Mdc * D (scale direction cosines by voxel sizes)
157 Matrix3f M;
158 M.col(0) = x_ras * xsize;
159 M.col(1) = y_ras * ysize;
160 M.col(2) = z_ras * zsize;
161
162 // Compute center voxel
163 Vector3f center(static_cast<float>(width) / 2.0f,
164 static_cast<float>(height) / 2.0f,
165 static_cast<float>(depth) / 2.0f);
166
167 // Compute P0 = c_ras - M * center
168 Vector3f P0 = c_ras - M * center;
169
170 // Build 4x4 matrix in meters (FreeSurfer uses mm, FIFF uses meters)
171 Matrix4f vox2ras = Matrix4f::Identity();
172 vox2ras.block<3, 3>(0, 0) = M / 1000.0f;
173 vox2ras.block<3, 1>(0, 3) = P0 / 1000.0f;
174
175 return vox2ras;
176}

180Matrix4f MriVolData::computeVox2RasTkr() const
181{
182 // Same as computeVox2Ras() but with c_ras = (0,0,0).
183 // This gives the tkregister (surface RAS) coordinate system
184 // that FreeSurfer surfaces use.
185 Matrix3f M;
186 M.col(0) = x_ras * xsize;
187 M.col(1) = y_ras * ysize;
188 M.col(2) = z_ras * zsize;
189
190 Vector3f center(static_cast<float>(width) / 2.0f,
191 static_cast<float>(height) / 2.0f,
192 static_cast<float>(depth) / 2.0f);
193
194 Vector3f P0 = -M * center; // c_ras = (0,0,0) for tkRAS
195
196 Matrix4f vox2ras_tkr = Matrix4f::Identity();
197 vox2ras_tkr.block<3, 3>(0, 0) = M / 1000.0f;
198 vox2ras_tkr.block<3, 1>(0, 3) = P0 / 1000.0f;
199
200 return vox2ras_tkr;
201}
