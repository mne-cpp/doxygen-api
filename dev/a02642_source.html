54int MNEChSelection::assign_channels(std::span<const FiffChInfo> chs, const MNEDeriv* deriv)
55{
56 chspick = chdef;
57 chspick_nospace = chdef;
58 for (auto& channel : chspick_nospace)
59 std::erase(channel, ' ');
60 const int nchan = this->nchan();
61
62 pick.setConstant(nchan, -1);
63 pick_deriv.setConstant(nchan, -1);
64 ch_kind.setConstant(nchan, -1);
65
66 const auto matches = [this](int c, std::string_view ch_name) {
67 return mne::iequals(chspick[c], ch_name) || mne::iequals(chspick_nospace[c], ch_name);
68 };
69 const auto assign = [this, &chs](int c, std::size_t rc) {
70 pick[c] = static_cast<int>(rc);
71 ch_kind[c] = chs[rc].kind;
72 };
73
74 for (int c = 0; c < nchan; c++) {
75 for (std::size_t rc = 0; rc < chs.size(); rc++) {
76 if (matches(c, chs[rc].ch_name)) {
77 assign(c, rc);
78 break;
79 }
80 }
81 }
82 /*
83 * Maybe the derivations will help
84 */
85 if (deriv && deriv->deriv_data) {
86 const StringList& deriv_names = deriv->deriv_data->rowlist;
87 for (int c = 0; c < nchan; c++) {
88 if (pick[c] != -1)
89 continue;
90 for (std::size_t d = 0; d < deriv_names.size(); d++) {
91 if (mne::iequals(chspick[c], deriv_names[d]) && d < static_cast<std::size_t>(deriv->valid.size()) && deriv->valid[static_cast<Eigen::Index>(d)]) {
92 pick_deriv[c] = static_cast<int>(d);
93 ch_kind[c] = deriv->chs[d].kind;
94 break;
95 }
96 }
97 }
98 }
99 /*
100 * Try simple channels again without the part after dashes
101 */
102 for (int c = 0; c < nchan; c++) {
103 if (pick[c] != -1 || pick_deriv[c] != -1)
104 continue;
105 for (std::size_t rc = 0; rc < chs.size(); rc++) {
106 const std::string_view ch_name = chs[rc].ch_name;
107 const auto dash = ch_name.find('-');
108 if (dash != std::string_view::npos && matches(c, ch_name.substr(0, dash))) {
109 assign(c, rc);
110 break;
111 }
112 }
113 }
114 if (const int matched = nderiv(); matched > 0)
115 mne::Log::info("Selection \"{}\" has {} matched derived channels.", name, matched);
116 return static_cast<int>(std::ranges::count_if(pick, [](int p) { return p >= 0; }));
117}
