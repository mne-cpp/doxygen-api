78ForwardPanel::ForwardPanel(const std::filesystem::path& atlasDirectory)
79: Panel("FwdSettingsView", "Forward Solution")
80{
81 SettingsForm& f = form();
82 f.add(make(kCompute, SettingField::Kind::Action, "Compute Forward Solution", false, kComputation));
83 f.add(make(kRecompute, SettingField::Kind::Bool, "Enable head position update", false, kComputation));
84 f.add(make(kCluster, SettingField::Kind::Bool, "Cluster forward solution", true, kComputation));
85 SettingField sources = make(kClusterSources, SettingField::Kind::Int, "Number of sources per cluster", 200, kComputation);
86 sources.minimum = 5;
87 sources.maximum = 999;
88 sources.commit = SettingField::Commit::WhenFinished;
89 f.add(std::move(sources));
90 f.add(make(kStatus, SettingField::Kind::Info, "Status computation:", std::string("Not Computed"), kComputation));
91 f.set_tone(kStatus, SettingField::Tone::Negative);
92
93 f.add(make(kAtlasDirectory, SettingField::Kind::Directory, "Brain atlas directory", utf8_from_path(atlasDirectory), kClustering));
94 f.add(make(kAtlasStatus, SettingField::Kind::Info, "(lh+rh).aparc.a2009s.annot", std::string("not loaded"), kClustering));
95
96 f.add(make("sourceSpaces", SettingField::Kind::Info, "Source spaces:", std::string("0"), kInformation));
97 f.add(make("sources", SettingField::Kind::Info, "Number of sources:", std::string("0"), kInformation));
98 f.add(make("clusteredSources", SettingField::Kind::Info, "Number of clustered sources:", std::string("Not Clustered"), kInformation));
99 f.add(make("sourceOrientation", SettingField::Kind::Info, "Source orientation:", std::string("fixed"), kInformation));
100 f.add(make("channels", SettingField::Kind::Info, "Number of channels:", std::string("0"), kInformation));
101 f.add(make("coordinateFrame", SettingField::Kind::Info, "Coordinate frame:", std::string("Head Space"), kInformation));
102
103 // Nothing can listen yet, so the start atlas is not reported
104 if (!atlasDirectory.empty()) {
105 set_atlas_directory(atlasDirectory, false);
106 }
107
108 m_listener = f.add_listener([this](const std::string& key, const std::optional<SettingValue>& value) {
109 if (!m_updating && (value || key == kCompute)) {
110 changed(key);
111 }
112 });
113}

124void ForwardPanel::changed(const std::string& key)
125{
126 if (key == kCompute) {
127 if (m_callbacks.compute) {
128 m_callbacks.compute();
129 }
130 } else if (key == kRecompute) {
131 if (m_callbacks.recomputation_changed) {
132 m_callbacks.recomputation_changed(recomputation());
133 }
134 } else if (key == kCluster) {
135 // Clustering needs the atlas
136 if (!m_atlasLoaded && clustering()) {
137 m_updating = true;
138 form().set_value(kCluster, false);
139 form().set_value(kAtlasStatus, std::string("not loaded: load an annotation set before clustering"));
140 form().set_tone(kAtlasStatus, SettingField::Tone::Warning);
141 m_updating = false;
142 return;
143 }
144 if (m_callbacks.clustering_changed) {
145 m_callbacks.clustering_changed(clustering());
146 }
147 } else if (key == kClusterSources) {
148 if (m_callbacks.cluster_sources_changed) {
149 m_callbacks.cluster_sources_changed(cluster_sources());
150 }
151 } else if (key == kAtlasDirectory) {
152 set_atlas_directory(path_from_utf8(form().get<std::string>(kAtlasDirectory)));
153 }
154}

158bool ForwardPanel::set_atlas_directory(const std::filesystem::path& directory, bool notify)
159{
160 m_updating = true;
161 form().set_value(kAtlasDirectory, utf8_from_path(directory));
162 m_updating = false;
163 auto annotations = std::make_shared<mne::FsAnnotationSet>(directory / "lh.aparc.a2009s.annot", directory / "rh.aparc.a2009s.annot");
164 m_atlasLoaded = !annotations->empty() && annotations->size() == 2;
165 m_updating = true;
166 form().set_value(kAtlasStatus, std::string(m_atlasLoaded ? "loaded" : "not loaded"));
167 form().set_tone(kAtlasStatus, m_atlasLoaded ? SettingField::Tone::Positive : SettingField::Tone::Normal);
168 m_updating = false;
169 if (m_atlasLoaded && notify && m_callbacks.atlas_changed) {
170 m_callbacks.atlas_changed(directory, annotations);
171 }
172 return m_atlasLoaded;
173}

177void ForwardPanel::set_status(int status)
178{
179 static const std::array<const char*, 7> kTexts = {"Initializing", "Computing", "Recomputing", "Clustering", "Not Computed", "Not Clustered", "Finished"};
180 const int index = (status >= 0 && status < static_cast<int>(kTexts.size())) ? status : static_cast<int>(Status::Finished);
181 SettingField::Tone tone = SettingField::Tone::Negative;
182 if (index == static_cast<int>(Status::NotClustered)) {
183 tone = SettingField::Tone::Warning;
184 } else if (index == static_cast<int>(Status::Finished)) {
185 tone = SettingField::Tone::Positive;
186 }
187 m_updating = true;
188 form().set_value(kStatus, std::string(kTexts[static_cast<std::size_t>(index)]));
189 form().set_tone(kStatus, tone);
190 m_updating = false;
191}
