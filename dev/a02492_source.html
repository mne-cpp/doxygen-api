79SettingsPanel::SettingsPanel(std::shared_ptr<SettingsForm> form, QWidget* parent)
80: QWidget(parent)
81, m_pForm(std::move(form))
82, m_form(*m_pForm)
83{
84 setWindowTitle(qt(m_form.title()));
85 auto* layout = new QVBoxLayout(this);
86 for (const std::string& group : m_form.groups()) {
87 QFormLayout* formLayout = nullptr;
88 if (group.empty()) {
89 auto* holder = new QWidget(this);
90 formLayout = new QFormLayout(holder);
91 formLayout->setContentsMargins(0, 0, 0, 0);
92 layout->addWidget(holder);
93 } else {
94 auto* box = new QGroupBox(qt(group), this);
95 formLayout = new QFormLayout(box);
96 layout->addWidget(box);
97 }
98 for (const SettingField& field : m_form.fields()) {
99 if (field.group == group) {
100 buildField(field, formLayout);
101 }
102 }
103 }
104 layout->addStretch(1);
105
106 m_listener = m_form.add_listener([this](const std::string& key, const std::optional<SettingValue>& value) {
107 if (value) {
108 showValue(key);
109 updateConditions();
110 }
111 });
112 m_structureListener = m_form.add_structure_listener([this](const std::string& key) {
113 updateField(key);
114 });
115 updateConditions();
116}

128void SettingsPanel::buildField(const SettingField& field, QFormLayout* layout)
129{
130 Row row;
131 const std::string key = field.key;
132 switch (field.kind) {
133 case SettingField::Kind::Bool: {
134 auto* box = new QCheckBox(qt(field.label), this);
135 connect(box, &QCheckBox::toggled, this, [this, key](bool on) {
136 m_form.set_value(key, on);
137 });
138 layout->addRow(box);
139 row.editor = box;
140 break;
141 }
142 case SettingField::Kind::Int:
143 case SettingField::Kind::Double:
144 if (field.style == SettingField::Style::Slider) {
145 auto* slider = new QSlider(Qt::Horizontal, this);
146 connect(slider, &QSlider::valueChanged, this, [this, key](int position) {
147 if (const SettingField* f = m_form.field(key)) {
148 m_form.set_value(key, f->kind == SettingField::Kind::Int ? SettingValue(position) : SettingValue(position / sliderFactor(*f)));
149 }
150 });
151 row.editor = slider;
152 } else if (field.kind == SettingField::Kind::Int) {
153 auto* spin = new QSpinBox(this);
154 if (field.commit == SettingField::Commit::WhenFinished) {
155 connect(spin, &QSpinBox::editingFinished, this, [this, key, spin] {
156 m_form.set_value(key, spin->value());
157 });
158 } else {
159 connect(spin, &QSpinBox::valueChanged, this, [this, key](int number) {
160 m_form.set_value(key, number);
161 });
162 }
163 row.editor = spin;
164 } else {
165 auto* spin = new QDoubleSpinBox(this);
166 if (field.commit == SettingField::Commit::WhenFinished) {
167 connect(spin, &QDoubleSpinBox::editingFinished, this, [this, key, spin] {
168 m_form.set_value(key, spin->value());
169 });
170 } else {
171 connect(spin, &QDoubleSpinBox::valueChanged, this, [this, key](double number) {
172 m_form.set_value(key, number);
173 });
174 }
175 row.editor = spin;
176 }
177 break;
178 case SettingField::Kind::Choice: {
179 auto* combo = new QComboBox(this);
180 connect(combo, &QComboBox::currentIndexChanged, this, [this, key, combo](int index) {
181 if (index >= 0) {
182 m_form.set_value(key, combo->itemData(index).toString().toStdString());
183 }
184 });
185 row.editor = combo;
186 break;
187 }
188 case SettingField::Kind::Text:
189 case SettingField::Kind::Path: {
190 auto* edit = new QLineEdit(this);
191 if (field.commit == SettingField::Commit::WhenFinished) {
192 connect(edit, &QLineEdit::editingFinished, this, [this, key, edit] {
193 m_form.set_value(key, edit->text().toStdString());
194 });
195 } else {
196 connect(edit, &QLineEdit::textEdited, this, [this, key](const QString& text) {
197 m_form.set_value(key, text.toStdString());
198 });
199 }
200 if (field.kind == SettingField::Kind::Text) {
201 row.editor = edit;
202 break;
203 }
204 auto* container = new QWidget(this);
205 auto* line = new QHBoxLayout(container);
206 line->setContentsMargins(0, 0, 0, 0);
207 edit->setParent(container);
208 line->addWidget(edit);
209 auto* browse = new QToolButton(container);
210 browse->setText(QStringLiteral("..."));
211 line->addWidget(browse);
212 connect(browse, &QToolButton::clicked, this, [this, key, edit] {
213 const SettingField* f = m_form.field(key);
214 const QString file = QFileDialog::getOpenFileName(this, f ? qt(f->label) : QString(), edit->text(), f ? qt(f->file_filter) : QString());
215 if (!file.isEmpty()) {
216 m_form.set_value(key, file.toStdString());
217 }
218 });
219 row.editor = container;
220 break;
221 }
222 case SettingField::Kind::Action: {
223 auto* button = new QPushButton(qt(field.label), this);
224 connect(button, &QPushButton::clicked, this, [this, key] {
225 m_form.trigger(key);
226 });
227 layout->addRow(button);
228 row.editor = button;
229 break;
230 }
231 case SettingField::Kind::Custom: {
232 auto* container = new QWidget(this);
233 auto* inner = new QVBoxLayout(container);
234 inner->setContentsMargins(0, 0, 0, 0);
235 if (field.label.empty()) {
236 layout->addRow(container);
237 } else {
238 row.label = new QLabel(qt(field.label), this);
239 layout->addRow(row.label, container);
240 }
241 row.editor = container;
242 break;
243 }
244 }
245 if (!row.label && field.kind != SettingField::Kind::Bool && field.kind != SettingField::Kind::Action && field.kind != SettingField::Kind::Custom) {
246 row.label = new QLabel(qt(field.label), this);
247 layout->addRow(row.label, row.editor);
248 }
249 row.editor->setObjectName(qt(key));
250 row.editor->setToolTip(qt(field.tooltip));
251 m_rows[key] = row;
252 updateField(key);
253}

257void SettingsPanel::updateField(const std::string& key)
258{
259 const auto it = m_rows.find(key);
260 const SettingField* field = m_form.field(key);
261 if (it == m_rows.end() || !field) {
262 return;
263 }
264 QWidget* editor = it->second.editor;
265 if (it->second.label) {
266 it->second.label->setText(qt(field->label));
267 }
268 const QSignalBlocker blocker(editor);
269 if (auto* spin = qobject_cast<QSpinBox*>(editor)) {
270 spin->setRange(static_cast<int>(std::lround(field->minimum)), static_cast<int>(std::lround(field->maximum)));
271 spin->setSingleStep(std::max(1, static_cast<int>(std::lround(field->step))));
272 spin->setPrefix(qt(field->prefix));
273 spin->setSuffix(qt(field->unit));
274 } else if (auto* doubleSpin = qobject_cast<QDoubleSpinBox*>(editor)) {
275 doubleSpin->setDecimals(field->decimals);
276 doubleSpin->setRange(field->minimum, field->maximum);
277 doubleSpin->setSingleStep(field->step);
278 doubleSpin->setPrefix(qt(field->prefix));
279 doubleSpin->setSuffix(qt(field->unit));
280 } else if (auto* slider = qobject_cast<QSlider*>(editor)) {
281 const double factor = sliderFactor(*field);
282 slider->setRange(static_cast<int>(std::lround(field->minimum * factor)), static_cast<int>(std::lround(field->maximum * factor)));
283 slider->setSingleStep(std::max(1, static_cast<int>(std::lround(field->step * factor))));
284 } else if (auto* combo = qobject_cast<QComboBox*>(editor)) {
285 combo->clear();
286 for (const SettingChoice& choice : field->choices) {
287 combo->addItem(qt(choice.label), qt(choice.value));
288 }
289 } else if (auto* button = qobject_cast<QAbstractButton*>(editor)) {
290 button->setText(qt(field->label));
291 }
292 showValue(key);
293}

297void SettingsPanel::showValue(const std::string& key)
298{
299 const auto it = m_rows.find(key);
300 const SettingField* field = m_form.field(key);
301 const std::optional<SettingValue> value = m_form.value(key);
302 if (it == m_rows.end() || !field || !value) {
303 return;
304 }
305 QWidget* editor = it->second.editor;
306 const QSignalBlocker blocker(editor);
307 if (auto* box = qobject_cast<QCheckBox*>(editor)) {
308 box->setChecked(m_form.get<bool>(key));
309 } else if (auto* spin = qobject_cast<QSpinBox*>(editor)) {
310 spin->setValue(m_form.get<int>(key));
311 } else if (auto* doubleSpin = qobject_cast<QDoubleSpinBox*>(editor)) {
312 doubleSpin->setValue(m_form.get<double>(key));
313 } else if (auto* slider = qobject_cast<QSlider*>(editor)) {
314 slider->setValue(static_cast<int>(std::lround(m_form.get<double>(key) * sliderFactor(*field))));
315 } else if (auto* combo = qobject_cast<QComboBox*>(editor)) {
316 combo->setCurrentIndex(combo->findData(qt(m_form.get<std::string>(key))));
317 } else if (auto* edit = qobject_cast<QLineEdit*>(editor)) {
318 edit->setText(qt(m_form.get<std::string>(key)));
319 } else if (auto* pathEdit = editor->findChild<QLineEdit*>()) {
320 const QSignalBlocker pathBlocker(pathEdit);
321 pathEdit->setText(qt(m_form.get<std::string>(key)));
322 }
323}

359void SettingsPanel::loadSettings(const QString& path)
360{
361 if (path.isEmpty()) {
362 return;
363 }
364 QSettings settings(QStringLiteral("MNECPP"));
365 for (const SettingField& field : m_form.fields()) {
366 const QString name = path + QLatin1Char('/') + qt(field.key);
367 if (!settings.contains(name)) {
368 continue;
369 }
370 const QVariant stored = settings.value(name);
371 switch (field.kind) {
372 case SettingField::Kind::Bool:
373 m_form.set_value(field.key, stored.toBool());
374 break;
375 case SettingField::Kind::Int:
376 m_form.set_value(field.key, stored.toInt());
377 break;
378 case SettingField::Kind::Double:
379 m_form.set_value(field.key, stored.toDouble());
380 break;
381 case SettingField::Kind::Choice:
382 case SettingField::Kind::Text:
383 case SettingField::Kind::Path:
384 m_form.set_value(field.key, stored.toString().toStdString());
385 break;
386 case SettingField::Kind::Action:
387 case SettingField::Kind::Custom:
388 break;
389 }
390 }
391}

406 if constexpr (std::is_same_v<T, std::string>) {
