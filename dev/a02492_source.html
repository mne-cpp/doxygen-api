41 const std::shared_ptr<FiffDirNode>& node,
42 int kind)
43{
44 std::shared_ptr<FiffDirNode> block;
45 std::optional<FiffTag> tag;
46 if (node->type == FIFFB_MNE_NAMED_MATRIX) {
47 if (tag = node->find_tag(stream, kind); tag) {
48 block = node;
49 }
50 } else {
51 for (const std::shared_ptr<FiffDirNode>& child : node->children) {
52 if (child->type != FIFFB_MNE_NAMED_MATRIX) {
53 continue;
54 }
55 if (tag = child->find_tag(stream, kind); tag) {
56 block = child;
57 break;
58 }
59 }
60 }
61 if (!block) {
62 return nullptr;
63 }
64 std::unique_ptr<FiffSparseMatrix> data = FiffSparseMatrix::fiff_get_float_sparse_matrix(*tag);
65 if (!data) {
66 return nullptr;
67 }
68 auto matrix = std::make_unique<MNESparseNamedMatrix>();
69 matrix->data = std::move(*data);
70 if (const auto nrow = block->find_tag(stream, FIFF_MNE_NROW); nrow && *nrow->to_int() != matrix->nrow()) {
71 mne::Log::critical("MNESparseNamedMatrix::read - FIFF_MNE_NROW conflicts with the matrix data.");
72 return nullptr;
73 }
74 if (const auto ncol = block->find_tag(stream, FIFF_MNE_NCOL); ncol && *ncol->to_int() != matrix->ncol()) {
75 mne::Log::critical("MNESparseNamedMatrix::read - FIFF_MNE_NCOL conflicts with the matrix data.");
76 return nullptr;
77 }
78 if (tag = block->find_tag(stream, FIFF_MNE_ROW_NAMES); tag) {
79 matrix->rowlist = FiffStream::split_name_list(tag->to_string());
80 if (static_cast<int>(matrix->rowlist.size()) != matrix->nrow()) {
81 mne::Log::critical("MNESparseNamedMatrix::read - Incorrect number of entries in the row name list.");
82 return nullptr;
83 }
84 }
85 if (tag = block->find_tag(stream, FIFF_MNE_COL_NAMES); tag) {
86 matrix->collist = FiffStream::split_name_list(tag->to_string());
87 if (static_cast<int>(matrix->collist.size()) != matrix->ncol()) {
88 mne::Log::critical("MNESparseNamedMatrix::read - Incorrect number of entries in the column name list.");
89 return nullptr;

91 }
92 return matrix;
93}

97void MNESparseNamedMatrix::write(FiffStream& stream, int kind) const
98{
99 stream.start_block(FIFFB_MNE_NAMED_MATRIX);
100 const fiff_int_t rows = nrow();
101 const fiff_int_t cols = ncol();
102 stream.write_int(FIFF_MNE_NROW, &rows);
103 stream.write_int(FIFF_MNE_NCOL, &cols);
104 if (!rowlist.empty()) {
105 stream.write_name_list(FIFF_MNE_ROW_NAMES, rowlist);
106 }
107 if (!collist.empty()) {
108 stream.write_name_list(FIFF_MNE_COL_NAMES, collist);
109 }
110 stream.write_float_sparse_rcs(kind, data.eigen());
111 stream.end_block(FIFFB_MNE_NAMED_MATRIX);
112}
