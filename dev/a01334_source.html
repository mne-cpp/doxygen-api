76AcqExpected<Config> parseConfig(const AcqUri& uri)
77{
78 Config c;
79 const auto simulate = paramBool(uri, "simulate", false);
80 const auto port = paramInt(uri, "port", 50000, 1, 65535);
81 const auto channels = paramInt(uri, "channels", 46, 1, 1024);
82 const auto rate = paramDouble(uri, "rate", 2048.0, 1.0, 1e5);
83 const auto block = paramInt(uri, "block", 16, 1, 4096);
84 const auto seed = paramInt(uri, "seed", 0, 0, INT64_MAX);
85 const auto realtime = paramBool(uri, "realtime", true);
86 const auto duration = paramDouble(uri, "duration", 0.0, 0.0, 1e7);
87 const auto gain = paramDouble(uri, "gain", 1.0, 0.0, 1e6);
88 auto thread = paramThreadPolicy(uri);
89 for (const AcqError& error : {simulate.error(), port.error(), channels.error(), rate.error(), block.error(), seed.error(), realtime.error(),
90 duration.error(), gain.error(), thread.error()}) {
91 if (!error.ok()) {
92 return error;
93 }
94 }
95 c.simulate = *simulate;
96 if (!uri.target.empty()) {
97 if (c.simulate) {
98 return invalid("target '" + uri.target + "': the simulator takes no address");
99 }
100 c.address = uri.target;
101 }
102 c.port = static_cast<std::uint16_t>(*port);
103 c.channels = static_cast<std::size_t>(*channels);
104 c.rate = *rate;
105 if ((kNatusHeaderFloats + static_cast<std::size_t>(*block) * c.channels) * sizeof(float) > kMaxDatagram) {
106 return invalid(std::format("block={}: {} channels of {} samples do not fit a datagram", *block, c.channels, *block));
107 }
108 c.simulation = {c.channels, c.rate, static_cast<std::size_t>(*block), static_cast<std::uint64_t>(*seed), *realtime, *duration, *gain};
109 c.thread = std::move(*thread);
110 return c;
111}

116 static AcqExpected<std::unique_ptr<AcqBackend>> open(const AcqUri& uri)
117 {
118 auto config = parseConfig(uri);
119 if (!config) {
120 return config.error();
121 }
122 std::unique_ptr<NatusTransport> transport;
123 if (config->simulate) {
124 transport = make_natus_simulator(config->simulation);
125 } else {
126 auto receiver = open_natus_receiver(config->address, config->port);
127 if (!receiver) {
128 return receiver.error();
129 }
130 transport = std::move(*receiver);
131 }
132
133 AcqStreamInfo info;
134 info.name = "eeg";
135 info.sourceId = transport->description();
136 info.format = AcqSampleFormat::Float32;
137 info.layout = AcqLayout::Interleaved;
138 info.sfreq = config->rate;
139 for (std::size_t c = 0; c < config->channels; ++c) {
140 AcqChannelInfo& channel = info.channels.emplace_back();
141 channel.name = std::format("EEG {:03}", c + 1);
142 channel.kind = AcqChannelKind::Eeg;
143 channel.unit = AcqUnit::Volt;
144 channel.scale = kNatusScale;
145 }
146
147 std::unique_ptr<NatusBackend> backend(new NatusBackend(std::move(*config), std::move(transport), uri.toString()));
148 AcqStreamOptions options;
149 options.capacitySamples = static_cast<std::size_t>(std::max(4.0 * backend->m_config.rate, 4096.0));
150 auto writer = backend->addInputStream(std::move(info), options);
151 if (!writer) {
152 return writer.error();
153 }
154 backend->m_writer = *writer;
155 return std::unique_ptr<AcqBackend>(std::move(backend));
156 }

171 AcqError start() override
172 {
173 const auto now = std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now().time_since_epoch()).count();
174 AcqClockModel clock;
175 clock.sfreq = m_config.rate;
176 clock.anchorSample = m_writer->writePosition();
177 clock.anchorHostNs = now;
178 clock.utcOffsetNs =

180 clock.source = AcqClockSource::Nominal;
181 m_writer->setClock(clock);

184 if (!thread) {
185 return thread.error();

188 return {};
189 }

209 AcqExpected<std::size_t> samples_of(std::size_t size) const
210 {
211 constexpr std::size_t kHeaderBytes = kNatusHeaderFloats * sizeof(float);
212 std::array<float, kNatusHeaderFloats> header{};
213 if (size >= kHeaderBytes) {
214 std::memcpy(header.data(), m_datagram.data(), kHeaderBytes);
215 }
216 const float samples = header[1];
217 const float channels = header[2];
218 if (size < kHeaderBytes || !(samples >= 0.0F) || samples != std::trunc(samples) || samples > static_cast<float>(kMaxDatagram)) {
219 return AcqError(AcqErrorCode::DeviceIo, "a datagram is not from the Natus streaming machine");
220 }
221 if (channels != static_cast<float>(m_config.channels)) {
222 return AcqError(AcqErrorCode::DeviceIo,
223 std::format("the Natus streaming machine sends {} channels, channels={} is set", channels, m_config.channels));
224 }
225 const auto count = static_cast<std::size_t>(samples);
226 if (size < kHeaderBytes + count * m_config.channels * sizeof(float)) {
227 return AcqError(AcqErrorCode::DeviceIo, "a datagram of the Natus streaming machine is truncated");
228 }
229 return count;
230 }
231
232 void run()
233 {
234 const std::size_t frame = m_config.channels * sizeof(float);
235 while (!m_stop.load(std::memory_order_acquire)) {
236 auto size = m_transport->receive(m_datagram, kWait);
237 if (!size) {
238 if (size.error() == AcqErrorCode::Timeout) {
239 continue;
240 }
241 if (size.error() == AcqErrorCode::EndOfStream) {
242 m_writer->finish();
243 } else {
244 reportError(size.error());
245 }
246 return;
247 }
248 const auto samples = samples_of(*size);
249 if (!samples) {
250 reportError(samples.error());
251 return;
252 }
253 AcqExpected<AcqWriteView> view = AcqError(AcqErrorCode::Timeout);
254 while (!view && view.error() == AcqErrorCode::Timeout) {
255 if (m_stop.load(std::memory_order_acquire)) {
256 return;
257 }
258 view = m_writer->beginWrite(*samples, kPoll); // a gating reader may not have freed the space yet
259 }
260 if (!view) {
261 reportError(view.error());
262 return;
263 }
264 const std::byte* source = m_datagram.data() + kNatusHeaderFloats * sizeof(float);
265 for (std::size_t p = 0; p < view->partCount(); ++p) {
266 const AcqWritePart& part = view->part(p);
267 std::memcpy(part.at(0, 0), source, part.samples * frame);
268 source += part.samples * frame;
269 }
270 m_writer->commit(*samples);
271 }
272 }
