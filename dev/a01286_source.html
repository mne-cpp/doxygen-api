70std::filesystem::path FigureExporter::save(QWidget& widget,
71 const std::string& name,
72 const std::string& caption,
73 const QSize& size)
74{
75 if (directory().empty()) {
76 return {};
77 }
78 widget.resize(size);
79 widget.setAttribute(Qt::WA_DontShowOnScreen);
80 widget.show();
81 QApplication::processEvents();

84 image.setDevicePixelRatio(kPixelRatio);
85 image.fill(Qt::white);
86 QPainter painter(&image);
87 widget.render(&painter);
88 painter.end();

95std::filesystem::path FigureExporter::save(const QImage& image,
96 const std::string& name,
97 const std::string& caption)
98{
99 const std::filesystem::path dir = directory();
100 std::error_code error;
101 if (dir.empty() || image.isNull() || (std::filesystem::create_directories(dir, error), error)) {
102 return {};
103 }
104 std::string stem;
105 for (const char c : name) {
106 const bool safe = (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '_' || c == '-';
107 if (safe) {
108 stem += c;
109 } else if ((static_cast<unsigned char>(c) & 0xC0) != 0x80) {
110 stem += '_'; // one '_' per character, not per UTF-8 byte
111 }
112 }
113 const std::filesystem::path path = dir / mne::path_from_utf8(std::format("{:02}_{}.png", figureCount.fetch_add(1) + 1, stem));
114 QImage figure = image.convertToFormat(QImage::Format_RGB32);
115 figure.setText("Description", QString::fromStdString(caption));
116 return figure.save(QString::fromStdString(mne::utf8_from_path(path)), "PNG") ? path : std::filesystem::path();
117}
