51TraceFrameBuilder::TraceFrameBuilder()
52{
53 GpuPipeline traces;
54 traces.shader = "channeldata";
55 traces.bindings = {{2 * sizeof(float), false}};
56 traces.attributes = {{0, 0, GpuVertexFormat::Float2, 0}};
57 traces.topology = GpuTopology::LineStrip;
58 traces.blend = GpuBlend::Alpha;
59 m_tracePipeline = m_resources.create_pipeline(std::move(traces));
60
61 GpuPipeline overlay;
62 overlay.shader = "overlay";
63 overlay.bindings = {{4 * sizeof(float), false}};
64 overlay.attributes = {{0, 0, GpuVertexFormat::Float2, 0}, {1, 0, GpuVertexFormat::Float2, 2 * sizeof(float)}};
65 overlay.topology = GpuTopology::TriangleStrip;
66 overlay.blend = GpuBlend::Alpha;
67 m_overlayPipeline = m_resources.create_pipeline(std::move(overlay));
68
69 m_quad = m_resources.create_buffer(GpuBufferUsage::Vertex);
70 m_resources.set_buffer_values(m_quad, std::span<const float>(kQuad));
71
72 m_overlayTexture = m_resources.create_texture(GpuTexture{Image::transparent(1, 1)});
73}

77void TraceFrameBuilder::upload_vertices()
78{
79 const std::vector<TraceVertices>& channels = m_geometry.channels();
80 while (m_channelBuffers.size() < channels.size()) {
81 m_channelBuffers.push_back(m_resources.create_buffer(GpuBufferUsage::Vertex));
82 }
83 for (std::size_t channel = 0; channel < channels.size(); ++channel) {
84 m_resources.set_buffer_values(m_channelBuffers[channel], std::span<const float>(channels[channel].xy));
85 }
86 for (std::size_t channel = channels.size(); channel < m_channelBuffers.size(); ++channel) {
87 m_resources.set_buffer_data(m_channelBuffers[channel], {});
88 }
89}

106GpuFrame TraceFrameBuilder::build(const TraceView& view, const TraceSource& source, const TraceStyle& style, const TraceMarkers& markers,
107 const TextMetrics& metrics, const GpuFrameTarget& target)
108{
109 GpuFrame frame;
110 frame.pixel_width = target.pixel_width;
111 frame.pixel_height = target.pixel_height;
112 frame.device_pixel_ratio = target.device_pixel_ratio;
113 GpuPass& pass = frame.passes.emplace_back();
114 pass.clear_color = style.background;
115 pass.clear_depth = false;
116
117 const int width = view.view_width();
118 const int height = view.view_height();
119 const double spp = view.samples_per_pixel();
120 const double scale = target.device_pixel_ratio > 0.0 ? target.device_pixel_ratio : 1.0;
121 if (target.pixel_width <= 0 || target.pixel_height <= 0 || width <= 0 || height <= 0) {
122 return frame;
123 }
124
125 // Traces
126 const bool hasTraces = source.channel_count() > 0 && spp > 0.0;
127 if (hasTraces) {
128 if (m_geometry.needs_stream(view) && m_geometry.stream(view, source, style.z_score)) {
129 upload_vertices();
130 }
131 for (const TraceDraw& trace : m_geometry.draws(view, source, style)) {
132 const auto channel = static_cast<std::size_t>(trace.channel);
133 if (channel >= m_channelBuffers.size() || channel >= m_geometry.channels().size() || m_geometry.channels()[channel].vertex_count() < 2) {
134 continue;
135 }
136 GpuDraw& draw = pass.draws.emplace_back();
137 draw.pipeline = m_tracePipeline;
138 draw.vertex_buffers = {m_channelBuffers[channel]};
139 draw.count = static_cast<std::uint32_t>(m_geometry.channels()[channel].vertex_count());
140 draw.set_uniforms(0, trace.uniforms);
141 }
142 }
143
144 // Annotations, events and epoch markers over the prefetch window, with the per-second bands
145 if (hasTraces && view.sfreq() > 0.0) {
146 const double visible = width * spp;
147 const double extra = overlay_prefetch() * visible;
148 const int overlayWidth = static_cast<int>(std::ceil((1.0 + 2.0 * overlay_prefetch()) * width));
149 const double scroll = view.scroll_sample();
150 if (m_overlayDirty || overlayWidth != m_overlayWidth || height != m_overlayHeight || scale != m_overlayScale || spp != m_overlaySpp || scroll < m_overlayFirstSample || scroll + visible > m_overlayFirstSample + m_overlayTotalSamples) {
151 m_overlayFirstSample = scroll - extra;
152 m_overlayTotalSamples = visible + 2.0 * extra;
153 m_overlayWidth = overlayWidth;
154 m_overlayHeight = height;
155 m_overlayScale = scale;
156 m_overlaySpp = spp;
157 m_overlayDirty = false;
158 Canvas canvas(overlayWidth, height);
159 markers.draw(canvas, metrics, Font{}, view.annotations(), m_overlayFirstSample, overlayWidth / m_overlayTotalSamples);
160 m_resources.set_texture(m_overlayTexture, canvas.display_list(), scale);
161 }
162 TraceOverlayUniforms uniforms;
163 uniforms.scroll_sample = static_cast<float>(scroll);
164 uniforms.samples_per_pixel = static_cast<float>(spp);
165 uniforms.view_width = static_cast<float>(width);
166 uniforms.sfreq = static_cast<float>(view.sfreq());
167 uniforms.first_file_sample = static_cast<float>(view.first_file_sample());
168 uniforms.grid_enabled = style.grid ? 1.f : 0.f;
169 uniforms.overlay_first_sample = static_cast<float>(m_overlayFirstSample);
170 uniforms.overlay_total_samples = static_cast<float>(m_overlayTotalSamples);
171 pass.draws.push_back(overlay_draw(m_overlayTexture, uniforms));
172 }
173
174 // Crosshair, scalebars, ruler and annotation selection, mapped 1:1 to the screen without bands
175 if (view.overlays_visible()) {
176 if (m_hudTexture == 0) {
177 m_hudTexture = m_resources.create_texture(GpuTexture{view.render_overlays(metrics), scale});
178 } else {
179 m_resources.set_texture(m_hudTexture, view.render_overlays(metrics), scale);
180 }
181 TraceOverlayUniforms uniforms;
182 uniforms.view_width = static_cast<float>(width);
183 uniforms.overlay_total_samples = static_cast<float>(width);
184 pass.draws.push_back(overlay_draw(m_hudTexture, uniforms));
185 }
186 return frame;
187}
