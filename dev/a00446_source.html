322int mne_apply_filter(const MNEFilterDef& filter, FilterData *d, float *data, int ns, int zero_pad, float dc_offset, int kind)
323/*
324 * Do the magick trick
325 */
326{
327 int k,p,n;
328 float *freq_resp;
329
330 if (ns != filter.size + 2*filter.taper_size) {
331 printf("Incorrect data length in apply_filter");
332 return FAIL;
333 }
334 /*
335 * Zero padding
336 */
337 if (zero_pad) {
338 for (k = 0; k < filter.taper_size; k++)
339 data[k] = 0.0;
340 for (k = filter.taper_size + filter.size; k < ns; k++)
341 data[k] = 0.0;
342 }
343 if (!filter.filter_on) /* Nothing else to do */
344 return OK;
345 /*
346 * Make things nice by compensating for the dc offset
347 */
348 if (dc_offset != 0.0) {
349 for (k = filter.taper_size; k < filter.taper_size + filter.size; k++)
350 data[k] = data[k] - dc_offset;
351 }
352 if (!d)
353 return OK;
354 if (d->freq_resp.empty())
355 return OK;
356 /*
357 * Next comes the FFT
358 */
359 mne_fft_ana(data,ns,d->precalc);
360 /*
361 * Multiply with the frequency response
362 * See FFTpack doc for details of the arrangement
363 */
364 n = ns % 2 == 0 ? ns/2 : (ns+1)/2;
365 p = 0;
366 /*
367 * No imaginary part for the DC component
368 */
369 if (kind == FIFFV_EOG_CH)
370 freq_resp = d->eog_freq_resp.data();
371 else
372 freq_resp = d->freq_resp.data();
373 data[p] = data[p]*freq_resp[0]; p++;
374 /*
375 * The other components
376 */
377 for (k = 1 ; k < n ; k++) {
378 data[p] = data[p]*freq_resp[k]; p++;
379 data[p] = data[p]*freq_resp[k]; p++;
380 }
381 /*
382 * Then the last value
383 */
384 if (ns % 2 == 0)
385 data[p] = data[p]*freq_resp[k];
386
387 mne_fft_syn(data,ns,d->precalc);
388
389 return OK;
390}

392std::unique_ptr<FilterData> mne_create_filter_response(const MNEFilterDef& filter,
393 float sfreq,
394 int *highpass_effective)
395/*
396 * Create a frequency response
397 */
398{
399 int resp_size;
400 int k,s,w,f;
401 int highpasss,lowpasss;
402 int highpass_widths,lowpass_widths;
403 float lowpass,highpass,lowpass_width,highpass_width;
404 float *freq_resp;
405 float pi4 = M_PI/4.0;
406 float mult,add,c;
407
408 resp_size = (filter.size + 2*filter.taper_size)/2 + 1;
409
410 auto filter_data = std::make_unique<FilterData>(resp_size);
411 *highpass_effective = FALSE;
412
413 for (f = 0; f < 2; f++) {
414 highpass = f == 0 ? filter.highpass : filter.eog_highpass;
415 highpass_width = f == 0 ? filter.highpass_width : filter.eog_highpass_width;
416 lowpass = f == 0 ? filter.lowpass : filter.eog_lowpass;
417 lowpass_width = f == 0 ? filter.lowpass_width : filter.eog_lowpass_width;
418 freq_resp = f == 0 ? filter_data->freq_resp.data() : filter_data->eog_freq_resp.data();
419 /*
420 * Start simple first
421 */
422 highpasss = ((resp_size-1)*highpass)/(0.5*sfreq);
423 lowpasss = ((resp_size-1)*lowpass)/(0.5*sfreq);
424
425 lowpass_widths = ((resp_size-1)*lowpass_width)/(0.5*sfreq);
426 lowpass_widths = (lowpass_widths+1)/2; /* What user specified */
427
428 if (filter.highpass_width > 0.0) {
429 highpass_widths = ((resp_size-1)*highpass_width)/(0.5*sfreq);
430 highpass_widths = (highpass_widths+1)/2; /* What user specified */
431 }
432 else
433 highpass_widths = 3; /* Minimal */
434
435 if (filter.filter_on) {
436 printf("filter : %7.3f ... %6.1f Hz bins : %d ... %d of %d hpw : %d lpw : %d\n",
437 highpass,
438 lowpass,
439 highpasss,
440 lowpasss,
441 resp_size,
442 highpass_widths,
443 lowpass_widths);
444 }
445 if (highpasss > highpass_widths + 1) {
446 w = highpass_widths;
447 mult = 1.0/w;
448 add = 3.0;
449 for (k = 0; k < highpasss-w+1; k++)
450 freq_resp[k] = 0.0;
451 for (k = -w+1, s = highpasss-w+1; k < w; k++, s++) {
452 if (s >= 0 && s < resp_size) {
453 c = cos(pi4*(k*mult+add));
454 freq_resp[s] = freq_resp[s]*c*c;
455 }
456 }
457 *highpass_effective = TRUE;
458 }
459 else
460 *highpass_effective = *highpass_effective || (filter.highpass == 0.0);
461
462 if (lowpass_widths > 0) {
463 w = lowpass_widths;
464 mult = 1.0/w;
465 add = 1.0;
466 for (k = -w+1, s = lowpasss-w+1; k < w; k++, s++) {
467 if (s >= 0 && s < resp_size) {
468 c = cos(pi4*(k*mult+add));
469 freq_resp[s] = freq_resp[s]*c*c;
470 }
471 }
472 for (k = s; k < resp_size; k++)
473 freq_resp[k] = 0.0;
474 }
475 else {
476 for (k = lowpasss; k < resp_size; k++)
477 freq_resp[k] = 0.0;
478 }
479 if (filter.filter_on) {
480 if (*highpass_effective)
481 printf("Highpass filter will work as specified.\n");
482 else
483 printf("NOTE: Highpass filter omitted due to a too low corner frequency.\n");
484 }
485 else
486 printf("NOTE: Filter is presently switched off.\n");
487 }
488 return filter_data;
489}

801#endif
802 this->filt_bufs.resize(nfilt_buf);

817 this->filt_ring = std::make_unique<RingBuffer>(nring_buf);
818 add_filter_response(&highpass_effective);
819
820 return;
821}

862int MNERawData::compensate_buffer(MNERawBufDef *buf)
863/*
864 * Apply compensation channels
865 */
866{
867
868 if (!comp)
869 return OK;
870 if (!comp->undo && !comp->current)
871 return OK;
872 if (buf->comp_status == comp_now)
873 return OK;
874 if (buf->vals.size() == 0)
875 return OK;
876 /*
877 * vals is now a RowMajorMatrixXf — wrap in a column-major MatrixXf for compensation
878 */
879 {
880 Eigen::MatrixXf dataMat = buf->vals; /* implicit copy/conversion */
881
882 if (comp->undo) {
883 std::swap(comp->current, comp->undo);
884 /*
885 * Undo the previous compensation
886 */
887 if (comp->apply_transpose(FALSE, dataMat) != OK) {
888 std::swap(comp->current, comp->undo);
889 goto bad;
890 }
891 std::swap(comp->current, comp->undo);
892 }
893 if (comp->current) {
894 /*
895 * Apply new compensation
896 */
897 if (comp->apply_transpose(TRUE, dataMat) != OK)
898 goto bad;
899 }
900 /*
901 * Copy result back to buf->vals
902 */
903 buf->vals = dataMat;
904 }
905 buf->comp_status = comp_now;
906 return OK;
907
908bad :
909 return FAIL;

914int MNERawData::pick_data(mneChSelection sel, int firsts, int ns, float **picked)
915/*
916 * Data from a selection of channels
917 */
918{

920 int ns2,s2;
921 MNERawBufDef* this_buf;
922 float *values;
923 int need_some;
924
925 float **deriv_vals = NULL;
926 int deriv_ns = 0;
927 int nderiv = 0;
928
929 if (firsts < first_samp) {
930 for (s = 0, p = firsts; p < first_samp; s++, p++) {
931 if (sel)
932 for (c = 0; c < sel->nchan; c++)
933 picked[c][s] = 0.0;
934 else
935 for (c = 0; c < info->nchan; c++)
936 picked[c][s] = 0.0;
937 }
938 ns = ns - s;
939 firsts = first_samp;
940 }
941 else
942 s = 0;
943 /*
944 * There is possibly nothing to do
945 */
946 if (sel) {
947 for (c = 0, need_some = FALSE; c < sel->nchan; c++) {
948 if (sel->pick[c] >= 0 || sel->pick_deriv[c] >= 0) {
949 need_some = TRUE;
950 break;
951 }
952 }
953 if (!need_some)
954 return OK;
955 }
956 /*
957 * Have to to the hard work
958 */
959 for (k = 0, this_buf = bufs.data(), s = 0; k < (int)bufs.size(); k++, this_buf++) {
960 if (this_buf->lasts >= firsts) {
961 start = firsts - this_buf->firsts;
962 if (start < 0)
963 start = 0;
964 if (this_buf->is_skip) {
965 for (p = start; p < this_buf->ns && ns > 0; p++, ns--, s++) {
966 if (sel) {
967 for (c = 0; c < sel->nchan; c++)
968 if (sel->pick[c] >= 0)
969 picked[c][s] = 0.0;
970 }
971 else {
972 for (c = 0; c < info->nchan; c++)
973 picked[c][s] = 0.0;
974 }
975 }
976 }
977 else {
978 /*
979 * Load the buffer
980 */
981 if (load_one_buffer(this_buf) != OK)
982 return FAIL;
983 /*
984 * Apply compensation
985 */
986 if (compensate_buffer(this_buf) != OK)
987 return FAIL;
988 ns2 = s2 = 0;
989 if (sel) {
990 /*
991 * Do we need the derived channels?
992 */
993 if (sel->nderiv > 0 && deriv_matched) {
994 if (deriv_ns < this_buf->ns || nderiv != deriv_matched->deriv_data->nrow) {
995 FREE_CMATRIX_36(deriv_vals);
996 deriv_vals = ALLOC_CMATRIX_36(deriv_matched->deriv_data->nrow,this_buf->ns);
997 nderiv = deriv_matched->deriv_data->nrow;
998 deriv_ns = this_buf->ns;
999 }
1000 if (mne_sparse_mat_mult2(deriv_matched->deriv_data->data.get(),this_buf->vals,this_buf->ns,deriv_vals) == FAIL) {
1001 FREE_CMATRIX_36(deriv_vals);
1002 return FAIL;
1003 }
1004 }
1005 for (c = 0; c < sel->nchan; c++) {
1006 /*
1007 * First pick the ordinary channels...
1008 */
1009 if (sel->pick[c] >= 0) {
1010 for (p = start, s2 = s, ns2 = ns; p < this_buf->ns && ns2 > 0; p++, ns2--, s2++)
1011 picked[c][s2] = this_buf->vals(sel->pick[c], p);
1012 }
1013 /*
1014 * ...then the derived ones
1015 */
1016 else if (sel->pick_deriv[c] >= 0 && deriv_matched) {
1017 values = deriv_vals[sel->pick_deriv[c]];
1018 for (p = start, s2 = s, ns2 = ns; p < this_buf->ns && ns2 > 0; p++, ns2--, s2++)
1019 picked[c][s2] = values[p];
1020 }
1021 }
1022 }
1023 else {
1024 for (c = 0; c < info->nchan; c++)
1025 for (p = start, s2 = s, ns2 = ns; p < this_buf->ns && ns2 > 0; p++, ns2--, s2++)
1026 picked[c][s2] = this_buf->vals(c, p);
1027 }
1028 s = s2;
1029 ns = ns2;
1030 }
1031 if (ns == 0)
1032 break;
1033 }
1034 }
1035 /*
1036 * Extend with the last available sample or zero if the request is beyond the data
1037 */
1038 if (s > 0) {
1039 fills = s-1;
1040 for (; ns > 0; ns--, s++) {
1041 if (sel)
1042 for (c = 0; c < sel->nchan; c++)
1043 picked[c][s] = picked[c][fills];
1044 else
1045 for (c = 0; c < info->nchan; c++)
1046 picked[c][s] = picked[c][fills];
1047 }
1048 }
1049 else {
1050 for (; ns > 0; ns--, s++) {
1051 if (sel)
1052 for (c = 0; c < sel->nchan; c++)
1053 picked[c][s] = 0;
1054 else
1055 for (c = 0; c < info->nchan; c++)
1056 picked[c][s] = 0;
1057 }
1058 }
1059 FREE_CMATRIX_36(deriv_vals);
1060 return OK;
1061}

1243 float dc_offset;
1244 int deriv_ns = 0;
1245 int nderiv = 0;
1246 int filter_was;
1247
1248 if (!filter->filter_on)
1249 return pick_data_proj(sel,firsts,ns,picked);
1250
1251 if (sel) {
1252 for (s = 0; s < ns; s++)
1253 for (c = 0; c < sel->nchan; c++)
1254 picked[c][s] = 0.0;
1255 }
1256 else {
1257 for (s = 0; s < ns; s++)
1258 for (c = 0; c < info->nchan; c++)
1259 picked[c][s] = 0.0;
1260 }
1261 lasts = firsts + ns - 1;
1262 /*
1263 * Take into account the initial dc offset (compensate and project)
1264 */
1265 if (first_sample_val.size() > 0) {
1266 dc = first_sample_val;
1267 /*
1268 * Is this correct??
1269 */
1270 if (comp && comp->current)
1271 if (comp->apply(TRUE,dc) != OK)
1272 goto bad;
1273 if (proj)
1274 if (proj->project_vector(dc.data(),info->nchan,TRUE) != OK)
1275 goto bad;
1276 }
1277 filter_was = filter->filter_on;
1278 /*
1279 * Find the first buffer to consider
1280 */
1281 for (k = 0, this_buf = filt_bufs.data(); k < (int)filt_bufs.size(); k++, this_buf++) {
1282 if (this_buf->lasts >= firsts)
1283 break;

1286#ifdef DEBUG

1290 * Load the buffer first and apply projection
1291 */
1292 if (load_one_filt_buf(this_buf) != OK)
1293 goto bad;
1294 /*
1295 * Then filter all relevant channels (not stimuli)
1296 */
1297 if (sel) {
1298 for (c = 0; c < sel->nchan; c++) {
1299 if (sel->pick[c] >= 0) {
1300 if (!this_buf->ch_filtered[sel->pick[c]]) {
1301 /*
1302 * Do not filter stimulus channels
1303 */
1304 dc_offset = 0.0;

1306 filter->filter_on = FALSE;
1307 else if (dc.size() > 0)
1308 dc_offset = dc[sel->pick[c]];

1311 filter->filter_on = filter_was;

1313 }
1314 this_buf->ch_filtered[sel->pick[c]] = TRUE;
1315 filter->filter_on = filter_was;
1316 }
1317 }
1318 }
1319 /*
1320 * Also check channels included in derivations if they are used
1321 */
1322 if (sel->nderiv > 0 && deriv_matched) {
1323 MNEDeriv* der = deriv_matched.get();
1324 for (c = 0; c < der->deriv_data->ncol; c++) {
1325 if (der->in_use[c] > 0 &&
1326 !this_buf->ch_filtered[c]) {
1327 /*
1328 * Do not filter stimulus channels
1329 */
1330 dc_offset = 0.0;
1331 if (info->chInfo[c].kind == FIFFV_STIM_CH)
1332 filter->filter_on = FALSE;
1333 else if (dc.size() > 0)
1334 dc_offset = dc[c];
1335 if (mne_apply_filter(*filter,filter_data.get(),this_buf->vals.row(c).data(),this_buf->ns,TRUE,
1336 dc_offset,info->chInfo[c].kind) != OK) {
1337 filter->filter_on = filter_was;
1338 goto bad;
1339 }
1340 this_buf->ch_filtered[c] = TRUE;
1341 filter->filter_on = filter_was;
1342 }
1343 }
1344 }
1345 }
1346 else {
1347 /*
1348 * Simply filter all channels if there is no selection
1349 */
1350 for (c = 0; c < info->nchan; c++) {
1351 if (!this_buf->ch_filtered[c]) {
1352 /*
1353 * Do not filter stimulus channels
1354 */
1355 dc_offset = 0.0;
1356 if (info->chInfo[c].kind == FIFFV_STIM_CH)
1357 filter->filter_on = FALSE;
1358 else if (dc.size() > 0)
1359 dc_offset = dc[c];
1360 if (mne_apply_filter(*filter,filter_data.get(),this_buf->vals.row(c).data(),this_buf->ns,TRUE,
1361 dc_offset,info->chInfo[c].kind) != OK) {
1362 filter->filter_on = filter_was;
1363 goto bad;
1364 }

1366 filter->filter_on = filter_was;
1367 }
1368 }
1369 }
1370 /*
1371 * Decide the picking limits
1372 */
1373 if (firsts >= this_buf->firsts) {
1374 bs1 = firsts - this_buf->firsts;
1375 s1 = 0;
1376 }
1377 else {
1378 bs1 = 0;
1379 s1 = this_buf->firsts - firsts;
1380 }
1381 if (lasts >= this_buf->lasts) {
1382 bs2 = this_buf->ns;
1383 s2 = this_buf->lasts - lasts + ns;
1384 }
1385 else {
1386 bs2 = lasts - this_buf->lasts + this_buf->ns;
1387 s2 = ns;
1388 }
1389#ifdef DEBUG
1390 printf("buf : %d..%d %d\n",bs1,bs2,bs2-bs1);
1391 printf("dest : %d..%d %d\n",s1,s2,s2-s1);
1392#endif
1393 /*
1394 * Then pick data from all relevant channels
1395 */
1396 if (sel) {
1397 if (sel->nderiv > 0 && deriv_matched) {
1398 /*
1399 * Compute derived data if we need it
1400 */
1401 if (deriv_ns < this_buf->ns || nderiv != deriv_matched->deriv_data->nrow) {
1402 FREE_CMATRIX_36(deriv_vals);
1403 deriv_vals = ALLOC_CMATRIX_36(deriv_matched->deriv_data->nrow,this_buf->ns);
1404 nderiv = deriv_matched->deriv_data->nrow;
1405 deriv_ns = this_buf->ns;
1406 }
1407 if (mne_sparse_mat_mult2(deriv_matched->deriv_data->data.get(),this_buf->vals,this_buf->ns,deriv_vals) == FAIL)
1408 goto bad;
1409 }
1410 for (c = 0; c < sel->nchan; c++) {
1411 /*
1412 * First the ordinary channels
1413 */
1414 if (sel->pick[c] >= 0) {
1415 values = this_buf->vals.row(sel->pick[c]).data();
1416 for (s = s1, bs = bs1; s < s2; s++, bs++)
1417 picked[c][s] += values[bs];
1418 }
1419 else if (sel->pick_deriv[c] >= 0 && deriv_matched) {
1420 values = deriv_vals[sel->pick_deriv[c]];
1421 for (s = s1, bs = bs1; s < s2; s++, bs++)
1422 picked[c][s] += values[bs];
1423 }
1424 }
1425 }
1426 else {
1427 for (c = 0; c < info->nchan; c++) {
1428 values = this_buf->vals.row(c).data();
1429 for (s = s1, bs = bs1; s < s2; s++, bs++)
1430 picked[c][s] += values[bs];
1431 }
1432 }
1433 }
1434 (void)bs2; // squash compiler warning, this is unused
1435 FREE_CMATRIX_36(deriv_vals);
1436 return OK;
1437
1438bad : {
1439 FREE_CMATRIX_36(deriv_vals);
1440 return FAIL;
1441 }
1442}
