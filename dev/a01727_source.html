69// Reads @p count big-endian values of type T and converts each one with @p convert into @p out
70template<typename T, typename Out, typename Convert>
71bool readBigEndianInto(mne::ByteStream& stream, std::vector<Out>& out, std::size_t count, Convert convert)
72{
73 std::vector<T> values(count);
74 if (!stream.read_big_endian(values.data(), count))
75 return false;
76 out.resize(count);
77 std::transform(values.begin(), values.end(), out.begin(), convert);
78 return true;
79}

87bool MriMghIO::read(const std::string& mgzFile,
88 MriVolData& volData,
89 std::vector<FiffCoordTrans>& additionalTrans,
90 const std::string& subjectMriDir,
91 bool verbose)
92{
93 volData.fileName = mgzFile;
94
95 // Step 1: Get raw (decompressed) bytes
96 const bool isCompressed = mne::to_lower(mgzFile).ends_with(".mgz");
97 const auto fileData = mne::read_file_bytes(mne::path_from_utf8(mgzFile), isCompressed ? mne::Compression::Gzip : mne::Compression::None);
98 if (!fileData) {
99 Log::critical("MriMghIO::read - {}", fileData.error().what());
100 return false;
101 }
102
103 if (fileData->size() < MRI_MGH_DATA_OFFSET) {
104 Log::critical("MriMghIO::read - File {} is too small to be a valid MGH file ({} bytes)", mgzFile, fileData->size());
105 return false;
106 }
107
108 // Step 2: Parse header
109 if (!parseHeader(*fileData, volData, verbose)) {
110 return false;
111 }
112
113 // Step 3: Build the voxel -> surface RAS transform
114 Matrix4f vox2ras = volData.computeVox2Ras();
115 volData.voxelSurfRasT = FiffCoordTrans(
116 FIFFV_COORD_MRI_SLICE, FIFFV_COORD_MRI, vox2ras, true);
117
118 if (verbose) {
119 Log::info("Voxel -> FsSurface RAS transform:");
120 for (int r = 0; r < 4; ++r) {
121 Log::info(" {:10.6f} {:10.6f} {:10.6f} {:10.6f}", vox2ras(r, 0), vox2ras(r, 1), vox2ras(r, 2), vox2ras(r, 3));
122 }
123 }
124
125 // Surface RAS -> scanner RAS: a shift by the volume centre (MNE-C mne_mri_add_transforms).
126 additionalTrans.push_back(FiffCoordTrans(FIFFV_COORD_MRI, FIFFV_MNE_COORD_RAS, Matrix3f::Identity(), Vector3f(volData.c_ras / 1000.0f)));
127
128 // Step 4: Read voxel data
129 if (!readVoxelData(*fileData, volData)) {
130 return false;
131 }
132
133 // Step 5: Parse footer (optional)
134 parseFooter(*fileData, volData, additionalTrans, subjectMriDir, verbose);
135
136 if (verbose) {
137 Log::info("Read {} slices from {} ({}x{} pixels)", volData.slices.size(), mgzFile, volData.width, volData.height);
138 }
139
140 return true;
141}

145bool MriMghIO::parseHeader(const std::vector<std::byte>& data, MriVolData& volData, bool verbose)
146{
147 //
148 // MGH header layout (all big-endian):
149 // Bytes 0-3: version (int32)
150 // Bytes 4-7: width (int32)
151 // Bytes 8-11: height (int32)
152 // Bytes 12-15: depth (int32)
153 // Bytes 16-19: nframes (int32)
154 // Bytes 20-23: type (int32)
155 // Bytes 24-27: dof (int32)
156 // Bytes 28-29: goodRASflag (int16)
157 // Bytes 30-41: spacingX/Y/Z (3×float32) — only if goodRASflag > 0
158 // Bytes 42-77: Mdc (9×float32) — direction cosines, only if goodRASflag > 0
159 // Bytes 78-89: c_ras (3×float32) — center RAS, only if goodRASflag > 0
160 // Bytes 90-283: unused (padding)
161 //
162
163 mne::MemoryByteStream stream({data.begin(), data.begin() + MRI_MGH_DATA_OFFSET});
164
165 std::int32_t header[7] = {};
166 stream.read_big_endian(header, 7);
167 const auto [version, width, height, depth, nframes, type, dof] = header;
168
169 if (version != MRI_MGH_VERSION) {
170 Log::critical("MriMghIO::parseHeader - Unknown MGH version: {}", version);
171 return false;
172 }
173
174 volData.version = version;
175 volData.width = width;
176 volData.height = height;
177 volData.depth = depth;
178 volData.nframes = nframes;
179 volData.type = type;
180 volData.dof = dof;
181
182 if (verbose) {
183 Log::info("MGH file: {}x{}x{}, {} frame(s), type={}", width, height, depth, nframes, type);
184 }
185
186 std::int16_t goodRASflag = 0;
187 stream.read_big_endian(goodRASflag);
188 volData.rasGood = (goodRASflag > 0);
189
190 if (goodRASflag > 0) {
191 // Voxel sizes, then the direction cosines (x, y, z columns) and the centre RAS
192 float geometry[15] = {};
193 stream.read_big_endian(geometry, 15);
194 volData.xsize = geometry[0];
195 volData.ysize = geometry[1];
196 volData.zsize = geometry[2];
197 volData.x_ras = Vector3f(geometry[3], geometry[4], geometry[5]);
198 volData.y_ras = Vector3f(geometry[6], geometry[7], geometry[8]);
199 volData.z_ras = Vector3f(geometry[9], geometry[10], geometry[11]);
200 volData.c_ras = Vector3f(geometry[12], geometry[13], geometry[14]);
201 }
202 // Else: default values from MriVolData constructor are used
203
204 if (verbose) {
205 Log::info("Voxel sizes: {:.4f} x {:.4f} x {:.4f} mm", volData.xsize, volData.ysize, volData.zsize);
206 Log::info("goodRAS: {}", int(goodRASflag));
207 Log::info("c_ras: {:.4f} {:.4f} {:.4f}", volData.c_ras[0], volData.c_ras[1], volData.c_ras[2]);
208 }
209
210 return true;
211}

215bool MriMghIO::readVoxelData(const std::vector<std::byte>& data, MriVolData& volData)
216{
217 //
218 // Read voxel data starting at byte 284 (MRI_MGH_DATA_OFFSET).
219 // Data layout in MGH: [width][height][depth][frames] in Fortran order (x fastest).
220 // Only the first frame is read.
221 //
222
223 const int bpv = bytesPerVoxel(volData.type);
224 if (bpv == 0) {
225 Log::critical("MriMghIO::readVoxelData - Unsupported MGH data type: {}", volData.type);
226 return false;
227 }
228
229 const std::size_t nPixels = static_cast<std::size_t>(volData.width) * static_cast<std::size_t>(volData.height);
230 const std::size_t nslice = static_cast<std::size_t>(volData.depth);
231 const std::size_t frameSize = nPixels * nslice * static_cast<std::size_t>(bpv);
232 if (data.size() < MRI_MGH_DATA_OFFSET + frameSize) {
233 Log::critical("MriMghIO::readVoxelData - File too small for expected data size");
234 return false;
235 }
236
237 mne::MemoryByteStream stream({data.begin() + MRI_MGH_DATA_OFFSET, data.begin() + std::ptrdiff_t(MRI_MGH_DATA_OFFSET + frameSize)});
238 volData.slices.resize(nslice);
239
240 // Build the vox2ras transform for per-slice transforms
241 Matrix4f vox2ras = volData.computeVox2Ras();
242
243 for (std::size_t k = 0; k < nslice; ++k) {
244 MriSlice& slice = volData.slices[k];
245 slice.width = volData.width;
246 slice.height = volData.height;
247 slice.dimx = volData.xsize / 1000.0f; // mm -> meters
248 slice.dimy = volData.ysize / 1000.0f;
249 slice.scale = 1.0f;
250
251 switch (volData.type) {
252 case MRI_UCHAR:
253 slice.pixelFormat = FIFFV_MRI_PIXEL_BYTE;
254 readBigEndianInto<std::uint8_t>(stream, slice.pixels, nPixels, [](std::uint8_t v) { return v; });
255 break;
256 case MRI_SHORT:
257 slice.pixelFormat = FIFFV_MRI_PIXEL_WORD;
258 readBigEndianInto<std::int16_t>(stream, slice.pixelsWord, nPixels, [](std::int16_t v) { return static_cast<unsigned short>(v < 0 ? 0 : v); });
259 break;
260 case MRI_INT:
261 // Convert INT to FLOAT
262 slice.pixelFormat = FIFFV_MRI_PIXEL_FLOAT;
263 readBigEndianInto<std::int32_t>(stream, slice.pixelsFloat, nPixels, [](std::int32_t v) { return static_cast<float>(v); });
264 break;
265 case MRI_FLOAT:
266 slice.pixelFormat = FIFFV_MRI_PIXEL_FLOAT;
267 readBigEndianInto<float>(stream, slice.pixelsFloat, nPixels, [](float v) { return v; });
268 break;
269 }
270
271 //
272 // Build per-slice coordinate transform (slice -> MRI surface RAS).
273 // For each slice k:
274 // sliceOrigin = vox2ras * [0, 0, k, 1]^T
275 // sliceRot = vox2ras rotation columns (x, y, z pixel axes)
276 //
277 const float kf = static_cast<float>(k);
278 Vector3f sliceOrigin;
279 sliceOrigin(0) = vox2ras(0, 2) * kf + vox2ras(0, 3);
280 sliceOrigin(1) = vox2ras(1, 2) * kf + vox2ras(1, 3);
281 sliceOrigin(2) = vox2ras(2, 2) * kf + vox2ras(2, 3);
282
283 Matrix3f sliceRot;
284 sliceRot.col(0) = vox2ras.block<3, 1>(0, 0); // x-pixel direction
285 sliceRot.col(1) = vox2ras.block<3, 1>(0, 1); // y-pixel direction
286 sliceRot.col(2) = vox2ras.block<3, 1>(0, 2); // z (normal) direction
287
288 slice.trans = FiffCoordTrans(FIFFV_COORD_MRI_SLICE, FIFFV_COORD_MRI, sliceRot, sliceOrigin);
289 }
290
291 return true;
292}

296bool MriMghIO::parseFooter(const std::vector<std::byte>& data,
297 MriVolData& volData,
298 std::vector<FiffCoordTrans>& additionalTrans,
299 const std::string& subjectMriDir,
300 bool verbose)
301{
302 //
303 // The footer starts after the voxel data.
304 // It contains (in order):
305 // 1. Scan parameters: TR(f32), flipAngle(f32), TE(f32), TI(f32), FoV(f32)
306 // 2. Tags: tagType(i32) + tagLen(i32 or i64) + tagData
307 //
308
309 int bpv = bytesPerVoxel(volData.type);
310 if (bpv == 0) {
311 bpv = 1; // Fallback for footer calculation
312 }
313
314 const std::size_t frameSize = static_cast<std::size_t>(volData.width) * static_cast<std::size_t>(volData.height) * static_cast<std::size_t>(volData.depth) * static_cast<std::size_t>(bpv);
315 const std::size_t footerPos = MRI_MGH_DATA_OFFSET + frameSize;
316
317 // Scan parameters take 5 x float32; without them there is no footer
318 constexpr std::size_t kScanParamBytes = 5 * sizeof(float);
319 if (data.size() < footerPos + kScanParamBytes) {
320 return true;
321 }
322
323 mne::MemoryByteStream stream({data.begin() + std::ptrdiff_t(footerPos), data.end()});
324 float scan[5] = {};
325 stream.read_big_endian(scan, 5);
326 volData.TR = scan[0];
327 volData.flipAngle = scan[1];
328 volData.TE = scan[2];
329 volData.TI = scan[3];
330 volData.FoV = scan[4];
331
332 // Parse tags
333 while (!stream.at_end()) {
334 std::int32_t tagType = 0;
335 if (!stream.read_big_endian(tagType))
336 break;
337
338 std::int64_t tagLen = 0;
339 // TAG_OLD_SURF_GEOM (20) and TAG_OLD_MGH_XFORM (30) have a 4-byte length, newer tags an 8-byte one
340 if (tagType == MGH_TAG_OLD_SURF_GEOM || tagType == MGH_TAG_OLD_MGH_XFORM) {
341 std::int32_t len32 = 0;
342 if (!stream.read_big_endian(len32))
343 break;
344 tagLen = len32;
345 } else if (!stream.read_big_endian(tagLen)) {
346 break;
347 }
348
349 if (tagLen <= 0 || static_cast<std::uint64_t>(tagLen) > data.size())
350 break;
351
352 std::string tagData(static_cast<std::size_t>(tagLen), '\0');
353 if (stream.read(tagData.data(), tagData.size()) != tagData.size())
354 break;
355
356 if (tagType == MGH_TAG_MGH_XFORM) {
357 // TAG_MGH_XFORM: path to talairach.xfm; FreeSurfer includes the terminating NUL in the length
358 std::string xfmPath(mne::trimmed(std::string_view(tagData).substr(0, tagData.find('\0'))));
359 volData.talairachXfmPath = xfmPath;
360
361 if (verbose) {
362 Log::info("Found Talairach transform reference: {}", xfmPath);
363 }
364
365 // Resolve relative paths using subject MRI directory
366 if (!mne::path_from_utf8(xfmPath).is_absolute() && !subjectMriDir.empty()) {
367 xfmPath = mne::utf8_from_path(mne::path_from_utf8(subjectMriDir) / "transforms" / mne::path_from_utf8(xfmPath));
368 }
369
370 FiffCoordTransSet talairach;
371 if (talairach.addTalairach(mne::path_from_utf8(xfmPath))) {
372 additionalTrans.push_back(talairach.RAS_MNI_tal_t);
373 additionalTrans.push_back(talairach.MNI_tal_tal_gtz_t);
374 additionalTrans.push_back(talairach.MNI_tal_tal_ltz_t);
375 if (verbose) {
376 Log::info("Read Talairach transform from {}", xfmPath);
377 }
378 } else if (verbose) {
379 Log::warning("Talairach transform not readable: {}", xfmPath);
380 }
381 }
382 }
383
384 return true;
385}
