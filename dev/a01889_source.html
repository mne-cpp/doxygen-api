49ChannelWindow::Api resolveApi(ChannelWindow::Api api)
50{
51 if (api != ChannelWindow::Api::Default)
52 return api;
53#if defined(Q_OS_MACOS) || defined(Q_OS_IOS)
54 return ChannelWindow::Api::Metal;

179void ChannelWindow::initRhi()
180{
181 switch (m_api) {
182 case Api::Null: {
183 QRhiNullInitParams params;
184 m_pRhi.reset(QRhi::create(QRhi::Null, &params));
185 break;
186 }
187#if QT_CONFIG(opengl)
188 case Api::OpenGL: {
189 m_pFallbackSurface.reset(QRhiGles2InitParams::newFallbackSurface());
190 QRhiGles2InitParams params;
191 params.fallbackSurface = m_pFallbackSurface.get();
192 params.window = this;
193 m_pRhi.reset(QRhi::create(QRhi::OpenGLES2, &params));
194 break;
195 }
196#endif
197#if QT_CONFIG(vulkan) && __has_include(<vulkan/vulkan.h>)
198 case Api::Vulkan: {
199 if (!vulkanInstance()) {
200 mne::Log::warning("ChannelWindow: Vulkan needs a QVulkanInstance set with QWindow::setVulkanInstance()");
201 break;
202 }
203 QRhiVulkanInitParams params;
204 params.inst = vulkanInstance();
205 params.window = this;
206 m_pRhi.reset(QRhi::create(QRhi::Vulkan, &params));
207 break;
208 }
209#endif
210#if QT_CONFIG(metal)
211 case Api::Metal: {
212 QRhiMetalInitParams params;
213 m_pRhi.reset(QRhi::create(QRhi::Metal, &params));
214 break;
215 }
216#endif
217#if defined(Q_OS_WIN)
218 case Api::Direct3D11: {
219 QRhiD3D11InitParams params;
220 m_pRhi.reset(QRhi::create(QRhi::D3D11, &params));
221 break;
222 }
223 case Api::Direct3D12: {
224 QRhiD3D12InitParams params;
225 m_pRhi.reset(QRhi::create(QRhi::D3D12, &params));
226 break;
227 }
228#endif
229 default:
230 break;
231 }
232
233 if (!m_pRhi) {
234 mne::Log::warning("ChannelWindow: failed to create a QRhi for the selected graphics API");
235 return;
236 }
237
238 m_pSwapChain.reset(m_pRhi->newSwapChain());
239 m_pSwapChain->setWindow(this);
240 m_pRenderPass.reset(m_pSwapChain->newCompatibleRenderPassDescriptor());
241 m_pSwapChain->setRenderPassDescriptor(m_pRenderPass.get());
242 m_pRenderer->initializeResources();
243}

266void ChannelWindow::renderNow()
267{
268 if (!m_hasSwapChain || m_notExposed)
269 return;
270
271 if (m_pSwapChain->currentPixelSize() != m_pSwapChain->surfacePixelSize() || m_newlyExposed) {
272 resizeSwapChain();
273 if (!m_hasSwapChain)
274 return;
275 m_newlyExposed = false;
276 }
277
278 QRhi::FrameOpResult result = m_pRhi->beginFrame(m_pSwapChain.get());
279 if (result == QRhi::FrameOpSwapChainOutOfDate) {
280 resizeSwapChain();
281 if (!m_hasSwapChain)
282 return;
283 result = m_pRhi->beginFrame(m_pSwapChain.get());
284 }
285 if (result != QRhi::FrameOpSuccess) {
286 requestUpdate();
287 return;
288 }
289
290 m_pRenderer->renderFrame(m_pRhi.get(), m_pSwapChain->currentFrameRenderTarget(),
291 m_pSwapChain->currentFrameCommandBuffer());
292 m_pRhi->endFrame(m_pSwapChain.get());
293 ++m_frameCount;
294}

298void ChannelWindow::exposeEvent(QExposeEvent* event)
299{
300 Q_UNUSED(event);
301
302 if (isExposed() && !m_initialized) {
303 initRhi();
304 resizeSwapChain();
305 m_initialized = true;
306 }
307
308 const QSize surfaceSize = m_hasSwapChain ? m_pSwapChain->surfacePixelSize() : QSize();
309
310 // Stop rendering while the window is hidden or minimised
311 if ((!isExposed() || (m_hasSwapChain && surfaceSize.isEmpty())) && m_initialized && !m_notExposed)
312 m_notExposed = true;
313
314 // Resize the swap chain on the next frame after the window became visible again
315 if (isExposed() && m_initialized && m_notExposed && !surfaceSize.isEmpty()) {
316 m_notExposed = false;
317 m_newlyExposed = true;
318 }
319
320 if (isExposed() && !surfaceSize.isEmpty())
321 renderNow();
322}

326bool ChannelWindow::event(QEvent* event)
327{
328 switch (event->type()) {
329 case QEvent::UpdateRequest:
330 renderNow();
331 break;
332 case QEvent::PlatformSurface:
333 if (static_cast<QPlatformSurfaceEvent*>(event)->surfaceEventType() == QPlatformSurfaceEvent::SurfaceAboutToBeDestroyed)
334 releaseSwapChain();
335 break;
336 default:
337 break;
338 }
339 return QWindow::event(event);
340}
