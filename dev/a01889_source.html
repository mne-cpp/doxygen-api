58class ChannelRhiRenderer::TraceSourceAdapter : public viz::TraceSource
59{
60public:
61 explicit TraceSourceAdapter(const ChannelRhiRenderer& renderer)
62 : m_renderer(renderer)
63 {
64 }
65
66 int channel_count() const override
67 {
68 return m_renderer.totalLogicalChannels();
69 }
70
71 std::optional<Channel> channel(int index) const override
72 {
73 const int ch = m_renderer.actualChannelAt(index);
74 const viz::TraceSource* source = buffer();
75 if (!source || ch < 0)
76 return std::nullopt;
77 return source->channel(ch);
78 }
79
80 float value_at(int index, int sample) const override
81 {
82 const int ch = m_renderer.actualChannelAt(index);
83 const viz::TraceSource* source = buffer();
84 return (source && ch >= 0) ? source->value_at(ch, sample) : 0.f;
85 }
86
87 int first_sample() const override
88 {
89 const viz::TraceSource* source = buffer();
90 return (source && source->sample_count() > 0) ? source->first_sample() : 0;
91 }
92
93 int sample_count() const override
94 {
95 const viz::TraceSource* source = buffer();
96 return source ? source->sample_count() : 0;
97 }
98
99 viz::TraceVertices vertices(int index, int first, int last, int pixelWidth) const override
100 {
101 const int ch = m_renderer.actualChannelAt(index);
102 const viz::TraceSource* source = buffer();
103 if (!source || ch < 0)
104 return {first, {}};
105 return source->vertices(ch, first, last, pixelWidth);
106 }
107
108private:
109 const viz::TraceSource* buffer() const
110 {
111 return m_renderer.m_model ? &m_renderer.m_model->buffer() : nullptr;
112 }
113
114 const ChannelRhiRenderer& m_renderer;
115};

142void ChannelRhiRenderer::connectView()
143{
144 viz::TraceView::Effects effects;
145 effects.scroll_changed = [this](double) {
146 emit scrollSampleChanged(scrollSample());
147 requestUpdate();
148 };
149 effects.zoom_changed = [this](double) {
150 m_frameBuilder.geometry().invalidate(); // zoom change → decimation changes
151 m_frameBuilder.invalidate_overlay();
152 emit samplesPerPixelChanged(samplesPerPixel());
153 requestUpdate();
154 };
155 effects.channel_offset_changed = [this](int firstChannel) {
156 m_frameBuilder.geometry().invalidate();
157 emit channelOffsetChanged(firstChannel);
158 requestUpdate();
159 };
160 effects.animate_scroll = [this](const viz::TraceView::ScrollAnimation& animation) {
161 m_animator.start_scroll(animation, sceneTimestampMs());
162 startAnimationTimer();
163 };
164 effects.animate_zoom = [this](const viz::TraceView::ZoomAnimation& animation) {
165 m_animator.start_zoom(animation, sceneTimestampMs());
166 startAnimationTimer();
167 };
168 effects.stop_animations = [this] {
169 m_animator.stop_scroll();
170 };
171 effects.sample_clicked = [this](int sample) {
172 emit sampleClicked(sample);
173 };
174 effects.range_selected = [this](int startSample, int endSample) {
175 emit sampleRangeSelected(startSample, endSample);
176 };
177 effects.annotation_boundary_moved = [this](int index, bool isStart, int sample) {
178 emit annotationBoundaryMoved(index, isStart, sample);
179 };
180 effects.annotations_changed = [this] {
181 m_frameBuilder.invalidate_overlay();
182 requestUpdate();
183 };
184 effects.channel_double_clicked = [this](int logicalChannel) {
185 const int ch = actualChannelAt(logicalChannel);
186 if (m_model && ch >= 0) {
187 m_model->setChannelBad(ch, !m_model->channelInfo(ch).bad);
188 }
189 };
190 effects.cursor_changed = [this](viz::CursorShape shape) {
191 if (!m_pHost)
192 return;
193 if (shape == viz::CursorShape::SizeHorizontal)
194 m_pHost->setCursorShape(Qt::SizeHorCursor);
195 else
196 m_pHost->unsetCursorShape();
197 };
198 effects.cursor_data_changed = [this](const viz::TraceView::CursorData& data) {
199 emit cursorDataChanged(data.time_sec, data.amplitude, QString::fromStdString(data.channel_name),
200 QString::fromStdString(data.unit_label));
201 };
202 effects.redraw = [this] {
203 requestUpdate();
204 };
205 m_view.set_effects(std::move(effects));
206}

690void ChannelRhiRenderer::renderFrame(QRhi* rhi, QRhiRenderTarget* target, QRhiCommandBuffer* cb)
691{
692 tickAnimations();
693 const int logicalW = viewWidth();
694 const qreal dpr = (logicalW > 0) ? (static_cast<qreal>(target->pixelSize().width()) / static_cast<qreal>(logicalW))
695 : (m_pHost ? m_pHost->surfaceDevicePixelRatio() : 1.0);
696 const QFont font = hostFont();
697 const QtTextMetrics metrics(font);
698 m_frameRenderer.setFont(font);
699 const viz::GpuFrame frame = m_frameBuilder.build(view(), *m_traceSource, traceStyle(), m_markers, metrics,
700 RhiFrameRenderer::frameTarget(rhi, target, dpr));
701 m_frameRenderer.render(rhi, target, cb, frame, m_frameBuilder.resources());
702}

713QImage ChannelRhiRenderer::renderToImage(const QSize& size) const
714{
715 const int pw = size.isValid() ? size.width() : viewWidth();
716 const int ph = size.isValid() ? size.height() : viewHeight();
717 if (!m_model || pw <= 0 || ph <= 0 || samplesPerPixel() <= 0.f) {
718 return QImage();
719 }
720
721 const QFont font = QGuiApplication::font();
722 const QtTextMetrics metrics(font);
723 viz::Canvas canvas(pw, ph);
724 viz::draw_trace_figure(canvas, metrics, viz::Font{}, view(), *m_traceSource, m_markers, traceStyle());
725
726 QImage img(pw, ph, QImage::Format_RGB32);
727 img.fill(m_bgColor.rgb());
728 QPainter p(&img);
729 paintScene(p, canvas.display_list(), font);
730 return img;
731}
