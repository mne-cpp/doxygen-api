62std::string format_amplitude(float amp, const std::string& unit)
63{
64 const float absAmp = std::abs(amp);
65 if (absAmp == 0.f) {
66 return "0 " + unit;
67 }
68 if (absAmp < 1e-9f) {
69 return fixed(amp * 1e12f, 1) + " p" + unit;
70 }
71 if (absAmp < 1e-6f) {
72 return fixed(amp * 1e9f, 1) + " n" + unit;
73 }
74 if (absAmp < 1e-3f) {
75 return fixed(amp * 1e6f, 1) + " " + std::string(kMicro) + unit;
76 }
77 if (absAmp < 1.f) {
78 return fixed(amp * 1e3f, 1) + " m" + unit;
79 }
80 return fixed(amp, 3) + " " + unit;
81}

186double TraceView::clamp_scroll_sample(double sample) const
187{
188 // Never before the first available sample
189 sample = std::max(sample, static_cast<double>(m_source ? m_source->first_sample() : 0));
190
191 // Never past the file end, once it is known
192 if (m_lastFileSample >= 0) {
193 double maxScroll = static_cast<double>(m_lastFileSample - visible_sample_count() + 1);
194 maxScroll = std::max(maxScroll, static_cast<double>(m_firstFileSample));
195 sample = std::min(sample, maxScroll);
196 }
197 return sample;
198}

425int TraceView::hit_test_annotation_boundary(double x, bool& is_start) const
426{
427 const int px = to_px(x);
428 for (int i = 0; i < std::ssize(m_annotations); ++i) {
429 const double xStart = (static_cast<double>(m_annotations[i].start_sample) - m_scrollSample) / m_samplesPerPixel;
430 const double xEnd = (static_cast<double>(m_annotations[i].end_sample + 1) - m_scrollSample) / m_samplesPerPixel;
431 if (std::abs(px - static_cast<int>(xStart)) <= kBoundaryHitPx) {
432 is_start = true;
433 return i;
434 }
435 if (std::abs(px - static_cast<int>(xEnd)) <= kBoundaryHitPx) {
436 is_start = false;
437 return i;
438 }
439 }
440 return -1;
441}

477TraceView::RulerSnap TraceView::ruler_snap(int dx, int dy)
478{
479 dx = std::abs(dx);
480 dy = std::abs(dy);

482 return RulerSnap::Free;
483 }
484 if (dx > dy * 2) {
485 return RulerSnap::Horizontal;

487 if (dy > dx * 2) {
488 return RulerSnap::Vertical;

503std::optional<TraceView::CursorData> TraceView::cursor_data() const
504{
505 if (m_crosshairX < 0 || m_crosshairY < 0 || !m_source || total_channels() == 0) {
506 return std::nullopt;
507 }
508
509 const int sample = static_cast<int>(sample_at(m_crosshairX));
510 CursorData data;
511 data.time_sec = m_sfreq > 0.0 ? static_cast<float>(sample - m_firstFileSample) / static_cast<float>(m_sfreq) : 0.f;
512
513 if (m_butterflyMode) {
514 // Lanes are the channel types
515 const std::vector<TraceSource::Group> groups = m_source->groups();
516 const int lanes = static_cast<int>(groups.size());
517 if (lanes <= 0) {
518 return std::nullopt;
519 }
520 const float laneH = static_cast<float>(m_height) / static_cast<float>(lanes);
521 const int lane = std::clamp(static_cast<int>(static_cast<float>(m_crosshairY) / laneH), 0, lanes - 1);
522 data.channel_name = groups[lane].type_label;
523 data.unit_label = unit_for_type(groups[lane].type_label);
524 return data;
525 }
526
527 const int rows = visible_rows();
528 if (rows <= 0) {
529 return std::nullopt;
530 }
531 const float laneH = static_cast<float>(m_height) / static_cast<float>(rows);
532 const int row = std::clamp(static_cast<int>(static_cast<float>(m_crosshairY) / laneH), 0, rows - 1);
533 const int index = m_firstVisibleChannel + row;
534 const std::optional<TraceSource::Channel> channel = m_source->channel(index);
535 if (!channel) {
536 return std::nullopt;
537 }
538 data.amplitude = m_source->value_at(index, sample);
539 data.channel_name = channel->name;
540 data.unit_label = unit_for_type(channel->type_label);
541 return data;
542}

577void TraceView::draw_crosshair(Canvas& canvas, const TextMetrics& metrics, const Font& font) const
578{
579 const int cx = m_crosshairX;
580 const int cy = m_crosshairY;
581 if (cx < 0 || cy < 0 || !m_source || total_channels() == 0) {
582 return;
583 }
584
585 canvas.set_pen(Pen{{80, 80, 80, 160}, 1.0, LineStyle::Dash});
586 canvas.draw_line(pt(cx, 0), pt(cx, m_height));
587 canvas.draw_line(pt(0, cy), pt(m_width, cy));
588
589 const std::optional<CursorData> data = cursor_data();
590 if (!data) {
591 return;
592 }
593
594 std::string timeText;
595 if (m_clockTime && data->time_sec >= 0.f) {
596 const int totalMs = static_cast<int>(std::lround(data->time_sec * 1000.f));
597 timeText = std::format("{:02}:{:02}.{:03}", totalMs / 60000, (totalMs % 60000) / 1000, totalMs % 1000);
598 } else {
599 timeText = fixed(static_cast<double>(data->time_sec), 3) + " s";
600 }
601 const std::string label = data->channel_name + " " + timeText + " " + format_amplitude(data->amplitude, data->unit_label);
602
603 const Font labelFont = sized_font(font, 8.5, false);
604 const TextBox box = measure(metrics, labelFont, label);
605 int lx = cx + 10;
606 int ly = cy - 10;
607 if (lx + box.width + 8 > m_width) {
608 lx = cx - box.width - 18;
609 }
610 if (ly - box.height < 4) {
611 ly = cy + box.height + 6;
612 }
613 const Rect labelRect = rect(lx, ly - box.height, box.width, box.height).adjusted(-4, -2, 4, 2);
614 canvas.fill_rect(labelRect, Brush::solid({255, 255, 255, 220}));
615 canvas.set_pen(Color{30, 30, 30});
616 canvas.set_font(labelFont);
617 canvas.draw_text(labelRect, Align::Center, label);
618}

622void TraceView::draw_scalebars(Canvas& canvas, const TextMetrics& metrics, const Font& font) const
623{
624 if (!m_source || total_channels() == 0) {
625 return;
626 }
627
628 std::vector<TraceSource::Group> groups;
629 int lanes = 0;
630 if (m_butterflyMode) {
631 groups = m_source->groups();
632 lanes = static_cast<int>(groups.size());
633 } else {
634 lanes = visible_rows();
635 }
636 if (lanes <= 0) {
637 return;
638 }
639 const float laneH = static_cast<float>(m_height) / static_cast<float>(lanes);
640
641 // The amplitude scale of each channel type in view
642 std::map<std::string, float> typeScales;
643 if (m_butterflyMode) {
644 for (const TraceSource::Group& group : groups) {
645 if (group.amplitude_max > 0.f) {
646 typeScales[group.type_label] = group.amplitude_max;
647 }
648 }
649 } else {
650 for (int i = 0; i < lanes; ++i) {
651 const std::optional<TraceSource::Channel> channel = m_source->channel(m_firstVisibleChannel + i);
652 if (channel && !typeScales.contains(channel->type_label) && channel->amplitude_max > 0.f) {
653 typeScales[channel->type_label] = channel->amplitude_max;
654 }
655 }
656 }
657 if (typeScales.empty()) {
658 return;
659 }
660
661 const Font labelFont = sized_font(font, 8.0, false);
662 canvas.set_font(labelFont);
663 const int lineHeight = to_px(metrics.height(labelFont));
664 const int ascent = to_px(metrics.ascent(labelFont));
665
666 // Stacked upwards from the bottom-right corner; amplitude_max spans 45 % of a row, so a bar of barHeight
667 // pixels stands for amplitude_max * barHeight / scalePx
668 const int margin = 12;
669 const float scalePx = laneH * 0.45f;
670 const int barHeight = std::clamp(static_cast<int>(std::lround(scalePx)), 20, 60);
671 const int x = m_width - margin;
672 int y = m_height - margin;
673
674 for (auto it = typeScales.crbegin(); it != typeScales.crend(); ++it) {
675 const float ampValue = it->second * static_cast<float>(barHeight) / scalePx;
676 const std::string label = it->first + ": " + format_amplitude(ampValue, unit_for_type(it->first));
677 const int textW = to_px(metrics.advance(labelFont, label));
678 const int barX = x - textW - 14;
679 const int barTop = y - barHeight;
680
681 canvas.fill_rect(rect(barX - 4, y - barHeight - lineHeight - 4, textW + 22, barHeight + lineHeight + 8),
682 Brush::solid({255, 255, 255, 200}));
683
684 canvas.set_pen(Pen{{40, 40, 40}, 2.0});
685 canvas.draw_line(pt(barX + 4, barTop), pt(barX + 4, y));
686 canvas.draw_line(pt(barX, barTop), pt(barX + 8, barTop));
687 canvas.draw_line(pt(barX, y), pt(barX + 8, y));
688
689 canvas.set_pen(Color{30, 30, 30});
690 canvas.draw_text(pt(barX + 14, y - barHeight / 2 + ascent / 2), label);
691
692 y -= barHeight + lineHeight + 16;
693 }
694}

698void TraceView::draw_ruler(Canvas& canvas, const TextMetrics& metrics, const Font& font) const
699{
700 const int x0 = m_ruler.x0;
701 const int y0 = m_ruler.y0;
702 const int x1 = m_ruler.x1;
703 const int y1 = m_ruler.y1;
704 const bool snapH = m_ruler.snap == RulerSnap::Horizontal;
705 const bool snapV = m_ruler.snap == RulerSnap::Vertical;
706
707 const Color activeColor{40, 120, 200, 220};
708 const Color dimColor{130, 160, 200, 120};
709 const int tickLen = 5;
710
711 // Light veil with a border over the measured area
712 Rect measured;
713 if (snapH) {
714 measured = inclusive_rect(std::min(x0, x1), 0, std::max(x0, x1), m_height);
715 } else if (snapV) {
716 measured = inclusive_rect(0, std::min(y0, y1), m_width, std::max(y0, y1));
717 } else {
718 measured = inclusive_rect(std::min(x0, x1), std::min(y0, y1), std::max(x0, x1), std::max(y0, y1));
719 }
720 canvas.fill_rect(measured, Brush::solid({255, 255, 255, 60}));
721 canvas.set_pen(Pen{{40, 120, 200, 80}, 1.0});
722 canvas.draw_rect(measured);
723
724 // Vertical guides at the two x positions
725 canvas.set_pen(Pen{snapV ? dimColor : activeColor, 1.0, LineStyle::Dash});
726 canvas.draw_line(pt(x0, 0), pt(x0, m_height));
727 if (!snapV) {
728 canvas.draw_line(pt(x1, 0), pt(x1, m_height));
729 }
730
731 // Horizontal guides at the two y positions while measuring amplitude only
732 if (snapV) {
733 canvas.set_pen(Pen{activeColor, 1.0, LineStyle::Dash});
734 canvas.draw_line(pt(0, y0), pt(m_width, y0));
735 canvas.draw_line(pt(0, y1), pt(m_width, y1));
736 }
737
738 // Time span at y0 and amplitude span at x0
739 canvas.set_pen(Pen{snapV ? dimColor : activeColor, snapH ? 2.0 : 1.0});
740 if (!snapV) {
741 canvas.draw_line(pt(std::min(x0, x1), y0), pt(std::max(x0, x1), y0));
742 }
743 canvas.set_pen(Pen{snapH ? dimColor : activeColor, snapV ? 2.0 : 1.0});
744 if (!snapH) {
745 canvas.draw_line(pt(x0, std::min(y0, y1)), pt(x0, std::max(y0, y1)));
746 }
747
748 // End ticks
749 canvas.set_pen(Pen{activeColor, 1.0});
750 if (!snapV) {
751 canvas.draw_line(pt(x0 - tickLen, y0), pt(x0 + tickLen, y0));

775 ampUnit = unit_for_type(channel->type_label);
776 }
777 }

779 const std::string ampLabel = std::string(kDelta) + "A = " + format_ruler_amplitude(deltaAmp, ampUnit);

784 const Brush pill = Brush::solid({255, 255, 255, 210});

788 int labelY = y0 - 6;
789 if (labelY < 14) {
790 labelY = y0 + 16;
791 }

794 canvas.fill_rect(r, pill);

796 canvas.draw_text(r, Align::Center, timeLabel);

800 const int labelY = (y0 + y1) / 2;

802 const Rect r = rect(labelX + box.width / 2 - (box.width - 1) / 2, labelY - (box.height - 1) / 2, box.width, box.height)

840bool TraceView::handle_wheel(const WheelEvent& event)
841{
842 const double dx = event.angle_delta_x;
843 const double dy = event.angle_delta_y;
844
845 if (has_flag(event.modifiers, Modifier::Ctrl)) {
846 // Zoom the time axis
847 zoom_to(m_samplesPerPixel * (dy > 0.0 ? 0.8 : 1.25), 150);
848 } else if (std::abs(dx) > std::abs(dy)) {
849 // Mostly horizontal (touch-pad swipe): scroll time by 10 % of the view
850 if (!m_frozen) {
851 const double step = m_width * m_samplesPerPixel * 0.1 * m_scrollSpeedFactor * (dx > 0.0 ? -1.0 : 1.0);
852 scroll_by(step, 100);
853 }
854 } else if (m_wheelScrollsChannels) {
855 // Vertical wheel scrolls channel rows: up = earlier
856 const int maxFirst = std::max(0, total_channels() - m_visibleChannelCount);
857 set_first_visible_channel(std::clamp(m_firstVisibleChannel + (dy > 0.0 ? -1 : 1), 0, maxFirst));
858 } else if (!m_frozen) {
859 // Vertical wheel scrolls time by 15 % of the view
860 const double step = m_width * m_samplesPerPixel * 0.15 * m_scrollSpeedFactor * (dy > 0.0 ? -1.0 : 1.0);
861 scroll_by(step, 100);
862 }
863 return true;
864}

884
885bool TraceView::handle_press(const PointerEvent& event)
886{
887 const int px = to_px(event.x);
888 const int py = to_px(event.y);
889
890 // Right button: annotation range selection in annotation mode, the ruler otherwise
891 if (event.button == MouseButton::Right) {
892 stop_scroll_animations();
893 if (m_annotationSelectionEnabled) {
894 m_selection = {true, px, px};
895 } else {
896 m_ruler = {true, RulerSnap::Free, px, py, px, py};
897 }
898 redraw();
899 return true;
900 }
901
902 // Middle button or Alt + left: plain drag scroll
903 if (!m_frozen && (event.button == MouseButton::Middle || (event.button == MouseButton::Left && has_flag(event.modifiers, Modifier::Alt)))) {
904 m_dragging = true;
905 m_dragStartX = px;
906 m_dragStartScroll = m_scrollSample;
907 return true;
908 }
909
910 if (event.button == MouseButton::Left) {
911 stop_scroll_animations();
912
913 // Grab an annotation boundary for resizing
914 if (m_annotationSelectionEnabled && !m_annotations.empty()) {
915 bool isStart = false;
916 const int hit = hit_test_annotation_boundary(event.x, isStart);
917 if (hit >= 0) {
918 m_annDragging = true;
919 m_annDragIndex = hit;
920 m_annDragIsStart = isStart;
921 return true;
922 }
923 }
924
925 if (m_frozen) {
926 // A click still reports its sample, but nothing scrolls
927 if (m_effects.sample_clicked) {
928 m_effects.sample_clicked(static_cast<int>(sample_at(event.x)));
929 }
930 return true;
931 }
932
933 // The drag starts once the pointer moved past the threshold
934 m_leftButtonDown = true;
935 m_leftDragActivated = false;
936 m_leftDownX = px;
937 m_leftDownScroll = m_scrollSample;
938 m_dragStartTime = event.timestamp_ms;
939 m_velocityHistory.clear();
940 m_velocityHistory.push_back({px, 0});
941 return true;
942 }
943 return false;
944}

947
948bool TraceView::handle_move(const PointerEvent& event)
949{
950 const int px = to_px(event.x);
951 const int py = to_px(event.y);
952
953 if (m_annDragging) {
954 if (m_annDragIndex >= 0 && m_annDragIndex < std::ssize(m_annotations)) {
955 const int sample = clamped_boundary_sample(px);
956 if (m_annDragIsStart) {
957 m_annotations[m_annDragIndex].start_sample = sample;
958 } else {
959 m_annotations[m_annDragIndex].end_sample = sample;
960 }
961 if (m_effects.annotations_changed) {
962 m_effects.annotations_changed();
963 }
964 }
965 return true;
966 }
967
968 if (m_selection.active) {
969 m_selection.x1 = px;
970 redraw();
971 return true;
972 }
973
974 if (m_ruler.active) {
975 m_ruler.snap = ruler_snap(px - m_ruler.x0, py - m_ruler.y0);
976 apply_ruler_end(px, py);
977 redraw();
978 return true;
979 }
980
981 if (m_dragging) {
982 set_scroll_sample(m_dragStartScroll - static_cast<double>(px - m_dragStartX) * m_samplesPerPixel);
983 return true;
984 }
985
986 if (m_leftButtonDown) {
987 const int dx = px - m_leftDownX;
988 if (!m_leftDragActivated && std::abs(dx) > kDragThresholdPx) {
989 m_leftDragActivated = true;
990 }
991 if (m_leftDragActivated) {
992 set_scroll_sample(m_leftDownScroll - static_cast<double>(dx) * m_samplesPerPixel);
993
994 // Keep the last 100 ms of positions for the release velocity
995 const std::int64_t now = event.timestamp_ms - m_dragStartTime;
996 m_velocityHistory.push_back({px, now});
997 while (m_velocityHistory.size() > 1 && now - m_velocityHistory.front().t > kVelocityWindowMs) {
998 m_velocityHistory.erase(m_velocityHistory.begin());
999 }
1000 return true;
1001 }
1002 }
1003
1004 // Hover: crosshair tracking
1005 if (m_crosshairEnabled) {
1006 m_crosshairX = px;
1007 m_crosshairY = py;
1008 redraw();
1009 if (m_effects.cursor_data_changed) {
1010 if (const std::optional<CursorData> data = cursor_data()) {
1011 m_effects.cursor_data_changed(*data);
1012 }
1013 }
1014 }
1015
1016 // Hover: resize cursor over annotation boundaries
1017 if (m_annotationSelectionEnabled && !m_annotations.empty()) {
1018 bool isStart = false;
1019 const int hit = hit_test_annotation_boundary(event.x, isStart);
1020 if (hit >= 0) {
1021 if (m_annHoverIndex != hit || m_annHoverIsStart != isStart) {
1022 m_annHoverIndex = hit;
1023 m_annHoverIsStart = isStart;
1024 if (m_effects.cursor_changed) {
1025 m_effects.cursor_changed(CursorShape::SizeHorizontal);
1026 }
1027 }
1028 } else if (m_annHoverIndex >= 0) {
1029 m_annHoverIndex = -1;
1030 if (m_effects.cursor_changed) {
1031 m_effects.cursor_changed(CursorShape::Default);
1032 }
1033 }
1034 }
1035 return false;
1036}

1039
1040bool TraceView::handle_release(const PointerEvent& event)
1041{
1042 const int px = to_px(event.x);
1043 const int py = to_px(event.y);
1044
1045 if (m_annDragging && event.button == MouseButton::Left) {
1046 if (m_effects.annotation_boundary_moved) {
1047 m_effects.annotation_boundary_moved(m_annDragIndex, m_annDragIsStart, clamped_boundary_sample(px));
1048 }
1049 m_annDragging = false;
1050 m_annDragIndex = -1;
1051 return true;
1052 }
1053
1054 if (m_selection.active && event.button == MouseButton::Right) {
1055 m_selection.x1 = px;
1056 m_selection.active = false;
1057 redraw();
1058 const int x0 = std::min(m_selection.x0, m_selection.x1);
1059 const int x1 = std::max(m_selection.x0, m_selection.x1);
1060 if (x1 - x0 > 3) {
1061 const int startSample = std::max(static_cast<int>(sample_at(x0)), m_firstFileSample);
1062 int endSample = static_cast<int>(sample_at(x1));
1063 if (m_lastFileSample >= 0) {
1064 endSample = std::min(endSample, m_lastFileSample);
1065 }
1066 if (endSample >= startSample && m_effects.range_selected) {
1067 m_effects.range_selected(startSample, endSample);
1068 }
1069 }
1070 return true;
1071 }
1072
1073 if (m_ruler.active && event.button == MouseButton::Right) {
1074 apply_ruler_end(px, py);
1075 m_ruler.active = false;
1076 redraw();
1077 return true;
1078 }
1079
1080 if (m_dragging && (event.button == MouseButton::Middle || event.button == MouseButton::Left)) {
1081 m_dragging = false;
1082 return true;
1083 }
1084
1085 if (event.button == MouseButton::Left && m_leftButtonDown) {
1086 if (!m_leftDragActivated) {
1087 // A tap: report the sample, no inertia
1088 if (m_effects.sample_clicked) {
1089 m_effects.sample_clicked(static_cast<int>(m_leftDownScroll + event.x * m_samplesPerPixel));
1090 }
1091 } else if (m_velocityHistory.size() >= 2) {
1092 // Glide on with the velocity of the last 100 ms
1093 const VelocitySample oldest = m_velocityHistory.front();
1094 const VelocitySample newest = m_velocityHistory.back();
1095 const double dt = static_cast<double>(newest.t - oldest.t);
1096 if (dt > 5.0) {
1097 // Dragging right scrolls back in time
1098 const double velocity = -(static_cast<double>(newest.x - oldest.x) / dt) * m_samplesPerPixel;
1099 const double speed = std::abs(velocity);
1100 if (speed > 0.3) {
1101 // Cubic ease-out starts at 3 times its mean speed: travel = v * duration / 3
1102 const double durationMs = std::clamp(speed, 500.0, 5000.0);
1103 const double target = std::max(m_scrollSample + velocity * durationMs / 3.0, static_cast<double>(m_firstFileSample));
1104 if (m_effects.animate_scroll) {
1105 m_effects.animate_scroll({m_scrollSample, target, static_cast<int>(durationMs), true});
1106 }
1107 }
1108 }
1109 }
1110 m_leftButtonDown = false;
1111 m_leftDragActivated = false;
1112 return true;
1113 }
1114 return false;
1115}
