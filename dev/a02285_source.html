57class ChannelRhiRenderer::TraceSourceAdapter : public scene::TraceSource
58{
59public:
60 explicit TraceSourceAdapter(const ChannelRhiRenderer& renderer)
61 : m_renderer(renderer)
62 {
63 }
64
65 int channel_count() const override
66 {
67 return m_renderer.totalLogicalChannels();
68 }
69
70 std::optional<Channel> channel(int index) const override
71 {
72 const int ch = m_renderer.actualChannelAt(index);
73 const scene::TraceSource* source = buffer();
74 if (!source || ch < 0)
75 return std::nullopt;
76 return source->channel(ch);
77 }
78
79 float value_at(int index, int sample) const override
80 {
81 const int ch = m_renderer.actualChannelAt(index);
82 const scene::TraceSource* source = buffer();
83 return (source && ch >= 0) ? source->value_at(ch, sample) : 0.f;
84 }
85
86 int first_sample() const override
87 {
88 const scene::TraceSource* source = buffer();
89 return (source && source->sample_count() > 0) ? source->first_sample() : 0;
90 }
91
92 int sample_count() const override
93 {
94 const scene::TraceSource* source = buffer();
95 return source ? source->sample_count() : 0;
96 }
97
98 scene::TraceVertices vertices(int index, int first, int last, int pixelWidth) const override
99 {
100 const int ch = m_renderer.actualChannelAt(index);
101 const scene::TraceSource* source = buffer();
102 if (!source || ch < 0)
103 return {first, {}};
104 return source->vertices(ch, first, last, pixelWidth);
105 }
106
107private:
108 const scene::TraceSource* buffer() const
109 {
110 return m_renderer.m_model ? &m_renderer.m_model->buffer() : nullptr;
111 }
112
113 const ChannelRhiRenderer& m_renderer;
114};

141void ChannelRhiRenderer::connectView()
142{
143 scene::TraceView::Effects effects;
144 effects.scroll_changed = [this](double) {
145 emit scrollSampleChanged(scrollSample());
146 requestUpdate();
147 };
148 effects.zoom_changed = [this](double) {
149 m_frameBuilder.geometry().invalidate(); // zoom change → decimation changes
150 m_frameBuilder.invalidate_overlay();
151 emit samplesPerPixelChanged(samplesPerPixel());
152 requestUpdate();
153 };
154 effects.channel_offset_changed = [this](int firstChannel) {
155 m_frameBuilder.geometry().invalidate();
156 emit channelOffsetChanged(firstChannel);
157 requestUpdate();
158 };
159 effects.animate_scroll = [this](const scene::TraceView::ScrollAnimation& animation) {
160 m_animator.start_scroll(animation, sceneTimestampMs());
161 startAnimationTimer();
162 };
163 effects.animate_zoom = [this](const scene::TraceView::ZoomAnimation& animation) {
164 m_animator.start_zoom(animation, sceneTimestampMs());
165 startAnimationTimer();
166 };
167 effects.stop_animations = [this] {
168 m_animator.stop_scroll();
169 };
170 effects.sample_clicked = [this](int sample) {
171 emit sampleClicked(sample);
172 };
173 effects.range_selected = [this](int startSample, int endSample) {
174 emit sampleRangeSelected(startSample, endSample);
175 };
176 effects.annotation_boundary_moved = [this](int index, bool isStart, int sample) {
177 emit annotationBoundaryMoved(index, isStart, sample);
178 };
179 effects.annotations_changed = [this] {
180 m_frameBuilder.invalidate_overlay();
181 requestUpdate();
182 };
183 effects.channel_double_clicked = [this](int logicalChannel) {
184 const int ch = actualChannelAt(logicalChannel);
185 if (m_model && ch >= 0) {
186 m_model->setChannelBad(ch, !m_model->channelInfo(ch).bad);
187 }
188 };
189 effects.cursor_changed = [this](scene::CursorShape shape) {
190 if (!m_pHost)
191 return;
192 if (shape == scene::CursorShape::SizeHorizontal)
193 m_pHost->setCursorShape(Qt::SizeHorCursor);
194 else
195 m_pHost->unsetCursorShape();
196 };
197 effects.cursor_data_changed = [this](const scene::TraceView::CursorData& data) {
198 emit cursorDataChanged(data.time_sec, data.amplitude, QString::fromStdString(data.channel_name),
199 QString::fromStdString(data.unit_label));
200 };
201 effects.redraw = [this] {
202 requestUpdate();
203 };
204 m_view.set_effects(std::move(effects));
205}

228{
229 // Animations advance on every frame and, for hosts that render no frames (e.g. hidden ones), on this timer

689void ChannelRhiRenderer::renderFrame(QRhi* rhi, QRhiRenderTarget* target, QRhiCommandBuffer* cb)
690{
691 tickAnimations();
692 const int logicalW = viewWidth();
693 const qreal dpr = (logicalW > 0) ? (static_cast<qreal>(target->pixelSize().width()) / static_cast<qreal>(logicalW))
694 : (m_pHost ? m_pHost->surfaceDevicePixelRatio() : 1.0);
695 const QFont font = hostFont();
696 const QtTextMetrics metrics(font);
697 m_frameRenderer.setFont(font);
698 const scene::GpuFrame frame = m_frameBuilder.build(view(), *m_traceSource, traceStyle(), m_markers, metrics,
699 RhiFrameRenderer::frameTarget(rhi, target, dpr));
700 m_frameRenderer.render(rhi, target, cb, frame, m_frameBuilder.resources());
701}

712QImage ChannelRhiRenderer::renderToImage(const QSize& size) const
713{
714 const int pw = size.isValid() ? size.width() : viewWidth();
715 const int ph = size.isValid() ? size.height() : viewHeight();
716 if (!m_model || pw <= 0 || ph <= 0 || samplesPerPixel() <= 0.f) {
717 return QImage();
718 }
719
720 const QFont font = QGuiApplication::font();
721 const QtTextMetrics metrics(font);
722 scene::Canvas canvas(pw, ph);
723 scene::draw_trace_figure(canvas, metrics, scene::Font{}, view(), *m_traceSource, m_markers, traceStyle());
724
725 QImage img(pw, ph, QImage::Format_RGB32);
726 img.fill(m_bgColor.rgb());
727 QPainter p(&img);
728 paintScene(p, canvas.display_list(), font);
729 return img;
730}
