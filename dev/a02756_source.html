56void DipoleObject::load(const mne::InvEcdSet& ecdSet)
57{
58 createGeometry();
59
60 m_instanceCount = ecdSet.size();
61 m_instanceData.resize(m_instanceCount * sizeof(InstanceData));
62 m_loadedModels.resize(m_instanceCount);
63 InstanceData* data = reinterpret_cast<InstanceData*>(m_instanceData.data());
64
65 const Eigen::Vector3f from = Eigen::Vector3f::UnitY(); // Cone points up Y axis
66
67 float maxMag = 0.0f;
68 for (int i = 0; i < ecdSet.size(); ++i) {
69 const float mag = ecdSet[i].Q.norm();
70 maxMag = std::max(maxMag, mag);
71 }
72
73 std::mt19937 rng(std::random_device{}());
74 std::uniform_real_distribution<float> colorDist(0.0f, 1.0f);
75
76 for (int i = 0; i < ecdSet.size(); ++i) {
77 const auto& dip = ecdSet[i];
78
79 const Eigen::Vector3f pos(dip.rd(0), dip.rd(1), dip.rd(2));
80 const Eigen::Vector3f Q(dip.Q(0), dip.Q(1), dip.Q(2));
81 const float mag = Q.norm();
82
83 const Eigen::Vector3f to = Q.normalized();
84 // Handle case where 'to' is parallel to 'from'
85 Eigen::Quaternionf rot = Eigen::Quaternionf::Identity();
86 if (from.dot(to) < -0.99f) {
87 rot = Eigen::Quaternionf(Eigen::AngleAxisf(std::numbers::pi_v<float>, Eigen::Vector3f::UnitX()));
88 } else if (from.dot(to) <= 0.99f) {
89 rot = rotationTo(from, to);
90 }
91
92 // Scale based on magnitude relative to max, with a minimum size of 20%
93 float scaleFactor = (maxMag > 0.0f) ? (0.2f + 0.8f * (mag / maxMag)) : 1.0f;
94
95 const Eigen::Matrix4f m = modelMatrix(pos, rot, Eigen::Vector3f::Constant(scaleFactor));
96 m_loadedModels[i] = m;
97 storeMatrix(m, data[i].model);
98
99 // Random Color
100 data[i].color[0] = colorDist(rng);
101 data[i].color[1] = colorDist(rng);
102 data[i].color[2] = colorDist(rng);
103 data[i].color[3] = 1.0f;
104
105 // Selection state
106 data[i].isSelected = 0.0f;
107 }
108
109 ++m_instanceGeneration;
110}

126bool DipoleObject::boundingBox(Eigen::Vector3f& min, Eigen::Vector3f& max) const
127{
128 if (m_instanceCount == 0)
129 return false;
130 const InstanceData* data = reinterpret_cast<const InstanceData*>(m_instanceData.data());
131 for (int i = 0; i < m_instanceCount; ++i) {
132 // Column-major model matrix: the translation is elements 12..14
133 const Eigen::Vector3f pos(data[i].model[12], data[i].model[13], data[i].model[14]);
134 min = i == 0 ? pos : Eigen::Vector3f(min.cwiseMin(pos));
135 max = i == 0 ? pos : Eigen::Vector3f(max.cwiseMax(pos));
136 }
137 return true;
138}

140void DipoleObject::createGeometry()
141{
142 if (!m_vertexData.empty())
143 return;
144
145 // Create a simple cone
146 // Radius 0.001, Height 0.003
147 // 32 segments
148
149 float radius = 0.005f; // Slightly larger for visibility
150 float height = 0.01f;
151 int segments = 16;
152
153 std::vector<VertexData> vertices;
154 std::vector<uint32_t> indices;
155
156 // Tip
157 VertexData tip = {0, height, 0, 0, 1, 0};
158 vertices.push_back(tip);
159
160 // Base center
161 VertexData baseCenter = {0, 0, 0, 0, -1, 0};
162 vertices.push_back(baseCenter); // Index 1
163
164 // Rim vertices
165 for (int i = 0; i < segments; ++i) {
166 const float angle = 2.0f * std::numbers::pi_v<float> * static_cast<float>(i) / static_cast<float>(segments);
167 float x = radius * std::cos(angle);
168 float z = radius * std::sin(angle);
169
170 // Side normal
171 // Slope vector: (cos, -h/r, sin) -> normalized
172 const Eigen::Vector3f n = Eigen::Vector3f(x, radius / height, z).normalized();
173
174 VertexData vSide = {x, 0, z, n.x(), n.y(), n.z()};
175 vertices.push_back(vSide);
176
177 VertexData vBase = {x, 0, z, 0, -1, 0};
178 vertices.push_back(vBase);
179 }
180
181 // Indices
182 for (int i = 0; i < segments; ++i) {
183 // Cone sides
184 // Tip (0), current(i), next(next)
185 // Side vertices start at firstRimIdx.
186 // Layout: Tip, BaseCenter, Side0, Base0, Side1, Base1...
187 // Wait, easier to keep separate lists or just flat
188
189 // Re-do layout for ease:
190 // 0: Tip
191 // 1: Base Center
192 // 2..2+segments-1: Side rim vertices
193 // 2+segments..2+2*segments-1: Base rim vertices
194 }
195
196 vertices.clear();
197 vertices.push_back(tip); // 0
198 vertices.push_back(baseCenter); // 1
199
200 for (int i = 0; i < segments; ++i) {
201 const float angle = 2.0f * std::numbers::pi_v<float> * static_cast<float>(i) / static_cast<float>(segments);
202 float x = radius * std::cos(angle);
203 float z = radius * std::sin(angle);
204
205 // Normal perpendicular to the slope down the side
206 const Eigen::Vector3f slope(x, -height, z);
207 const Eigen::Vector3f tangent(-std::sin(angle), 0.0f, std::cos(angle));
208 const Eigen::Vector3f sideNormal = tangent.cross(slope).normalized();
209 // Actually simple: normal y component is radius/height ratio related.
210
211 vertices.push_back({x, 0, z, sideNormal.x(), sideNormal.y(), sideNormal.z()}); // Side rim
212 }
213
214 for (int i = 0; i < segments; ++i) {
215 const float angle = 2.0f * std::numbers::pi_v<float> * static_cast<float>(i) / static_cast<float>(segments);
216 float x = radius * std::cos(angle);
217 float z = radius * std::sin(angle);
218 vertices.push_back({x, 0, z, 0, -1, 0}); // Base rim
219 }
220
221 int sideStart = 2;
222 int baseStart = 2 + segments;
223
224 for (int i = 0; i < segments; ++i) {
225 int next = (i + 1) % segments;
226
227 // Cone face
228 indices.push_back(0);
229 indices.push_back(sideStart + next);
230 indices.push_back(sideStart + i);
231
232 // Base face
233 indices.push_back(1);
234 indices.push_back(baseStart + i);
235 indices.push_back(baseStart + next);
236 }
237
238 m_indexCount = static_cast<int>(indices.size());
239
240 m_vertexData.resize(vertices.size() * sizeof(VertexData));
241 memcpy(m_vertexData.data(), vertices.data(), m_vertexData.size());
242
243 m_indexData.resize(indices.size() * sizeof(uint32_t));
244 memcpy(m_indexData.data(), indices.data(), m_indexData.size());
245
246 ++m_geometryGeneration;
247}

249int DipoleObject::intersect(const Eigen::Vector3f& rayOrigin, const Eigen::Vector3f& rayDir, float& dist) const
250{
251 if (m_instanceCount == 0)
252 return -1;
253
254 int closestIdx = -1;
255 float closestDist = std::numeric_limits<float>::max();
256
257 const InstanceData* data = reinterpret_cast<const InstanceData*>(m_instanceData.data());
258
259 // Geometry radius ~ 0.005 (base) to 0.01 (height).
260 // Let's use a slightly larger hit radius to make selection easier and more accurate.
261 const float baseRadius = 0.02f;
262 const Ray ray{rayOrigin, rayDir};
263
264 for (int i = 0; i < m_instanceCount; ++i) {
265 // Translation (last column) and scale (length of the first column, roughly uniform)
266 const Eigen::Vector3f center(data[i].model[12], data[i].model[13], data[i].model[14]);
267 const float scale = Eigen::Vector3f(data[i].model[0], data[i].model[1], data[i].model[2]).norm();
268 const float radius = baseRadius * scale;
269
270 // Spheres behind the ray origin do not count; from inside a sphere the exit point is hit
271 const std::optional<SphereCrossing> crossing = intersect_ray_sphere(ray, center, radius);
272 if (!crossing || crossing->entry + crossing->exit < 0.0f)
273 continue;
274 const float t = crossing->entry >= 0.0f ? crossing->entry : crossing->exit;
275 if (t < 0.0f)
276 continue;
277
278 if (t < closestDist) {
279 closestDist = t;
280 closestIdx = i;
281 }
282 }
283
284 if (closestIdx != -1) {
285 dist = closestDist;
286 return closestIdx;
287 }
288
289 return -1;
290}
