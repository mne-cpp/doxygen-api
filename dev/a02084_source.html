61std::vector<AverageLayoutWidget::Channel> AverageLayoutWidget::channelsFrom(const SelectionItem& selection)
62{
63 std::vector<Channel> channels;
64 for (std::size_t i = 0; i < selection.m_sChannelName.size(); ++i) {
65 Channel channel;
66 channel.name = selection.m_sChannelName[i];
67 channel.number = i < selection.m_iChannelNumber.size() ? selection.m_iChannelNumber[i] : -1;
68 channel.position = i < selection.m_qpChannelPosition.size() ? selection.m_qpChannelPosition[i] : QPointF();
69 channel.kind = i < selection.m_iChannelKind.size() ? selection.m_iChannelKind[i] : 0;
70 channel.unit = i < selection.m_iChannelUnit.size() ? selection.m_iChannelUnit[i] : 0;
71 channels.push_back(std::move(channel));
72 }
73 return channels;

140mne::viz::AverageLayout AverageLayoutWidget::averageLayout() const
141{
142 std::vector<viz::AverageTile> tiles;
143 tiles.reserve(m_channels.size());
144 for (std::size_t i = 0; i < m_channels.size(); ++i) {
145 const Channel& channel = m_channels[i];
146 viz::AverageTile tile;
147 tile.box = viz::AverageLayout::box_at({channel.position.x(), channel.position.y()});
148 tile.name = channel.name;
149 tile.name_size = static_cast<int>(i) == m_pressedTile ? kPressedNameSize : kNameSize;
150 tile.bad = channel.bad;
151 tile.max_value = getScalingValue(m_scaleMap, channel.kind, channel.unit);
152 tile.stim_sample = channel.firstSample;
153 tile.columns = channel.averages.empty() ? 0 : channel.averages.front().samples;
154 for (const Average& average : channel.averages) {
155 // Averages missing from the activation map are drawn
156 if (const auto active = m_activation.find(average.comment); active != m_activation.end() && !active->second) {
157 continue;
158 }
159 viz::AverageTile::Trace trace;
160 const auto color = m_colors.find(average.comment);
161 trace.color = toSceneColor(color != m_colors.end() ? color->second : m_signalColor);
162 if (average.data) {
163 trace.samples.reserve(static_cast<std::size_t>(average.samples));
164 for (int s = 0; s < average.samples; ++s) {
165 trace.samples.push_back(average.data[static_cast<std::ptrdiff_t>(s) * average.stride]);
166 }
167 }
168 tile.traces.push_back(std::move(trace));
169 }
170 tiles.push_back(std::move(tile));
171 }
172 viz::AverageLayout layout;
173 layout.set_tiles(std::move(tiles));
174 return layout;
175}

180{
181 viz::Rect united;

183 const viz::Rect area = viz::AverageLayout::box_at({channel.position.x(), channel.position.y()}).adjusted(0, -kNameSize, 0, 0);
184 united = united.empty() ? area : united.united(area);
