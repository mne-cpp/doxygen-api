79{
80 QStringList lMethods = connectivitySettings.getConnectivityMethods();

103 }
104

106 results.append(Coherence::calculate(connectivitySettings));
107 }
108
109 if(lMethods.contains("IMAGCOH")) {
110 results.append(ImagCoherence::calculate(connectivitySettings));
111 }
112

133 qWarning() << "Total" << timer.elapsed();
134 qDebug() << "Connectivity::calculateMultiMethods - Calculated"<< lMethods <<"for" << connectivitySettings.size() << "trials in"<< timer.elapsed() << "msecs.";
