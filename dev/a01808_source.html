105void PolylineObject::createSegmentGeometry()
106{
107 if(!m_vertexData.isEmpty()) {
108 return;
109 }
110
111 // Unit cylinder along Y, height 1 and radius 1, scaled and rotated per
112 // instance. Eight segments is enough for a thin line and keeps the vertex
113 // count low, since a long path can have thousands of instances.
114 const int iSegments = 8;
115 const float fHalfHeight = 0.5f;
116
117 std::vector<VertexData> vertices;
118 std::vector<quint32> indices;
119
120 for(int i = 0; i < iSegments; ++i) {
121 const float fAngle = 2.0f * static_cast<float>(M_PI) * static_cast<float>(i) / static_cast<float>(iSegments);
122 const float x = std::cos(fAngle);
123 const float z = std::sin(fAngle);
124
125 QVector3D normal(x, 0.0f, z);
126 normal.normalize();
127
128 vertices.push_back({x, fHalfHeight, z, normal.x(), normal.y(), normal.z()});
129 vertices.push_back({x, -fHalfHeight, z, normal.x(), normal.y(), normal.z()});
130 }
131
132 for(int i = 0; i < iSegments; ++i) {
133 const quint32 uTop = static_cast<quint32>(i * 2);
134 const quint32 uBottom = uTop + 1;
135 const quint32 uNextTop = static_cast<quint32>(((i + 1) % iSegments) * 2);
136 const quint32 uNextBottom = uNextTop + 1;
137
138 indices.push_back(uTop);
139 indices.push_back(uBottom);
140 indices.push_back(uNextTop);
141
142 indices.push_back(uNextTop);
143 indices.push_back(uBottom);
144 indices.push_back(uNextBottom);
145 }
146
147 m_vertexData = QByteArray(reinterpret_cast<const char*>(vertices.data()),
148 static_cast<qsizetype>(vertices.size() * sizeof(VertexData)));
149 m_indexData = QByteArray(reinterpret_cast<const char*>(indices.data()),
150 static_cast<qsizetype>(indices.size() * sizeof(quint32)));
151 m_iIndexCount = static_cast<int>(indices.size());
152 m_bGeometryDirty = true;
153}

157void PolylineObject::buildInstances()
158{
159 if(m_vecPoints.size() < 2) {
160 m_instanceData.clear();
161 m_iInstanceCount = 0;
162 return;
163 }
164
165 const int iSegmentCount = m_vecPoints.size() - 1;
166 std::vector<InstanceData> instances;
167 instances.reserve(static_cast<size_t>(iSegmentCount));
168
169 const float fLastStep = static_cast<float>(iSegmentCount - 1);
170
171 for(int i = 0; i < iSegmentCount; ++i) {
172 const Eigen::Vector3f& vecStart = m_vecPoints.at(i);
173 const Eigen::Vector3f& vecEnd = m_vecPoints.at(i + 1);
174
175 const QVector3D start(vecStart.x(), vecStart.y(), vecStart.z());
176 const QVector3D end(vecEnd.x(), vecEnd.y(), vecEnd.z());
177
178 const QVector3D direction = end - start;
179 const float fLength = direction.length();
180
181 // Two consecutive fits can land on the same position when the subject
182 // does not move. Such a segment has no direction to orient a cylinder
183 // by, so it is skipped instead of producing a NaN rotation.
184 //
185 // The threshold has to be far below anything physical rather than
186 // qFuzzyIsNull, whose tolerance for float is 1e-5. Positions are in
187 // metres, so that would discard every step under ten micrometres and
188 // punch gaps into the path of a slowly drifting subject. A picometre
189 // is small enough that only genuinely coincident points are dropped,
190 // and large enough to keep the division below well conditioned.
191 constexpr float fMinSegmentLength = 1.0e-12f;
192 if(fLength < fMinSegmentLength) {
193 continue;
194 }
195
196 QMatrix4x4 model;
197 model.translate((start + end) * 0.5f);
198 model.rotate(QQuaternion::rotationTo(QVector3D(0.0f, 1.0f, 0.0f),
199 direction / fLength));
200 model.scale(m_fRadius, fLength, m_fRadius);
201
202 InstanceData instance{};
203 memcpy(instance.model, model.constData(), 16 * sizeof(float));
204
205 // Grade from the start colour to the end colour so the direction the
206 // path was travelled in is visible without extra decoration.
207 const float fRatio = (fLastStep > 0.0f) ? (static_cast<float>(i) / fLastStep) : 0.0f;
208 instance.color[0] = static_cast<float>(m_startColor.redF() + (m_endColor.redF() - m_startColor.redF()) * fRatio);
209 instance.color[1] = static_cast<float>(m_startColor.greenF() + (m_endColor.greenF() - m_startColor.greenF()) * fRatio);
210 instance.color[2] = static_cast<float>(m_startColor.blueF() + (m_endColor.blueF() - m_startColor.blueF()) * fRatio);
211 instance.color[3] = 1.0f;
212 instance.isSelected = 0.0f;
213
214 instances.push_back(instance);
215 }
216
217 m_instanceData = QByteArray(reinterpret_cast<const char*>(instances.data()),
218 static_cast<qsizetype>(instances.size() * sizeof(InstanceData)));
219 m_iInstanceCount = static_cast<int>(instances.size());
220 m_bInstancesDirty = m_iInstanceCount > 0;
221}

225void PolylineObject::updateBuffers(QRhi *rhi, QRhiResourceUpdateBatch *u)
226{
227 if(m_bGeometryDirty) {
228 if(!m_gpu->vertexBuffer) {
229 m_gpu->vertexBuffer.reset(rhi->newBuffer(QRhiBuffer::Immutable,
230 QRhiBuffer::VertexBuffer,
231 m_vertexData.size()));
232 m_gpu->vertexBuffer->create();
233 }
234 if(!m_gpu->indexBuffer) {
235 m_gpu->indexBuffer.reset(rhi->newBuffer(QRhiBuffer::Immutable,
236 QRhiBuffer::IndexBuffer,
237 m_indexData.size()));
238 m_gpu->indexBuffer->create();
239 }
240
241 u->uploadStaticBuffer(m_gpu->vertexBuffer.get(), m_vertexData.constData());
242 u->uploadStaticBuffer(m_gpu->indexBuffer.get(), m_indexData.constData());
243 m_bGeometryDirty = false;
244 }
245
246 if(m_bInstancesDirty && m_iInstanceCount > 0) {
247 const int iRequiredSize = m_instanceData.size();
248
249 if(!m_gpu->instanceBuffer || m_gpu->instanceBuffer->size() < static_cast<quint32>(iRequiredSize)) {
250 m_gpu->instanceBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
251 QRhiBuffer::VertexBuffer,
252 iRequiredSize));
253 m_gpu->instanceBuffer->create();
254 }
255
256 u->updateDynamicBuffer(m_gpu->instanceBuffer.get(), 0, iRequiredSize, m_instanceData.constData());
257 m_bInstancesDirty = false;
258 }
259}
