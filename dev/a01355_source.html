39RtSensorStream::RtSensorStream()
40{
41 // Create data worker
42 // streamData() runs on the streaming thread, which also calls the result callbacks.
43 m_pWorker = std::make_unique<RtSensorDataWorker>();
44
45 // Forward worker results to the callbacks of this stream
46 m_pWorker->setNewRtSensorColorsCallback([this](const std::string& surfaceKey, const std::vector<uint32_t>& colors) {
47 if (m_newSensorColorsAvailableCallback) {
48 m_newSensorColorsAvailableCallback(surfaceKey, colors);
49 }
50 });
51 m_pWorker->setNewRtRawSensorDataCallback([this](const Eigen::VectorXf& data) {
52 if (m_newRawSensorDataAvailableCallback) {
53 m_newRawSensorDataAvailableCallback(data);
54 }
55 });
56
57 // Streaming thread driving the cadence
58 m_pStreamTask = std::make_unique<PeriodicTask>([worker = m_pWorker.get()] { worker->streamData(); });
59
60 // Create interpolation matrix worker
61 // One pool thread runs the recomputations in request order.
62 m_pInterpWorker = std::make_unique<RtSensorInterpolationMatWorker>();
63 m_pInterpPool = std::make_unique<mne::ThreadPool>(1);
64
65 // Forward interpolation results: auto-apply to data worker + re-emit for external listeners
66 m_pInterpWorker->setNewMegMappingAvailableCallback([this](const std::string& surfaceKey, const std::shared_ptr<Eigen::MatrixXf>& mappingMat, const std::vector<int>& pick) {
67 onNewMegMapping(surfaceKey, mappingMat, pick);
68 });
69 m_pInterpWorker->setNewEegMappingAvailableCallback([this](const std::string& surfaceKey, const std::shared_ptr<Eigen::MatrixXf>& mappingMat, const std::vector<int>& pick) {
70 onNewEegMapping(surfaceKey, mappingMat, pick);
71 });
72
73 mne::Log::debug("RtSensorStream: Initialized with {} ms interval", m_iTimeInterval);
74}

78RtSensorStream::~RtSensorStream()
79{

84 m_pInterpPool.reset();

280void RtSensorStream::onNewMegMapping(const std::string& surfaceKey,
281 std::shared_ptr<Eigen::MatrixXf> mappingMat,
282 const std::vector<int>& pick)
283{
284 // Auto-forward the new matrix to the data worker
285 if (m_pWorker && mappingMat) {
286 m_pWorker->setMappingMatrix(surfaceKey, mappingMat);
287 }
288
289 // Re-emit for external listeners (e.g. BrainView)
290 if (m_newMegMappingAvailableCallback) {
291 m_newMegMappingAvailableCallback(surfaceKey, mappingMat, pick);
292 }
293
294 mne::Log::debug("RtSensorStream: New MEG mapping received and forwarded ({}x{})", mappingMat->rows(), mappingMat->cols());
295}
