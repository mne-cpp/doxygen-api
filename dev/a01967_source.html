56// Each pair of a trial is taken out of the matching running sum, as long as both have the same length
57template<typename T>
58void subtractPairs(QVector<QPair<int, T>>& sums, const QVector<QPair<int, T>>& trial, int rows)
59{
60 if (sums.size() != trial.size())
61 return;
62 for (int i = 0; i < rows && i < sums.size(); ++i)
63 sums[i].second -= trial.at(i).second;
64}
65
66// Removes the influence of one trial from the sums over all trials
67void subtractFromSums(ConnectivitySettings::IntermediateSumData& sums, const ConnectivitySettings::IntermediateTrialData& trial)
68{
69 const int rows = static_cast<int>(trial.matData.rows());
70 subtractPairs(sums.vecPairCsdSum, trial.vecPairCsd, rows);
71 subtractPairs(sums.vecPairCsdNormalizedSum, trial.vecPairCsdNormalized, rows);
72 subtractPairs(sums.vecPairCsdImagSignSum, trial.vecPairCsdImagSign, rows);
73 subtractPairs(sums.vecPairCsdImagAbsSum, trial.vecPairCsdImagAbs, rows);
74 subtractPairs(sums.vecPairCsdImagSqrdSum, trial.vecPairCsdImagSqrd, rows);
75
76 if (sums.matPsdSum.rows() == trial.matPsd.rows() && sums.matPsdSum.cols() == trial.matPsd.cols())
77 sums.matPsdSum -= trial.matPsd;
78}

179void ConnectivitySettings::removeFirst(int iAmount)
180{
181 if (m_trialData.isEmpty()) {
182 qDebug() << "ConnectivitySettings::removeFirst - No elements to delete. Returning.";
183 return;
184 }
185
186 if (m_trialData.size() < iAmount) {
187 qDebug() << "ConnectivitySettings::removeFirst - Not enough elements stored in list in order to delete them. Returning.";
188 return;
189 }
190
191 for (int j = 0; j < iAmount; ++j) {
192 subtractFromSums(m_intermediateSumData, m_trialData.first());
193 m_trialData.removeFirst();
194 }
195}

310void ConnectivitySettings::setNodePositions(const MNEForwardSolution& forwardSolution, const FsSurfaceSet& surfSet)
311{
312 //Generate node vertices
313 MatrixX3f matNodeVertLeft, matNodeVertRight;
314
315 if (forwardSolution.isClustered()) {
316 auto* lhHemi = forwardSolution.src.hemisphereAt(0);
317 auto* rhHemi = forwardSolution.src.hemisphereAt(1);
318 matNodeVertLeft.resize(lhHemi->cluster_info.centroidVertno.size(), 3);
319 for (int j = 0; j < matNodeVertLeft.rows(); ++j) {
320 matNodeVertLeft.row(j) = surfSet[0].rr().row(lhHemi->cluster_info.centroidVertno.at(j)) - surfSet[0].offset().transpose();
321 }
322
323 matNodeVertRight.resize(rhHemi->cluster_info.centroidVertno.size(), 3);
324 for (int j = 0; j < matNodeVertRight.rows(); ++j) {
325 matNodeVertRight.row(j) = surfSet[1].rr().row(rhHemi->cluster_info.centroidVertno.at(j)) - surfSet[1].offset().transpose();
326 }
327 } else {
328 matNodeVertLeft.resize(forwardSolution.src[0].vertno.rows(), 3);
329 for (int j = 0; j < matNodeVertLeft.rows(); ++j) {
330 matNodeVertLeft.row(j) = surfSet[0].rr().row(forwardSolution.src[0].vertno(j)) - surfSet[0].offset().transpose();
331 }
332
333 matNodeVertRight.resize(forwardSolution.src[1].vertno.rows(), 3);
334 for (int j = 0; j < matNodeVertRight.rows(); ++j) {
335 matNodeVertRight.row(j) = surfSet[1].rr().row(forwardSolution.src[1].vertno(j)) - surfSet[1].offset().transpose();
336 }
337 }
338
339 m_matNodePositions.resize(matNodeVertLeft.rows() + matNodeVertRight.rows(), 3);
340 m_matNodePositions << matNodeVertLeft, matNodeVertRight;
341}
