64AcqExpected<Message> receive(TcpSocket& socket, std::chrono::milliseconds timeout, std::vector<std::byte>& body, const std::string& name)
65{
66 std::array<std::byte, 8> header{};
67 if (const std::error_code e = socket.read(header, timeout)) {
68 return e == std::errc::connection_reset ? AcqError(AcqErrorCode::EndOfStream)
69 : transport_error(e, AcqErrorCode::DeviceIo, std::format("{} failed", name));
70 }
71 std::uint32_t length = 0;
72 for (std::size_t i = 4; i < 8; ++i) {
73 length = (length << 8) | std::to_integer<std::uint32_t>(header[i]);
74 }
75 if (length > kMaxBody) {
76 return AcqError(AcqErrorCode::DeviceIo, std::format("{} sent a message of {} bytes", name, length));
77 }
78 body.resize(length);
79 if (socket.read(body, kRest)) {
80 return AcqError(AcqErrorCode::DeviceIo, std::format("{} sent an incomplete message", name));
81 }
82 return Message{std::string(reinterpret_cast<const char*>(header.data()), 4), {}};
83}

111 AcqExpected<std::string> info(std::chrono::milliseconds timeout) override
112 {
113 // The command port serves one request per connection
114 auto socket = TcpSocket::connect(m_host, m_command_port, timeout);
115 if (!socket) {
116 return AcqError(AcqErrorCode::DeviceNotFound,
117 std::format("no BabyMEG command port: cannot connect to {}:{} ({})", m_host, m_command_port, socket.error().message()));
118 }
119 if (const std::error_code e = socket->write(bytes("INFO"))) {
120 return transport_error(e, AcqErrorCode::DeviceIo, "cannot ask the BabyMEG server for its channels");
121 }
122 std::vector<std::byte> body;
123 while (true) {
124 auto message = receive(*socket, timeout, body, m_name);
125 if (!message) {
126 return message.error() == AcqErrorCode::Timeout ? AcqError(AcqErrorCode::Timeout, std::format("{} does not describe its channels", m_name))
127 : message.error();
128 }
129 if (message->command == "INFO") {
130 return std::string(reinterpret_cast<const char*>(body.data()), body.size());
131 }
132 }
133 }

136 {

139 if (!socket) {
140 return {AcqErrorCode::DeviceNotFound, std::format("cannot connect to {} ({})", m_name, socket.error().message())};

143 }
144 return send("DATA");

147 AcqExpected<std::span<const std::byte>> next(std::chrono::milliseconds timeout) override
148 {
149 if (!m_data) {
150 return AcqError(AcqErrorCode::InvalidState, "the BabyMEG data connection is closed");
151 }
152 while (true) {
153 auto message = receive(*m_data, timeout, m_body, m_name);
154 if (!message) {
155 return message.error();
156 }
157 if (message->command == "DATR") {
158 if (AcqError e = send("DATA"); !e.ok()) {
159 return e;
160 }
161 return std::span<const std::byte>(m_body);
162 }
163 if (message->command == "QUIT") {
164 static_cast<void>(send("QREL"));
165 m_data.reset();
166 return AcqError(AcqErrorCode::EndOfStream);
167 }
168 timeout = kRest; // a message for someone else; the package may follow right away
169 }
170 }

172 void stop() override
173 {
174 if (!m_data) {
175 return;
176 }
177 if (send("QUIT").ok()) {
178 // Packages already on their way come before the server's answer
179 const auto deadline = std::chrono::steady_clock::now() + kGoodbye;
180 while (std::chrono::steady_clock::now() < deadline) {
181 auto message = receive(*m_data, kGoodbye, m_body, m_name);
182 if (!message || message->command == "QUIT") {
183 break;
184 }
185 }
186 static_cast<void>(send("QREL"));
187 }
188 m_data.reset();
189 }
