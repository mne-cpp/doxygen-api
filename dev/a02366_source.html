171double MNESurfaceOrVolume::solid_angle(const Eigen::Vector3f& from, const MNETriangle& tri) /* ...to this triangle */
172/*
173 * Compute the solid angle according to van Oosterom's
174 * formula
175 */
176{
177 Eigen::Vector3d v1 = (tri.r1 - from).cast<double>();
178 Eigen::Vector3d v2 = (tri.r2 - from).cast<double>();
179 Eigen::Vector3d v3 = (tri.r3 - from).cast<double>();
180
181 double triple = v1.cross(v2).dot(v3);
182
183 double l1 = v1.norm();
184 double l2 = v2.norm();
185 double l3 = v3.norm();
186 double s = (l1 * l2 * l3 + v1.dot(v2) * l3 + v1.dot(v3) * l2 + v2.dot(v3) * l1);
187
188 return (2.0 * atan2(triple, s));
189}

193void MNESurfaceOrVolume::add_triangle_data()
194/*
195 * Add the triangle data structures
196 */
197{
198 int k;
199 MNETriangle* tri;
200
201 if (type != MNE_SOURCE_SPACE_SURFACE)
202 return;
203
204 tris.clear();
205 use_tris.clear();
206 /*
207 * Add information for the complete triangulation
208 */
209 if (itris.rows() > 0 && ntri > 0) {
210 tris.resize(ntri);
211 tot_area = 0.0;
212 for (k = 0, tri = tris.data(); k < ntri; k++, tri++) {
213 tri->vert = &itris(k, 0);
214 tri->r1 = rr.row(tri->vert[0]).transpose();
215 tri->r2 = rr.row(tri->vert[1]).transpose();
216 tri->r3 = rr.row(tri->vert[2]).transpose();
217 tri->compute_data();
218 tot_area += tri->area;
219 }
220#ifdef TRIANGLE_SIZE_WARNING
221 for (k = 0, tri = tris.data(); k < ntri; k++, tri++)
222 if (tri->area < 1e-5 * tot_area / ntri)
223 mne::Log::warning("Warning: Triangle area is only {:g} um^2 ({:.5f} % of expected average)\n", 1e12 * tri->area, 100 * ntri * tri->area / tot_area);
224#endif
225 }
226#ifdef DEBUG
227 mne::Log::info("\ttotal area = {:<.1f} cm^2\n", 1e4 * tot_area);
228#endif
229 /*
230 * Add information for the selected subset if applicable
231 */
232 if (use_itris.rows() > 0 && nuse_tri > 0) {
233 use_tris.resize(nuse_tri);
234 for (k = 0, tri = use_tris.data(); k < nuse_tri; k++, tri++) {
235 tri->vert = &use_itris(k, 0);
236 tri->r1 = rr.row(tri->vert[0]).transpose();
237 tri->r2 = rr.row(tri->vert[1]).transpose();
238 tri->r3 = rr.row(tri->vert[2]).transpose();
239 tri->compute_data();
240 }
241 }
242 return;
243}

247void MNESurfaceOrVolume::compute_cm(const MNESurfaceOrVolume::PointsT& rr, int np, float (&cm)[3])
248/*
249 * Compute the center of mass of a set of points
250 */
251{
252 int q;
253 cm[0] = cm[1] = cm[2] = 0.0;
254 for (q = 0; q < np; q++) {
255 cm[0] += rr(q, 0);
256 cm[1] += rr(q, 1);
257 cm[2] += rr(q, 2);
258 }
259 if (np > 0) {
260 cm[0] = cm[0] / np;
261 cm[1] = cm[1] / np;
262 cm[2] = cm[2] / np;
263 }
264 return;
265}

280void MNESurfaceOrVolume::calculate_vertex_distances()
281{
282 int k, p, ndist;
283 int nneigh;
284
285 if (neighbor_vert.empty() || nneighbor_vert.size() == 0)
286 return;
287
288 vert_dist.clear();
289 vert_dist.resize(np);
290 mne::Log::info("\tDistances between neighboring vertices...");
291 for (k = 0, ndist = 0; k < np; k++) {
292 nneigh = nneighbor_vert[k];
293 vert_dist[k] = Eigen::VectorXf(nneigh);
294 const Eigen::VectorXi& neigh = neighbor_vert[k];
295 for (p = 0; p < nneigh; p++) {
296 if (neigh[p] >= 0) {
297 vert_dist[k][p] = (rr.row(neigh[p]) - rr.row(k)).norm();
298 } else
299 vert_dist[k][p] = -1.0;
300 ndist++;
301 }
302 }
303 mne::Log::info("[{} distances done]\n", ndist);
304 return;
305}

309int MNESurfaceOrVolume::add_vertex_normals()
310{
311 int k, c, p;
312 int* ii;
313 float w, size;
314 MNETriangle* tri;
315
316 if (type != MNE_SOURCE_SPACE_SURFACE)
317 return OK;
318 /*
319 * Reallocate the stuff and initialize
320 */
321 nn = MNESurfaceOrVolume::NormalsT::Zero(np, 3);
322 /*
323 * One pass through the triangles will do it
324 */
325 add_triangle_data();
326 for (p = 0, tri = tris.data(); p < ntri; p++, tri++) {
327 ii = tri->vert;
328 w = 1.0; /* This should be related to the triangle size */
329 /*
330 * Then the vertex normals
331 */
332 for (k = 0; k < 3; k++)
333 for (c = 0; c < 3; c++)
334 nn(ii[k], c) += w * tri->nn[c];
335 }
336 for (k = 0; k < np; k++) {
337 size = nn.row(k).norm();
338 if (size > 0.0)
339 nn.row(k) /= size;
340 }
341 compute_surface_cm();
342 return OK;
343}

347int MNESurfaceOrVolume::add_geometry_info(bool do_normals, bool check_too_many_neighbors)
348/*
349 * Add vertex normals and neighbourhood information
350 */
351{
352 int k, c, p, q;
353 int vert;
354 int* ii;
355 int nneighbors;
356 float w, size;
357 int found;
358 int nfix_distinct, nfix_no_neighbors, nfix_defect;
359 MNETriangle* tri;
360
361 if (type == MNE_SOURCE_SPACE_VOLUME) {
362 calculate_vertex_distances();
363 return OK;
364 }
365 if (type != MNE_SOURCE_SPACE_SURFACE)
366 return OK;
367 /*
368 * Reallocate the stuff and initialize
369 */
370 if (do_normals) {
371 nn = MNESurfaceOrVolume::NormalsT::Zero(np, 3);
372 }
373 neighbor_tri.clear();
374 neighbor_tri.resize(np);
375 nneighbor_tri = Eigen::VectorXi::Zero(np);
376
377 /* nn is already zero-initialized above */
378 /*
379 * One pass through the triangles will do it
380 */
381 add_triangle_data();
382 for (p = 0, tri = tris.data(); p < ntri; p++, tri++)
383 if (tri->area == 0)
384 mne::Log::warning("\tWarning : zero size triangle # {}\n", p);
385 mne::Log::info("\tTriangle ");
386 if (do_normals)
387 mne::Log::info("and vertex ");
388 mne::Log::info("normals and neighboring triangles...");
389 for (p = 0, tri = tris.data(); p < ntri; p++, tri++) {
390 ii = tri->vert;
391 w = 1.0; /* This should be related to the triangle size */
392 for (k = 0; k < 3; k++) {
393 /*
394 * Then the vertex normals
395 */
396 if (do_normals)
397 for (c = 0; c < 3; c++)
398 nn(ii[k], c) += w * tri->nn[c];
399 /*
400 * Add to the list of neighbors
401 */
402 neighbor_tri[ii[k]].conservativeResize(nneighbor_tri[ii[k]] + 1);
403 neighbor_tri[ii[k]][nneighbor_tri[ii[k]]] = p;
404 nneighbor_tri[ii[k]]++;
405 }
406 }
407 nfix_no_neighbors = 0;
408 nfix_defect = 0;
409 for (k = 0; k < np; k++) {
410 if (nneighbor_tri[k] <= 0) {
411#ifdef STRICT_ERROR
412 err_printf_set_error("Vertex %d does not have any neighboring triangles!", k);
413 return FAIL;
414#else
415#ifdef REPORT_WARNINGS
416 mne::Log::warning("Warning: Vertex {} does not have any neighboring triangles!\n", k);
417#endif
418#endif
419 nfix_no_neighbors++;
420 } else if (nneighbor_tri[k] < 3) {
421#ifdef REPORT_WARNINGS
422 mne::Log::warning("\n\tTopological defect: Vertex {} has only {} neighboring triangle{} Vertex omitted.\n\t", k, nneighbor_tri[k], nneighbor_tri[k] > 1 ? "s." : ".");
423#endif
424 nfix_defect++;
425 nneighbor_tri[k] = 0;
426 neighbor_tri[k].resize(0);
427 }
428 }
429 /*
430 * Scale the vertex normals to unit length
431 */
432 for (k = 0; k < np; k++)
433 if (nneighbor_tri[k] > 0) {
434 size = nn.row(k).norm();
435 if (size > 0.0)
436 nn.row(k) /= size;
437 }
438 mne::Log::info("[done]\n");
439 /*
440 * Determine the neighboring vertices
441 */
442 mne::Log::info("\tVertex neighbors...");
443 neighbor_vert.clear();
444 neighbor_vert.resize(np);
445 nneighbor_vert = VectorXi::Zero(np);
446 /*
447 * We know the number of neighbors beforehand
448 */
449 for (k = 0; k < np; k++) {
450 if (nneighbor_tri[k] > 0) {
451 neighbor_vert[k] = VectorXi(nneighbor_tri[k]);
452 nneighbor_vert[k] = nneighbor_tri[k];
453 } else {
454 nneighbor_vert[k] = 0;
455 }
456 }
457 nfix_distinct = 0;
458 for (k = 0; k < np; k++) {
459 Eigen::VectorXi& neighbors = neighbor_vert[k];
460 nneighbors = 0;
461 for (p = 0; p < nneighbor_tri[k]; p++) {
462 /*
463 * Fit in the other vertices of the neighboring triangle
464 */
465 for (c = 0; c < 3; c++) {
466 vert = tris[neighbor_tri[k][p]].vert[c];
467 if (vert != k) {
468 for (q = 0, found = false; q < nneighbors; q++) {
469 if (neighbors[q] == vert) {
470 found = true;
471 break;
472 }
473 }
474 if (!found) {
475 if (nneighbors < nneighbor_vert[k])
476 neighbors[nneighbors++] = vert;
477 else {
478 if (check_too_many_neighbors) {
479 mne::Log::critical("Too many neighbors for vertex {}.", k);
480 return FAIL;
481 } else
482 mne::Log::warning("\tWarning: Too many neighbors for vertex {}\n", k);
483 }
484 }
485 }
486 }
487 }
488 if (nneighbors != nneighbor_vert[k]) {
489#ifdef REPORT_WARNINGS
490 mne::Log::warning("\n\tIncorrect number of distinct neighbors for vertex {} ({} instead of {}) [fixed].", k, nneighbors, nneighbor_vert[k]);
491#endif
492 nfix_distinct++;
493 nneighbor_vert[k] = nneighbors;
494 }
495 }
496 mne::Log::info("[done]\n");
497 /*
498 * Distance calculation follows
499 */
500 calculate_vertex_distances();
501 compute_surface_cm();
502 /*
503 * Summarize the defects
504 */
505 if (nfix_defect > 0)
506 mne::Log::warning("\tWarning: {} topological defects were fixed.\n", nfix_defect);
507 if (nfix_distinct > 0)
508 mne::Log::warning("\tWarning: {} vertices had incorrect number of distinct neighbors (fixed).\n", nfix_distinct);
509 if (nfix_no_neighbors > 0)
510 mne::Log::warning("\tWarning: {} vertices did not have any neighboring triangles (fixed)\n", nfix_no_neighbors);
511#ifdef DEBUG
512 for (k = 0; k < np; k++) {
513 if (nneighbor_vert[k] <= 0)
514 mne::Log::critical("No neighbors for vertex {}\n", k);
515 if (nneighbor_tri[k] <= 0)
516 mne::Log::critical("No neighbor tris for vertex {}\n", k);
517 }
518#endif
519 return OK;
520}
