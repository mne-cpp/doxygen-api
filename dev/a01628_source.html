103struct RegionData
104{
105 Eigen::MatrixXd matRoiG;
106 Eigen::MatrixXd matRoiGWhitened;
107 bool bUseWhitened;
108
109 Eigen::MatrixXd matRoiGOrig;
110
111 qint32 nClusters;
112
113 Eigen::VectorXi idcs;
114 qint32 iLabelIdxIn;
115 QString sDistMeasure;
116
117 RegionDataOut cluster() const
118 {
119 QString t_sDistMeasure;
120 if(sDistMeasure.isEmpty())
121 t_sDistMeasure = QString("cityblock");
122 else
123 t_sDistMeasure = sDistMeasure;
124
125 // Kmeans Reduction
126 RegionDataOut p_RegionDataOut;
127
128 UTILSLIB::KMeans t_kMeans(t_sDistMeasure, QString("sample"), 5);
129
130 if(bUseWhitened)
131 {
132 t_kMeans.calculate(this->matRoiGWhitened, this->nClusters, p_RegionDataOut.roiIdx, p_RegionDataOut.ctrs, p_RegionDataOut.sumd, p_RegionDataOut.D);
133
134 Eigen::MatrixXd newCtrs = Eigen::MatrixXd::Zero(p_RegionDataOut.ctrs.rows(), p_RegionDataOut.ctrs.cols());
135 for(qint32 c = 0; c < p_RegionDataOut.ctrs.rows(); ++c)
136 {
137 qint32 num = 0;
138
139 for(qint32 idx = 0; idx < p_RegionDataOut.roiIdx.size(); ++idx)
140 {
141 if(c == p_RegionDataOut.roiIdx[idx])
142 {
143 newCtrs.row(c) += this->matRoiG.row(idx); //just take whitened to get indeces calculate centroids using the original matrix
144 ++num;
145 }
146 }
147
148 if(num > 0)
149 newCtrs.row(c) /= num;
150 }
151 p_RegionDataOut.ctrs = newCtrs; //Replace whitened with original
152 }
153 else
154 t_kMeans.calculate(this->matRoiG, this->nClusters, p_RegionDataOut.roiIdx, p_RegionDataOut.ctrs, p_RegionDataOut.sumd, p_RegionDataOut.D);
155
156 p_RegionDataOut.iLabelIdxOut = this->iLabelIdxIn;
157
158 return p_RegionDataOut;
159 }
160};
