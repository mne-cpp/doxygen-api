175template<typename Table, typename Key>
176constexpr auto bids_lookup(const Table& table, const Key& key) -> std::optional<typename Table::value_type::second_type>
177{
178 const auto it = std::ranges::find(table, key, &Table::value_type::first);
179 return it == table.end() ? std::nullopt : std::optional(it->second);
180}
