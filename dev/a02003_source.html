46{
47public:
48 explicit NatusSimulator(const NatusSimulation& simulation)
49 : m_simulation(simulation)
50 {
51 }
52
53 std::string description() const override
54 {
55 return std::format("simulated Natus streaming machine, seed {}", m_simulation.seed);
56 }
57
58 AcqExpected<std::size_t> receive(std::span<std::byte> buffer, std::chrono::milliseconds timeout) override
59 {
60 if (!m_reader.isValid()) {
61 if (AcqError e = start(); !e.ok()) {
62 return e;
63 }
64 }
65 const std::size_t samples = m_simulation.block;
66 const std::size_t channels = m_simulation.channels;
67 auto view = m_reader.acquire(samples, timeout, samples);
68 if (!view) {
69 return view.error();
70 }
71 if (view->sampleCount() < samples) {
72 static_cast<void>(m_reader.release(*view)); // whole datagrams only; the end of the session cuts the rest
73 return AcqError(AcqErrorCode::EndOfStream);
74 }
75 const std::size_t bytes = (kNatusHeaderFloats + samples * channels) * sizeof(float);
76 if (buffer.size() < bytes) {
77 static_cast<void>(m_reader.release(*view));
78 return AcqError(AcqErrorCode::Internal, "datagram buffer too small");
79 }
80 const std::array<float, kNatusHeaderFloats> header = {static_cast<float>(m_sent), static_cast<float>(samples), static_cast<float>(channels)};
81 std::memcpy(buffer.data(), header.data(), sizeof(header));
82 auto* out = buffer.data() + sizeof(header);
83 for (std::size_t p = 0; p < view->partCount(); ++p) {
84 const AcqBlockPart& part = view->part(p);
85 for (std::size_t s = 0; s < part.samples; ++s) {
86 for (std::size_t c = 0; c < channels; ++c) {
87 float volts = 0.0F;
88 std::memcpy(&volts, part.at(c, s), sizeof(volts));
89 const auto value = static_cast<float>(m_simulation.gain * volts / kNatusScale);
90 std::memcpy(out, &value, sizeof(value));
91 out += sizeof(value);
92 }
93 }
94 }
95 ++m_sent;
96 if (AcqError e = m_reader.release(*view); !e.ok()) {
97 return e;
98 }
99 return bytes;
100 }
101
102private:
103 AcqError start()
104 {
105 std::string uri = std::format("synthetic://?sfreq={}&stim=0&eeg={}&block={}&seed={}&realtime={}", m_simulation.rate, m_simulation.channels,
106 m_simulation.block, m_simulation.seed, m_simulation.realtime ? 1 : 0);
107 if (m_simulation.duration > 0.0) {
108 uri += std::format("&duration={}", m_simulation.duration);
109 }
110 auto session = AcqDevice::open(uri);
111 if (!session) {
112 return session.error();
113 }
114 AcqReaderOptions gating;
115 gating.gating = true; // the session waits for this reader instead of overwriting unread samples
116 auto reader = session->stream().reader(AcqReader::ReadMode::Lossless, gating);
117 if (!reader) {
118 return reader.error();
119 }
120 if (AcqError e = session->start(); !e.ok()) {
121 return e;
122 }
123 m_session = std::move(*session);
124 m_reader = std::move(*reader);
125 return {};
126 }
127
128 NatusSimulation m_simulation;
129 AcqDevice m_session;
130 AcqReader m_reader;
131 std::uint64_t m_sent = 0;
132};
