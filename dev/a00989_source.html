50{

94std::expected<TcpSocket, std::error_code> TcpSocket::connect(const std::string& host, std::uint16_t port, std::chrono::milliseconds timeout)
95{
96 if (!started()) {
97 return std::unexpected(std::make_error_code(std::errc::not_supported));
98 }
99 addrinfo hints{};
100 hints.ai_family = AF_UNSPEC;
101 hints.ai_socktype = SOCK_STREAM;
102 addrinfo* found = nullptr;
103 const std::string service = std::to_string(port);
104 if (::getaddrinfo(host.c_str(), service.c_str(), &hints, &found) != 0 || found == nullptr) {
105 return std::unexpected(std::make_error_code(std::errc::host_unreachable));
106 }
107 std::error_code why = std::make_error_code(std::errc::address_not_available);
108 TcpSocket socket;
109 for (const addrinfo* a = found; a != nullptr; a = a->ai_next) {
110 const Native s = ::socket(a->ai_family, a->ai_socktype, a->ai_protocol);
111 if (s == kInvalid) {
112 why = last_error();
113 continue;
114 }
115 socket = TcpSocket();
116 socket.m_socket = static_cast<Handle>(s);
117 if (!set_non_blocking(s)) {
118 why = last_error();
119 continue;
120 }
121#ifdef SO_NOSIGPIPE
122 int no_signal = 1; // a closed peer is an error, not a signal
123 ::setsockopt(s, SOL_SOCKET, SO_NOSIGPIPE, &no_signal, sizeof(no_signal));
124#endif
125 if (::connect(s, a->ai_addr, static_cast<socklen_t>(a->ai_addrlen)) != 0 && !would_block(last_error())) {
126 why = last_error();
127 continue;
128 }
129 if (wait_for(s, POLLOUT, timeout) <= 0) {
130 why = std::make_error_code(std::errc::timed_out);
131 continue;
132 }
133 int error = 0;
134 socklen_t length = sizeof(error);
135 ::getsockopt(s, SOL_SOCKET, SO_ERROR, reinterpret_cast<char*>(&error), &length);
136 if (error != 0) {
137#ifdef _WIN32
138 why = {error, std::system_category()};
139#else
140 why = {error, std::generic_category()};
141#endif
142 continue;
143 }
144 int on = 1; // requests are small and answered one by one
145 ::setsockopt(s, IPPROTO_TCP, TCP_NODELAY, reinterpret_cast<const char*>(&on), sizeof(on));
146 ::freeaddrinfo(found);
147 return socket;
148 }
149 ::freeaddrinfo(found);
150 return std::unexpected(why);
151}

156{
157 const auto s = static_cast<Native>(m_socket);
158 while (!data.empty()) {
159 const auto sent = send_some(s, data);
160 if (sent < 0) {

164 }
165 return error;
166 }
167 data = data.subspan(static_cast<std::size_t>(sent));

174std::error_code TcpSocket::write(std::span<const std::byte> data, std::chrono::milliseconds timeout)
175{
176 const auto s = static_cast<Native>(m_socket);
177 const auto deadline = std::chrono::steady_clock::now() + timeout;
178 while (!data.empty()) {
179 const auto sent = send_some(s, data);
180 if (sent < 0) {
181 const std::error_code error = last_error();
182 if (!would_block(error)) {
183 return error;
184 }
185 const auto left = std::chrono::duration_cast<std::chrono::milliseconds>(deadline - std::chrono::steady_clock::now());
186 if (left <= std::chrono::milliseconds::zero()) {
187 return std::make_error_code(std::errc::timed_out);
188 }
189 const int ready = wait_for(s, POLLOUT, left);
190 if (ready == 0) {
191 return std::make_error_code(std::errc::timed_out);
192 }
193 if (ready < 0) {
194 return last_error();
195 }
196 continue;
197 }
198 data = data.subspan(static_cast<std::size_t>(sent));
199 }
200 return {};
201}

205std::error_code TcpSocket::read(std::span<std::byte> buffer, std::chrono::milliseconds timeout)
206{
207 const auto s = static_cast<Native>(m_socket);
208 bool started_reading = false;
209 while (!buffer.empty()) {
210 const int ready = wait_for(s, POLLIN, timeout);
211 if (ready == 0) {
212 return std::make_error_code(started_reading ? std::errc::io_error : std::errc::timed_out);
213 }
214 if (ready < 0) {
215 return last_error();
216 }
217#ifdef _WIN32
218 const int got = ::recv(s, reinterpret_cast<char*>(buffer.data()), static_cast<int>(buffer.size()), 0);
219#else
220 const auto got = ::recv(s, buffer.data(), buffer.size(), 0);
221#endif
222 if (got == 0) {
223 return std::make_error_code(std::errc::connection_reset);
224 }
225 if (got < 0) {
226 const std::error_code error = last_error();
227 if (would_block(error)) {
228 continue;
229 }
230 return error;
231 }
232 started_reading = true;
233 buffer = buffer.subspan(static_cast<std::size_t>(got));
234 }
235 return {};
236}
