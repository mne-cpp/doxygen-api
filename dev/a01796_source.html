52CompensatorPanel::CompensatorPanel()
53: Panel("CompensatorView", "Compensators")
54{
55 set_width(kPanelWidth);
56 m_listener = form().add_listener([this](const std::string& key, const std::optional<SettingValue>& value) {
57 if (m_updating || !value || !key.starts_with(kFieldPrefix)) {
58 return;
59 }
60 int kind = 0;
61 const char* first = key.data() + std::char_traits<char>::length(kFieldPrefix);
62 std::from_chars(first, key.data() + key.size(), kind);
63 if (const bool* on = std::get_if<bool>(&*value)) {

143void CompensatorPanel::load(const SettingsStore& store, const std::string& path)
144{
145 if (path.empty()) {
146 return;
147 }
148 // "kind:on" pairs separated by commas, e.g. "1:0,3:1"
149 const std::string stored = store.read_string(settings_key(path, "active"), {});
150 std::size_t begin = 0;
151 while (begin < stored.size()) {
152 const std::size_t end = std::min(stored.find(',', begin), stored.size());
153 const std::string_view entry(stored.data() + begin, end - begin);
154 const std::size_t colon = entry.find(':');
155 int kind = 0;
156 if (colon != std::string_view::npos && std::from_chars(entry.data(), entry.data() + colon, kind).ec == std::errc()) {
157 m_active[kind] = entry.substr(colon + 1) == "1";
158 }
159 begin = end + 1;
160 }
161}

174 store.set_value(settings_key(path, "active"), stored);
