73BadChannelsMaxwellResult UTILSLIB::findBadChannelsMaxwell(
74 const MatrixXd& matData,
75 const FiffInfo& info,
76 const BadChannelsMaxwellParams& params)
77{
78 BadChannelsMaxwellResult result;
79
80 // Build SSS basis
81 SSSParams sssParams;
82 sssParams.iOrderIn = params.iOrderIn;
83 sssParams.iOrderOut = params.iOrderOut;
84 sssParams.origin = params.origin;
85
86 SSS::Basis basis = SSS::computeBasis(info, sssParams);
87
88 if (basis.megChannelIdx.isEmpty()) {
89 qWarning() << "[findBadChannelsMaxwell] No MEG channels found.";
90 return result;
91 }
92
93 const int nMeg = basis.megChannelIdx.size();
94
95 // Extract MEG data
96 int nTimes = static_cast<int>(matData.cols());
97 MatrixXd megData(nMeg, nTimes);
98 for (int i = 0; i < nMeg; ++i) {
99 megData.row(i) = matData.row(basis.megChannelIdx[i]);
100 }
101
102 // Apply SSS reconstruction
103 MatrixXd reconData = SSS::apply(matData, basis);
104
105 // Extract reconstructed MEG channels
106 MatrixXd megRecon(nMeg, nTimes);
107 for (int i = 0; i < nMeg; ++i) {
108 megRecon.row(i) = reconData.row(basis.megChannelIdx[i]);
109 }
110
111 // Compute per-channel residual (RMS of difference)
112 result.residuals.resize(nMeg);
113 for (int i = 0; i < nMeg; ++i) {
114 VectorXd diff = megData.row(i) - megRecon.row(i);
115 result.residuals(i) = std::sqrt(diff.squaredNorm() / nTimes);
116 }
117
118 // Compute z-scores
119 // Use median/MAD for robustness
120 VectorXd sorted = result.residuals;
121 std::sort(sorted.data(), sorted.data() + sorted.size());
122
123 double median = sorted(nMeg / 2);
124 if (nMeg % 2 == 0 && nMeg > 1) {
125 median = (sorted(nMeg / 2 - 1) + sorted(nMeg / 2)) / 2.0;
126 }
127
128 VectorXd absDev(nMeg);
129 for (int i = 0; i < nMeg; ++i) {
130 absDev(i) = std::abs(result.residuals(i) - median);
131 }
132 std::sort(absDev.data(), absDev.data() + absDev.size());
133
134 double mad = absDev(nMeg / 2);
135 if (nMeg % 2 == 0 && nMeg > 1) {
136 mad = (absDev(nMeg / 2 - 1) + absDev(nMeg / 2)) / 2.0;
137 }
138
139 // Robust scale: MAD * 1.4826 (consistency constant for normal distribution)
140 double scale = mad * 1.4826;
141 if (scale < 1e-30) scale = 1e-30;
142
143 result.zScores.resize(nMeg);
144 for (int i = 0; i < nMeg; ++i) {
145 result.zScores(i) = (result.residuals(i) - median) / scale;
146 }
147
148 // Flag channels exceeding threshold
149 for (int i = 0; i < nMeg; ++i) {
150 if (result.zScores(i) > params.dZThreshold) {
151 int origIdx = basis.megChannelIdx[i];
152 result.badIndices.append(origIdx);
153 if (origIdx < info.ch_names.size()) {
154 result.badChannels.append(info.ch_names[origIdx]);
155 }
156 }
157 }
158
159qInfo("[findBadChannelsMaxwell] Detected %lld bad channel(s) out of %d MEG channels.",
160 static_cast<long long>(result.badChannels.size()), nMeg);
161
162 return result;
163}
