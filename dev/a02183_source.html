53void FilterPlot::setFilter(const FilterKernel& filter)
54{
55 const Eigen::RowVectorXcd& spectrum = filter.fft_coefficients();
56 if (filter.coefficients().cols() == 0 || spectrum.cols() == 0) {
57 return;
58 }
59 std::vector<double> magnitude(static_cast<std::size_t>(spectrum.cols()));
60 for (Eigen::Index i = 0; i < spectrum.cols(); ++i) {
61 magnitude[static_cast<std::size_t>(i)] = std::abs(spectrum(i));
62 }
63 const double nyquist = filter.sampling_frequency() / 2.0;
64 std::vector<double> cutoffs;
65 for (double hz : {filter.highpass_freq(), filter.lowpass_freq()}) {
66 if (hz > 0.0 && hz < nyquist) {
67 cutoffs.push_back(hz);
68 }
69 }
70 m_plot.set_response(std::move(magnitude), nyquist);
71 m_plot.set_cutoffs(std::move(cutoffs));
72 m_plot.set_title(filter.name());
73 update();
74}
