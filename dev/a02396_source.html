80std::shared_ptr<FiffDirNode> FiffIO::readRaw(FiffStream& stream, std::unique_ptr<mne::ByteStream> rawBytes)
81{
82 FiffInfo info;
83 std::shared_ptr<FiffDirNode> infoNode;
84 if (!stream.open() || !stream.read_meas_info(stream.dirtree(), info, infoNode) || !infoNode) {
85 mne::Log::warning("[FiffIO::read] Dir tree could not be read");
86 return nullptr;
87 }
88 const std::shared_ptr<FiffDirNode>& tree = stream.dirtree();
89 if (tree->has_kind(FIFFB_RAW_DATA) || tree->has_kind(FIFFB_PROCESSED_DATA) || tree->has_kind(FIFFB_CONTINUOUS_DATA) ||
90 tree->has_kind(FIFFB_SMSH_RAW_DATA)) {
91 try {
92 raw.push_back(std::make_shared<FiffRawData>(std::move(rawBytes)));
93 mne::Log::info("Finished reading raw data!\n");
94 } catch (const std::exception& e) {
95 mne::Log::warning("[FiffIO::read] Could not read raw data: {}", e.what());
96 }
97 }
98 return tree;
99}

103bool FiffIO::read(const std::filesystem::path& path)
104{
105 FiffStream stream(path);
106 const std::shared_ptr<FiffDirNode> tree = readRaw(stream, mne::FileByteStream::open(path, mne::FileByteStream::Mode::Read));
107 if (!tree) {
108 return false;
109 }
110 if (tree->has_kind(FIFFB_EVOKED)) {
111 if (auto set = FiffEvokedSet::read(path)) {
112 for (const FiffEvoked& one : set->evoked) {
113 evoked.push_back(std::make_shared<FiffEvoked>(one));
114 }
115 } else {
116 mne::Log::warning("[FiffIO::read] {}", set.error().what());
117 }
118 }
119 return true;
120}

136bool FiffIO::write(const std::filesystem::path& path, fiff_int_t type, fiff_int_t idx) const
137{
138 switch (type) {
139 case FIFFB_RAW_DATA: {
140 if (idx >= 0) {
141 return write_raw(path, idx);
142 }
143 bool ok = true;
144 for (fiff_int_t i = 0; i < static_cast<fiff_int_t>(raw.size()); ++i) {
145 std::filesystem::path one = path;
146 one.replace_filename(std::format("{}_raw-{}{}", mne::utf8_from_path(path.stem()), i, mne::utf8_from_path(path.extension())));
147 mne::Log::info("Writing raw data set {} to {} ...", i, mne::utf8_from_path(one));
148 ok &= write_raw(one, i);
149 }
150 return ok;
151 }
152 case FIFFB_EVOKED: {
153 if (evoked.empty() || idx >= static_cast<fiff_int_t>(evoked.size())) {
154 return false;
155 }
156 FiffEvokedSet set;
157 set.info = evoked.front()->info;
158 for (fiff_int_t i = 0; i < static_cast<fiff_int_t>(evoked.size()); ++i) {
159 if (idx < 0 || i == idx) {
160 set.evoked.push_back(*evoked[i]);
161 }
162 }
163 if (auto saved = set.save(path); !saved) {
164 mne::Log::warning("[FiffIO::write] {}", saved.error().what());
165 return false;
166 }
167 return true;
168 }
169 default:
170 return false;
171 }
172}

176bool FiffIO::write_raw(const std::filesystem::path& path, fiff_int_t idx) const
177{
178 if (idx < 0 || idx >= static_cast<fiff_int_t>(raw.size())) {
179 return false;
180 }
181 const FiffRawData& data = *raw[idx];
182 RowVectorXd cals;
183 auto outfid = FiffStream::start_writing_raw(path, data.info, cals);
184 if (!outfid) {
185 return false;
186 }
187
188 // Read and write in 30 s chunks
189 const fiff_int_t quantum = static_cast<fiff_int_t>(std::ceil(30.0 * data.info.sfreq));
190 SparseMatrix<double> mult;
191 RowVectorXi sel;
192 MatrixXd buffer;
193 MatrixXd times;
194 for (fiff_int_t first = data.first_samp; first < data.last_samp; first += quantum) {
195 const fiff_int_t last = std::min(first + quantum - 1, data.last_samp);
196 if (!data.read_raw_segment(buffer, times, mult, first, last, sel)) {
197 mne::Log::warning("error during read_raw_segment\n");
198 return false;
199 }
200 if (first == data.first_samp && first > 0) {
201 fiff_int_t firstSample = first;
202 outfid->write_int(FIFF_FIRST_SAMPLE, &firstSample);
203 }
204 outfid->write_raw_buffer(buffer, cals);
205 }
206 outfid->finish_writing_raw();
207 return true;
208}
