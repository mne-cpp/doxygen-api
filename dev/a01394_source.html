169 constexpr DWORD kFlags = FILE_ATTRIBUTE_NORMAL | FILE_FLAG_WRITE_THROUGH;

172 if (device == INVALID_HANDLE_VALUE) {

178 }
179 DWORD version = 0;
180 DWORD returned = 0;
181 ::DeviceIoControl(device, IOCTL_BA_DRIVERVERSION, nullptr, 0, &version, sizeof(version), &returned, nullptr);
182 return std::unique_ptr<BrainAmpTransport>(std::make_unique<BrainAmpDriver>(device, usb, version));
