78JsonObject MnaNode::to_json() const
79{
80 JsonObject json = extras;
81 json["id"] = id;
82 json["op_type"] = op_type;
83 json["exec_mode"] = execModeToString(exec_mode);
84 json["attributes"] = attributes;
85 json["inputs"] = detail::to_json_array(inputs);
86 json["outputs"] = detail::to_json_array(outputs);
87
88 if (exec_mode == MnaNodeExecMode::Ipc) {
89 json["ipc_command"] = ipc_command;
90 json["ipc_args"] = JsonArray::from_string_list(ipc_args);
91 json["ipc_work_dir"] = ipc_work_dir;
92 json["ipc_transport"] = ipc_transport;
93 }
94 if (exec_mode == MnaNodeExecMode::Script) {
95 json["script"] = script.to_json();
96 }
97 if (JsonObject verificationJson = verification.to_json(); !verificationJson.empty()) {
98 json["verification"] = std::move(verificationJson);
99 }
100 if (!tool_version.empty())
101 json["tool_version"] = tool_version;
102 if (executed_at)
103 json["executed_at"] = format_iso8601(*executed_at, false);
104 json["dirty"] = dirty;
105 return json;
106}

110MnaNode MnaNode::from_json(const JsonObject& json)
111{
112 MnaNode node;
113 node.id = json["id"].to_std_string();
114 node.op_type = json["op_type"].to_std_string();
115 node.exec_mode = execModeFromString(json["exec_mode"].to_string());
116 node.attributes = json["attributes"].to_object();
117 node.inputs = detail::from_json_array<MnaPort>(json["inputs"]);
118 node.outputs = detail::from_json_array<MnaPort>(json["outputs"]);
119 node.ipc_command = json["ipc_command"].to_std_string();
120 node.ipc_args = json["ipc_args"].to_array().to_string_list();
121 node.ipc_work_dir = json["ipc_work_dir"].to_std_string();
122 node.ipc_transport = json["ipc_transport"].to_std_string();
123 if (json.contains("script")) {
124 node.script = MnaScript::from_json(json["script"].to_object());
125 }
126 if (json.contains("verification")) {
127 node.verification = MnaVerification::from_json(json["verification"].to_object());
128 }
129 node.tool_version = json["tool_version"].to_std_string();
130 node.executed_at = parse_iso8601(json["executed_at"].to_string());
131 node.dirty = json["dirty"].to_bool(true);
132 node.extras = json.without({"id", "op_type", "exec_mode", "attributes", "inputs", "outputs", "ipc_command", "ipc_args",
133 "ipc_work_dir", "ipc_transport", "script", "verification", "tool_version", "executed_at",
134 "dirty"});
135 return node;
136}
