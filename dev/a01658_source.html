102std::optional<std::pair<int, BrainFlowDescription>> mne::detail::builtin_brainflow_board(std::string_view board)
103{
104 const std::optional<int> id = parse_id(board);
105 const std::string key = brainflow_board_key(board);
106 for (const auto& [known, json] : kBoards) {
107 auto description = parse_brainflow_description(json);
108 if (description && (id ? *id == known : brainflow_board_key(description->name) == key)) {
109 return std::pair{known, std::move(*description)};
110 }
111 }
112 return std::nullopt;
113}

117AcqExpected<std::pair<int, BrainFlowDescription>> mne::detail::brainflow_board(std::string_view board)
118{
119 if (auto builtin = builtin_brainflow_board(board)) {
120 return std::move(*builtin);
121 }
122 auto library = BrainFlowLibrary::load(brainflow_library_candidates());
123 if (!library) {
124 return AcqError(AcqErrorCode::InvalidParameter,
125 std::format("the layout of BrainFlow board '{}' needs BrainFlow ({}); without it the known boards are "
126 "synthetic, cyton, ganglion and cytondaisy",
127 board, library.error().text()));
128 }
129 const std::optional<int> parsed = parse_id(board);
130 auto id = parsed ? AcqExpected<int>(*parsed) : (*library)->find_board(board);
131 if (!id) {
132 return id.error();
133 }
134 auto description = (*library)->description(*id);
135 if (!description) {
136 return description.error();
137 }
138 return std::pair{*id, std::move(*description)};
139}

155AcqStreamInfo mne::detail::brainflow_stream_info(const BrainFlowDescription& description, std::string sourceId)
156{
157 AcqStreamInfo info;
158 info.name = "eeg";
159 info.sourceId = std::move(sourceId);
160 info.format = AcqSampleFormat::Float64;
161 info.layout = AcqLayout::Interleaved;
162 info.sfreq = description.sampling_rate;
163 info.channels.resize(description.rows);
164 std::vector<bool> named(description.rows, false);
165 for (const Role& role : kRoles) {
166 const std::span<const int> rows = description.role(role.key);
167 const bool eeg = std::string_view(role.key) == "eeg_channels" && description.eeg_names.size() == rows.size();
168 for (std::size_t r = 0; r < rows.size(); ++r) {
169 const auto row = static_cast<std::size_t>(rows[r]);
170 if (named[row]) {
171 continue;
172 }
173 named[row] = true;
174 AcqChannelInfo& channel = info.channels[row];
175 channel.name = eeg ? description.eeg_names[r] : std::string(role.key).ends_with("_channel") ? role.prefix
176 : std::format("{} {:03}", role.prefix, r + 1);
177 channel.kind = role.kind;
178 channel.unit = role.unit;
179 channel.scale = role.scale;
180 }
181 }
182 for (std::size_t row = 0; row < description.rows; ++row) {
183 if (!named[row]) {
184 info.channels[row].name = std::format("ROW {:03}", row);
185 info.channels[row].kind = AcqChannelKind::Misc;
186 }
187 }
188 return info;
189}

210AcqError BrainFlowRowWriter::commit(const AcqWriteView& view, const double* data, std::size_t samples, std::size_t sampleStride, std::size_t rowStride)
211{
212 std::size_t done = 0;
213 for (std::size_t p = 0; p < view.partCount(); ++p) {
214 const AcqWritePart& part = view.part(p);
215 for (std::size_t s = 0; s < part.samples; ++s, ++done) {
216 const double* sample = data + done * sampleStride;
217 if (rowStride == 1) {
218 std::memcpy(part.at(0, s), sample, m_rows * sizeof(double));
219 continue;
220 }
221 for (std::size_t r = 0; r < m_rows; ++r) {
222 std::memcpy(part.at(r, s), sample + r * rowStride, sizeof(double));
223 }
224 }
225 }
226 const std::uint64_t first = m_writer->writePosition();
227 if (m_marker_row) {
228 for (std::size_t s = 0; s < samples; ++s) {
229 const double marker = data[s * sampleStride + *m_marker_row * rowStride];
230 if (marker != 0.0) {
231 if (AcqError e = m_writer->pushEvent({first + s, 0, static_cast<std::int32_t>(std::lround(marker))}); !e.ok()) {
232 return e;
233 }
234 }
235 }
236 }
237 m_writer->commit(samples);
238 if (m_timestamp_row) {
239 publishClock(first + samples - 1, data[(samples - 1) * sampleStride + *m_timestamp_row * rowStride]);
240 }
241 return {};
242}

246void BrainFlowRowWriter::publishClock(std::uint64_t sample, double timestamp)
247{
248 const auto steady = std::chrono::steady_clock::now();
249 if (m_clocked && steady - m_last_clock < kClockRefresh) {
250 return;
251 }
252 m_clocked = true;
253 m_last_clock = steady;
254 const auto steadyNs = std::chrono::duration_cast<std::chrono::nanoseconds>(steady.time_since_epoch()).count();
255 const auto utcNs = std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::system_clock::now().time_since_epoch()).count();
256 AcqClockModel clock;
257 clock.sfreq = m_sfreq;
258 clock.anchorSample = sample;
259 clock.utcOffsetNs = utcNs - steadyNs;
260 clock.anchorHostNs = std::llround(timestamp * 1e9) - clock.utcOffsetNs;
261 clock.source = AcqClockSource::Device;
262 m_writer->setClock(clock);
263}

267AcqExpected<BrainFlowStreamer> BrainFlowStreamer::open(const BrainFlowDescription& description, std::string address, std::uint16_t port,
268 std::size_t batch)
269{
270 auto socket = mne::UdpSocket::bind("0.0.0.0", 0);
271 if (!socket) {
272 return AcqError(AcqErrorCode::DeviceIo, "cannot open a UDP socket: " + socket.error().message());
273 }
274 BrainFlowStreamer streamer;
275 streamer.m_socket = std::move(*socket);
276 streamer.m_address = std::move(address);
277 streamer.m_port = port;
278 streamer.m_rows = description.rows;
279 streamer.m_batch = std::max<std::size_t>(batch, 1);
280 const std::span<const int> eeg = description.role("eeg_channels");
281 streamer.m_eeg_rows.assign(eeg.begin(), eeg.end());
282 streamer.m_package_row = row_of(description, "package_num_channel");
283 streamer.m_timestamp_row = row_of(description, "timestamp_channel");
284 streamer.m_marker_row = row_of(description, "marker_channel");
285 streamer.m_datagram.reserve(streamer.m_batch * streamer.m_rows);
286 return streamer;
287}

291AcqError BrainFlowStreamer::push(std::span<const double> eeg, double marker, double timestamp)
292{
293 const std::size_t at = m_datagram.size();
294 m_datagram.resize(at + m_rows, 0.0);
295 double* row = m_datagram.data() + at;
296 for (std::size_t c = 0; c < std::min(eeg.size(), m_eeg_rows.size()); ++c) {
297 row[m_eeg_rows[c]] = eeg[c];
298 }
299 if (m_package_row >= 0) {
300 row[m_package_row] = static_cast<double>(m_sample % 256);
301 }
302 if (m_timestamp_row >= 0) {
303 row[m_timestamp_row] = timestamp;
304 }
305 if (m_marker_row >= 0) {
306 row[m_marker_row] = marker;
307 }
308 ++m_sample;
309 if (m_datagram.size() < m_batch * m_rows) {
310 return {};
311 }
312 const std::error_code error = m_socket.send_to(std::as_bytes(std::span(m_datagram)), m_address, m_port);
313 m_datagram.clear();
314 if (error) {
315 return {AcqErrorCode::DeviceIo, std::format("cannot send to {}:{}: {}", m_address, m_port, error.message())};
316 }
317 return {};
