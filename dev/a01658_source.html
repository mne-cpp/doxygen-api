47std::string escape(const std::string& text)
48{
49 std::string out;
50 for (const char c : text) {
51 if (c == '\\' || c == '\n' || c == '=') {
52 out += '\\';
53 }
54 out += c == '\n' ? 'n' : c;

65ProjectorsPanel::ProjectorsPanel()
66: Panel("ProjectorsView", "Projectors")
67{
68 set_width(kPanelWidth);
69 m_listener = form().add_listener([this](const std::string& key, const std::optional<SettingValue>& value) {
70 const bool* on = value ? std::get_if<bool>(&*value) : nullptr;
71 if (m_updating || !on || !key.starts_with(kFieldPrefix)) {
72 return;

102void ProjectorsPanel::set_projectors(const std::vector<mne::FiffProj>& projs)
103{
104 m_updating = true;
105 for (std::size_t i = 0; i < m_projs.size(); ++i) {
106 form().remove(field_key(i));
107 }
108 form().remove(kAllKey);
109 m_projs = projs;
110 for (std::size_t i = 0; i < m_projs.size(); ++i) {
111 mne::FiffProj& proj = m_projs[i];
112 proj.active = m_active.try_emplace(proj.desc, proj.active).first->second;
113 SettingField field;
114 field.key = field_key(i);
115 field.kind = SettingField::Kind::Bool;
116 field.label = proj.desc;
117 field.default_value = proj.active;
118 field.persistent = false;
119 form().add(std::move(field));
120 }
121 if (!m_projs.empty()) {
122 SettingField all;
123 all.key = kAllKey;
124 all.kind = SettingField::Kind::Bool;
125 all.label = "Enable all";
126 all.default_value = false;
127 all.persistent = false;
128 form().add(std::move(all));
129 }
130 m_updating = false;
131 changed();

186void ProjectorsPanel::load(const SettingsStore& store, const std::string& path)
187{
188 if (path.empty()) {
189 return;
190 }
191 // One "description=0|1" per line, with '\', '=' and line breaks escaped
192 const std::string stored = store.read_string(settings_key(path, "active"), {});
193 std::string desc;
194 bool escaped = false;
195 for (std::size_t i = 0; i < stored.size(); ++i) {
196 const char c = stored[i];
197 if (escaped) {
198 desc += c == 'n' ? '\n' : c;
199 escaped = false;
200 } else if (c == '\\') {
201 escaped = true;
202 } else if (c == '=' && i + 1 < stored.size()) {
203 m_active[desc] = stored[i + 1] == '1';
204 desc.clear();
205 i = std::min(stored.find('\n', i), stored.size());
206 } else {
207 desc += c;
208 }
209 }
210}

214void ProjectorsPanel::save(SettingsStore& store, const std::string& path) const
215{
216 if (path.empty()) {
217 return;
218 }
219 std::string stored;
220 for (const auto& [desc, on] : m_active) {
221 stored += escape(desc) + (on ? "=1\n" : "=0\n");
222 }
223 store.set_value(settings_key(path, "active"), stored);
224}
