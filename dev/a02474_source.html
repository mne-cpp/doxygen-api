80bool FiffTag::matrix_dimensions(std::int32_t& p_ndim, std::vector<std::int32_t>& p_Dims) const
81{
82 p_Dims.clear();
83 p_ndim = 0;
84 if (!this->is_matrix() || this->data() == nullptr || this->size() < 4)
85 return false;
86
87 //
88 // The dimensions trail the data: [nz,] dims..., ndim. A corrupt file must not make the readers
89 // index outside the tag, so the trailer has to fit and a dense matrix has to fit before it.
90 //
91 const std::int32_t* t_pInt32 = reinterpret_cast<const std::int32_t*>(this->data());
92 const std::ptrdiff_t nInts = this->size() / 4;
93 const std::int32_t ndim = t_pInt32[nInts - 1];
94 const int coding = fiff_type_matrix_coding(this->type);
95 const bool sparse = coding == FIFFTS_MC_CCS || coding == FIFFTS_MC_RCS;
96 if (coding != FIFFTS_MC_DENSE && !sparse) {
97 mne::Log::warning("Cannot handle other than dense or sparse matrices yet.");
98 return false;
99 }
100 const std::int32_t nTrailer = ndim + (sparse ? 2 : 1);
101 if (ndim <= 0 || ndim > FIFFC_MATRIX_MAX_DIM || nTrailer > nInts)
102 return false;
103
104 std::vector<std::int32_t> dims;
105 std::int64_t nValues = 1;
106 for (std::ptrdiff_t i = nInts - nTrailer; i < nInts - 1; ++i) {
107 if (t_pInt32[i] < 0)
108 return false;
109 dims.push_back(t_pInt32[i]);
110 nValues = std::min<std::int64_t>(nValues * t_pInt32[i], this->size()); // capped: enough to compare, no overflow
111 }
112 if (!sparse) {
113 const int base = this->data_type();
114 const std::int64_t valueSize = (base == FIFFT_DOUBLE || base == FIFFT_COMPLEX_FLOAT) ? 8 : (base == FIFFT_COMPLEX_DOUBLE ? 16 : 4);
115 if (nValues * valueSize > static_cast<std::int64_t>(nInts - nTrailer) * 4)
116 return false;
117 }
118
119 p_ndim = ndim;
120 p_Dims = dims;
121 return true;
122}

293void FiffTag::convert_matrix_from_file_data(FiffTag& tag)
294/*
295 * Assumes that the input is in the non-native byte order and needs to be swapped to the other one
296 */
297{
298 int ndim;
299 int k;
300 int *dimp, *data, kind, np, nz;
301 float* fdata;
302 double* ddata;
303 const std::size_t tsize = tag.size();
304
305 if (fiff_type_fundamental(tag.type) != FIFFTS_FS_MATRIX)
306 return;
307 if (tag.data() == nullptr)
308 return;
309 if (tsize < sizeof(fiff_int_t))
310 return;
311
312 dimp = (reinterpret_cast<fiff_int_t*>((tag.data()) + tag.size() - sizeof(fiff_int_t)));
313 swap_intp(dimp);
314 ndim = *dimp;
315 if (fiff_type_matrix_coding(tag.type) == FIFFTS_MC_DENSE) {
316 if (tsize < (ndim + 1) * sizeof(fiff_int_t))
317 return;
318 dimp = dimp - ndim;
319 for (k = 0, np = 1; k < ndim; k++) {
320 swap_intp(dimp + k);
321 np = np * dimp[k];
322 }
323 } else {
324 if (tsize < (ndim + 2) * sizeof(fiff_int_t))
325 return;
326 if (ndim > 2) /* Not quite sure what to do */
327 return;
328 dimp = dimp - ndim - 1;
329 for (k = 0; k < ndim + 1; k++)
330 swap_intp(dimp + k);
331 nz = dimp[0];
332 if (fiff_type_matrix_coding(tag.type) == FIFFTS_MC_CCS)
333 np = nz + dimp[2] + 1; /* nz + n + 1 */
334 else if (fiff_type_matrix_coding(tag.type) == FIFFTS_MC_RCS)
335 np = nz + dimp[1] + 1; /* nz + m + 1 */
336 else
337 return; /* Don't know what to do */
338 /*
339 * Take care of the indices
340 */
341 for (data = reinterpret_cast<int*>(tag.data()) + nz, k = 0; k < np; k++)
342 swap_intp(data + k);
343 np = nz;
344 }
345 /*
346 * Now convert data...
347 */
348 kind = fiff_type_base(tag.type);
349 if (kind == FIFFT_INT) {
350 for (data = reinterpret_cast<int*>(tag.data()), k = 0; k < np; k++)
351 swap_intp(data + k);
352 } else if (kind == FIFFT_FLOAT) {
353 for (fdata = reinterpret_cast<float*>(tag.data()), k = 0; k < np; k++)
354 swap_floatp(fdata + k);
355 } else if (kind == FIFFT_DOUBLE) {
356 for (ddata = reinterpret_cast<double*>(tag.data()), k = 0; k < np; k++)
357 swap_doublep(ddata + k);
358 } else if (kind == FIFFT_COMPLEX_FLOAT) {
359 for (fdata = reinterpret_cast<float*>(tag.data()), k = 0; k < 2 * np; k++)
360 swap_floatp(fdata + k);
361 } else if (kind == FIFFT_COMPLEX_DOUBLE) {
362 for (ddata = reinterpret_cast<double*>(tag.data()), k = 0; k < 2 * np; k++)
363 swap_doublep(ddata + k);
364 }
365 return;
366}

370void FiffTag::convert_matrix_to_file_data(FiffTag& tag)
371/*
372 * Assumes that the input is in the NATIVE_ENDIAN byte order and needs to be swapped to the other one
373 */
374{
375 int ndim;
376 int k;
377 int *dimp, *data, kind, np;
378 float* fdata;
379 double* ddata;
380 const std::size_t tsize = tag.size();
381
382 if (fiff_type_fundamental(tag.type) != FIFFTS_FS_MATRIX)
383 return;
384 if (tag.data() == nullptr)
385 return;
386 if (tsize < sizeof(fiff_int_t))
387 return;
388
389 dimp = (reinterpret_cast<fiff_int_t*>((tag.data()) + tag.size() - sizeof(fiff_int_t)));
390 ndim = *dimp;
391 swap_intp(dimp);
392
393 if (fiff_type_matrix_coding(tag.type) == FIFFTS_MC_DENSE) {
394 if (tsize < (ndim + 1) * sizeof(fiff_int_t))
395 return;
396 dimp = dimp - ndim;
397 for (k = 0, np = 1; k < ndim; k++) {
398 np = np * dimp[k];
399 swap_intp(dimp + k);
400 }
401 } else {
402 if (tsize < (ndim + 2) * sizeof(fiff_int_t))
403 return;
404 if (ndim > 2) /* Not quite sure what to do */
405 return;
406 dimp = dimp - ndim - 1;
407 const int nz = dimp[0];
408 if (fiff_type_matrix_coding(tag.type) == FIFFTS_MC_CCS)
409 np = nz + dimp[2] + 1; /* nz + n + 1 */
410 else if (fiff_type_matrix_coding(tag.type) == FIFFTS_MC_RCS)
411 np = nz + dimp[1] + 1; /* nz + m + 1 */
412 else
413 return; /* Don't know what to do */
414 for (k = 0; k < ndim + 1; k++)
415 swap_intp(dimp + k);
416 for (data = reinterpret_cast<int*>(tag.data()) + nz, k = 0; k < np; k++)
417 swap_intp(data + k);
418 np = nz;
419 }
420 /*
421 * Now convert data...
422 */
423 kind = fiff_type_base(tag.type);
424 if (kind == FIFFT_INT) {
425 for (data = reinterpret_cast<int*>(tag.data()), k = 0; k < np; k++)
426 swap_intp(data + k);
427 } else if (kind == FIFFT_FLOAT) {
428 for (fdata = reinterpret_cast<float*>(tag.data()), k = 0; k < np; k++)
429 swap_floatp(fdata + k);
430 } else if (kind == FIFFT_DOUBLE) {
431 for (ddata = reinterpret_cast<double*>(tag.data()), k = 0; k < np; k++)
432 swap_doublep(ddata + k);
433 } else if (kind == FIFFT_COMPLEX_FLOAT) {
434 for (fdata = reinterpret_cast<float*>(tag.data()), k = 0; k < 2 * np; k++)
435 swap_floatp(fdata + k);
436 } else if (kind == FIFFT_COMPLEX_DOUBLE) {
437 for (ddata = reinterpret_cast<double*>(tag.data()), k = 0; k < 2 * np; k++)
438 swap_doublep(ddata + k);
439 }
440 return;
441}

443//=============================================================================================================
444void FiffTag::convert_tag_data(FiffTag& tag, int from_endian, int to_endian)
445{
446 // Element counts derived from tag->size(), which is a size type. A FIFF tag
447 // never holds more than INT_MAX elements, so narrowing is safe, but it has
448 // to be explicit: on Win64 size_t is 64-bit while int stays 32-bit.
449 int np;
450 int k, r;
451 std::byte* offset;
452 fiff_int_t* ithis;
453 fiff_short_t* sthis;
454 fiff_long_t* lthis;
455 float* fthis;
456 double* dthis;
457 FiffDataRef* drthis;
458
459 if (tag.data() == nullptr || tag.size() == 0)
460 return;
461
462 if (from_endian == FIFFV_NATIVE_ENDIAN)
463 from_endian = NATIVE_ENDIAN;
464 if (to_endian == FIFFV_NATIVE_ENDIAN)
465 to_endian = NATIVE_ENDIAN;
466
467 if (from_endian == to_endian)
468 return;
469
470 if (fiff_type_fundamental(tag.type) == FIFFTS_FS_MATRIX) {
471 if (from_endian == NATIVE_ENDIAN)
472 convert_matrix_to_file_data(tag);
473 else
474 convert_matrix_from_file_data(tag);
475 return;
476 }
477
478 switch (tag.type) {
479 case FIFFT_INT:
480 case FIFFT_UINT:
481 case FIFFT_JULIAN:
482 np = static_cast<int>(tag.size() / sizeof(fiff_int_t));
483 for (ithis = reinterpret_cast<fiff_int_t*>(tag.data()), k = 0; k < np; k++, ithis++)
484 swap_intp(ithis);
485 break;
486
487 case FIFFT_LONG:
488 case FIFFT_ULONG:
489 np = static_cast<int>(tag.size() / sizeof(fiff_long_t));
490 for (lthis = reinterpret_cast<fiff_long_t*>(tag.data()), k = 0; k < np; k++, lthis++)
491 swap_longp(lthis);
492 break;
493
494 case FIFFT_SHORT:
495 case FIFFT_DAU_PACK16:
496 case FIFFT_USHORT:
497 np = static_cast<int>(tag.size() / sizeof(fiff_short_t));
498 for (sthis = reinterpret_cast<fiff_short_t*>(tag.data()), k = 0; k < np; k++, sthis++)
499 *sthis = swap_short(*sthis);
500 break;
501
502 case FIFFT_FLOAT:
503 case FIFFT_COMPLEX_FLOAT:
504 np = static_cast<int>(tag.size() / sizeof(fiff_float_t));
505 for (fthis = reinterpret_cast<fiff_float_t*>(tag.data()), k = 0; k < np; k++, fthis++)
506 swap_floatp(fthis);
507 break;
508
509 case FIFFT_DOUBLE:
510 case FIFFT_COMPLEX_DOUBLE:
511 np = static_cast<int>(tag.size() / sizeof(fiff_double_t));
512 for (dthis = reinterpret_cast<fiff_double_t*>(tag.data()), k = 0; k < np; k++, dthis++)
513 swap_doublep(dthis);
514 break;
515
516 case FIFFT_OLD_PACK:
517 fthis = reinterpret_cast<float*>(tag.data());
518 /*
519 * Offset and scale...
520 */
521 swap_floatp(fthis + 0);
522 swap_floatp(fthis + 1);
523 sthis = reinterpret_cast<short*>(fthis + 2);
524 np = static_cast<int>((tag.size() - 2 * sizeof(float)) / sizeof(short));
525 for (k = 0; k < np; k++, sthis++)
526 *sthis = swap_short(*sthis);
527 break;
528
529 case FIFFT_DIR_ENTRY_STRUCT:
530 np = static_cast<int>(tag.size() / FiffDirEntry::storage_size());
531 for (k = 0; k < np; k++) {
532 offset = tag.data() + k * FiffDirEntry::storage_size();
533 ithis = reinterpret_cast<fiff_int_t*>(offset);
534 ithis[0] = swap_int(ithis[0]); //kind
535 ithis[1] = swap_int(ithis[1]); //type
536 ithis[2] = swap_int(ithis[2]); //size
537 ithis[3] = swap_int(ithis[3]); //pos
538 }
539 break;
540
541 case FIFFT_ID_STRUCT:
542 np = static_cast<int>(tag.size() / FiffId::storage_size());
543 for (k = 0; k < np; k++) {
544 offset = tag.data() + k * FiffId::storage_size();
545 ithis = reinterpret_cast<fiff_int_t*>(offset);
546 ithis[0] = swap_int(ithis[0]); //version
547 ithis[1] = swap_int(ithis[1]); //machid[0]
548 ithis[2] = swap_int(ithis[2]); //machid[1]
549 ithis[3] = swap_int(ithis[3]); //time.secs
550 ithis[4] = swap_int(ithis[4]); //time.usecs
551 }
552 break;
553
554 case FIFFT_CH_INFO_STRUCT:
555
556 np = static_cast<int>(tag.size() / FiffChInfo::storage_size());
557 for (k = 0; k < np; k++) {
558 offset = tag.data() + k * FiffChInfo::storage_size();
559 ithis = reinterpret_cast<fiff_int_t*>(offset);
560 fthis = reinterpret_cast<float*>(offset);
561
562 ithis[0] = swap_int(ithis[0]); //scanno
563 ithis[1] = swap_int(ithis[1]); //logno
564 ithis[2] = swap_int(ithis[2]); //kind
565 swap_floatp(&fthis[3]); //range
566 swap_floatp(&fthis[4]); //cal
567 ithis[5] = swap_int(ithis[5]); //coil_type
568 for (r = 0; r < 12; ++r)
569 swap_floatp(&fthis[6 + r]); //loc
570 ithis[18] = swap_int(ithis[18]); //unit
571 ithis[19] = swap_int(ithis[19]); //unit_mul
572 }
573
574 break;
575
576 case FIFFT_CH_POS_STRUCT:
577
578 np = static_cast<int>(tag.size() / FiffChPos::storage_size());
579 for (k = 0; k < np; ++k) {
580 offset = tag.data() + k * FiffChPos::storage_size();
581 ithis = reinterpret_cast<fiff_int_t*>(offset);
582 fthis = reinterpret_cast<float*>(offset);
583
584 ithis[0] = swap_int(ithis[0]); //coil_type
585 for (r = 0; r < 12; r++)
586 swap_floatp(&fthis[1 + r]); //r0, ex, ey, ez
587 }
588
589 break;
590
591 case FIFFT_DIG_POINT_STRUCT:
592
593 np = static_cast<int>(tag.size() / FiffDigPoint::storage_size());
594
595 for (k = 0; k < np; k++) {
596 offset = tag.data() + k * FiffDigPoint::storage_size();
597 ithis = reinterpret_cast<fiff_int_t*>(offset);
598 fthis = reinterpret_cast<float*>(offset);
599
600 ithis[0] = swap_int(ithis[0]); //kind
601 ithis[1] = swap_int(ithis[1]); //ident
602
603 for (r = 0; r < 3; ++r)
604 swap_floatp(&fthis[2 + r]); //r
605 }
606 break;
607
608 case FIFFT_COORD_TRANS_STRUCT:
609
610 np = static_cast<int>(tag.size() / FiffCoordTrans::storage_size());
611
612 for (k = 0; k < np; ++k) {
613 offset = tag.data() + k * FiffCoordTrans::storage_size();
614 ithis = reinterpret_cast<fiff_int_t*>(offset);
615 fthis = reinterpret_cast<float*>(offset);
616
617 ithis[0] = swap_int(ithis[0]);
618 ithis[1] = swap_int(ithis[1]);
619
620 for (r = 0; r < 24; ++r)
621 swap_floatp(&fthis[2 + r]);
622 }
623 break;
624
625 case FIFFT_DATA_REF_STRUCT:
626 np = static_cast<int>(tag.size() / FiffDataRef::storage_size());
627 for (drthis = reinterpret_cast<FiffDataRef*>(tag.data()), k = 0; k < np; k++, drthis++) {
628 drthis->type = swap_int(drthis->type);
629 drthis->endian = swap_int(drthis->endian);
630 drthis->size = swap_long(drthis->size);
631 drthis->offset = swap_long(drthis->offset);
632 }
633 break;
634
635 default:
636 break;
637 }
638 return;
639}

676const std::int32_t* FiffTag::to_julian() const
677{
678 if (this->is_matrix() || this->data_type() != FIFFT_JULIAN) {
679 mne::Log::warning("Expected a julian tag : {} (found data type {} instead)\n", this->kind, this->data_type());
