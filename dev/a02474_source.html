90 KeyEvent key;

92 key.timestamp_ms = now;
93 switch (event.key) {
94 case SDLK_ESCAPE:
95 key.key = Key::Escape;
96 break;
97 case SDLK_RETURN:
98 case SDLK_KP_ENTER:
99 key.key = Key::Return;
100 break;
101 case SDLK_TAB:
102 key.key = Key::Tab;
103 break;
104 case SDLK_BACKSPACE:
105 key.key = Key::Backspace;
106 break;
107 case SDLK_DELETE:
108 key.key = Key::Delete;
109 break;
110 case SDLK_LEFT:
111 key.key = Key::Left;
112 break;
113 case SDLK_RIGHT:
114 key.key = Key::Right;
115 break;
116 case SDLK_UP:
117 key.key = Key::Up;
118 break;
119 case SDLK_DOWN:
120 key.key = Key::Down;
121 break;
122 case SDLK_HOME:
123 key.key = Key::Home;
124 break;
125 case SDLK_END:
126 key.key = Key::End;
127 break;
128 case SDLK_PAGEUP:
129 key.key = Key::PageUp;
130 break;
131 case SDLK_PAGEDOWN:
132 key.key = Key::PageDown;
133 break;
134 default:
135 // SDL names printable keys by their unshifted character; letters are reported in upper case
136 if (event.key >= 0x20 && event.key < 0x7f) {

165SdlHost::SdlHost(Options options)
166: m_options(std::move(options))
167{
168 if (!SDL_Init(SDL_INIT_VIDEO)) {
169 m_error = SDL_GetError();
170 return;
171 }
172 m_window = SDL_CreateWindow(m_options.title.c_str(), m_options.width, m_options.height, SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
173 // Any renderer only shows a texture; SDL's software renderer works without a GPU, e.g. on the dummy driver
174 m_renderer = m_window ? SDL_CreateRenderer(m_window, nullptr) : nullptr;
175 if (m_window && !m_renderer) {
176 m_renderer = SDL_CreateRenderer(m_window, SDL_SOFTWARE_RENDERER);
177 }
178 if (!m_renderer) {
179 m_error = SDL_GetError();
180 }
181}

205int SdlHost::run(SurfaceClient& client, const std::function<void(std::int64_t)>& before_frame)
206{
207 if (!is_open()) {
208 return 0;
209 }
210
211 client.attach(this);
212 m_size = SurfaceSize{};
213 m_updateRequested = true;
214
215 int frames = 0;
216 while (m_options.frame_limit < 0 || frames < m_options.frame_limit) {
217 if (!dispatch(client)) {
218 break;
219 }
220 const std::int64_t now = now_ms();
221 if (before_frame) {
222 before_frame(now);
223 }
224
225 SurfaceSize size;
226 SDL_GetWindowSize(m_window, &size.width, &size.height);
227 SDL_GetWindowSizeInPixels(m_window, &size.pixel_width, &size.pixel_height);
228 size.device_pixel_ratio = size.width > 0 ? static_cast<double>(size.pixel_width) / size.width : 1.0;
229 if (size.width != m_size.width || size.height != m_size.height || size.pixel_width != m_size.pixel_width) {
230 m_size = size;
231 client.resize(m_size);
232 }
233
234 if (client.tick(now)) {
235 m_updateRequested = true;
236 }
237 // Without a limit an idle window waits for input instead of redrawing unchanged frames
238 if (!m_updateRequested && m_options.frame_limit < 0) {
239 SDL_WaitEventTimeout(nullptr, 16);
240 continue;
241 }
242 m_updateRequested = false;
243 present(client);
244 ++frames;
245 }
246
247 client.attach(nullptr);
248 return frames;
249}

281bool SdlHost::dispatch(SurfaceClient& client)
282{
283 SDL_Event event;
284 while (SDL_PollEvent(&event)) {
285 const std::int64_t now = now_ms();
286 switch (event.type) {
287 case SDL_EVENT_QUIT:
288 return false;
289 case SDL_EVENT_MOUSE_BUTTON_DOWN:
290 case SDL_EVENT_MOUSE_BUTTON_UP: {
291 PointerEvent pointer;
292 pointer.type = event.type == SDL_EVENT_MOUSE_BUTTON_UP ? PointerEvent::Type::Release
293 : event.button.clicks == 2 ? PointerEvent::Type::DoubleClick
294 : PointerEvent::Type::Press;
295 pointer.x = event.button.x;
296 pointer.y = event.button.y;
297 pointer.button = button_of(event.button.button);
298 pointer.buttons = buttons_of(SDL_GetMouseState(nullptr, nullptr));
299 pointer.modifiers = modifiers_of(SDL_GetModState());
300 pointer.timestamp_ms = now;
301 client.handle_pointer(pointer);
302 break;
303 }
304 case SDL_EVENT_MOUSE_MOTION: {
305 PointerEvent pointer;
306 pointer.x = event.motion.x;
307 pointer.y = event.motion.y;
308 pointer.buttons = buttons_of(event.motion.state);
309 pointer.modifiers = modifiers_of(SDL_GetModState());
310 pointer.timestamp_ms = now;
311 client.handle_pointer(pointer);
312 break;
313 }
314 case SDL_EVENT_MOUSE_WHEEL: {
315 WheelEvent wheel;
316 wheel.x = event.wheel.mouse_x;
317 wheel.y = event.wheel.mouse_y;
318 // SDL counts notches, mne_viz eighths of a degree (120 per notch)
319 wheel.angle_delta_x = -120.0 * event.wheel.x;
320 wheel.angle_delta_y = 120.0 * event.wheel.y;
321 wheel.modifiers = modifiers_of(SDL_GetModState());
322 wheel.timestamp_ms = now;
323 client.handle_wheel(wheel);
324 break;
325 }
326 case SDL_EVENT_KEY_DOWN: {
327 const KeyEvent key = key_of(event.key, now);
328 const bool used = (m_keyHandler && m_keyHandler(key)) || client.handle_key(key);
329 if (!used && key.key == Key::Escape) {
330 return false;
331 }
332 break;
333 }
334 case SDL_EVENT_WINDOW_EXPOSED:
335 m_updateRequested = true;
336 break;
337 default:
338 break;
339 }
340 }
341 return true;
342}

346void SdlHost::present(SurfaceClient& client)
347{
348 GpuFrameTarget target;
349 target.pixel_width = m_size.pixel_width;
350 target.pixel_height = m_size.pixel_height;
351 target.device_pixel_ratio = m_size.device_pixel_ratio;
352 const Image image = m_software.render(client.frame(target), client.resources());
353 if (image.width <= 0 || image.height <= 0) {
354 return;
355 }
356
357 if (!m_texture || m_textureWidth != image.width || m_textureHeight != image.height) {
358 SDL_DestroyTexture(m_texture);
359 m_texture = SDL_CreateTexture(m_renderer, SDL_PIXELFORMAT_RGBA32, SDL_TEXTUREACCESS_STREAMING, image.width, image.height);
360 m_textureWidth = image.width;
361 m_textureHeight = image.height;
362 }
363 SDL_UpdateTexture(m_texture, nullptr, image.rgba.data(), image.width * 4);
364 SDL_RenderClear(m_renderer);
365 SDL_RenderTexture(m_renderer, m_texture, nullptr, nullptr);
366 SDL_RenderPresent(m_renderer);
367}
