169void ScalingPanel::rebuild()
170{
171 const bool updating = std::exchange(m_updating, true);
172 const bool linked = form().get<bool>(kLink);
173 const double factor = form().get<double>(kFactor, 2.0);
174 std::vector<std::string> keys;
175 keys.reserve(form().fields().size());
176 for (const SettingField& field : form().fields()) {
177 keys.push_back(field.key);
178 }
179 for (const std::string& key : keys) {
180 form().remove(key);
181 }
182 for (const ScaleSpec& spec : kScales) {
183 if (!m_scales.contains(spec.id) || !shown(spec.type)) {
184 continue;
185 }
186 SettingField field;
187 field.key = spec.key;
188 field.kind = SettingField::Kind::Double;
189 field.label = spec.label;
190 field.unit = spec.unit;
191 field.default_value = spec.fallback;
192 field.minimum = spec.minimum;
193 field.maximum = spec.maximum;
194 field.step = spec.step;
195 field.decimals = spec.decimals;
196 field.persistent = false;
197 form().add(std::move(field));
198 }
199 if (form().field("scaleMAG") && form().field("scaleGRAD")) {
200 SettingField link;
201 link.key = kLink;
202 link.kind = SettingField::Kind::Bool;
203 link.label = "Link MAGs -> GRADs";
204 link.default_value = linked;
205 link.persistent = false;
206 form().add(std::move(link));
207 SettingField distance;
208 distance.key = kFactor;
209 distance.kind = SettingField::Kind::Double;
210 distance.label = "MAG-GRAD link";
211 distance.unit = " cm";
212 distance.default_value = factor;
213 distance.minimum = 0.1;
214 distance.maximum = 8.0;
215 distance.step = 0.1;
216 distance.decimals = 2;
217 distance.visible_if = kLink;
218 distance.persistent = false;
219 form().add(std::move(distance));
220 }
221 show_scales();
222 m_updating = updating;
223}

246void ScalingPanel::changed(const std::string& key)
247{
248 const bool linked = form().get<bool>(kLink);
249 const auto follow = [this](int id, double shown) {
250 const ScaleSpec& spec = spec_of_id(id);
251 const bool updating = std::exchange(m_updating, true);
252 form().set_value(spec.key, shown);
253 m_updating = updating;
254 m_scales[id] = static_cast<float>(form().get<double>(spec.key) * spec.factor);
255 };
256 if (const ScaleSpec* spec = spec_of_key(key)) {
257 m_scales[spec->id] = static_cast<float>(form().get<double>(key) * spec->factor);
258 if (linked && spec->id == FIFF_UNIT_T) {
259 follow(FIFF_UNIT_T_M, form().get<double>("scaleMAG") / (form().get<double>(kFactor) / 100.0));
260 } else if (linked && spec->id == FIFF_UNIT_T_M) {
261 follow(FIFF_UNIT_T, form().get<double>("scaleGRAD") * (form().get<double>(kFactor) / 100.0));
262 }
263 report();
264 } else if (key == kFactor) {
265 if (linked) {
266 follow(FIFF_UNIT_T_M, form().get<double>("scaleMAG") / (form().get<double>(kFactor) / 100.0));
267 report();
268 }
269 } else if (key == kLink && linked) {
270 set_mag_grad_link(true);
271 }
272}
