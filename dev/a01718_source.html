119template<AcqSampleFormat F>
120std::uint32_t loadWord(const std::byte* element, int k) noexcept
121{
122 const std::byte* p = element + static_cast<std::size_t>(k) * componentBytes<F>();
123 if constexpr (F == AcqSampleFormat::Int24Packed) {
124 return static_cast<std::uint32_t>(std::to_integer<std::uint8_t>(p[0])) |
125 (static_cast<std::uint32_t>(std::to_integer<std::uint8_t>(p[1])) << 8U) |
126 (static_cast<std::uint32_t>(std::to_integer<std::uint8_t>(p[2])) << 16U);
127 } else if constexpr (Native<F>::bits == 32) {
128 std::uint32_t w = 0;
129 std::memcpy(&w, p, sizeof(w));
130 return w;
131 } else {
132 std::uint16_t w = 0;
133 std::memcpy(&w, p, sizeof(w));
134 return w;
135 }
136}

138template<AcqSampleFormat F>
139void storeWord(std::byte* element, int k, std::uint32_t w) noexcept
140{
141 std::byte* p = element + static_cast<std::size_t>(k) * componentBytes<F>();
142 if constexpr (F == AcqSampleFormat::Int24Packed) {
143 p[0] = static_cast<std::byte>(w & 0xFFU);
144 p[1] = static_cast<std::byte>((w >> 8U) & 0xFFU);
145 p[2] = static_cast<std::byte>((w >> 16U) & 0xFFU);
146 } else if constexpr (Native<F>::bits == 32) {
147 std::memcpy(p, &w, sizeof(w));
148 } else {
149 const auto h = static_cast<std::uint16_t>(w);
150 std::memcpy(p, &h, sizeof(h));
151 }
152}

155template<AcqSampleFormat F>
156std::int64_t loadInteger(const std::byte* element, int k, int d) noexcept
157{
158 constexpr int kBits = Native<F>::bits;
159 const auto w = static_cast<std::uint64_t>(loadWord<F>(element, k));
160 const int analog = kBits - d;
161 const std::uint64_t low = w & ((std::uint64_t(1) << analog) - 1);
162 const std::uint64_t sign = std::uint64_t(1) << (analog - 1);
163 const auto value = static_cast<std::int64_t>(low ^ sign) - static_cast<std::int64_t>(sign);
164 return value * (std::int64_t(1) << d);
165}

167template<AcqSampleFormat F>
168double loadFloating(const std::byte* element, int k) noexcept
169{
170 if constexpr (F == AcqSampleFormat::Float64) {
171 double v = 0;
172 std::memcpy(&v, element, sizeof(v));
173 return v;
174 } else {
175 float v = 0;
176 std::memcpy(&v, element + static_cast<std::size_t>(k) * sizeof(float), sizeof(v));
177 return v;

182template<AcqSampleFormat F, typename Real>
183Real component(const std::byte* element, int k, int d, bool physical, Real scale, Real offset) noexcept
184{
185 Real v = 0;
186 if constexpr (Native<F>::integer) {
187 v = static_cast<Real>(loadInteger<F>(element, k, d));
188 } else {
189 v = static_cast<Real>(loadFloating<F>(element, k));
190 }
191 if (physical) {
192 v = v * scale;
193 if (k == 0) {
194 v = v + offset;
195 }
196 }
197 return v;
198}

204template<AcqSampleFormat F, typename T>
205void decodeRun(const AcqBlockPart& part, std::size_t firstChannel, std::span<const double> scales,
206 std::span<const double> offsets, std::span<const std::uint8_t> digital, const AcqMatrixRef<T>& out,
207 std::size_t firstColumn, AcqScaling scaling) noexcept
208{
209 using Real = std::conditional_t<IsComplex<T>::value, float, T>;
210 const bool physical = scaling == AcqScaling::Physical;
211 for (std::size_t s0 = 0; s0 < part.samples; s0 += kBlock) {
212 const std::size_t s1 = std::min(part.samples, s0 + kBlock);
213 for (std::size_t r = 0; r < out.rows; ++r) {
214 const std::size_t c = firstChannel + r;
215 const Real scale = physical ? static_cast<Real>(scales[c]) : Real(1);
216 const Real offset = physical ? static_cast<Real>(offsets[c]) : Real(0);
217 const int d = digital.empty() ? 0 : digital[c];
218 T* row = out.data + static_cast<std::ptrdiff_t>(r) * out.rowStride;
219 for (std::size_t s = s0; s < s1; ++s) {
220 const std::byte* element = part.at(c, s);
221 T& dst = row[static_cast<std::ptrdiff_t>(firstColumn + s) * out.colStride];
222 if constexpr (IsComplex<T>::value) {
223 const Real re = component<F, Real>(element, 0, d, physical, scale, offset);
224 Real im = 0;
225 if constexpr (Native<F>::complex) {
226 im = component<F, Real>(element, 1, 0, physical, scale, offset);
227 }
228 dst = T(re, im);
229 } else {
230 dst = component<F, Real>(element, 0, d, physical, scale, offset);
231 }
232 }
233 }
234 }
235}

283// The general kernels above handle every case one element at a time. The kernels below cover the common case
284// with the format and the scaling as template parameters, typed loads and stores, and loops whose innermost
285// dimension is contiguous in both memories wherever the layouts allow it, so the compiler vectorises them.
286// The arithmetic is the same expression as above, so the results are bit-identical.
287
289constexpr std::size_t kChannelChunk = 256;
290
292template<AcqSampleFormat F>
293auto loadReal(const std::byte* p) noexcept
294{
295 if constexpr (F == AcqSampleFormat::Int16) {
296 std::int16_t v = 0;
297 std::memcpy(&v, p, sizeof(v));
298 return static_cast<std::int32_t>(v);
299 } else if constexpr (F == AcqSampleFormat::Int24Packed) {
300 const auto u = static_cast<std::uint32_t>(std::to_integer<std::uint8_t>(p[0])) |
301 (static_cast<std::uint32_t>(std::to_integer<std::uint8_t>(p[1])) << 8U) |
302 (static_cast<std::uint32_t>(std::to_integer<std::uint8_t>(p[2])) << 16U);
303 return static_cast<std::int32_t>(u << 8U) >> 8;
304 } else if constexpr (F == AcqSampleFormat::Int32) {
305 std::int32_t v = 0;
306 std::memcpy(&v, p, sizeof(v));
307 return v;
308 } else if constexpr (F == AcqSampleFormat::Float32) {
309 float v = 0;
310 std::memcpy(&v, p, sizeof(v));
311 return v;
312 } else {
313 double v = 0;
314 std::memcpy(&v, p, sizeof(v));
315 return v;
316 }
317}

347template<AcqSampleFormat F, typename Real, bool Physical>
348void decodeFastRun(const AcqBlockPart& part, std::size_t firstChannel, std::span<const double> scales,
349 std::span<const double> offsets, const AcqMatrixRef<Real>& out, std::size_t firstColumn) noexcept
350{
351 constexpr auto kBytes = static_cast<std::ptrdiff_t>(realBytes<F>());
352 const auto factor = [&](std::span<const double> v, std::size_t c, double fallback) {
353 return static_cast<Real>(Physical ? v[c] : fallback);
354 };
355 if (part.sampleStride == kBytes && out.colStride == 1) {
356 // Samples contiguous in the ring (planar) and in the matrix (row-major): one stream per channel.
357 for (std::size_t r = 0; r < out.rows; ++r) {
358 const std::byte* src = part.at(firstChannel + r, 0);
359 Real* dst = out.data + static_cast<std::ptrdiff_t>(r) * out.rowStride + static_cast<std::ptrdiff_t>(firstColumn);
360 const Real scale = factor(scales, firstChannel + r, 1.0);
361 const Real offset = factor(offsets, firstChannel + r, 0.0);
362 for (std::size_t s = 0; s < part.samples; ++s) {
363 dst[s] = decodeValue<F, Real, Physical>(src + static_cast<std::ptrdiff_t>(s) * kBytes, scale, offset);
364 }
365 }
366 return;
367 }
368 if (part.channelStride == kBytes && out.rowStride == 1) {
369 // Channels contiguous in the ring (interleaved) and in the matrix (column-major): one frame per column.
370 std::array<Real, kChannelChunk> scale{};
371 std::array<Real, kChannelChunk> offset{};
372 for (std::size_t c0 = 0; c0 < out.rows; c0 += kChannelChunk) {
373 const std::size_t n = std::min(kChannelChunk, out.rows - c0);
374 for (std::size_t i = 0; i < n; ++i) {
375 scale[i] = factor(scales, firstChannel + c0 + i, 1.0);
376 offset[i] = factor(offsets, firstChannel + c0 + i, 0.0);
377 }
378 for (std::size_t s = 0; s < part.samples; ++s) {
379 const std::byte* src = part.at(firstChannel + c0, s);
380 Real* dst = out.data + static_cast<std::ptrdiff_t>(firstColumn + s) * out.colStride +
381 static_cast<std::ptrdiff_t>(c0);
382 for (std::size_t i = 0; i < n; ++i) {
383 dst[i] = decodeValue<F, Real, Physical>(src + static_cast<std::ptrdiff_t>(i) * kBytes, scale[i],
384 offset[i]);
385 }
386 }
387 }
388 return;
389 }
390 if (part.channelStride == kBytes && out.colStride == 1) {
391 // Interleaved ring into a row-major matrix: decode a tile of frames along the contiguous channels, then
392 // write it out row by row, so both memories are read and written in order.
393 constexpr std::size_t kTileSamples = 64;
394 constexpr std::size_t kTileChannels = 32;
395 std::array<Real, kTileSamples * kTileChannels> tile{};
396 for (std::size_t c0 = 0; c0 < out.rows; c0 += kTileChannels) {
397 const std::size_t n = std::min(kTileChannels, out.rows - c0);
398 std::array<Real, kTileChannels> scale{};
399 std::array<Real, kTileChannels> offset{};
400 for (std::size_t i = 0; i < n; ++i) {
401 scale[i] = factor(scales, firstChannel + c0 + i, 1.0);
402 offset[i] = factor(offsets, firstChannel + c0 + i, 0.0);
403 }
404 for (std::size_t s0 = 0; s0 < part.samples; s0 += kTileSamples) {
405 const std::size_t m = std::min(kTileSamples, part.samples - s0);
406 for (std::size_t s = 0; s < m; ++s) {
407 const std::byte* src = part.at(firstChannel + c0, s0 + s);
408 Real* row = tile.data() + s * kTileChannels;
409 for (std::size_t i = 0; i < n; ++i) {
410 row[i] = decodeValue<F, Real, Physical>(src + static_cast<std::ptrdiff_t>(i) * kBytes, scale[i],
411 offset[i]);
412 }
413 }
414 for (std::size_t i = 0; i < n; ++i) {
415 Real* dst = out.data + static_cast<std::ptrdiff_t>(c0 + i) * out.rowStride +
416 static_cast<std::ptrdiff_t>(firstColumn + s0);
417 for (std::size_t s = 0; s < m; ++s) {
418 dst[s] = tile[s * kTileChannels + i];
419 }
420 }
421 }
422 }
423 return;
424 }
425 // Other strides: blocks of kBlock samples, so a block's source and destination stay in cache.
426 for (std::size_t s0 = 0; s0 < part.samples; s0 += kBlock) {
427 const std::size_t s1 = std::min(part.samples, s0 + kBlock);
428 for (std::size_t r = 0; r < out.rows; ++r) {
429 const Real scale = factor(scales, firstChannel + r, 1.0);
430 const Real offset = factor(offsets, firstChannel + r, 0.0);
431 const std::byte* src = part.at(firstChannel + r, 0);
432 Real* dst = out.data + static_cast<std::ptrdiff_t>(r) * out.rowStride;
433 for (std::size_t s = s0; s < s1; ++s) {
434 dst[static_cast<std::ptrdiff_t>(firstColumn + s) * out.colStride] =
435 decodeValue<F, Real, Physical>(src + static_cast<std::ptrdiff_t>(s) * part.sampleStride, scale, offset);
436 }
437 }
438 }
439}

446template<AcqSampleFormat F, bool Physical>
447void decodeComplexFastRun(const AcqBlockPart& part, std::size_t firstChannel, std::span<const double> scales,
448 std::span<const double> offsets, const AcqMatrixRef<std::complex<float>>& out,
449 std::size_t firstColumn) noexcept
450{
451 using Component = std::conditional_t<F == AcqSampleFormat::ComplexInt16, std::int16_t, float>;
452 constexpr auto kBytes = static_cast<std::ptrdiff_t>(2 * sizeof(Component));
453 const auto element = [](const std::byte* p, [[maybe_unused]] float scale, [[maybe_unused]] float offset) {
454 std::array<Component, 2> iq{};
455 std::memcpy(iq.data(), p, sizeof(iq));
456 auto re = static_cast<float>(iq[0]);
457 auto im = static_cast<float>(iq[1]);
458 if constexpr (Physical) {
459 re = re * scale;
460 re = re + offset;
461 im = im * scale;
462 }
463 return std::complex<float>(re, im);
464 };
465 const auto factor = [&](std::span<const double> v, std::size_t c, double fallback) {
466 return static_cast<float>(Physical ? v[c] : fallback);
467 };
468 if (part.channelStride == kBytes && out.rowStride == 1) {
469 std::array<float, kChannelChunk> scale{};
470 std::array<float, kChannelChunk> offset{};
471 for (std::size_t c0 = 0; c0 < out.rows; c0 += kChannelChunk) {
472 const std::size_t n = std::min(kChannelChunk, out.rows - c0);
473 for (std::size_t i = 0; i < n; ++i) {
474 scale[i] = factor(scales, firstChannel + c0 + i, 1.0);
475 offset[i] = factor(offsets, firstChannel + c0 + i, 0.0);
476 }
477 for (std::size_t s = 0; s < part.samples; ++s) {
478 const std::byte* src = part.at(firstChannel + c0, s);
479 std::complex<float>* dst =
480 out.data + static_cast<std::ptrdiff_t>(firstColumn + s) * out.colStride + static_cast<std::ptrdiff_t>(c0);
481 for (std::size_t i = 0; i < n; ++i) {
482 dst[i] = element(src + static_cast<std::ptrdiff_t>(i) * kBytes, scale[i], offset[i]);
483 }
484 }
485 }
486 return;
487 }
488 for (std::size_t r = 0; r < out.rows; ++r) {
489 const float scale = factor(scales, firstChannel + r, 1.0);
490 const float offset = factor(offsets, firstChannel + r, 0.0);
491 const std::byte* src = part.at(firstChannel + r, 0);
492 std::complex<float>* dst = out.data + static_cast<std::ptrdiff_t>(r) * out.rowStride;
493 for (std::size_t s = 0; s < part.samples; ++s) {
494 dst[static_cast<std::ptrdiff_t>(firstColumn + s) * out.colStride] =
495 element(src + static_cast<std::ptrdiff_t>(s) * part.sampleStride, scale, offset);
496 }
497 }
498}

525template<AcqSampleFormat F, bool Physical>
526std::uint64_t encodeValue(double x, std::byte* p, double scale, double offset, double lo, double hi) noexcept
527{
528 const bool nan = std::isnan(x);
529 double v = nan ? 0.0 : x;
530 std::uint64_t clip = nan ? 1U : 0U;
531 if constexpr (Physical) {
532 const double c = std::fmin(std::fmax(v, lo), hi);
533 clip += !nan && c != v ? 1U : 0U;
534 v = nan ? 0.0 : (c - offset) / scale;
535 }
536 if constexpr (F == AcqSampleFormat::Float64) {
537 const bool inf = std::isinf(v);
538 clip += inf ? 1U : 0U;
539 v = inf ? std::copysign(std::numeric_limits<double>::max(), v) : v;
540 std::memcpy(p, &v, sizeof(v));
541 } else if constexpr (F == AcqSampleFormat::Float32) {
542 constexpr double kMax = std::numeric_limits<float>::max();
543 const double c = std::min(std::max(v, -kMax), kMax);
544 clip += c != v ? 1U : 0U;
545 const auto f = static_cast<float>(c);
546 std::memcpy(p, &f, sizeof(f));
547 } else {
548 constexpr double kTop = F == AcqSampleFormat::Int16 ? 32768.0 : (F == AcqSampleFormat::Int24Packed ? 8388608.0 : 2147483648.0);
549 // Round to nearest, ties to even: exact for |v| < 2^51; anything larger saturates anyway.
550 constexpr double kRound = 0x1.8p52;
551 const double bounded = std::fmin(std::fmax(v, -0x1p40), 0x1p40);
552 const double rounded = (bounded + kRound) - kRound;
553 const double q = std::fmin(std::fmax(rounded, -kTop), kTop - 1.0);
554 clip += q != rounded ? 1U : 0U;
555 const auto i = static_cast<std::int32_t>(q);
556 if constexpr (F == AcqSampleFormat::Int16) {
557 const auto h = static_cast<std::int16_t>(i);
558 std::memcpy(p, &h, sizeof(h));
559 } else if constexpr (F == AcqSampleFormat::Int24Packed) {
560 const auto u = static_cast<std::uint32_t>(i);
561 p[0] = static_cast<std::byte>(u & 0xFFU);
562 p[1] = static_cast<std::byte>((u >> 8U) & 0xFFU);
563 p[2] = static_cast<std::byte>((u >> 16U) & 0xFFU);
564 } else {
565 std::memcpy(p, &i, sizeof(i));
566 }
567 }
568 return clip;
569}

571template<AcqSampleFormat F, typename T, bool Physical>
572void encodeFastRun(const AcqMatrixRef<const T>& in, std::size_t firstColumn, const AcqWritePart& part,
573 std::size_t firstChannel, std::span<const double> scales, std::span<const double> offsets,
574 std::span<const double> rangeMin, std::span<const double> rangeMax, std::uint64_t& clipped) noexcept
575{
576 constexpr auto kBytes = static_cast<std::ptrdiff_t>(realBytes<F>());
577 constexpr double kInf = std::numeric_limits<double>::infinity();
578 struct Limits
579 {
580 double lo;
581 double hi;
582 };
583 // A plain struct, not std::pair, whose by-value passing GCC flags as changed in its ABI (-Wpsabi note).
584 const auto limits = [&](std::size_t c) {
585 return Limits{std::isnan(rangeMin[c]) ? -kInf : rangeMin[c], std::isnan(rangeMax[c]) ? kInf : rangeMax[c]};
586 };
587 std::uint64_t clip = 0;
588 if (part.sampleStride == kBytes && in.colStride == 1) {
589 for (std::size_t r = 0; r < in.rows; ++r) {
590 const std::size_t c = firstChannel + r;
591 const auto [lo, hi] = limits(c);
592 const double scale = scales[c];
593 const double offset = offsets[c];
594 const T* src = in.data + static_cast<std::ptrdiff_t>(r) * in.rowStride + static_cast<std::ptrdiff_t>(firstColumn);
595 std::byte* dst = part.at(c, 0);
596 for (std::size_t s = 0; s < part.samples; ++s) {
597 clip += encodeValue<F, Physical>(static_cast<double>(src[s]), dst + static_cast<std::ptrdiff_t>(s) * kBytes,
598 scale, offset, lo, hi);
599 }
600 }
601 } else if (part.channelStride == kBytes && in.rowStride == 1) {
602 // Per-channel factors on the stack: the byte stores below could alias the spans' memory.
603 std::array<double, kChannelChunk> lo{};
604 std::array<double, kChannelChunk> hi{};
605 std::array<double, kChannelChunk> scale{};
606 std::array<double, kChannelChunk> offset{};
607 for (std::size_t c0 = 0; c0 < in.rows; c0 += kChannelChunk) {
608 const std::size_t n = std::min(kChannelChunk, in.rows - c0);
609 for (std::size_t i = 0; i < n; ++i) {
610 const Limits l = limits(firstChannel + c0 + i);
611 lo[i] = l.lo;
612 hi[i] = l.hi;
613 scale[i] = scales[firstChannel + c0 + i];
614 offset[i] = offsets[firstChannel + c0 + i];
615 }
616 for (std::size_t s = 0; s < part.samples; ++s) {
617 const T* src = in.data + static_cast<std::ptrdiff_t>(firstColumn + s) * in.colStride + static_cast<std::ptrdiff_t>(c0);
618 std::byte* dst = part.at(firstChannel + c0, s);
619 for (std::size_t i = 0; i < n; ++i) {
620 clip += encodeValue<F, Physical>(static_cast<double>(src[i]), dst + static_cast<std::ptrdiff_t>(i) * kBytes,
621 scale[i], offset[i], lo[i], hi[i]);
622 }
623 }
624 }
625 } else {
626 for (std::size_t s0 = 0; s0 < part.samples; s0 += kBlock) {
627 const std::size_t s1 = std::min(part.samples, s0 + kBlock);
628 for (std::size_t r = 0; r < in.rows; ++r) {
629 const std::size_t c = firstChannel + r;
630 const auto [lo, hi] = limits(c);
631 const double scale = scales[c];
632 const double offset = offsets[c];
633 const T* src = in.data + static_cast<std::ptrdiff_t>(r) * in.rowStride;
634 for (std::size_t s = s0; s < s1; ++s) {
635 clip += encodeValue<F, Physical>(static_cast<double>(src[static_cast<std::ptrdiff_t>(firstColumn + s) * in.colStride]),
636 part.at(c, s), scale, offset, lo, hi);
637 }
638 }
639 }
640 }
641 clipped += clip;
642}

668template<typename T>
669AcqError decodeView(const AcqBlockView& view, const AcqMatrixRef<T>& out, const AcqConvertOptions& options) noexcept
670{
671 if (!view.memory().hostAccessible()) {
672 return AcqErrorCode::DeviceMemory;
673 }
674 std::size_t first = 0;
675 std::size_t count = 0;
676 if (!resolveRange(options.channels, view.channelCount(), first, count) || out.data == nullptr ||
677 out.rows != count || out.cols < options.firstColumn + view.sampleCount()) {
678 return AcqErrorCode::ShapeMismatch;
679 }
680 const DecodeFn<T> fn = decoderFor<T>(view.format());
681 if (fn == nullptr) {
682 return AcqErrorCode::UnsupportedFormat;
683 }
684 const AcqMatrixRef<T> rows{out.data, count, out.cols, out.rowStride, out.colStride};
685 std::size_t column = options.firstColumn;
686 DecodeFastFn<T> fast = nullptr;
687 const bool physical = options.scaling == AcqScaling::Physical;
688 if (!anyDigitalBits(view.digitalBits(), first, count)) {
689 if constexpr (IsComplex<T>::value) {
690 if (view.format() == AcqSampleFormat::ComplexInt16) {
691 fast = physical ? &decodeComplexFastRun<AcqSampleFormat::ComplexInt16, true>
692 : &decodeComplexFastRun<AcqSampleFormat::ComplexInt16, false>;
693 } else if (view.format() == AcqSampleFormat::ComplexFloat32) {
694 fast = physical ? &decodeComplexFastRun<AcqSampleFormat::ComplexFloat32, true>
695 : &decodeComplexFastRun<AcqSampleFormat::ComplexFloat32, false>;
696 }
697 } else {
698 fast = physical ? fastDecoderFor<T, true>(view.format()) : fastDecoderFor<T, false>(view.format());
699 }
700 }
701 if (fast != nullptr) {
702 for (std::size_t p = 0; p < view.partCount(); ++p) {
703 fast(view.part(p), first, view.scales(), view.offsets(), rows, column);
704 column += view.part(p).samples;
705 }
706 return {};
707 }
708 for (std::size_t p = 0; p < view.partCount(); ++p) {
709 fn(view.part(p), first, view.scales(), view.offsets(), view.digitalBits(), rows, column, options.scaling);
710 column += view.part(p).samples;
711 }
712 return {};
713}

720template<AcqSampleFormat F>
721void encodeComponent(double value, std::byte* element, int k, int d, bool physical, double scale, double offset,
722 double lo, double hi, std::uint64_t& clipped) noexcept
723{
724 if (std::isnan(value)) {
725 value = 0.0;
726 physical = false;
727 ++clipped;
728 }
729 if (physical) {
730 if (value < lo || value > hi) {
731 value = value < lo ? lo : hi;
732 ++clipped;
733 }
734 value = (value - (k == 0 ? offset : 0.0)) / scale;
735 }
736 if constexpr (F == AcqSampleFormat::Float64) {
737 if (std::isinf(value)) {
738 value = std::copysign(std::numeric_limits<double>::max(), value);
739 ++clipped;
740 }
741 std::memcpy(element, &value, sizeof(value));
742 } else if constexpr (!Native<F>::integer) {
743 constexpr double kMax = std::numeric_limits<float>::max();
744 if (value > kMax || value < -kMax) {
745 value = std::copysign(kMax, value);
746 ++clipped;
747 }
748 const auto v = static_cast<float>(value);
749 std::memcpy(element + static_cast<std::size_t>(k) * sizeof(float), &v, sizeof(v));
750 } else {
751 constexpr int kBits = Native<F>::bits;
752 const int analog = kBits - d;
753 const double top = std::ldexp(1.0, analog - 1);
754 // Default rounding mode: nearest, ties to even.
755 double q = std::nearbyint(std::ldexp(value, -d));
756 if (q < -top || q > top - 1.0) {
757 q = q < -top ? -top : top - 1.0;
758 ++clipped;
759 }
760 const std::uint64_t lowMask = (std::uint64_t(1) << analog) - 1;
761 const auto low = static_cast<std::uint64_t>(static_cast<std::int64_t>(q)) & lowMask;
762 const auto old = static_cast<std::uint64_t>(loadWord<F>(element, k));
763 storeWord<F>(element, k, static_cast<std::uint32_t>((old & ~lowMask) | low));
764 }
765}

767template<AcqSampleFormat F, typename T>
768void encodeRun(const AcqMatrixRef<const T>& in, std::size_t firstColumn, const AcqWritePart& part,
769 std::size_t firstChannel, std::span<const double> scales, std::span<const double> offsets,
770 std::span<const double> rangeMin, std::span<const double> rangeMax, std::span<const std::uint8_t> digital,
771 AcqScaling scaling, std::uint64_t& clipped) noexcept
772{
773 const bool physical = scaling == AcqScaling::Physical;
774 for (std::size_t s0 = 0; s0 < part.samples; s0 += kBlock) {
775 const std::size_t s1 = std::min(part.samples, s0 + kBlock);
776 for (std::size_t r = 0; r < in.rows; ++r) {
777 const std::size_t c = firstChannel + r;
778 const double scale = physical ? scales[c] : 1.0;
779 const double offset = physical ? offsets[c] : 0.0;
780 const double lo = std::isnan(rangeMin[c]) ? -std::numeric_limits<double>::infinity() : rangeMin[c];
781 const double hi = std::isnan(rangeMax[c]) ? std::numeric_limits<double>::infinity() : rangeMax[c];
782 const int d = digital.empty() ? 0 : digital[c];
783 const T* row = in.data + static_cast<std::ptrdiff_t>(r) * in.rowStride;
784 for (std::size_t s = s0; s < s1; ++s) {
785 std::byte* element = part.at(c, s);
786 const T& value = row[static_cast<std::ptrdiff_t>(firstColumn + s) * in.colStride];
787 if constexpr (IsComplex<T>::value) {
788 encodeComponent<F>(value.real(), element, 0, d, physical, scale, offset, lo, hi, clipped);
789 if constexpr (Native<F>::complex) {
790 encodeComponent<F>(value.imag(), element, 1, 0, physical, scale, offset, lo, hi, clipped);
791 } else if (value.imag() != 0.0F) {
792 ++clipped;
793 }
794 } else {
795 encodeComponent<F>(static_cast<double>(value), element, 0, d, physical, scale, offset, lo, hi,
796 clipped);
797 if constexpr (Native<F>::complex) {
798 std::uint64_t ignored = 0;
799 encodeComponent<F>(0.0, element, 1, 0, physical, scale, offset, lo, hi, ignored);
800 }
801 }
802 }
803 }
804 }

897template<typename View>
898AcqError checkFields(const View& view, std::span<const AcqBitField> fields) noexcept
899{
900 const int bits = integerWordBits(view.format());
901 if (bits == 0) {
902 return AcqErrorCode::UnsupportedFormat;
903 }
904 for (const AcqBitField& field : fields) {
905 if (field.channel >= view.channelCount() || field.width == 0) {
906 return AcqErrorCode::InvalidParameter;
907 }
908 const int digital = view.digitalBits()[field.channel];
909 if (field.firstBit + field.width > bits || field.firstBit < bits - digital) {
910 return AcqErrorCode::InvalidParameter;
911 }
912 }
913 return {};
914}

1024AcqExpected<AcqEncodeResult> mne::encodeBits(const AcqMatrixRef<const std::uint32_t>& in,
1025 std::span<const AcqBitField> fields, const AcqWriteView& view,
1026 std::size_t firstColumn) noexcept
1027{
1028 if (!view.memory().hostAccessible()) {
1029 return AcqErrorCode::DeviceMemory;
1030 }
1031 if (AcqError error = checkFields(view, fields); !error.ok()) {

1052 }

1054 }
