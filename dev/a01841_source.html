83{
84 m_pProjs = projs;
85
86 for (int i = 0; i < std::ssize(m_pProjs); ++i) {
87 if (!m_mapProjActive.contains(m_pProjs.at(i).desc)) {

99void ProjectorsView::redrawGUI()
100{
101 if (m_pProjs.empty()) {
102 return;
103 }
104
105 m_qListProjCheckBox.clear();
106 detail::resetLayout(this);
107
108 // Projection Selection
109 QGridLayout* topLayout = new QGridLayout(this);
110
111 bool bAllActivated = true;
112
113 qint32 i = 0;
114
115 for (; i < std::ssize(m_pProjs); ++i) {
116 QCheckBox* checkBox = new QCheckBox(QString::fromStdString(m_pProjs.at(i).desc), this);
117
118 if (m_pProjs.at(i).active == false) {
119 bAllActivated = false;
120 }
121
122 m_qListProjCheckBox.append(checkBox);
123
124 connect(checkBox, &QCheckBox::toggled,
125 this, &ProjectorsView::onCheckProjStatusChanged);
126
127 checkBox->setChecked(m_pProjs.at(i).active);
128
129 topLayout->addWidget(checkBox, i, 0);
130 }
131
132 QFrame* line = new QFrame(this);
133 line->setFrameShape(QFrame::HLine);
134 line->setFrameShadow(QFrame::Sunken);
135
136 topLayout->addWidget(line, i + 1, 0);
137
138 m_pEnableDisableProjectors = new QCheckBox("Enable all", this);
139 m_pEnableDisableProjectors->setChecked(bAllActivated);
140 topLayout->addWidget(m_pEnableDisableProjectors, i + 2, 0);
141 connect(m_pEnableDisableProjectors, static_cast<void (QCheckBox::*)(bool)>(&QCheckBox::clicked),
142 this, &ProjectorsView::onEnableDisableAllProj);
143
144 onCheckProjStatusChanged();
145}

149void ProjectorsView::saveSettings()
150{
151 if (m_sSettingsPath.isEmpty()) {
152 return;
153 }
154
155 QSettings settings("MNECPP");
156
157 settings.beginGroup(m_sSettingsPath + QString("/ProjectorsView/projectorsActive"));
158
159 std::map<std::string, bool>::const_iterator iProj = m_mapProjActive.cbegin();
160 while (iProj != m_mapProjActive.cend()) {
161 settings.setValue(QString::fromStdString(iProj->first), iProj->second);
162 ++iProj;

164

212void ProjectorsView::onEnableDisableAllProj(bool status)
213{
214 //Set all checkboxes to status
215 for (int i = 0; i < m_qListProjCheckBox.size(); i++) {
216 m_qListProjCheckBox.at(i)->setChecked(status);
217 }
218
219 //Set all projection activation states to status
220 for (int i = 0; i < std::ssize(m_pProjs); ++i) {
221 m_pProjs[i].active = status;
222 m_mapProjActive[m_pProjs.at(i).desc] = status;
223 }
224
225 if (m_pEnableDisableProjectors) {
226 m_pEnableDisableProjectors->setChecked(status);
227 }
228
229 emit projSelectionChanged(m_pProjs);
230
231 saveSettings();
232}

236void ProjectorsView::onCheckProjStatusChanged()
237{

239
240 for (qint32 i = 0; i < m_qListProjCheckBox.size(); ++i) {
241 if (m_qListProjCheckBox.at(i)->isChecked() == false) {
242 bAllActivated = false;

244
245 m_pProjs[i].active = m_qListProjCheckBox.at(i)->isChecked();
246 m_mapProjActive[m_pProjs.at(i).desc] = m_qListProjCheckBox.at(i)->isChecked();
247 }
248
249 if (m_pEnableDisableProjectors) {
250 m_pEnableDisableProjectors->setChecked(bAllActivated);
251 }
252
253 emit projSelectionChanged(m_pProjs);
