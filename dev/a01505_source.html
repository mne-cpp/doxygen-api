91Sha256& Sha256::update(std::span<const std::byte> data)
92{
93 m_length += data.size();
94 if (m_blockSize > 0) {
95 const std::size_t take = std::min(data.size(), m_block.size() - m_blockSize);
96 std::ranges::copy(data.first(take), m_block.begin() + static_cast<std::ptrdiff_t>(m_blockSize));
97 m_blockSize += take;
98 data = data.subspan(take);
99 if (m_blockSize < m_block.size())
100 return *this;
101 compress(m_block.data());
102 m_blockSize = 0;
103 }
104 for (; data.size() >= m_block.size(); data = data.subspan(m_block.size()))
105 compress(data.data());
106 std::ranges::copy(data, m_block.begin());
107 m_blockSize = data.size();
108 return *this;
109}

121{
122 Sha256 last = *this;
123 const std::uint64_t bits = m_length * 8;
124 const std::size_t padding = (m_blockSize < 56 ? 56 : 120) - m_blockSize;
125 std::array<std::byte, 72> tail{};
126 tail[0] = std::byte{0x80};
127 for (std::size_t i = 0; i < 8; ++i)
128 tail[padding + i] = static_cast<std::byte>(bits >> (56 - (8 * i)));
129 last.update(std::span(tail).first(padding + 8));

131 Digest out{};
132 for (std::size_t i = 0; i < 8; ++i) {
133 for (std::size_t k = 0; k < 4; ++k)
