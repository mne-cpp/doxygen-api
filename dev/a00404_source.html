72bool MNELogger::setLogFile(const QString& sFilePath)
73{
74 std::lock_guard<std::mutex> lock(m_mutex);
75
76 if(sFilePath.isEmpty()) {
77 m_sLogFilePath.clear();
78 return true;
79 }
80
81 //
82 // Make sure the destination is writable now rather than silently dropping
83 // every message later on.
84 //
85 QFile file(sFilePath);
86 if(!file.open(QIODevice::WriteOnly | QIODevice::Append | QIODevice::Text)) {
87 return false;
88 }
89 file.close();
90
91 m_sLogFilePath = sFilePath;
92
93 return true;
94}

105void MNELogger::customLogWriter(QtMsgType type,
106 const QMessageLogContext &context,
107 const QString &msg)
108{
109 Q_UNUSED(context)
110
111 // Map Qt message type to numeric level (must match CMake MNE_LOG_LEVEL values)
112 // 0=DEBUG, 1=INFO, 2=WARN, 3=CRIT, 4=FATAL
113 int msgLevel = 0;
114 switch(type) {
115 case QtDebugMsg: msgLevel = 0; break;
116 case QtInfoMsg: msgLevel = 1; break;
117 case QtWarningMsg: msgLevel = 2; break;
118 case QtCriticalMsg: msgLevel = 3; break;
119 case QtFatalMsg: msgLevel = 4; break;
120 default: msgLevel = 0; break;
121 }
122
123#ifdef MNE_LOG_LEVEL
124 if (msgLevel < MNE_LOG_LEVEL)
125 return;
126#endif
127
128 std::string text = msg.toStdString();
129
130 std::lock_guard<std::mutex> lock(m_mutex);
131
132 //
133 // Mirror the message into the log file when one was configured. This is
134 // kept free of the colour escapes so the file stays readable, and carries
135 // a timestamp because a log that outlives the session needs one.
136 //
137 if(!m_sLogFilePath.isEmpty()) {
138 QFile logFile(m_sLogFilePath);
139 if(logFile.open(QIODevice::WriteOnly | QIODevice::Append | QIODevice::Text)) {
140 const char* fileLevel = "";
141 switch(type) {
142 case QtDebugMsg: fileLevel = "DEBUG"; break;
143 case QtInfoMsg: fileLevel = "INFO"; break;
144 case QtWarningMsg: fileLevel = "WARN"; break;
145 case QtCriticalMsg: fileLevel = "CRIT"; break;
146 case QtFatalMsg: fileLevel = "FATAL"; break;
147 default: fileLevel = "LOG"; break;
148 }
149
150 QTextStream stream(&logFile);
151 stream << QDateTime::currentDateTime().toString("yyyy-MM-dd hh:mm:ss.zzz")
152 << " [" << fileLevel << "] "
153 << QString::fromStdString(text).trimmed() << '\n';
154 logFile.close();
155 }
156 }
157
158#ifdef MNE_LOG_MODE_FORMATTED
159 static bool s_startOfLine = true;
160
161 // Determine color and level tag
162 const char* color = COLOR_RESET;
163 const char* level = "";
164 switch(type) {
165 case QtDebugMsg: color = COLOR_DEBUG; level = "[DEBUG] "; break;
166 case QtInfoMsg: color = COLOR_INFO; level = "[INFO] "; break;
167 case QtWarningMsg: color = COLOR_WARN; level = "[WARN] "; break;
168 case QtCriticalMsg: color = COLOR_CRITICAL; level = "[CRIT] "; break;
169 case QtFatalMsg: color = COLOR_FATAL; level = "[FATAL] "; break;
170 default: level = ""; break;
171 }
172
173 // Uncomment the following line to include date/time in log prefix:
174 // std::string sDateTime = QDateTime::currentDateTime().toString("dd/MM/yyyy hh:mm:ss.z ").toStdString();
175 std::string sDateTime;
176
177 std::string prefix = std::string(color) + sDateTime + level + COLOR_RESET;
178
179 // Walk through the message character by character.
180 // Print the prefix only at the start of each new line.
181 for (size_t i = 0; i < text.size(); ++i) {
182 if (s_startOfLine && text[i] != '\n') {
183 std::cout << prefix;
184 s_startOfLine = false;
185 }
186 std::cout << text[i];
187 if (text[i] == '\n') {
188 s_startOfLine = true;
189 }
190 }
191
192 if (text.empty() || text.back() != '\n') {
193 std::cout << '\n';
194 s_startOfLine = true;
195 }
196#else
197 // PLAIN mode: raw text, like printf
198 std::cout << text;
199 if (text.empty() || text.back() != '\n') {
200 std::cout << '\n';
201 }
202#endif
203
204 std::cout.flush();
205}
