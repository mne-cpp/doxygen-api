67bool ComputeFwdSettings::checkIntegrity() const
68{
69 if (srcname.empty()) {
70 mne::Log::critical("Source space name is missing. Use the --src option to specify it.");
71 return false;
72 }
73 if (!mri_head_ident) {
74 if (mriname.empty() && transname.empty()) {
75 mne::Log::critical("MRI <-> head coordinate transformation is missing. Use the --mri or --trans option to specify it.");
76 return false;
77 }
78 }
79 if (measname.empty()) {
80 mne::Log::critical("Source of coil and electrode locations is missing. Use the --meas option to specify it.");
81 return false;
82 }
83 if (solname.empty()) {
84 mne::Log::critical("Solution name is missing. Use the --fwd option to specify it.");
85 return false;
86 }
87 if (!(include_meg || include_eeg)) {
88 mne::Log::critical("Employ the --meg and --eeg options to select MEG and/or EEG");
89 return false;
90 }
91 return true;
92}
