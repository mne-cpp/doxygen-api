136 void enqueue(const std::vector<std::vector<float>>& samples)
137 {
138 if (samples.empty()) {
139 return;
140 }
141 const double rate = m_info.nominal_srate();
142 const double now = local_clock();
143 const double first = rate > 0.0 ? now - static_cast<double>(samples.size() - 1) / rate : now;
144 std::string block;
145 std::vector<double> values(static_cast<std::size_t>(m_info.channel_count()));
146 for (std::size_t s = 0; s < samples.size(); ++s) {
147 for (std::size_t c = 0; c < values.size(); ++c) {
148 values[c] = c < samples[s].size() ? static_cast<double>(samples[s][c]) : 0.0;
149 }
150 appendSample(block, m_info.channel_format(), values, first, s > 0 && rate > 0.0);
151 }
152 std::lock_guard<std::mutex> lock(m_queueMutex);
153 m_blocks.push_back(std::move(block));
154 }

166private:
167 void run()
168 {
169 auto server = TcpServer::listen("0.0.0.0", 0);
170 if (!server) {
171 m_listening.set_value();
172 return;
173 }
174 // The service port has the data port's number where it is free, as in liblsl.
175 auto service = UdpSocket::bind("0.0.0.0", server->port());
176 if (!service) {
177 service = UdpSocket::bind("0.0.0.0", 0);
178 }
179 auto discovery = UdpSocket::bind("0.0.0.0", kDiscoveryPort, true);
180 if (discovery) {
181 joinDiscoveryGroups(*discovery);
182 }
183 m_info.set_data_port(server->port());
184 m_info.set_service_port(service ? service->port() : 0);
185 m_shortinfo = m_info.to_string();
186 m_listening.set_value();
187
188 std::vector<Client> clients;
189 std::array<std::byte, 4096> chunk{};
190 while (m_bRunning) {
191 while (auto connection = server->accept(std::chrono::milliseconds(0))) {
192 Client client;
193 client.socket = std::move(*connection);
194 clients.push_back(std::move(client));
195 }
196 if (discovery) {
197 answerQueries(*discovery, false);
198 }
199 if (service) {
200 answerQueries(*service, true);
201 }
202 for (Client& client : clients) {
203 if (client.closing) {
204 continue;
205 }
206 // A streaming inlet sends nothing; reading still notices when it goes away
207 while (true) {
208 const auto got = client.socket.read_some(chunk, std::chrono::milliseconds(0));
209 if (!got) {
210 if (got.error() != std::errc::timed_out) {
211 client.socket.close();
212 }
213 break;
214 }
215 if (!client.streaming) {
216 client.request.append(reinterpret_cast<const char*>(chunk.data()), *got);
217 }
218 }
219 if (!client.streaming && client.socket.is_open()) {
220 handleRequest(client);
221 }
222 }
223 std::vector<std::string> blocks;
224 {
225 std::lock_guard<std::mutex> lock(m_queueMutex);
226 blocks.swap(m_blocks);
227 }
228 for (Client& client : clients) {
229 if (!client.streaming || !client.socket.is_open()) {
230 continue;
231 }
232 for (const std::string& block : blocks) {
233 if (client.socket.write(bytes(block))) {
234 client.socket.close();
235 break;
236 }
237 }
238 }
239 std::erase_if(clients, [](const Client& client) { return client.closing || !client.socket.is_open(); });
240 m_nClients.store(static_cast<int>(std::count_if(clients.begin(), clients.end(),
241 [](const Client& c) { return c.streaming; })));
242 std::this_thread::sleep_for(std::chrono::milliseconds(1));
243 }
244 m_nClients.store(0);
245 }

248 void answerQueries(UdpSocket& socket, bool timedata)
249 {
250 std::array<std::byte, 65536> buffer{};
251 while (const auto datagram = socket.receive_from(buffer, std::chrono::milliseconds(0))) {
252 const std::vector<std::string> lines = split(std::string_view(reinterpret_cast<const char*>(buffer.data()), datagram->size), '\n');
253 if (lines.size() < 2) {
254 continue;
255 }
256 const std::string_view kind = trimmed(lines[0]);
257 const std::vector<std::string> words = lines.size() > 2 ? split(trimmed(lines[2]), ' ') : std::vector<std::string>();
258 if (kind == "LSL:shortinfo" && words.size() == 2) {
259 const std::optional<int> port = to_int(words[0]);
260 if (matchesQuery(m_info, std::string(trimmed(lines[1]))) && port && *port > 0 && *port <= 65535) {
261 reply(socket, words[1] + "\r\n" + m_shortinfo, datagram->address, static_cast<std::uint16_t>(*port));
262 }
263 } else if (timedata && kind == "LSL:timedata") {
264 const double received = local_clock();
265 const std::vector<std::string> request = split(trimmed(lines[1]), ' ');
266 if (request.size() == 2) {
267 reply(socket, " " + request[0] + " " + request[1] + " " + number(received) + " " + number(local_clock()), datagram->address, datagram->port);
268 }
269 }
270 }

282 void handleRequest(Client& client)
283 {
284 const std::string& r = client.request;
285 if (r.starts_with("LSL:streamfeed/")) {
286 if (r.find("\r\n\r\n") == std::string::npos) {
287 client.closing = r.size() > kMaxRequestBytes;
288 return;
289 }
290 const std::vector<std::string> head = split(std::string_view(r).substr(0, r.find("\r\n")), ' ');
291 const bool ours = head.size() < 2 || head[1].empty() || head[1] == m_info.uid();
292 if (!ours) {
293 client.closing = true;
294 if (const std::error_code error = client.socket.write(bytes("LSL/110 404 Not found\r\n\r\n"))) {
295 mne::Log::debug("[lsl::stream_outlet] Cannot refuse a stream feed: {}", error.message());
296 }
297 return;
298 }
299 std::string reply = "LSL/110 200 OK\r\nUID: " + m_info.uid() + "\r\nByte-Order: " + byteOrderName() +
300 "\r\nSuppress-Subnormals: 0\r\nData-Protocol-Version: " + std::to_string(kProtocolVersion) + "\r\n\r\n";
301 std::vector<double> pattern(static_cast<std::size_t>(m_info.channel_count()));
302 for (const int offset : {4, 2}) {
303 for (std::size_t k = 0; k < pattern.size(); ++k) {
304 pattern[k] = testPatternValue(m_info.channel_format(), static_cast<int>(k), offset);
305 }
306 appendSample(reply, m_info.channel_format(), pattern, kTestPatternTime, false);
307 }
308 if (client.socket.write(bytes(reply))) {
309 client.closing = true;
310 return;
311 }
312 client.streaming = true;
313 client.request.clear();
314 } else if (r.starts_with("LSL:shortinfo\r\n")) {
315 const std::size_t end = r.find("\r\n", 15);
316 if (end == std::string::npos) {
317 client.closing = r.size() > kMaxRequestBytes;
318 return;
319 }
320 client.closing = true;
321 if (matchesQuery(m_info, std::string(trimmed(std::string_view(r).substr(15, end - 15))))) {
322 sendInfo(client);
323 }
324 } else if (r.starts_with("LSL:fullinfo")) {
325 if (r.find('\n') != std::string::npos) {
326 client.closing = true;
327 sendInfo(client);
328 }
329 } else if (r.find('\n') != std::string::npos || r.size() > kMaxRequestBytes) {
330 client.closing = true; // e.g. a protocol 1.00 stream feed, which is not served
331 }
332 }
