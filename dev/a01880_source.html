92template<typename T>
93struct AcqSpanPair
94{
95 std::span<T> first;
96 std::span<T> second;
97
99 [[nodiscard]] constexpr std::size_t size() const noexcept
100 {
101 return first.size() + second.size();
102 }

105 [[nodiscard]] constexpr bool empty() const noexcept
106 {
107 return size() == 0;
108 }
109
115 constexpr T& operator[](std::size_t i) const noexcept
116 {
117 return i < first.size() ? first[i] : second[i - first.size()];
