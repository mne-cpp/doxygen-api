217static QImage tightlyPackedRgba(const QImage& source)
218{
219 if (source.isNull())
220 return {};
221
222 QImage rgba = source.convertToFormat(QImage::Format_RGBA8888);
223 const qsizetype tightStride = qsizetype(rgba.width()) * 4;
224 if (rgba.bytesPerLine() == tightStride)
225 return rgba.copy();
226
227 QImage packed(rgba.size(), QImage::Format_RGBA8888);
228 for (int y = 0; y < rgba.height(); ++y) {
229 memcpy(packed.scanLine(y), rgba.constScanLine(y), size_t(tightStride));
230 }
231 return packed;
232}

287int BrainRenderer::Impl::claimUniformSlot(const char* caller)
288{
289 if (insidePass(caller))
290 return -1;
291 const int offset = currentUniformOffset;
292 // Once a pass has begun the buffer can no longer grow this frame
293 const qint64 capacity = passBegunThisFrame ? qint64(uniformBuffer->size()) : qint64(kMaxUniformSlots) * uniformSlotStride;
294 if (offset + uniformSlotStride > capacity) {
295 if (!overflowReported) {
296 qWarning("BrainRenderer: uniform buffer full at %s; further draws this frame are skipped", caller);
297 overflowReported = true;
298 }
299 return -1;
300 }
301 currentUniformOffset += uniformSlotStride;
302 if (uniformShadow.size() < currentUniformOffset)
303 uniformShadow.resize(currentUniformOffset);
304 return offset;
305}

309void BrainRenderer::Impl::flushUniforms(QRhiResourceUpdateBatch* u)
310{
311 if (currentUniformOffset <= flushedUniformBytes)
312 return;
313 if (quint32(currentUniformOffset) > uniformBuffer->size()) {
314 int slotCount = int(uniformBuffer->size()) / uniformSlotStride;
315 while (slotCount * uniformSlotStride < currentUniformOffset)
316 slotCount *= 2;
317 uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, quint32(slotCount * uniformSlotStride)));
318 uniformBuffer->create();
319 srb->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage, uniformBuffer.get(), kUniformBlockSize)});
320 srb->create();
321 flushedUniformBytes = 0;
322 }
323 u->updateDynamicBuffer(uniformBuffer.get(), quint32(flushedUniformBytes), quint32(currentUniformOffset - flushedUniformBytes),
324 uniformShadow.constData() + flushedUniformBytes);
325 flushedUniformBytes = currentUniformOffset;
326}

353QRhiBuffer* BrainRenderer::Impl::upload(GpuMesh& m, int slot, QRhiBuffer::UsageFlags usage, const GpuStream& stream, QRhiResourceUpdateBatch* u)
354{
355 auto& buffer = m.buffers[slot];
356 if (!stream.data || stream.bytes == 0) {
357 buffer.reset();
358 return nullptr;
359 }
360 // Static buffers are updated in place; only a size change reallocates
361 if (!buffer || buffer->size() != stream.bytes) {
362 buffer.reset(rhi->newBuffer(QRhiBuffer::Static, usage, stream.bytes));
363 buffer->create();
364 m.generations[slot] = std::numeric_limits<quint64>::max();
365 }
366 if (m.generations[slot] != stream.generation) {
367 u->uploadStaticBuffer(buffer.get(), 0, stream.bytes, stream.data);
368 m.generations[slot] = stream.generation;
369 }
370 return buffer.get();
371}

403void BrainRenderer::Impl::createResources(QRhi* newRhi, QRhiRenderPassDescriptor* rp, int sampleCount)
404{
405 rhi = newRhi;
406 uniformSlotStride = rhi->ubufAligned(kUniformBlockSize);
407
408 if (!uniformBuffer) {
409 uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, kInitialUniformSlots * uniformSlotStride));
410 uniformBuffer->create();
411 }
412
413 // Create SRB
414 if (!srb) {
415 srb.reset(rhi->newShaderResourceBindings());
416 srb->setBindings({// Use dynamic offset for the uniform buffer.
417 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage, uniformBuffer.get(), kUniformBlockSize)});
418 srb->create();
419 }
420
421 // Shader Loader
422 auto getShader = [](const QString& name) {
423 QFile f(name);
424 return f.open(QIODevice::ReadOnly) ? QShader::fromSerialized(f.readAll()) : QShader();
425 };
426
427 // List of modes to initialize
428 QList<ShaderMode> modes = {Standard, Holographic, Anatomical, Dipole, XRay, ShowNormals};
429
430 for (ShaderMode mode : modes) {
431 QString vert = (mode == Holographic || mode == XRay) ? ":/holographic.vert.qsb" : (mode == Anatomical) ? ":/anatomical.vert.qsb"
432 : (mode == Dipole) ? ":/dipole.vert.qsb"
433 : (mode == ShowNormals) ? ":/shownormals.vert.qsb"
434 : ":/standard.vert.qsb";
435
436 QString frag = (mode == Holographic || mode == XRay) ? ":/holographic.frag.qsb" : (mode == Anatomical) ? ":/anatomical.frag.qsb"
437 : (mode == Dipole) ? ":/dipole.frag.qsb"
438 : (mode == ShowNormals) ? ":/shownormals.frag.qsb"
439 : ":/standard.frag.qsb";
440
441 QShader vS = getShader(vert);
442 QShader fS = getShader(frag);
443
444 if (!vS.isValid() || !fS.isValid()) {
445 qWarning() << "BrainRenderer: Could not load shaders for mode" << mode << vert << frag;
446 continue;
447 }
448
449 // Setup Pipeline
450 auto pipeline = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
451
452 QRhiGraphicsPipeline::TargetBlend blend;
453 if (mode == Holographic || mode == XRay) {
454 blend.enable = true;
455 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
456 blend.dstColor = QRhiGraphicsPipeline::One;
457 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
458 blend.dstAlpha = QRhiGraphicsPipeline::One;
459 } else if (mode == Dipole) {
460 blend.enable = true;
461 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
462 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
463 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
464 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
465 }
466
467 auto setup = [&](QRhiGraphicsPipeline* p, QRhiGraphicsPipeline::CullMode cull) {
468 p->setShaderStages({{QRhiShaderStage::Vertex, vS}, {QRhiShaderStage::Fragment, fS}});
469
470 p->setVertexInputLayout(mode == Dipole ? instancedInputLayout() : surfaceInputLayout());
471 p->setShaderResourceBindings(srb.get());
472 p->setRenderPassDescriptor(rp);
473 p->setSampleCount(sampleCount);
474 p->setCullMode(cull);
475 if (mode == Holographic) {
476 p->setTargetBlends({blend});
477 p->setDepthTest(true);
478 p->setDepthWrite(false);
479 } else if (mode == XRay) {
480 p->setTargetBlends({blend});
481 p->setDepthTest(false); // Disable Depth Test to see through head
482 p->setDepthWrite(false);
483 } else if (mode == Dipole) {
484 p->setTargetBlends({blend});
485 p->setCullMode(QRhiGraphicsPipeline::None);
486 p->setDepthTest(true);
487 p->setDepthWrite(false);
488 } else {
489 p->setDepthTest(true);
490 p->setDepthWrite(true);
491 }
492 p->setFlags(QRhiGraphicsPipeline::UsesScissor);
493 p->create();
494 };
495
496 if (mode == Holographic || mode == XRay) { // Handle XRay back-faces same as Holographic
497 auto pipelineBack = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
498 setup(pipelineBack.get(), QRhiGraphicsPipeline::Front);
499 pipelinesBackColor[mode] = std::move(pipelineBack);
500 setup(pipeline.get(), QRhiGraphicsPipeline::Back); // Front faces
501 } else {
502 // Culling: None (Double-sided) to be safe for FreeSurfer meshes
503 setup(pipeline.get(), QRhiGraphicsPipeline::None);
504 }
505 pipelines[mode] = std::move(pipeline);
506 }
507
508 // Electrodes: opaque and depth-written, unlike the translucent dipole glyphs
509 const QShader evS = getShader(QStringLiteral(":/electrode.vert.qsb"));
510 const QShader efS = getShader(QStringLiteral(":/electrode.frag.qsb"));
511 electrodePipeline.reset();
512 if (evS.isValid() && efS.isValid()) {
513 electrodePipeline.reset(rhi->newGraphicsPipeline());
514 electrodePipeline->setShaderStages({{QRhiShaderStage::Vertex, evS}, {QRhiShaderStage::Fragment, efS}});
515 electrodePipeline->setVertexInputLayout(instancedInputLayout());
516 electrodePipeline->setShaderResourceBindings(srb.get());
517 electrodePipeline->setRenderPassDescriptor(rp);
518 electrodePipeline->setSampleCount(sampleCount);
519 electrodePipeline->setCullMode(QRhiGraphicsPipeline::None);
520 electrodePipeline->setDepthTest(true);
521 electrodePipeline->setDepthWrite(true);
522 electrodePipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);
523 electrodePipeline->create();
524 } else {
525 qWarning() << "BrainRenderer: Could not load the electrode shaders";
526 }
527
528 resourcesDirty = false;
529}

535void BrainRenderer::ensureRenderTargets(QRhi* rhi, QRhiTexture* colorTex, const QSize& pixelSize)
536{
537 // Rebuild when size changes OR when the backing texture changes
538 // (QRhiWidget may return a different colorTexture() each frame).
539 if (d->rtClear && d->rtSize == pixelSize && d->rtColorTex == colorTex)
540 return;
541
542 d->rtSize = pixelSize;
543 d->rtColorTex = colorTex;
544
545 // Shared depth-stencil buffer
546 d->dsBuffer.reset(rhi->newRenderBuffer(QRhiRenderBuffer::DepthStencil, pixelSize));
547 d->dsBuffer->create();
548
549 QRhiColorAttachment colorAtt(colorTex);
550 QRhiTextureRenderTargetDescription desc(colorAtt);
551 desc.setDepthStencilBuffer(d->dsBuffer.get());
552
553 // RT 1: Clearing (no preserve flags) — used for the first pass of each frame
554 d->rtClear.reset(rhi->newTextureRenderTarget(desc));
555 d->rpClear.reset(d->rtClear->newCompatibleRenderPassDescriptor());
556 d->rtClear->setRenderPassDescriptor(d->rpClear.get());
557 d->rtClear->create();
558
559 // RT 2: Preserving (load previous contents) — used for passes 2+
560 d->rtPreserve.reset(rhi->newTextureRenderTarget(desc,
561 QRhiTextureRenderTarget::PreserveColorContents | QRhiTextureRenderTarget::PreserveDepthStencilContents));
562 d->rpPreserve.reset(d->rtPreserve->newCompatibleRenderPassDescriptor());
563 d->rtPreserve->setRenderPassDescriptor(d->rpPreserve.get());
564 d->rtPreserve->create();
565}

581void BrainRenderer::newFrame()
582{
583 d->currentUniformOffset = 0;
584 d->flushedUniformBytes = 0;
585 d->overflowReported = false;
586 d->passBegunThisFrame = false;
587 d->videoOverlay.currentUniformOffset = 0;
588 d->sliceRes.currentUniformOffset = 0;
589
590 ++d->frameIndex;
591 for (auto it = d->meshes.begin(); it != d->meshes.end();) {
592 if (it->second.owner.expired() || d->frameIndex - it->second.lastUsedFrame > kEvictAfterFrames)
593 it = d->meshes.erase(it);
594 else
595 ++it;
596 }
597}

608void BrainRenderer::Impl::beginPass(QRhiCommandBuffer* cb, QRhiRenderTarget* rt, QRhiResourceUpdateBatch* updates)
609{
610 if (currentUniformOffset > flushedUniformBytes) {
611 if (!updates)
612 updates = rhi->nextResourceUpdateBatch();
613 flushUniforms(updates);
614 }
615 cb->beginPass(rt, QColor(0, 0, 0), {1.0f, 0}, updates);
616 passRecording = true;
617 passBegunThisFrame = true;
618 const int w = rt->pixelSize().width();
619 const int h = rt->pixelSize().height();
620 cb->setViewport(QRhiViewport(0, 0, w, h));
621 cb->setScissor(QRhiScissor(0, 0, w, h));
622}

683 return;
684 }

686 QShader fShader = QShader::fromSerialized(fFile.readAll());
687 if (!vShader.isValid() || !fShader.isValid()) {
688 qWarning() << "BrainRenderer: invalid video overlay shaders";
689 return;
690 }
691
692 // Uniform buffer (mat4 + vec4 + 4 floats = 96 bytes, pad to 256)
693 k.uniformSlotStride = rhi->ubufAligned(kUniformBlockSize);
694 k.uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
695 QRhiBuffer::UniformBuffer,
696 kVideoOverlaySlots * k.uniformSlotStride));

698
699 // Vertex buffer: one billboard quad per uniform slot — refilled each frame

701 QRhiBuffer::VertexBuffer, kVideoOverlaySlots * kVideoQuadBytes));
702 k.vertexBuffer->create();
703
704 // Index buffer: 2 triangles, 6 indices — uploaded once

707 QRhiBuffer::IndexBuffer, kIbSize));

712 QRhiSampler::None,
713 QRhiSampler::ClampToEdge,
714 QRhiSampler::ClampToEdge));
715 k.sampler->create();
716
717 // Initial texture uses the first real frame size.
718 k.texture.reset(rhi->newTexture(QRhiTexture::RGBA8, frame.size()));
719 k.texture->create();
720 k.textureSize = frame.size();
721
722 // SRB binding: uniform @ 0, sampled image @ 1
723 k.srb.reset(rhi->newShaderResourceBindings());
724 k.srb->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
725 QRhiShaderResourceBinding::VertexStage |
726 QRhiShaderResourceBinding::FragmentStage,
727 k.uniformBuffer.get(), kUniformBlockSize),

764 }

768 qWarning() << "BrainRenderer: invalid video decal shaders";

770 }

772 QRhiGraphicsPipeline::TargetBlend decalBlend;
773 decalBlend.enable = true;
774 decalBlend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
775 decalBlend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
776 decalBlend.srcAlpha = QRhiGraphicsPipeline::One;
777 decalBlend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
778
779 k.surfacePipeline.reset(rhi->newGraphicsPipeline());
780 k.surfacePipeline->setShaderStages({{QRhiShaderStage::Vertex, dvShader},
781 {QRhiShaderStage::Fragment, dfShader}});
782 k.surfacePipeline->setVertexInputLayout(surfaceInputLayout());
783 k.surfacePipeline->setShaderResourceBindings(k.srb.get());

785 k.surfacePipeline->setSampleCount(d->rtClear->sampleCount());
786 k.surfacePipeline->setCullMode(QRhiGraphicsPipeline::None);
787 k.surfacePipeline->setTargetBlends({decalBlend});

789 k.surfacePipeline->setDepthWrite(false);
790 k.surfacePipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);

792
793 k.initialized = true;
794 }

796 // ── Lazy depth-enhanced pipeline creation ───────────────────────

810 QRhiSampler::ClampToEdge));

815 QRhiTexture::RGBA8, QSize(1, 1), 1, QRhiTexture::MipMapped));

823 QRhiShaderResourceBinding::FragmentStage,

829 QRhiShaderResourceBinding::VertexStage |
830 QRhiShaderResourceBinding::FragmentStage,

832 k.srbDepth->create();

836 depthDecalBlend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
837 depthDecalBlend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
838 depthDecalBlend.srcAlpha = QRhiGraphicsPipeline::One;
839 depthDecalBlend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;

841 k.surfaceDepthPipeline.reset(rhi->newGraphicsPipeline());

843 {QRhiShaderStage::Fragment, ddfShader}});

898 QRhiTextureSubresourceUploadDescription sub(frame);

900 u->uploadTexture(k.texture.get(), desc);

902 }
903
904 // ── Depth texture upload ────────────────────────────────────────
905 if (overlay->isDepthEnabled() && overlay->hasDepthFrame() && k.depthInitialized) {

908 if (depthFrame.size() != k.depthTextureSize) {

914 k.srbDepth->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
915 QRhiShaderResourceBinding::VertexStage |
916 QRhiShaderResourceBinding::FragmentStage,
917 k.uniformBuffer.get(), kUniformBlockSize),
918 QRhiShaderResourceBinding::sampledTexture(1,
919 QRhiShaderResourceBinding::FragmentStage,
920 k.texture.get(), k.sampler.get()),
921 QRhiShaderResourceBinding::sampledTexture(2,
922 QRhiShaderResourceBinding::VertexStage |
923 QRhiShaderResourceBinding::FragmentStage,
924 k.depthTexture.get(), k.depthSampler.get())});
925 k.srbDepth->create();
926 k.uploadedDepthFrameGen = std::numeric_limits<quint64>::max();
927 }
928 if (overlay->depthFrameGeneration() != k.uploadedDepthFrameGen) {
929 QRhiTextureSubresourceUploadDescription depthSub(depthFrame);
930 QRhiTextureUploadDescription depthDesc({0, 0, depthSub});
931 u->uploadTexture(k.depthTexture.get(), depthDesc);
932 u->generateMips(k.depthTexture.get());
933 k.uploadedDepthFrameGen = overlay->depthFrameGeneration();
934 }
935 }

938
939int BrainRenderer::prepareVideoOverlayDraw(QRhiResourceUpdateBatch* u,
940 const SceneData& data,
941 VideoOverlay* overlay,
942 BrainSurface* surface)
943{
944 if (!overlay || !overlay->isEnabled() || !overlay->hasFrame())
945 return -1;
946
947 auto& k = d->videoOverlay;
948 if (!k.initialized || (surface && !k.surfacePipeline))
949 return -1;
950 if (d->insidePass("prepareVideoOverlayDraw"))
951 return -1;
952 if (surface && !d->uploadSurface(surface, u))
953 return -1;
954
955 const int uniformOffset = k.currentUniformOffset;
956 if (static_cast<quint32>(uniformOffset + kUniformBlockSize) > k.uniformBuffer->size())
957 return -1;
958 k.currentUniformOffset += k.uniformSlotStride;
959
960 if (!surface) {
961 // ── Billboard the quad to face the camera ──────────────────────
962 const QVector3D centre = overlay->focusPosition();
963 const float sizeM = overlay->sizeMeters();
964
965 // Preserve the video frame's aspect ratio: sizeMeters controls the
966 // width, and the height is derived from the frame dimensions.
967 const QImage& frm = overlay->frame();
968 const float aspect = (frm.height() > 0)
969 ? static_cast<float>(frm.width()) / frm.height()
970 : 1.0f;
971 const float halfW = 0.5f * sizeM;
972 const float halfH = (aspect > 0.0f) ? halfW / aspect : halfW;
973
974 QVector3D viewDir = centre - data.cameraPos;
975 if (viewDir.lengthSquared() < 1e-12f)
976 viewDir = QVector3D(0, 0, -1);
977 viewDir.normalize();
978
979 // Use the upHint (tracker→objective axis) when available so that
980 // the quad's long edge is perpendicular to the optical axis.
981 const QVector3D hint = overlay->upHint();
982 QVector3D up;
983 if (hint.lengthSquared() > 1e-8f) {
984 // Project the hint onto the plane perpendicular to viewDir
985 up = (hint - QVector3D::dotProduct(hint, viewDir) * viewDir).normalized();
986 if (up.lengthSquared() < 1e-8f)
987 up = QVector3D(0.0f, 0.0f, 1.0f);
988 } else {
989 QVector3D worldUp(0.0f, 0.0f, 1.0f);
990 if (std::abs(QVector3D::dotProduct(viewDir, worldUp)) > 0.95f)
991 worldUp = QVector3D(0.0f, 1.0f, 0.0f);
992 up = QVector3D::crossProduct(
993 QVector3D::crossProduct(viewDir, worldUp), viewDir)
994 .normalized();
995 }
996 QVector3D right = QVector3D::crossProduct(viewDir, up).normalized();
997
998 const QVector3D c00 = centre - right * halfW - up * halfH;
999 const QVector3D c10 = centre + right * halfW - up * halfH;
1000 const QVector3D c01 = centre - right * halfW + up * halfH;
1001 const QVector3D c11 = centre + right * halfW + up * halfH;
1002
1003 const float verts[4 * 5] = {
1004 c00.x(), c00.y(), c00.z(), 0.0f, 1.0f, // image y is flipped vs uv
1005 c10.x(), c10.y(), c10.z(), 1.0f, 1.0f,
1006 c01.x(), c01.y(), c01.z(), 0.0f, 0.0f,
1007 c11.x(), c11.y(), c11.z(), 1.0f, 0.0f};
1008 // Each slot has its own quad: every pane faces its own camera
1009 static_assert(sizeof(verts) == kVideoQuadBytes);
1010 u->updateDynamicBuffer(k.vertexBuffer.get(), quint32(uniformOffset / k.uniformSlotStride) * kVideoQuadBytes, sizeof(verts), verts);
1011
1012 // ── Uniforms (mat4 mvp, vec4 borderColor, float opacity, 3 pad) ─
1013 struct
1014 {
1015 float mvp[16];
1016 float borderColor[4];
1017 float opacity;
1018 float pad0, pad1, pad2;
1019 } ub;
1020 memcpy(ub.mvp, data.mvp.constData(), 64);
1021 ub.borderColor[0] = 0.0f;
1022 ub.borderColor[1] = 0.85f;
1023 ub.borderColor[2] = 1.0f;
1024 ub.borderColor[3] = 1.0f;
1025 ub.opacity = overlay->opacity();
1026 ub.pad0 = ub.pad1 = ub.pad2 = 0.0f;
1027 u->updateDynamicBuffer(k.uniformBuffer.get(), uniformOffset, sizeof(ub), &ub);
1028 return uniformOffset;
1029 }
1030
1031 // Approximate the local scalp normal from the focus position vector.
1032 QVector3D localNormal = overlay->focusPosition();
1033 if (localNormal.lengthSquared() < 1e-12f) {
1034 localNormal = QVector3D(0.0f, 0.0f, 1.0f);
1035 }
1036 localNormal.normalize();
1037
1038 // Use the tracker-derived up hint when available — this keeps the
1039 // decal orientation locked to the physical microscope and avoids
1040 // the 180° flip singularity that occurs with a fixed reference axis.
1041 QVector3D referenceUp = overlay->upHint();
1042 if (referenceUp.lengthSquared() < 1e-6f) {
1043 // Fallback when no tracker up is available.
1044 referenceUp = QVector3D(0.0f, 0.0f, 1.0f);
1045 if (std::abs(QVector3D::dotProduct(localNormal, referenceUp)) > 0.95f) {
1046 referenceUp = QVector3D(1.0f, 0.0f, 0.0f);
1047 }
1048 }
1049 // Orthogonalise against the surface normal so the frame is tangent.
1050 referenceUp = (referenceUp - QVector3D::dotProduct(referenceUp, localNormal) * localNormal);
1051 if (referenceUp.lengthSquared() < 1e-12f) {
1052 referenceUp = QVector3D(0.0f, 0.0f, 1.0f);
1053 }
1054 referenceUp.normalize();
1055
1056 const QVector3D axisV = referenceUp;
1057 const QVector3D axisU = QVector3D::crossProduct(axisV, localNormal).normalized();
1058
1059 const bool useDepth = d->videoOverlayUsesDepth(overlay);
1060
1061 // Uniform block — the depth-enhanced shader has an extra vec4 depthParams
1062 // but the base 7×vec4 layout (112 bytes) still fits within kUniformBlockSize
1063 // (256 bytes) even with the extra vec4 (128 bytes total).
1064 struct
1065 {
1066 float mvp[16];
1067 float focusAndSize[4];
1068 float axisUAndOpacity[4];
1069 float axisVAndOffset[4];
1070 float axisNAndDepth[4];
1071 float cameraPosAndFacing[4];
1072 float borderColor[4];
1073 float depthParams[4];
1074 } ub;
1075 memcpy(ub.mvp, data.mvp.constData(), 64);
1076 ub.focusAndSize[0] = overlay->focusPosition().x();
1077 ub.focusAndSize[1] = overlay->focusPosition().y();
1078 ub.focusAndSize[2] = overlay->focusPosition().z();
1079 ub.focusAndSize[3] = overlay->sizeMeters();
1080 ub.axisUAndOpacity[0] = axisU.x();
1081 ub.axisUAndOpacity[1] = axisU.y();
1082 ub.axisUAndOpacity[2] = axisU.z();
1083 ub.axisUAndOpacity[3] = overlay->opacity();
1084 ub.axisVAndOffset[0] = axisV.x();
1085 ub.axisVAndOffset[1] = axisV.y();
1086 ub.axisVAndOffset[2] = axisV.z();
1087 ub.axisVAndOffset[3] = 0.00045f;
1088 ub.axisNAndDepth[0] = localNormal.x();
1089 ub.axisNAndDepth[1] = localNormal.y();
1090 ub.axisNAndDepth[2] = localNormal.z();
1091 ub.axisNAndDepth[3] = std::max(0.015f, overlay->sizeMeters() * 0.45f);
1092 ub.cameraPosAndFacing[0] = data.cameraPos.x();
1093 ub.cameraPosAndFacing[1] = data.cameraPos.y();
1094 ub.cameraPosAndFacing[2] = data.cameraPos.z();
1095 const QImage& decalFrame = overlay->frame();
1096 ub.cameraPosAndFacing[3] = (decalFrame.height() > 0)
1097 ? static_cast<float>(decalFrame.width()) / decalFrame.height()
1098 : 1.0f;
1099 ub.borderColor[0] = 0.0f;
1100 ub.borderColor[1] = 0.85f;
1101 ub.borderColor[2] = 1.0f;
1102 ub.borderColor[3] = 1.0f;
1103 ub.depthParams[0] = useDepth ? overlay->depthScale() : 0.0f;
1104 ub.depthParams[1] = useDepth ? static_cast<float>(overlay->depthSteps()) : 0.0f;
1105 ub.depthParams[2] = useDepth ? 1.0f : 0.0f;
1106 ub.depthParams[3] = 0.0f;
1107 u->updateDynamicBuffer(k.uniformBuffer.get(), uniformOffset, sizeof(ub), &ub);
1108 return uniformOffset;
1109}

1187 QRhiBuffer::UniformBuffer,
1188 64 * k.uniformSlotStride));
1189 k.uniformBuffer->create();
1190
1191 // Index buffer (shared by all 3 slots): 2 triangles = 6 indices
1192 k.indexBuffer.reset(rhi->newBuffer(QRhiBuffer::Immutable,
1193 QRhiBuffer::IndexBuffer,
1194 6 * sizeof(quint32)));
1195 k.indexBuffer->create();
1196
1197 k.sampler.reset(rhi->newSampler(QRhiSampler::Linear, QRhiSampler::Linear,
1198 QRhiSampler::None,
1199 QRhiSampler::ClampToEdge,
1200 QRhiSampler::ClampToEdge));
1201 k.sampler->create();

1203 // Per-slot resources: vertex buffer + texture + SRB
1204 for (int i = 0; i < Impl::kMaxSliceSlots; ++i) {
1205 constexpr int kVbSize = 4 * 5 * sizeof(float); // 4 verts × (3 pos + 2 uv)
1206 k.vertexBuffer[i].reset(rhi->newBuffer(QRhiBuffer::Dynamic,
1207 QRhiBuffer::VertexBuffer, kVbSize));
1208 k.vertexBuffer[i]->create();
1209
1210 // 1×1 placeholder texture — real size set on first data
1211 k.sliceSlots[i].texture.reset(rhi->newTexture(QRhiTexture::R8, QSize(1, 1)));
1212 k.sliceSlots[i].texture->create();
1213 k.sliceSlots[i].textureSize = QSize(1, 1);
1214
1215 k.srb[i].reset(rhi->newShaderResourceBindings());
1216 k.srb[i]->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
1217 QRhiShaderResourceBinding::VertexStage |
1218 QRhiShaderResourceBinding::FragmentStage,
1219 k.uniformBuffer.get(), kUniformBlockSize),
1220 QRhiShaderResourceBinding::sampledTexture(1,
1221 QRhiShaderResourceBinding::FragmentStage,
1222 k.sliceSlots[i].texture.get(), k.sampler.get())});
1223 k.srb[i]->create();
1224 }
1225
1226 // Pipeline: alpha-blended, depth-test on, depth-write off
1227 QRhiGraphicsPipeline::TargetBlend blend;
1228 blend.enable = true;
1229 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
1230 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
1231 blend.srcAlpha = QRhiGraphicsPipeline::One;
1232 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
1233
1234 k.pipeline.reset(rhi->newGraphicsPipeline());
1235 k.pipeline->setShaderStages({{QRhiShaderStage::Vertex, vShader},
1236 {QRhiShaderStage::Fragment, fShader}});
1237 QRhiVertexInputLayout il;
1238 il.setBindings({{5 * sizeof(float)}});
1239 il.setAttributes({{0, 0, QRhiVertexInputAttribute::Float3, 0},
1240 {0, 1, QRhiVertexInputAttribute::Float2, 3 * sizeof(float)}});
1241 k.pipeline->setVertexInputLayout(il);
1242 k.pipeline->setShaderResourceBindings(k.srb[0].get());
1243 k.pipeline->setRenderPassDescriptor(d->rtClear->renderPassDescriptor());
1244 k.pipeline->setSampleCount(d->rtClear->sampleCount());
1245 k.pipeline->setCullMode(QRhiGraphicsPipeline::None);
1246 k.pipeline->setTargetBlends({blend});
1247 k.pipeline->setDepthTest(true);
1248 k.pipeline->setDepthWrite(false);
1249 k.pipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);
1250 k.pipeline->create();
1251
1252 k.initialized = true;
1253 }
1254
1255 // ── Index buffer (one-shot upload) ──────────────────────────────
1256 if (!k.indexUploaded) {
1257 const quint32 idx[6] = {0, 1, 2, 2, 1, 3};
1258 u->uploadStaticBuffer(k.indexBuffer.get(), idx);
1259 k.indexUploaded = true;
1260 }
1261
1262 // ── Texture (re-create on size change, upload data) ─────────────
1263 // Convert to R8 (grayscale) for the slice shader
1264 QImage gray = img.convertToFormat(QImage::Format_Grayscale8);
1265 const QSize imgSize = gray.size();
1266
1267 if (imgSize != sliceSlot.textureSize) {
1268 sliceSlot.texture.reset(rhi->newTexture(QRhiTexture::R8, imgSize));
1269 sliceSlot.texture->create();
1270 sliceSlot.textureSize = imgSize;
1271 // Rebuild SRB to point at new texture
1272 k.srb[slotIndex]->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
1273 QRhiShaderResourceBinding::VertexStage |
1274 QRhiShaderResourceBinding::FragmentStage,
1275 k.uniformBuffer.get(), kUniformBlockSize),
1276 QRhiShaderResourceBinding::sampledTexture(1,
1277 QRhiShaderResourceBinding::FragmentStage,
1278 sliceSlot.texture.get(), k.sampler.get())});
1279 k.srb[slotIndex]->create();
1280 }
1281
1282 // Upload texture — R8 needs tightly-packed row data
1283 {
1284 const qsizetype tightStride = qsizetype(gray.width());
1285 QByteArray texData;
1286 texData.resize(gray.height() * tightStride);
1287 for (int y = 0; y < gray.height(); ++y) {
1288 memcpy(texData.data() + y * tightStride,
1289 gray.constScanLine(y),
1290 size_t(tightStride));
1291 }
1292 QRhiTextureSubresourceUploadDescription sub(texData.constData(), texData.size());
1293 sub.setSourceSize(imgSize);
1294 QRhiTextureUploadDescription desc({0, 0, sub});
1295 u->uploadTexture(sliceSlot.texture.get(), desc);
1296 }
1297
1298 // ── Vertex data ─────────────────────────────────────────────────
1299 slice->generateQuadVertices(sliceSlot.vertices);
1300 u->updateDynamicBuffer(k.vertexBuffer[slotIndex].get(), 0,
1301 static_cast<quint32>(sliceSlot.vertices.size() * sizeof(float)),
1302 sliceSlot.vertices.constData());
1303
1304 sliceSlot.opacity = slice->opacity();
1305 sliceSlot.windowCenter = slice->windowCenter();
1306 sliceSlot.windowWidth = slice->windowWidth();
1307 sliceSlot.visible = true;
1308}

1314 int slotIndex)
1315{

1317 return -1;

1319 auto& k = d->sliceRes;
1320 if (!k.initialized || !k.pipeline)
1321 return -1;

1324 if (!sliceSlot.visible || d->insidePass("prepareSliceDraw"))
1325 return -1;

1328 k.currentUniformOffset += k.uniformSlotStride;
1329 if (static_cast<quint32>(uniformOffset + kUniformBlockSize) > k.uniformBuffer->size())
1330 return -1;

1332 // Uniform block matches slice.vert / slice.frag layout:
1333 // mat4 mvp (64 bytes)
1334 // mat4 sliceToWorld (64 bytes)
1335 // float opacity (4)
1336 // float windowCenter(4)
1337 // float windowWidth (4)
1338 // float _pad0 (4)

1340 {
1341 float mvp[16];
1342 float sliceToWorld[16];
1343 float opacity;
1344 float windowCenter;
1345 float windowWidth;
1346 float _pad0;
1347 } ub;
1348 memcpy(ub.mvp, data.mvp.constData(), 64);
1349 // SliceToWorld is identity for pre-transformed vertices (already in world coords)
1350 QMatrix4x4 identity;
1351 identity.setToIdentity();
1352 memcpy(ub.sliceToWorld, identity.constData(), 64);
1353 ub.opacity = sliceSlot.opacity;
1354 ub.windowCenter = sliceSlot.windowCenter;
1355 ub.windowWidth = sliceSlot.windowWidth;
1356 ub._pad0 = 0.0f;
1357 u->updateDynamicBuffer(k.uniformBuffer.get(), uniformOffset, sizeof(ub), &ub);
1358
1359 return uniformOffset;
