56 // Specify which channel types are needed

59 m_lChannelTypeList << "MAG";

62 m_lChannelTypeList << "GRAD";

66 }

70 if(lChannelList.at(i).kind == FIFFV_STIM_CH && !m_lChannelTypeList.contains("STIM")) {

190void ModalitySelectionView::redrawGUI()
191{
192 m_qListModalityCheckBox.clear();
193
194 //Delete all widgets in the averages layout
195 QGridLayout* topLayout = static_cast<QGridLayout*>(this->layout());
196 if(!topLayout) {
197 topLayout = new QGridLayout();
198 }
199
200 QLayoutItem *child;
201 while ((child = topLayout->takeAt(0)) != 0) {
202 delete child->widget();
203 delete child;
204 }
205
206 QMapIterator<QString, bool> i(m_modalityMap);
207
208 int count = 0;
209 while (i.hasNext()) {
210 i.next();
211
212 if(m_lChannelTypeList.contains(i.key(), Qt::CaseInsensitive)) {
213 QCheckBox* t_pCheckBoxModality = new QCheckBox(i.key());
214 t_pCheckBoxModality->setChecked(i.value());
215 m_qListModalityCheckBox << t_pCheckBoxModality;
216 connect(t_pCheckBoxModality,&QCheckBox::checkStateChanged,
217 this, &ModalitySelectionView::onUpdateModalityCheckbox);
218 topLayout->addWidget(t_pCheckBoxModality,count,0);
219 count++;
220 }
221 }
222
223 //Find Modalities tab and add current layout
224 this->setLayout(topLayout);
225}
