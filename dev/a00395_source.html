42FiffAnnotations mne::annotations_from_events(const MatrixXi& events,
43 double sfreq,
44 const std::map<int, std::string>& event_desc,
45 int first_sample)
46{
47 FiffAnnotations annotations;
48 if (events.rows() == 0 || events.cols() < 3) {
49 return annotations;
50 }
51 if (sfreq <= 0.0) {
52 mne::Log::warning("annotations_from_events: sfreq must be positive, got {:f}", sfreq);
53 return annotations;
54 }
55 for (Index i = 0; i < events.rows(); ++i) {
56 const int id = events(i, 2);
57 const auto label = event_desc.find(id);
58 annotations.append(static_cast<double>(events(i, 0) - first_sample) / sfreq, 0.0, label != event_desc.end() ? label->second : std::to_string(id));
59 }
60 return annotations;
61}

65MatrixXi mne::events_from_annotations(const FiffAnnotations& annotations,
66 double sfreq,
67 const std::map<std::string, int, std::less<>>& event_id,
68 int first_sample)
69{
70 if (annotations.empty()) {
71 return MatrixXi(0, 3);
72 }
73 if (sfreq <= 0.0) {
74 mne::Log::warning("events_from_annotations: sfreq must be positive, got {:f}", sfreq);
75 return MatrixXi(0, 3);
76 }
77 MatrixXi result(static_cast<Index>(annotations.size()), 3);
78 Index row = 0;
79 for (const FiffAnnotation& a : annotations) {
80 const auto id = event_id.find(a.description);
81 result(row, 0) = static_cast<int>(std::round(a.onset * sfreq)) + first_sample;
82 result(row, 1) = 0;
83 result(row, 2) = id != event_id.end() ? id->second : mne::to_int(a.description).value_or(0);
84 ++row;
85 }
86 return result;
87}
