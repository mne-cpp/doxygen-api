82Eigen::Quaternionf rotation_about_axis(const Eigen::Vector3f& axis, float degrees)
83{
84 const float norm = axis.norm();
85 if (norm <= 0.0f) {
86 return Eigen::Quaternionf::Identity();
87 }
88 return Eigen::Quaternionf(Eigen::AngleAxisf(radians(degrees), axis / norm)).normalized();
89}

93Eigen::Quaternionf rotation_from_euler_angles(float pitch, float yaw, float roll)
94{
95 // Half angles; the product yaw(y) * pitch(x) * roll(z) written out
96 const float c1 = std::cos(radians(yaw) * 0.5f);
97 const float s1 = std::sin(radians(yaw) * 0.5f);
98 const float c2 = std::cos(radians(roll) * 0.5f);
99 const float s2 = std::sin(radians(roll) * 0.5f);
100 const float c3 = std::cos(radians(pitch) * 0.5f);
101 const float s3 = std::sin(radians(pitch) * 0.5f);
102 const float c1c2 = c1 * c2;
103 const float s1s2 = s1 * s2;
104 return {c1c2 * c3 + s1s2 * s3, c1c2 * s3 + s1s2 * c3, s1 * c2 * c3 - c1 * s2 * s3, c1 * s2 * c3 - s1 * c2 * s3};
105}

116Eigen::Quaternionf view_preset_rotation(int preset)
117{
118 const Eigen::Vector3f x = Eigen::Vector3f::UnitX();
119 const Eigen::Vector3f y = Eigen::Vector3f::UnitY();
120 const Eigen::Vector3f z = Eigen::Vector3f::UnitZ();
121 switch (preset) {
122 case 1:
123 return perspective_preset_rotation();
124 case 2:
125 return rotation_about_axis(y, 180.0f) * rotation_about_axis(z, 180.0f) * rotation_about_axis(x, 90.0f) *
126 rotation_about_axis(z, 180.0f);
127 case 3:
128 return rotation_about_axis(x, 90.0f) * rotation_about_axis(y, -90.0f);
129 case 4:
130 return rotation_about_axis(x, 180.0f) * rotation_about_axis(z, 180.0f);
131 case 5:
132 return rotation_about_axis(x, 90.0f);
133 case 6:
134 return rotation_about_axis(x, 90.0f) * rotation_about_axis(y, 90.0f);
135 default:
136 return rotation_about_axis(z, 180.0f);
137 }
138}

142Eigen::Matrix4f perspective_matrix(float fov_y_degrees, float aspect, float near_plane, float far_plane)
143{
144 Eigen::Matrix4f m = Eigen::Matrix4f::Identity();
145 if (near_plane == far_plane || aspect == 0.0f) {
146 return m;
147 }
148 const float half = radians(fov_y_degrees / 2.0f);
149 const float sine = std::sin(half);
150 if (sine == 0.0f) {
151 return m;
152 }
153 const float cotan = std::cos(half) / sine;
154 const float clip = far_plane - near_plane;
155 m(0, 0) = cotan / aspect;
156 m(1, 1) = cotan;
157 m(2, 2) = -(near_plane + far_plane) / clip;
158 m(2, 3) = -(2.0f * near_plane * far_plane) / clip;
159 m(3, 2) = -1.0f;
160 m(3, 3) = 0.0f;
161 return m;
162}

166Eigen::Matrix4f look_at_matrix(const Eigen::Vector3f& eye, const Eigen::Vector3f& center, const Eigen::Vector3f& up)
167{
168 Eigen::Vector3f forward = center - eye;
169 constexpr float fuzzy = 0.00001f;
170 if (std::abs(forward.x()) <= fuzzy && std::abs(forward.y()) <= fuzzy && std::abs(forward.z()) <= fuzzy) {
171 return Eigen::Matrix4f::Identity();
172 }
173 forward.normalize();
174 const Eigen::Vector3f side = forward.cross(up).normalized();
175 const Eigen::Vector3f true_up = side.cross(forward);
176
177 Eigen::Matrix4f m = Eigen::Matrix4f::Identity();
178 m.block<1, 3>(0, 0) = side.transpose();
179 m.block<1, 3>(1, 0) = true_up.transpose();
180 m.block<1, 3>(2, 0) = -forward.transpose();
181 return m * translation_matrix(-eye);
182}

195CameraMatrices OrbitCamera::compute(const Eigen::Quaternionf& rotation, float zoom, const Eigen::Vector2f& pan,
196 bool apply_pan, float aspect) const
197{
198 CameraMatrices result;
199 const float far_plane = std::max(m_scene_size * 20.0f, 100.0f);
200 result.projection = perspective_matrix(45.0f, aspect, m_scene_size * 0.01f, far_plane);
201
202 const float base_distance = m_scene_size * 1.5f;
203 result.distance = base_distance - zoom * (m_scene_size * 0.05f);
204 result.eye = rotation * Eigen::Vector3f(0.0f, 0.0f, result.distance);
205 result.up = rotation * Eigen::Vector3f::UnitY();
206 result.look_at = Eigen::Vector3f::Zero();
207
208 // Pan shifts camera and look-at point along the view plane
209 if (apply_pan && (pan.x() != 0.0f || pan.y() != 0.0f)) {
210 const Eigen::Vector3f right = (rotation * Eigen::Vector3f::UnitX()).normalized();
211 const Eigen::Vector3f offset = right * pan.x() + result.up.normalized() * pan.y();
212 result.look_at += offset;
213 result.eye += offset;
214 }
215
216 result.view = look_at_matrix(result.eye, result.look_at, result.up);
217 result.model = translation_matrix(-m_scene_center);
218 return result;
219}

241Eigen::Quaternionf OrbitCamera::orbit(const Eigen::Quaternionf& rotation, double dx, double dy, float degrees_per_pixel)
242{
243 const Eigen::Quaternionf preset = perspective_preset_rotation();
244 const Eigen::Quaternionf effective = rotation * preset;
245
246 const Eigen::Vector3f up_axis = (effective * Eigen::Vector3f::UnitY()).normalized();
247 const Eigen::Vector3f right_axis = (effective * Eigen::Vector3f::UnitX()).normalized();
248
249 const Eigen::Quaternionf yaw = rotation_about_axis(up_axis, static_cast<float>(-dx) * degrees_per_pixel);
250 const Eigen::Quaternionf pitch = rotation_about_axis(right_axis, static_cast<float>(-dy) * degrees_per_pixel);
251
252 return (yaw * pitch * effective * preset.conjugate()).normalized();
253}
