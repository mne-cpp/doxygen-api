106void NetworkObject::createNodeGeometry()
107{
108 if (!m_nodeVertexData.empty())
109 return;
110
111 // Create an icosphere (subdivision level 1) for nodes
112 const int subdivisions = 1;
113 const float radius = 1.0f; // Unit sphere, scaled per-instance
114
115 // Start with icosahedron
116 const float t = (1.0f + std::sqrt(5.0f)) / 2.0f;
117
118 std::vector<Eigen::Vector3f> vertices = {
119 Eigen::Vector3f(-1, t, 0).normalized() * radius,
120 Eigen::Vector3f(1, t, 0).normalized() * radius,
121 Eigen::Vector3f(-1, -t, 0).normalized() * radius,
122 Eigen::Vector3f(1, -t, 0).normalized() * radius,
123 Eigen::Vector3f(0, -1, t).normalized() * radius,
124 Eigen::Vector3f(0, 1, t).normalized() * radius,
125 Eigen::Vector3f(0, -1, -t).normalized() * radius,
126 Eigen::Vector3f(0, 1, -t).normalized() * radius,
127 Eigen::Vector3f(t, 0, -1).normalized() * radius,
128 Eigen::Vector3f(t, 0, 1).normalized() * radius,
129 Eigen::Vector3f(-t, 0, -1).normalized() * radius,
130 Eigen::Vector3f(-t, 0, 1).normalized() * radius,
131 };
132
133 std::vector<uint32_t> indices = {
134 0,
135 11,
136 5,
137 0,
138 5,

195
196 // Subdivide
197 for (int s = 0; s < subdivisions; ++s) {
198 std::vector<uint32_t> newIndices;
199 std::map<uint64_t, uint32_t> midpointCache;
200
201 auto getMidpoint = [&](uint32_t i0, uint32_t i1) -> uint32_t {
202 uint64_t key = static_cast<uint64_t>(std::min(i0, i1)) << 32 | std::max(i0, i1);
203 auto it = midpointCache.find(key);
204 if (it != midpointCache.end())
205 return it->second;
206
207 const Eigen::Vector3f mid = ((vertices[i0] + vertices[i1]) / 2.0f).normalized() * radius;
208 uint32_t idx = static_cast<uint32_t>(vertices.size());
209 vertices.push_back(mid);
210 midpointCache[key] = idx;
211 return idx;
212 };
213
214 for (size_t i = 0; i < indices.size(); i += 3) {
215 uint32_t a = indices[i], b = indices[i + 1], c = indices[i + 2];
216 uint32_t ab = getMidpoint(a, b);
217 uint32_t bc = getMidpoint(b, c);
218 uint32_t ca = getMidpoint(c, a);
219
220 newIndices.insert(newIndices.end(), {a, ab, ca});
221 newIndices.insert(newIndices.end(), {b, bc, ab});
222 newIndices.insert(newIndices.end(), {c, ca, bc});
223 newIndices.insert(newIndices.end(), {ab, bc, ca});
224 }
225
226 indices = std::move(newIndices);
227 }
228
229 // Build vertex data with normals (normal = normalized position for sphere)
230 std::vector<VertexData> vd;
231 vd.reserve(vertices.size());
232 for (const auto& v : vertices) {
233 const Eigen::Vector3f n = v.normalized();
234 vd.push_back({v.x(), v.y(), v.z(), n.x(), n.y(), n.z()});
235 }
236
237 m_nodeIndexCount = static_cast<int>(indices.size());
238
239 m_nodeVertexData.resize(vd.size() * sizeof(VertexData));
240 memcpy(m_nodeVertexData.data(), vd.data(), m_nodeVertexData.size());
241
242 m_nodeIndexData.resize(indices.size() * sizeof(uint32_t));
243 memcpy(m_nodeIndexData.data(), indices.data(), m_nodeIndexData.size());
244
245 ++m_nodeGeometryGeneration;
246}

250void NetworkObject::createEdgeGeometry()
251{
252 if (!m_edgeVertexData.empty())
253 return;
254
255 // Create a unit cylinder along Y axis (height=1, radius=1, scaled per-instance)
256 const int segments = 8;
257 const float radius = 1.0f;
258 const float halfHeight = 0.5f;
259
260 std::vector<VertexData> vertices;
261 std::vector<uint32_t> indices;
262
263 // Top center (0)
264 vertices.push_back({0, halfHeight, 0, 0, 1, 0});
265 // Bottom center (1)
266 vertices.push_back({0, -halfHeight, 0, 0, -1, 0});
267
268 // Side vertices: top ring (2..2+segments-1), bottom ring (2+segments..2+2*segments-1)
269 for (int i = 0; i < segments; ++i) {
270 float angle = 2.0f * static_cast<float>(std::numbers::pi) * i / segments;
271 float x = radius * std::cos(angle);
272 float z = radius * std::sin(angle);
273
274 const Eigen::Vector3f normal = Eigen::Vector3f(x, 0.0f, z).normalized();
275
276 // Top side vertex
277 vertices.push_back({x, halfHeight, z, normal.x(), normal.y(), normal.z()});
278 // Bottom side vertex
279 vertices.push_back({x, -halfHeight, z, normal.x(), normal.y(), normal.z()});
280 }
281
282 // Top cap vertices (for proper normals)
283 int topCapStart = static_cast<int>(vertices.size());
284 for (int i = 0; i < segments; ++i) {
285 float angle = 2.0f * static_cast<float>(std::numbers::pi) * i / segments;
286 float x = radius * std::cos(angle);
287 float z = radius * std::sin(angle);
288 vertices.push_back({x, halfHeight, z, 0, 1, 0});
289 }
290
291 // Bottom cap vertices
292 int botCapStart = static_cast<int>(vertices.size());
293 for (int i = 0; i < segments; ++i) {
294 float angle = 2.0f * static_cast<float>(std::numbers::pi) * i / segments;
295 float x = radius * std::cos(angle);
296 float z = radius * std::sin(angle);
297 vertices.push_back({x, -halfHeight, z, 0, -1, 0});
298 }
299
300 // Side faces
301 int sideStart = 2;
302 for (int i = 0; i < segments; ++i) {
303 int next = (i + 1) % segments;
304 int topCur = sideStart + i * 2;
305 int botCur = sideStart + i * 2 + 1;
306 int topNext = sideStart + next * 2;
307 int botNext = sideStart + next * 2 + 1;
308
309 indices.insert(indices.end(), {static_cast<uint32_t>(topCur), static_cast<uint32_t>(topNext), static_cast<uint32_t>(botCur)});
310 indices.insert(indices.end(), {static_cast<uint32_t>(botCur), static_cast<uint32_t>(topNext), static_cast<uint32_t>(botNext)});
311 }
312
313 // Top cap
314 for (int i = 0; i < segments; ++i) {
315 int next = (i + 1) % segments;
316 indices.push_back(0); // center
317 indices.push_back(topCapStart + i);
318 indices.push_back(topCapStart + next);
319 }
320
321 // Bottom cap
322 for (int i = 0; i < segments; ++i) {
323 int next = (i + 1) % segments;
324 indices.push_back(1); // center
325 indices.push_back(botCapStart + next);
326 indices.push_back(botCapStart + i);
327 }
328
329 m_edgeIndexCount = static_cast<int>(indices.size());
330
331 m_edgeVertexData.resize(vertices.size() * sizeof(VertexData));
332 memcpy(m_edgeVertexData.data(), vertices.data(), m_edgeVertexData.size());
333
334 m_edgeIndexData.resize(indices.size() * sizeof(uint32_t));
335 memcpy(m_edgeIndexData.data(), indices.data(), m_edgeIndexData.size());
336
337 ++m_edgeGeometryGeneration;
338}

342void NetworkObject::buildNodeInstances()
343{
344 if (m_network.empty()) {
345 m_nodeInstanceCount = 0;
346 ++m_nodeInstanceGeneration;
347 return;
348 }
349
350 const auto& nodes = m_network.nodes();
351 int iMaxDegree = m_network.degree_range(EdgeSet::thresholded).second;
352 if (iMaxDegree == 0)
353 iMaxDegree = 1;
354
355 const VisualizationInfo& vizInfo = m_network.visualization_info();
356 const std::string& colormap = vizInfo.colormap.empty() ? m_colormap : vizInfo.colormap;
357
358 std::vector<InstanceData> instances;
359 instances.reserve(nodes.size());
360
361 for (int i = 0; i < static_cast<int>(nodes.size()); ++i) {
362 const int degree = m_network.degree(i, EdgeSet::thresholded);
363 if (degree == 0)
364 continue;
365
366 const Vector3f& vert = nodes[i].position();
367
368 // Scale: nodes with higher degree are larger
369 // Range: 0.0006 to 0.005 (same as mne_viz_qt)
370 float scaleFactor = (static_cast<float>(degree) / static_cast<float>(iMaxDegree)) * (0.005f - 0.0006f) + 0.0006f;
371
372 InstanceData inst;
373 storeMatrix(modelMatrix(vert, Eigen::Quaternionf::Identity(), Eigen::Vector3f::Constant(scaleFactor)), inst.model);
374
375 // Color: colormap-based or fixed
376 if (vizInfo.coloring == VisualizationInfo::Coloring::colormap) {
377 float normalized = static_cast<float>(degree) / static_cast<float>(iMaxDegree);
378 storeColor(map_color(colormap, normalized), inst.color);
379 inst.color[3] = std::pow(normalized, 4.0f);
380 } else {
381 inst.color[0] = vizInfo.node_color[0] / 255.0f;
382 inst.color[1] = vizInfo.node_color[1] / 255.0f;
383 inst.color[2] = vizInfo.node_color[2] / 255.0f;
384 inst.color[3] = vizInfo.node_color[3] / 255.0f;
385 }
386 inst.isSelected = 0.0f;
387
388 instances.push_back(inst);
389 }
390
391 m_nodeInstanceCount = static_cast<int>(instances.size());
392 m_nodeInstanceData.resize(m_nodeInstanceCount * sizeof(InstanceData));
393 if (m_nodeInstanceCount > 0) {
394 memcpy(m_nodeInstanceData.data(), instances.data(), m_nodeInstanceData.size());
395 }
396 ++m_nodeInstanceGeneration;
397}

401void NetworkObject::buildEdgeInstances()
402{
403 if (m_network.empty()) {
404 m_edgeInstanceCount = 0;
405 ++m_edgeInstanceGeneration;
406 return;
407 }
408
409 const auto& nodes = m_network.nodes();
410
411 const auto [dMinWeight, dMaxWeight] = m_network.weight_range(EdgeSet::thresholded);
412 double dWeightRange = dMaxWeight - dMinWeight;
413 if (dWeightRange == 0.0)
414 dWeightRange = 1.0;
415
416 const VisualizationInfo& vizInfo = m_network.visualization_info();
417 const std::string& colormap = vizInfo.colormap.empty() ? m_colormap : vizInfo.colormap;
418
419 std::vector<InstanceData> instances;
420 instances.reserve(m_network.edge_count(EdgeSet::thresholded));
421
422 for (const NetworkEdge& edge : m_network.edges()) {
423 if (!edge.active())
424 continue;
425
426 const Vector3f& vStart = nodes[edge.source()].position();
427 const Vector3f& vEnd = nodes[edge.target()].position();
428
429 if (vStart == vEnd)
430 continue;
431
432 double dWeight = std::fabs(edge.weight());
433 if (dWeight == 0.0)
434 continue;
435
436 float normalizedWeight = static_cast<float>(std::fabs((dWeight - dMinWeight) / dWeightRange));
437
438 // Cylinder radius: proportional to weight, range 0.0001 to 0.001
439 float edgeRadius = 0.0001f + normalizedWeight * 0.0009f;
440
441 InstanceData inst;
442 storeMatrix(segmentMatrix(vStart, vEnd, edgeRadius), inst.model);
443
444 // Color
445 if (vizInfo.coloring == VisualizationInfo::Coloring::colormap) {
446 float normalized = (dMaxWeight != 0.0) ? static_cast<float>(std::fabs(dWeight / dMaxWeight)) : 0.0f;
447 storeColor(map_color(colormap, normalized), inst.color);
448 inst.color[3] = std::pow(normalized, 1.5f);
449 } else {
450 inst.color[0] = vizInfo.edge_color[0] / 255.0f;
451 inst.color[1] = vizInfo.edge_color[1] / 255.0f;
452 inst.color[2] = vizInfo.edge_color[2] / 255.0f;
453 inst.color[3] = vizInfo.edge_color[3] / 255.0f;
454 }
455 inst.isSelected = 0.0f;
456
457 instances.push_back(inst);
458 }
459
460 m_edgeInstanceCount = static_cast<int>(instances.size());
461 m_edgeInstanceData.resize(m_edgeInstanceCount * sizeof(InstanceData));
462 if (m_edgeInstanceCount > 0) {
463 memcpy(m_edgeInstanceData.data(), instances.data(), m_edgeInstanceData.size());
464 }
465 ++m_edgeInstanceGeneration;
466}
