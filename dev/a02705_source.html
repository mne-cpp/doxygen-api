74VectorXd computeRowPsd(const VectorXd& row, int nFft, double sfreq)
75{
76 // Zero-pad or truncate to nFft
77 VectorXd segment = VectorXd::Zero(nFft);
78 int copyLen = std::min(static_cast<int>(row.size()), nFft);
79 segment.head(copyLen) = row.head(copyLen);
80
81 // Apply Hann window
82 for (int i = 0; i < copyLen; ++i) {
83 double w = 0.5 * (1.0 - std::cos(2.0 * std::numbers::pi * i / (copyLen - 1)));
84 segment(i) *= w;
85 }
86
87 // Compute FFT via correlation (real FFT using DFT)
88 int nFreqs = nFft / 2 + 1;
89 VectorXd psd(nFreqs);
90
91 for (int f = 0; f < nFreqs; ++f) {
92 double re = 0.0, im = 0.0;
93 for (int t = 0; t < nFft; ++t) {
94 double angle = -2.0 * std::numbers::pi * f * t / nFft;
95 re += segment(t) * std::cos(angle);
96 im += segment(t) * std::sin(angle);
97 }
98 psd(f) = (re * re + im * im) / (sfreq * nFft);
99 // Double for non-DC/Nyquist bins (one-sided spectrum)
100 if (f > 0 && f < nFreqs - 1)
101 psd(f) *= 2.0;
102 }
103
104 return psd;
105}

113std::vector<InvSourceEstimate> mne::apply_inverse_epochs(
114 const std::vector<MatrixXd>& epochs,
115 const MNEInverseOperator& inverse,
116 float lambda2,
117 std::string_view method,
118 float tmin,
119 float tstep,
120 bool pickNormal)
121{
122 std::vector<InvSourceEstimate> results;
123
124 if (epochs.empty()) {
125 mne::Log::warning("[apply_inverse_epochs] No epochs provided.");
126 return results;
127 }
128
129 // Create the minimum norm estimator
130 InvMinimumNorm mn(inverse, lambda2, method);
131
132 // Setup once with nave=1 (per-epoch)
133 mn.doInverseSetup(1, pickNormal);
134
135 results.reserve(epochs.size());
136 for (const MatrixXd& epoch : epochs) {
137 results.push_back(mn.calculateInverse(epoch, tmin, tstep, pickNormal));
138 if (results.back().empty()) {
139 mne::Log::warning("[apply_inverse_epochs] Epoch {} produced empty source estimate.", results.size() - 1);
140 }
141 }
142
143 return results;
144}

148InvSourceEstimate mne::apply_inverse_raw(
149 const FiffRawData& raw,
150 const MNEInverseOperator& inverse,
151 float lambda2,
152 std::string_view method,
153 int from,
154 int to,
155 bool pickNormal)
156{
157 // Default: use full range
158 if (from < 0)
159 from = raw.first_samp;
160 if (to < 0)
161 to = raw.last_samp;
162
163 // Pick channels matching the inverse operator
164 RowVectorXi picks = FiffInfo::pick_channels(raw.info.ch_names, inverse.noise_cov.names);
165 if (picks.size() == 0) {
166 mne::Log::warning("[apply_inverse_raw] No channels match the inverse operator.");
167 return InvSourceEstimate();
168 }
169
170 // Read raw data
171 MatrixXd data, times;
172 raw.read_raw_segment(data, times, from, to, picks);
173
174 if (data.cols() == 0) {
175 mne::Log::warning("[apply_inverse_raw] No data read from raw file.");
176 return InvSourceEstimate();
177 }
178
179 float tmin = static_cast<float>(from) / raw.info.sfreq;
180 float tstep = 1.0f / raw.info.sfreq;
181
182 // Apply inverse
183 InvMinimumNorm mn(inverse, lambda2, method);
184 mn.doInverseSetup(1, pickNormal);
185
186 return mn.calculateInverse(data, tmin, tstep, pickNormal);
187}

191FiffEvoked mne::apply_forward(const MNEForwardSolution& fwd, const InvSourceEstimate& stc, const FiffInfo& info)
192{
193 // Adapted from mne.forward.forward.apply_forward / _apply_forward (MNE-Python, BSD-3-Clause).
194 if (fwd.sol.data.size() == 0 || stc.empty()) {
195 mne::Log::warning("[apply_forward] Empty forward solution or source estimate.");
196 return FiffEvoked();
197 }
198 MNEForwardSolution fixed = fwd;
199 const int nHemi = fixed.src.size();
200 if (nHemi > 0 && !fixed.isFixedOrient()) {
201 fixed.convert_to_surf_ori();
202 fixed.to_fixed_ori();
203 }
204 const MatrixXd& gain = fixed.sol.data;
205
206 std::vector<int> columns(static_cast<std::size_t>(stc.data.rows()));
207 if (nHemi == 0) {
208 if (gain.cols() != stc.data.rows()) {
209 mne::Log::warning("[apply_forward] Forward columns {} != source estimate rows {}", gain.cols(), stc.data.rows());
210 return FiffEvoked();
211 }
212 std::iota(columns.begin(), columns.end(), 0);
213 } else {
214 const int nLh = stc.nVerticesLh >= 0 ? stc.nVerticesLh : static_cast<int>(stc.vertices.size());
215 for (int r = 0; r < std::ssize(columns); ++r) {
216 const int hemi = (r < nLh || nHemi == 1) ? 0 : 1;
217 const VectorXi& vertno = fixed.src.hemisphereAt(hemi)->vertno;
218 int offset = 0;
219 for (int h = 0; h < hemi; ++h)
220 offset += static_cast<int>(fixed.src.hemisphereAt(h)->vertno.size());
221 const auto* hit = std::find(vertno.data(), vertno.data() + vertno.size(), stc.vertices(r));
222 if (hit == vertno.data() + vertno.size()) {
223 mne::Log::warning("[apply_forward] Vertex {} is not in the forward solution.", stc.vertices(r));
224 return FiffEvoked();
225 }
226 columns[r] = offset + static_cast<int>(hit - vertno.data());
227 }
228 }
229
230 // A gain matrix without channel names is taken to have the info's channels in order
231 RowVectorXi sel(fixed.sol.row_names.empty() ? gain.rows() : fixed.sol.row_names.size());
232 if (fixed.sol.row_names.empty()) {
233 if (info.nchan != gain.rows()) {
234 mne::Log::warning("[apply_forward] Unnamed forward rows {} != info channels {}", gain.rows(), info.nchan);
235 return FiffEvoked();
236 }
237 std::iota(sel.begin(), sel.end(), 0);
238 }
239 for (int c = 0; c < static_cast<int>(fixed.sol.row_names.size()); ++c) {
240 sel(c) = static_cast<int>(mne::channel_index(info.ch_names, fixed.sol.row_names[c]));
241 if (sel(c) < 0) {
242 mne::Log::warning("[apply_forward] Channel {} of the forward is not in the info.", fixed.sol.row_names[c]);
243 return FiffEvoked();
244 }
245 }
246
247 FiffEvoked evoked;
248 MatrixXd selected(gain.rows(), std::ssize(columns));
249 for (int r = 0; r < std::ssize(columns); ++r)
250 selected.col(r) = gain.col(columns[r]);
251 evoked.data = selected * stc.data;
252 evoked.info = info.pick_info(sel);
253 evoked.times = RowVectorXf::LinSpaced(stc.data.cols(), stc.tmin, stc.tmin + (stc.data.cols() - 1) * stc.tstep);
254 evoked.first = static_cast<int>(std::lround(stc.tmin * info.sfreq));
255 evoked.last = evoked.first + static_cast<int>(stc.data.cols()) - 1;
256 evoked.nave = 1;
257 evoked.comment = "Simulated";
258 return evoked;
259}

263std::pair<VectorXd, VectorXd> mne::estimate_snr(
264 const FiffEvoked& evoked,
265 const MNEInverseOperator& inverse)
266{
267 // Adapted from MNE-Python mne.minimum_norm.estimate_snr (BSD-3-Clause), itself after
268 // compute_regularization in MNE-C's mne_analyze/regularization.c.
269 const MNEInverseOperator inv = inverse.prepare_inverse_operator(evoked.nave, 1.0f / 9.0f, false);
270 if (inv.eigen_fields.empty()) {
271 mne::Log::warning("[estimate_snr] Could not prepare the inverse operator.");
272 return std::pair<VectorXd, VectorXd>();
273 }
274 const MatrixXd white = inv.whitener * inv.proj * evoked.pick_channels(inv.noise_cov.names).data;
275 const MatrixXd whiteEf = inv.eigen_fields.data * white;
276
277 int rank = 0;
278 if (!inv.noise_cov.diag) {
279 rank = static_cast<int>((inv.noise_cov.eig.array() > 0).count());
280 } else {
281 MatrixXd proj;
282 rank = inv.noise_cov.dim - FiffProj::make_projector(inv.projs, inv.noise_cov.names, proj);
283 }
284
285 const int nTimes = static_cast<int>(white.cols());
286 VectorXd snr = white.colwise().squaredNorm().transpose() / rank;
287 VectorXd lambda2 = VectorXd::Constant(nTimes, 10.0);
288 std::vector<bool> remaining(nTimes, true);
289 for (int t = 0; t < nTimes; ++t) {
290 if (snr(t) <= 1.0) {
291 lambda2(t) = std::numeric_limits<double>::infinity();
292 remaining[t] = false;
293 }
294 }
295
296 const ArrayXd sing2 = inv.sing.array().square();
297 const double limit = mne::Numerics::chi2Isf(1e-3, rank);
298 bool converged = false;
299 for (int iter = 0; iter < 1000 && !converged; ++iter) {
300 converged = true;
301 for (int t = 0; t < nTimes; ++t) {
302 if (!remaining[t])
303 continue;
304 const ArrayXd keep = (inv.sing.array() == 0).select(1.0, lambda2(t) / (sing2 + lambda2(t)));
305 if ((whiteEf.col(t).array() * keep).matrix().squaredNorm() < limit) {
306 remaining[t] = false;
307 } else {
308 lambda2(t) *= 0.99;
309 converged = false;
310 }
311 }
312 }
313 if (!converged)
314 mne::Log::warning("[estimate_snr] SNR estimation did not converge.");
315
316 return std::pair<VectorXd, VectorXd>(snr.cwiseSqrt(), lambda2.cwiseSqrt().cwiseInverse());
317}

321std::pair<MatrixXd, int> mne::compute_whitener(
322 const FiffCov& noiseCov,
323 int rank)
324{
325 const int dim = noiseCov.dim;
326
327 if (dim <= 0) {
328 mne::Log::warning("[compute_whitener] Empty noise covariance.");
329 return std::pair<MatrixXd, int>(MatrixXd(), 0);
330 }
331
332 // Use pre-computed eigendecomposition if available
333 VectorXd eig;
334 MatrixXd eigvec;
335
336 // eigvec holds one eigenvector per row, as FiffCov does.
337 if (noiseCov.eig.size() > 0 && noiseCov.eigvec.size() > 0) {
338 eig = noiseCov.eig;
339 eigvec = noiseCov.eigvec;
340 } else {
341 SelfAdjointEigenSolver<MatrixXd> solver(noiseCov.data);
342 eig = solver.eigenvalues();
343 eigvec = solver.eigenvectors().transpose();
344 }
345
346 // Auto-detect rank from eigenvalue spectrum. FiffCov::prepare_noise_cov already zeroed the null space per
347 // channel type; a threshold relative to the largest (EEG) eigenvalue would drop every MEG component.
348 const bool prepared = noiseCov.eig.size() > 0 && noiseCov.eigvec.size() > 0;
349 if (rank <= 0) {
350 double maxEig = eig.maxCoeff();
351 double threshold = prepared ? 0.0 : maxEig * 1e-10;
352 rank = 0;
353 for (int i = 0; i < eig.size(); ++i) {
354 if (eig(i) > threshold)
355 ++rank;
356 }
357 if (rank == 0)
358 rank = 1;
359 }
360
361 // Build whitening matrix: W = diag(1/sqrt(eig)) @ eigvec
362 // Only use the top 'rank' eigenvalues
363 VectorXd invSqrtEig = VectorXd::Zero(eig.size());
364 int effectiveRank = 0;
365
366 // Eigenvalues are in ascending order — use last 'rank' values
367 for (int i = eig.size() - 1; i >= 0 && effectiveRank < rank; --i) {
368 if (eig(i) > 1e-30) {
369 invSqrtEig(i) = 1.0 / std::sqrt(eig(i));
370 ++effectiveRank;
371 }
372 }
373
374 MatrixXd whitener = invSqrtEig.asDiagonal() * eigvec;
375
376 return std::pair<MatrixXd, int>(whitener, effectiveRank);
377}

381std::pair<MatrixXd, VectorXd> mne::compute_source_psd(
382 const InvSourceEstimate& stc,
383 float sfreq,
384 float fmin,
385 float fmax,
386 int nFft)
387{
388 if (stc.empty()) {
389 mne::Log::warning("[compute_source_psd] Empty source estimate.");
390 return std::pair<MatrixXd, VectorXd>();
391 }
392
393 const int nSources = static_cast<int>(stc.data.rows());
394 const int nTimes = static_cast<int>(stc.data.cols());
395
396 if (nFft <= 0)
397 nFft = nTimes;
398 if (fmax < 0)
399 fmax = sfreq / 2.0f;
400
401 const int nFreqs = nFft / 2 + 1;
402
403 // Build frequency vector
404 VectorXd freqs(nFreqs);
405 for (int f = 0; f < nFreqs; ++f) {
406 freqs(f) = static_cast<double>(f) * sfreq / nFft;
407 }
408
409 // Find frequency range indices
410 int fminIdx = 0, fmaxIdx = nFreqs - 1;
411 for (int f = 0; f < nFreqs; ++f) {
412 if (freqs(f) >= fmin) {
413 fminIdx = f;
414 break;
415 }
416 }
417 for (int f = nFreqs - 1; f >= 0; --f) {
418 if (freqs(f) <= fmax) {
419 fmaxIdx = f;
420 break;
421 }
422 }
423
424 int nBandFreqs = fmaxIdx - fminIdx + 1;
425 if (nBandFreqs <= 0) {
426 mne::Log::warning("[compute_source_psd] No frequencies in range.");
427 return std::pair<MatrixXd, VectorXd>();
428 }
429
430 // Compute PSD for each source
431 MatrixXd psd(nSources, nBandFreqs);
432
433 for (int s = 0; s < nSources; ++s) {
434 VectorXd fullPsd = computeRowPsd(stc.data.row(s).transpose(), nFft, sfreq);
435 psd.row(s) = fullPsd.segment(fminIdx, nBandFreqs).transpose();
436 }
437
438 VectorXd bandFreqs = freqs.segment(fminIdx, nBandFreqs);
439
440 return std::pair<MatrixXd, VectorXd>(psd, bandFreqs);
441}

445std::map<std::string, VectorXd> mne::compute_source_band_power(
446 const InvSourceEstimate& stc,
447 float sfreq,
448 const std::map<std::string, std::pair<float, float>>& bands)
449{
450 std::map<std::string, VectorXd> result;
451
452 if (stc.empty() || bands.empty()) {
453 return result;
454 }
455
456 // Compute full PSD
457 auto [psd, freqs] = compute_source_psd(stc, sfreq);
458 if (psd.size() == 0)
459 return result;
460
461 const int nSources = static_cast<int>(psd.rows());
462 const int nFreqs = static_cast<int>(freqs.size());
463 double df = (nFreqs > 1) ? (freqs(1) - freqs(0)) : 1.0;
464
465 for (const auto& [name, range] : bands) {
466 const float bfmin = range.first;
467 const float bfmax = range.second;
468
469 VectorXd bandPower = VectorXd::Zero(nSources);
470
471 for (int f = 0; f < nFreqs; ++f) {
472 if (freqs(f) >= bfmin && freqs(f) <= bfmax) {
473 bandPower += psd.col(f) * df;
474 }
475 }
476
477 result[name] = bandPower;
478 }
479
480 return result;
481}
