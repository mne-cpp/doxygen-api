109bool MNEBemSurface::addTriangleData()
110{
111 //
112 // Main triangulation
113 //
114 mne::Log::info("\tCompleting triangulation info...");
115 this->tri_cent = MatrixX3d::Zero(this->ntri, 3);
116 this->tri_nn = MatrixX3d::Zero(this->ntri, 3);
117 this->tri_area = VectorXd::Zero(this->ntri);
118
119 Matrix3d r;
120 Vector3d a, b;
121 int k = 0;
122 float size = 0;
123 for (std::int32_t i = 0; i < this->ntri; ++i) {
124 for (std::int32_t j = 0; j < 3; ++j) {
125 k = this->itris(i, j);
126
127 r(j, 0) = this->rr(k, 0);
128 r(j, 1) = this->rr(k, 1);
129 r(j, 2) = this->rr(k, 2);
130
131 this->tri_cent(i, 0) += this->rr(k, 0);
132 this->tri_cent(i, 1) += this->rr(k, 1);
133 this->tri_cent(i, 2) += this->rr(k, 2);
134 }
135 this->tri_cent.row(i) /= 3.0f;
136
137 //cross product {cross((r2-r1),(r3-r1))}
138 a = (r.row(1) - r.row(0)).transpose();
139 b = (r.row(2) - r.row(0)).transpose();
140 this->tri_nn(i, 0) = a(1) * b(2) - a(2) * b(1);
141 this->tri_nn(i, 1) = a(2) * b(0) - a(0) * b(2);
142 this->tri_nn(i, 2) = a(0) * b(1) - a(1) * b(0);
143
144 //area
145 size = this->tri_nn.row(i) * this->tri_nn.row(i).transpose();
146 size = std::pow(size, 0.5f);
147
148 this->tri_area(i) = size / 2.0f;
149 this->tri_nn.row(i) /= size;
150 }
151
152 mne::Log::info("Adding additional geometry info\n");
153 add_geometry_info();
154
155 mne::Log::info("[done]\n");
156
157 return true;
158}

162bool MNEBemSurface::add_geometry_info()
163{
164 int k, c, p, q;
165 bool found;
166
167 //Create neighboring triangle vector using temporary std::vector for efficient appending
168 {
169 // One list per vertex, as mne-python's neighbor_tri
170 std::vector<std::vector<int>> temp_ntri(this->np);
171 for (p = 0; p < this->itris.rows(); p++) {
172 for (k = 0; k < 3; k++) {
173 temp_ntri[this->itris(p, k)].push_back(p);
174 }
175 }
176 neighbor_tri.resize(this->np);
177 for (k = 0; k < static_cast<int>(temp_ntri.size()); k++) {
178 neighbor_tri[k] = Eigen::Map<Eigen::VectorXi>(temp_ntri[k].data(), temp_ntri[k].size());
179 }
180 }
181
182 //Create the neighboring vertices vector using temporary std::vector
183 {
184 std::vector<std::vector<int>> temp_nvert(this->np);
185 for (k = 0; k < this->np; k++) {
186 for (p = 0; p < neighbor_tri[k].size(); p++) {
187 //Fit in the other vertices of the neighboring triangle
188 for (c = 0; c < 3; c++) {
189 int vert = this->itris(neighbor_tri[k][p], c);
190
191 if (vert != k) {
192 found = false;
193
194 for (q = 0; q < static_cast<int>(temp_nvert[k].size()); q++) {
195 if (temp_nvert[k][q] == vert) {
196 found = true;
197 break;
198 }
199 }
200
201 if (!found) {
202 temp_nvert[k].push_back(vert);
203 }
204 }
205 }
206 }
207 }
208 neighbor_vert.resize(this->np);
209 for (k = 0; k < this->np; k++) {
210 neighbor_vert[k] = Eigen::Map<Eigen::VectorXi>(temp_nvert[k].data(), temp_nvert[k].size());
211 }
212 }
213
214 return true;
215}

219bool MNEBemSurface::addVertexNormals()
220{
221 //
222 // Accumulate the vertex normals from scratch (MNE-C mne_add_vertex_normals)
223 //
224 this->nn = NormalsT::Zero(this->np, 3);
225
226 for (std::int32_t p = 0; p < this->ntri; ++p) //check each triangle
227 {
228 for (std::int32_t j = 0; j < 3; ++j) {
229 int nodenr;
230 nodenr = this->itris(p, j); //find the corners(nodes) of the triangles
231 this->nn(nodenr, 0) += this->tri_nn(p, 0); //add the triangle normal to the nodenormal
232 this->nn(nodenr, 1) += this->tri_nn(p, 1);
233 this->nn(nodenr, 2) += this->tri_nn(p, 2);
234 }
235 }
236
237 // normalize
238 for (std::int32_t p = 0; p < this->np; ++p) {
239 const float size = this->nn.row(p).norm();
240 if (size > 0.0f)
241 this->nn.row(p) /= size;
242 }
243
244 return true;
245}

278std::vector<MNEBemSurface> MNEBemSurface::makeScalpSurfaces(
279 const MNEBemSurface& outerSkin,
280 const std::vector<int>& targetVertices)
281{
282 std::vector<MNEBemSurface> results;
283
284 if (outerSkin.np <= 0 || outerSkin.ntri <= 0) {
285 mne::Log::warning("MNEBemSurface::makeScalpSurfaces - Input surface is empty.");
286 return results;
287 }
288
289 for (int target : targetVertices) {
290 if (target >= outerSkin.np) {
291 // No decimation needed, just copy
292 results.push_back(MNEBemSurface(outerSkin));
293 continue;
294 }
295
296 // Work on a copy
297 MNEBemSurface surf(outerSkin);
298
299 // Build vertex-referenced rr as doubles, triangles as ints
300 // Use iterative edge collapse: find shortest edge, collapse to midpoint
301 Eigen::MatrixX3d rr = surf.rr.cast<double>();
302 Eigen::MatrixX3d nn = surf.nn.cast<double>();
303
304 // Build triangle list as std::vector for easy manipulation
305 int nTri = surf.ntri;
306 std::vector<std::array<int, 3>> tris(nTri);
307 for (int i = 0; i < nTri; ++i) {
308 tris[i] = {surf.itris(i, 0), surf.itris(i, 1), surf.itris(i, 2)};
309 }
310
311 // Track which vertices are alive
312 int currentNp = surf.np;
313 std::vector<bool> vertAlive(currentNp, true);
314 // Redirect map: when a vertex is collapsed, redirect to its replacement
315 std::vector<int> redirect(currentNp);
316 for (int i = 0; i < currentNp; ++i)
317 redirect[i] = i;
318
319 auto resolve = [&](int v) -> int {
320 while (redirect[v] != v)
321 v = redirect[v];
322 return v;
323 };
324
325 while (currentNp > target) {
326 // Find shortest edge in the mesh
327 double bestLen = std::numeric_limits<double>::max();
328 int bestTri = -1;
329 int bestEdge = -1;
330
331 for (int t = 0; t < static_cast<int>(tris.size()); ++t) {
332 int v0 = resolve(tris[t][0]);
333 int v1 = resolve(tris[t][1]);
334 int v2 = resolve(tris[t][2]);
335 // Skip degenerate triangles
336 if (v0 == v1 || v1 == v2 || v0 == v2)
337 continue;
338
339 double d01 = edgeLenSq(rr, v0, v1);
340 double d12 = edgeLenSq(rr, v1, v2);
341 double d20 = edgeLenSq(rr, v2, v0);
342
343 if (d01 < bestLen) {
344 bestLen = d01;
345 bestTri = t;
346 bestEdge = 0;
347 }
348 if (d12 < bestLen) {
349 bestLen = d12;
350 bestTri = t;
351 bestEdge = 1;
352 }
353 if (d20 < bestLen) {
354 bestLen = d20;
355 bestTri = t;
356 bestEdge = 2;
357 }
358 }
359
360 if (bestTri < 0)
361 break;
362
363 // Collapse edge: merge the two vertices into the midpoint
364 int va = resolve(tris[bestTri][bestEdge]);
365 int vb = resolve(tris[bestTri][(bestEdge + 1) % 3]);
366
367 // Place midpoint at va
368 rr.row(va) = (rr.row(va) + rr.row(vb)) * 0.5;
369 nn.row(va) = (nn.row(va) + nn.row(vb)).normalized();
370
371 // Redirect vb -> va
372 redirect[vb] = va;
373 vertAlive[vb] = false;
374 currentNp--;
375 }
376
377 // Rebuild compact vertex array and triangle array
378 std::vector<int> oldToNew(surf.np, -1);
379 int newNp = 0;
380 for (int i = 0; i < surf.np; ++i) {
381 if (vertAlive[i] && resolve(i) == i) {
382 oldToNew[i] = newNp++;
383 }
384 }
385
386 MNEBemSurface decimated;
387 decimated.id = surf.id;
388 decimated.coord_frame = surf.coord_frame;
389 decimated.sigma = surf.sigma;
390 decimated.np = newNp;
391
392 // Use Eigen types matching MNESurfaceOrVolume
393 MNESurfaceOrVolume::PointsT newRr(newNp, 3);
394 Eigen::MatrixX3f newNn(newNp, 3);
395 for (int i = 0; i < surf.np; ++i) {
396 if (oldToNew[i] >= 0) {
397 newRr.row(oldToNew[i]) = rr.row(i).cast<float>();
398 newNn.row(oldToNew[i]) = nn.row(i).cast<float>();
399 }
400 }
401 decimated.rr = newRr;
402 decimated.nn = newNn;
403
404 // Rebuild triangles
405 std::vector<std::array<int, 3>> newTris;
406 for (const auto& tri : tris) {
407 int a = oldToNew[resolve(tri[0])];
408 int b = oldToNew[resolve(tri[1])];
409 int c = oldToNew[resolve(tri[2])];
410 if (a >= 0 && b >= 0 && c >= 0 && a != b && b != c && a != c) {
411 newTris.push_back({a, b, c});
412 }
413 }
414
415 decimated.ntri = static_cast<int>(newTris.size());
416 MNESurfaceOrVolume::TrianglesT newItris(decimated.ntri, 3);
417 for (int i = 0; i < decimated.ntri; ++i) {
418 newItris(i, 0) = newTris[i][0];
419 newItris(i, 1) = newTris[i][1];
420 newItris(i, 2) = newTris[i][2];
421 }
422 decimated.itris = newItris;
423
424 // Recompute triangle data
425 decimated.addTriangleData();
426
427 results.push_back(decimated);
428 }
429
430 return results;
431}
