61VectorXd Linalg::combine_xyz(const VectorXd& vec)
62{
63 if (vec.size() % 3 != 0) {
64 return VectorXd();
65 }
66
67 MatrixXd tmp = MatrixXd(vec.transpose());
68 SparseMatrix<double> s = make_block_diag(tmp, 3);
69
70 SparseMatrix<double> sC = s * s.transpose();
71 VectorXd comb(sC.rows());
72
73 for (std::int32_t i = 0; i < sC.rows(); ++i)
74 comb[i] = sC.coeff(i, i);
75

87// Thin SVD of a matrix with at least as many rows as columns
88Linalg::ThinSvd thin_svd_tall(const MatrixXd& A)
89{
90 const Index m = A.rows();
91 const Index n = A.cols();
92 ThreadPool& pool = ThreadPool::global();
93 // Each block must hold at least two square factors' worth of rows to pay for the extra reduction
94 // A fixed number of blocks, not one per core, so every machine computes the same result
95 const Index blocks = std::min<Index>(kBlocks, n > 0 ? m / (2 * n) : 1);
96
97 Linalg::ThinSvd result;
98 if (blocks < 2 || m * n < 1000000) {
99 BDCSVD<MatrixXd, ComputeThinU | ComputeThinV> svd(A);
100 result.singularValues = svd.singularValues();
101 result.U = svd.matrixU();
102 result.V = svd.matrixV();
103 return result;
104 }
105
106 std::vector<Index> starts(static_cast<std::size_t>(blocks) + 1);
107 for (Index b = 0; b <= blocks; ++b)
108 starts[static_cast<std::size_t>(b)] = b * m / blocks;
109
110 std::vector<HouseholderQR<MatrixXd>> local(static_cast<std::size_t>(blocks));
111 MatrixXd stacked(blocks * n, n);
112 pool.parallel_for(0, static_cast<std::size_t>(blocks), [&](std::size_t b) {
113 const Index rows = starts[b + 1] - starts[b];
114 local[b].compute(A.middleRows(starts[b], rows));
115 stacked.middleRows(static_cast<Index>(b) * n, n) =
116 local[b].matrixQR().topRows(n).triangularView<Upper>().toDenseMatrix();
117 });
118
119 const HouseholderQR<MatrixXd> top(stacked);
120 const MatrixXd R = top.matrixQR().topRows(n).triangularView<Upper>();
121 BDCSVD<MatrixXd, ComputeFullU | ComputeFullV> small(R);
122
123 // The columns of W are independent, so the reduction's orthogonal factor is applied block by block in parallel
124 MatrixXd W = MatrixXd::Zero(blocks * n, n);
125 W.topRows(n) = small.matrixU();
126 pool.parallel_for(0, static_cast<std::size_t>(blocks), [&](std::size_t b) {
127 const Index first = static_cast<Index>(b) * n / blocks;
128 const Index last = (static_cast<Index>(b) + 1) * n / blocks;
129 auto columns = W.middleCols(first, last - first);
130 columns.applyOnTheLeft(top.householderQ());
131 });
132
133 result.singularValues = small.singularValues();
134 result.V = small.matrixV();
135 result.U.resize(m, n);
136 pool.parallel_for(0, static_cast<std::size_t>(blocks), [&](std::size_t b) {
137 const Index rows = starts[b + 1] - starts[b];
138 MatrixXd Y = MatrixXd::Zero(rows, n);
139 Y.topRows(n) = W.middleRows(static_cast<Index>(b) * n, n);
140 Y.applyOnTheLeft(local[b].householderQ());
141 result.U.middleRows(starts[b], rows) = Y;
142 });
143 return result;
144}

150Linalg::ThinSvd Linalg::thin_svd(const MatrixXd& A)
151{
152 if (A.rows() >= A.cols())
153 return thin_svd_tall(A);
154 ThinSvd transposed = thin_svd_tall(A.transpose());
155 std::swap(transposed.U, transposed.V);
156 return transposed;
157}

161MatrixXd Linalg::parallel_product(const MatrixXd& A, const MatrixXd& B)
162{
163 ThreadPool& pool = ThreadPool::global();
164 // Below a few million multiply-adds the hand-over costs more than it saves
165 const Index blocks = A.rows() * A.cols() * B.cols() < 4000000
166 ? 1
167 : std::min<Index>(kBlocks, B.cols());
168 if (blocks < 2)
169 return A * B;
170 MatrixXd C(A.rows(), B.cols());
171 pool.parallel_for(0, static_cast<std::size_t>(blocks), [&](std::size_t b) {
172 const Index first = static_cast<Index>(b) * B.cols() / blocks;
173 const Index last = (static_cast<Index>(b) + 1) * B.cols() / blocks;
174 C.middleCols(first, last - first).noalias() = A * B.middleCols(first, last - first);
175 });
176 return C;
177}

207void Linalg::get_whitener(MatrixXd& A,
208 bool pca,
209 const std::string& /*ch_type*/,
210 VectorXd& eig,
211 MatrixXd& eigvec)
212{
213 // Eigenvalues come in ascending order; for a symmetric matrix they are the singular values up to sign
214 SelfAdjointEigenSolver<MatrixXd> t_eigenSolver(A);
215 eig = t_eigenSolver.eigenvalues();
216 eigvec = t_eigenSolver.eigenvectors().transpose();
217
218 const double threshold = 1e-8 * eig.cwiseAbs().maxCoeff();
219 const auto rnk = static_cast<std::int32_t>((eig.array().abs() > threshold).count());
220 eig.head(eig.size() - rnk).setZero();
221
222 if (pca) {
223 eigvec = eigvec.bottomRows(rnk).eval(); // eval: resizing eigvec frees the rows being read
224 }
225}

229VectorXi Linalg::intersect(const VectorXi& v1,
230 const VectorXi& v2,
231 VectorXi& idx_sel)
232{
233 std::vector<int> tmp;
234
235 std::vector<std::pair<int, int>> t_vecIntIdxValue;
236
237 for (std::int32_t i = 0; i < v1.size(); ++i)
238 tmp.push_back(v1[i]);
239
240 std::vector<int>::iterator it;
241 for (std::int32_t i = 0; i < v2.size(); ++i) {
242 it = std::search(tmp.begin(), tmp.end(), &v2[i], &v2[i] + 1);
243 if (it != tmp.end())
244 t_vecIntIdxValue.push_back(std::pair<int, int>(v2[i], static_cast<int>(it - tmp.begin())));
245 }
246
247 std::sort(t_vecIntIdxValue.begin(), t_vecIntIdxValue.end(), Linalg::compareIdxValuePairSmallerThan<int>);
248
249 VectorXi p_res(t_vecIntIdxValue.size());
250 idx_sel = VectorXi(t_vecIntIdxValue.size());
251
252 for (std::uint32_t i = 0; i < t_vecIntIdxValue.size(); ++i) {
253 p_res[i] = t_vecIntIdxValue[i].first;
254 idx_sel[i] = t_vecIntIdxValue[i].second;
255 }
256
257 return p_res;
258}

262SparseMatrix<double> Linalg::make_block_diag(const MatrixXd& A,
263 std::int32_t n)
264{
265 const auto ma = static_cast<std::int32_t>(A.rows());
266 const auto na = static_cast<std::int32_t>(A.cols());
267 float bdn = static_cast<float>(na) / n;
268
269 if (bdn != std::floor(bdn)) {
270 return SparseMatrix<double>();
271 }
272
273 typedef Eigen::Triplet<double> T;
274 std::vector<T> tripletList;
275 tripletList.reserve(static_cast<size_t>(bdn * ma * n));
276
277 for (std::int32_t i = 0; i < static_cast<std::int32_t>(bdn); ++i) {
278 std::int32_t current_col = i * n;
279 std::int32_t current_row = i * ma;
280
281 for (std::int32_t r = 0; r < ma; ++r)
282 for (std::int32_t c = 0; c < n; ++c)
283 tripletList.push_back(T(r + current_row, c + current_col, A(r, c + current_col)));
284 }
285
286 SparseMatrix<double> bd(static_cast<int>(std::floor(ma * bdn + 0.5f)), na);
287 bd.setFromTriplets(tripletList.begin(), tripletList.end());
288
289 return bd;
290}

294std::int32_t Linalg::rank(const MatrixXd& A,
295 double tol)
296{
297 JacobiSVD<MatrixXd> t_svdA(A);
298 VectorXd s = t_svdA.singularValues();
299 double t_dMax = s.maxCoeff();
300 t_dMax *= tol;
301 std::int32_t sum = 0;
302 for (std::int32_t i = 0; i < s.size(); ++i)
303 sum += s[i] > t_dMax ? 1 : 0;
304 return sum;
305}
