37RtSensorDataController::RtSensorDataController(QObject* parent)
38: QObject(parent)
39{
40 // Create data worker
41 m_pWorker = new mne::disp::RtSensorDataWorker();
42
43 // Create and configure data thread
44 m_pWorkerThread = new QThread(this);
45 m_pWorker->moveToThread(m_pWorkerThread);
46
47 // Clean up worker when thread finishes
48 connect(m_pWorkerThread, &QThread::finished, m_pWorker, &QObject::deleteLater);
49
50 // Forward worker signals to controller signals
51 connect(m_pWorker, &mne::disp::RtSensorDataWorker::newRtSensorColors,
52 this, &RtSensorDataController::newSensorColorsAvailable);
53 connect(m_pWorker, &mne::disp::RtSensorDataWorker::newRtRawSensorData,
54 this, &RtSensorDataController::newRawSensorDataAvailable);
55
56 // Create timer for driving the streaming cadence
57 m_pTimer = new QTimer(this);
58 m_pTimer->setTimerType(Qt::PreciseTimer);
59 connect(m_pTimer, &QTimer::timeout, m_pWorker, &mne::disp::RtSensorDataWorker::streamData);
60
61 // Start the data worker thread
62 m_pWorkerThread->start();
63
64 // Create interpolation matrix worker
65 m_pInterpWorker = new mne::disp::RtSensorInterpolationMatWorker();
66 m_pInterpThread = new QThread(this);
67 m_pInterpWorker->moveToThread(m_pInterpThread);
68
69 connect(m_pInterpThread, &QThread::finished, m_pInterpWorker, &QObject::deleteLater);
70
71 // Forward interpolation results: auto-apply to data worker + re-emit for external listeners
72 connect(m_pInterpWorker, &mne::disp::RtSensorInterpolationMatWorker::newMegMappingAvailable,
73 this, &RtSensorDataController::onNewMegMapping);
74 connect(m_pInterpWorker, &mne::disp::RtSensorInterpolationMatWorker::newEegMappingAvailable,
75 this, &RtSensorDataController::onNewEegMapping);
76
77 m_pInterpThread->start();
78
79 qDebug() << "RtSensorDataController: Initialized with" << m_iTimeInterval << "ms interval";
80}

295void RtSensorDataController::onNewMegMapping(const QString& surfaceKey,
296 std::shared_ptr<Eigen::MatrixXf> mappingMat,
297 const QVector<int>& pick)
298{
299 // Auto-forward the new matrix to the data worker
300 if (m_pWorker && mappingMat) {
301 m_pWorker->setMappingMatrix(surfaceKey, mappingMat);
302 }
303
304 // Re-emit for external listeners (e.g. BrainView)
305 emit newMegMappingAvailable(surfaceKey, mappingMat, pick);
306
307 qDebug() << "RtSensorDataController: New MEG mapping received and forwarded"
308 << "(" << mappingMat->rows() << "x" << mappingMat->cols() << ")";
309}
