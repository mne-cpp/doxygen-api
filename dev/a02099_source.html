166 schema.category = obj["category"].to_std_string();

206int MnaRegistryLoader::load_file(const std::filesystem::path& path, MnaOpRegistry& registry)
207{
208 const Expected<std::string> text = read_text_file(path);
209 if (!text) {
210 Log::warning("MnaRegistryLoader: Cannot open {}", utf8_from_path(path));
211 return -1;
212 }
213 JsonParseError parseError;
214 const JsonDocument doc = JsonDocument::from_json(*text, &parseError);
215 if (parseError.error != JsonParseError::NoError) {
216 Log::warning("MnaRegistryLoader: JSON parse error in {} : {}", utf8_from_path(path), parseError.error_string());
217 return -1;
218 }
219 const JsonObject& root = doc.object();
220 if (!root.contains("mna_registry_version")) {
221 Log::warning("MnaRegistryLoader: Missing 'mna_registry_version' in {}", utf8_from_path(path));
222 return -1;
223 }
224
225 int count = 0;
226 for (const JsonValue& op : root["ops"].to_array()) {
227 MnaOpSchema schema = parseOp(op.to_object());
228 if (!schema.op_type.empty()) {
229 registry.register_op(schema);
230 ++count;
231 }
232 }
233 Log::info("MnaRegistryLoader: Loaded {} ops from {} (provider: {} )", count, utf8_from_path(path), root["provider"].to_string());
234 return count;
235}

239int MnaRegistryLoader::load_directory(const std::filesystem::path& dir, MnaOpRegistry& registry)
240{
241 int total = 0;
242 std::error_code ec;
243 if (const std::filesystem::path master = dir / "mna-registry.json"; std::filesystem::exists(master, ec)) {
244 total += std::max(load_file(master, registry), 0);
245 }
246
247 std::vector<std::filesystem::path> dropIns;
248 for (const auto& entry : std::filesystem::directory_iterator(dir / "mna-registry.d", ec)) {
249 if (entry.is_regular_file(ec) && entry.path().extension() == ".json") {
250 dropIns.push_back(entry.path());
251 }
252 }
253 std::ranges::sort(dropIns);
254 for (const std::filesystem::path& file : dropIns) {
255 total += std::max(load_file(file, registry), 0);
256 }
257 return total;
258}

262bool MnaRegistryLoader::save_file(const std::filesystem::path& path, std::string_view provider, const MnaOpRegistry& registry)
263{
264 JsonObject root;
265 root["mna_registry_version"] = "1.0";
266 root["provider"] = provider;
267 JsonArray ops;
268 for (const std::string& op : registry.registered_ops()) {
269 ops.push_back(opToJson(registry.schema(op)));
270 }
271 root["ops"] = std::move(ops);
272
273 if (!write_text_file(path, JsonDocument(root).to_json(JsonDocument::Indented))) {
274 Log::warning("MnaRegistryLoader: Cannot write {}", utf8_from_path(path));
275 return false;
276 }
277 return true;
278}
