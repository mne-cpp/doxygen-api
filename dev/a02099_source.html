72int mne_read_meg_comp_eeg_ch_info_32(const QString& name,
73 QList<FIFFLIB::FiffChInfo>& megp, /* MEG channels */
74 int *nmegp,
75 QList<FIFFLIB::FiffChInfo>& meg_compp,
76 int *nmeg_compp,
77 QList<FIFFLIB::FiffChInfo>& eegp, /* EEG channels */
78 int *neegp,
79 FiffCoordTrans *meg_head_t,
80 fiffId *idp) /* The measurement ID */
81/*
82 * Read the channel information and split it into three arrays,
83 * one for MEG, one for MEG compensation channels, and one for EEG
84 */
85{
86 QFile file(name);
87 FiffStream::SPtr stream(new FiffStream(&file));
88
89 QList<FIFFLIB::FiffChInfo> chs;
90 int nchan = 0;
91 QList<FIFFLIB::FiffChInfo> meg;
92 int nmeg = 0;
93 QList<FIFFLIB::FiffChInfo> meg_comp;
94 int nmeg_comp = 0;
95 QList<FIFFLIB::FiffChInfo> eeg;
96 int neeg = 0;
97 std::unique_ptr<FiffId> id;
98 QList<FiffDirNode::SPtr> nodes;
99 FiffDirNode::SPtr info;
100 FiffTag::UPtr t_pTag;
101 FIFFLIB::FiffChInfo this_ch;
102 FiffCoordTrans t;
103 fiff_int_t kind, pos;
104 int j,k,to_find;
105
106 if(!stream->open())
107 goto bad;
108
109 nodes = stream->dirtree()->dir_tree_find(FIFFB_MNE_PARENT_MEAS_FILE);
110
111 if (nodes.size() == 0) {
112 nodes = stream->dirtree()->dir_tree_find(FIFFB_MEAS_INFO);
113 if (nodes.size() == 0) {
114 qCritical ("Could not find the channel information.");
115 goto bad;
116 }
117 }
118 info = nodes[0];
119 to_find = 0;
120 for (k = 0; k < info->nent(); k++) {
121 kind = info->dir[k]->kind;
122 pos = info->dir[k]->pos;
123 switch (kind) {
124 case FIFF_NCHAN :
125 if (!stream->read_tag(t_pTag,pos))
126 goto bad;
127 nchan = *t_pTag->toInt();
128
129 for (j = 0; j < nchan; j++) {
130 chs.append(FiffChInfo());
131 chs[j].scanNo = -1;
132 }
133 to_find = nchan;
134 break;
135
136 case FIFF_PARENT_BLOCK_ID :
137 if(!stream->read_tag(t_pTag, pos))
138 goto bad;
139 id = std::make_unique<FiffId>(*(FiffId*)t_pTag->data());
140 break;
141
142 case FIFF_COORD_TRANS :
143 if(!stream->read_tag(t_pTag, pos))
144 goto bad;
145// t = t_pTag->toCoordTrans();
146 t = FiffCoordTrans::readFromTag( t_pTag );
147 if (t.from != FIFFV_COORD_DEVICE || t.to != FIFFV_COORD_HEAD)
148 t = FiffCoordTrans();
149 break;
150
151 case FIFF_CH_INFO : /* Information about one channel */
152 if(!stream->read_tag(t_pTag, pos))
153 goto bad;
154
155 this_ch = t_pTag->toChInfo();
156 if (this_ch.scanNo <= 0 || this_ch.scanNo > nchan) {
157 printf ("FIFF_CH_INFO : scan # out of range %d (%d)!",this_ch.scanNo,nchan);
158 goto bad;
159 }
160 else
161 chs[this_ch.scanNo-1] = this_ch;
162 to_find--;
163 break;
164 }
165 }
166 if (to_find != 0) {
167 qCritical("Some of the channel information was missing.");
168 goto bad;
169 }
170 if (t.isEmpty() && meg_head_t != nullptr) {
171 /*
172 * Try again in a more general fashion
173 */
174 t = FiffCoordTrans::readMeasTransform(name);
175 if (t.isEmpty()) {
176 qCritical("MEG -> head coordinate transformation not found.");
177 goto bad;
178 }
179 }
180 /*
181 * Sort out the channels
182 */
183 for (k = 0; k < nchan; k++) {
184 if (chs[k].kind == FIFFV_MEG_CH) {
185 meg.append(chs[k]);
186 nmeg++;
187 } else if (chs[k].kind == FIFFV_REF_MEG_CH) {
188 meg_comp.append(chs[k]);
189 nmeg_comp++;
190 } else if (chs[k].kind == FIFFV_EEG_CH) {
191 eeg.append(chs[k]);
192 neeg++;
193 }
194 }
195// fiff_close(in);
196 stream->close();
197
198 megp = meg;
199 if(nmegp) {
200 *nmegp = nmeg;
201 }
202
203 meg_compp = meg_comp;
204 if(nmeg_compp) {
205 *nmeg_compp = nmeg_comp;
206 }
207
208 eegp = eeg;
209 if(neegp) {
210 *neegp = neeg;
211 }
212
213 if (idp == nullptr) {
214 /* id is auto-deleted by unique_ptr */
215 }
216 else
217 *idp = id.release();
218 if (meg_head_t == nullptr) {
219 }
220 else
221 *meg_head_t = t;
222
223 return FIFF_OK;
224
225bad : {
226 stream->close();
227 return FIFF_FAIL;
228 }
229}

260std::unique_ptr<FiffSparseMatrix> mne_convert_to_sparse(const Eigen::MatrixXf& dense, /* The dense matrix to be converted */
261 int stor_type, /* Either FIFFTS_MC_CCS or FIFFTS_MC_RCS */
262 float small) /* How small elements should be ignored? */
263/*
264 * Create the compressed row or column storage sparse matrix representation
265 * including a vector containing the nonzero matrix element values,
266 * the row or column pointer vector and the appropriate index vector(s).
267 */
268{
269 int j,k;
270 int nz;
271 int ptr;
272 int nrow = static_cast<int>(dense.rows());
273 int ncol = static_cast<int>(dense.cols());
274
275 if (small < 0) { /* Automatic scaling */
276 float maxval = dense.cwiseAbs().maxCoeff();
277 if (maxval > 0)
278 small = maxval*std::fabs(small);
279 else
280 small = std::fabs(small);
281 }
282 for (j = 0, nz = 0; j < nrow; j++)
283 for (k = 0; k < ncol; k++) {
284 if (std::fabs(dense(j,k)) > small)
285 nz++;
286 }
287
288 if (nz <= 0) {
289 qWarning("No nonzero elements found.");
290 return nullptr;
291 }
292 if (stor_type != FIFFTS_MC_CCS && stor_type != FIFFTS_MC_RCS) {
293 qWarning("Unknown sparse matrix storage type: %d",stor_type);
294 return nullptr;
295 }
296 auto sparse = std::make_unique<FiffSparseMatrix>();
297 sparse->coding = stor_type;
298 sparse->m = nrow;
299 sparse->n = ncol;
300 sparse->nz = nz;
301 sparse->data.resize(nz);
302 sparse->inds.resize(nz);
303
304 if (stor_type == FIFFTS_MC_RCS) {
305 sparse->ptrs.resize(nrow + 1);
306 for (j = 0, nz = 0; j < nrow; j++) {
307 ptr = -1;
308 for (k = 0; k < ncol; k++)
309 if (std::fabs(dense(j,k)) > small) {
310 sparse->data[nz] = dense(j,k);
311 if (ptr < 0)
312 ptr = nz;
313 sparse->inds[nz++] = k;
314 }
315 sparse->ptrs[j] = ptr;
316 }
317 sparse->ptrs[nrow] = nz;
318 for (j = nrow - 1; j >= 0; j--) /* Take care of the empty rows */
319 if (sparse->ptrs[j] < 0)
320 sparse->ptrs[j] = sparse->ptrs[j+1];
321 }
322 else if (stor_type == FIFFTS_MC_CCS) {
323 sparse->ptrs.resize(ncol + 1);
324 for (k = 0, nz = 0; k < ncol; k++) {
325 ptr = -1;
326 for (j = 0; j < nrow; j++)
327 if (std::fabs(dense(j,k)) > small) {
328 sparse->data[nz] = dense(j,k);
329 if (ptr < 0)
330 ptr = nz;
331 sparse->inds[nz++] = j;
332 }
333 sparse->ptrs[k] = ptr;
334 }
335 sparse->ptrs[ncol] = nz;
336 for (k = ncol-1; k >= 0; k--) /* Take care of the empty columns */
337 if (sparse->ptrs[k] < 0)
338 sparse->ptrs[k] = sparse->ptrs[k+1];
339 }
340 return sparse;
341}

343int mne_sparse_vec_mult2_32(FiffSparseMatrix* mat, /* The sparse matrix */
344 float *vector, /* Vector to be multiplied */
345 float *res) /* Result of the multiplication */
346/*
347 * Multiply a vector by a sparse matrix.
348 */
349{
350 int i,j;
351
352 if (mat->coding == FIFFTS_MC_RCS) {
353 for (i = 0; i < mat->m; i++) {
354 res[i] = 0.0;
355 for (j = mat->ptrs[i]; j < mat->ptrs[i+1]; j++)
356 res[i] += mat->data[j]*vector[mat->inds[j]];
357 }
358 return 0;
359 }
360 else if (mat->coding == FIFFTS_MC_CCS) {
361 for (i = 0; i < mat->m; i++)
362 res[i] = 0.0;
363 for (i = 0; i < mat->n; i++)
364 for (j = mat->ptrs[i]; j < mat->ptrs[i+1]; j++)
365 res[mat->inds[j]] += mat->data[j]*vector[i];
366 return 0;
367 }
368 else {
369 qWarning("mne_sparse_vec_mult2: unknown sparse matrix storage type: %d",mat->coding);
370 return -1;
371 }

418std::unique_ptr<MNECTFCompDataSet> MNECTFCompDataSet::read(const QString &name)
419/*
420 * Read all CTF compensation data from a given file
421 */
422{
423 QFile file(name);
424 FiffStream::SPtr stream(new FiffStream(&file));
425
426 std::unique_ptr<MNECTFCompDataSet> set;
427 QList<FiffDirNode::SPtr> nodes;
428 QList<FiffDirNode::SPtr> comps;
429 int ncomp;
430 int kind,k;
431 FiffTag::UPtr t_pTag;
432 QList<FiffChInfo> chs;
433 int nch = 0;
434 int calibrated;
435 /*
436 * Read the channel information
437 */
438 {
439 QList<FiffChInfo> comp_chs, temp;
440 int ncompch = 0;
441
442 if (mne_read_meg_comp_eeg_ch_info_32(name,
443 chs,
444 &nch,
445 comp_chs,
446 &ncompch,
447 temp,
448 nullptr,
449 nullptr,
450 nullptr) == FAIL)
451 goto bad;
452 if (ncompch > 0) {
453 for (k = 0; k < ncompch; k++)
454 chs.append(comp_chs[k]);
455 nch = nch + ncompch;
456 }
457 }
458 /*
459 * Read the rest of the stuff
460 */
461 if(!stream->open())
462 goto bad;
463 set = std::make_unique<MNECTFCompDataSet>();
464 /*
465 * Locate the compensation data sets
466 */
467 nodes = stream->dirtree()->dir_tree_find(FIFFB_MNE_CTF_COMP);
468 if (nodes.size() == 0)
469 goto good; /* Nothing more to do */
470 comps = nodes[0]->dir_tree_find(FIFFB_MNE_CTF_COMP_DATA);
471 if (comps.size() == 0)
472 goto good;
473 ncomp = comps.size();
474 /*
475 * Set the channel info
476 */
477 set->chs = chs;
478 set->nch = nch;
479 /*
480 * Read each data set
481 */
482 for (k = 0; k < ncomp; k++) {
483 auto mat = MNENamedMatrix::read(stream,comps[k],FIFF_MNE_CTF_COMP_DATA);
484 if (!mat)
485 goto bad;
486 comps[k]->find_tag(stream, FIFF_MNE_CTF_COMP_KIND, t_pTag);
487 if (t_pTag) {
488 kind = *t_pTag->toInt();
489 }
490 else
491 goto bad;
492 comps[k]->find_tag(stream, FIFF_MNE_CTF_COMP_CALIBRATED, t_pTag);
493 if (t_pTag) {
494 calibrated = *t_pTag->toInt();
495 }
496 else
497 calibrated = 0;
498 /*
499 * Add these data to the set
500 */
501 auto one = std::make_unique<MNECTFCompData>();
502 one->data = std::move(mat);
503 one->kind = kind;
504 one->mne_kind = mne_unmap_ctf_comp_kind(one->kind);
505 one->calibrated = calibrated;
506
507 if (one->calibrate(set->chs,set->nch,true) == FAIL) {
508 printf("Warning: Compensation data for '%s' omitted\n", explain_comp(one->kind).toUtf8().constData());
509 }
510 else {
511 set->comps.push_back(std::move(one));
512 set->ncomp++;
513 }
514 }
515#ifdef DEBUG
516 printf("%d CTF compensation data sets read from %s\n",set->ncomp,name);
517#endif
518 goto good;
519
520bad : {
521 stream->close();
522 return nullptr;
523 }
524
525good : {
526 stream->close();
527 return set;
528 }
529}

533int MNECTFCompDataSet::make_comp(const QList<FiffChInfo>& chs,
534 int nch,
535 QList<FiffChInfo> compchs,
536 int ncomp) /* How many of these */
537/*
538 * Make compensation data to apply to a set of channels to yield (or uncompensated) compensated data
539 */
540{
541 Eigen::VectorXi comps;
542 int need_comp;
543 int first_comp;
544 MNECTFCompData* this_comp;
545 Eigen::VectorXi comp_sel;
546 QStringList names;
547 QString name;
548 int j,k,p;
549
550 std::unique_ptr<FiffSparseMatrix> presel;
551 std::unique_ptr<FiffSparseMatrix> postsel;
552 std::unique_ptr<MNENamedMatrix> data;
553
554 QStringList emptyList;
555
556 if (compchs.isEmpty()) {
557 compchs = chs;
558 ncomp = nch;
559 }
560 printf("Setting up compensation data...\n");
561 if (nch == 0)
562 return OK;
563 current.reset();
564 comps.resize(nch);
565 for (k = 0, need_comp = 0, first_comp = MNE_CTFV_COMP_NONE; k < nch; k++) {
566 if (chs[k].kind == FIFFV_MEG_CH) {
567 comps[k] = chs[k].chpos.coil_type >> 16;
568 if (comps[k] != MNE_CTFV_COMP_NONE) {
569 if (first_comp == MNE_CTFV_COMP_NONE)
570 first_comp = comps[k];
571 else {
572 if (comps[k] != first_comp) {
573 printf("We do not support nonuniform compensation yet.");
574 return FAIL;
575 }
576 }
577 need_comp++;
578 }
579 }
580 else
581 comps[k] = MNE_CTFV_COMP_NONE;
582 }
583 if (need_comp == 0) {
584 printf("\tNo compensation set. Nothing more to do.\n");
585 return OK;
586 }
587 printf("\t%d out of %d channels have the compensation set.\n",need_comp,nch);
588 /*
589 * Find the desired compensation data matrix
590 */
591 for (k = 0, this_comp = nullptr; k < this->ncomp; k++) {
592 if (this->comps[k]->mne_kind == first_comp) {
593 this_comp = this->comps[k].get();
594 break;
595 }
596 }
597 if (!this_comp) {
598 printf("Did not find the desired compensation data : %s",
599 explain_comp(map_comp_kind(first_comp)).toUtf8().constData());
600 return FAIL;
601 }
602 printf("\tDesired compensation data (%s) found.\n",explain_comp(map_comp_kind(first_comp)).toUtf8().constData());
603 /*
604 * Find the compensation channels
605 */
606 comp_sel.resize(this_comp->data->ncol);
607 for (k = 0; k < this_comp->data->ncol; k++) {
608 comp_sel[k] = -1;
609 name = this_comp->data->collist[k];
610 for (p = 0; p < ncomp; p++)
611 if (QString::compare(name,compchs[p].ch_name) == 0) {
612 comp_sel[k] = p;
613 break;
614 }
615 if (comp_sel[k] < 0) {
616 printf("Compensation channel %s not found",name.toUtf8().constData());
617 return FAIL;
618 }
619 }
620 printf("\tAll compensation channels found.\n");
621 /*
622 * Create the preselector
623 */
624 {
625 Eigen::MatrixXf sel = Eigen::MatrixXf::Zero(this_comp->data->ncol, ncomp);
626 for (j = 0; j < this_comp->data->ncol; j++)
627 sel(j, comp_sel[j]) = 1.0f;
628 presel = mne_convert_to_sparse(sel, FIFFTS_MC_RCS, 1e-30f);
629 if (!presel)
630 return FAIL;
631 printf("\tPreselector created.\n");
632 }
633 /*
634 * Pick the desired channels
635 */
636 for (k = 0; k < nch; k++) {
637 if (comps[k] != MNE_CTFV_COMP_NONE)
638 names.append(chs[k].ch_name);
639 }
640
641 {
642 auto d = this_comp->data->pick(names, need_comp, emptyList, 0);
643 if (!d)
644 return FAIL;
645 data = std::move(d);
646 }
647 printf("\tCompensation data matrix created.\n");
648 /*
649 * Create the postselector
650 */
651 {
652 Eigen::MatrixXf sel = Eigen::MatrixXf::Zero(nch, data->nrow);
653 for (j = 0, p = 0; j < nch; j++) {
654 if (comps[j] != MNE_CTFV_COMP_NONE)
655 sel(j, p++) = 1.0f;
656 }
657 postsel = mne_convert_to_sparse(sel, FIFFTS_MC_RCS, 1e-30f);
658 if (!postsel)
659 return FAIL;
660 printf("\tPostselector created.\n");
661 }
662 current = std::make_unique<MNECTFCompData>();
663 current->kind = this_comp->kind;
664 current->mne_kind = this_comp->mne_kind;
665 current->data = std::move(data);
666 current->presel = std::move(presel);
667 current->postsel = std::move(postsel);
668
669 printf("\tCompensation set up.\n");
670 return OK;
671}

704int MNECTFCompDataSet::apply(bool do_it, Eigen::Ref<Eigen::VectorXf> data, Eigen::Ref<const Eigen::VectorXf> compdata)
705/*
706 * Apply compensation or revert to uncompensated data
707 */
708{
709 MNECTFCompData* this_comp;
710 int ndata = static_cast<int>(data.size());
711 int ncompdata = static_cast<int>(compdata.size());
712
713 if (!current)
714 return OK;
715 this_comp = current.get();
716 /*
717 * Dimension checks
718 */
719 if (this_comp->presel) {
720 if (this_comp->presel->n != ncompdata) {
721 printf("Compensation data dimension mismatch. Expected %d, got %d channels.",
722 this_comp->presel->n,ncompdata);
723 return FAIL;
724 }
725 }
726 else if (this_comp->data->ncol != ncompdata) {
727 printf("Compensation data dimension mismatch. Expected %d, got %d channels.",
728 this_comp->data->ncol,ncompdata);
729 return FAIL;
730 }
731 if (this_comp->postsel) {
732 if (this_comp->postsel->m != ndata) {
733 printf("Data dimension mismatch. Expected %d, got %d channels.",
734 this_comp->postsel->m,ndata);
735 return FAIL;
736 }
737 }
738 else if (this_comp->data->nrow != ndata) {
739 printf("Data dimension mismatch. Expected %d, got %d channels.",
740 this_comp->data->nrow,ndata);
741 return FAIL;
742 }
743 /*
744 * Preselection is optional
745 */
746 const float *presel;
747 if (this_comp->presel) {
748 if (this_comp->presel_data.size() == 0)
749 this_comp->presel_data.resize(this_comp->presel->m);
750 if (mne_sparse_vec_mult2_32(this_comp->presel.get(),const_cast<float*>(compdata.data()),this_comp->presel_data.data()) != OK)
751 return FAIL;
752 presel = this_comp->presel_data.data();
753 }
754 else
755 presel = compdata.data();
756 /*
757 * This always happens
758 */
759 if (this_comp->comp_data.size() == 0)
760 this_comp->comp_data.resize(this_comp->data->nrow);
761 {
762 Eigen::Map<const Eigen::VectorXf> preselVec(presel, this_comp->data->ncol);
763 Eigen::Map<Eigen::VectorXf> compVec(this_comp->comp_data.data(), this_comp->data->nrow);
764 compVec = this_comp->data->data * preselVec;
765 }
766 /*
767 * Optional postselection
768 */
769 const float *comp;
770 if (!this_comp->postsel)
771 comp = this_comp->comp_data.data();
772 else {
773 if (this_comp->postsel_data.size() == 0)
774 this_comp->postsel_data.resize(this_comp->postsel->m);
775 if (mne_sparse_vec_mult2_32(this_comp->postsel.get(),this_comp->comp_data.data(),this_comp->postsel_data.data()) != OK)
776 return FAIL;
777 comp = this_comp->postsel_data.data();
778 }
779 /*
780 * Compensate or revert compensation?
781 */
782 Eigen::Map<const Eigen::VectorXf> compVec(comp, ndata);
783 if (do_it)
784 data -= compVec;
785 else
786 data += compVec;
787 return OK;
788}

791
792int MNECTFCompDataSet::apply_transpose(bool do_it, Eigen::MatrixXf& data)
793/*
794 * Apply compensation or revert to uncompensated data
795 */
796{
797 using RowMatrixXf = Eigen::Matrix<float, Eigen::Dynamic, Eigen::Dynamic, Eigen::RowMajor>;
798
799 MNECTFCompData* this_comp;
800 int ndata = static_cast<int>(data.rows());
801 int ns = static_cast<int>(data.cols());
802 int ncompdata = ndata;
803
804 if (!current)
805 return OK;
806 this_comp = current.get();
807 /*
808 * Dimension checks
809 */
810 if (this_comp->presel) {
811 if (this_comp->presel->n != ncompdata) {
812 printf("Compensation data dimension mismatch. Expected %d, got %d channels.",
813 this_comp->presel->n,ncompdata);
814 return FAIL;
815 }
816 }
817 else if (this_comp->data->ncol != ncompdata) {
818 printf("Compensation data dimension mismatch. Expected %d, got %d channels.",
819 this_comp->data->ncol,ncompdata);
820 return FAIL;
821 }
822 if (this_comp->postsel) {
823 if (this_comp->postsel->m != ndata) {
824 printf("Data dimension mismatch. Expected %d, got %d channels.",
825 this_comp->postsel->m,ndata);
826 return FAIL;
827 }
828 }
829 else if (this_comp->data->nrow != ndata) {
830 printf("Data dimension mismatch. Expected %d, got %d channels.",
831 this_comp->data->nrow,ndata);
832 return FAIL;
833 }
834 /*
835 * Preselection is optional — sparse matrix * data
836 */
837 Eigen::MatrixXf preselMat;
838 if (this_comp->presel) {
839 preselMat.resize(this_comp->presel->m, ns);
840 FiffSparseMatrix* sp = this_comp->presel.get();
841 if (sp->coding == FIFFTS_MC_RCS) {
842 for (int i = 0; i < sp->m; i++)
843 for (int c = 0; c < ns; c++) {
844 float val = 0.0f;
845 for (int j = sp->ptrs[i]; j < sp->ptrs[i+1]; j++)
846 val += sp->data[j] * data(sp->inds[j], c);
847 preselMat(i, c) = val;
848 }
849 } else if (sp->coding == FIFFTS_MC_CCS) {
850 preselMat.setZero();
851 for (int c = 0; c < ns; c++)
852 for (int i = 0; i < sp->n; i++)
853 for (int j = sp->ptrs[i]; j < sp->ptrs[i+1]; j++)
854 preselMat(sp->inds[j], c) += sp->data[j] * data(i, c);
855 } else {
856 printf("Unknown sparse matrix storage type: %d", sp->coding);
857 return FAIL;
858 }
859 }
860 else {
861 preselMat = data;
862 }
863 /*
864 * Compensation: comp = data_matrix * preselMat
865 * data_matrix is float** (contiguous row-major via mne_cmatrix)
866 */
867 Eigen::MatrixXf comp = this_comp->data->data * preselMat;
868 /*
869 * Optional postselection — sparse matrix * comp
870 */
871 if (this_comp->postsel) {
872 Eigen::MatrixXf postselMat(this_comp->postsel->m, ns);
873 FiffSparseMatrix* sp = this_comp->postsel.get();
874 if (sp->coding == FIFFTS_MC_RCS) {
875 for (int i = 0; i < sp->m; i++)
876 for (int c = 0; c < ns; c++) {
877 float val = 0.0f;
878 for (int j = sp->ptrs[i]; j < sp->ptrs[i+1]; j++)
879 val += sp->data[j] * comp(sp->inds[j], c);
880 postselMat(i, c) = val;
881 }
882 } else if (sp->coding == FIFFTS_MC_CCS) {
883 postselMat.setZero();
884 for (int c = 0; c < ns; c++)
885 for (int i = 0; i < sp->n; i++)
886 for (int j = sp->ptrs[i]; j < sp->ptrs[i+1]; j++)
887 postselMat(sp->inds[j], c) += sp->data[j] * comp(i, c);
888 } else {
889 printf("Unknown sparse matrix storage type: %d", sp->coding);
890 return FAIL;
891 }
892 comp = std::move(postselMat);
893 }
894 /*
895 * Compensate or revert compensation?
896 */
897 if (do_it)
898 data -= comp;
899 else
900 data += comp;
901 return OK;
902}
