480inline qint32 FiffCoordTrans::storageSize()
481{
482 // On-disk layout: from, to, rot[3][3], move[3], invrot[3][3], invmove[3]
483 // (C++ class uses Matrix4f but on-disk stores 3x3 rot + 3-vec separately)
484 return 2 * sizeof(fiff_int_t) + 2 * 12 * sizeof(fiff_float_t);
485}

491 return (a.from == b.from &&
492 a.to == b.to &&
493 a.trans.isApprox(b.trans, 0.0001f) &&
