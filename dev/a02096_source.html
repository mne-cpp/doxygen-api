61 return value.to_bool() ? 1.0 : 0.0;

70class Evaluator
71{
72public:
73 using Lookup = std::function<JsonValue(std::string_view)>;
74
75 Evaluator(std::string_view text, Lookup lookup)
76 : m_text(text)
77 , m_lookup(std::move(lookup))
78 {
79 }
80
81 JsonValue run()
82 {
83 JsonValue value = comparison();
84 skipSpace();
85 return m_pos == m_text.size() ? value : JsonValue(JsonValue::Undefined);
86 }
87
88private:
89 static bool isNameStart(char c)
90 {
91 return std::isalpha(static_cast<unsigned char>(c)) != 0 || c == '_';
92 }
93 static bool isNameChar(char c)
94 {
95 return std::isalnum(static_cast<unsigned char>(c)) != 0 || c == '_' || c == '.' || c == ':' || c == '/';
96 }
97
98 void skipSpace()
99 {
100 while (m_pos < m_text.size() && std::isspace(static_cast<unsigned char>(m_text[m_pos])) != 0) {
101 ++m_pos;
102 }
103 }
104
105 bool accept(std::string_view token)
106 {
107 skipSpace();
108 if (m_text.substr(m_pos).starts_with(token)) {
109 m_pos += token.size();
110 return true;
111 }
112 return false;
113 }
114
115 JsonValue comparison()
116 {
117 JsonValue left = sum();
118 for (const std::string_view op : {">=", "<=", "==", "!=", ">", "<"}) {
119 if (!accept(op)) {
120 continue;
121 }
122 const JsonValue right = sum();
123 if (left.is_undefined() || right.is_undefined()) {
124 return JsonValue(JsonValue::Undefined);
125 }
126 if (left.is_string() && right.is_string() && (op == "==" || op == "!=")) {
127 return (left.to_string() == right.to_string()) == (op == "==");
128 }
129 const std::optional<double> a = toNumber(left);
130 const std::optional<double> b = toNumber(right);
131 if (!a || !b) {
132 return JsonValue(JsonValue::Undefined);
133 }
134 return op == ">=" ? *a >= *b : op == "<=" ? *a <= *b
135 : op == "==" ? *a == *b
136 : op == "!=" ? *a != *b
137 : op == ">" ? *a > *b
138 : *a < *b;
139 }
140 return left;
141 }
142
143 JsonValue sum()
144 {
145 JsonValue value = product();
146 for (;;) {
147 const bool plus = accept("+");
148 if (!plus && !accept("-")) {
149 return value;
150 }
151 value = arithmetic(value, product(), plus ? '+' : '-');
152 }
153 }
154
155 JsonValue product()
156 {
157 JsonValue value = unary();
158 for (;;) {
159 const bool times = accept("*");
160 if (!times && !accept("/")) {
161 return value;
162 }
163 value = arithmetic(value, unary(), times ? '*' : '/');
164 }
165 }
166
167 JsonValue unary()
168 {
169 if (accept("-")) {
170 const std::optional<double> v = toNumber(unary());
171 return v ? JsonValue(-*v) : JsonValue(JsonValue::Undefined);
172 }
173 return primary();
174 }
175
176 static JsonValue arithmetic(const JsonValue& left, const JsonValue& right, char op)
177 {
178 const std::optional<double> a = toNumber(left);
179 const std::optional<double> b = toNumber(right);
180 if (left.is_undefined() || right.is_undefined() || !a || !b) {
181 return JsonValue(JsonValue::Undefined);
182 }
183 switch (op) {
184 case '+':
185 return *a + *b;
186 case '-':
187 return *a - *b;
188 case '*':
189 return *a * *b;
190 default:
191 return std::abs(*b) < 1e-15 ? JsonValue(JsonValue::Undefined) : JsonValue(*a / *b);
192 }
193 }
194
195 JsonValue primary()
196 {
197 skipSpace();
198 if (m_pos >= m_text.size()) {
199 return JsonValue(JsonValue::Undefined);
200 }
201 const char c = m_text[m_pos];
202 if (c == '(') {
203 ++m_pos;
204 JsonValue value = comparison();
205 return accept(")") ? value : JsonValue(JsonValue::Undefined);
206 }
207 if (c == '\'') {
208 const std::size_t end = m_text.find('\'', m_pos + 1);
209 if (end == std::string_view::npos) {
210 return JsonValue(JsonValue::Undefined);
211 }
212 JsonValue text(m_text.substr(m_pos + 1, end - m_pos - 1));
213 m_pos = end + 1;
214 return text;
215 }
216 if (std::isdigit(static_cast<unsigned char>(c)) != 0 || c == '.') {
217 double number = 0.0;
218 const auto [next, ec] = std::from_chars(m_text.data() + m_pos, m_text.data() + m_text.size(), number);
219 if (ec != std::errc{}) {
220 return JsonValue(JsonValue::Undefined);
221 }
222 m_pos = static_cast<std::size_t>(next - m_text.data());
223 return number;
224 }
225 if (!isNameStart(c)) {
226 return JsonValue(JsonValue::Undefined);
227 }
228 const std::size_t start = m_pos;
229 while (m_pos < m_text.size() && isNameChar(m_text[m_pos])) {
230 ++m_pos;
231 }
232 const std::string_view name = m_text.substr(start, m_pos - start);
233 if (!accept("(")) {
234 if (name == "true" || name == "false") {
235 return name == "true";
236 }
237 return m_lookup(name);
238 }
239 std::vector<JsonValue> args;
240 if (!accept(")")) {
241 do {
242 args.push_back(comparison());
243 } while (accept(","));
244 if (!accept(")")) {
245 return JsonValue(JsonValue::Undefined);
246 }
247 }
248 return call(name, args);
249 }
250
251 JsonValue call(std::string_view name, const std::vector<JsonValue>& args) const
252 {
253 if (name == "ref") {
254 return args.size() == 1 && args[0].is_string() ? m_lookup(args[0].to_string()) : JsonValue(JsonValue::Undefined);
255 }
256 std::vector<double> x;
257 for (const JsonValue& arg : args) {
258 const std::optional<double> v = toNumber(arg);
259 if (arg.is_undefined() || !v) {
260 return JsonValue(JsonValue::Undefined);
261 }
262 x.push_back(*v);
263 }
264 if (name == "clamp" && x.size() == 3) {
265 return std::clamp(x[0], x[1], x[2]);
266 }
267 if (name == "scale" && x.size() == 2) {
268 return x[0] * x[1];
269 }
270 if (name == "threshold" && x.size() == 4) {
271 return x[0] > x[1] ? x[2] : x[3];
272 }
273 if (name == "lerp" && x.size() == 3) {
274 return x[0] + ((x[1] - x[0]) * x[2]);
275 }
276 return JsonValue(JsonValue::Undefined);
277 }
278
279 std::string_view m_text;
280 std::size_t m_pos = 0;
281 Lookup m_lookup;
282};

359StringList MnaParamTree::evaluate(const JsonObject& results)
360{
361 StringList changed;
362 for (const auto& [path, binding] : m_bindings) {
363 if (binding.trigger == "manual") { // manual bindings need explicit invocation
364 continue;
365 }
366 JsonValue value = evaluate_expression(binding.expression, results);
367 if (value.is_undefined() || param(path) == value) {
368 continue;
369 }
370 m_params.insert_or_assign(path, std::move(value));
371 changed.push_back(path);
372 }
373 return changed;
374}

390{

394 bindingList.push_back(binding.to_json());
395 }
396 json["bindings"] = std::move(bindingList);
397
398 JsonObject params; // only parameters without a binding; bound ones are recomputed
399 for (const auto& [path, value] : m_params) {
400 if (!m_bindings.contains(path)) {
401 params.insert(path, value);
402 }
403 }
404 if (!params.empty()) {
405 json["parameters"] = std::move(params);
406 }
407 return json;
