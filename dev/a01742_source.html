62ApplyToPanel::ApplyToPanel()
63: Panel("ApplyToView", "Apply to View")
64{
65 set_width(330);
66 SettingsForm& f = form();
67 for (const ViewSpec& view : kViews) {
68 SettingField field;
69 field.key = view.key;
70 field.kind = SettingField::Kind::Bool;
71 field.label = view.label;
72 field.default_value = true;
73 field.group = "Apply to View";
74 field.persistent = false;
75 f.add(std::move(field));
76 }
77 f.add_action(kAll, "Select All", "Apply to View");
78 f.add_action(kClear, "Clear Selected", "Apply to View");
79 m_listener = f.add_listener([this](const std::string& key, const std::optional<SettingValue>& value) {
80 if (value) {
81 return;
82 }
83 if (key == kAll) {
84 select_all();
85 } else if (key == kClear) {
86 clear();
87 }
88 });
89}
