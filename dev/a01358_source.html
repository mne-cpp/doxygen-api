163std::shared_ptr<Eigen::SparseMatrix<float>> RtSourceInterpolationMatWorker::computeHemi(
164 const Eigen::MatrixX3f& matVertices,
165 const std::vector<Eigen::VectorXi>& vecNeighborVertices,
166 Eigen::VectorXi vecSourceVertices,
167 double dCancelDist,
168 double (*interpFunc)(double))
169{
170 if (matVertices.rows() == 0 || vecSourceVertices.size() == 0) {
171 return std::shared_ptr<Eigen::SparseMatrix<float>>();
172 }
173
174 // Compute surface-constrained distances (SCDC / geodesic)
175 std::shared_ptr<Eigen::MatrixXd> distTable = GeometryInfo::scdc(
176 matVertices,
177 vecNeighborVertices,
178 vecSourceVertices,
179 dCancelDist);
180
181 if (!distTable || distTable->rows() == 0) {
182 mne::Log::warning("RtSourceInterpolationMatWorker: SCDC computation failed.");
183 return std::shared_ptr<Eigen::SparseMatrix<float>>();
184 }
185
186 // Create interpolation matrix
187 auto interpMat = Interpolation::createInterpolationMat(
188 vecSourceVertices,
189 distTable,
190 interpFunc,
191 dCancelDist);
192
193 return interpMat;
194}

198void RtSourceInterpolationMatWorker::computeInterpolationMatrix()
199{
200 // ── Snapshot parameters under lock ─────────────────────────────────
201 Eigen::MatrixX3f vertsLh, vertsRh;
202 std::vector<Eigen::VectorXi> neighborsLh, neighborsRh;
203 Eigen::VectorXi srcVertsLh, srcVertsRh;
204 bool hasLh, hasRh;
205 double cancelDist;
206 std::string interpFuncName;
207 int visType;
208
209 // FsAnnotation data snapshots
210 std::vector<mne::FsLabel> labelsLh, labelsRh;
211 std::map<std::int32_t, std::int32_t> mapLabelIdSrcLh, mapLabelIdSrcRh;
212 std::vector<int> vertNosLh, vertNosRh;
213 bool annotLhInit, annotRhInit;
214 int nAnnotVertsLh, nAnnotVertsRh;
215
216 {
217 std::scoped_lock locker(m_mutex);
218 vertsLh = m_matVerticesLh;
219 vertsRh = m_matVerticesRh;
220 neighborsLh = m_vecNeighborsLh;
221 neighborsRh = m_vecNeighborsRh;
222 srcVertsLh = m_vecSourceVerticesLh;
223 srcVertsRh = m_vecSourceVerticesRh;
224 hasLh = m_hasLh;
225 hasRh = m_hasRh;
226 cancelDist = m_dCancelDist;
227 interpFuncName = m_sInterpolationFunction;
228 visType = m_iVisualizationType;
229
230 labelsLh = m_lLabelsLh;
231 labelsRh = m_lLabelsRh;
232 mapLabelIdSrcLh = m_mapLabelIdSourcesLh;
233 mapLabelIdSrcRh = m_mapLabelIdSourcesRh;
234 vertNosLh = m_vertNosLh;
235 vertNosRh = m_vertNosRh;
236 annotLhInit = m_bAnnotationLhInit;
237 annotRhInit = m_bAnnotationRhInit;
238 nAnnotVertsLh = m_nAnnotationVerticesLh;
239 nAnnotVertsRh = m_nAnnotationVerticesRh;
240 }
241
242 // ── FsAnnotation-based mode ──────────────────────────────────────────
243 if (visType == AnnotationBased) {
244 mne::Log::debug("RtSourceInterpolationMatWorker: Computing annotation matrices...");
245
246 if (annotLhInit) {
247 auto mat = computeAnnotationOperator(labelsLh, mapLabelIdSrcLh, vertNosLh, nAnnotVertsLh);
248 if (mat && mat->rows() > 0) {
249 mne::Log::debug("RtSourceInterpolationMatWorker: LH annotation matrix: {} x {}", mat->rows(), mat->cols());
250 if (m_newInterpolationMatrixLeftAvailableCallback) {
251 m_newInterpolationMatrixLeftAvailableCallback(mat);
252 }
253 }
254 }
255
256 if (annotRhInit) {
257 auto mat = computeAnnotationOperator(labelsRh, mapLabelIdSrcRh, vertNosRh, nAnnotVertsRh);
258 if (mat && mat->rows() > 0) {
259 mne::Log::debug("RtSourceInterpolationMatWorker: RH annotation matrix: {} x {}", mat->rows(), mat->cols());
260 if (m_newInterpolationMatrixRightAvailableCallback) {
261 m_newInterpolationMatrixRightAvailableCallback(mat);
262 }
263 }
264 }
265
266 mne::Log::debug("RtSourceInterpolationMatWorker: FsAnnotation matrix computation complete.");
267 return;
268 }
269
270 // ── Interpolation-based mode (default) ─────────────────────────────
271 if (!hasLh && !hasRh) {
272 mne::Log::debug("RtSourceInterpolationMatWorker: No hemisphere data set, skipping.");
273 return;
274 }
275
276 auto interpFunc = resolveInterpolationFunction(interpFuncName);
277
278 mne::Log::debug("RtSourceInterpolationMatWorker: Computing interpolation matrices (cancelDist={}, func={})...", cancelDist, interpFuncName);
279
280 // ── LH ─────────────────────────────────────────────────────────────
281 if (hasLh) {
282 mne::Log::debug("RtSourceInterpolationMatWorker: Computing LH matrix ({} verts, {} sources)...", vertsLh.rows(), srcVertsLh.size());
283
284 auto mat = computeHemi(vertsLh, neighborsLh, srcVertsLh, cancelDist, interpFunc);
285
286 if (mat && mat->rows() > 0) {
287 mne::Log::debug("RtSourceInterpolationMatWorker: LH matrix computed: {} x {}", mat->rows(), mat->cols());
288 if (m_newInterpolationMatrixLeftAvailableCallback) {
289 m_newInterpolationMatrixLeftAvailableCallback(mat);
290 }
291 } else {
292 mne::Log::warning("RtSourceInterpolationMatWorker: LH interpolation matrix computation failed.");
293 }
294 }
295
296 // ── RH ─────────────────────────────────────────────────────────────
297 if (hasRh) {
298 mne::Log::debug("RtSourceInterpolationMatWorker: Computing RH matrix ({} verts, {} sources)...", vertsRh.rows(), srcVertsRh.size());
299
300 auto mat = computeHemi(vertsRh, neighborsRh, srcVertsRh, cancelDist, interpFunc);
301
302 if (mat && mat->rows() > 0) {
303 mne::Log::debug("RtSourceInterpolationMatWorker: RH matrix computed: {} x {}", mat->rows(), mat->cols());
304 if (m_newInterpolationMatrixRightAvailableCallback) {
305 m_newInterpolationMatrixRightAvailableCallback(mat);
306 }
307 } else {
308 mne::Log::warning("RtSourceInterpolationMatWorker: RH interpolation matrix computation failed.");
309 }
310 }
311
312 mne::Log::debug("RtSourceInterpolationMatWorker: Interpolation matrix computation complete.");
313}

317std::shared_ptr<Eigen::SparseMatrix<float>> RtSourceInterpolationMatWorker::computeAnnotationOperator(
318 const std::vector<mne::FsLabel>& lLabels,
319 const std::map<std::int32_t, std::int32_t>& mapLabelIdSrc,
320 const std::vector<int>& vertNos,
321 int nVertices)
322{
323 if (lLabels.empty() || vertNos.empty() || nVertices <= 0) {
324 return std::shared_ptr<Eigen::SparseMatrix<float>>();
325 }
326
327 // One row per surface vertex: labels need not cover the surface, and their vertex indices are surface indices
328 auto mat = std::make_shared<Eigen::SparseMatrix<float>>(nVertices, static_cast<Eigen::Index>(vertNos.size()));
329
330 // For each label: assign uniform weight to all its vertices from sources in that label
331 for (int i = 0; i < std::ssize(lLabels); ++i) {
332 const mne::FsLabel& label = lLabels.at(i);
333 std::vector<std::int32_t> listSourcesVertNoLabel;
334 for (const auto& [vertNo, labelId] : mapLabelIdSrc) {
335 if (labelId == label.label_id) {
336 listSourcesVertNoLabel.push_back(vertNo);
337 }
338 }
339
340 for (int j = 0; j < label.vertices.rows(); ++j) {
341 for (const std::int32_t sourceVertNo : listSourcesVertNoLabel) {
342 const auto col = std::ranges::find(vertNos, sourceVertNo);
343 const int colIdx = col == vertNos.end() ? -1 : static_cast<int>(col - vertNos.begin());
344 if (colIdx >= 0 && label.vertices(j) >= 0 && label.vertices(j) < nVertices) {
345 mat->coeffRef(label.vertices(j), colIdx) = 1.0f / static_cast<float>(listSourcesVertNoLabel.size());
346 }
347 }
348 }
349 }
350
351 return mat;
352}
