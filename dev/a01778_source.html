208bool ChannelSelectionPanel::load_layout(const std::string& file)
209{
210 set_field(kLayoutFile, file);
211 const std::optional<mne::MNELayout> layout = file.empty() ? std::nullopt : mne::MNELayout::read(resolve(m_layouts, file));
212 m_layout.clear();
213 if (layout) {
214 for (const auto& [name, position] : layout->channelPositions()) {
215 m_layout.insert_or_assign(name, Point{position.x(), position.y()});
216 }
217 }
218
219 // EEG channels the file does not have are laid out from their digitiser points
220 std::vector<Eigen::Vector3d> positions;
221 std::vector<std::string> names;
222 const int rows = m_table ? m_table->row_count() : 0;
223 for (int row = 0; row < rows; ++row) {
224 if (m_table->info()->chs[static_cast<std::size_t>(row)].kind == FIFFV_EEG_CH) {
225 positions.push_back(m_table->digitizer_cm(row).cast<double>());
226 names.push_back(m_table->text(row, ChannelTable::Column::OriginalName));
227 }
228 }
229 if (!positions.empty()) {
230 if (const auto corners = mne::LayoutMaker::make_layout(positions, {.radius = 60.0, .box_width = 5.0, .box_height = 4.0, .mirror_y = true})) {
231 for (std::size_t i = 0; i < corners->size() && i < names.size(); ++i) {
232 m_layout.try_emplace(names[i], Point{(*corners)[i].x(), (*corners)[i].y()});
233 }
234 }
235 }
236
237 update_bad_channels();
238 if (layout && m_callbacks.layout_loaded) {
239 m_callbacks.layout_loaded(m_layout);
240 }
241 return layout.has_value();
242}

253bool ChannelSelectionPanel::load_selection_groups(const std::string& file)
254{
255 set_field(kGroupFile, file);
256 const std::filesystem::path path = resolve(m_selections, file);
257 std::optional<std::vector<mne::ChannelSelection>> read;
258 if (!file.empty()) {
259 if (has_extension(path, ".sel")) {
260 read = mne::SelectionIO::read_sel(path);
261 }
262 if (has_extension(path, ".mon")) {
263 if (auto group = mne::SelectionIO::read_mon(path)) {
264 read = std::vector<mne::ChannelSelection>{std::move(*group)};
265 }
266 }
267 }
268 m_groups.clear();
269 if (read) {
270 for (mne::ChannelSelection& group : *read) {
271 m_groups.insert_or_assign(group.name, std::move(group.channels));
272 }
273 }
274 show_groups();
275 return read.has_value();
276}

280bool ChannelSelectionPanel::save_selection_groups(const std::filesystem::path& path) const
281{
282 std::vector<mne::ChannelSelection> groups;
283 for (const auto& [name, channels] : m_groups) {
284 if (name != kAll && name != kAllEeg) {
285 groups.push_back(mne::ChannelSelection{name, channels});
286 }
287 }
288 if (has_extension(path, ".sel")) {
289 return mne::SelectionIO::write_sel(path, groups);
290 }
291 if (has_extension(path, ".mon")) {
292 return mne::SelectionIO::write_mon(path.parent_path(), groups);
293 }
294 return false;
295}

341void ChannelSelectionPanel::update_bad_channels()
342{
343 std::vector<std::string> bad;
344 if (m_table && form().get<bool>(kShowBad)) {
345 const StringList badNames = m_table->bad_channels();
346 for (int row = 0; row < m_table->row_count(); ++row) {
347 if (std::ranges::contains(badNames, m_table->text(row, ChannelTable::Column::OriginalName))) {
348 bad.push_back(m_table->text(row, ChannelTable::Column::MappedLayoutName));
349 }
350 }
351 }
352 m_sensors->set_layout(m_layout, bad);
353 m_sensors->set_selectable(m_channelList->items());
354 update_data_view();
355}

411void ChannelSelectionPanel::show_groups()
412{
413 // "All" depends on the measurement and "All EEG" on its channels, not on the group file
414 m_groups.insert_or_assign(kAll, m_mapped);

421 const bool updating = std::exchange(m_updating, true);
422 m_groupList->set_items(std::move(names));
423 m_updating = updating;

425 m_groupList->set_selected({m_groupList->find(kAll)});
