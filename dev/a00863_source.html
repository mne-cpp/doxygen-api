53{
54 return static_cast<std::size_t>(std::clamp(n / 4096, 1, 64));

57// Centroid, normal (unit length if normalize) and area of the first n triangles, in parallel chunks
58void triangle_geometry(const MNESurfaceOrVolume::PointsT& rr,
59 const MNESurfaceOrVolume::TrianglesT& tris,
60 int n,
61 MatrixX3d& cent,
62 MatrixX3d& nn,
63 VectorXd& area,
64 bool normalize)
65{
66 cent = MatrixX3d::Zero(n, 3);
67 nn = MatrixX3d::Zero(n, 3);
68 area = VectorXd::Zero(n);
69 const std::size_t chunks = chunk_count(n);
70 ThreadPool::global().parallel_for(0, chunks, [&](std::size_t chunk) {
71 const int first = static_cast<int>(chunk * static_cast<std::size_t>(n) / chunks);
72 const int last = static_cast<int>((chunk + 1) * static_cast<std::size_t>(n) / chunks);
73 Matrix3d r;
74 Vector3d a, b;
75 for (int i = first; i < last; ++i) {
76 for (int j = 0; j < 3; ++j) {
77 const int k = tris(i, j);
78 r(j, 0) = rr(k, 0);
79 r(j, 1) = rr(k, 1);
80 r(j, 2) = rr(k, 2);
81 cent(i, 0) += rr(k, 0);
82 cent(i, 1) += rr(k, 1);
83 cent(i, 2) += rr(k, 2);
84 }
85 cent.row(i) /= 3.0f;
86
87 // cross((r2 - r1), (r3 - r1))
88 a = (r.row(1) - r.row(0)).transpose();
89 b = (r.row(2) - r.row(0)).transpose();
90 nn(i, 0) = a(1) * b(2) - a(2) * b(1);
91 nn(i, 1) = a(2) * b(0) - a(0) * b(2);
92 nn(i, 2) = a(0) * b(1) - a(1) * b(0);
93
94 float size = nn.row(i) * nn.row(i).transpose();
95 size = std::pow(size, 0.5f);
96 area(i) = size / 2.0f;
97 if (normalize)
98 nn.row(i) /= size;
99 }
100 });
101}

293bool MNEHemisphere::compute_patch_info()
294{
295 if (nearest.empty()) {
296 pinfo.clear();
297 patch_inds = VectorXi();
298 return false;
299 }
300
301 Log::info("\tComputing patch statistics...");
302
303 std::vector<std::pair<int, int>> t_vIndn;
304
305 for (size_t i = 0; i < nearest.size(); ++i) {
306 std::pair<int, int> t_pair(static_cast<int>(i), nearest[i].nearest);
307 t_vIndn.push_back(t_pair);
308 }
309 std::sort(t_vIndn.begin(), t_vIndn.end(), Linalg::compareIdxValuePairSmallerThan<int>);
310
311 VectorXi nearest_sorted(t_vIndn.size());
312
313 int current = 0;
314 std::vector<int> t_vfirsti;
315 t_vfirsti.push_back(current);
316 std::vector<int> t_vlasti;
317
318 for (int i = 0; i < static_cast<int>(t_vIndn.size()); ++i) {
319 nearest_sorted[i] = t_vIndn[i].second;
320 if (t_vIndn[current].second != t_vIndn[i].second) {
321 current = i;
322 t_vlasti.push_back(i - 1);
323 t_vfirsti.push_back(current);
324 }
325 }
326 t_vlasti.push_back(static_cast<int>(t_vIndn.size() - 1));
327
328 for (int k = 0; k < static_cast<int>(t_vfirsti.size()); ++k) {
329 Eigen::VectorXi t_vIndex(t_vlasti[k] - t_vfirsti[k] + 1);
330
331 for (int l = t_vfirsti[k]; l <= t_vlasti[k]; ++l)
332 t_vIndex[l - t_vfirsti[k]] = t_vIndn[l].first;
333
334 std::sort(t_vIndex.data(), t_vIndex.data() + t_vIndex.size());
335
336 pinfo.push_back(t_vIndex);
337 }
338
339 // compute patch indices of the in-use source space vertices
340 Eigen::VectorXi patch_verts(t_vlasti.size());
341 for (int i = 0; i < static_cast<int>(t_vlasti.size()); ++i)
342 patch_verts[i] = nearest_sorted[t_vlasti[i]];
343
344 patch_inds.resize(vertno.size());
345 for (int i = 0; i < vertno.size(); ++i) {
346 const int* ptr = std::find(patch_verts.data(), patch_verts.data() + patch_verts.size(), vertno[i]);
347 patch_inds[i] = static_cast<int>(ptr - patch_verts.data());
348 }
349
350 return true;
351}

355bool MNEHemisphere::add_geometry_info()
356{
357 // The triangles of each vertex in increasing order, one entry per vertex (a volume has no triangles)
358 const auto ntris = static_cast<int>(this->itris.rows());
359 const auto nverts = static_cast<std::size_t>(std::max(this->np, 0));
360 std::vector<int> counts(nverts, 0);
361 for (int t = 0; t < ntris; ++t)
362 for (int corner = 0; corner < 3; ++corner)
363 ++counts.at(static_cast<std::size_t>(this->itris(t, corner)));
364 neighbor_tri.assign(nverts, Eigen::VectorXi());
365 for (std::size_t vertex = 0; vertex < nverts; ++vertex)
366 neighbor_tri[vertex].resize(counts[vertex]);
367 std::ranges::fill(counts, 0);
368 for (int t = 0; t < ntris; ++t) {
369 for (int corner = 0; corner < 3; ++corner) {
370 const auto vertex = static_cast<std::size_t>(this->itris(t, corner));
371 neighbor_tri[vertex][counts[vertex]++] = t;
372 }
373 }
374
375 // The neighboring vertices of each vertex, in the order its triangles name them; vertices are independent
376 neighbor_vert.resize(this->np);
377 const std::size_t chunks = chunk_count(this->np);
378 ThreadPool::global().parallel_for(0, chunks, [&](std::size_t chunk) {
379 const int first = static_cast<int>(chunk * static_cast<std::size_t>(this->np) / chunks);
380 const int last = static_cast<int>((chunk + 1) * static_cast<std::size_t>(this->np) / chunks);
381 std::vector<int> verts;
382 for (int vertex = first; vertex < last; ++vertex) {
383 verts.clear();
384 for (Eigen::Index t = 0; t < neighbor_tri[vertex].size(); ++t) {
385 for (int corner = 0; corner < 3; ++corner) {
386 const int vert = this->itris(neighbor_tri[vertex][t], corner);
387 if (vert != vertex && std::find(verts.begin(), verts.end(), vert) == verts.end())
388 verts.push_back(vert);
389 }
390 }
391 neighbor_vert[vertex] = Eigen::Map<Eigen::VectorXi>(verts.data(), static_cast<Eigen::Index>(verts.size()));
392 }
393 });
394
395 return true;
396}

439bool MNEHemisphere::transform_hemisphere_to(fiff_int_t dest, const FiffCoordTrans& p_Trans)
440{
441 FiffCoordTrans trans(p_Trans);
442
443 if (this->coord_frame == dest) {
444 // res = src;
445 return true;
446 }
447
448 if (trans.to == this->coord_frame && trans.from == dest)
449 trans.invert_transform();
450 else if (trans.from != this->coord_frame || trans.to != dest) {
451 Log::warning("Cannot transform the source space using this coordinate transformation"); //Consider throw
452 return false;
453 }
454
455 MatrixXf t = trans.trans.block(0, 0, 3, 4);
456 // res = src;
457 this->coord_frame = dest;
458 MatrixXf t_rr = MatrixXf::Ones(this->np, 4);
459 t_rr.block(0, 0, this->np, 3) = this->rr;
460 MatrixXf t_nn = MatrixXf::Zero(this->np, 4);
461 t_nn.block(0, 0, this->np, 3) = this->nn;
462
463 this->rr = (t * t_rr.transpose()).transpose();
464 this->nn = (t * t_nn.transpose()).transpose();
465
466 return true;
467}
