119CborMap MnaProject::to_cbor() const
120{
121 CborMap cbor = CborMap::from_json_object(metadata(*this));
122 const auto date = [](const std::optional<UtcTime>& time) {
123 return time ? CborValue::tagged(kDateTimeTag, format_iso8601(*time, true)) : CborValue(nullptr);
124 };
125 cbor.insert("created", date(created));
126 cbor.insert("modified", date(modified));
127 cbor.insert("subjects", detail::to_cbor_array(subjects));
128 cbor.insert("pipeline", detail::to_cbor_array(pipeline));
