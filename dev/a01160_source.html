83bool BidsEvent::write_tsv(const std::filesystem::path& path,
84 std::span<const BidsEvent> events)
85{
86 const std::vector<std::string> headers{"onset", "duration", "trial_type", "value", "sample"};
87
88 std::vector<BidsTsvRow> rows;
89 rows.reserve(events.size());
90
91 for (const BidsEvent& ev : events) {
92 rows.push_back({{"onset", std::format("{:.6f}", static_cast<double>(ev.onset))},
93 {"duration", std::format("{:.6f}", static_cast<double>(ev.duration))},
94 {"trial_type", ev.trialType.empty() ? std::string("n/a") : ev.trialType},
95 {"value", std::to_string(ev.value)},
96 {"sample", std::to_string(ev.sample)}});
97 }
98
99 return BidsTsv::write_tsv(path, headers, rows);
