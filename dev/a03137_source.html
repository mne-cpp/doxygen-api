83mne::Expected<FiffDigPointSet> FiffDigPointSet::read(FiffStream& stream)
84{
85 const bool openHere = !stream.is_open();
86 if (openHere && !stream.open()) {
87 return mne::fail(mne::ErrorCode::Io, "cannot open {}", stream.stream_name());
88 }
89
90 FiffDigPointSet dig;
91 fiff_int_t coordFrame = FIFFV_COORD_HEAD;
92 const std::vector<std::shared_ptr<FiffDirNode>> isotrak = stream.dirtree()->dir_tree_find(FIFFB_ISOTRAK);
93 if (isotrak.size() == 1) {
94 for (const FiffDirEntry& entry : isotrak[0]->dir) {
95 if (entry.kind != FIFF_DIG_POINT && entry.kind != FIFF_MNE_COORD_FRAME) {
96 continue;
97 }
98 const std::optional<FiffTag> tag = stream.read_tag(entry.pos);
99 if (!tag) {
100 return mne::fail(mne::ErrorCode::Format, "cannot read the digitizer tag at {}", entry.pos);
101 }
102 if (entry.kind == FIFF_DIG_POINT) {
103 dig.m_points.push_back(tag->to_dig_point());
104 } else {
105 // Applies to every point of the block, as mne's _read_dig_fif
106 coordFrame = *tag->to_int();
107 }
108 }
109 }
110 for (FiffDigPoint& point : dig.m_points) {
111 point.coord_frame = coordFrame;
112 }
113
114 if (openHere) {
115 stream.close();
116 }
117 return dig;
118}

122mne::Expected<void> FiffDigPointSet::write(const std::filesystem::path& path) const
123{
124 if (path.empty()) {
125 return mne::fail(mne::ErrorCode::Io, "the output path is empty");
126 }
127 std::error_code ec;
128 const std::filesystem::path dir = path.has_parent_path() ? path.parent_path() : std::filesystem::path(".");
129 if (!std::filesystem::is_directory(dir, ec)) {
130 return mne::fail(mne::ErrorCode::NotFound, "the directory {} does not exist", mne::utf8_from_path(dir));
131 }
132 auto stream = FiffStream::start_file(path);
133 if (!stream) {
134 return mne::fail(mne::ErrorCode::Io, "cannot open {} for writing", mne::utf8_from_path(path));
135 }
136 write_to_stream(*stream);
137 stream->end_file();
138 stream->close();
139 return {};
140}

172void FiffDigPointSet::apply_transform(const FiffCoordTrans& coordTrans, bool inverse)
173{
174 const auto& trans = inverse ? coordTrans.invtrans : coordTrans.trans;
175 for (FiffDigPoint& point : m_points) {
176 const Vector4f r = trans * Vector4f(point.r[0], point.r[1], point.r[2], 1.0f);
177 point.r[0] = r(0);
178 point.r[1] = r(1);
179 point.r[2] = r(2);
180 }
181}

185bool FiffDigPointSet::transform_to_head()
186{
187 const FiffDigPoint* fid[3] = {nullptr, nullptr, nullptr};
188 for (const FiffDigPoint& p : m_points) {
189 if (p.kind == FIFFV_POINT_CARDINAL && p.ident >= FIFFV_POINT_LPA && p.ident <= FIFFV_POINT_RPA) {
190 fid[p.ident - FIFFV_POINT_LPA] = &p;
191 }
192 }
193 if (!fid[0] || !fid[1] || !fid[2] || fid[0]->coord_frame != fid[1]->coord_frame || fid[0]->coord_frame != fid[2]->coord_frame) {
194 mne::Log::warning("[FiffDigPointSet::transform_to_head] LPA, nasion and RPA in one frame are required.");
195 return false;
196 }
197 const int frame = fid[0]->coord_frame;
198 if (frame == FIFFV_COORD_HEAD) {
199 return true;
200 }
201 // head -> native, built from the fiducials given in the native frame
202 const FiffCoordTrans headNative = FiffCoordTrans::from_cardinal_points(FIFFV_COORD_HEAD, frame, fid[0]->r, fid[1]->r, fid[2]->r);
203 for (FiffDigPoint& p : m_points) {
204 if (p.coord_frame == frame) {
205 FiffCoordTrans::apply_inverse_trans(p.r, headNative, true);
206 p.coord_frame = FIFFV_COORD_HEAD;
207 }
208 }
209 return true;
210}
