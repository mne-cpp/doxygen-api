142std::int32_t FiffInfo::get_current_comp()
143{
144 std::int32_t comp = 0;
145 std::int32_t first_comp = -1;
146
147 std::int32_t k = 0;
148 for (k = 0; k < this->nchan; ++k) {
149 if (this->chs[k].kind == FIFFV_MEG_CH) {
150 comp = this->chs[k].chpos.coil_type >> 16;
151 if (first_comp < 0)
152 first_comp = comp;
153 else if (comp != first_comp)
154 Log::warning("Compensation is not set equally on all MEG channels");
155 }
156 }

162bool FiffInfo::make_compensator(fiff_int_t from, fiff_int_t to, FiffCtfComp& ctf_comp, bool exclude_comp_chs) const
163{
164 MatrixXd C1, C2, comp_tmp;
165
166 // if(ctf_comp.data)
167 // delete ctf_comp.data;
168 ctf_comp.data.clear();
169
170 if (from == to) {
171 ctf_comp.data.data = MatrixXd::Identity(this->nchan, this->nchan);
172 return false;
173 }
174
175 if (from == 0)
176 C1 = MatrixXd::Zero(this->nchan, this->nchan);
177 else {
178 if (!this->make_compensator(from, C1)) {
179 Log::warning("Cannot create compensator C1\n");
180 Log::warning("Desired compensation matrix (kind = {}) not found\n", from);
181 return false;
182 }
183 }
184
185 if (to == 0)
186 C2 = MatrixXd::Zero(this->nchan, this->nchan);
187 else {
188 if (!this->make_compensator(to, C2)) {
189 Log::warning("Cannot create compensator C2\n");
190 Log::warning("Desired compensation matrix (kind = {}) not found\n", to);
191 return false;
192 }
193 }
194 //
195 // s_from = (I - C1)*s_orig => s_orig = (I - C1)^-1 * s_from
196 // s_to = (I - C2)*s_orig
197 // (I + C1) is only the inverse when C1*C1 = 0, which fails as soon as
198 // reference channels are themselves compensated (e.g. CTF grade 1).
199 //
200 const MatrixXd eye = MatrixXd::Identity(this->nchan, this->nchan);
201 comp_tmp = (eye - C2) * (eye - C1).partialPivLu().inverse();
202
203 std::int32_t k;
204 if (exclude_comp_chs) {
205 VectorXi pick = VectorXi::Zero(this->nchan);
206 std::int32_t npick = 0;
207 for (k = 0; k < this->nchan; ++k) {
208 if (this->chs[k].kind != FIFFV_REF_MEG_CH) {
209 pick(npick) = k;
210 ++npick;
211 }
212 }
213 if (npick == 0) {
214 Log::warning("Nothing remains after excluding the compensation channels\n");
215 return false;
216 }
217
218 ctf_comp.data.data.resize(npick, this->nchan);
219 for (k = 0; k < npick; ++k)
220 ctf_comp.data.data.row(k) = comp_tmp.block(pick(k), 0, 1, this->nchan);
221 } else {
222 ctf_comp.data.data = comp_tmp;
223 }
224 // FiffRawData applies the compensator only when kind is set.
225 ctf_comp.kind = to;
226
227 return true;
228}

232bool FiffInfo::make_compensator(fiff_int_t kind, MatrixXd& this_comp) const //private method
233{
234 const FiffNamedMatrix* this_data = nullptr;
235 MatrixXd presel, postsel;
236 std::int32_t k, col, c, ch = 0, row, row_ch = 0, channelAvailable;
237
238 for (k = 0; k < std::ssize(this->comps); ++k) {
239 if (this->comps[k].kind == kind) {
240 this_data = &this->comps[k].data;
241
242 //
243 // Create the preselector
244 //
245 presel = MatrixXd::Zero(this_data->ncol, this->nchan);
246
247 for (col = 0; col < this_data->ncol; ++col) {
248 channelAvailable = 0;
249 for (c = 0; c < static_cast<int>(this->ch_names.size()); ++c) {
250 if ((this_data->col_names.at(col) == this->ch_names.at(c))) {
251 ++channelAvailable;
252 ch = c;
253 }
254 }
255 if (channelAvailable == 0) {
256 Log::warning("Channel {} is not available in data\n", this_data->col_names.at(col));
257 return false;
258 } else if (channelAvailable > 1) {
259 Log::warning("Ambiguous channel {}", this_data->col_names.at(col));
260 return false;
261 }
262 presel(col, ch) = 1.0;
263 }
264 //
265 // Create the postselector
266 //
267 postsel = MatrixXd::Zero(this->nchan, this_data->nrow);
268
269 for (c = 0; c < this->nchan; ++c) {
270 channelAvailable = 0;
271 for (row = 0; row < static_cast<int>(this_data->row_names.size()); ++row) {
272 if ((this->ch_names.at(c) == this_data->row_names.at(row))) {
273 ++channelAvailable;
274 row_ch = row;
275 }
276 }
277 if (channelAvailable > 1) {
278 Log::warning("Ambiguous channel {}", this->ch_names.at(c));
279 return false;
280 } else if (channelAvailable == 1) {
281 postsel(c, row_ch) = 1.0;
282 }
283 }
284 this_comp = postsel * this_data->data * presel;
285 return true;
286 }
287 }
288 this_comp = defaultMatrixXd;
289 return false;
290}

294FiffInfo FiffInfo::pick_info(const RowVectorXi& sel) const
295{
296 FiffInfo res = *this; //new FiffInfo(this);
297 if (sel.size() == 0)
298 return res;
299
300 //ToDo when pointer List do delation
301 res.chs.clear();
302 res.ch_names.clear();
303
304 std::int32_t idx;
305 for (std::int32_t i = 0; i < sel.size(); ++i) {
306 idx = sel[i];
307 res.chs.push_back(this->chs[idx]);
308 res.ch_names.push_back(this->ch_names[idx]);
309 }
310 res.nchan = static_cast<int>(sel.size());
311 std::erase_if(res.bads, [&res](const std::string& bad) { return !mne::has_channel(res.ch_names, bad); });
312
313 return res;

318std::vector<FiffChInfo> FiffInfo::set_current_comp(std::vector<FiffChInfo>& listFiffChInfo, fiff_int_t value)
319{
320 std::vector<FiffChInfo> newList;
321 std::int32_t k;
322 fiff_int_t coil_type;
323
324 for (k = 0; k < std::ssize(listFiffChInfo); ++k)
325 newList.push_back(listFiffChInfo[k]);
326
327 std::int32_t lower_half = 65535; // hex2dec('FFFF');
328 for (k = 0; k < std::ssize(listFiffChInfo); ++k) {
329 if (listFiffChInfo[k].kind == FIFFV_MEG_CH) {
330 coil_type = listFiffChInfo[k].chpos.coil_type & lower_half;
331 newList[k].chpos.coil_type = (coil_type | (value << 16));
332 }
333 }
334 return newList;
335}

339bool FiffInfo::readMegEegChannels(const std::string& name,
340 bool do_meg,
341 bool do_eeg,
342 const StringList& bads,
343 std::vector<FiffChInfo>& chsp,
344 int& nmegp,
345 int& neegp)
346{
347 FiffStream stream(name);
348
349 if (!stream.open())
350 return false;
351
352 FiffInfo info;
353 std::shared_ptr<FiffDirNode> infoNode;
354 if (!stream.read_meas_info(stream.dirtree(), info, infoNode)) {
355 Log::critical("{} : could not read measurement info", name);
356 stream.close();
357 return false;
358 }
359 stream.close();
360
361 std::vector<FiffChInfo> meg;
362 std::vector<FiffChInfo> eeg;
363
364 for (int k = 0; k < std::ssize(info.chs); k++) {
365 if (mne::has_channel(bads, info.chs[k].ch_name))
366 continue;
367 if (do_meg && info.chs[k].kind == FIFFV_MEG_CH)
368 meg.push_back(info.chs[k]);
369 else if (do_eeg && info.chs[k].isValidEeg())
370 eeg.push_back(info.chs[k]);
371 }
372
373 chsp.clear();
374 chsp.reserve(meg.size() + eeg.size());
375 chsp.insert(chsp.end(), meg.begin(), meg.end());
376 chsp.insert(chsp.end(), eeg.begin(), eeg.end());
377
378 nmegp = static_cast<int>(meg.size());
379 neegp = static_cast<int>(eeg.size());
380 return true;
381}

385void FiffInfo::writeToStream(FiffStream& stream) const
386{
387 //
388 // We will always write floats
389 //
390 fiff_int_t data_type = 4;
391 std::vector<FiffChInfo> chsToWrite(this->chs.begin(), this->chs.begin() + this->nchan);
392
393 fiff_int_t nchanToWrite = static_cast<int>(chsToWrite.size());
394
395 //
396 // write the essentials
397 //
398 stream.start_block(FIFFB_MEAS); //4
399 stream.write_id(FIFF_BLOCK_ID); //5
400 if (this->meas_id.version != -1) {
401 stream.write_id(FIFF_PARENT_BLOCK_ID, this->meas_id); //6
402 }
403 //
404 // Measurement info
405 //
406 stream.start_block(FIFFB_MEAS_INFO); //7
407
408 //
409 // Blocks from the original -> skip this
410 //
411 // std::vector<fiff_int_t> blocks;
412 // blocks << FIFFB_SUBJECT << FIFFB_HPI_MEAS << FIFFB_HPI_RESULT << FIFFB_ISOTRAK << FIFFB_PROCESSING_HISTORY;
413 bool have_hpi_result = false;
414 bool have_isotrak = false;
415 //
416 // megacq parameters
417 //
418 if (!this->acq_pars.empty() || !this->acq_stim.empty()) {
419 stream.start_block(FIFFB_DACQ_PARS);
420 if (!this->acq_pars.empty())
421 stream.write_string(FIFF_DACQ_PARS, this->acq_pars);
422
423 if (!this->acq_stim.empty())
424 stream.write_string(FIFF_DACQ_STIM, this->acq_stim);
425
426 stream.end_block(FIFFB_DACQ_PARS);
427 }
428 stream.write_device_info(this->device_info);
429 //
430 // Coordinate transformations if the HPI result block was not there
431 //
432 if (!have_hpi_result) {
433 if (!this->dev_head_t.empty())
434 stream.write_coord_trans(this->dev_head_t);
435
436 if (!this->ctf_head_t.empty())
437 stream.write_coord_trans(this->ctf_head_t);
438 }
439 //
440 // Polhemus data
441 //
442 if (this->dig.size() > 0 && !have_isotrak) {
443 stream.start_block(FIFFB_ISOTRAK);
444 for (std::int32_t k = 0; k < std::ssize(this->dig); ++k)
445 stream.write_dig_point(this->dig[k]);
446
447 stream.end_block(FIFFB_ISOTRAK);
448 }
449 //
450 // Projectors
451 //
452 stream.write_proj(this->projs);
453 //
454 // CTF compensation info
455 //
456 stream.write_ctf_comp(this->comps);
457 //
458 // Bad channels
459 //
460 if (this->bads.size() > 0) {
461 stream.start_block(FIFFB_MNE_BAD_CHANNELS);
462 stream.write_name_list(FIFF_MNE_CH_NAME_LIST, this->bads);
463 stream.end_block(FIFFB_MNE_BAD_CHANNELS);
464 }
465 //
466 // General
467 //
468 stream.write_float(FIFF_SFREQ, &this->sfreq);
469 if (this->linefreq >= 0.0f)
470 stream.write_float(FIFF_LINE_FREQ, &this->linefreq);
471 stream.write_float(FIFF_HIGHPASS, &this->highpass);
472 stream.write_float(FIFF_LOWPASS, &this->lowpass);
473 stream.write_string(FIFF_EXPERIMENTER, this->experimenter);
474 stream.write_string(FIFF_DESCRIPTION, this->description);
475 stream.write_string(FIFF_UTC_OFFSET, this->utc_offset);
476 stream.write_string(FIFF_PROJ_NAME, this->proj_name);
477 stream.write_int(FIFF_PROJ_ID, &this->proj_id);
478 stream.write_int(FIFF_GANTRY_ANGLE, &this->gantry_angle);
479 stream.write_int(FIFF_NCHAN, &nchanToWrite);
480 stream.write_int(FIFF_DATA_PACK, &data_type);
481 if (this->meas_date[0] != -1)
482 stream.write_int(FIFF_MEAS_DATE, this->meas_date, 2);
483 //
484 // Channel info
485 //
486 MatrixXd cals(1, nchanToWrite);
487
488 for (std::int32_t k = 0; k < nchanToWrite; ++k) {
489 //
490 // Scan numbers may have been messed up
491 //
492 chsToWrite[k].scanNo = k + 1; //+1 because
493 // chs[k].range = 1.0f;//Why? -> cause its already calibrated through reading
494 cals(0, k) = static_cast<double>(chsToWrite[k].cal); //ToDo whats going on with cals?
495 stream.write_ch_info(chsToWrite[k]);
496 }
497 //
498 //
499 stream.end_block(FIFFB_MEAS_INFO);
500}
