44std::string lower(std::string_view text)
45{
46 std::string out(text);
47 std::transform(out.begin(), out.end(), out.begin(), [](char c) {
48 return (c >= 'A' && c <= 'Z') ? static_cast<char>(c - 'A' + 'a') : c;
49 });
50 return out;
51}

53std::uint32_t be32(const std::vector<std::uint8_t>& b, std::size_t offset)
54{
55 if (offset + 4 > b.size()) {
56 return 0;
57 }
58 return (static_cast<std::uint32_t>(b[offset]) << 24) | (static_cast<std::uint32_t>(b[offset + 1]) << 16) |
59 (static_cast<std::uint32_t>(b[offset + 2]) << 8) | b[offset + 3];
60}

62std::uint16_t be16(const std::vector<std::uint8_t>& b, std::size_t offset)
63{
64 return offset + 2 <= b.size() ? static_cast<std::uint16_t>((b[offset] << 8) | b[offset + 1]) : 0;
65}

67std::vector<std::uint8_t> readAt(std::ifstream& file, std::uint64_t offset, std::size_t size)
68{
69 std::vector<std::uint8_t> bytes(size);
70 file.clear();
71 file.seekg(static_cast<std::streamoff>(offset));
72 file.read(reinterpret_cast<char*>(bytes.data()), static_cast<std::streamsize>(size));
73 bytes.resize(file ? size : static_cast<std::size_t>(std::max<std::streamsize>(file.gcount(), 0)));

77// Appends the TrueType faces of a font file, reading only the table directory, name and OS/2
78void indexFile(const fs::path& path, std::vector<SystemFontInfo>& out)
79{
80 std::ifstream file(path, std::ios::binary);
81 if (!file) {
82 return;
83 }
84 const std::vector<std::uint8_t> header = readAt(file, 0, 12);
85 std::vector<std::uint32_t> directories;
86 if (be32(header, 0) == 0x74746366u) { // ttcf
87 const std::uint32_t count = std::min<std::uint32_t>(be32(header, 8), 256);
88 const std::vector<std::uint8_t> offsets = readAt(file, 12, 4 * static_cast<std::size_t>(count));
89 for (std::size_t i = 0; i + 4 <= offsets.size(); i += 4) {
90 directories.push_back(be32(offsets, i));
91 }
92 } else {
93 directories.push_back(0);
94 }
95 for (std::size_t faceIndex = 0; faceIndex < directories.size(); ++faceIndex) {
96 const std::vector<std::uint8_t> directory = readAt(file, directories[faceIndex], 12);
97 const std::uint32_t version = be32(directory, 0);
98 if (version != 0x00010000u && version != 0x74727565u) { // TrueType or 'true'; CFF fonts are skipped
99 continue;
100 }
101 const std::size_t tableCount = be16(directory, 4);
102 const std::vector<std::uint8_t> records = readAt(file, directories[faceIndex] + 12ull, 16 * tableCount);
103 std::uint32_t nameOffset = 0;
104 std::uint32_t nameLength = 0;
105 std::uint32_t os2Offset = 0;
106 std::uint32_t os2Length = 0;
107 bool glyf = false;
108 for (std::size_t r = 0; r + 16 <= records.size(); r += 16) {
109 const std::uint32_t tableTag = be32(records, r);
110 if (tableTag == 0x6E616D65u) { // name
111 nameOffset = be32(records, r + 8);
112 nameLength = be32(records, r + 12);
113 } else if (tableTag == 0x4F532F32u) { // OS/2
114 os2Offset = be32(records, r + 8);
115 os2Length = be32(records, r + 12);
116 } else if (tableTag == 0x676C7966u && be32(records, r + 12) > 0) { // glyf
117 glyf = true;
118 }
119 }
120 if (!glyf || nameLength == 0) {
121 continue;
122 }
123 SystemFontInfo info;
124 info.path = path;
125 info.face_index = static_cast<int>(faceIndex);
126 const std::vector<std::uint8_t> name = readAt(file, nameOffset, std::min<std::size_t>(nameLength, kMaxNameTable));
127 std::tie(info.family, info.style) = detail::font_names(name);
128 if (info.family.empty()) {
129 continue;
130 }
131 if (os2Length >= 64) {
132 const std::vector<std::uint8_t> os2 = readAt(file, os2Offset, 64);
133 if (os2.size() == 64) {
134 info.weight = std::clamp<int>(be16(os2, 4), 1, 1000);
135 info.italic = (be16(os2, 62) & 0x1) != 0;
136 for (std::size_t k = 0; k < 4; ++k) {
137 info.unicode_ranges[k] = be32(os2, 42 + 4 * k);
138 }
139 }
140 } else {
141 const std::string style = lower(info.style);
142 info.weight = style.find("bold") != std::string::npos ? 700 : 400;
143 info.italic = style.find("italic") != std::string::npos || style.find("oblique") != std::string::npos;
144 }
145 out.push_back(std::move(info));
146 }
147}

220// Lower is better: italic as requested, then the weight
221int matchScore(const SystemFontInfo& info, bool bold, bool italic, bool hasHeavy)
222{
223 int score = info.italic == italic ? 0 : 10000;
224 if (bold && hasHeavy) {
225 score += info.weight >= 600 ? info.weight - 600 : 5000;
226 } else {
227 score += 2 * std::abs(info.weight - 400) + (info.weight < 400 ? 1 : 0);
228 }
229 return score;
230}

254std::vector<fs::path> SystemFonts::default_directories()
255{
256 std::vector<fs::path> directories;
257 const auto env = [](const char* name) -> std::string {
258 const char* value = std::getenv(name); // NOLINT(concurrency-mt-unsafe): read-only
259 return value ? std::string(value) : std::string();
260 };
261#if defined(_WIN32)
262 if (const std::string windir = env("WINDIR"); !windir.empty()) {
263 directories.push_back(fs::path(windir) / "Fonts");
264 }
265 if (const std::string local = env("LOCALAPPDATA"); !local.empty()) {
266 directories.push_back(fs::path(local) / "Microsoft" / "Windows" / "Fonts");
267 }
268 constexpr char kSeparator = ';';
269#elif defined(__APPLE__)
270 directories.push_back("/System/Library/Fonts");
271 directories.push_back("/Library/Fonts");
272 if (const std::string home = env("HOME"); !home.empty()) {
273 directories.push_back(fs::path(home) / "Library" / "Fonts");
274 }
275 constexpr char kSeparator = ':';
276#else
277 directories.push_back("/usr/share/fonts");
278 directories.push_back("/usr/local/share/fonts");
279 if (const std::string home = env("HOME"); !home.empty()) {
280 directories.push_back(fs::path(home) / ".local" / "share" / "fonts");
281 directories.push_back(fs::path(home) / ".fonts");
282 }
283 constexpr char kSeparator = ':';
284#endif
285 const std::string extra = env("MNE_FONT_PATH");
286 std::size_t start = 0;
287 while (start <= extra.size()) {
288 const std::size_t end = std::min(extra.find(kSeparator, start), extra.size());
289 if (end > start) {
290 directories.emplace_back(extra.substr(start, end - start));
291 }
292 start = end + 1;
293 }
294 return directories;
295}

306const std::vector<SystemFontInfo>& SystemFonts::faces() const
307{
308 std::call_once(m_indexed, [this] {
309 std::set<fs::path> files;
310 for (const fs::path& directory : m_directories) {
311 std::error_code error;
312 fs::recursive_directory_iterator it(directory, fs::directory_options::skip_permission_denied, error);
313 for (; !error && it != fs::recursive_directory_iterator(); it.increment(error)) {
314 std::error_code statError;
315 if (!it->is_regular_file(statError)) {
316 continue;
317 }
318 const std::string extension = lower(it->path().extension().string());
319 if (extension == ".ttf" || extension == ".ttc" || extension == ".otf") {
320 std::error_code canonicalError;
321 const fs::path canonical = fs::weakly_canonical(it->path(), canonicalError);
322 files.insert(canonicalError ? it->path() : canonical);
323 }
324 }
325 }
326 for (const fs::path& file : files) {
327 indexFile(file, m_faces);
328 }
329 });
330 return m_faces;

350std::shared_ptr<const FontFace> SystemFonts::find(std::string_view family, bool bold, bool italic) const
351{
352 const std::string wanted = lower(family);
353 std::vector<const SystemFontInfo*> candidates;
354 for (const SystemFontInfo& info : faces()) {
355 if (lower(info.family) == wanted) {
356 candidates.push_back(&info);
357 }
358 }
359 const bool hasHeavy = std::any_of(candidates.begin(), candidates.end(), [](const SystemFontInfo* info) { return info->weight >= 600; });
360 std::stable_sort(candidates.begin(), candidates.end(), [&](const SystemFontInfo* a, const SystemFontInfo* b) {
361 return matchScore(*a, bold, italic, hasHeavy) < matchScore(*b, bold, italic, hasHeavy);
362 });
363 const std::lock_guard lock(m_mutex);
364 for (const SystemFontInfo* info : candidates) {
365 if (auto face = load(*info)) {
366 return face;
367 }
368 }
369 return nullptr;
370}

374std::shared_ptr<const FontFace> SystemFonts::face(const Font& font) const
375{
376 const std::string key = lower(font.family) + (font.bold ? "|b" : "|") + (font.italic ? "i" : "");
377 {
378 const std::lock_guard lock(m_mutex);
379 if (const auto it = m_resolved.find(key); it != m_resolved.end()) {
380 return it->second;
381 }
382 }
383 std::shared_ptr<const FontFace> resolved;
384 if (!font.family.empty()) {
385 resolved = find(font.family, font.bold, font.italic);
386 }
387 for (std::size_t i = 0; !resolved && i < m_preferredFamilies.size(); ++i) {
388 resolved = find(m_preferredFamilies[i], font.bold, font.italic);
389 }
390 if (!resolved) {
391 // Any regular face with Latin letters
392 std::vector<const SystemFontInfo*> latin;
393 for (const SystemFontInfo& info : faces()) {
394 if (hasRangeBit(info, 0) && !info.italic) {
395 latin.push_back(&info);
396 }
397 }
398 std::stable_sort(latin.begin(), latin.end(), [](const SystemFontInfo* a, const SystemFontInfo* b) {
399 return std::abs(a->weight - 400) < std::abs(b->weight - 400);
400 });
401 const std::lock_guard lock(m_mutex);
402 for (std::size_t i = 0; !resolved && i < latin.size(); ++i) {
403 if (auto candidate = load(*latin[i]); candidate && candidate->glyph_index(U'A') != 0) {
404 resolved = std::move(candidate);
405 }
406 }
407 }
408 const std::lock_guard lock(m_mutex);
409 m_resolved.emplace(key, resolved);
410 return resolved;
411}

415std::shared_ptr<const FontFace> SystemFonts::fallback_face(const Font& /*font*/, char32_t code_point) const
416{
417 {
418 const std::lock_guard lock(m_mutex);
419 if (const auto it = m_fallbacks.find(code_point); it != m_fallbacks.end()) {
420 return it->second;
421 }
422 }
423 // One regular face per family, preferred families first, that claims the code point's Unicode range
424 const int bit = unicodeRangeBit(code_point);
425 std::vector<const SystemFontInfo*> candidates;
426 for (const SystemFontInfo& info : faces()) {
427 if (info.italic || lower(info.family) == "lastresort" || (bit >= 0 && !hasRangeBit(info, bit) && !rangesUnknown(info))) {
428 continue;
429 }
430 const auto same = std::find_if(candidates.begin(), candidates.end(), [&](const SystemFontInfo* c) { return c->family == info.family; });
431 if (same == candidates.end()) {
432 candidates.push_back(&info);
433 } else if (std::abs(info.weight - 400) < std::abs((*same)->weight - 400)) {
434 *same = &info;
435 }
436 }
437 const auto rank = [&](const SystemFontInfo* info) {
438 const std::string family = lower(info->family);
439 for (std::size_t i = 0; i < m_preferredFamilies.size(); ++i) {
440 if (lower(m_preferredFamilies[i]) == family) {
441 return static_cast<int>(i);
442 }
443 }
444 return static_cast<int>(m_preferredFamilies.size()) + (rangesUnknown(*info) ? 1 : 0);
445 };
446 std::stable_sort(candidates.begin(), candidates.end(), [&](const SystemFontInfo* a, const SystemFontInfo* b) { return rank(a) < rank(b); });
447 const std::lock_guard lock(m_mutex);
448 std::shared_ptr<const FontFace> found;
449 int loads = 0;
450 for (const SystemFontInfo* info : candidates) {
451 if (loads++ >= kMaxFallbackLoads) {
452 break;
453 }
454 if (auto face = load(*info); face && face->glyph_index(code_point) != 0) {
455 found = std::move(face);
456 break;
457 }
458 }
459 m_fallbacks.emplace(code_point, found);
460 return found;
461}
