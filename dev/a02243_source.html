59Expected<MNEHemisphere> read_source_space(FiffStream& stream, const std::shared_ptr<FiffDirNode>& p_Tree, const std::string& name)
60{
61 MNEHemisphere p_Hemisphere;
62
63 std::optional<FiffTag> t_pTag;
64
65 //=====================================================================
66 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_ID); !t_pTag)
67 p_Hemisphere.id = FIFFV_MNE_SURF_UNKNOWN;
68 else
69 p_Hemisphere.id = *t_pTag->to_int();
70
71
72 //=====================================================================
73 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NPOINTS); !t_pTag) {
74 return fail(ErrorCode::Format, "a source space in {} has no vertex count", name);
75 }
76 p_Hemisphere.np = *t_pTag->to_int();
77
78 //=====================================================================
79 if (t_pTag = p_Tree->find_tag(stream, FIFF_BEM_SURF_NTRI); !t_pTag) {
80 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NTRI); !t_pTag)
81 p_Hemisphere.ntri = 0;
82 else
83 p_Hemisphere.ntri = *t_pTag->to_int();
84 } else {
85 p_Hemisphere.ntri = *t_pTag->to_int();
86 }
87
88 //=====================================================================
89 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_COORD_FRAME); !t_pTag) {
90 return fail(ErrorCode::Format, "a source space in {} has no coordinate frame", name);
91 }
92 p_Hemisphere.coord_frame = *t_pTag->to_int();
93
94 //=====================================================================
95 //
96 // Vertices, normals, and triangles
97 //
98 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_POINTS); !t_pTag) {
99 return fail(ErrorCode::Format, "a source space in {} has no vertices", name);
100 }
101
102 p_Hemisphere.rr = t_pTag->to_float_matrix().transpose();
103 std::int32_t rows_rr = p_Hemisphere.rr.rows();
104
105 if (rows_rr != p_Hemisphere.np) {
106 return fail(ErrorCode::Format, "a source space in {} has a vertex count that does not match its vertices", name);
107 }
108
109 //=====================================================================
110 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NORMALS); !t_pTag) {
111 return fail(ErrorCode::Format, "a source space in {} has no vertex normals", name);
112 }
113
114 p_Hemisphere.nn = t_pTag->to_float_matrix().transpose();
115 std::int32_t rows_nn = p_Hemisphere.nn.rows();
116
117 if (rows_nn != p_Hemisphere.np) {
118 return fail(ErrorCode::Format, "a source space in {} has a normal count that does not match its vertices", name);
119 }
120
121 //=====================================================================
122 if (p_Hemisphere.ntri > 0) {
123 if (t_pTag = p_Tree->find_tag(stream, FIFF_BEM_SURF_TRIANGLES); !t_pTag) {
124 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_TRIANGLES); !t_pTag) {
125 return fail(ErrorCode::Format, "a source space in {} has no triangles", name);
126 } else {
127 p_Hemisphere.itris = t_pTag->to_int_matrix().transpose();
128 p_Hemisphere.itris.array() -= 1; //0 based indizes
129 }
130 } else {
131 p_Hemisphere.itris = t_pTag->to_int_matrix().transpose();
132 p_Hemisphere.itris.array() -= 1; //0 based indizes
133 }
134 if (p_Hemisphere.itris.rows() != p_Hemisphere.ntri) {
135 return fail(ErrorCode::Format, "a source space in {} has a triangle count that does not match its triangles", name);
136 }
137 } else {
138 p_Hemisphere.itris.resize(0, 3);
139 }
140
141 //
142 // Which vertices are active
143 //
144 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NUSE); !t_pTag) {
145 p_Hemisphere.nuse = 0;
146 p_Hemisphere.inuse = VectorXi::Zero(p_Hemisphere.nuse);
147 VectorXi p_defaultVector;
148 p_Hemisphere.vertno = p_defaultVector;
149 } else {
150 p_Hemisphere.nuse = *t_pTag->to_int();
151 if (t_pTag = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_SELECTION); !t_pTag) {
152 return fail(ErrorCode::Format, "a source space in {} has no source selection", name);
153 }
154 p_Hemisphere.inuse = VectorXi(Map<const VectorXi>(t_pTag->to_int(), t_pTag->size() / 4, 1)); //use copy constructor, for the sake of easy memory management
155
156 p_Hemisphere.vertno = VectorXi::Zero(p_Hemisphere.nuse);
157 if (p_Hemisphere.inuse.rows() != p_Hemisphere.np) {
158 return fail(ErrorCode::Format, "a source space in {} has a selection that does not match its vertices", name);
159 }
160 int pp = 0;
161 for (int p = 0; p < p_Hemisphere.np; ++p) {
162 if (p_Hemisphere.inuse(p) == 1) {
163 p_Hemisphere.vertno(pp) = p;
164 ++pp;
165 }
166 }
167 }
168
169 //
170 // Use triangulation
171 //
172 std::optional<FiffTag> t_pTag1;
173 std::optional<FiffTag> t_pTag2;
174 t_pTag1 = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NUSE_TRI);
175 t_pTag2 = t_pTag1 ? p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_USE_TRIANGLES) : std::nullopt;
176 if (!t_pTag1 || !t_pTag2) {
177 MatrixX3i p_defaultMatrix;
178 p_Hemisphere.nuse_tri = 0;
179 p_Hemisphere.use_itris = p_defaultMatrix;
180 } else {
181 p_Hemisphere.nuse_tri = *t_pTag1->to_int();
182 p_Hemisphere.use_itris = t_pTag2->to_int_matrix().transpose();
183 p_Hemisphere.use_itris.array() -= 1; //0 based indizes
184 }
185
186 //
187 // Patch-related information
188 //
189 t_pTag1 = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NEAREST);
190 t_pTag2 = t_pTag1 ? p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NEAREST_DIST) : std::nullopt;
191 if (!t_pTag1 || !t_pTag2) {
192 p_Hemisphere.nearest.clear();
193 } else {
194 //res.nearest = tag1.data + 1;
195 VectorXi nearestIdx = VectorXi(Map<const VectorXi>(t_pTag1->to_int(), t_pTag1->size() / 4, 1));
196 VectorXd nearestDist = VectorXd((Map<const VectorXf>(t_pTag2->to_float(), t_pTag2->size() / 4, 1)).cast<double>());
197 p_Hemisphere.setNearestData(nearestIdx, nearestDist);
198 }
199
200 // patch_info(p_Hemisphere.nearest, p_Hemisphere.pinfo);
201 if (p_Hemisphere.compute_patch_info())
202 Log::info("\tPatch information added...");
203 //
204 // Distances
205 //
206 t_pTag1 = p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_DIST);
207 t_pTag2 = t_pTag1 ? p_Tree->find_tag(stream, FIFF_MNE_SOURCE_SPACE_DIST_LIMIT) : std::nullopt;
208 if (!t_pTag1 || !t_pTag2) {
209 p_Hemisphere.dist = FiffSparseMatrix();
210 p_Hemisphere.dist_limit = 0;
211 } else {
212 auto dist_lower = FiffSparseMatrix::fiff_get_float_sparse_matrix(*t_pTag1);
213 if (dist_lower) {
214 auto dist_full = dist_lower->mne_add_upper_triangle_rcs();
215 if (dist_full) {
216 p_Hemisphere.dist = std::move(*dist_full);
217 }
218 }
219 p_Hemisphere.dist_limit = *t_pTag2->to_float(); //ToDo Check if this is realy always a float and not a matrix
220 }
221
222 return p_Hemisphere;
223}

352MNESourceSpaces MNESourceSpaces::pick_regions(const std::vector<FsLabel>& p_qListLabels) const
353{
354 MNESourceSpaces selectedSrc(*this);
355
356 for (std::int32_t h = 0; h < 2; ++h) {
357 auto& srcSpace = *selectedSrc.m_sourceSpaces[h];
358 const auto& origSpace = *this->m_sourceSpaces[h];
359 auto* selHemi = dynamic_cast<MNEHemisphere*>(&srcSpace);
360 auto* origHemi = dynamic_cast<const MNEHemisphere*>(&origSpace);
361
362 VectorXi selVertices;
363
364 //get vertices indeces for new selection
365 std::int32_t iSize = 0;
366 for (const auto& label : p_qListLabels) {
367 if (label.hemi == h) {
368 VectorXi currentSelection;
369
370 Linalg::intersect(origSpace.vertno, label.vertices, currentSelection);
371
372 selVertices.conservativeResize(iSize + currentSelection.size());
373 selVertices.block(iSize, 0, currentSelection.size(), 1) = currentSelection;
374 iSize = selVertices.size();
375 }
376 }
377
378 Linalg::sort(selVertices, false);
379
380 VectorXi newVertno(selVertices.size());
381
382 srcSpace.inuse = VectorXi::Zero(srcSpace.np);
383
384 for (std::int32_t i = 0; i < selVertices.size(); ++i) {
385 newVertno[i] = origSpace.vertno[selVertices[i]];
386 srcSpace.inuse[newVertno[i]] = 1;
387 }
388
389 srcSpace.nuse = selVertices.size();
390 srcSpace.vertno = newVertno;
391
392 //
393 // Tris
394 //
395 VectorXi idx_select = VectorXi::Zero(origSpace.use_itris.rows());
396 for (std::int32_t i = 0; i < 3; ++i) {
397 VectorXi tri_dim = origSpace.use_itris.col(i);
398 VectorXi idx_dim;
399 Linalg::intersect(tri_dim, newVertno, idx_dim);
400
401 for (std::int32_t j = 0; j < idx_dim.size(); ++j)
402 idx_select[idx_dim[j]] = 1;
403 }
404
405 std::int32_t countSel = 0;
406 for (std::int32_t i = 0; i < idx_select.size(); ++i)
407 if (idx_select[i] == 1)
408 ++countSel;
409
410 srcSpace.nuse_tri = countSel;
411
412 MatrixX3i use_tris_new(countSel, 3);
413 MatrixX3d use_tri_cent_new(countSel, 3);
414 MatrixX3d use_tri_nn_new(countSel, 3);
415 VectorXd use_tri_area_new(countSel);
416
417 countSel = 0;
418 for (std::int32_t i = 0; i < idx_select.size(); ++i) {
419 if (idx_select[i] == 1) {
420 use_tris_new.row(countSel) = origSpace.use_itris.row(i);
421 if (origHemi && selHemi) {
422 use_tri_cent_new.row(countSel) = origHemi->use_tri_cent.row(i);
423 use_tri_nn_new.row(countSel) = origHemi->use_tri_nn.row(i);
424 use_tri_area_new[countSel] = origHemi->use_tri_area[i];
425 }
426 ++countSel;
427 }
428 }
429
430 srcSpace.use_itris = use_tris_new;
431 if (selHemi) {
432 selHemi->use_tri_cent = use_tri_cent_new;
433 selHemi->use_tri_nn = use_tri_nn_new;
434 selHemi->use_tri_area = use_tri_area_new;
435 // patch_inds is indexed by used source, so it must follow the selection
436 if (origHemi && origHemi->patch_inds.size() == origSpace.nuse) {
437 selHemi->patch_inds.resize(selVertices.size());
438 for (std::int32_t i = 0; i < selVertices.size(); ++i)
439 selHemi->patch_inds[i] = origHemi->patch_inds[selVertices[i]];
440 }
441 }
442 }
443
444 return selectedSrc;
445}

460Expected<MNESourceSpaces> MNESourceSpaces::read(FiffStream& stream, bool add_geom)
461{
462 const std::string name = stream.stream_name();
463 if (!stream.is_open() && !stream.open()) {
464 return fail(ErrorCode::Io, "cannot open {}", name);
465 }
466 const std::vector<std::shared_ptr<FiffDirNode>> nodes = stream.dirtree()->dir_tree_find(FIFFB_MNE_SOURCE_SPACE);
467 if (nodes.empty()) {
468 return fail(ErrorCode::NotFound, "no source spaces in {}", name);
469 }
470
471 MNESourceSpaces spaces;
472 spaces.m_sourceSpaces.reserve(nodes.size());
473 for (const std::shared_ptr<FiffDirNode>& node : nodes) {
474 auto hemisphere = read_source_space(stream, node, name);
475 if (!hemisphere) {
476 return std::unexpected(hemisphere.error());
477 }
478 if (add_geom) {
479 hemisphere->complete_source_space_info();
480 }
481 spaces.m_sourceSpaces.push_back(std::make_shared<MNEHemisphere>(std::move(*hemisphere)));
482 }
483 Log::info("\t{} source spaces read", nodes.size());
484 return spaces;
485}

510 }
511 }
512 }

519{
