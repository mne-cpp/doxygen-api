58VectorXd Linalg::combine_xyz(const VectorXd& vec)
59{
60 if (vec.size() % 3 != 0) {
61 return VectorXd();
62 }
63
64 MatrixXd tmp = MatrixXd(vec.transpose());
65 SparseMatrix<double> s = make_block_diag(tmp, 3);
66
67 SparseMatrix<double> sC = s * s.transpose();
68 VectorXd comb(sC.rows());
69
70 for (std::int32_t i = 0; i < sC.rows(); ++i)
71 comb[i] = sC.coeff(i, i);
72
73 return comb;
74}

104void Linalg::get_whitener(MatrixXd& A,
105 bool pca,
106 const std::string& /*ch_type*/,
107 VectorXd& eig,
108 MatrixXd& eigvec)
109{
110 // Eigenvalues come in ascending order; for a symmetric matrix they are the singular values up to sign
111 SelfAdjointEigenSolver<MatrixXd> t_eigenSolver(A);
112 eig = t_eigenSolver.eigenvalues();
113 eigvec = t_eigenSolver.eigenvectors().transpose();
114
115 const double threshold = 1e-8 * eig.cwiseAbs().maxCoeff();
116 const auto rnk = static_cast<std::int32_t>((eig.array().abs() > threshold).count());
117 eig.head(eig.size() - rnk).setZero();
118
119 if (pca) {
120 eigvec = eigvec.bottomRows(rnk).eval(); // eval: resizing eigvec frees the rows being read
121 }
122}

126VectorXi Linalg::intersect(const VectorXi& v1,
127 const VectorXi& v2,
128 VectorXi& idx_sel)
129{
130 std::vector<int> tmp;
131
132 std::vector<std::pair<int, int>> t_vecIntIdxValue;
133
134 for (std::int32_t i = 0; i < v1.size(); ++i)
135 tmp.push_back(v1[i]);
136
137 std::vector<int>::iterator it;
138 for (std::int32_t i = 0; i < v2.size(); ++i) {
139 it = std::search(tmp.begin(), tmp.end(), &v2[i], &v2[i] + 1);
140 if (it != tmp.end())
141 t_vecIntIdxValue.push_back(std::pair<int, int>(v2[i], static_cast<int>(it - tmp.begin())));
142 }
143
144 std::sort(t_vecIntIdxValue.begin(), t_vecIntIdxValue.end(), Linalg::compareIdxValuePairSmallerThan<int>);
145
146 VectorXi p_res(t_vecIntIdxValue.size());
147 idx_sel = VectorXi(t_vecIntIdxValue.size());
148
149 for (std::uint32_t i = 0; i < t_vecIntIdxValue.size(); ++i) {
150 p_res[i] = t_vecIntIdxValue[i].first;
151 idx_sel[i] = t_vecIntIdxValue[i].second;
152 }
153
154 return p_res;
155}

159SparseMatrix<double> Linalg::make_block_diag(const MatrixXd& A,
160 std::int32_t n)
161{
162 const auto ma = static_cast<std::int32_t>(A.rows());
163 const auto na = static_cast<std::int32_t>(A.cols());
164 float bdn = static_cast<float>(na) / n;
165
166 if (bdn != std::floor(bdn)) {
167 return SparseMatrix<double>();
168 }
169
170 typedef Eigen::Triplet<double> T;
171 std::vector<T> tripletList;
172 tripletList.reserve(static_cast<size_t>(bdn * ma * n));
173
174 for (std::int32_t i = 0; i < static_cast<std::int32_t>(bdn); ++i) {
175 std::int32_t current_col = i * n;
176 std::int32_t current_row = i * ma;
177
178 for (std::int32_t r = 0; r < ma; ++r)
179 for (std::int32_t c = 0; c < n; ++c)
180 tripletList.push_back(T(r + current_row, c + current_col, A(r, c + current_col)));
181 }
182
183 SparseMatrix<double> bd(static_cast<int>(std::floor(ma * bdn + 0.5f)), na);
184 bd.setFromTriplets(tripletList.begin(), tripletList.end());
185
186 return bd;
187}

191std::int32_t Linalg::rank(const MatrixXd& A,
192 double tol)
193{
194 JacobiSVD<MatrixXd> t_svdA(A);
195 VectorXd s = t_svdA.singularValues();
196 double t_dMax = s.maxCoeff();
197 t_dMax *= tol;
198 std::int32_t sum = 0;
199 for (std::int32_t i = 0; i < s.size(); ++i)
200 sum += s[i] > t_dMax ? 1 : 0;
201 return sum;
202}
