112
127int mne_get_values_from_data_3 (float time,
128 float integ,
129 float **data,
130 int nsamp,
131 int nch,
132 float tmin,
133 float sfreq,
134 int use_abs,
135 float *value)
136{
137 int n1,n2,k;
138 float s1,s2;
139 float f1,f2;
140 float sum;
141 float width;
142 int ch;
143
144 for (ch = 0; ch < nch; ch++) {
145 /*
146 * Find out the correct samples
147 */
148 if (std::fabs(sfreq*integ) < EPS_3) { /* This is the single-sample case */
149 s1 = sfreq*(time - tmin);
150 n1 = floor(s1);
151 f1 = 1.0 + n1 - s1;
152 if (n1 < 0 || n1 > nsamp-1) {
153 qWarning("Sample value out of range %d (0..%d)",n1,nsamp-1);
154 return(-1);
155 }
156 /*
157 * Avoid rounding error
158 */
159 if (n1 == nsamp-1) {
160 if (std::fabs(f1 - 1.0) < 1e-3)
161 f1 = 1.0;
162 }
163 if (f1 < 1.0 && n1 > nsamp-2) {
164 qWarning("Sample value out of range %d (0..%d) %.4f",n1,nsamp-1,f1);
165 return(-1);
166 }
167 if (f1 < 1.0) {
168 if (use_abs)
169 sum = f1*std::fabs(data[n1][ch]) + (1.0-f1)*std::fabs(data[n1+1][ch]);
170 else
171 sum = f1*data[n1][ch] + (1.0-f1)*data[n1+1][ch];
172 }
173 else {
174 if (use_abs)
175 sum = std::fabs(data[n1][ch]);
176 else
177 sum = data[n1][ch];
178 }
179 }
180 else { /* Multiple samples */
181 s1 = sfreq*(time - 0.5*integ - tmin);
182 s2 = sfreq*(time + 0.5*integ - tmin);
183 n1 = ceil(s1); n2 = floor(s2);
184 if (n2 < n1) { /* We are within one sample interval */
185 n1 = floor(s1);
186 if (n1 < 0 || n1 > nsamp-2)
187 return (-1);
188 f1 = s1 - n1;
189 f2 = s2 - n1;
190 if (use_abs)
191 sum = 0.5*((f1+f2)*std::fabs(data[n1+1][ch]) + (2.0-f1-f2)*std::fabs(data[n1][ch]));
192 else
193 sum = 0.5*((f1+f2)*data[n1+1][ch] + (2.0-f1-f2)*data[n1][ch]);
194 }
195 else {
196 f1 = n1 - s1;
197 f2 = s2 - n2;
198 if (n1 < 0 || n1 > nsamp-1) {
199 qWarning("Sample value out of range %d (0..%d)",n1,nsamp-1);
200 return(-1);
201 }
202 if (n2 < 0 || n2 > nsamp-1) {
203 qWarning("Sample value out of range %d (0..%d)",n2,nsamp-1);
204 return(-1);
205 }
206 if (f1 != 0.0 && n1 < 1)
207 return(-1);
208 if (f2 != 0.0 && n2 > nsamp-2)
209 return(-1);
210 sum = 0.0;
211 width = 0.0;
212 if (n2 > n1) { /* Do the whole intervals */
213 if (use_abs) {
214 sum = 0.5 * std::fabs(data[n1][ch]);
215 for (k = n1+1; k < n2; k++)
216 sum = sum + std::fabs(data[k][ch]);
217 sum = sum + 0.5 * std::fabs(data[n2][ch]);
218 }
219 else {
220 sum = 0.5*data[n1][ch];
221 for (k = n1+1; k < n2; k++)
222 sum = sum + data[k][ch];
223 sum = sum + 0.5*data[n2][ch];
224 }
225 width = n2 - n1;
226 }
227 /*
228 * Add tails
229 */
230 if (use_abs) {
231 if (f1 != 0.0)
232 sum = sum + 0.5*f1*(f1*std::fabs(data[n1-1][ch]) + (2.0-f1)*std::fabs(data[n1][ch]));
233 if (f2 != 0.0)
234 sum = sum + 0.5*f2*(f2*std::fabs(data[n2+1][ch]) + (2.0-f2)*std::fabs(data[n2][ch]));
235 }
236 else {
237 if (f1 != 0.0)
238 sum = sum + 0.5*f1*(f1*data[n1-1][ch] + (2.0-f1)*data[n1][ch]);
239 if (f2 != 0.0)
240 sum = sum + 0.5*f2*(f2*data[n2+1][ch] + (2.0-f2)*data[n2][ch]);
241 }
242 width = width + f1 + f2;
243 sum = sum/width;
244 }
245 }
246 value[ch] = sum;
247 }
248 return (0);
249}

396static int read_meg_eeg_ch_info(const QString& name,
397 int do_meg,
398 int do_eeg,
399 const QStringList& bads,
400 int nbad,
401 QList<FiffChInfo>& chsp,
402 int &nmegp,
403 int &neegp)
404{
405 QFile file(name);
406 FiffStream::SPtr stream(new FiffStream(&file));
407
408 if (!stream->open())
409 return FIFF_FAIL;
410
411 FiffInfo info;
412 FiffDirNode::SPtr infoNode;
413 if (!stream->read_meas_info(stream->dirtree(), info, infoNode)) {
414 qCritical("%s : could not read measurement info", name.toUtf8().data());
415 stream->close();
416 return FIFF_FAIL;
417 }
418 stream->close();
419
420 QList<FiffChInfo> meg;
421 int nmeg = 0;
422 QList<FiffChInfo> eeg;
423 int neeg = 0;
424
425 for (int k = 0; k < info.chs.size(); k++) {
426 if (accept_ch(info.chs[k], bads, nbad)) {
427 if (do_meg && info.chs[k].kind == FIFFV_MEG_CH) {
428 meg.append(info.chs[k]);
429 nmeg++;
430 } else if (do_eeg && info.chs[k].kind == FIFFV_EEG_CH && is_valid_eeg_ch(info.chs[k])) {
431 eeg.append(info.chs[k]);
432 neeg++;
433 }
434 }
435 }
436
437 chsp.clear();
438 chsp.reserve(nmeg + neeg);
439 chsp.append(meg);
440 chsp.append(eeg);
441
442 nmegp = nmeg;
443 neegp = neeg;
444 return FIFF_OK;
445}

484int InvDipoleFitData::setup_forward_model(InvDipoleFitData *d, MNECTFCompDataSet* comp_data, FwdCoilSet *comp_coils)
485{
486 FwdCompData* comp;
487 dipoleFitFuncs f;
488 int fit_sphere_to_bem = true;
489
490 if (!d->bemname.isEmpty()) {
491 /*
492 * Set up the boundary-element model
493 */
494 QString bemsolname = FwdBemModel::fwd_bem_make_bem_sol_name(d->bemname);
495 d->bemname = bemsolname;
496
497 qInfo("\nSetting up the BEM model using %s...",d->bemname.toUtf8().constData());
498 qInfo("\nLoading surfaces...");
499 d->bem_model = FwdBemModel::fwd_bem_load_three_layer_surfaces(d->bemname);
500 if (d->bem_model) {
501 qInfo("Three-layer model surfaces loaded.");
502 }
503 else {
504 d->bem_model = FwdBemModel::fwd_bem_load_homog_surface(d->bemname);
505 if (!d->bem_model)
506 return FAIL;
507 qInfo("Homogeneous model surface loaded.");
508 }
509 if (d->neeg > 0 && d->bem_model->nsurf == 1) {
510 qCritical("Cannot use a homogeneous model in EEG calculations.");
511 return FAIL;
512 }
513 qInfo("\nLoading the solution matrix...");
514 if (d->bem_model->fwd_bem_load_recompute_solution(d->bemname,FWD_BEM_UNKNOWN,false) == FAIL)
515 return FAIL;
516 qInfo("Employing the head->MRI coordinate transform with the BEM model.");
517 Q_ASSERT(d->mri_head_t);
518 if (d->bem_model->fwd_bem_set_head_mri_t(*d->mri_head_t) == FAIL)
519 return FAIL;
520 qInfo("BEM model %s is now set up",d->bem_model->sol_name.toUtf8().constData());
521 /*
522 * Find the best-fitting sphere
523 */
524 if (fit_sphere_to_bem) {
525 MNESurface* inner_skull;
526 float simplex_size = 2e-2;
527 float R;
528 VectorXf r0_vec;
529
530 if ((inner_skull = d->bem_model->fwd_bem_find_surface(FIFFV_BEM_SURF_ID_BRAIN)) == nullptr)
531 return FAIL;
532
533 if (!UTILSLIB::Sphere::fit_sphere_to_points(inner_skull->rr,simplex_size,r0_vec,R))
534 return FAIL;
535 d->r0 = r0_vec.head<3>();
536
537 FiffCoordTrans::apply_trans(d->r0.data(),*d->mri_head_t,true);
538 qInfo("Fitted sphere model origin : %6.1f %6.1f %6.1f mm rad = %6.1f mm.",
539 1000*d->r0[0],1000*d->r0[1],1000*d->r0[2],1000*R);
540 }
541 d->bem_funcs = f = new_dipole_fit_funcs();
542 if (d->nmeg > 0) {
543 /*
544 * Use the new compensated field computation
545 * It works the same way independent of whether or not the compensation is in effect
546 */
547 comp = FwdCompData::fwd_make_comp_data(comp_data,d->meg_coils.get(),comp_coils,
548 FwdBemModel::fwd_bem_field,nullptr,nullptr,d->bem_model.get());
549 if (!comp)
550 return FAIL;
551 qInfo("Compensation setup done.");
552
553 qInfo("MEG solution matrix...");
554 if (d->bem_model->fwd_bem_specify_coils(d->meg_coils.get()) == FAIL)
555 return FAIL;
556 if (d->bem_model->fwd_bem_specify_coils(comp->comp_coils) == FAIL)
557 return FAIL;
558 qInfo("[done]");
559
560 f->meg_field = FwdCompData::fwd_comp_field;
561 f->meg_vec_field = nullptr;
562 f->meg_client = comp;
563 f->meg_client_free = [](void* d) { delete static_cast<FwdCompData*>(d); };
564 }
565 if (d->neeg > 0) {
566 qInfo("\tEEG solution matrix...");
567 if (d->bem_model->fwd_bem_specify_els(d->eeg_els.get()) == FAIL)
568 return FAIL;
569 qInfo("[done]");
570 f->eeg_pot = FwdBemModel::fwd_bem_pot_els;
571 f->eeg_vec_pot = nullptr;
572 f->eeg_client = d->bem_model.get();
573 }
574 }
575 if (d->neeg > 0 && !d->eeg_model) {
576 qCritical("EEG sphere model not defined.");
577 return FAIL;
578 }
579 d->sphere_funcs = f = new_dipole_fit_funcs();
580 if (d->neeg > 0) {
581 d->eeg_model->r0 = d->r0;
582 f->eeg_pot = FwdEegSphereModel::fwd_eeg_spherepot_coil;
583 f->eeg_vec_pot = FwdEegSphereModel::fwd_eeg_spherepot_coil_vec;
584 f->eeg_client = d->eeg_model.get();
585 }
586 if (d->nmeg > 0) {
587 /*
588 * Use the new compensated field computation
589 * It works the same way independent of whether or not the compensation is in effect
590 */
591 comp = FwdCompData::fwd_make_comp_data(comp_data,d->meg_coils.get(),comp_coils,
592 FwdBemModel::fwd_sphere_field,
593 FwdBemModel::fwd_sphere_field_vec,
594 nullptr,
595 d->r0.data());
596 if (!comp)
597 return FAIL;
598 f->meg_field = FwdCompData::fwd_comp_field;
599 f->meg_vec_field = FwdCompData::fwd_comp_field_vec;
600 f->meg_client = comp;
601 f->meg_client_free = [](void* d) { delete static_cast<FwdCompData*>(d); };
602 }
603 qInfo("Sphere model origin : %6.1f %6.1f %6.1f mm.",
604 1000*d->r0[0],1000*d->r0[1],1000*d->r0[2]);
605 /*
606 * Finally add the magnetic dipole fitting functions (for special purposes)
607 */
608 d->mag_dipole_funcs = f = new_dipole_fit_funcs();
609 if (d->nmeg > 0) {
610 /*
611 * Use the new compensated field computation
612 * It works the same way independent of whether or not the compensation is in effect
613 */
614 comp = FwdCompData::fwd_make_comp_data(comp_data,d->meg_coils.get(),comp_coils,
615 FwdBemModel::fwd_mag_dipole_field,
616 FwdBemModel::fwd_mag_dipole_field_vec,
617 nullptr,
618 nullptr);
619 if (!comp)
620 return FAIL;
621 f->meg_field = FwdCompData::fwd_comp_field;
622 f->meg_vec_field = FwdCompData::fwd_comp_field_vec;
623 f->meg_client = comp;
624 f->meg_client_free = [](void* d) { delete static_cast<FwdCompData*>(d); };
625 }
626 f->eeg_pot = FwdBemModel::fwd_mag_dipole_field;
627 f->eeg_vec_pot = FwdBemModel::fwd_mag_dipole_field_vec;
628 /*
629 * Select the appropriate fitting function
630 */
631 d->funcs = !d->bemname.isEmpty() ? d->bem_funcs : d->sphere_funcs;
632
633 qWarning("");
634 return OK;
635}

642std::unique_ptr<MNECovMatrix> InvDipoleFitData::ad_hoc_noise(FwdCoilSet *meg, FwdCoilSet *eeg, float grad_std, float mag_std, float eeg_std)
643{
644 int nchan;
645 Eigen::VectorXd stds;
646 QStringList names, ch_names;
647 int k,n;
648
649 qInfo("Using standard noise values "
650 "(MEG grad : %6.1f fT/cm MEG mag : %6.1f fT EEG : %6.1f uV)\n",
651 1e13*grad_std,1e15*mag_std,1e6*eeg_std);
652
653 nchan = 0;
654 if (meg)
655 nchan = nchan + meg->ncoil();
656 if (eeg)
657 nchan = nchan + eeg->ncoil();
658
659 stds.resize(nchan);
660
661 n = 0;
662 if (meg) {
663 for (k = 0; k < meg->ncoil(); k++, n++) {
664 if (meg->coils[k]->is_axial_coil()) {
665 stds[n] = mag_std*mag_std;
666#ifdef TEST_REF
667 if (meg->coils[k]->type == FIFFV_COIL_CTF_REF_MAG ||
668 meg->coils[k]->type == FIFFV_COIL_CTF_REF_GRAD ||
669 meg->coils[k]->type == FIFFV_COIL_CTF_OFFDIAG_REF_GRAD)
670 stds[n] = 1e6*stds[n];
671#endif
672 }
673 else
674 stds[n] = grad_std*grad_std;
675 ch_names.append(meg->coils[k]->chname);
676 }
677 }
678 if (eeg) {
679 for (k = 0; k < eeg->ncoil(); k++, n++) {
680 stds[n] = eeg_std*eeg_std;
681 ch_names.append(eeg->coils[k]->chname);
682 }
683 }
684 names = ch_names;
685 return MNECovMatrix::create(FIFFV_MNE_NOISE_COV,nchan,names,Eigen::VectorXd(),stds);
686}

851{
852 int nave,j,k;
853 float nonsel_w = 30;
854 int min_nchan = 20;
855
856 if (!f || !f->noise_orig)
857 return OK;
858 if (!d)
859 nave = 1;
860 else {
861 if (d->nave < 0)
862 nave = d->meas->current->nave;
863 else
864 nave = d->nave;
865 }
866 /*
867 * Channel selection
868 */
869 if (d) {
870 std::vector<float> wVec(f->noise_orig->ncov);
871 float *w = wVec.data();
872 int nomit_meg,nomit_eeg,nmeg,neeg;
873
874 nmeg = neeg = 0;
875 nomit_meg = nomit_eeg = 0;
876 for (k = 0; k < f->noise_orig->ncov; k++) {
877 if (f->noise_orig->ch_class[k] == MNE_COV_CH_EEG)
878 neeg++;
879 else
880 nmeg++;
881 if (is_selected_in_data(d,f->noise_orig->names[k]))
882 w[k] = 1.0;
883 else {
884 w[k] = nonsel_w;
885 if (f->noise_orig->ch_class[k] == MNE_COV_CH_EEG)
886 nomit_eeg++;
887 else
888 nomit_meg++;
889 }
890 }
891 f->noise.reset();
892 if (nmeg > 0 && nmeg-nomit_meg > 0 && nmeg-nomit_meg < min_nchan) {
893 qCritical("Too few MEG channels remaining");
894 return FAIL;
895 }
896 if (neeg > 0 && neeg-nomit_eeg > 0 && neeg-nomit_eeg < min_nchan) {
897 qCritical("Too few EEG channels remaining");
898 return FAIL;
899 }

901 if (nomit_meg+nomit_eeg > 0) {
902 if (f->noise->cov.size() > 0) {
903 for (j = 0; j < f->noise->ncov; j++)
904 for (k = 0; k <= j; k++) {
905 f->noise->cov[MNECovMatrix::lt_packed_index(j,k)] *= w[j]*w[k];
906 }
907 }
908 else {
909 for (j = 0; j < f->noise->ncov; j++) {
910 f->noise->cov_diag[j] *= w[j]*w[j];
911 }
912 }
913 }
914 }
915 else {
916 if (f->noise && f->nave == nave)
917 return OK;
918 f->noise = f->noise_orig->dup();
919 }
920
921 return scale_dipole_fit_noise_cov(f,nave);
922}

1312
1316InvDipoleForward* dipole_forward(InvDipoleFitData* d,
1317 float **rd,
1318 int ndip,
1319 InvDipoleForward* old)
1320{
1321 InvDipoleForward* res;
1322 float S[3];
1323 int k,p;
1324 /*
1325 * Allocate data if necessary
1326 */
1327 if (old && old->ndip == ndip && old->nch == d->nmeg+d->neeg) {
1328 res = old;
1329 }
1330 else {
1331 delete old; old = nullptr;
1332 res = new InvDipoleForward;
1333 int nch = d->nmeg + d->neeg;
1334 int m = 3 * ndip;
1335 res->fwd.resize(m, nch);
1336 res->uu.resize(m, nch);
1337 res->vv.resize(m, m);
1338 res->sing.resize(m);
1339 res->scales.resize(m);
1340 res->rd.resize(ndip, 3);
1341 res->nch = nch;
1342 res->ndip = ndip;
1343 }
1344
1345 for (k = 0; k < ndip; k++) {
1346 res->rd.row(k) = Eigen::Map<const Eigen::Vector3f>(rd[k]);
1347 /*
1348 * Calculate the field of three orthogonal dipoles
1349 */
1350 Eigen::MatrixXf this_fwd(d->nmeg + d->neeg, 3);
1351 Eigen::Map<const Eigen::Vector3f> rd_k(rd[k]);
1352 if ((InvDipoleFitData::compute_dipole_field(*d,rd_k,true,this_fwd)) == FAIL) {
1353 if (!old)
1354 delete res;
1355 return nullptr;
1356 }
1357 for (int p = 0; p < 3; p++)
1358 res->fwd.row(3*k+p) = this_fwd.col(p).transpose();
1359 /*
1360 * Choice of column normalization
1361 * (componentwise normalization is not recommended)
1362 */
1363 if (d->column_norm == COLUMN_NORM_LOC || d->column_norm == COLUMN_NORM_COMP) {
1364 for (p = 0; p < 3; p++)
1365 S[p] = res->fwd.row(3*k+p).squaredNorm();
1366 if (d->column_norm == COLUMN_NORM_COMP) {
1367 for (p = 0; p < 3; p++)
1368 res->scales[3*k+p] = sqrt(S[p]);
1369 }
1370 else {
1371 /*
1372 * Divide by three or not?
1373 */
1374 res->scales[3*k+0] = res->scales[3*k+1] = res->scales[3*k+2] = sqrt(S[0]+S[1]+S[2])/3.0;
1375 }
1376 for (p = 0; p < 3; p++) {
1377 if (res->scales[3*k+p] > 0.0) {
1378 res->scales[3*k+p] = 1.0/res->scales[3*k+p];
1379 res->fwd.row(3*k+p) *= res->scales[3*k+p];
1380 }
1381 else
1382 res->scales[3*k+p] = 1.0;
1383 }
1384 }
1385 else {
1386 res->scales[3*k] = 1.0;
1387 res->scales[3*k+1] = 1.0;
1388 res->scales[3*k+2] = 1.0;
1389 }
1390 }
1391
1392 /*
1393 * SVD: A = U · Σ · V^T where A is m×n (3*ndip × nch)
1394 * uu stores right singular vectors (V^T rows, length nch) for data-space projections
1395 * vv stores left singular vectors (U^T rows, length m) for dipole-moment reconstruction
1396 */
1397 {
1398 int m = 3*ndip;
1399 int n = d->nmeg+d->neeg;
1400 int udim = std::min(m,n);
1401 JacobiSVD<MatrixXf> svd(res->fwd, ComputeFullU | ComputeFullV);
1402 res->sing = svd.singularValues();

1453static int find_best_guess(const Eigen::Ref<const Eigen::VectorXf>& B,
1454 int nch,

1495static MatrixXf make_initial_dipole_simplex(const Eigen::Vector3f& r0,
1496 float size)
1497{
1498 /*
1499 * For this definition of a regular tetrahedron, see
1500 *
1501 * http://mathworld.wolfram.com/Tetrahedron.html
1502 *
1503 */
1504 float x = sqrt(3.0f)/3.0f;
1505 float r = sqrt(6.0f)/12.0f;
1506 float R = 3*r;
1507 float d = x/2.0f;
1508 float rr[][3] = { { x , 0.0f, -r },
1509 { -d, 0.5f, -r },
1510 { -d, -0.5f, -r },
1511 { 0.0f, 0.0f, R } };
1512
1513 MatrixXf simplex = MatrixXf::Zero(4, 3);
1514
1515 for (int j = 0; j < 4; j++) {
1516 simplex.row(j) = Eigen::Map<const Vector3f>(rr[j]).transpose() * size + r0.transpose();
1517 }
1518 return simplex;
1519}

1536static int fit_Q(InvDipoleFitData* fit,
1537 const Eigen::Ref<const Eigen::VectorXf>& B,
1538 const Eigen::Vector3f& rd,
1539 float limit,
1540 Eigen::Vector3f& Q,
1541 int &ncomp,
1542 float &res)
1543{
1544 int c;
1545 InvDipoleForward* fwd = InvDipoleFitData::dipole_forward_one(fit,rd,nullptr);
1546 float Bm2,one;
1547
1548 if (!fwd)
1549 return FAIL;
1550
1551 ncomp = fwd->sing[2]/fwd->sing[0] > limit ? 3 : 2;
1552
1553 Q.setZero();
1554 for (c = 0, Bm2 = 0.0; c < ncomp; c++) {
1555 one = fwd->uu.row(c).dot(B);
1556 Q += (one/fwd->sing[c]) * fwd->vv.row(c).head(3).transpose();
1557 Bm2 = Bm2 + one*one;
1558 }
1559 /*
1560 * Counteract the effect of column normalization
1561 */
1562 for (c = 0; c < 3; c++)
1563 Q[c] = fwd->scales[c]*Q[c];
1564 res = B.squaredNorm() - Bm2;
1565
1566 delete fwd;
1567
1568 return OK;
1569}

1572// Dipole fitting - main entry point
1584bool InvDipoleFitData::fit_one(InvDipoleFitData* fit,
1585 InvGuessData* guess,
1586 float time,
1587 Eigen::Ref<Eigen::VectorXf> B,
1588 int verbose,
1589 InvEcd& res
1590 )
1591{
1592 VectorXf vals(4); /* Values at the vertices */
1593 float limit = 0.2f; /* (pseudo) radial component omission limit */
1594 float size = 1e-2f; /* Size of the initial simplex */
1595 float ftol[] = { 1e-2f, 1e-2f }; /* Tolerances on the the two passes */
1596 float atol[] = { 0.2e-3f, 0.2e-3f }; /* If dipole movement between two iterations is less than this,
1597 we consider to have converged */
1598 int ntol = 2;
1599 int max_eval = 1000; /* Limit for fit function evaluations */
1600 int report_interval = verbose ? 1 : -1; /* How often to report the intermediate result */
1601
1602 int best;
1603 float good,final_val;
1604 Eigen::Vector3f rd_final, Q;
1605 FitDipUserRec user;
1606 int k,neval,neval_tot,nchan,ncomp;
1607 int fit_fail;
1608 Vector3f rd_guess;
1609
1610 nchan = fit->nmeg+fit->neeg;
1611 user.fwd = nullptr;
1612
1613 if (fit->proj && fit->proj->project_vector(B.data(),nchan,true) == FAIL)
1614 return false;
1615
1616 if (fit->noise->whiten_vector(B,B,nchan) == FAIL)
1617 return false;
1618 /*
1619 * Get the initial guess
1620 */
1621 if (find_best_guess(B,nchan,guess,limit,best,good) < 0)
1622 return false;
1623
1624 user.limit = limit;
1625 user.B = B.data();
1626 user.B2 = B.squaredNorm();
1627 user.fwd = nullptr;
1628 user.report_dim = false;
1629 fit->user = &user;
1630
1631 rd_guess = guess->rr.row(best).transpose();
1632 rd_final = rd_guess;
1633
1634 neval_tot = 0;
1635 fit_fail = false;
1636 for (k = 0; k < ntol; k++) {
1637 /*
1638 * Do first pass with the sphere model
1639 */
1640 if (k == 0)
1641 fit->funcs = fit->sphere_funcs;
1642 else
1643 fit->funcs = !fit->bemname.isEmpty() ? fit->bem_funcs : fit->sphere_funcs;
1644
1645 MatrixXf simplexMat = make_initial_dipole_simplex(rd_guess,size);
1646 for (int p = 0; p < 4; p++)
1647 vals[p] = fit_eval(simplexMat.row(p),fit);
1648 if (!UTILSLIB::SimplexAlgorithm::simplex_minimize<float>(
1649 simplexMat, /* The initial simplex */
1650 vals, /* Function values at the vertices */
1651 ftol[k], /* Relative convergence tolerance for the target function */
1652 atol[k], /* Absolute tolerance for the change in the parameters */
1653 fit_eval, /* The function to be evaluated */
1654 fit, /* Data to be passed to the above function in each evaluation */
1655 max_eval, /* Maximum number of function evaluations */
1656 neval, /* Number of function evaluations */
1657 report_interval, /* How often to report (-1 = no_reporting) */
1658 dipole_report_func)) {
1659 if (k == 0) {
1660 delete user.fwd;
1661 return false;
1662 }
1663 else {
1664 float rv = 2.0f*(vals.maxCoeff()-vals.minCoeff())/(vals.maxCoeff()+vals.minCoeff());
1665 qWarning("Warning (t = %8.1f ms) : g = %6.1f %% final val = %7.3f rtol = %f",
1666 1000*time,100*(1 - vals[0]/user.B2),vals[0],rv);
1667 fit_fail = true;
1668 }
1669 }
1670 rd_final = simplexMat.row(0).transpose();
1671 rd_guess = simplexMat.row(0).transpose();
1672
1673 neval_tot += neval;
1674 final_val = vals[0];
1675 }
1676 /*
1677 * Confidence limits should be computed here
1678 */
1679 /*
1680 * Compute the dipole moment at the final point
1681 */
1682 if (fit_Q(fit,B,rd_final,user.limit,Q,ncomp,final_val) == OK) {
1683 res.time = time;
1684 res.valid = true;
1685 res.rd = rd_final;
1686 res.Q = Q;
1687 res.good = 1.0 - final_val/user.B2;
1688 if (fit_fail)
1689 res.good = -res.good;
1690 res.khi2 = final_val;
1691 if (fit->proj)
1692 res.nfree = nchan-3-ncomp-fit->proj->nvec;
1693 else
1694 res.nfree = nchan-3-ncomp;
1695 res.neval = neval_tot;
1696 }
1697 else {
1698 delete user.fwd;
1699 return false;
1700 }
1701 delete user.fwd;
1702
1703 return true;
1704}
