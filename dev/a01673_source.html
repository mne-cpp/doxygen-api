83void appendJsonString(std::string& out, std::string_view text)
84{
85 out += '"';
86 for (const char c : text) {
87 if (c == '"' || c == '\\') {
88 out += '\\';
89 out += c;
90 } else if (static_cast<unsigned char>(c) < 0x20) {
91 out += std::format("\\u{:04x}", static_cast<unsigned>(c));
92 } else {
93 out += c;
94 }
95 }
96 out += '"';
97}

99AcqExpected<Config> parseConfig(const AcqUri& uri)
100{
101 auto thread = paramThreadPolicy(uri);
102 if (!thread) {
103 return thread.error();
104 }
105 Config c;
106 c.board = uri.target;
107 if (c.board.empty()) {
108 return invalid("brainflow:// needs a board id or name, e.g. brainflow://synthetic");
109 }
110 if (const auto library = uri.param("library")) {
111 c.library = {std::string(*library)};
112 } else {
113 c.library = brainflow_library_candidates();
114 }
115 // Every parameter is in the JSON: BrainFlow refuses a missing one
116 c.params = "{";
117 for (const char* key : kTextParams) {
118 c.params += std::format("\"{}\":", key);
119 appendJsonString(c.params, uri.param(key).value_or(""));
120 c.params += ',';
121 }
122 for (const char* key : kIntParams) {
123 const std::string_view name(key);
124 const auto value = paramInt(uri, key, name == "master_board" ? kNoBoard : 0, name == "master_board" ? kNoBoard : 0, INT32_MAX);
125 if (!value) {
126 return value.error();
127 }
128 c.params += std::format("\"{}\":{},", key, *value);
129 }
130 c.params.back() = '}';
131 c.thread = std::move(*thread);
132 return c;
133}

136AcqExpected<int> boardId(const BrainFlowLibrary& library, const std::string& board)
137{
138 int id = 0;
139 const auto [end, error] = std::from_chars(board.data(), board.data() + board.size(), id);
140 if (error == std::errc() && end == board.data() + board.size()) {
141 return id;
142 }
143 return library.find_board(board);

146class BrainFlowBackend final : public AcqBackend
147{
148public:
149 static AcqExpected<std::unique_ptr<AcqBackend>> open(const AcqUri& uri)
150 {
151 auto config = parseConfig(uri);
152 if (!config) {
153 return config.error();
154 }
155 auto library = BrainFlowLibrary::load(config->library);
156 if (!library) {
157 return library.error();
158 }
159 auto board = boardId(**library, config->board);
160 if (!board) {
161 return board.error();
162 }
163 auto description = (*library)->description(*board);
164 if (!description) {
165 return description.error();
166 }
167 if (description->sampling_rate <= 0 || description->rows == 0) {
168 return AcqError(AcqErrorCode::DeviceIo, std::format("BrainFlow board {} describes no samples", *board));
169 }
170 if (const int code = (*library)->prepare_session(*board, config->params.c_str()); code != 0) {
171 if (code == kUnsupportedBoard) { // described, but not built into this platform's BrainFlow
172 return AcqError(AcqErrorCode::PlatformUnsupported,
173 std::format("BrainFlow {} on this platform does not support board {} ({})", (*library)->version(), description->name, *board));
174 }
175 return brainflow_error(code, "prepare_session");
176 }
177 std::unique_ptr<BrainFlowBackend> backend(new BrainFlowBackend(std::move(*config), std::move(*library), *board, *description, uri.toString()));
178 AcqStreamOptions options;
179 options.capacitySamples = static_cast<std::size_t>(std::max(4 * description->sampling_rate, 1024));
180 auto writer = backend->addInputStream(brainflow_stream_info(*description, std::format("brainflow:{}", *board)), options);
181 if (!writer) {
182 return writer.error();
183 }
184 backend->m_writer = BrainFlowRowWriter(**writer, *description);
185 return std::unique_ptr<AcqBackend>(std::move(backend));
186 }

188 ~BrainFlowBackend() override
189 {
190 stop();
191 m_library->release_session(m_board, m_config.params.c_str());
192 }

194 BrainFlowBackend(const BrainFlowBackend&) = delete;
195 BrainFlowBackend& operator=(const BrainFlowBackend&) = delete;

197 AcqDeviceDescription description() const override
198 {
199 return {"brainflow", m_library->version(), std::format("BrainFlow {} (board {})", m_name, m_board), m_uri};
200 }

202 AcqError start() override
203 {
204 if (const int code = m_library->start_stream(kBrainFlowBuffer, "", m_board, m_config.params.c_str()); code != 0) {
205 return brainflow_error(code, "start_stream");
206 }
207 m_stop.store(false, std::memory_order_relaxed);
208 m_writer.restart();
209 auto thread = start_thread(m_config.thread, [this] { run(); });
210 if (!thread) {
211 m_library->stop_stream(m_board, m_config.params.c_str());
212 return thread.error();
213 }
214 m_thread = std::move(*thread);
215 return {};
216 }
217
218 void stop() override
219 {
220 m_stop.store(true, std::memory_order_release);
221 if (m_thread.joinable()) {
222 m_thread.join();
223 m_library->stop_stream(m_board, m_config.params.c_str());
224 }
225 }

227private:
228 BrainFlowBackend(Config config, std::shared_ptr<const BrainFlowLibrary> library, int board, const BrainFlowDescription& description, std::string uri)
229 : m_config(std::move(config))
230 , m_library(std::move(library))
231 , m_board(board)
232 , m_name(description.name)
233 , m_uri(std::move(uri))
234 , m_rows(description.rows)
235 , m_max_block(static_cast<std::size_t>(std::max(description.sampling_rate, 256)))
236 {
237 m_data.reserve(m_rows * m_max_block);
238 }
239
240 void run()
241 {
242 while (!m_stop.load(std::memory_order_acquire)) {
243 int available = 0;
244 if (const int code = m_library->get_board_data_count(0, &available, m_board, m_config.params.c_str()); code != 0) {
245 reportError(brainflow_error(code, "get_board_data_count"));
246 return;
247 }
248 if (available <= 0) {
249 std::this_thread::sleep_for(kPoll);
250 continue;
251 }
252 const std::size_t samples = std::min(static_cast<std::size_t>(available), m_max_block);
253 m_data.resize(m_rows * samples);
254 if (const int code = m_library->get_board_data(static_cast<int>(samples), 0, m_data.data(), m_board, m_config.params.c_str()); code != 0) {
255 reportError(brainflow_error(code, "get_board_data"));
256 return;
257 }
258 // BrainFlow's buffer is row after row
259 const AcqError written = m_writer.write(m_data.data(), samples, 1, samples, [this] { return m_stop.load(std::memory_order_acquire); });
260 if (written.code() == AcqErrorCode::DeviceStopped) {
261 return;
262 }
263 if (!written.ok()) {
264 reportError(written);
265 return;
266 }
267 }
268 }
