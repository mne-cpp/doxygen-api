118std::optional<SpectrumFigure::Readout> SpectrumFigure::readout() const
119{
120 const double plotWidth = m_width - name_width;
121 if (!m_pointer || plotWidth <= 0.0 || m_pointer->x < name_width || m_pointer->x > m_width || m_pointer->y < 0.0) {
122 return std::nullopt;
123 }
124 Readout readout;
125 readout.row = static_cast<int>(m_pointer->y / row_height);
126 if (readout.row >= m_table.row_count()) {
127 return std::nullopt;
128 }
129 const std::optional<Eigen::RowVectorXd> spectrum = m_table.spectrum(readout.row);
130 const Eigen::RowVectorXd& bounded = m_table.freq_scale_bound();
131 const int lower = m_table.lower_bin();
132 const int upper = m_table.upper_bin();
133 if (!spectrum || spectrum->size() == 0 || bounded.size() <= upper || upper < lower) {
134 return std::nullopt;
135 }
136 // The bin nearest to the pointer between the bounds
137 const double rate = (m_pointer->x - name_width) / plotWidth;
138 double best = std::numeric_limits<double>::max();
139 for (int bin = lower; bin <= upper && bin < spectrum->size(); ++bin) {
140 if (const double distance = std::abs(bounded[bin] - rate); distance < best) {
141 best = distance;
142 readout.bin = bin;
143 }
144 }
145 if (readout.bin < 0) {
146 return std::nullopt;
147 }
148 readout.value = (*spectrum)[readout.bin];
149 readout.frequency = bin_frequency(readout.bin);
150 return readout;
151}

162void SpectrumFigure::draw(Canvas& canvas, const TextMetrics& metrics, const Font& font) const
163{
164 canvas.set_font(font);
165 // Only the rows in the canvas, as long as the host only scrolls and scales
166 int first = 0;
167 int last = m_table.row_count() - 1;
168 const Transform& transform = canvas.transform();
169 if (transform.m12 == 0.0 && transform.m21 == 0.0 && transform.m22 > 0.0) {
170 first = std::max(first, static_cast<int>(std::floor(-transform.dy / transform.m22 / row_height)));
171 last = std::min(last, static_cast<int>(std::floor((canvas.height() - transform.dy) / transform.m22 / row_height)));
172 }
173 for (int row = first; row <= last; ++row) {
174 draw_row(canvas, metrics, font, row);
175 }
176}

213std::vector<std::pair<int, double>> SpectrumFigure::grid() const
214{
215 std::vector<std::pair<int, double>> lines;
216 const int bins = m_table.bins();
217 if (!m_table.info() || bins == 0) {
218 return lines;
219 }
220 const double nyquist = m_table.info()->sfreq / 2.0;
221 const double resolution = nyquist / bins;
222 const auto bin_of = [resolution](double frequency) {
223 return static_cast<int>(std::floor(frequency / resolution));
224 };
225 lines.emplace_back(0, 0.0);
226 if (m_scale == SpectrumTable::Scale::Log) {
227 // Decades: 1, 10, 100 Hz, ...
228 const auto decades = static_cast<int>(std::ceil(std::log10(nyquist)));
229 for (int decade = 0; decade < decades; ++decade) {
230 const double frequency = std::pow(10.0, decade);
231 lines.emplace_back(bin_of(frequency), frequency);
232 }
233 } else {
234 for (int line = 1; line < kLinearLines; ++line) {
235 const int bin = bin_of(line * nyquist / kLinearLines);
236 lines.emplace_back(bin, bin_frequency(bin));
237 }
238 }
239 return lines;
240}

244double SpectrumFigure::bin_frequency(int bin) const
245{
246 const Eigen::RowVectorXd& axis = m_table.freq_scale();
247 if (!m_table.info() || bin < 0 || bin >= axis.size()) {
248 return 0.0;
249 }
250 // Where the axis places the bin: its last bin is at the Nyquist frequency
251 const double nyquist = m_table.info()->sfreq / 2.0;
252 if (m_scale == SpectrumTable::Scale::Log) {
253 return std::pow(10.0, axis[bin] * std::log10(nyquist + 1.0)) - 1.0;
254 }
255 return axis[bin] * nyquist;
256}

260void SpectrumFigure::draw_row(Canvas& canvas, const TextMetrics& metrics, const Font& font, int row) const
261{
262 const double top = row * row_height;
263 const double plotWidth = m_width - name_width;
264 const Color text{0, 0, 0};
265
266 canvas.set_pen(text);
267 canvas.draw_text({4.0, top, name_width - 8.0, row_height}, Align::Left | Align::VCenter, m_table.channel_name(row));
268 canvas.set_pen(Color{220, 220, 220});
269 canvas.draw_line({0.0, top + row_height - 0.5}, {m_width, top + row_height - 0.5});
270
271 const std::optional<Eigen::RowVectorXd> spectrum = m_table.spectrum(row);
272 const Eigen::RowVectorXd& bounded = m_table.freq_scale_bound();
273 const int lower = m_table.lower_bin();
274 const int upper = std::min(m_table.upper_bin(), static_cast<int>(bounded.size()) - 1);
275 if (!spectrum || spectrum->size() <= upper || upper <= lower || plotWidth <= 0.0) {
276 return;
277 }
278
279 // Frequency grid with labels at the top
280 Pen grid{text, 0.5, LineStyle::Dot};
281 const double ascent = metrics.ascent(font);
282 for (const auto& [bin, frequency] : this->grid()) {
283 if (bin <= lower || bin >= upper) {
284 continue;
285 }
286 const double x = name_width + bounded[bin] * plotWidth;
287 canvas.set_pen(grid);
288 canvas.draw_line({x, top}, {x, top + row_height});
289 canvas.set_pen(text);
290 canvas.draw_text({x + 2.0, top + ascent}, number(frequency) + "Hz");
291 }
292
293 // The spectrum between the bounds, scaled to its range with a pixel to spare at both edges
294 const Eigen::RowVectorXd visible = spectrum->segment(lower, upper - lower + 1);
295 const double minimum = visible.minCoeff();
296 const double maximum = visible.maxCoeff();
297 const double scale = (row_height - 2.0) / (maximum > minimum ? maximum - minimum : 1.0);
298 const double bottom = top + row_height - 1.0;
299 std::vector<Point> points;
300 points.reserve(static_cast<std::size_t>(upper - lower) + 1);
301 for (int bin = lower; bin <= upper; ++bin) {
302 points.push_back({name_width + bounded[bin] * plotWidth, bottom - ((*spectrum)[bin] - minimum) * scale});
303 }
304 const std::optional<Readout> current = readout();
305 const bool hovered = current && current->row == row;
306 Color curve = m_table.frozen() ? Color{128, 128, 128} : Color{0, 0, 128};
307 if (hovered) {
308 curve = m_table.frozen() ? Color{128, 0, 0} : Color{255, 0, 0};
309 }
310 canvas.save();
311 canvas.set_antialiasing(true);
312 canvas.set_pen(Pen{curve, 1.0});
313 canvas.draw_polyline(points);
314 canvas.restore();
315
316 if (!hovered) {
317 return;
318 }
319 // Read-out: a line at the pointer and the bin's value and frequency beside it, inside the row
320 const Point at = *m_pointer;
321 canvas.set_pen(Pen{Color{128, 128, 128}, 1.0, LineStyle::Dash});
322 canvas.draw_line({at.x, top}, {at.x, top + row_height});
323 const std::string label = number(current->value) + " [dB], " + number(current->frequency) + " [Hz]";
324 const double width = metrics.advance(font, label);
325 const double x = at.x + kReadoutGap + width > m_width ? at.x - kReadoutGap - width : at.x + kReadoutGap;
326 const double y = std::clamp(at.y - kReadoutGap, top + ascent, top + row_height - 2.0);
327 canvas.set_pen(text);
328 canvas.draw_text({x, y}, label);
329}
