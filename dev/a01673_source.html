39std::vector<std::string_view> split_lines(std::string_view text)
40{
41 std::vector<std::string_view> lines;
42 std::size_t start = 0;
43 while (true) {
44 const std::size_t end = text.find('\n', start);
45 lines.push_back(text.substr(start, end == std::string_view::npos ? std::string_view::npos : end - start));
46 if (end == std::string_view::npos) {
47 return lines;
48 }
49 start = end + 1;
50 }
51}

188PixelSize BrainViewOverlay::label_size(const LabelStyle& style, std::string_view text, const TextMetrics& metrics)
189{
190 double width = 0.0;
191 const std::vector<std::string_view> lines = split_lines(text);
192 for (std::string_view line : lines) {
193 width = std::max(width, metrics.advance(style.font, line));
194 }
195 const double height = static_cast<double>(lines.size()) * metrics.height(style.font);
196 const int frame = style.border_px;
197 return {static_cast<int>(std::ceil(width)) + 2 * (style.pad_x + frame),
198 static_cast<int>(std::ceil(height)) + 2 * (style.pad_y + frame)};
199}

203void BrainViewOverlay::update_layout(const LayoutInput& input, const TextMetrics& metrics)
204{
205 const PaneLayout default_layout;
206 const PaneLayout& layout = input.layout ? *input.layout : default_layout;
207 const PixelSize area = input.size;
208
209 m_fps.visible = input.info_visible;
210 const PixelSize fps_size = label_size(style(Item::FpsLabel), m_fps.text, metrics);
211 m_fps.geometry = {area.width - fps_size.width - 10, area.height - fps_size.height - 2, fps_size.width, fps_size.height};
212
213 m_single_info.visible = !input.multi_view && input.info_visible;
214 if (m_single_info.visible) {
215 const PixelSize info_size = label_size(style(Item::SingleViewInfoLabel), m_single_info.text, metrics);
216 m_single_info.geometry = {area.width - info_size.width - 8, 8, info_size.width, info_size.height};
217 }
218
219 // The region label keeps a fixed size; long names are clipped
220 m_region.geometry = {10, input.multi_view ? 38 : 10, 300, 30};
221
222 for (std::size_t i = 0; i < m_names.size(); ++i) {
223 m_names[i].visible = false;
224 m_names[i].selected = input.multi_view && input.edit_target == static_cast<int>(i);
225 m_infos[i].visible = false;
226 }
227 m_vertical.visible = false;
228 m_horizontal.visible = false;
229
230 const int pane_count = static_cast<int>(input.viewports.size());
231 if (!input.multi_view) {
232 return;
233 }
234
235 if (pane_count > 1) {
236 const PaneLayout::Separators separators = layout.separator_geometries(pane_count, area);
237 m_vertical.geometry = separators.vertical;
238 m_vertical.visible = !separators.vertical.empty();
239 m_horizontal.geometry = separators.horizontal;
240 m_horizontal.visible = !separators.horizontal.empty();
241 }
242
243 for (int slot = 0; slot < pane_count; ++slot) {
244 const int vp = input.viewports[static_cast<std::size_t>(slot)];
245 if (vp < 0 || vp >= viewport_count()) {
246 continue;
247 }
248 Label& name = m_names[static_cast<std::size_t>(vp)];
249 Label& info = m_infos[static_cast<std::size_t>(vp)];
250
251 const int preset = vp < static_cast<int>(input.presets.size()) ? input.presets[static_cast<std::size_t>(vp)] : 0;
252 name.text = std::string(view_preset_name(clamp_view_preset(preset)));
253
254 const PixelRect pane = layout.slot_rect(slot, pane_count, area);
255 const PixelSize name_size = label_size(style(Item::ViewportNameLabel, name.selected), name.text, metrics);
256 name.geometry = {pane.x + 8, pane.y + 8, name_size.width, name_size.height};
257 name.visible = true;
258
259 const PixelSize info_size = label_size(style(Item::ViewportInfoLabel), info.text, metrics);
260 info.geometry = {pane.x + pane.width - info_size.width - 8, pane.y + 8, info_size.width, info_size.height};
261 info.visible = input.info_visible;
262 }
263}

267void BrainViewOverlay::draw_label(Canvas& canvas, const LabelStyle& style, const Label& label, const TextMetrics& metrics) const
268{
269 const Rect box = label.geometry.to_rect();
270 canvas.save();
271 canvas.clip_rect(box);
272 if (style.background.a > 0 || style.border_px > 0) {
273 const double inset = style.border_px * 0.5;
274 canvas.set_pen(style.border_px > 0 ? Pen{style.border, static_cast<double>(style.border_px)} : Pen::none());
275 canvas.set_brush(Brush::solid(style.background));
276 canvas.draw_rounded_rect(box.adjusted(inset, inset, -inset, -inset), style.radius, style.radius);
277 }
278 const double inset_x = style.pad_x + style.border_px;
279 const double inset_y = style.pad_y + style.border_px;
280 const Rect text_box = box.adjusted(inset_x, inset_y, -inset_x, -inset_y);
281 canvas.set_pen(style.color);
282 canvas.set_font(style.font);
283
284 const std::vector<std::string_view> lines = split_lines(label.text);
285 if (lines.size() == 1) {
286 canvas.draw_text(text_box, style.align, label.text);
287 } else {
288 // One text per line keeps the display list free of line breaks
289 const double line_height = metrics.height(style.font);
290 const Align horizontal = has_flag(style.align, Align::Right) ? Align::Right
291 : has_flag(style.align, Align::HCenter) ? Align::HCenter
292 : Align::Left;
293 for (std::size_t i = 0; i < lines.size(); ++i) {
294 const Rect line_box{text_box.x, text_box.y + static_cast<double>(i) * line_height, text_box.width, line_height};
295 canvas.draw_text(line_box, horizontal | Align::Top, lines[i]);
296 }
297 }
298 canvas.restore();
299}

303DisplayList BrainViewOverlay::draw(int width, int height, const TextMetrics& metrics) const
304{
305 Canvas canvas(width, height);
306 canvas.set_antialiasing(true);
307
308 // Separators below the labels
309 for (const Label* separator : {&m_vertical, &m_horizontal}) {
310 if (separator->visible) {
311 canvas.fill_rect(separator->geometry.to_rect(), Brush::solid(m_separator_color));
312 }
313 }
314 if (m_fps.visible) {
315 draw_label(canvas, style(Item::FpsLabel), m_fps, metrics);
316 }
317 if (m_single_info.visible) {
318 draw_label(canvas, style(Item::SingleViewInfoLabel), m_single_info, metrics);
319 }
320 if (m_region.visible) {
321 draw_label(canvas, style(Item::RegionLabel), m_region, metrics);
322 }
323 for (std::size_t i = 0; i < m_names.size(); ++i) {
324 if (m_names[i].visible) {
325 draw_label(canvas, style(Item::ViewportNameLabel, m_names[i].selected), m_names[i], metrics);
326 }
327 if (m_infos[i].visible) {
328 draw_label(canvas, style(Item::ViewportInfoLabel), m_infos[i], metrics);
329 }
330 }
331 return canvas.take();
332}
