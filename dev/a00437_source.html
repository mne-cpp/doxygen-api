57void MNEMneData::computeRegularization(const MNEInverseOperator& inv, const MatrixXd& data)
58{
59 const MatrixXd white = inv.whitener * inv.proj * data;
60 datap = inv.eigen_fields.data * white;
61
62 // Projected-out channels give zero noise eigenvalues; only the others count.
63 const int rank = inv.noise_cov.diag ? inv.noise_cov.dim : static_cast<int>((inv.noise_cov.eig.array() > 0).count());
64 const int ncomp = std::min(rank, static_cast<int>(inv.sing.size()));
65 SNR = white.colwise().squaredNorm().transpose() / rank;
66
67 const ArrayXd sing2 = inv.sing.head(ncomp).array().square();
68 const double limit = mne::Numerics::chi2Isf(1e-3, ncomp - 1);
69 lambda2_est.resize(SNR.size());
70 for (Index t = 0; t < SNR.size(); ++t) {
71 if (SNR(t) < 1.0) {
72 lambda2_est(t) = 100.0;
73 continue;
74 }
75 const ArrayXd alpha = datap.col(t).head(ncomp).array();
76 double lambda2t = 10.0;
77 for (int iter = 0; iter <= 1000; ++iter) {
78 const ArrayXd error = (sing2 > 0.0).select(alpha * lambda2t / (sing2 + lambda2t), alpha);
79 if (error.square().sum() < limit) {
80 break;
81 }
82 lambda2t *= 0.9;
83 }
84 lambda2_est(t) = lambda2t;
85 }
86}

101void MNEMneData::computePredicted(const MNEInverseOperator& inv)
102{
103 const ArrayXd sing2 = inv.sing.array().square();
104 MatrixXd weighted(datap.rows(), datap.cols());
105 for (Index t = 0; t < datap.cols(); ++t) {
106 weighted.col(t) = datap.col(t).array() * sing2 / (sing2 + lambda2(t));
107 }
108 const MatrixXd white = inv.eigen_fields.data.transpose() * weighted;
109 if (inv.noise_cov.diag) {
110 predicted = inv.noise_cov.data.col(0).cwiseSqrt().asDiagonal() * white;
111 } else {
112 // Colorer C^(1/2) = eigvec^T sqrt(eig), the rows of eigvec being the eigenvectors.
113 predicted = inv.noise_cov.eigvec.transpose() * (inv.noise_cov.eig.cwiseMax(0.0).cwiseSqrt().asDiagonal() * white);
114 }
115}
