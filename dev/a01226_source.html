57void LayoutScene::wheelEvent(QGraphicsSceneWheelEvent* event) {
58 m_qvView->setTransformationAnchor(QGraphicsView::AnchorUnderMouse);
59
60 // Scale the view / do the zoom
61 double scaleFactor = 1.15;
62 if(event->delta() > 0) {
63 // Zoom in
64 m_qvView->scale(scaleFactor, scaleFactor);
65 } else {
66 // Zooming out
67 m_qvView->scale(1.0 / scaleFactor, 1.0 / scaleFactor);
