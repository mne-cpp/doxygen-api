300mne::viz::DisplayList ButterflyView::figure() const
301{
302 viz::ButterflyPlot plot;
303 plot.set_background(toSceneColor(m_colCurrentBackgroundColor));
304 if (m_bIsInit && m_pEvokedSetModel) {
305 if (m_pEvokedSetModel->getNumSamples() == 0) {
306 mne::Log::debug("Unable to get data. Returning early.");
307 }
308 const auto [from, to] = m_pEvokedSetModel->getBaselineInfo();
309 plot.set_timing(m_pEvokedSetModel->getNumSamples(), m_pEvokedSetModel->getNumPreStimSamples(), m_pEvokedSetModel->getSamplingFrequency());
310 plot.set_baseline(from, to);
311 plot.set_time_spacers(m_pEvokedSetModel->getNumberOfTimeSpacers() > 0);
312 plot.set_frozen(m_pEvokedSetModel->isFreezed());
313
314 // The selected channels of the shown modalities
315 const auto shown = [this](qint32 row) {
316 switch (m_pEvokedSetModel->getKind(row)) {
317 case FIFFV_MEG_CH: {
318 const qint32 unit = m_pEvokedSetModel->getUnit(row);
319 return (unit == FIFF_UNIT_T_M && detail::valueOr(m_modalityMap, std::string("GRAD"))) || (unit == FIFF_UNIT_T && detail::valueOr(m_modalityMap, std::string("MAG")));
320 }
321 case FIFFV_EEG_CH:
322 return detail::valueOr(m_modalityMap, std::string("EEG"));
323 case FIFFV_EOG_CH:
324 return detail::valueOr(m_modalityMap, std::string("EOG"));
325 case FIFFV_MISC_CH:
326 return detail::valueOr(m_modalityMap, std::string("MISC"));
327 default:
328 return false;
329 }
330 };
331 std::vector<viz::ButterflyPlot::Channel> channels;
332 for (qint32 r = 0; r < m_pEvokedSetModel->rowCount(); ++r) {
333 if (!std::ranges::contains(m_lSelectedChannels, r) || !shown(r)) {
334 continue;
335 }
336 viz::ButterflyPlot::Channel channel;
337 const qint32 kind = m_pEvokedSetModel->getKind(r);
338 channel.max_value = mne::viz::qt::getScalingValue(m_scaleMap, kind, m_pEvokedSetModel->getUnit(r));
339 channel.bad = m_pEvokedSetModel->getIsChannelBad(r);
340 for (const mne::viz::qt::AvrTypeRowVector& average : m_pEvokedSetModel->data(r, 1).value<std::vector<mne::viz::qt::AvrTypeRowVector>>()) {
341 if (detail::valueOr(*m_qMapAverageActivation, average.first)) {
342 channel.traces.push_back({std::vector<double>(average.second.data(), average.second.data() + average.second.size()),
343 toSceneColor(detail::valueOr(*m_qMapAverageColor, average.first))});
344 }
345 }
346 channels.push_back(std::move(channel));
347 }
348 plot.set_channels(std::move(channels));
349 }
350 return plot.render(width(), height(), toSceneFont(font()));
351}
