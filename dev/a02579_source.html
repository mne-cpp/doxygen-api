72Expected<MNEBemSurface> read_surface(FiffStream& stream, const std::shared_ptr<FiffDirNode>& node, const std::string& name)
73{
74 MNEBemSurface surface;
75 const auto id = find_tag(stream, node, {FIFF_BEM_SURF_ID});
76 surface.id = id ? *id->to_int() : FIFFV_BEM_SURF_ID_UNKNOWN;
77 const auto sigma = find_tag(stream, node, {FIFF_BEM_SIGMA});
78 surface.sigma = sigma ? *sigma->to_float() : 1.0f;
79
80 const auto np = find_tag(stream, node, {FIFF_BEM_SURF_NNODE});
81 if (!np) {
82 return fail(ErrorCode::Format, "a BEM surface in {} has no vertex count", name);
83 }
84 surface.np = *np->to_int();
85 const auto ntri = find_tag(stream, node, {FIFF_BEM_SURF_NTRI});
86 if (!ntri) {
87 return fail(ErrorCode::Format, "a BEM surface in {} has no triangle count", name);
88 }
89 surface.ntri = *ntri->to_int();
90
91 // MNE-C files carry FIFF_BEM_COORD_FRAME, newer ones FIFF_MNE_COORD_FRAME
92 const auto frame = find_tag(stream, node, {FIFF_MNE_COORD_FRAME, FIFF_BEM_COORD_FRAME});
93 if (!frame) {
94 return fail(ErrorCode::Format, "a BEM surface in {} has no coordinate frame", name);
95 }
96 surface.coord_frame = *frame->to_int();
97
98 const auto nodes = find_tag(stream, node, {FIFF_BEM_SURF_NODES});
99 if (!nodes) {
100 return fail(ErrorCode::Format, "a BEM surface in {} has no vertices", name);
101 }
102 surface.rr = nodes->to_float_matrix().transpose();
103 if (surface.rr.rows() != surface.np) {
104 return fail(ErrorCode::Format, "a BEM surface in {} has {} vertices instead of {}", name, surface.rr.rows(), surface.np);
105 }
106
107 // Stored normals are optional (mne-python, MNE-C); read() computes missing ones
108 if (const auto normals = find_tag(stream, node, {FIFF_BEM_SURF_NORMALS, FIFF_MNE_SOURCE_SPACE_NORMALS})) {
109 surface.nn = normals->to_float_matrix().transpose();
110 if (surface.nn.rows() != surface.np) {
111 return fail(ErrorCode::Format, "a BEM surface in {} has {} normals for {} vertices", name, surface.nn.rows(), surface.np);
112 }
113 } else {
114 surface.nn.resize(0, 3);
115 }
116
117 if (surface.ntri > 0) {
118 const auto triangles = find_tag(stream, node, {FIFF_BEM_SURF_TRIANGLES, FIFF_MNE_SOURCE_SPACE_TRIANGLES});
119 if (!triangles) {
120 return fail(ErrorCode::Format, "a BEM surface in {} has no triangles", name);
121 }
122 surface.itris = triangles->to_int_matrix().transpose().array() - 1; // 1-based in the file
123 if (surface.itris.rows() != surface.ntri) {
124 return fail(ErrorCode::Format, "a BEM surface in {} has {} triangles instead of {}", name, surface.itris.rows(), surface.ntri);
125 }
126 } else {
127 surface.itris.resize(0, 3);
128 }
129 return surface;
130}

154Expected<MNEBem> MNEBem::read(FiffStream& stream, bool patch_stats)
155{
156 const std::string name = stream.stream_name();
157 if (!stream.is_open() && !stream.open()) {
158 return fail(ErrorCode::Io, "cannot open {}", name);
159 }
160 if (stream.dirtree()->dir_tree_find(FIFFB_BEM).empty()) {
161 return fail(ErrorCode::NotFound, "no BEM in {}", name);
162 }
163 const std::vector<std::shared_ptr<FiffDirNode>> nodes = stream.dirtree()->dir_tree_find(FIFFB_BEM_SURF);
164 if (nodes.empty()) {
165 return fail(ErrorCode::NotFound, "no BEM surfaces in {}", name);
166 }
167
168 MNEBem bem;
169 bem.m_surfaces.reserve(nodes.size());
170 for (const std::shared_ptr<FiffDirNode>& node : nodes) {
171 auto surface = read_surface(stream, node, name);
172 if (!surface) {
173 return std::unexpected(surface.error());
174 }
175 surface->addTriangleData();
176 if (patch_stats || surface->nn.rows() == 0) {
177 surface->addVertexNormals();
178 }
179 bem.m_surfaces.push_back(std::move(*surface));
180 }
181 Log::info("\t{} BEM surfaces read", bem.size());
182 return bem;
183}

200void MNEBem::write_to_stream(FiffStream& stream) const
201{
202 stream.start_block(FIFFB_BEM);
203 for (const MNEBemSurface& surface : m_surfaces) {
204 stream.start_block(FIFFB_BEM_SURF);
205 surface.write_to_stream(stream);
206 stream.end_block(FIFFB_BEM_SURF);
207 }
208 stream.end_block(FIFFB_BEM);

213void MNEBem::warp(const MatrixXf& sLm, const MatrixXf& dLm)
214{
215 std::vector<MatrixXf> vertices;
216 vertices.reserve(m_surfaces.size());
217 for (const MNEBemSurface& surface : m_surfaces) {
218 vertices.push_back(surface.rr);
219 }
220 Warp().calculate(sLm, dLm, vertices);
221 for (std::size_t i = 0; i < m_surfaces.size(); ++i) {
222 m_surfaces[i].rr = vertices[i];
223 }
224}
