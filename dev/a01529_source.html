48Expected<FilterKernel> FilterIO::read_filter(const std::filesystem::path& path)
49{
50 const Expected<std::string> text = read_text_file(path);
51 if (!text) {
52 return std::unexpected(text.error());
53 }
54 FilterKernel filter;
55 std::vector<double> coefficients;
56 std::string_view rest = *text;
57 while (!rest.empty()) {
58 const std::size_t end = rest.find('\n');
59 const std::string_view line = trimmed(rest.substr(0, end));
60 rest = end == std::string_view::npos ? std::string_view() : rest.substr(end + 1);
61 if (line.empty()) {
62 continue;
63 }
64 if (!line.starts_with('#')) {
65 const std::optional<double> value = to_double(line);
66 if (!value) {
67 return fail(ErrorCode::Format, "{} has a filter coefficient that is no number: \"{}\"",
68 utf8_from_path(path), line);
69 }
70 coefficients.push_back(*value);
71 continue;
72 }
73 const std::string_view entry = line.substr(1);
74 const std::size_t space = entry.find_first_of(" \t");
75 const std::string_view key = entry.substr(0, space);
76 const std::string_view value = space == std::string_view::npos ? std::string_view() : trimmed(entry.substr(space));
77 if (key == "name") {
78 filter.set_name(std::string(value));
79 } else if (key == "type") {
80 filter.set_filter_type(FilterKernel::filter_type_index(value));
81 } else if (key == "DesignMethod") {
82 filter.set_design_method(FilterKernel::design_method_index(value));
83 } else if (key == "order") {
84 continue; // the coefficients give the order
85 } else {
86 const auto setter = key == "sFreq" ? &FilterKernel::set_sampling_frequency
87 : key == "HPFreq" ? &FilterKernel::set_highpass_freq
88 : key == "LPFreq" ? &FilterKernel::set_lowpass_freq
89 : key == "CenterFreq" ? &FilterKernel::set_center_frequency
90 : key == "ParksWidth" ? &FilterKernel::set_parks_width
91 : nullptr;
92 if (!setter) {
93 continue;
94 }
95 const std::optional<double> number = to_double(value);
96 if (!number) {
97 return fail(ErrorCode::Format, "{} has no number for {}: \"{}\"", utf8_from_path(path), key, value);
98 }
99 (filter.*setter)(*number);
100 }
101 }
102 filter.set_filter_order(static_cast<int>(coefficients.size()));
103 filter.set_coefficients(Map<const RowVectorXd>(coefficients.data(), std::ssize(coefficients)));
104 return filter;
105}

109Expected<void> FilterIO::write_filter(const std::filesystem::path& path, const FilterKernel& filter)
110{
111 std::string text = std::format("#sFreq {}\n#name {}\n#type {}\n#order {}\n#HPFreq {}\n#LPFreq {}\n#CenterFreq {}\n"
112 "#DesignMethod {}\n#ParksWidth {}\n",
113 filter.sampling_frequency(), filter.name(), filter.filter_type().name(),
114 filter.filter_order(), filter.highpass_freq(), filter.lowpass_freq(),
115 filter.center_frequency(), filter.design_method().name(), filter.parks_width());
116 for (const double coefficient : filter.coefficients()) {
117 std::format_to(std::back_inserter(text), "{}\n", coefficient);
118 }
119 return write_text_file(path, text);
