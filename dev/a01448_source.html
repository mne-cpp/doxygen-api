47void ImagePlot::set_data(int rows, int cols, std::vector<double> values)
48{
49 if (rows <= 0 || cols <= 0 || values.size() != static_cast<std::size_t>(rows) * static_cast<std::size_t>(cols)) {
50 return;
51 }
52 m_rows = rows;
53 m_cols = cols;
54 m_values = std::move(values);
55 if (!m_rangeSet) {
56 const auto [lo, hi] = std::minmax_element(m_values.begin(), m_values.end());
57 m_min = *lo;
58 m_max = *hi;
59 }
60 update();
61}

122void ImagePlot::update()
123{
124 if (!has_data()) {
125 return;
126 }
127 m_normalized = m_values;
128 for (double& v : m_normalized) {
129 v -= m_min;
130 // A constant matrix has no range: keep it at 0 instead of dividing 0 by 0
131 if (m_max > m_min) {
132 v /= m_max - m_min;
133 }
134 v = std::clamp(v, 0.0, 1.0);
135 }
136 const auto opaque = [](Color c) {
137 c.a = 255;
138 return c;
139 };
140 auto image = std::make_shared<Image>(Image::transparent(m_cols, m_rows));
141 for (int r = 0; r < m_rows; ++r) {
142 for (int c = 0; c < m_cols; ++c) {
143 image->set_pixel(c, r, opaque(map_color(m_colormap, m_normalized[static_cast<std::size_t>(r) * static_cast<std::size_t>(m_cols) + static_cast<std::size_t>(c)])));
144 }
145 }
146 m_image = std::move(image);
147 auto colorbar = std::make_shared<Image>(Image::transparent(1, m_colorbarSteps));
148 for (int j = 0; j < m_colorbarSteps; ++j) {
149 colorbar->set_pixel(0, j, opaque(map_color(m_colormap, static_cast<double>(m_colorbarSteps - 1 - j) / (m_colorbarSteps - 1))));
150 }
151 m_colorbar = std::move(colorbar);
152}

165void ImagePlot::draw(Canvas& canvas, const TextMetrics& metrics, const Font& font) const
166{
167 if (!m_image) {
168 return;
169 }
170 const double width = canvas.width();
171 const double height = canvas.height();
172 const Color text{51, 51, 51};
173 const double line = metrics.height(font);
174
175 // Column and row centres in data units: without an extent, column and row indices
176 const double left = m_extentSet ? m_left : 0.0;
177 const double right = m_extentSet ? m_right : m_cols - 1;
178 const double top = m_extentSet ? m_top : 0.0;
179 const double bottom = m_extentSet ? m_bottom : m_rows - 1;
180 // Half a cell beyond the first and last centre, so each cell is centred on its value
181 const double halfX = m_cols > 1 ? (right - left) / (m_cols - 1) / 2.0 : 0.5;
182 const double halfY = m_rows > 1 ? (bottom - top) / (m_rows - 1) / 2.0 : 0.5;
183 const double x0 = left - halfX;
184 const double x1 = right + halfX;
185 const double y0 = top - halfY;
186 const double y1 = bottom + halfY;
187
188 // No ticks between cell centres: whole numbers for indices, at least one cell apart in data units
189 const std::vector<double> xTicks = nice_ticks(std::min(x0, x1), std::max(x0, x1), 8, std::fabs(2.0 * halfX));
190 const std::vector<double> yTicks = nice_ticks(std::min(y0, y1), std::max(y0, y1), 6, std::fabs(2.0 * halfY));
191 const std::vector<double> cTicks = nice_ticks(m_min, m_max, 6);
192 double yTickWidth = 0.0;
193 for (double v : yTicks) {
194 yTickWidth = std::max(yTickWidth, metrics.advance(font, tick_label(v)));
195 }
196 double cTickWidth = 0.0;
197 for (double v : cTicks) {
198 cTickWidth = std::max(cTickWidth, metrics.advance(font, tick_label(v)));
199 }
200 const double mLeft = yTickWidth + 14 + (m_yLabel.empty() ? 0 : line + 6);
201 const double mRight = 16 + m_colorbarWidth + 14 + cTickWidth + (m_colorbarLabel.empty() ? 0 : line + 6);
202 const double mTop = m_title.empty() ? 12 : 2 * line + 8;
203 const double mBottom = line + 14 + (m_xLabel.empty() ? 0 : line + 6);
204 const double areaW = std::floor(width - mLeft - mRight);
205 const double areaH = std::floor(height - mTop - mBottom);
206 if (areaW <= 0 || areaH <= 0) {
207 return;
208 }
209 const Rect area{mLeft, mTop, areaW, areaH};
210 // Last pixel row and column, and the centre, as QRect reports them
211 const double areaRight = area.x + areaW - 1;
212 const double areaBottom = area.y + areaH - 1;
213 const double areaCenterY = std::floor((area.y + areaBottom) / 2);
214
215 const auto toPixelX = [&](double x) {
216 return area.x + (x - x0) / (x1 - x0) * areaW;
217 };
218 const auto toPixelY = [&](double y) {
219 return area.y + (y - y0) / (y1 - y0) * areaH;
220 };
221
222 canvas.save();
223 canvas.set_font(font);
224 canvas.draw_image(area, m_image);
225 canvas.set_pen(Pen{{136, 136, 136}, 1});
226 canvas.set_brush(Brush{});
227 canvas.draw_rect(area.adjusted(0, 0, -1, -1));
228
229 canvas.set_pen(text);
230 for (double v : xTicks) {
231 const double px = toPixelX(v);
232 canvas.draw_line({px, areaBottom}, {px, areaBottom + 4});
233 canvas.draw_text(Rect{px - 50, areaBottom + 6, 100, line}, Align::HCenter | Align::Top, tick_label(v));
234 }
235 for (double v : yTicks) {
236 const double py = toPixelY(v);
237 canvas.draw_line({area.x - 4, py}, {area.x, py});
238 canvas.draw_text(Rect{0, py - line, area.x - 8, 2 * line}, Align::Right | Align::VCenter, tick_label(v));
239 }
240
241 const Rect bar{areaRight + 16, area.y, static_cast<double>(m_colorbarWidth), areaH};
242 const double barRight = bar.x + bar.width - 1;
243 const double barBottom = bar.y + bar.height - 1;
244 canvas.draw_image(bar, m_colorbar);
245 canvas.set_pen(Pen{{136, 136, 136}, 1});
246 canvas.draw_rect(bar.adjusted(0, 0, -1, -1));
247 canvas.set_pen(text);
248 if (m_max > m_min) {
249 for (double v : cTicks) {
250 const double py = barBottom - (v - m_min) / (m_max - m_min) * bar.height;
251 canvas.draw_line({barRight, py}, {barRight + 4, py});
252 canvas.draw_text(Rect{barRight + 7, py - line, cTickWidth + 4, 2 * line}, Align::Left | Align::VCenter, tick_label(v));
253 }
254 }
255 if (!m_colorbarLabel.empty()) {
256 canvas.save();
257 canvas.translate(barRight + 13 + cTickWidth, std::floor((bar.y + barBottom) / 2));
258 canvas.rotate(-90);
259 canvas.draw_text(Rect{-bar.height / 2.0, 0, bar.height, line}, Align::Center, m_colorbarLabel);
260 canvas.restore();
261 }
262
263 if (!m_title.empty()) {
264 Font titleFont = scaled_font(font, 1.25, metrics);
265 titleFont.bold = true;
266 canvas.set_font(titleFont);
267 canvas.draw_text(Rect{0, 4, width, mTop - 4}, Align::Center, m_title);
268 canvas.set_font(font);
269 }
270 if (!m_xLabel.empty()) {
271 canvas.draw_text(Rect{area.x, height - line - 6, areaW, line}, Align::Center, m_xLabel);
272 }
273 if (!m_yLabel.empty()) {
274 canvas.save();
275 canvas.translate(6, areaCenterY);
276 canvas.rotate(-90);
277 canvas.draw_text(Rect{-areaH / 2.0, 0, areaH, line}, Align::Center, m_yLabel);
278 canvas.restore();
279 }
280 canvas.restore();
281}
