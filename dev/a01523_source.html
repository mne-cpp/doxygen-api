105void CommandManager::insert(const QJsonDocument &p_jsonDocument)
106{
107 QJsonObject t_jsonObjectCommand;
108
109 //Switch to command object
110 if(p_jsonDocument.isObject() && p_jsonDocument.object().value(QString("commands")) != QJsonValue::Undefined)
111 t_jsonObjectCommand = p_jsonDocument.object().value(QString("commands")).toObject();
112 else
113 return;
114
115 QJsonObject::Iterator it;
116 for(it = t_jsonObjectCommand.begin(); it != t_jsonObjectCommand.end(); ++it)
117 {
118 if(!m_qMapCommands.contains(it.key()))
119 m_qMapCommands.insert(it.key(), Command(it.key(), it.value().toObject(), true, this));
120 else
121 qWarning("Warning: CommandMap contains command %s already. Insertion skipped.\n", it.key().toUtf8().constData());
122 }
123
124 emit commandMapChanged();
125}

147void CommandManager::update(UTILSLIB::Subject* p_pSubject)
148{
149 // If Manager is not active do not parse commands
150 if(!m_bIsActive)
151 return;
152
153 CommandParser* t_pCommandParser = static_cast<CommandParser*>(p_pSubject);
154
155 RawCommand t_rawCommand(t_pCommandParser->getRawCommand());
156 QString t_sCommandName = t_rawCommand.command();
157
158 if(!this->hasCommand(t_sCommandName))
159 return;
160
161 // check if number of parameters is right
162 if(t_rawCommand.count() >= m_qMapCommands[t_sCommandName].count())
163 {
164 m_qMapCommands[t_sCommandName].isJson() = t_rawCommand.isJson();
165
166 //Parse Parameters
167 for(quint32 i = 0; i < m_qMapCommands[t_sCommandName].count(); ++i)
168 {
169 QMetaType t_metaType = m_qMapCommands[t_sCommandName][i].metaType();
170
171 QVariant t_qVariantParam(t_rawCommand.pValues()[i]);
172
173 if(t_qVariantParam.canConvert(t_metaType) && t_qVariantParam.convert(t_metaType))
174 m_qMapCommands[t_sCommandName][i] = t_qVariantParam;
175 else
176 return;
177 }
178
179 m_qMapCommands[t_sCommandName].execute();
180 }
181}
