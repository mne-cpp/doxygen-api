72// Keeps the part of the polygon where dot(normal, p) >= offset (Sutherland-Hodgman, one edge)
73Polygon clipHalfPlane(const Polygon& polygon, Point normal, double offset)
74{
75 Polygon out;
76 for (std::size_t i = 0; i < polygon.size(); ++i) {
77 const Point a = polygon[i];
78 const Point b = polygon[(i + 1) % polygon.size()];
79 const double da = dot(normal, a) - offset;
80 const double db = dot(normal, b) - offset;
81 if (da >= 0.0) {
82 out.push_back(a);
83 }
84 if ((da >= 0.0) != (db >= 0.0)) {
85 out.push_back(a + (b - a) * (da / (da - db)));
86 }
87 }
88 return out;
89}

91// Cuts the polygon with a convex clip polygon of either orientation
92Polygon clipConvex(Polygon polygon, const Polygon& clip)
93{
94 const double orientation = signedArea(clip) >= 0.0 ? 1.0 : -1.0;
95 for (std::size_t i = 0; i < clip.size() && polygon.size() >= 3; ++i) {
96 const Point a = clip[i];
97 const Point b = clip[(i + 1) % clip.size()];
98 // The interior lies on the same side of every edge
99 const Point normal{-(b.y - a.y) * orientation, (b.x - a.x) * orientation};
100 polygon = clipHalfPlane(polygon, normal, dot(normal, a));
101 }
102 return polygon;
103}

105bool insideConvex(Point p, const Polygon& clip)
106{
107 const double orientation = signedArea(clip) >= 0.0 ? 1.0 : -1.0;
108 for (std::size_t i = 0; i < clip.size(); ++i) {
109 const Point a = clip[i];
110 const Point b = clip[(i + 1) % clip.size()];
111 if (cross(b - a, p - a) * orientation < 0.0) {
112 return false;
113 }
114 }
115 return true;
116}

118Polygon ellipsePolygon(Point center, double rx, double ry, double startDegrees, double spanDegrees, int segments)
119{
120 Polygon points;
121 for (int i = 0; i <= segments; ++i) {
122 const double angle = (startDegrees + spanDegrees * i / segments) * std::numbers::pi / 180.0;
123 points.push_back({center.x + rx * std::cos(angle), center.y + ry * std::sin(angle)});
124 }
125 return points;
126}

133Polygon roundedRectPolygon(const Rect& r, double rx, double ry)
134{
135 rx = std::min(rx, r.width / 2.0);
136 ry = std::min(ry, r.height / 2.0);
137 if (rx <= 0.0 || ry <= 0.0) {
138 return {{r.left(), r.top()}, {r.right(), r.top()}, {r.right(), r.bottom()}, {r.left(), r.bottom()}};
139 }
140 const int n = std::max(2, segmentsFor(std::max(rx, ry)) / 4);
141 Polygon points;
142 const auto corner = [&](Point c, double start) {
143 const Polygon arc = ellipsePolygon(c, rx, ry, start, 90.0, n);
144 points.insert(points.end(), arc.begin(), arc.end());
145 };
146 corner({r.right() - rx, r.top() + ry}, 270.0);
147 corner({r.right() - rx, r.bottom() - ry}, 0.0);
148 corner({r.left() + rx, r.bottom() - ry}, 90.0);
149 corner({r.left() + rx, r.top() + ry}, 180.0);
150 return points;
151}

169// Splits a polyline into the pieces its dash pattern draws
170std::vector<Polygon> dashPieces(const Polygon& line, const std::vector<double>& pattern, double unit)
171{
172 if (pattern.empty()) {
173 return {line};
174 }
175 std::vector<Polygon> pieces;
176 std::size_t index = 0;
177 double left = pattern[0] * unit;
178 bool on = true;
179 Polygon current{line.front()};
180 for (std::size_t i = 1; i < line.size(); ++i) {
181 Point a = line[i - 1];
182 const Point b = line[i];
183 double length = std::hypot(b.x - a.x, b.y - a.y);
184 while (length > 0.0) {
185 const double step = std::min(left, length);
186 const Point p = a + (b - a) * (step / length);
187 if (on) {
188 current.push_back(p);
189 }
190 length -= step;
191 left -= step;
192 a = p;
193 if (left <= 1e-9) {
194 if (on && current.size() >= 2) {
195 pieces.push_back(current);
196 }
197 on = !on;
198 index = (index + 1) % pattern.size();
199 left = pattern[index] * unit;
200 current = {a};
201 }
202 }
203 }
204 if (on && current.size() >= 2) {
205 pieces.push_back(current);
206 }
207 return pieces;
208}

210// The outline of a stroke as polygons of one orientation
211std::vector<Polygon> strokePolygons(const Polygon& points, bool closed, const Pen& pen, double width)
212{
213 std::vector<Polygon> polygons;
214 const double h = width / 2.0;
215 Polygon line = points;
216 if (closed && points.size() >= 2) {
217 line.push_back(points.front());
218 }
219 for (const Polygon& piece : dashPieces(line, dashPattern(pen.style), width)) {
220 const bool wholeClosed = closed && pen.style == LineStyle::Solid;
221 for (std::size_t i = 1; i < piece.size(); ++i) {
222 Point a = piece[i - 1];
223 Point b = piece[i];
224 const double length = std::hypot(b.x - a.x, b.y - a.y);
225 if (length <= 0.0) {
226 continue;
227 }
228 const Point d = (b - a) * (1.0 / length);
229 if (pen.cap == LineCap::Square && !wholeClosed) {
230 if (i == 1) {
231 a = a - d * h;
232 }
233 if (i + 1 == piece.size()) {
234 b = b + d * h;
235 }
236 }
237 const Point n{-d.y * h, d.x * h};
238 polygons.push_back({a + n, b + n, b - n, a - n});
239 }
240 // Joins: a disc where segments meet (round), else the triangles that close the outer gap
241 const std::size_t first = wholeClosed ? 0 : 1;
242 for (std::size_t i = first; i + 1 < piece.size(); ++i) {
243 const Point v = piece[i];
244 const Point prev = i == 0 ? piece[piece.size() - 2] : piece[i - 1];
245 const Point next = piece[i + 1];
246 if (pen.join == LineJoin::Round) {
247 polygons.push_back(ellipsePolygon(v, h, h, 0.0, 360.0, segmentsFor(h)));
248 continue;
249 }
250 const double l1 = std::hypot(v.x - prev.x, v.y - prev.y);
251 const double l2 = std::hypot(next.x - v.x, next.y - v.y);
252 if (l1 <= 0.0 || l2 <= 0.0) {
253 continue;
254 }
255 const Point n1{-(v.y - prev.y) / l1 * h, (v.x - prev.x) / l1 * h};
256 const Point n2{-(next.y - v.y) / l2 * h, (next.x - v.x) / l2 * h};
257 polygons.push_back({v, v + n1, v + n2});
258 polygons.push_back({v, v - n2, v - n1});
259 }
260 if (pen.cap == LineCap::Round && !wholeClosed) {
261 polygons.push_back(ellipsePolygon(piece.front(), h, h, 0.0, 360.0, segmentsFor(h)));
262 polygons.push_back(ellipsePolygon(piece.back(), h, h, 0.0, 360.0, segmentsFor(h)));
263 }
264 }
265 return polygons;
266}

285 {

374 template<class ColorAt>
375 void paintCoverage(const std::vector<Polygon>& polygons, ColorAt colorAt)
376 {
377 if (polygons.empty()) {
378 return;
379 }
380 double minX = m_image.width();
381 double minY = m_image.height();
382 double maxX = 0.0;
383 double maxY = 0.0;
384 for (const Polygon& polygon : polygons) {
385 for (const Point& p : polygon) {
386 minX = std::min(minX, p.x);
387 minY = std::min(minY, p.y);
388 maxX = std::max(maxX, p.x);
389 maxY = std::max(maxY, p.y);
390 }
391 }
392 const int x0 = std::max(0, static_cast<int>(std::floor(minX)));
393 const int y0 = std::max(0, static_cast<int>(std::floor(minY)));
394 const int x1 = std::min(m_image.width(), static_cast<int>(std::ceil(maxX)) + 1);
395 const int y1 = std::min(m_image.height(), static_cast<int>(std::ceil(maxY)) + 1);
396 if (x0 >= x1 || y0 >= y1) {
397 return;
398 }
399 RasterImage mask(x1 - x0, y1 - y0);
400 std::vector<std::vector<PointF>> contours;
401 for (const Polygon& polygon : polygons) {
402 std::vector<PointF> contour;
403 for (const Point& p : polygon) {
404 contour.push_back({p.x - x0, p.y - y0});
405 }
406 contours.push_back(std::move(contour));
407 }
408 mask.fill_polygons(contours, {255, 255, 255});
409 for (int y = y0; y < y1; ++y) {
410 for (int x = x0; x < x1; ++x) {
411 const int coverage = mask.pixel(x - x0, y - y0).r;
412 if (coverage > 0) {
413 const Color c = colorAt(x + 0.5, y + 0.5);
414 m_image.blend_pixel(x, y, rgb(c), alpha(coverage * c.a / 255));
415 }
416 }
417 }

450 // Cubic segments as polylines, with steps of about two device pixels
451 Polygon flattenCubic(const Polygon& points) const
452 {
453 Polygon line{points.front()};
454 for (std::size_t i = 1; i + 2 < points.size(); i += 3) {
455 const Point p0 = points[i - 1];
456 const Point c1 = points[i];
457 const Point c2 = points[i + 1];
458 const Point p1 = points[i + 2];
459 double length = 0.0;
460 for (const auto& [a, b] : {std::pair{p0, c1}, std::pair{c1, c2}, std::pair{c2, p1}}) {
461 const Point d = m_device.map(b) - m_device.map(a);
462 length += std::hypot(d.x, d.y);
463 }
464 const int steps = std::clamp(static_cast<int>(length / 2.0), 1, 256);
465 for (int k = 1; k <= steps; ++k) {
466 const double t = static_cast<double>(k) / steps;
467 const double u = 1.0 - t;
468 line.push_back(p0 * (u * u * u) + c1 * (3 * u * u * t) + c2 * (3 * u * t * t) + p1 * (t * t * t));
469 }
470 }
471 return line;
472 }

491 template<class Sample>
492 void paintMapped(double w, double h, const Transform& place, Sample sample)
493 {
494 const Transform toDevice = place * m_device;
495 const Transform fromDevice = toDevice.inverted();
496 const Point corners[4] = {toDevice.map({0, 0}), toDevice.map({w, 0}), toDevice.map({w, h}), toDevice.map({0, h})};
497 double minX = corners[0].x;
498 double maxX = corners[0].x;
499 double minY = corners[0].y;
500 double maxY = corners[0].y;
501 for (const Point& c : corners) {
502 minX = std::min(minX, c.x);
503 maxX = std::max(maxX, c.x);
504 minY = std::min(minY, c.y);
505 maxY = std::max(maxY, c.y);
506 }
507 const int x0 = std::max(0, static_cast<int>(std::floor(minX)));
508 const int y0 = std::max(0, static_cast<int>(std::floor(minY)));
509 const int x1 = std::min(m_image.width(), static_cast<int>(std::ceil(maxX)));
510 const int y1 = std::min(m_image.height(), static_cast<int>(std::ceil(maxY)));
511 for (int y = y0; y < y1; ++y) {
512 for (int x = x0; x < x1; ++x) {
513 const Point center{x + 0.5, y + 0.5};
514 if (std::any_of(m_clips.begin(), m_clips.end(), [&](const Polygon& clip) { return !insideConvex(center, clip); })) {
515 continue;
516 }
517 const Point source = fromDevice.map(center);
518 if (source.x < 0.0 || source.y < 0.0 || source.x >= w || source.y >= h) {
519 continue;
520 }
521 const Color c = sample(source.x, source.y);
522 if (c.a > 0) {
523 m_image.blend_pixel(x, y, rgb(c), alpha(c.a));
524 }
525 }
526 }

559 void draw(const ImageShape& shape)
560 {
561 const Image& image = *shape.image;
562 const Transform place = Transform::scaling(shape.target.width / image.width, shape.target.height / image.height) *
563 Transform::translation(shape.target.x, shape.target.y);
564 paintMapped(image.width, image.height, place, [&](double sx, double sy) {
565 if (!shape.smooth) {
566 return image.pixel(static_cast<int>(sx), static_cast<int>(sy));
567 }
568 // Bilinear between the four nearest pixel centres
569 const double fx = std::clamp(sx - 0.5, 0.0, image.width - 1.0);
570 const double fy = std::clamp(sy - 0.5, 0.0, image.height - 1.0);
571 const int ix = static_cast<int>(fx);
572 const int iy = static_cast<int>(fy);
573 const int jx = std::min(ix + 1, image.width - 1);
574 const int jy = std::min(iy + 1, image.height - 1);
575 const double tx = fx - ix;
576 const double ty = fy - iy;
577 const Color a = image.pixel(ix, iy);
578 const Color b = image.pixel(jx, iy);
579 const Color c = image.pixel(ix, jy);
580 const Color d = image.pixel(jx, jy);
581 const auto mix = [&](std::uint8_t p, std::uint8_t q, std::uint8_t r, std::uint8_t s) {
582 return static_cast<std::uint8_t>(std::lround((p * (1 - tx) + q * tx) * (1 - ty) + (r * (1 - tx) + s * tx) * ty));
583 };
584 return Color{mix(a.r, b.r, c.r, d.r), mix(a.g, b.g, c.g, d.g), mix(a.b, b.b, c.b, d.b), mix(a.a, b.a, c.a, d.a)};
585 });
