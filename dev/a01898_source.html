296template<typename T>
297void Spline::updatePlot(const Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& matClassLimitData,
298 const Eigen::VectorXi& matClassFrequencyData)
299{
300 int iClassAmount = std::min(static_cast<int>(matClassFrequencyData.rows()),
301 static_cast<int>(matClassLimitData.rows()) - 1);
302 if (iClassAmount < 1)
303 return;
304 std::vector<viz::Point> classMarks;
305 for (int ir = 0; ir < iClassAmount; ++ir) {
306 const double classMark = (static_cast<double>(matClassLimitData(ir)) + static_cast<double>(matClassLimitData(ir + 1))) / 2.0;
307 classMarks.push_back({classMark, static_cast<double>(matClassFrequencyData(ir))});
308 }
309 m_plot.set_classes(std::move(classMarks), static_cast<double>(matClassLimitData(0)), static_cast<double>(matClassLimitData(iClassAmount)));
310
311 QWidget::update();
312}
