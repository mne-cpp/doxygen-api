201void BidsPanel::update_actions()
202{
203 SettingsForm& f = form();
204 const int node = selected();
205 const bool has = m_tree.contains(node);
206 std::vector<SettingChoice> choices;
207 std::string label = "Move to";
208 if (has && m_tree.node(node).type == BidsTree::Session) {
209 label = "Move session to";
210 for (const int subject : m_tree.subjects()) {
211 if (subject != m_tree.node(node).subject) {
212 choices.push_back({std::to_string(subject), m_tree.node(subject).name});
213 }
214 }
215 } else if (has && is_movable_data(m_tree.node(node).type)) {
216 label = "Move data to";
217 for (const int subject : m_tree.subjects()) {
218 for (const int session : m_tree.node(subject).children) {
219 if (session != m_tree.node(node).session && m_tree.node(session).type == BidsTree::Session) {
220 choices.push_back({std::to_string(session), m_tree.node(subject).name + "/" + m_tree.node(session).name});
221 }
222 }
223 }
224 }
225 m_updating = true;
226 f.set_value(kHasSelection, has);
227 f.set_value(kCanMove, !choices.empty());
228 f.set_label(kTarget, label);
229 if (const SettingField* target = f.field(kTarget); target && target->choices != choices) {
230 f.set_choices(kTarget, std::move(choices));
231 }
232 m_updating = false;
233}

237void BidsPanel::changed(const std::string& key)
238{
239 const std::string name = form().get<std::string>(kName);
240 const int node = selected();
241 const bool has = m_tree.contains(node);
242 if (key == kAddSubject && !name.empty()) {
243 if (m_callbacks.add_subject) {
244 m_callbacks.add_subject(name);
245 } else {
246 const int subject = m_tree.add_subject(name);
247 rebuild();
248 picked(subject);
249 }
250 } else if (key == kAddSession && has && !name.empty()) {
251 const int subject = m_tree.node(node).subject;
252 if (m_callbacks.add_session) {
253 m_callbacks.add_session(subject, name);
254 } else {
255 const int session = m_tree.add_session(subject, name);
256 rebuild();
257 picked(session);
258 }
259 } else if (key == kRename && has && !name.empty()) {
260 if (m_callbacks.rename) {
261 m_callbacks.rename(node, name);
262 } else {
263 m_tree.rename(node, name);
264 rebuild();
265 }
266 } else if (key == kMove && has && form().get<bool>(kCanMove)) {
267 const std::string value = form().get<std::string>(kTarget);
268 int target = -1;
269 std::from_chars(value.data(), value.data() + value.size(), target);
270 if (!m_tree.contains(target)) {
271 return;
272 }
273 if (m_tree.node(node).type == BidsTree::Session) {
274 if (m_callbacks.move_session) {
275 m_callbacks.move_session(node, target);
276 return;
277 }
278 m_tree.move_session(node, target);
279 } else {
280 if (m_callbacks.move_data) {
281 m_callbacks.move_data(node, target);
282 return;
283 }
284 m_tree.move_data(node, target);
285 }
286 rebuild();
287 picked(node);
288 } else if (key == kRemove && has) {
289 if (m_callbacks.remove) {
290 m_callbacks.remove(node);
291 } else {
292 m_tree.remove(node);
293 rebuild();
294 picked(-1);
295 }
296 }
297}
