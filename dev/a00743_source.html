73constexpr std::int16_t DT_UINT8 = 2;

75constexpr std::int16_t DT_INT32 = 8;

77constexpr std::int16_t DT_INT8 = 256;

80constexpr std::size_t NIFTI_HDR_SIZE = 348;

82// Value of type T at @p p, stored with the file's byte order
83template<mne::Swappable T>
84T readValue(const std::byte* p, bool bigEndian)
85{
86 T v;
87 std::memcpy(&v, p, sizeof(T));
88 return bigEndian != (std::endian::native == std::endian::big) ? mne::byte_swap(v) : v;
89}
90
91// Decodes @p count values of type T from @p p and converts each one with @p convert into @p out
92template<mne::Swappable T, typename Out, typename Convert>
93void readValues(const std::byte*& p, bool bigEndian, std::vector<Out>& out, std::size_t count, Convert convert)
94{
95 out.resize(count);
96 for (std::size_t i = 0; i < count; ++i, p += sizeof(T))
97 out[i] = convert(readValue<T>(p, bigEndian));
98}

117bool MriNiftiIO::read(const std::string& niiFile, MriVolData& volData, bool verbose)
118{
119 volData.fileName = niiFile;
120
121 const bool compressed = mne::to_lower(niiFile).ends_with(".gz");
122 const auto content = mne::read_file_bytes(mne::path_from_utf8(niiFile), compressed ? mne::Compression::Gzip : mne::Compression::None);
123 if (!content) {
124 Log::critical("MriNiftiIO::read - {}", content.error().what());
125 return false;
126 }
127 const std::vector<std::byte>& bytes = *content;
128
129 if (bytes.size() < NIFTI_HDR_SIZE) {
130 Log::critical("MriNiftiIO::read - file too small ({} bytes) : {}", bytes.size(), niiFile);
131 return false;
132 }
133
134 const std::byte* hdr = bytes.data();
135
136 // Endianness detection via sizeof_hdr (must be 348).
137 bool bigEndian = false;
138 std::int32_t sizeofHdr = readValue<std::int32_t>(hdr, bigEndian);
139 if (sizeofHdr != std::int32_t(NIFTI_HDR_SIZE)) {
140 bigEndian = true;
141 sizeofHdr = readValue<std::int32_t>(hdr, bigEndian);
142 if (sizeofHdr != std::int32_t(NIFTI_HDR_SIZE)) {
143 Log::critical("MriNiftiIO::read - not a NIfTI-1 file (sizeof_hdr = {}): {}", sizeofHdr, niiFile);
144 return false;
145 }
146 }
147
148 auto i16 = [&](std::size_t off) {
149 return readValue<std::int16_t>(hdr + off, bigEndian);
150 };
151 auto f32 = [&](std::size_t off) {
152 return readValue<float>(hdr + off, bigEndian);
153 };
154
155 // dim[0..7] starts at offset 40; dim[0] = ndim, dim[1..3] = nx,ny,nz, dim[4] = nt.
156 const std::int16_t ndim = i16(40);
157 const std::int16_t nx = i16(42);
158 const std::int16_t ny = i16(44);
159 const std::int16_t nz = i16(46);
160 const std::int16_t nt = i16(48);
161 if (ndim < 3 || nx <= 0 || ny <= 0 || nz <= 0) {
162 Log::critical("MriNiftiIO::read - degenerate dims {} {} {} {}", ndim, nx, ny, nz);
163 return false;
164 }
165
166 const std::int16_t datatype = i16(70);
167
168 // pixdim[0..7] starts at offset 76. pixdim[1..3] = spacing in mm; pixdim[0] is the qfac sign.
169 float qfac = f32(76);
170 if (qfac == 0.0f) {
171 qfac = 1.0f;
172 }
173 const float dx = std::fabs(f32(80));
174 const float dy = std::fabs(f32(84));
175 const float dz = std::fabs(f32(88));
176
177 const float voxOffset = f32(108);
178 const float sclSlope = f32(112);
179 const float sclInter = f32(116);
180
181 const std::int16_t qformCode = i16(252);
182 const std::int16_t sformCode = i16(254);
183
184 // Map NIfTI datatype to MRI_* + bytes-per-voxel.
185 int mriType = MRI_UCHAR;
186 int bpv = 1;
187 switch (datatype) {
188 case DT_UINT8:
189 mriType = MRI_UCHAR;
190 bpv = 1;
191 break;
192 case DT_INT8:
193 mriType = MRI_UCHAR;
194 bpv = 1;
195 break; // promoted to unsigned, negatives clamped
196 case DT_INT16:
197 mriType = MRI_SHORT;
198 bpv = 2;
199 break;
200 case DT_UINT16:
201 mriType = MRI_SHORT;
202 bpv = 2;
203 break;
204 case DT_INT32:
205 mriType = MRI_INT;
206 bpv = 4;
207 break;
208 case DT_FLOAT32:
209 mriType = MRI_FLOAT;
210 bpv = 4;
211 break;
212 default:
213 Log::critical("MriNiftiIO::read - unsupported datatype {}", datatype);
214 return false;
215 }
216
217 // Build the 4×4 voxel→RAS transform (mm) using sform → qform → pixdim fallback.
218 Matrix4f vox2ras = Matrix4f::Identity();
219 if (sformCode > 0) {
220 for (int c = 0; c < 4; ++c) {
221 const std::size_t col = static_cast<std::size_t>(c) * 4;
222 vox2ras(0, c) = f32(280 + col);
223 vox2ras(1, c) = f32(296 + col);
224 vox2ras(2, c) = f32(312 + col);
225 }
226 } else if (qformCode > 0) {
227 const float b = f32(256);
228 const float c = f32(260);
229 const float d = f32(264);
230 const float aSq = 1.0f - (b * b + c * c + d * d);
231 const float a = aSq > 0.0f ? std::sqrt(aSq) : 0.0f;
232 Matrix3f R;
233 R << a * a + b * b - c * c - d * d, 2.0f * (b * c - a * d), 2.0f * (b * d + a * c),
234 2.0f * (b * c + a * d), a * a + c * c - b * b - d * d, 2.0f * (c * d - a * b),
235 2.0f * (b * d - a * c), 2.0f * (c * d + a * b), a * a + d * d - b * b - c * c;
236 Matrix3f S = Matrix3f::Zero();
237 S(0, 0) = dx;
238 S(1, 1) = dy;
239 S(2, 2) = dz * qfac;
240 Matrix3f M = R * S;
241 vox2ras.block<3, 3>(0, 0) = M;
242 vox2ras(0, 3) = f32(268);
243 vox2ras(1, 3) = f32(272);
244 vox2ras(2, 3) = f32(276);
245 } else {
246 // Method 1: pixdim diagonal, origin at volume centre.
247 vox2ras(0, 0) = dx;
248 vox2ras(1, 1) = dy;
249 vox2ras(2, 2) = dz;
250 vox2ras(0, 3) = -0.5f * dx * nx;
251 vox2ras(1, 3) = -0.5f * dy * ny;
252 vox2ras(2, 3) = -0.5f * dz * nz;
253 }
254
255 // Decompose vox2ras → (Mdc, spacing, c_ras) so MriVolData::computeVox2Ras round-trips.
256 Vector3f col0 = vox2ras.block<3, 1>(0, 0);
257 Vector3f col1 = vox2ras.block<3, 1>(0, 1);
258 Vector3f col2 = vox2ras.block<3, 1>(0, 2);
259 float sx = col0.norm();
260 float sy = col1.norm();
261 float sz = col2.norm();
262 if (sx <= 0.0f)
263 sx = dx > 0.0f ? dx : 1.0f;
264 if (sy <= 0.0f)
265 sy = dy > 0.0f ? dy : 1.0f;
266 if (sz <= 0.0f)
267 sz = dz > 0.0f ? dz : 1.0f;
268
269 Vector3f xRas = col0 / sx;
270 Vector3f yRas = col1 / sy;
271 Vector3f zRas = col2 / sz;
272
273 Vector4f centreVox(nx * 0.5f, ny * 0.5f, nz * 0.5f, 1.0f);
274 Vector4f cRas4 = vox2ras * centreVox;
275
276 volData.version = MRI_MGH_VERSION; // mark as a valid volume for isValid()
277 volData.width = nx;
278 volData.height = ny;
279 volData.depth = nz;
280 volData.nframes = nt > 0 ? nt : 1;
281 volData.type = mriType;
282 volData.dof = 0;
283 volData.rasGood = true;
284 volData.xsize = sx;
285 volData.ysize = sy;
286 volData.zsize = sz;
287 volData.x_ras = xRas;
288 volData.y_ras = yRas;
289 volData.z_ras = zRas;
290 volData.c_ras = cRas4.head<3>();
291 volData.voxelSurfRasT = FiffCoordTrans(
292 FIFFV_COORD_MRI_SLICE, FIFFV_COORD_MRI, volData.computeVox2Ras(), true);
293
294 // Read voxel data starting at vox_offset (typically 352 for single-file).
295 const std::size_t dataOff = static_cast<std::size_t>(voxOffset > 0.0f ? voxOffset : float(NIFTI_HDR_SIZE + 4));
296 const std::size_t nPixels = static_cast<std::size_t>(nx) * static_cast<std::size_t>(ny);
297 const std::size_t nslice = static_cast<std::size_t>(nz);
298 const std::size_t frameSize = nPixels * nslice * static_cast<std::size_t>(bpv);
299 if (bytes.size() < dataOff + frameSize) {
300 Log::critical("MriNiftiIO::read - data section truncated (need {} bytes, got {})", dataOff + frameSize, bytes.size());
301 return false;
302 }
303
304 const std::byte* voxels = bytes.data() + dataOff;
305 volData.slices.resize(nslice);
306
307 const Matrix4f vox2rasFinal = volData.computeVox2Ras();
308 const bool useScale = (sclSlope != 0.0f) && !(sclSlope == 1.0f && sclInter == 0.0f);
309
310 for (std::size_t k = 0; k < nslice; ++k) {
311 MriSlice& slice = volData.slices[k];
312 slice.width = nx;
313 slice.height = ny;
314 slice.dimx = sx / 1000.0f;
315 slice.dimy = sy / 1000.0f;
316 slice.scale = 1.0f;
317
318 switch (mriType) {
319 case MRI_UCHAR:
320 slice.pixelFormat = FIFFV_MRI_PIXEL_BYTE;
321 if (datatype == DT_INT8) {
322 readValues<std::int8_t>(voxels, bigEndian, slice.pixels, nPixels, [](std::int8_t v) { return static_cast<unsigned char>(v < 0 ? 0 : v); });
323 } else {
324 readValues<std::uint8_t>(voxels, bigEndian, slice.pixels, nPixels, [](std::uint8_t v) { return v; });
325 }
326 break;
327 case MRI_SHORT:
328 slice.pixelFormat = FIFFV_MRI_PIXEL_WORD;
329 if (datatype == DT_UINT16) {
330 readValues<std::uint16_t>(voxels, bigEndian, slice.pixelsWord, nPixels, [](std::uint16_t v) { return v; });
331 } else {
332 readValues<std::int16_t>(voxels, bigEndian, slice.pixelsWord, nPixels, [](std::int16_t v) { return static_cast<unsigned short>(v < 0 ? 0 : v); });
333 }
334 break;
335 case MRI_INT:
336 slice.pixelFormat = FIFFV_MRI_PIXEL_FLOAT;
337 readValues<std::int32_t>(voxels, bigEndian, slice.pixelsFloat, nPixels, [](std::int32_t v) { return static_cast<float>(v); });
338 break;
339 case MRI_FLOAT:
340 slice.pixelFormat = FIFFV_MRI_PIXEL_FLOAT;
341 readValues<float>(voxels, bigEndian, slice.pixelsFloat, nPixels, [](float v) { return v; });
342 break;
343 }
344
345 // Apply scl_slope / scl_inter for floating-point output (the NIfTI spec
346 // states the transform produces the "physical" value; we only honour it
347 // for floats — integer outputs stay as-is to preserve label maps).
348 if (useScale && slice.pixelFormat == FIFFV_MRI_PIXEL_FLOAT) {
349 for (float& v : slice.pixelsFloat) {
350 v = v * sclSlope + sclInter;
351 }
352 }
353
354 const float kf = static_cast<float>(k);
355 Vector3f sliceOrigin;
356 sliceOrigin(0) = vox2rasFinal(0, 2) * kf + vox2rasFinal(0, 3);
357 sliceOrigin(1) = vox2rasFinal(1, 2) * kf + vox2rasFinal(1, 3);
358 sliceOrigin(2) = vox2rasFinal(2, 2) * kf + vox2rasFinal(2, 3);
359
360 Matrix3f sliceRot;
361 sliceRot.col(0) = vox2rasFinal.block<3, 1>(0, 0);
362 sliceRot.col(1) = vox2rasFinal.block<3, 1>(0, 1);
363 sliceRot.col(2) = vox2rasFinal.block<3, 1>(0, 2);
364
365 slice.trans = FiffCoordTrans(FIFFV_COORD_MRI_SLICE, FIFFV_COORD_MRI, sliceRot, sliceOrigin);
366 }
367
368 if (verbose) {
369 Log::info("NIfTI file: {}x{}x{}, datatype={}, voxel {:.3f}x{:.3f}x{:.3f} mm, c_ras=({:.2f}, {:.2f}, {:.2f})",
370 nx, ny, nz, datatype, sx, sy, sz, volData.c_ras[0], volData.c_ras[1], volData.c_ras[2]);
371 }
372
373 return true;
374}
