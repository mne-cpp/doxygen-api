123/*
124 * Read the strange 3-byte integer
125 */

127 unsigned int s = 0;
128

130 qCritical("read_int3 could not read data");

133 s = (unsigned int)FIFFLIB::swap_int(s);

219MNEVolGeom* read_vol_geom(QFile &fp)
220/*
221 * This the volume geometry reading code from FreeSurfer
222 */
223{
224 char param[64];
225 char eq[2];
226 char buf[256];
227 int vgRead = 0;
228 int counter = 0;
229 qint64 pos = 0;
230
231 MNEVolGeom* vg = new MNEVolGeom();
232
233 while (!fp.atEnd() && counter < 8)
234 {
235 QByteArray lineData = fp.readLine(256);
236 if (lineData.isEmpty())
237 break;
238 const char *line = lineData.constData();
239 if (strlen(line) == 0)
240 break ;
241 sscanf(line, "%s %s %*s", param, eq);
242 if (!strcmp(param, "valid")) {
243 sscanf(line, "%s %s %d", param, eq, &vg->valid);
244 vgRead = 1;
245 counter++;
246 }
247 else if (!strcmp(param, "filename")) {
248 if (sscanf(line, "%s %s %s", param, eq, buf) >= 3)
249 vg->filename = strdup(buf);
250 counter++;
251 }
252 else if (!strcmp(param, "volume")) {
253 sscanf(line, "%s %s %d %d %d",
254 param, eq, &vg->width, &vg->height, &vg->depth);
255 counter++;
256 }
257 else if (!strcmp(param, "voxelsize")) {
258 sscanf(line, "%s %s %f %f %f",
259 param, eq, &vg->xsize, &vg->ysize, &vg->zsize);
260 /*
261 * We like these to be in meters
262 */
263 vg->xsize = vg->xsize/1000.0;
264 vg->ysize = vg->ysize/1000.0;
265 vg->zsize = vg->zsize/1000.0;
266 counter++;
267 }
268 else if (!strcmp(param, "xras")) {
269 sscanf(line, "%s %s %f %f %f",
270 param, eq, vg->x_ras, vg->x_ras+1, vg->x_ras+2);
271 counter++;
272 }
273 else if (!strcmp(param, "yras")) {
274 sscanf(line, "%s %s %f %f %f",
275 param, eq, vg->y_ras, vg->y_ras+1, vg->y_ras+2);
276 counter++;
277 }
278 else if (!strcmp(param, "zras")) {
279 sscanf(line, "%s %s %f %f %f",
280 param, eq, vg->z_ras, vg->z_ras+1, vg->z_ras+2);
281 counter++;
282 }
283 else if (!strcmp(param, "cras")) {
284 sscanf(line, "%s %s %f %f %f",
285 param, eq, vg->c_ras, vg->c_ras+1, vg->c_ras+2);
286 vg->c_ras[0] = vg->c_ras[0]/1000.0;
287 vg->c_ras[1] = vg->c_ras[1]/1000.0;
288 vg->c_ras[2] = vg->c_ras[2]/1000.0;
289 counter++;
290 }
291 /* remember the current position */
292 pos = fp.pos();
293 };
294 if (!fp.atEnd()) { /* we read one more line */
295 if (pos > 0 ) /* if success in getting pos, then */
296 fp.seek(pos); /* restore the position */
297 /* note that this won't allow compression using pipe */
298 }
299 if (!vgRead) {
300 delete vg;
301 vg = new MNEVolGeom();
302 }
303 return vg;
304}

310int read_tag_data(QFile &fp, int tag, long long nbytes, unsigned char *&val, long long &nbytesp)
311/*
312 * Read the data of one tag
313 */
314{
315 unsigned char *dum = nullptr;
316 size_t snbytes = nbytes;
317
318 val = nullptr;
319 if (nbytes > 0) {
320 dum = new unsigned char[nbytes+1]();
321 if (fp.read(reinterpret_cast<char*>(dum), nbytes) != static_cast<qint64>(snbytes)) {
322 printf("Failed to read %d bytes of tag data",static_cast<int>(nbytes));
323 delete[] dum;
324 return FAIL;
325 }
326 dum[nbytes] = '\0'; /* Ensure null termination */
327 val = dum;
328 nbytesp = nbytes;
329 }
330 else { /* Need to handle special cases */
331 if (tag == TAG_OLD_SURF_GEOM) {
332 MNEVolGeom* g = read_vol_geom(fp);
333 if (!g)
334 return FAIL;
335 val = reinterpret_cast<unsigned char *>(g);
336 nbytesp = sizeof(MNEVolGeom);
337 }
338 else if (tag == TAG_OLD_USEREALRAS || tag == TAG_USEREALRAS) {
339 int *vi = new int[1]();
340 if (read_int(fp,*vi) == FAIL)
341 vi = 0;
342 val = reinterpret_cast<unsigned char *>(vi);
343 nbytesp = sizeof(int);
344 }
345 else {
346 printf("Encountered an unknown tag with no length specification : %d\n",tag);
347 val = nullptr;
348 nbytesp = 0;
349 }
350 }
351 return OK;
352}

374int read_next_tag(QFile &fp, int &tagp, long long &lenp, unsigned char *&datap)
375/*
376 * Read the next tag in the file
377 */
378{
379 int ilen,tag;
380 long long len;
381
382 if (read_int(fp,tag) == FAIL) {
383 tagp = 0;
384 return OK;
385 }
386 if (fp.atEnd()) {
387 tagp = 0;
388 return OK;
389 }
390 switch (tag) {
391 case TAG_OLD_MGH_XFORM: /* This is obviously a burden of the past */
392 if (read_int(fp,ilen) == FAIL)
393 return FAIL;
394 len = ilen - 1;
395 break ;
396 case TAG_OLD_SURF_GEOM:
397 case TAG_OLD_USEREALRAS:
398 case TAG_OLD_COLORTABLE:
399 len = 0 ;
400 break ;
401 default:
402 if (read_long(fp,len) == FAIL)
403 return FAIL;
404 break;
405 }
406 lenp = len;
407 tagp = tag;
408 if (read_tag_data(fp,tag,len,datap,lenp) == FAIL)
409 return FAIL;
410 return OK;
411}

440int read_curvature_file(const QString& fname,
441 Eigen::VectorXf& curv)
442
443{
444 QFile fp(fname);
445 int magic;
446
447 float curvmin,curvmax;
448 int ncurv = 0;
449 int nface,val_pervert;
450 int val,k;
451 float fval;
452
453 if (!fp.open(QIODevice::ReadOnly)) {
454 qCritical() << fname;
455 goto bad;
456 }
457 if (read_int3(fp,magic) != 0) {
458 qCritical() << "Bad magic in" << fname;
459 goto bad;
460 }
461 if (magic == CURVATURE_FILE_MAGIC_NUMBER) { /* A new-style curvature file */
462 /*
463 * How many and faces
464 */
465 if (read_int(fp,ncurv) != 0)
466 goto bad;
467 if (read_int(fp,nface) != 0)
468 goto bad;
469#ifdef DEBUG
470 printf("nvert = %d nface = %d\n",ncurv,nface);
471#endif
472 if (read_int(fp,val_pervert) != 0)
473 goto bad;
474 if (val_pervert != 1) {
475 qCritical("Values per vertex not equal to one.");
476 goto bad;
477 }
478 /*
479 * Read the curvature values
480 */
481 curv.resize(ncurv);
482 curvmin = curvmax = 0.0;
483 for (k = 0; k < ncurv; k++) {
484 if (read_float(fp,fval) != 0)
485 goto bad;
486 curv[k] = fval;
487 if (curv[k] > curvmax)
488 curvmax = curv[k];
489 if (curv[k] < curvmin)
490 curvmin = curv[k];
491 }
492 }
493 else { /* An old-style curvature file */
494 ncurv = magic;
495 /*
496 * How many vertices
497 */
498 if (read_int3(fp,nface) != 0)
499 goto bad;
500#ifdef DEBUG
501 printf("nvert = %d nface = %d\n",ncurv,nface);
502#endif
503 /*
504 * Read the curvature values
505 */
506 curv.resize(ncurv);
507 curvmin = curvmax = 0.0;
508 for (k = 0; k < ncurv; k++) {
509 if (read_int2(fp,val) != 0)
510 goto bad;
511 curv[k] = (float)val/100.0;
512 if (curv[k] > curvmax)
513 curvmax = curv[k];
514 if (curv[k] < curvmin)
515 curvmin = curv[k];
516
517 }
518 }
519#ifdef DEBUG
520 printf("Curvature range: %f...%f\n",curvmin,curvmax);
521#endif
522 return OK;
523
524bad : {
525 curv.resize(0);
526 return FAIL;
527 }
528}

534int read_triangle_file(const QString& fname,
535 PointsT& vertices,
536 TrianglesT& triangles,
537 std::optional<MNEMghTagGroup>* tagsp)
538/*
539 * Read the FS triangulated surface
540 */
541{
542 QFile fp(fname);
543 int magic;
544 char c;
545
546 qint32 nvert,ntri,nquad;
547 PointsT vert;
548 TrianglesT tri;
549 int k,p;
550 int quad[4];
551 int val;
552 int which;
553
554 if (!fp.open(QIODevice::ReadOnly)) {
555 qCritical() << fname;
556 goto bad;
557 }
558 if (read_int3(fp,magic) != 0) {
559 qCritical() << "Bad magic in" << fname;
560 goto bad;
561 }
562 if (magic != TRIANGLE_FILE_MAGIC_NUMBER &&
563 magic != QUAD_FILE_MAGIC_NUMBER &&
564 magic != NEW_QUAD_FILE_MAGIC_NUMBER) {
565 qCritical() << "Bad magic in" << fname;
566 goto bad;
567 }
568 if (magic == TRIANGLE_FILE_MAGIC_NUMBER) {
569 /*
570 * Get the comment
571 */
572 printf("Triangle file : ");
573 for (fp.getChar(&c); c != '\n'; fp.getChar(&c)) {
574 if (fp.atEnd()) {
575 qCritical()<<"Bad triangle file.";
576 goto bad;
577 }
578 putc(c,stderr);
579 }
580 fp.getChar(&c);
581 /*
582 * How many vertices and triangles?
583 */
584 if (read_int(fp,nvert) != 0)
585 goto bad;
586 if (read_int(fp,ntri) != 0)
587 goto bad;
588 printf(" nvert = %d ntri = %d\n",nvert,ntri);
589 vert.resize(nvert, 3);
590 tri.resize(ntri, 3);
591 /*
592 * Read the vertices
593 */
594 for (k = 0; k < nvert; k++) {
595 if (read_float(fp,vert(k,X_17)) != 0)
596 goto bad;
597 if (read_float(fp,vert(k,Y_17)) != 0)
598 goto bad;
599 if (read_float(fp,vert(k,Z_17)) != 0)
600 goto bad;
601 }
602 /*
603 * Read the triangles
604 */
605 for (k = 0; k < ntri; k++) {
606 if (read_int(fp,tri(k,X_17)) != 0)
607 goto bad;
608 if (check_vertex(tri(k,X_17),nvert) != OK)
609 goto bad;
610 if (read_int(fp,tri(k,Y_17)) != 0)
611 goto bad;
612 if (check_vertex(tri(k,Y_17),nvert) != OK)
613 goto bad;
614 if (read_int(fp,tri(k,Z_17)) != 0)
615 goto bad;
616 if (check_vertex(tri(k,Z_17),nvert) != OK)
617 goto bad;
618 }
619 }
620 else if (magic == QUAD_FILE_MAGIC_NUMBER ||
621 magic == NEW_QUAD_FILE_MAGIC_NUMBER) {
622 if (read_int3(fp,nvert) != 0)
623 goto bad;
624 if (read_int3(fp,nquad) != 0)
625 goto bad;
626 printf("%s file : nvert = %d nquad = %d\n",
627 magic == QUAD_FILE_MAGIC_NUMBER ? "Quad" : "New quad",
628 nvert,nquad);
629 vert.resize(nvert, 3);
630 if (magic == QUAD_FILE_MAGIC_NUMBER) {
631 for (k = 0; k < nvert; k++) {
632 if (read_int2(fp,val) != 0)
633 goto bad;
634 vert(k,X_17) = val/100.0;
635 if (read_int2(fp,val) != 0)
636 goto bad;
637 vert(k,Y_17) = val/100.0;
638 if (read_int2(fp,val) != 0)
639 goto bad;
640 vert(k,Z_17) = val/100.0;
641 }
642 }
643 else { /* NEW_QUAD_FILE_MAGIC_NUMBER */
644 for (k = 0; k < nvert; k++) {
645 if (read_float(fp,vert(k,X_17)) != 0)
646 goto bad;
647 if (read_float(fp,vert(k,Y_17)) != 0)
648 goto bad;
649 if (read_float(fp,vert(k,Z_17)) != 0)
650 goto bad;
651 }
652 }
653 ntri = 2*nquad;
654 tri.resize(ntri, 3);
655 for (k = 0, ntri = 0; k < nquad; k++) {
656 for (p = 0; p < 4; p++) {
657 if (read_int3(fp,quad[p]) != 0)
658 goto bad;
659 }
660
661 /*
662 * The randomization is borrowed from FreeSurfer code
663 * Strange...
664 */
665#define EVEN(n) ((((n) / 2) * 2) == n)
666#ifdef FOO
667#define WHICH_FACE_SPLIT(vno0, vno1) \
668 (1*nearbyint(sqrt(1.9*vno0) + sqrt(3.5*vno1)))
669
670 which = WHICH_FACE_SPLIT(quad[0], quad[1]) ;
671#endif
672 which = quad[0];
673 /*
674 printf("%f ",sqrt(1.9*quad[0]) + sqrt(3.5*quad[1]));
675 */
676
677 if (EVEN(which)) {
678 tri(ntri,X_17) = quad[0];
679 tri(ntri,Y_17) = quad[1];
680 tri(ntri,Z_17) = quad[3];
681 ntri++;
682
683 tri(ntri,X_17) = quad[2];
684 tri(ntri,Y_17) = quad[3];
685 tri(ntri,Z_17) = quad[1];
686 ntri++;
687 }
688 else {
689 tri(ntri,X_17) = quad[0];
690 tri(ntri,Y_17) = quad[1];
691 tri(ntri,Z_17) = quad[2];
692 ntri++;
693
694 tri(ntri,X_17) = quad[0];
695 tri(ntri,Y_17) = quad[2];
696 tri(ntri,Z_17) = quad[3];
697 ntri++;
698 }
699 }
700 }
701 /*
702 * Optionally read the tags
703 */
704 if (tagsp) {
705 std::optional<MNEMghTagGroup> tags;
706 if (read_mgh_tags(fp, tags) == FAIL) {
707 goto bad;
708 }
709 *tagsp = std::move(tags);
710 }
711 /*
712 * Convert mm to m and store as Eigen matrices
713 */
714 vert /= 1000.0f;
715 vertices = std::move(vert);
716 triangles = std::move(tri);
717 return OK;
718
719bad : {
720 return FAIL;
721 }
722}

901int MNESourceSpace::transform_source_space(const FiffCoordTrans& t)
902/*
903 * Transform source space data into another coordinate frame
904 */
905{
906 int k;
907 if (coord_frame == t.to)
908 return OK;

910 printf("Coordinate transformation does not match with the source space coordinate system.");
911 return FAIL;
912 }
913 for (k = 0; k < np; k++) {
914 FiffCoordTrans::apply_trans(&rr(k,0),t,FIFFV_MOVE);
915 FiffCoordTrans::apply_trans(&nn(k,0),t,FIFFV_NO_MOVE);
916 }
917 if (!tris.empty()) {
918 for (k = 0; k < ntri; k++)
919 FiffCoordTrans::apply_trans(tris[k].nn.data(),t,FIFFV_NO_MOVE);
920 }
921 coord_frame = t.to;
922 return OK;
923}

927int MNESourceSpace::add_patch_stats()
928{
929 MNENearest* nearest_data = nearest.data();
930 MNENearest* this_patch;
931 std::vector<std::optional<MNEPatchInfo>> pinfo(nuse);
932 int nave,p,q,k;
933
934 printf("Computing patch statistics...\n");
935 if (neighbor_tri.empty())
936 if (add_geometry_info(false) != OK)
937 return FAIL;
938
939 if (nearest.empty()) {
940 qCritical("The patch information is not available.");
941 return FAIL;
942 }
943 if (nuse == 0) {
944 patches.clear();
945 return OK;
946 }
947 /*
948 * Calculate the average normals and the patch areas
949 */
950 printf("\tareas, average normals, and mean deviations...");
951 std::sort(nearest.begin(), nearest.end(),
952 [](const MNENearest& a, const MNENearest& b) { return a.nearest < b.nearest; });
953 nearest_data = nearest.data(); // refresh after sort
954 nave = 1;
955 for (p = 1, q = 0; p < np; p++) {
956 if (nearest_data[p].nearest != nearest_data[p-1].nearest) {
957 if (nave == 0) {
958 qCritical("No vertices belong to the patch of vertex %d",nearest_data[p-1].nearest);
959 return FAIL;
960 }
961 if (vertno[q] == nearest_data[p-1].nearest) { /* Some source space points may have been omitted since
962 * the patch information was computed */
963 pinfo[q] = MNEPatchInfo();
964 pinfo[q]->vert = nearest_data[p-1].nearest;
965 this_patch = nearest_data+p-nave;
966 pinfo[q]->memb_vert.resize(nave);
967 for (k = 0; k < nave; k++) {
968 pinfo[q]->memb_vert[k] = this_patch[k].vert;
969 this_patch[k].patch = &(*pinfo[q]);
970 }
971 pinfo[q]->calculate_area(this);
972 pinfo[q]->calculate_normal_stats(this);
973 q++;
974 }
975 nave = 0;
976 }
977 nave++;
978 }
979 if (nave == 0) {
980 qCritical("No vertices belong to the patch of vertex %d",nearest_data[p-1].nearest);
981 return FAIL;
982 }
983 if (vertno[q] == nearest_data[p-1].nearest) {
984 pinfo[q] = MNEPatchInfo();
985 pinfo[q]->vert = nearest_data[p-1].nearest;
986 this_patch = nearest_data+p-nave;
987 pinfo[q]->memb_vert.resize(nave);
988 for (k = 0; k < nave; k++) {
989 pinfo[q]->memb_vert[k] = this_patch[k].vert;
990 this_patch[k].patch = &(*pinfo[q]);
991 }
992 pinfo[q]->calculate_area(this);
993 pinfo[q]->calculate_normal_stats(this);
994 q++;
995 }
996 printf(" %d/%d [done]\n",q,nuse);
997
998 patches = std::move(pinfo);
999
1000 return OK;
1001}

1180
1181MNESourceSpace* MNESourceSpace::make_volume_source_space(const MNESurface& surf, float grid, float exclude, float mindist)
1182/*
1183 * Make a source space which covers the volume bounded by surf
1184 */
1185{
1186 float min[3],max[3],cm[3];
1187 int minn[3],maxn[3];
1188 const float *node;
1189 float maxdist,dist,diff[3];
1190 int k,c;
1191 std::unique_ptr<MNESourceSpace> sp;
1192 int np,nplane,nrow;
1193 int nneigh;
1194 int x,y,z;
1195 /*
1196 * Figure out the grid size
1197 */
1198 cm[X_17] = cm[Y_17] = cm[Z_17] = 0.0;
1199 node = &surf.rr(0,0);
1200 for (c = 0; c < 3; c++)
1201 min[c] = max[c] = node[c];
1202
1203 for (k = 0; k < surf.np; k++) {
1204 node = &surf.rr(k,0);
1205 for (c = 0; c < 3; c++) {
1206 cm[c] += node[c];
1207 if (node[c] < min[c])
1208 min[c] = node[c];
1209 if (node[c] > max[c])
1210 max[c] = node[c];
1211 }
1212 }
1213 for (c = 0; c < 3; c++)
1214 cm[c] = cm[c]/surf.np;
1215 /*
1216 * Define the sphere which fits the surface
1217 */
1218 maxdist = 0.0;
1219 for (k = 0; k < surf.np; k++) {
1220 VEC_DIFF_17(cm,&surf.rr(k,0),diff);
1221 dist = VEC_LEN_17(diff);
1222 if (dist > maxdist)
1223 maxdist = dist;
1224 }
1225 printf("FsSurface CM = (%6.1f %6.1f %6.1f) mm\n",
1226 1000*cm[X_17], 1000*cm[Y_17], 1000*cm[Z_17]);
1227 printf("FsSurface fits inside a sphere with radius %6.1f mm\n",1000*maxdist);
1228 printf("FsSurface extent:\n"
1229 "\tx = %6.1f ... %6.1f mm\n"
1230 "\ty = %6.1f ... %6.1f mm\n"
1231 "\tz = %6.1f ... %6.1f mm\n",
1232 1000*min[X_17],1000*max[X_17],
1233 1000*min[Y_17],1000*max[Y_17],
1234 1000*min[Z_17],1000*max[Z_17]);
1235 for (c = 0; c < 3; c++) {
1236 if (max[c] > 0)
1237 maxn[c] = floor(std::fabs(max[c])/grid)+1;
1238 else
1239 maxn[c] = -floor(std::fabs(max[c])/grid)-1;
1240 if (min[c] > 0)
1241 minn[c] = floor(std::fabs(min[c])/grid)+1;
1242 else
1243 minn[c] = -floor(std::fabs(min[c])/grid)-1;
1244 }
1245 printf("Grid extent:\n"
1246 "\tx = %6.1f ... %6.1f mm\n"
1247 "\ty = %6.1f ... %6.1f mm\n"
1248 "\tz = %6.1f ... %6.1f mm\n",
1249 1000*(minn[X_17]*grid),1000*(maxn[X_17]*grid),
1250 1000*(minn[Y_17]*grid),1000*(maxn[Y_17]*grid),
1251 1000*(minn[Z_17]*grid),1000*(maxn[Z_17]*grid));
1252 /*
1253 * Now make the initial grid
1254 */
1255 np = 1;
1256 for (c = 0; c < 3; c++)
1257 np = np*(maxn[c]-minn[c]+1);
1258 nplane = (maxn[X_17]-minn[X_17]+1)*(maxn[Y_17]-minn[Y_17]+1);
1259 nrow = (maxn[X_17]-minn[X_17]+1);
1260 sp = MNESourceSpace::create_source_space(np);
1261 sp->type = MNE_SOURCE_SPACE_VOLUME;
1262 sp->nneighbor_vert = Eigen::VectorXi::Constant(sp->np, NNEIGHBORS);
1263 sp->neighbor_vert.resize(sp->np);
1264 for (k = 0; k < sp->np; k++) {
1265 sp->inuse[k] = 1;
1266 sp->vertno[k] = k;
1267 sp->nn(k,X_17) = sp->nn(k,Y_17) = 0.0; /* Source orientation is immaterial */
1268 sp->nn(k,Z_17) = 1.0;
1269 sp->neighbor_vert[k] = Eigen::VectorXi::Constant(NNEIGHBORS, -1);
1270 sp->nuse++;
1271 }
1272 for (k = 0, z = minn[Z_17]; z <= maxn[Z_17]; z++) {
1273 for (y = minn[Y_17]; y <= maxn[Y_17]; y++) {
1274 for (x = minn[X_17]; x <= maxn[X_17]; x++, k++) {
1275 sp->rr(k,X_17) = x*grid;
1276 sp->rr(k,Y_17) = y*grid;
1277 sp->rr(k,Z_17) = z*grid;
1278 /*
1279 * Figure out the neighborhood:
1280 * 6-neighborhood first
1281 */
1282 Eigen::VectorXi& neigh = sp->neighbor_vert[k];
1283 if (z > minn[Z_17])
1284 neigh[0] = k - nplane;
1285 if (x < maxn[X_17])
1286 neigh[1] = k + 1;
1287 if (y < maxn[Y_17])
1288 neigh[2] = k + nrow;
1289 if (x > minn[X_17])
1290 neigh[3] = k - 1;
1291 if (y > minn[Y_17])
1292 neigh[4] = k - nrow;
1293 if (z < maxn[Z_17])
1294 neigh[5] = k + nplane;
1295 /*
1296 * Then the rest to complete the 26-neighborhood
1297 * First the plane below
1298 */
1299 if (z > minn[Z_17]) {
1300 if (x < maxn[X_17]) {
1301 neigh[6] = k + 1 - nplane;
1302 if (y < maxn[Y_17])
1303 neigh[7] = k + 1 + nrow - nplane;
1304 }
1305 if (y < maxn[Y_17])
1306 neigh[8] = k + nrow - nplane;
1307 if (x > minn[X_17]) {
1308 if (y < maxn[Y_17])
1309 neigh[9] = k - 1 + nrow - nplane;
1310 neigh[10] = k - 1 - nplane;
1311 if (y > minn[Y_17])
1312 neigh[11] = k - 1 - nrow - nplane;
1313 }
1314 if (y > minn[Y_17]) {
1315 neigh[12] = k - nrow - nplane;
1316 if (x < maxn[X_17])
1317 neigh[13] = k + 1 - nrow - nplane;
1318 }
1319 }
1320 /*
1321 * Then the same plane
1322 */
1323 if (x < maxn[X_17] && y < maxn[Y_17])
1324 neigh[14] = k + 1 + nrow;
1325 if (x > minn[X_17]) {
1326 if (y < maxn[Y_17])
1327 neigh[15] = k - 1 + nrow;
1328 if (y > minn[Y_17])
1329 neigh[16] = k - 1 - nrow;
1330 }
1331 if (y > minn[Y_17] && x < maxn[X_17])
1332 neigh[17] = k + 1 - nrow - nplane;
1333 /*
1334 * Finally one plane above
1335 */
1336 if (z < maxn[Z_17]) {
1337 if (x < maxn[X_17]) {
1338 neigh[18] = k + 1 + nplane;
1339 if (y < maxn[Y_17])
1340 neigh[19] = k + 1 + nrow + nplane;
1341 }
1342 if (y < maxn[Y_17])
1343 neigh[20] = k + nrow + nplane;
1344 if (x > minn[X_17]) {
1345 if (y < maxn[Y_17])
1346 neigh[21] = k - 1 + nrow + nplane;
1347 neigh[22] = k - 1 + nplane;
1348 if (y > minn[Y_17])
1349 neigh[23] = k - 1 - nrow + nplane;
1350 }
1351 if (y > minn[Y_17]) {
1352 neigh[24] = k - nrow + nplane;
1353 if (x < maxn[X_17])
1354 neigh[25] = k + 1 - nrow + nplane;
1355 }
1356 }
1357 }
1358 }
1359 }
1360 printf("%d sources before omitting any.\n",sp->nuse);
1361 /*
1362 * Exclude infeasible points
1363 */
1364 for (k = 0; k < sp->np; k++) {
1365 VEC_DIFF_17(cm,&sp->rr(k,0),diff);
1366 dist = VEC_LEN_17(diff);
1367 if (dist < exclude || dist > maxdist) {
1368 sp->inuse[k] = 0;
1369 sp->nuse--;
1370 }
1371 }
1372 printf("%d sources after omitting infeasible sources.\n",sp->nuse);
1373 {
1374 std::vector<std::unique_ptr<MNESourceSpace>> sp_vec;
1375 sp_vec.push_back(std::move(sp));
1376 if (filter_source_spaces(surf,mindist,FiffCoordTrans(),sp_vec,nullptr) != OK) {
1377 sp = std::move(sp_vec[0]);
1378 goto bad;
1379 }
1380 sp = std::move(sp_vec[0]);
1381 }
1382 printf("%d sources remaining after excluding the sources outside the surface and less than %6.1f mm inside.\n",sp->nuse,1000*mindist);
1383 /*
1384 * Omit unused vertices from the neighborhoods
1385 */
1386 printf("Adjusting the neighborhood info...");
1387 for (k = 0; k < sp->np; k++) {
1388 Eigen::VectorXi& neigh = sp->neighbor_vert[k];
1389 nneigh = sp->nneighbor_vert[k];
1390 if (sp->inuse[k]) {
1391 for (c = 0; c < nneigh; c++)
1392 if (!sp->inuse[neigh[c]])
1393 neigh[c] = -1;
1394 }
1395 else {
1396 for (c = 0; c < nneigh; c++)
1397 neigh[c] = -1;
1398 }
1399 }
1400 printf("[done]\n");
1401 /*
1402 * Set up the volume data (needed for creating the interpolation matrix)
1403 */
1404 {
1405 float r0[3],voxel_size[3],x_ras[3],y_ras[3],z_ras[3];
1406 int width,height,depth;
1407
1408 r0[X_17] = minn[X_17]*grid;
1409 r0[Y_17] = minn[Y_17]*grid;
1410 r0[Z_17] = minn[Z_17]*grid;
1411
1412 voxel_size[0] = grid;
1413 voxel_size[1] = grid;
1414 voxel_size[2] = grid;
1415
1416 width = (maxn[X_17]-minn[X_17]+1);
1417 height = (maxn[Y_17]-minn[Y_17]+1);
1418 depth = (maxn[Z_17]-minn[Z_17]+1);
1419
1420 for (k = 0; k < 3; k++)
1421 x_ras[k] = y_ras[k] = z_ras[k] = 0.0;
1422
1423 x_ras[0] = 1.0;
1424 y_ras[1] = 1.0;
1425 z_ras[2] = 1.0;
1426
1427 sp->voxel_surf_RAS_t = make_voxel_ras_trans(r0,x_ras,y_ras,z_ras,voxel_size);
1428 if (!sp->voxel_surf_RAS_t || sp->voxel_surf_RAS_t->isEmpty())
1429 goto bad;
1430
1431 sp->vol_dims[0] = width;
1432 sp->vol_dims[1] = height;
1433 sp->vol_dims[2] = depth;
1434 VEC_COPY_17(sp->voxel_size,voxel_size);
1435 }
1436
1437 return sp.release();
1438
1439bad : {
1440 return nullptr;
1441 }
1442}

1699int MNESourceSpace::read_source_spaces(const QString &name, std::vector<std::unique_ptr<MNESourceSpace>>& spaces)
1700/*
1701 * Read source spaces from a FIFF file
1702 */
1703{
1704 QFile file(name);

1706
1707 std::vector<std::unique_ptr<MNESourceSpace>> local_spaces;
1708 std::unique_ptr<MNESourceSpace> new_space;
1709 QList<FiffDirNode::SPtr> sources;
1710 FiffDirNode::SPtr node;
1711 FiffTag::UPtr t_pTag;
1712 int j,k,p,q;
1713 int ntri;
1714
1715 if(!stream->open())
1716 goto bad;
1717
1718 sources = stream->dirtree()->dir_tree_find(FIFFB_MNE_SOURCE_SPACE);
1719 if (sources.size() == 0) {
1720 printf("No source spaces available here");
1721 goto bad;
1722 }
1723 for (j = 0; j < sources.size(); j++) {
1724 new_space = MNESourceSpace::create_source_space(0);
1725 node = sources[j];
1726 /*
1727 * Get the mandatory data first
1728 */
1729 if (!node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NPOINTS, t_pTag)) {
1730 goto bad;
1731 }
1732 new_space->np = *t_pTag->toInt();

1734 printf("No points in this source space");
1735 goto bad;
1736 }
1737 if (!node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_POINTS, t_pTag)) {
1738 goto bad;
1739 }
1740 MatrixXf tmp_rr = t_pTag->toFloatMatrix().transpose();
1741 new_space->rr = tmp_rr;
1742 if (!node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NORMALS, t_pTag)) {
1743 goto bad;
1744 }
1745 MatrixXf tmp_nn = t_pTag->toFloatMatrix().transpose();
1746 new_space->nn = tmp_nn;
1747 if (!node->find_tag(stream, FIFF_MNE_COORD_FRAME, t_pTag)) {
1748 new_space->coord_frame = FIFFV_COORD_MRI;
1749 }
1750 else {
1751 new_space->coord_frame = *t_pTag->toInt();
1752 }
1753 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_ID, t_pTag)) {
1754 new_space->id = *t_pTag->toInt();
1755 }
1756 if (node->find_tag(stream, FIFF_SUBJ_HIS_ID, t_pTag)) {
1757 new_space->subject = t_pTag->toString();
1758 }
1759 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_TYPE, t_pTag)) {
1760 new_space->type = *t_pTag->toInt();
1761 }
1762 ntri = 0;
1763 if (node->find_tag(stream, FIFF_BEM_SURF_NTRI, t_pTag)) {
1764 ntri = *t_pTag->toInt();
1765 }
1766 else if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NTRI, t_pTag)) {
1767 ntri = *t_pTag->toInt();
1768 }
1769 if (ntri > 0) {
1770
1771 if (!node->find_tag(stream, FIFF_BEM_SURF_TRIANGLES, t_pTag)) {
1772 if (!node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_TRIANGLES, t_pTag)) {
1773 goto bad;
1774 }
1775 }
1776
1777 MatrixXi tmp_itris = t_pTag->toIntMatrix().transpose();
1778 tmp_itris.array() -= 1;
1779 new_space->itris = tmp_itris;
1780 new_space->ntri = ntri;
1781 }
1782 if (!node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NUSE, t_pTag)) {
1783 if (new_space->type == FIFFV_MNE_SPACE_VOLUME) {
1784 /*
1785 * Use all
1786 */
1787 new_space->nuse = new_space->np;
1788 new_space->inuse = Eigen::VectorXi::Ones(new_space->nuse);
1789 new_space->vertno = Eigen::VectorXi::LinSpaced(new_space->nuse, 0, new_space->nuse - 1);
1790 }
1791 else {
1792 /*
1793 * None in use
1794 * NOTE: The consequences of this change have to be evaluated carefully
1795 */
1796 new_space->nuse = 0;
1797 new_space->inuse = Eigen::VectorXi::Zero(new_space->np);
1798 new_space->vertno.resize(0);
1799 }
1800 }
1801 else {
1802 new_space->nuse = *t_pTag->toInt();
1803 if (!node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_SELECTION, t_pTag)) {
1804 goto bad;
1805 }
1806
1807 {
1808 Eigen::Map<Eigen::VectorXi> inuseMap(t_pTag->toInt(), new_space->np);
1809 new_space->inuse = inuseMap;
1810 }
1811 if (new_space->nuse > 0) {
1812 new_space->vertno = Eigen::VectorXi::Zero(new_space->nuse);
1813 for (k = 0, p = 0; k < new_space->np; k++) {
1814 if (new_space->inuse[k])
1815 new_space->vertno[p++] = k;
1816 }
1817 }
1818 else {
1819 new_space->vertno.resize(0);
1820 }
1821 /*
1822 * Selection triangulation
1823 */
1824 ntri = 0;
1825 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NUSE_TRI, t_pTag)) {
1826 ntri = *t_pTag->toInt();
1827 }
1828 if (ntri > 0) {
1829
1830 if (!node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_USE_TRIANGLES, t_pTag)) {
1831 goto bad;
1832 }
1833
1834 MatrixXi tmp_itris = t_pTag->toIntMatrix().transpose();
1835 tmp_itris.array() -= 1;
1836 new_space->use_itris = tmp_itris;
1837 new_space->nuse_tri = ntri;
1838 }
1839 /*
1840 * The patch information becomes relevant here
1841 */
1842 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NEAREST, t_pTag)) {
1843 Eigen::Map<Eigen::VectorXi> nearestMap(t_pTag->toInt(), new_space->np);
1844 new_space->nearest.resize(new_space->np);
1845 for (k = 0; k < new_space->np; k++) {
1846 new_space->nearest[k].vert = k;
1847 new_space->nearest[k].nearest = nearestMap[k];
1848 new_space->nearest[k].patch = nullptr;
1849 }
1850
1851 if (!node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NEAREST_DIST, t_pTag)) {
1852 goto bad;
1853 }
1854 Eigen::Map<Eigen::VectorXf> nearestDistMap(t_pTag->toFloat(), new_space->np);
1855 for (k = 0; k < new_space->np; k++) {
1856 new_space->nearest[k].dist = nearestDistMap[k];
1857 }
1858 }
1859 /*
1860 * We may have the distance matrix
1861 */
1862 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_DIST_LIMIT, t_pTag)) {
1863 new_space->dist_limit = *t_pTag->toFloat();
1864 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_DIST, t_pTag)) {
1865 // SparseMatrix<double> tmpSparse = t_pTag->toSparseFloatMatrix();

1867 if (!dist_lower) {
1868 goto bad;
1869 }
1870 auto dist_full = dist_lower->mne_add_upper_triangle_rcs();
1871 if (!dist_full) {
1872 goto bad;
1873 }
1874 new_space->dist = std::move(*dist_full);
1875 }
1876 else
1877 new_space->dist_limit = 0.0;
1878 }
1879 }
1880 /*
1881 * For volume source spaces we might have the neighborhood information
1882 */
1883 if (new_space->type == FIFFV_MNE_SPACE_VOLUME) {
1884 int ntot,nvert,ntot_count,nneigh;
1885

1893 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_NNEIGHBORS, t_pTag)) {

1896 }
1897 if (neighborsVec.size() > 0 && nneighborsVec.size() > 0) {
1898 if (nvert != new_space->np) {
1899 printf("Inconsistent neighborhood data in file.");
1900 goto bad;
1901 }
1902 for (k = 0, ntot_count = 0; k < nvert; k++)
1903 ntot_count += nneighborsVec[k];
1904 if (ntot_count != ntot) {
1905 printf("Inconsistent neighborhood data in file.");
1906 goto bad;
1907 }

1915 }
1916 }

1919 */
1920 new_space->voxel_surf_RAS_t = FiffCoordTrans(FiffCoordTrans::readTransformFromNode(stream, node, FIFFV_MNE_COORD_MRI_VOXEL, FIFFV_MNE_COORD_SURFACE_RAS));

1930 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_MRI_FILE, t_pTag)) {
1931 new_space->MRI_volume = t_pTag->toString();
1932 }
1933 if (node->find_tag(stream, FIFF_MNE_SOURCE_SPACE_INTERPOLATOR, t_pTag)) {

1940 }
