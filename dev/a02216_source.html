40RtSourceDataController::RtSourceDataController(QObject* parent)
41: QObject(parent)
42{
43 // Create data worker
44 // The worker stays on the GUI thread; streamData() runs on the streaming thread, so its signals reach
45 // this controller queued.
46 m_pWorker = new mne::viz::qt::RtSourceDataWorker(this);
47
48 // Forward worker signals to controller signals
49 connect(m_pWorker, &mne::viz::qt::RtSourceDataWorker::newRtSmoothedData,
50 this, &RtSourceDataController::newSmoothedDataAvailable);
51 connect(m_pWorker, &mne::viz::qt::RtSourceDataWorker::newRtRawData,
52 this, &RtSourceDataController::newRawDataAvailable);
53
54 // Streaming thread driving the cadence
55 m_pStreamTask = std::make_unique<PeriodicTask>([worker = m_pWorker] { worker->streamData(); });
56
57 // Create interpolation matrix worker
58 // One pool thread runs the recomputations in request order.
59 m_pInterpWorker = new mne::viz::qt::RtSourceInterpolationMatWorker(this);
60 m_pInterpPool = std::make_unique<mne::ThreadPool>(1);
61
62 // Forward interpolation results: auto-apply to data worker + re-emit
63 connect(m_pInterpWorker, &mne::viz::qt::RtSourceInterpolationMatWorker::newInterpolationMatrixLeftAvailable,
64 this, &RtSourceDataController::onNewInterpolationMatrixLeft);
65 connect(m_pInterpWorker, &mne::viz::qt::RtSourceInterpolationMatWorker::newInterpolationMatrixRightAvailable,
66 this, &RtSourceDataController::onNewInterpolationMatrixRight);
67
68 mne::Log::debug("RtSourceDataController: Initialized with {} ms interval", m_iTimeInterval);
69}

73RtSourceDataController::~RtSourceDataController()
74{
75 // Join the threads while the workers (children of this object) are still alive. Pending
76 // recomputations are skipped.
77 m_bShuttingDown.store(true, std::memory_order_release);
78 m_pStreamTask.reset();
79 m_pInterpPool.reset();
80}
