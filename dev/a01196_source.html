213bool PolhemusCoregistration::compute_registration()
214{
215 if (!has_all_pen_fiducials()) {
216 return false;
217 }
218
219 // Pen fiducial positions in Polhemus world frame (metres)
220 const Eigen::Vector3f pNas = m_penFid[static_cast<int>(FiducialId::NAS)];
221 const Eigen::Vector3f pLpa = m_penFid[static_cast<int>(FiducialId::LPA)];
222 const Eigen::Vector3f pRpa = m_penFid[static_cast<int>(FiducialId::RPA)];
223
224 // Spread check: reject degenerate input
225 const float dNL = (pNas - pLpa).norm() * 1000.0f;
226 const float dNR = (pNas - pRpa).norm() * 1000.0f;
227 const float dLR = (pLpa - pRpa).norm() * 1000.0f;
228 const float minSpread = std::min({dNL, dNR, dLR});
229 if (minSpread < 20.0f) {
230 mne::Log::warning("Registration failed: pen fiducials too close (min spread {} mm, need > 20 mm)", minSpread);
231 m_registrationValid = false;
232 notify(m_listener.registration_changed);
233 return false;
234 }
235
236 // --- Paired path: SVD-based rigid registration (Kabsch / Procrustes) ---
237 //
238 // Finds the optimal rotation R and translation t that map pen
239 // fiducials to model fiducials: model_pos ≈ R * pen_pos + t
240 //
241 // The Kabsch algorithm guarantees det(R) = +1 (proper rotation,
242 // no reflection) regardless of how the two coordinate systems are
243 // oriented. This avoids the left-right / up-down inversion bugs
244 // that plagued the old buildFrame approach, where an asymmetric
245 // vertex correction could flip ey or ez in one frame but not the
246 // other.
247 if (has_all_model_fiducials()) {
248 const Eigen::Vector3f mNas = m_modelFid[static_cast<int>(FiducialId::NAS)];
249 const Eigen::Vector3f mLpa = m_modelFid[static_cast<int>(FiducialId::LPA)];
250 const Eigen::Vector3f mRpa = m_modelFid[static_cast<int>(FiducialId::RPA)];
251
252 // Compare pen vs model fiducial distances — should be similar for the same head
253 const float mNL = (mNas - mLpa).norm() * 1000.0f;
254 const float mNR = (mNas - mRpa).norm() * 1000.0f;
255 const float mLR = (mLpa - mRpa).norm() * 1000.0f;
256
257 const float maxRatio = std::max({dNL / mNL, dNR / mNR, dLR / mLR});
258 const float minRatio = std::min({dNL / mNL, dNR / mNR, dLR / mLR});
259 if (maxRatio / minRatio > 2.0f) {
260 mne::Log::warning("Registration: shape mismatch — pen fiducial triangle has very different proportions from model. Check fiducial placement!");
261 }
262
263 // --- Kabsch algorithm (SVD least-squares rigid alignment) ---
264
265 // 1. Centroids
266 const Eigen::Vector3d penC = (pNas + pLpa + pRpa).cast<double>() / 3.0;
267 const Eigen::Vector3d modC = (mNas + mLpa + mRpa).cast<double>() / 3.0;
268
269 // 2. Centered point matrices (3×3, each column = one centered point)
270 Eigen::Matrix3d P, Q;
271 P.col(0) = pNas.cast<double>() - penC;
272 P.col(1) = pLpa.cast<double>() - penC;
273 P.col(2) = pRpa.cast<double>() - penC;
274
275 Q.col(0) = mNas.cast<double>() - modC;
276 Q.col(1) = mLpa.cast<double>() - modC;
277 Q.col(2) = mRpa.cast<double>() - modC;
278
279 // 3. Cross-covariance matrix H = P * Qᵀ
280 const Eigen::Matrix3d H = P * Q.transpose();
281
282 // 4. SVD of H
283 Eigen::JacobiSVD<Eigen::Matrix3d> svd(H, Eigen::ComputeFullU | Eigen::ComputeFullV);
284 const Eigen::Matrix3d U = svd.matrixU();
285 const Eigen::Matrix3d V = svd.matrixV();
286
287 // 5. Optimal rotation — ensure proper rotation (det = +1)
288 const double d = (V * U.transpose()).determinant();
289 Eigen::Matrix3d D = Eigen::Matrix3d::Identity();
290 D(2, 2) = (d > 0.0) ? 1.0 : -1.0;
291 Eigen::Matrix3d R = V * D * U.transpose();
292
293 // 6. Translation
294 Eigen::Vector3d t = modC - R * penC;
295
296 // 6b. Vertex disambiguation for coplanar degeneracy.
297 //
298 // With only 3 coplanar fiducials the SVD's 3rd singular value
299 // is ~0, leaving the out-of-plane rotation direction ambiguous.
300 // The det(V*Uᵀ) sign heuristic picks one direction but may
301 // choose wrong, inverting superior ↔ inferior.
302 //
303 // Fix: if a pen vertex (CZ) and model vertex are available,
304 // test both candidate rotations (D₃₃ = +1 and D₃₃ = -1) and
305 // keep whichever maps pen CZ closer to model CZ.
306 if (m_hasPenVertex && m_hasModelVertex) {
307 const Eigen::Vector3d penCZ = m_penVertex.cast<double>();
308 const Eigen::Vector3d modCZ = m_modelVertex.cast<double>();
309
310 const Eigen::Vector3d mappedCZ = R * penCZ + t;
311 const double errCurrent = (mappedCZ - modCZ).norm();
312
313 // Try the alternative sign
314 Eigen::Matrix3d D_alt = D;
315 D_alt(2, 2) = -D(2, 2);
316 const Eigen::Matrix3d R_alt = V * D_alt * U.transpose();
317 const Eigen::Vector3d t_alt = modC - R_alt * penC;
318 const Eigen::Vector3d mappedCZ_alt = R_alt * penCZ + t_alt;
319 const double errAlt = (mappedCZ_alt - modCZ).norm();
320
321 if (errAlt < errCurrent) {
322 R = R_alt;
323 t = t_alt;
324 D = D_alt;
325 }
326 }
327
328 // 7. world_to_model = [R | t]
329 m_worldToModel.setIdentity();
330 m_worldToModel.topLeftCorner<3, 3>() = R.cast<float>();
331 m_worldToModel.topRightCorner<3, 1>() = t.cast<float>();
332
333 m_headToWorld.setIdentity();
334 m_headToDevice = m_deviceToWorld.inverse() * m_headToWorld;
335 m_registrationValid = true;
336 mne::Log::info("Registration succeeded (analytical paired).");
337 notify(m_listener.registration_changed);
338 return true;
339 }
340
341 // --- Fallback: head-frame-only registration ---
342 const Eigen::Matrix4f headFrame = buildHeadFrame();
343 if (headFrame.block<3, 1>(0, 2).norm() < 1e-6f) {
344 return false;
345 }
346
347 m_headToWorld = headFrame;
348 m_headToDevice = m_deviceToWorld.inverse() * m_headToWorld;
349 m_worldToModel.setIdentity();
350 m_registrationValid = true;
351 mne::Log::info("Registration succeeded (head-frame fallback).");
352 notify(m_listener.registration_changed);
353 return true;
354}

358void PolhemusCoregistration::on_pose(int station, const Eigen::Vector3f& position, const Eigen::Quaternionf& orientation)
359{
360 // Apply axis mirroring to compensate for transmitter placement
361 const Eigen::Vector3f pos(m_mirrorX ? -position.x() : position.x(),
362 m_mirrorY ? -position.y() : position.y(),
363 position.z());
364
365 if (station == m_trackerStation) {
366 m_deviceToWorld = buildDevicePose(pos, orientation);
367 notify(m_listener.device_pose_changed, m_deviceToWorld);
368 } else if (station == m_penStation) {
369 // Gimbal-lock guard: for ZYX Euler, sin(el) = 2*(w*y - x*z).
370 // When |el| > 80° the Euler→quaternion conversion is unreliable,
371 // so freeze the pen pose at its last good value.
372 const float sinEl = 2.0f * (orientation.w() * orientation.y() - orientation.x() * orientation.z());
373 constexpr float kGimbalSinEl = 0.9848f; // sin(80°)
374 const bool gimbalLock = (std::abs(sinEl) > kGimbalSinEl);
375
376 if (!gimbalLock) {
377 const Eigen::Vector3f tipAdj = m_tipOffsetEnabled
378 ? Eigen::Vector3f(orientation * m_penTipOffset)
379 : Eigen::Vector3f::Zero();
380 m_penPosition = pos + tipAdj;
381 m_penOrientation = orientation;
382 m_havePenPos = true;
383 }
384
385 // Collect raw (un-offset) samples during pivot calibration.
386 // Only keep samples with sufficient angular change from the last
387 // accepted sample to avoid redundant near-identical rows in the SVD.
388 // Also reject position jumps and gimbal-lock orientations.
389 if (m_pivotState == PivotState::Collecting) {
390 constexpr float kMinAngleDeg = 3.0f;
391 constexpr float kMaxPosJumpM = 0.05f; // 5 cm
392 constexpr float kGimbalSinEl2 = 0.9848f; // sin(80°) — reject |el|>80°
393
394 // Gimbal-lock check: for ZYX Euler, sin(el) = 2*(w*y - x*z)
395 const float sinEl2 = 2.0f * (orientation.w() * orientation.y() - orientation.x() * orientation.z());
396 if (std::abs(sinEl2) > kGimbalSinEl2) {
397 // Near gimbal lock — skip this sample silently
398 } else {
399 bool accept = m_pivotOrientations.empty();
400 if (!accept) {
401 const Eigen::Quaternionf& prev = m_pivotOrientations.back();
402 float dot = std::abs(prev.dot(orientation));
403 float angleDeg = 2.0f * std::acos(std::min(dot, 1.0f)) * (180.0f / 3.14159265f);
404 float posDelta = (pos - m_pivotPositions.back()).norm();
405 accept = (angleDeg >= kMinAngleDeg) && (posDelta < kMaxPosJumpM);
406 }
407 if (accept) {
408 m_pivotPositions.push_back(pos);
409 m_pivotOrientations.push_back(orientation);
410
411 // Compute angular span for live feedback
412 float spanDeg = 0.0f;
413 if (m_pivotOrientations.size() > 1) {
414 float minDot = 1.0f;
415 const auto& first = m_pivotOrientations.front();
416 for (size_t k = 1; k < m_pivotOrientations.size(); ++k) {
417 float d = std::abs(first.dot(m_pivotOrientations[k]));
418 if (d < minDot)
419 minDot = d;
420 }
421 spanDeg = 2.0f * std::acos(std::min(minDot, 1.0f)) * (180.0f / 3.14159265f);
422 }
423 notify(m_listener.pivot_sample_collected, static_cast<int>(m_pivotPositions.size()), spanDeg);
424 }
425 }
426 }
427
428 notify(m_listener.pen_pose_changed, m_penPosition, m_penOrientation);
429 } else if (station == m_probeStation) {
430 m_probePosition = pos;
431 m_probeOrientation = orientation;
432 m_haveProbePos = true;
433 notify(m_listener.probe_pose_changed, m_probePosition, m_probeOrientation);
434 }
435}

478Eigen::Matrix4f PolhemusCoregistration::buildHeadFrame() const
479{
480 const Eigen::Vector3f nas = m_points.fiducial(FiducialId::NAS);
481 const Eigen::Vector3f lpa = m_points.fiducial(FiducialId::LPA);
482 const Eigen::Vector3f rpa = m_points.fiducial(FiducialId::RPA);
483
484 const Eigen::Vector3f origin = (lpa + rpa) * 0.5f;
485 const Eigen::Vector3f ex = (nas - origin).normalized();
486 const Eigen::Vector3f eyApprox = (lpa - origin).normalized();
487 // Gram-Schmidt: orthogonalize ey against ex, preserving LPA direction
488 const Eigen::Vector3f ey = (eyApprox - eyApprox.dot(ex) * ex).normalized();
489 const Eigen::Vector3f ez = ex.cross(ey).normalized();
490
491 Eigen::Matrix4f frame = Eigen::Matrix4f::Identity();
492 frame.block<3, 1>(0, 0) = ex;
493 frame.block<3, 1>(0, 1) = ey;
494 frame.block<3, 1>(0, 2) = ez;
495 frame.block<3, 1>(0, 3) = origin;
496 return frame;
497}

521bool PolhemusCoregistration::solvePivotCalibration()
522{
523 const int N = static_cast<int>(m_pivotPositions.size());
524 if (N < 10) {
525 mne::Log::warning("Pivot calibration: only {} samples, need at least 10", N);
526 m_pivotState = PivotState::Idle;
527 notify(m_listener.pivot_state_changed, m_pivotState);
528 return false;
529 }
530
531 // Build the linear system A * x = b (double precision)
532 // where x = [offset(3); tipPos(3)]
533 // For each sample i: R_i * offset - I * tipPos = -p_i
534 Eigen::MatrixXd A(3 * N, 6);
535 Eigen::VectorXd b(3 * N);
536
537 for (int i = 0; i < N; ++i) {
538 const Eigen::Vector3f& p = m_pivotPositions[static_cast<size_t>(i)];
539 const Eigen::Quaternionf& q = m_pivotOrientations[static_cast<size_t>(i)];
540
541 const Eigen::Matrix3f rm = q.toRotationMatrix();
542
543 const int row = 3 * i;
544 for (int r = 0; r < 3; ++r) {
545 for (int c = 0; c < 3; ++c) {
546 A(row + r, c) = static_cast<double>(rm(r, c)); // R_i
547 A(row + r, 3 + c) = (r == c) ? -1.0 : 0.0; // -I
548 }
549 }
550 b(row + 0) = -static_cast<double>(p.x());
551 b(row + 1) = -static_cast<double>(p.y());
552 b(row + 2) = -static_cast<double>(p.z());
553 }
554
555 // Solve via SVD least squares
556 Eigen::JacobiSVD<Eigen::MatrixXd> svd(A, Eigen::ComputeThinU | Eigen::ComputeThinV);
557
558 // Check condition number for rank deficiency (insufficient angular diversity)
559 const auto& sv = svd.singularValues();
560 double cond = sv(0) / sv(sv.size() - 1);
561 if (cond > 1e6) {
562 mne::Log::warning("Pivot calibration: ill-conditioned (cond = {} ). Pivot the pen through a wider range of angles.", cond);
563 m_pivotState = PivotState::Idle;
564 notify(m_listener.pivot_state_changed, m_pivotState);
565 return false;
566 }
567
568 Eigen::VectorXd x = svd.solve(b);
569
570 const Eigen::Vector3f offset = x.head<3>().cast<float>();
571 const Eigen::Vector3f tipPos = x.tail<3>().cast<float>();
572
573 // Compute RMS residual
574 double sumSq = 0.0;
575 for (int i = 0; i < N; ++i) {
576 const Eigen::Vector3f& p = m_pivotPositions[static_cast<size_t>(i)];
577 const Eigen::Quaternionf& q = m_pivotOrientations[static_cast<size_t>(i)];
578 const Eigen::Vector3f computed = p + q * offset;
579 float err = (computed - tipPos).norm();
580 sumSq += static_cast<double>(err) * static_cast<double>(err);
581 }
582 float rms = static_cast<float>(std::sqrt(sumSq / static_cast<double>(N))) * 1000.0f;
583 m_pivotResidualMm = rms;
584
585 mne::Log::info("Pivot calibration: {} samples, cond = {} , offset = {} {} {} mm, RMS residual = {} mm", N, cond, offset.x() * 1000.0f, offset.y() * 1000.0f, offset.z() * 1000.0f, rms);
586
587 m_penTipOffset = offset;
588 m_pivotState = PivotState::Done;
589 notify(m_listener.pivot_state_changed, m_pivotState);
590 notify(m_listener.pivot_calibration_done, offset, rms);
591 return true;
592}

598bool PolhemusCoregistration::capture_optical_calib_sample()
599{
600 if (!m_havePenPos) {
601 mne::Log::warning("Optical calibration: no pen position available");
602 return false;
603 }
604
605 const Eigen::Matrix4f& dev = m_deviceToWorld;
606 if (isIdentity(dev)) {
607 mne::Log::warning("Optical calibration: no tracker data available");
608 return false;
609 }
610
611 // Extract tracker position and orientation from device_to_world.
612 // device_to_world = trackerToWorld * offset, but we want the raw tracker
613 // pose (before offset). Reconstruct from the current raw tracker data
614 // by removing the offset: trackerToWorld = device_to_world * offset^-1
615 Eigen::Vector3f tracker_pos;
616 Eigen::Quaternionf tracker_ori;
617 trackerPose(dev, m_offsetTranslation, m_offsetRotation, tracker_pos, tracker_ori);
618
619 OpticalCalibSample sample;
620 sample.tracker_pos = tracker_pos;
621 sample.tracker_ori = tracker_ori;
622 sample.focus_point = m_penPosition;
623 m_opticalCalibSamples.push_back(sample);
624
625 const int n = static_cast<int>(m_opticalCalibSamples.size());
626 const Eigen::Vector3f localFocus = tracker_ori.inverse() * (m_penPosition - tracker_pos);
627 const float distToTracker = localFocus.norm() * 1000.0f;
628
629 mne::Log::info("Optical calibration sample {}:\n tracker pos: ({}, {}, {}) mm\n focus point: ({}, {}, {}) mm\n focus (local): ({}, {}, {}) mm\n distance tracker\u2194focus: {} mm", n, tracker_pos.x() * 1000.f, tracker_pos.y() * 1000.f, tracker_pos.z() * 1000.f, m_penPosition.x() * 1000.f, m_penPosition.y() * 1000.f, m_penPosition.z() * 1000.f, localFocus.x() * 1000.f, localFocus.y() * 1000.f, localFocus.z() * 1000.f, distToTracker);
630
631 // Log convergence angle between tracker→focus directions for successive samples.
632 // As the OPMI moves farther away, this angle shrinks (lines become parallel);
633 // at close range the angle is large because the ~20 cm tracker offset dominates.
634 if (n >= 2) {
635 const auto& prev = m_opticalCalibSamples[static_cast<size_t>(n - 2)];
636 const Eigen::Vector3f prevLocal = prev.tracker_ori.inverse() * (prev.focus_point - prev.tracker_pos);
637 const Eigen::Vector3f curDir = localFocus.normalized();
638 const Eigen::Vector3f prevDir = prevLocal.normalized();
639 const float dot = std::clamp(curDir.dot(prevDir), -1.0f, 1.0f);
640 const float angleDeg = std::acos(dot) * (180.0f / 3.14159265f);
641 const float prevDist = prevLocal.norm() * 1000.0f;
642 mne::Log::info(" convergence angle (sample {}\u2194{}): {}\u00b0 (distances: {} / {} mm)", n - 1, n, angleDeg, prevDist, distToTracker);
643 }
644
645 return true;
646}

693bool PolhemusCoregistration::solve_optical_calibration()
694{
695 const int N = static_cast<int>(m_opticalCalibSamples.size());
696 if (N < 2) {
697 mne::Log::warning("Optical calibration: need at least 2 samples, have {}", N);
698 return false;
699 }
700
701 // Step 1: Transform all focus points into the tracker's local frame
702 std::vector<Eigen::Vector3d> localPoints(static_cast<size_t>(N));
703 for (int i = 0; i < N; ++i) {
704 const auto& s = m_opticalCalibSamples[static_cast<size_t>(i)];
705 const Eigen::Vector3f local = s.tracker_ori.inverse() * (s.focus_point - s.tracker_pos);
706 localPoints[static_cast<size_t>(i)] = local.cast<double>();
707 }
708
709 // Step 2: PCA initial estimate — centroid + SVD for starting axis & center
710 Eigen::Vector3d centroid = Eigen::Vector3d::Zero();
711 for (const auto& p : localPoints)
712 centroid += p;
713 centroid /= static_cast<double>(N);
714
715 Eigen::MatrixXd centered(3, N);
716 for (int i = 0; i < N; ++i) {
717 centered.col(i) = localPoints[static_cast<size_t>(i)] - centroid;
718 }
719
720 Eigen::JacobiSVD<Eigen::MatrixXd> svd(centered, Eigen::ComputeThinU);
721 Eigen::Vector3d axisDir = svd.matrixU().col(0);
722 if (centroid.dot(axisDir) < 0.0)
723 axisDir = -axisDir;
724
725 const double t0 = -centroid.dot(axisDir);
726 Eigen::Vector3d opticalCenter = centroid + t0 * axisDir;
727
728 // Step 3: Constrained refinement.
729 //
730 // Priority: (a) directly captured objective center, (b) known distance
731 // constraint, (c) unconstrained PCA fallback.
732 //
733 // (a) If the user touched the pen to the objective lens, we know the
734 // optical center exactly in tracker-local frame. Only the axis
735 // direction needs to be determined from the focus samples.
736 // (b) If only the distance is known, optimise O on the sphere |O| = R.
737 // (c) Otherwise fall back to unconstrained PCA.
738 const double knownDist = static_cast<double>(m_knownTrackerToObjectiveDist);
739 if (m_hasObjectiveCenter) {
740 // Direct objective center — use it as-is, recompute axis from it
741 opticalCenter = m_objectiveCenterLocal.cast<double>();
742
743 Eigen::MatrixXd rays(3, N);
744 for (int i = 0; i < N; ++i)
745 rays.col(i) = localPoints[static_cast<size_t>(i)] - opticalCenter;
746
747 Eigen::JacobiSVD<Eigen::MatrixXd> raySvd(rays, Eigen::ComputeThinU);
748 axisDir = raySvd.matrixU().col(0);
749 // Axis must point from objective toward focus points
750 Eigen::Vector3d meanRay = Eigen::Vector3d::Zero();
751 for (int i = 0; i < N; ++i)
752 meanRay += rays.col(i);
753 if (meanRay.dot(axisDir) < 0.0)
754 axisDir = -axisDir;
755
756 mne::Log::info("Optical calibration: using directly captured objective center, |O|={} mm", opticalCenter.norm() * 1000.0);
757 } else if (knownDist > 0.0 && N >= 2) {
758 // Start where the PCA axis meets the sphere |O| = R on the side the axis points away from the
759 // tracker; the descent below only takes small steps, so it cannot travel from a distant start.
760 double R = knownDist;
761 Eigen::Vector3d O = opticalCenter;
762 const double along = centroid.dot(axisDir);
763 const double disc = along * along - centroid.squaredNorm() + R * R;
764 if (disc >= 0.0) {
765 O = centroid + (-along + std::sqrt(disc)) * axisDir;
766 } else if (O.norm() > 1e-9) {
767 O = O * (R / O.norm());
768 } else {
769 O = centroid.normalized() * R;
770 }
771
772 // Gauss-Newton iterations: optimise O on the sphere, recompute axis each step
773 for (int iter = 0; iter < 30; ++iter) {
774 // Recompute axis direction from O: SVD of (F_i - O)
775 Eigen::MatrixXd rays(3, N);
776 for (int i = 0; i < N; ++i)
777 rays.col(i) = localPoints[static_cast<size_t>(i)] - O;
778
779 Eigen::JacobiSVD<Eigen::MatrixXd> raySvd(rays, Eigen::ComputeThinU);
780 Eigen::Vector3d d = raySvd.matrixU().col(0);
781 if (O.dot(d) < 0.0)
782 d = -d; // axis points away from tracker
783
784 // Compute gradient of cost w.r.t. O (on the tangent plane of the sphere)
785 // Cost: E = sum_i |(F_i-O) - ((F_i-O).d)d|^2
786 // dE/dO = sum_i -2 * perp_i where perp_i = (F_i-O) - ((F_i-O).d)d
787 // but we project gradient onto tangent plane of sphere at O
788 Eigen::Vector3d grad = Eigen::Vector3d::Zero();
789 double cost = 0.0;
790 for (int i = 0; i < N; ++i) {
791 Eigen::Vector3d v = localPoints[static_cast<size_t>(i)] - O;
792 Eigen::Vector3d perp = v - v.dot(d) * d;
793 cost += perp.squaredNorm();
794 grad -= 2.0 * perp;
795 }
796
797 // Project gradient onto tangent plane of sphere at O
798 Eigen::Vector3d normal = O.normalized();
799 Eigen::Vector3d tangentGrad = grad - grad.dot(normal) * normal;
800
801 double gradNorm = tangentGrad.norm();
802 if (gradNorm < 1e-12)
803 break;
804
805 // Line search with backtracking
806 double step = 0.01 * R / gradNorm; // conservative initial step
807 for (int ls = 0; ls < 10; ++ls) {
808 Eigen::Vector3d candidate = O - step * tangentGrad;
809 // Project back onto sphere
810 candidate = candidate.normalized() * R;
811
812 // Recompute cost at candidate
813 Eigen::MatrixXd cRays(3, N);
814 for (int i = 0; i < N; ++i)
815 cRays.col(i) = localPoints[static_cast<size_t>(i)] - candidate;
816 Eigen::JacobiSVD<Eigen::MatrixXd> cSvd(cRays, Eigen::ComputeThinU);
817 Eigen::Vector3d cd = cSvd.matrixU().col(0);
818
819 double cCost = 0.0;
820 for (int i = 0; i < N; ++i) {
821 Eigen::Vector3d v = localPoints[static_cast<size_t>(i)] - candidate;
822 Eigen::Vector3d perp = v - v.dot(cd) * cd;
823 cCost += perp.squaredNorm();
824 }
825
826 if (cCost < cost) {
827 O = candidate;
828 break;
829 }
830 step *= 0.5;
831 }
832 }
833
834 opticalCenter = O;
835
836 // Final axis from refined center
837 Eigen::MatrixXd finalRays(3, N);
838 for (int i = 0; i < N; ++i)
839 finalRays.col(i) = localPoints[static_cast<size_t>(i)] - opticalCenter;
840 Eigen::JacobiSVD<Eigen::MatrixXd> finalSvd(finalRays, Eigen::ComputeThinU);
841 axisDir = finalSvd.matrixU().col(0);
842 if (opticalCenter.dot(axisDir) < 0.0)
843 axisDir = -axisDir;
844
845 mne::Log::info("Optical calibration: constrained refinement (R={} mm) converged, |O|={} mm", knownDist * 1000.0, opticalCenter.norm() * 1000.0);
846 }
847
848 // Step 4: Compute RMS residual (perpendicular distance from each point to the ray from O)
849 double sumSq = 0.0;
850 for (const auto& p : localPoints) {
851 const Eigen::Vector3d diff = p - opticalCenter;
852 const double along = diff.dot(axisDir);
853 const double perpSq = (diff - along * axisDir).squaredNorm();
854 sumSq += perpSq;
855 }
856 const double rms = std::sqrt(sumSq / static_cast<double>(N)) * 1000.0; // mm
857
858 // Step 5: Depth spread along the optical axis from the refined center
859 double minDepth = 1e30, maxDepth = -1e30;
860 for (const auto& p : localPoints) {
861 const double depth = (p - opticalCenter).dot(axisDir);
862 minDepth = std::min(minDepth, depth);
863 maxDepth = std::max(maxDepth, depth);
864 }
865 const double depthSpreadMm = (maxDepth - minDepth) * 1000.0;
866
867 m_opticalAxisLocal = axisDir.cast<float>();
868 m_opticalCenterLocal = opticalCenter.cast<float>();
869 m_opticalCalibResidualMm = static_cast<float>(rms);
870 m_opticalCalibDepthSpreadMm = static_cast<float>(depthSpreadMm);
871 m_opticalCalibValid = true;
872
873 const float distMm = m_opticalCenterLocal.norm() * 1000.0f;
874
875 const std::string constraint = knownDist > 0.0 ? std::format(", constrained R={:.1f} mm", knownDist * 1000.0) : std::string();
876 mne::Log::info("Optical calibration solved ({} samples{}):\n"
877 " axis (local): ({:.4f}, {:.4f}, {:.4f})\n"
878 " center (local): ({:.1f}, {:.1f}, {:.1f}) mm\n"
879 " tracker\u2194center: {:.1f} mm\n"
880 " RMS residual: {:.2f} mm\n"
881 " depth spread: {:.1f} mm (range {:.0f} .. {:.0f} mm)",
882 N, constraint,
883 m_opticalAxisLocal.x(), m_opticalAxisLocal.y(), m_opticalAxisLocal.z(),
884 m_opticalCenterLocal.x() * 1000.f, m_opticalCenterLocal.y() * 1000.f, m_opticalCenterLocal.z() * 1000.f,
885 distMm, rms, depthSpreadMm, minDepth * 1000.0, maxDepth * 1000.0);
886
887 // Per-sample: report convergence angle between tracker→focus direction
888 // and fitted optical axis. This angle shrinks as the OPMI moves farther
889 // from the head (lines become parallel) and grows at close range where
890 // the tracker offset dominates.
891 mne::Log::info(" Per-sample convergence angles (tracker\u2192focus vs optical axis):");
892 for (int i = 0; i < N; ++i) {
893 const Eigen::Vector3d& p = localPoints[static_cast<size_t>(i)];
894 const Eigen::Vector3d trkToFocus = p.normalized();
895 const double cosAngle = std::clamp(trkToFocus.dot(axisDir), -1.0, 1.0);
896 const double angleDeg = std::acos(cosAngle) * (180.0 / 3.14159265358979);
897 const double depth = (p - opticalCenter).dot(axisDir);
898 const Eigen::Vector3d diff = p - opticalCenter;
899 const double perpDist = (diff - diff.dot(axisDir) * axisDir).norm() * 1000.0;
900 mne::Log::info(" sample {}: depth={} mm, convergence angle={}\u00b0, perp err={} mm", (i + 1), depth * 1000.0, angleDeg, perpDist);
901 }
902
903 if (depthSpreadMm < 50.0) {
904 mne::Log::warning("Optical calibration: depth spread is only {} mm. For a reliable axis direction, move the OPMI to vary the focal distance by at least 50 mm between samples.", depthSpreadMm);
905 }
906
907 if (rms > 10.0) {
908 mne::Log::warning("Optical calibration: RMS residual {} mm is large. Check that the stylus accurately marks the microscope focus point.", rms);
909 }
910
911 // Plausibility check: when not using constrained mode, verify the
912 // tracker-to-optical-center distance is in a plausible range.
913 if (knownDist <= 0.0) {
914 constexpr float kMinPlausibleMm = 100.0f;
915 constexpr float kMaxPlausibleMm = 500.0f;
916 if (distMm < kMinPlausibleMm || distMm > kMaxPlausibleMm) {
917 mne::Log::warning("Optical calibration: tracker\u2194axis distance {} mm is outside the plausible range [{}, {}] mm. Calibration data may be unreliable.", distMm, kMinPlausibleMm, kMaxPlausibleMm);
918 }
919 }
920
921 notify(m_listener.optical_calibration_changed);
922 return true;
923}

969bool PolhemusCoregistration::apply_optical_axis_fine_adjust(const Eigen::Vector3f& targetWorldPos,
970 float& correctionDeg)
971{
972 correctionDeg = 0.0f;
973 if (!m_opticalCalibValid)
974 return false;
975
976 // Get tracker-to-world transform (same as optical_ray_in_world)
977 const Eigen::Matrix4f& dev = m_deviceToWorld;
978 if (isIdentity(dev))
979 return false;
980
981 Eigen::Vector3f tracker_pos;
982 Eigen::Quaternionf tracker_ori;
983 trackerPose(dev, m_offsetTranslation, m_offsetRotation, tracker_pos, tracker_ori);
984
985 // Current optical center in world frame
986 const Eigen::Vector3f optCenterWorld = tracker_pos + tracker_ori * m_opticalCenterLocal;
987
988 // Desired axis direction: from optical center to the target point
989 const Eigen::Vector3f toTarget = (targetWorldPos - optCenterWorld);
990 if (toTarget.norm() < 0.001f)
991 return false; // target too close to optical center
992
993 const Eigen::Vector3f desiredDirWorld = toTarget.normalized();
994
995 // Current axis direction in world frame
996 const Eigen::Vector3f currentDirWorld = (tracker_ori * m_opticalAxisLocal).normalized();
997
998 // Compute correction angle
999 const float cosAngle = std::clamp(currentDirWorld.dot(desiredDirWorld), -1.0f, 1.0f);
1000 correctionDeg = std::acos(cosAngle) * (180.0f / 3.14159265358979f);
1001
1002 // Sanity: reject corrections > 10° — likely a user error
1003 if (correctionDeg > 10.0f) {
1004 mne::Log::warning("Optical fine adjust: correction {}° exceeds 10° limit — rejected.", correctionDeg);
1005 return false;
1006 }
1007
1008 // Save pre-adjustment axis for undo
1009 if (!m_opticalFineAdjustApplied)
1010 m_opticalAxisPreFineAdjust = m_opticalAxisLocal;
1011
1012 // Compute the rotation from current to desired direction, in world frame
1013 const Eigen::Quaternionf worldCorrection = Eigen::Quaternionf::FromTwoVectors(currentDirWorld, desiredDirWorld);
1014
1015 // Transform the correction into tracker-local frame:
1016 // localCorrection = tracker_ori⁻¹ * worldCorrection * tracker_ori
1017 const Eigen::Quaternionf trackerOriInv = tracker_ori.inverse();
1018 const Eigen::Quaternionf localCorrection = trackerOriInv * worldCorrection * tracker_ori;
1019
1020 // Apply to the local axis
1021 m_opticalAxisLocal = (localCorrection * m_opticalAxisLocal).normalized();
1022 m_opticalFineAdjustDeg = correctionDeg;
1023 m_opticalFineAdjustApplied = true;
1024
1025 mne::Log::info("Optical fine adjust applied: {}°\n axis (local): ({}, {}, {})", correctionDeg, m_opticalAxisLocal.x(), m_opticalAxisLocal.y(), m_opticalAxisLocal.z());
1026
1027 notify(m_listener.optical_calibration_changed);
1028 return true;
1029}
