74 std::size_t estimate(const JsonValue& value, std::size_t depth) const
75 {
76 const std::size_t indent = m_indented ? (depth + 1) * kIndent + 1 : 0;
77 if (value.is_string()) {
78 return value.to_string().size() + 2;
79 }
80 if (value.is_array()) {
81 std::size_t size = 2 + indent;
82 for (const JsonValue& element : value.to_array()) {
83 size += 1 + indent + estimate(element, depth + 1);
84 }
85 return size;
86 }
87 if (value.is_object()) {
88 std::size_t size = 2 + indent;
89 for (const auto& [key, member] : value.to_object()) {
90 size += key.size() + 5 + indent + estimate(member, depth + 1);
91 }
92 return size;
93 }
94 return 24; // the longest number, true, false and null
95 }

193 void number(double value)
194 {
195 if (!std::isfinite(value)) {
196 m_out += "null";
197 return;
198 }
199 if (std::abs(value) <= kMaxExactInteger && std::trunc(value) == value) {
200 integer(static_cast<std::int64_t>(value));
201 return;
202 }
203 // The shortest round-trip digits, written in plain or exponent notation, whichever is shorter (plain on a tie).
204 // std::to_chars without a format would print every digit of large numbers in plain notation, Qt pads zeros.
205 char scientific[32];
206 const auto result = std::to_chars(scientific, scientific + sizeof(scientific), value, std::chars_format::scientific);
207 const std::string_view text(scientific, static_cast<std::size_t>(result.ptr - scientific));
208 const std::size_t e = text.find('e');
209 const bool negative = text.front() == '-';
210 char digits[24];
211 std::size_t digitCount = 0;
212 for (const char c : text.substr(negative ? 1 : 0, e - (negative ? 1 : 0))) {
213 if (c != '.') {
214 digits[digitCount++] = c;
215 }
216 }
217 int exponent = 0;
218 const std::string_view exponentText = text.substr(e + (text[e + 1] == '+' ? 2 : 1));
219 std::from_chars(exponentText.data(), exponentText.data() + exponentText.size(), exponent);
220
221 const std::size_t plainSize = (negative ? 1 : 0) + (exponent < 0 ? 2 + static_cast<std::size_t>(-exponent - 1) + digitCount : std::max(static_cast<std::size_t>(exponent) + 1, digitCount) + (digitCount > static_cast<std::size_t>(exponent) + 1 ? 1 : 0));
222 if (plainSize > text.size()) {
223 m_out += text;
224 return;
225 }
226 char plain[32]; // plainSize <= text.size() < 32
227 std::size_t size = 0;
228 if (negative) {
229 plain[size++] = '-';
230 }
231 if (exponent < 0) {
232 plain[size++] = '0';
233 plain[size++] = '.';
234 for (int i = -1; i > exponent; --i) {
235 plain[size++] = '0';
236 }
237 for (std::size_t i = 0; i < digitCount; ++i) {
238 plain[size++] = digits[i];
239 }
240 } else {
241 const std::size_t integerDigits = static_cast<std::size_t>(exponent) + 1;
242 for (std::size_t i = 0; i < std::max(integerDigits, digitCount); ++i) {
243 if (i == integerDigits) {
244 plain[size++] = '.';
245 }
246 plain[size++] = i < digitCount ? digits[i] : '0';
247 }
248 }
249 m_out.append(plain, size);
250 }

253 void string(std::string_view text)
254 {
255 m_out += '"';
256 std::size_t run = 0;
257 for (std::size_t i = 0; i < text.size(); ++i) {
258 const unsigned char c = static_cast<unsigned char>(text[i]);
259 if (c >= 0x20 && c != '"' && c != '\\') {
260 continue;
261 }
262 m_out.append(text.data() + run, i - run);
263 run = i + 1;
264 if (c == '"') {
265 m_out += "\\\"";
266 } else if (c == '\\') {
267 m_out += "\\\\";
268 } else if (!kControlEscapes[c].empty()) {
269 m_out += kControlEscapes[c];
270 } else {
271 constexpr char kHex[] = "0123456789abcdef";
272 const char escape[6] = {'\\', 'u', '0', '0', kHex[c >> 4], kHex[c & 0xF]};
273 m_out.append(escape, sizeof(escape));
274 }
275 }
276 m_out.append(text.data() + run, text.size() - run);
277 m_out += '"';
278 }

633void JsonObject::sort_members()
634{
635 const auto byKey = [](const Member& a, const Member& b) {
636 return a.m_key < b.m_key;
637 };
638 if (std::is_sorted(m_members.begin(), m_members.end(), byKey) && std::adjacent_find(m_members.begin(), m_members.end(), [](const Member& a, const Member& b) { return a.m_key == b.m_key; }) == m_members.end()) {
639 return;
640 }
641 if (m_members.size() <= 32) {
642 // Insertion sort: stable and allocation-free, and fastest for the small objects that dominate real data.
643 for (auto it = std::next(m_members.begin()); it != m_members.end(); ++it) {
644 for (auto hole = it; hole != m_members.begin() && byKey(*hole, *std::prev(hole)); --hole) {
645 std::iter_swap(hole, std::prev(hole));
646 }
647 }
648 } else {
649 std::stable_sort(m_members.begin(), m_members.end(), byKey);
650 }
651 // Of equal keys the stable sort keeps input order, so the last one is the one to keep, as in Qt.
652 auto out = m_members.begin();
653 for (auto it = m_members.begin(); it != m_members.end(); ++it) {
654 const auto next = std::next(it);
655 if (next != m_members.end() && next->m_key == it->m_key) {
656 continue;
657 }
658 if (out != it) {
659 *out = std::move(*it);
660 }
661 ++out;
662 }
663 m_members.erase(out, m_members.end());
664}

668JsonDocument JsonDocument::from_json(std::string_view json, JsonParseError* error)
669{
670 constexpr std::string_view kByteOrderMark = "\xEF\xBB\xBF";
671 const std::size_t skipped = json.starts_with(kByteOrderMark) ? kByteOrderMark.size() : 0;
672
673 JsonStreamParser::Options options;
674 options.maxDepth = 1024;
675 options.maxBufferSize = std::numeric_limits<std::size_t>::max();
676 options.maxDocumentSize = std::numeric_limits<std::size_t>::max();
677 options.singleDocument = true;
678 JsonStreamParser parser(options);
679 parser.feed(json.substr(skipped));
680 parser.finish();
681
682 JsonParseError result = parser.error();
683 JsonDocument document;
684 if (result.error == JsonParseError::NoError) {
685 if (parser.has_document()) {
686 document = parser.take_document();
687 } else {
688 result = {json.size() - skipped, JsonParseError::IllegalValue}; // nothing but whitespace
689 }
690 }
691 if (error) {
692 *error = result;
693 if (result.error != JsonParseError::NoError) {
694 error->offset += skipped;
695 }
696 }
697 return document;
698}
