498inline qint32 FiffCoordTrans::storageSize()
499{
500 // On-disk layout: from, to, rot[3][3], move[3], invrot[3][3], invmove[3]
501 // (C++ class uses Matrix4f but on-disk stores 3x3 rot + 3-vec separately)
502 return 2 * sizeof(fiff_int_t) + 2 * 12 * sizeof(fiff_float_t);
503}

507inline bool operator== (const FiffCoordTrans &a, const FiffCoordTrans &b)
508{
509 return (a.from == b.from &&
510 a.to == b.to &&
511 a.trans.isApprox(b.trans, 0.0001f) &&
512 a.invtrans.isApprox(b.invtrans, 0.0001f));
