68 AcqExpected<std::string> command(std::string_view line, std::chrono::milliseconds /*timeout*/) override
69 {
70 const std::lock_guard lock(m_mutex); // commands come from another thread than the reader
71 const std::size_t space = line.find(' ');
72 const std::string_view name = line.substr(0, space);
73 const std::string_view argument = space == std::string_view::npos ? std::string_view() : line.substr(space + 1);
74 std::int64_t value = -1;
75 std::from_chars(argument.data(), argument.data() + argument.size(), value);
76 if (name == "bufsize") {
77 if (value < 1 || m_source) {
78 return std::string("Buffer size not set\r\n");
79 }
80 m_buffer = static_cast<std::size_t>(value);
81 return std::format("Buffer size set to {}\r\n", value);
82 }
83 if (name == "selcon") {
84 return std::string(" * (1) Fiff File Simulator\r\n");
85 }
86 if (name == "measinfo" && value == kId) {
87 if (AcqError e = load(); !e.ok()) {
88 return e;
89 }
90 m_queue.push_back(int_tag(fiff::kBlockStart, fiff::kBlockMeas));
91 m_queue.insert(m_queue.end(), m_meas_info.begin(), m_meas_info.end());
92 return std::format("Sending measurement info to FiffStreamClient {}\r\n", kId);
93 }
94 if (name == "start" && value == kId && !m_meas_info.empty()) {
95 m_queue.push_back(int_tag(fiff::kBlockStart, fiff::kBlockRawData));
96 m_streaming = true;
97 return std::format("FiffStreamClient {} starts.\r\n", kId);
98 }
99 if ((name == "stop" && value == kId) || name == "stop-all") {
100 if (m_streaming) {
101 m_queue.push_back(int_tag(fiff::kBlockEnd, fiff::kBlockRawData));
102 m_streaming = false;
103 }
104 return std::string("Stopped.\r\n");
105 }
106 return std::format("{} is no command\r\n", name);
107 }
108
109 AcqError send(std::int32_t command, std::string_view /*text*/) override
110 {
111 const std::lock_guard lock(m_mutex);
112 if (command == rt::kGetClientId) {
113 m_queue.push_back(int_tag(rt::kClientId, kId));
114 }
115 return {};
116 }
117
118 AcqExpected<const NeuromagTag*> next(std::chrono::milliseconds timeout) override
119 {
120 std::unique_lock lock(m_mutex);
121 if (!m_queue.empty()) {
122 m_tag = std::move(m_queue.front());
123 m_queue.pop_front();
124 return &m_tag;
125 }
126 if (!m_streaming) {
127 lock.unlock();
128 std::this_thread::sleep_for(timeout);
129 return AcqError(AcqErrorCode::Timeout);
130 }
131 auto tag = m_source->next(timeout);
132 if (!tag) {
133 return tag.error();
134 }
135 const NeuromagTag& t = **tag;
136 if (t.kind != fiff::kDataBuffer) {
137 return AcqError(AcqErrorCode::Timeout);
138 }
139 // The connector sends the buffers calibrated, as floats
140 const std::size_t count = t.data.size() / sizeof(std::int32_t);
141 m_tag = {fiff::kDataBuffer, fiff::kTypeFloat, std::vector<std::byte>(count * sizeof(float))};
142 for (std::size_t i = 0; i < count; ++i) {
143 std::int32_t raw = 0;
144 std::memcpy(&raw, t.data.data() + i * sizeof(raw), sizeof(raw));
145 const auto value = static_cast<float>(m_scales[i % m_scales.size()] * raw);
146 std::memcpy(m_tag.data.data() + i * sizeof(value), &value, sizeof(value));
147 }
148 return &m_tag;
149 }
150
151private:
153 AcqError load()
154 {
155 if (m_source) {
156 return {};
157 }
158 m_source = make_neuromag_simulator({.meg = m_simulation.meg,
159 .eeg = m_simulation.eeg,
160 .rate = m_simulation.rate,
161 .buffer = m_buffer,
162 .seed = m_simulation.seed,
163 .realtime = m_simulation.realtime,
164 .duration = m_simulation.duration,
165 .gain = 1.0});
166 if (AcqError e = m_source->start(); !e.ok()) {
167 return e;
168 }
169 std::vector<NeuromagTag> tags;
170 while (true) {
171 auto tag = m_source->next(std::chrono::milliseconds(1000));
172 if (!tag) {
173 return tag.error();
174 }
175 tags.push_back(**tag);
176 if (tags.back().kind == fiff::kBlockStart && tags.back().as_int() == fiff::kBlockRawData) {
177 break;
178 }
179 }
180 m_meas_info = block_of(tags, fiff::kBlockMeasInfo);
181 for (const NeuromagTag& tag : m_meas_info) {
182 if (const auto ch = tag.kind == fiff::kChInfo ? NeuromagChannel::parse(tag.data) : std::nullopt) {
183 m_scales.push_back(static_cast<double>(ch->cal) * static_cast<double>(ch->range));
184 }
185 }
186 return {};
187 }
188
189 RtServerSimulation m_simulation;
190 std::size_t m_buffer = kDefaultBuffer;
191 std::unique_ptr<NeuromagTransport> m_source;
192 std::vector<NeuromagTag> m_meas_info;
193 std::vector<double> m_scales; // calibration times range per channel
194 std::deque<NeuromagTag> m_queue;
195 NeuromagTag m_tag;
196 bool m_streaming = false;
197 std::mutex m_mutex;
198};
