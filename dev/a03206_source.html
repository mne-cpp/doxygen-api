62std::vector<std::vector<int>> FsLabelUtils::buildAdjacency(const MatrixX3i& tris, int nVerts)
63{
64 std::vector<std::vector<int>> adj(static_cast<std::size_t>(std::max(nVerts, 0)));
65 for (Index t = 0; t < tris.rows(); ++t) {
66 const int v[3] = {tris(t, 0), tris(t, 1), tris(t, 2)};
67 if (std::ranges::all_of(v, [nVerts](int x) { return x >= 0 && x < nVerts; })) {
68 for (int a = 0; a < 3; ++a) {
69 auto& neighbours = adj[static_cast<std::size_t>(v[a])];
70 neighbours.push_back(v[(a + 1) % 3]);
71 neighbours.push_back(v[(a + 2) % 3]);
72 }
73 }
74 }
75 for (auto& neighbours : adj) {
76 std::ranges::sort(neighbours);
77 neighbours.erase(std::unique(neighbours.begin(), neighbours.end()), neighbours.end());
78 }
79 return adj;
80}

84FsLabel FsLabelUtils::growLabel(const FsLabel& label,
85 const FsSurface& surface,
86 int nSteps)
87{
88 if (label.empty() || surface.empty() || nSteps <= 0)
89 return label;
90
91 const int nVerts = static_cast<int>(surface.rr().rows());
92 const auto adj = buildAdjacency(surface.tris(), nVerts);
93
94 // Breadth-first by rings; out-of-range seed vertices are kept but not expanded
95 std::vector<char> inLabel(static_cast<std::size_t>(nVerts), 0);
96 std::vector<int> all(label.vertices.data(), label.vertices.data() + label.vertices.size());
97 std::vector<int> current;
98 for (const int v : all) {
99 if (v >= 0 && v < nVerts && !inLabel[static_cast<std::size_t>(v)]) {
100 inLabel[static_cast<std::size_t>(v)] = 1;
101 current.push_back(v);
102 }
103 }
104 for (int step = 0; step < nSteps && !current.empty(); ++step) {
105 std::vector<int> frontier;
106 for (const int v : current) {
107 for (const int neighbour : adj[static_cast<std::size_t>(v)]) {
108 if (!inLabel[static_cast<std::size_t>(neighbour)]) {
109 inLabel[static_cast<std::size_t>(neighbour)] = 1;
110 frontier.push_back(neighbour);
111 }
112 }
113 }
114 all.insert(all.end(), frontier.begin(), frontier.end());
115 current = std::move(frontier);
116 }
117 std::ranges::sort(all);
118 all.erase(std::unique(all.begin(), all.end()), all.end());
119
120 FsLabel result = makeLabel(all, label.hemi, label.name + "_grown");
121 for (Index i = 0; i < result.vertices.size(); ++i) {
122 if (result.vertices[i] >= 0 && result.vertices[i] < nVerts)
123 result.pos.row(i) = surface.rr().row(result.vertices[i]);
124 }
125 return result;
126}

130FsLabel FsLabelUtils::growCircularLabel(const FsSurface& surface,
131 std::span<const int> seeds,
132 double extentMm,
133 int hemi,
134 const std::string& name)
135{
136 // Adapted from mne.label._verts_within_dist / _grow_labels (MNE-Python, BSD-3-Clause).
137 const MatrixX3f& rr = surface.rr();
138 const int nVerts = static_cast<int>(rr.rows());
139 const auto adj = buildAdjacency(surface.tris(), nVerts);
140
141 // NaN marks a vertex not reached yet; neighbours are visited in ascending order, as in mne's CSR graph
142 std::vector<double> dist(static_cast<std::size_t>(std::max(nVerts, 0)), std::numeric_limits<double>::quiet_NaN());
143 std::vector<int> reached;
144 std::vector<int> addedLast;
145 for (const int seed : seeds) {
146 if (seed >= 0 && seed < nVerts && std::isnan(dist[static_cast<std::size_t>(seed)])) {
147 dist[static_cast<std::size_t>(seed)] = 0.0;
148 addedLast.push_back(seed);
149 }
150 }
151 reached = addedLast;
152 while (!addedLast.empty()) {
153 std::vector<int> added;
154 for (const int i : addedLast) {
155 const double vDist = dist[static_cast<std::size_t>(i)];
156 for (const int j : adj[static_cast<std::size_t>(i)]) {
157 const double nDist = vDist + 1000.0 * (rr.row(j) - rr.row(i)).cast<double>().norm();
158 double& known = dist[static_cast<std::size_t>(j)];
159 if (!std::isnan(known)) {
160 known = std::min(known, nDist);
161 } else if (nDist <= extentMm) {
162 known = nDist;
163 added.push_back(j);
164 }
165 }
166 }
167 reached.insert(reached.end(), added.begin(), added.end());
168 addedLast = std::move(added);
169 }
170 std::ranges::sort(reached);
171
172 const char* suffix = hemi == 0 ? "lh" : "rh";
173 FsLabel label = makeLabel(reached, hemi, name.ends_with(suffix) ? name : std::format("{}-{}", name, suffix));
174 for (Index k = 0; k < label.vertices.size(); ++k) {
175 label.pos.row(k) = rr.row(label.vertices[k]);
176 label.values[k] = std::trunc(dist[static_cast<std::size_t>(label.vertices[k])]);
177 }
178 return label;
179}

183std::vector<FsLabel> FsLabelUtils::splitLabel(const FsLabel& label,
184 const FsSurface& surface)
185{
186 std::vector<FsLabel> components;
187 if (label.empty())
188 return components;
189
190 // Label row of each vertex: the parts keep the label's positions and values
191 std::unordered_map<int, Index> rowOf;
192 rowOf.reserve(static_cast<std::size_t>(label.vertices.size()));
193 for (Index i = 0; i < label.vertices.size(); ++i)
194 rowOf.insert_or_assign(label.vertices[i], i);
195
196 // Connected components over the surface edges; without a surface every vertex stands alone
197 const int nVerts = surface.empty() ? 0 : static_cast<int>(surface.rr().rows());
198 const auto adj = surface.empty() ? std::vector<std::vector<int>>() : buildAdjacency(surface.tris(), nVerts);
199 std::vector<char> visited(static_cast<std::size_t>(label.vertices.size()), 0);
200 std::vector<std::vector<int>> parts;
201 for (Index i = 0; i < label.vertices.size(); ++i) {
202 const int seed = label.vertices[i];
203 char& seen = visited[static_cast<std::size_t>(rowOf[seed])];
204 if (seen)
205 continue;
206 seen = 1;
207 std::vector<int> part{seed};
208 for (std::size_t head = 0; head < part.size(); ++head) {
209 const int v = part[head];
210 if (v < 0 || v >= nVerts)
211 continue;
212 for (const int neighbour : adj[static_cast<std::size_t>(v)]) {
213 const auto it = rowOf.find(neighbour);
214 if (it != rowOf.end() && !visited[static_cast<std::size_t>(it->second)]) {
215 visited[static_cast<std::size_t>(it->second)] = 1;
216 part.push_back(neighbour);
217 }
218 }
219 }
220 std::ranges::sort(part);
221 parts.push_back(std::move(part));
222 }
223
224 // As mne Label.split("contiguous"): largest part first, named <name>_div<i>[-lh|-rh]
225 std::ranges::stable_sort(parts, std::ranges::greater{}, &std::vector<int>::size);
226 const bool hemiSuffix = label.name.ends_with("lh") || label.name.ends_with("rh");
227 const std::string_view base = std::string_view(label.name).substr(0, hemiSuffix ? label.name.size() - 3 : label.name.size());
228 const std::string_view ext = hemiSuffix ? std::string_view(label.name).substr(label.name.size() - 3) : std::string_view();
229
230 components.reserve(parts.size());
231 for (std::size_t p = 0; p < parts.size(); ++p) {
232 FsLabel comp = makeLabel(parts[p], label.hemi, std::format("{}_div{}{}", base, p + 1, ext));
233 for (Index j = 0; j < comp.vertices.size(); ++j) {
234 const Index row = rowOf[comp.vertices[j]];
235 comp.pos.row(j) = label.pos.row(row);
236 comp.values[j] = row < label.values.size() ? label.values[row] : 1.0;
237 }
238 components.push_back(std::move(comp));
239 }
240
241 return components;
242}

246std::vector<FsLabel> FsLabelUtils::stcToLabel(const MatrixXd& stcData,
247 const VectorXi& vertices,
248 const FsSurface& surface,
249 double dThreshold,
250 int iHemi)
251{
252 if (stcData.size() == 0 || vertices.size() == 0 || surface.empty())
253 return {};
254
255 // Vertices whose maximum absolute value over time exceeds the threshold, with their first STC row
256 const VectorXd maxAbs = stcData.cwiseAbs().rowwise().maxCoeff();
257 std::unordered_map<int, Index> firstRow;
258 for (Index i = 0; i < vertices.size(); ++i) {
259 if (maxAbs[i] > dThreshold)
260 firstRow.try_emplace(vertices[i], i);
261 }
262 if (firstRow.empty())
263 return {};
264
265 std::vector<int> sortedVerts;
266 sortedVerts.reserve(firstRow.size());
267 for (const auto& entry : firstRow)
268 sortedVerts.push_back(entry.first);
269 std::ranges::sort(sortedVerts);
270
271 const int nSurfVerts = static_cast<int>(surface.rr().rows());
272 FsLabel fullLabel = makeLabel(sortedVerts, iHemi, "stc_label");
273 for (Index i = 0; i < fullLabel.vertices.size(); ++i) {
274 const int v = fullLabel.vertices[i];
275 if (v >= 0 && v < nSurfVerts)
276 fullLabel.pos.row(i) = surface.rr().row(v);
277 fullLabel.values[i] = maxAbs[firstRow[v]];
278 }
279
280 // Split into connected components
281 return splitLabel(fullLabel, surface);
282}

286MatrixXd FsLabelUtils::labelsToStc(std::span<const FsLabel> labels,
287 const VectorXi& stcVertices,
288 int nTimes)
289{
290 const Index nVerts = stcVertices.size();
291 MatrixXd mask = MatrixXd::Zero(nVerts, nTimes);
292
293 std::unordered_map<int, Index> vertToRow;
294 vertToRow.reserve(static_cast<std::size_t>(nVerts));
295 for (Index i = 0; i < nVerts; ++i)
296 vertToRow.insert_or_assign(stcVertices[i], i);
297
298 for (const FsLabel& label : labels) {
299 for (Index i = 0; i < label.vertices.size(); ++i) {
300 const auto it = vertToRow.find(label.vertices[i]);
301 if (it != vertToRow.end())
302 mask.row(it->second).setOnes();
303 }
304 }
305
306 return mask;
307}
