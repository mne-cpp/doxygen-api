80InvSourceEstimate UTILSLIB::simulateStc(
81 const VectorXi& activeVertices,
82 const VectorXi& allVertices,
83 const SimulateStcParams& params)
84{
85 const int nAll = static_cast<int>(allVertices.size());
86 const int nActive = static_cast<int>(activeVertices.size());
87 const float tstep = 1.0f / params.sfreq;
88 const int nTimes = static_cast<int>(params.duration * params.sfreq);
89
90 if (nTimes <= 0 || nAll <= 0) {
91 qWarning() << "[simulateStc] Invalid parameters.";
92 return InvSourceEstimate();
93 }
94
95 // Build mapping: active vertex -> row in allVertices
96 QList<int> activeRows;
97 for (int a = 0; a < nActive; ++a) {
98 bool found = false;
99 for (int v = 0; v < nAll; ++v) {
100 if (allVertices(v) == activeVertices(a)) {
101 activeRows.append(v);
102 found = true;
103 break;
104 }
105 }
106 if (!found) {
107 qWarning() << "[simulateStc] Active vertex" << activeVertices(a) << "not in allVertices.";
108 return InvSourceEstimate();
109 }
110 }
111
112 // Generate Gaussian-envelope waveforms for each active source
113 MatrixXd data = MatrixXd::Zero(nAll, nTimes);
114
115 std::mt19937 gen(params.seed);
116 std::uniform_real_distribution<double> timeDist(0.2, 0.8);
117
118 for (int a = 0; a < nActive; ++a) {
119 // Center the Gaussian at a random fraction of the duration
120 double centerFrac = timeDist(gen);
121 double centerSample = centerFrac * nTimes;
122 double sigma = nTimes * 0.1; // 10% of duration
123
124 for (int t = 0; t < nTimes; ++t) {
125 double exponent = -0.5 * std::pow((t - centerSample) / sigma, 2.0);
126 data(activeRows[a], t) = std::exp(exponent);
127 }
128 }
129
130 return InvSourceEstimate(data, allVertices, params.tmin, tstep);
131}

135InvSourceEstimate UTILSLIB::simulateStcFromWaveforms(
136 const MatrixXd& waveforms,
137 const VectorXi& activeVertices,
138 const VectorXi& allVertices,
139 float tmin,
140 float tstep)
141{
142 const int nAll = static_cast<int>(allVertices.size());
143 const int nActive = static_cast<int>(activeVertices.size());
144 const int nTimes = static_cast<int>(waveforms.cols());
145
146 if (waveforms.rows() != nActive) {
147 qWarning() << "[simulateStcFromWaveforms] waveforms.rows() != activeVertices.size()";
148 return InvSourceEstimate();
149 }
150
151 MatrixXd data = MatrixXd::Zero(nAll, nTimes);
152
153 for (int a = 0; a < nActive; ++a) {
154 for (int v = 0; v < nAll; ++v) {
155 if (allVertices(v) == activeVertices(a)) {
156 data.row(v) = waveforms.row(a);
157 break;
158 }
159 }
160 }
161
162 return InvSourceEstimate(data, allVertices, tmin, tstep);
163}

167FiffEvoked UTILSLIB::simulateEvoked(
168 const MNEForwardSolution& fwd,
169 const InvSourceEstimate& stc,
170 const FiffInfo& info,
171 const FiffCov& noiseCov,
172 int nave,
173 int seed)
174{
175 // First generate noiseless data
176 FiffEvoked evoked = simulateEvokedNoiseless(fwd, stc, info);
177
178 if (evoked.data.size() == 0) return evoked;
179
180 const int nChan = static_cast<int>(evoked.data.rows());
181 const int nTimes = static_cast<int>(evoked.data.cols());
182
183 // Add noise from covariance
184 if (noiseCov.data.size() > 0 && noiseCov.data.rows() == nChan) {
185 // Decompose noise covariance: Cov = V * D * V^T
186 // Noise samples: V * sqrt(D) * randn / sqrt(nave)
187 SelfAdjointEigenSolver<MatrixXd> solver(noiseCov.data);
188 VectorXd eigvals = solver.eigenvalues();
189 MatrixXd eigvecs = solver.eigenvectors();
190
191 // Clamp negative eigenvalues
192 for (int i = 0; i < eigvals.size(); ++i) {
193 if (eigvals(i) < 0) eigvals(i) = 0;
194 }
195
196 MatrixXd sqrtCov = eigvecs * eigvals.cwiseSqrt().asDiagonal();
197
198 std::mt19937 gen(seed);
199 std::normal_distribution<double> dist(0.0, 1.0);
200
201 MatrixXd noise(nChan, nTimes);
202 for (int i = 0; i < nChan; ++i) {
203 for (int j = 0; j < nTimes; ++j) {
204 noise(i, j) = dist(gen);
205 }
206 }
207
208 double scaleFactor = 1.0 / std::sqrt(static_cast<double>(nave));
209 evoked.data += sqrtCov * noise * scaleFactor;
210 }
211
212 evoked.nave = nave;
213 return evoked;
214}

218FiffEvoked UTILSLIB::simulateEvokedNoiseless(
219 const MNEForwardSolution& fwd,
220 const InvSourceEstimate& stc,
221 const FiffInfo& info)
222{
223 FiffEvoked evoked;
224
225 if (!fwd.sol || fwd.sol->data.size() == 0) {
226 qWarning() << "[simulateEvokedNoiseless] Forward solution has no gain matrix.";
227 return evoked;
228 }
229
230 if (stc.isEmpty()) {
231 qWarning() << "[simulateEvokedNoiseless] Source estimate is empty.";
232 return evoked;
233 }
234
235 MatrixXd G = fwd.sol->data; // (n_channels x n_dipoles)
236 const int nChan = static_cast<int>(G.rows());
237 const int nDipoles = static_cast<int>(G.cols());
238 const int nSrc = static_cast<int>(stc.data.rows());
239 const int nTimes = static_cast<int>(stc.data.cols());
240
241 if (nDipoles != nSrc) {
242 qWarning() << "[simulateEvokedNoiseless] Leadfield columns" << nDipoles
243 << "!= source estimate rows" << nSrc;
244 return evoked;
245 }
246
247 // Sensor data = G * stc.data
248 evoked.data = G * stc.data;
249
250 // Set times
251 evoked.times.resize(nTimes);
252 for (int t = 0; t < nTimes; ++t) {
253 evoked.times(t) = stc.tmin + t * stc.tstep;
254 }
255
256 evoked.first = 0;
257 evoked.last = nTimes - 1;
258 evoked.nave = 1;
259 evoked.comment = "Simulated";
260 evoked.info = info;
261
262 return evoked;
263}
