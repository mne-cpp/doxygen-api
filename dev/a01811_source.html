85std::vector<std::string_view> split_words(std::string_view text)
86{
87 std::vector<std::string_view> words;
88 while (true) {
89 const std::size_t begin = text.find_first_not_of(whitespace);
90 if (begin == std::string_view::npos) {
91 return words;
92 }
93 text.remove_prefix(begin);
94 const std::size_t end = text.find_first_of(whitespace);
95 words.push_back(text.substr(0, end));
96 text = (end == std::string_view::npos) ? std::string_view{} : text.substr(end);
97 }
98}

102bool parse_floats(std::string_view line, std::vector<float>& values)
103{
104 values.clear();
105 for (const std::string_view word : split_words(line)) {
106 float value = 0.0f;
107 const auto [ptr, ec] = std::from_chars(word.data(), word.data() + word.size(), value);
108 if (ec != std::errc{} || ptr != word.data() + word.size()) {
109 return false;
110 }
111 values.push_back(value);
112 }
113 return true;
114}

122InvEcdSet InvEcdSet::read_dipoles_dip(const std::filesystem::path& path)
123{
124 std::ifstream in(path);
125 if (!in) {
126 mne::Log::critical("Not able to read from: {}", path.string());
127 return {};
128 }
129
130 // begin end x y z Q Qx Qy Qz g/% come first; MNE-Python may append khi^2, free and confidence columns.
131 constexpr std::size_t n_fixed = 10;
132 std::ptrdiff_t khi2_column = -1;
133 std::ptrdiff_t free_column = -1;
134 bool data_seen = false;
135
136 InvEcdSet set;
137 std::string line;
138 std::vector<float> values;
139 while (std::getline(in, line)) {
140 const std::size_t first = line.find_first_not_of(whitespace);
141 if (first == std::string::npos) {
142 continue;
143 }
144 if (line[first] == '#' || line[first] == '%') {
145 if (const std::size_t gof = line.find("g/%"); !data_seen && gof != std::string::npos) {
146 const std::vector<std::string_view> extra = split_words(std::string_view(line).substr(gof + 3));
147 for (std::size_t k = 0; k < extra.size(); ++k) {
148 const auto column = static_cast<std::ptrdiff_t>(n_fixed + k);
149 if (extra[k] == "khi^2") {
150 khi2_column = column;
151 } else if (extra[k] == "free") {
152 free_column = column;
153 }
154 }
155 }
156 continue;
157 }
158 data_seen = true;
159 if (!parse_floats(line, values) || values.size() < n_fixed) {
160 mne::Log::critical("Malformed dipole line in {}: {}", path.string(), line);
161 return {};
162 }
163 InvEcd one;
164 one.valid = true;
165 one.time = values[0] / 1000.0f;
166 one.rd = Eigen::Vector3f(values[2], values[3], values[4]) / 1000.0f;
167 one.Q = Eigen::Vector3f(values[6], values[7], values[8]) / 1e9f;
168 one.good = values[9] / 100.0f;
169 const auto n_values = static_cast<std::ptrdiff_t>(values.size());
170 if (khi2_column >= 0 && khi2_column < n_values) {
171 one.khi2 = values[static_cast<std::size_t>(khi2_column)];
172 }
173 if (free_column >= 0 && free_column < n_values) {
174 one.nfree = static_cast<int>(values[static_cast<std::size_t>(free_column)]);
175 }
176 set.push_back(one);
177 }
178
179 mne::Log::info("Read {} dipoles in dip format from {}", set.size(), path.string());
180 return set;
181}

185bool InvEcdSet::save_dipoles_bdip(const std::filesystem::path& path) const
186{
187 if (path.empty() || this->size() == 0) {
188 return true;
189 }
190
191 std::ofstream out(path, std::ios::binary);
192 if (!out) {
193 mne::Log::critical("Not able to write to: {}", path.string());
194 return false;
195 }
196
197 int nsave = 0;
198 for (const InvEcd& one : m_dipoles) {
199 if (!one.valid) {
200 continue;
201 }
202 // Fields xfit fills from an error analysis stay zero.
203 bdipEcdRec one_out{};
204 one_out.dipole = mne::swap_int(1);
205 one_out.begin = mne::swap_float(one.time);
206 one_out.end = mne::swap_float(one.time);
207 for (int p = 0; p < 3; p++) {
208 one_out.r0[p] = mne::swap_float(0.0f);
209 one_out.rd[p] = mne::swap_float(one.rd[p]);
210 one_out.Q[p] = mne::swap_float(one.Q[p]);
211 }
212 one_out.goodness = mne::swap_float(one.good);
213 one_out.errors_computed = mne::swap_int(0);
214 one_out.khi2 = mne::swap_float(one.khi2);
215 out.write(reinterpret_cast<const char*>(&one_out), sizeof(bdipEcdRec));
216 nsave++;
217 }
218 if (!out.flush()) {
219 mne::Log::critical("Failed to write the dipoles to {}", path.string());
220 out.close();
221 std::error_code ec;
222 std::filesystem::remove(path, ec);
223 return false;
224 }
225 mne::Log::info("Save {} dipoles in bdip format to {}", nsave, path.string());
226 return true;
227}

231bool InvEcdSet::save_dipoles_dip(const std::filesystem::path& path) const
232{
233 if (path.empty() || this->size() == 0) {
234 return true;
235 }
236 std::ofstream out(path, std::ios::binary);
237 if (!out) {
238 mne::Log::critical("Not able to write to: {}", path.string());
239 return false;
240 }
241 // Column widths as in MNE-C save_dipoles_dip; other tools read this layout back.
242 out << "# CoordinateSystem \"Head\"\n";
243 out << std::format("# {:>7} {:>7} {:>8} {:>8} {:>8} {:>8} {:>8} {:>8} {:>8} {:>6}\n",
244 "begin", "end", "X (mm)", "Y (mm)", "Z (mm)", "Q(nAm)", "Qx(nAm)", "Qy(nAm)", "Qz(nAm)", "g/%");
245 int nsave = 0;
246 for (const InvEcd& one : m_dipoles) {
247 if (one.valid) {
248 const double time = 1000.0 * one.time;
249 out << std::format(" {:7.1f} {:7.1f} {:8.2f} {:8.2f} {:8.2f} {:8.3f} {:8.3f} {:8.3f} {:8.3f} {:6.1f}\n",
250 time, time,
251 1000.0 * one.rd[X], 1000.0 * one.rd[Y], 1000.0 * one.rd[Z],
252 1e9 * one.Q.norm(), 1e9 * one.Q[X], 1e9 * one.Q[Y], 1e9 * one.Q[Z], 100.0 * one.good);
253 nsave++;
254 }
255 }
256 out << "## Name \"ALL dipoles\" Style \"Dipoles\"\n";
257 if (!out.flush()) {
258 mne::Log::critical("Failed to write the dipoles to {}", path.string());
259 return false;
260 }
261 mne::Log::info("Save {} dipoles in dip format to {}", nsave, path.string());
262 return true;
263}
