775{
776 std::vector<std::pair<int, double>> triggerIndices;
777
778 if (m_bIsFreezed)
779 return m_qMapDetectedTriggerFreeze.contains(m_iCurrentTriggerChIndex) ? m_qMapDetectedTriggerFreeze.at(m_iCurrentTriggerChIndex) : triggerIndices;
780
781 if (m_bTriggerDetectionActive) {
782 return m_qMapDetectedTrigger.contains(m_iCurrentTriggerChIndex) ? m_qMapDetectedTrigger.at(m_iCurrentTriggerChIndex) : triggerIndices;
