44MatrixXd InvTfMxne::buildGaborDictionary(int iNSamples, int iNFreqs,
45 double dFMin, double dFMax,
46 double dSFreq)
47{
48 // Build a set of Gabor atoms: windowed complex exponentials at different frequencies
49 // Returns real-valued matrix: for each frequency, we store cos and sin rows
50 const int nAtoms = 2 * iNFreqs; // cos + sin per frequency
51 MatrixXd dict = MatrixXd::Zero(nAtoms, iNSamples);
52
53 VectorXd timeVec(iNSamples);
54 for (int t = 0; t < iNSamples; ++t) {
55 timeVec(t) = static_cast<double>(t) / dSFreq;
56 }
57
58 // Log-spaced frequencies between fMin and fMax
59 for (int f = 0; f < iNFreqs; ++f) {
60 double freq;
61 if (iNFreqs > 1) {
62 double logMin = std::log(dFMin);
63 double logMax = std::log(dFMax);
64 freq = std::exp(logMin + (logMax - logMin) * f / (iNFreqs - 1));
65 } else {
66 freq = (dFMin + dFMax) / 2.0;
67 }
68
69 // Gaussian window width: ~3 cycles at this frequency
70 double sigma = 3.0 / (2.0 * M_PI * freq);
71
72 double tCenter = timeVec(iNSamples / 2);
73
74 for (int t = 0; t < iNSamples; ++t) {
75 double dt = timeVec(t) - tCenter;
76 double envelope = std::exp(-0.5 * dt * dt / (sigma * sigma));
77 dict(2 * f, t) = envelope * std::cos(2.0 * M_PI * freq * timeVec(t));
78 dict(2 * f + 1, t) = envelope * std::sin(2.0 * M_PI * freq * timeVec(t));
79 }
80
81 // Normalize each atom to unit norm
82 double normCos = dict.row(2 * f).norm();
83 if (normCos > 1e-12) dict.row(2 * f) /= normCos;
84
85 double normSin = dict.row(2 * f + 1).norm();
86 if (normSin > 1e-12) dict.row(2 * f + 1) /= normSin;
87 }
88
89 return dict;
90}

94InvTfMxneResult InvTfMxne::compute(const MatrixXd& matGain,
95 const MatrixXd& matData,
96 const InvTfMxneParams& params)
97{
98 InvTfMxneResult result;
99
100 const int nChannels = matGain.rows();
101 const int nSources = matGain.cols();
102 const int nTimes = matData.cols();
103
104 if (nChannels == 0 || nSources == 0 || nTimes == 0) {
105 return result;
106 }
107 const int nAtoms = 2 * params.iNFreqs;
108
109 if (matData.rows() != nChannels) {
110 qWarning() << "[InvTfMxne::compute] Dimension mismatch: data rows" << matData.rows()
111 << "!= gain rows" << nChannels;
112 return result;
113 }
114
115 // Build Gabor dictionary: (nAtoms × nTimes)
116 MatrixXd Phi = buildGaborDictionary(nTimes, params.iNFreqs,
117 params.dFMin, params.dFMax,
118 params.dSFreq);
119
120 // TF coefficients: Z (nSources × nAtoms)
121 // The model is: M = G * X, where X = Z * Phi (each source has TF representation)
122 // Equivalently in expanded form: M = G_expanded * z_vec
123 // where G_expanded = G ⊗ Phi^T, z_vec = vec(Z)
124 // But we solve iteratively using Block Coordinate Descent.
125
126 // Initialize Z to zero
127 MatrixXd Z = MatrixXd::Zero(nSources, nAtoms);
128
129 // Precompute G^T * G diagonal for Lipschitz constants
130 VectorXd lipschitz(nSources);
131 for (int j = 0; j < nSources; ++j) {
132 lipschitz(j) = matGain.col(j).squaredNorm();
133 }
134 // Scale by dictionary energy
135 double phiEnergy = 0.0;
136 for (int a = 0; a < nAtoms; ++a) {
137 phiEnergy += Phi.row(a).squaredNorm();
138 }
139 lipschitz *= phiEnergy;
140
141 // Proximal gradient descent (ISTA-like)
142 MatrixXd residual = matData;
143 double prevObj = std::numeric_limits<double>::max();
144
145 for (int iter = 0; iter < params.iMaxIterations; ++iter) {
146 // Compute residual: R = M - G * (Z * Phi)
147 MatrixXd X = Z * Phi; // (nSources × nTimes)
148 residual = matData - matGain * X;
149
150 // Compute objective: ||R||^2_F + alpha_space * ||Z||_21 + alpha_time * ||Z||_1
151 double dataFit = residual.squaredNorm();
152 double l21Norm = 0.0;
153 double l1Norm = 0.0;
154 for (int j = 0; j < nSources; ++j) {
155 l21Norm += Z.row(j).norm();
156 l1Norm += Z.row(j).lpNorm<1>();
157 }
158 double objective = 0.5 * dataFit + params.dAlphaSpace * l21Norm + params.dAlphaTime * l1Norm;
159
160 // Check convergence
161 if (std::abs(prevObj - objective) / (std::abs(prevObj) + 1e-12) < params.dTolerance) {
162 result.nIterations = iter + 1;
163 break;
164 }
165 prevObj = objective;
166 result.nIterations = iter + 1;
167
168 // Gradient step + proximal operator for each source
169 // Gradient of data term w.r.t. Z_j: -G_j^T * R * Phi^T
170 MatrixXd GtR = matGain.transpose() * residual; // (nSources × nTimes)
171 MatrixXd grad = GtR * Phi.transpose(); // (nSources × nAtoms)
172
173 for (int j = 0; j < nSources; ++j) {
174 if (lipschitz(j) < 1e-12) continue;
175
176 double stepSize = 1.0 / lipschitz(j);
177
178 // Gradient step
179 RowVectorXd zNew = Z.row(j) + stepSize * grad.row(j);
180
181 // Proximal operator: soft-threshold (L1) then group-threshold (L21)
182 // L1 soft threshold
183 double threshL1 = params.dAlphaTime * stepSize;
184 for (int a = 0; a < nAtoms; ++a) {
185 double val = zNew(a);
186 double sign = (val > 0.0) ? 1.0 : -1.0;
187 zNew(a) = sign * std::max(0.0, std::abs(val) - threshL1);
188 }
189
190 // L21 group threshold
191 double groupNorm = zNew.norm();
192 double threshL21 = params.dAlphaSpace * stepSize;
193 if (groupNorm > threshL21) {
194 zNew *= (1.0 - threshL21 / groupNorm);
195 } else {
196 zNew.setZero();
197 }
198
199 Z.row(j) = zNew;
200 }
201 }
202
203 // Reconstruct time-domain source estimate from TF coefficients
204 MatrixXd X = Z * Phi; // (nSources × nTimes)
205
206 // Find active sources
207 QVector<int> activeVertices;
208 for (int j = 0; j < nSources; ++j) {
209 if (Z.row(j).norm() > 1e-12) {
210 activeVertices.append(j);
211 }
212 }
213
214 // Optional debiasing: re-estimate amplitudes on active set
215 MatrixXd finalX;
216 if (params.bDebias && !activeVertices.isEmpty()) {
217 MatrixXd Gactive(nChannels, activeVertices.size());
218 for (int i = 0; i < activeVertices.size(); ++i) {
219 Gactive.col(i) = matGain.col(activeVertices[i]);
220 }
221 // Least-squares on active set: X_active = pinv(G_active) * M
222 finalX = Gactive.bdcSvd<ComputeThinU | ComputeThinV>().solve(matData);
223 } else if (!activeVertices.isEmpty()) {
224 finalX = MatrixXd(activeVertices.size(), nTimes);
225 for (int i = 0; i < activeVertices.size(); ++i) {
226 finalX.row(i) = X.row(activeVertices[i]);
227 }
228 } else {
229 finalX = MatrixXd::Zero(0, nTimes);
230 }
231
232 // Build result
233 VectorXi vertices(activeVertices.size());
234 for (int i = 0; i < activeVertices.size(); ++i) {
235 vertices(i) = activeVertices[i];
236 }
237
238 result.stc = InvSourceEstimate(finalX, vertices, 0.0f,
239 static_cast<float>(1.0 / params.dSFreq));
240 result.stc.method = InvEstimateMethod::MixedNorm;
241 result.activeVertices = activeVertices;
242 result.residualNorm = residual.norm();
243
244 // Store TF coefficients for active sources
245 if (!activeVertices.isEmpty()) {
246 result.tfCoefficients = MatrixXd(activeVertices.size(), nAtoms);
247 for (int i = 0; i < activeVertices.size(); ++i) {
248 result.tfCoefficients.row(i) = Z.row(activeVertices[i]);
249 }
250 }
251
252 return result;
253}
