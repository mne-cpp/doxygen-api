122void TraceViewClient::install_effects()
123{
124 // Each entry first does what every host needs, then calls the application's callback
125 TraceView::Effects effects;
126 effects.scroll_changed = [this](double sample) {
127 update();
128 if (m_effects.scroll_changed) {
129 m_effects.scroll_changed(sample);
130 }
131 };
132 effects.zoom_changed = [this](double samples_per_pixel) {
133 // Another zoom decimates differently and moves the overlay's time labels
134 m_builder.geometry().invalidate();
135 m_builder.invalidate_overlay();
136 update();
137 if (m_effects.zoom_changed) {
138 m_effects.zoom_changed(samples_per_pixel);
139 }
140 };
141 effects.channel_offset_changed = [this](int first_channel) {
142 m_builder.geometry().invalidate();
143 update();
144 if (m_effects.channel_offset_changed) {
145 m_effects.channel_offset_changed(first_channel);
146 }
147 };
148 effects.animate_scroll = [this](const TraceView::ScrollAnimation& animation) {
149 m_animator.start_scroll(animation, now());
150 update();
151 };
152 effects.animate_zoom = [this](const TraceView::ZoomAnimation& animation) {
153 m_animator.start_zoom(animation, now());
154 update();
155 };
156 effects.stop_animations = [this] {
157 m_animator.stop_scroll();
158 };
159 effects.sample_clicked = [this](int sample) {
160 if (m_effects.sample_clicked) {
161 m_effects.sample_clicked(sample);
162 }
163 };
164 effects.range_selected = [this](int start, int end) {
165 if (m_effects.range_selected) {
166 m_effects.range_selected(start, end);
167 }
168 };
169 effects.annotation_boundary_moved = [this](int index, bool start, int sample) {
170 if (m_effects.annotation_boundary_moved) {
171 m_effects.annotation_boundary_moved(index, start, sample);
172 }
173 };
174 effects.annotations_changed = [this] {
175 m_builder.invalidate_overlay();
176 update();
177 if (m_effects.annotations_changed) {
178 m_effects.annotations_changed();
179 }
180 };
181 effects.channel_double_clicked = [this](int channel) {
182 if (m_effects.channel_double_clicked) {
183 m_effects.channel_double_clicked(channel);
184 }
185 };
186 effects.cursor_changed = [this](CursorShape shape) {
187 if (surface()) {
188 surface()->set_cursor(shape);
189 }
190 if (m_effects.cursor_changed) {
191 m_effects.cursor_changed(shape);
192 }
193 };
194 effects.cursor_data_changed = [this](const TraceView::CursorData& data) {
195 if (m_effects.cursor_data_changed) {
196 m_effects.cursor_data_changed(data);
197 }
198 };
199 effects.redraw = [this] {
200 update();
201 if (m_effects.redraw) {
202 m_effects.redraw();
203 }
204 };
205 m_view.set_effects(std::move(effects));
206}
