133std::expected<SerialPort, std::error_code> SerialPort::open(const std::string& name, int baud)
134{
135 if (baud <= 0) {
136 return std::unexpected(std::make_error_code(std::errc::invalid_argument));
137 }
138 const std::string path = name.starts_with("\\\\.\\") ? name : "\\\\.\\" + name; // COM10 and above need the prefix
139 HANDLE handle = CreateFileA(path.c_str(), GENERIC_READ | GENERIC_WRITE, 0, nullptr, OPEN_EXISTING, 0, nullptr);
140 if (handle == INVALID_HANDLE_VALUE) {
141 const DWORD error = GetLastError();
142 if (error == ERROR_ACCESS_DENIED) {
143 return std::unexpected(std::make_error_code(std::errc::device_or_resource_busy)); // ports are not shared
144 }
145 return std::unexpected(std::error_code(static_cast<int>(error), std::system_category()));
146 }
147 SerialPort port;
148 port.m_handle = handle;
149 DCB dcb{};
150 dcb.DCBlength = sizeof(dcb);
151 COMMTIMEOUTS timeouts{};
152 timeouts.ReadIntervalTimeout = MAXDWORD; // read() returns what is there
153 if (!GetCommState(handle, &dcb)) {
154 return std::unexpected(std::make_error_code(std::errc::inappropriate_io_control_operation));
155 }
156 dcb.BaudRate = static_cast<DWORD>(baud);
157 dcb.ByteSize = 8;
158 dcb.Parity = NOPARITY;
159 dcb.StopBits = ONESTOPBIT;
160 dcb.fBinary = TRUE;
161 dcb.fParity = FALSE;
162 dcb.fOutxCtsFlow = FALSE;
163 dcb.fOutxDsrFlow = FALSE;
164 dcb.fDtrControl = DTR_CONTROL_ENABLE;
165 dcb.fRtsControl = RTS_CONTROL_ENABLE;
166 dcb.fOutX = FALSE;
167 dcb.fInX = FALSE;
168 if (!SetCommState(handle, &dcb) || !SetCommTimeouts(handle, &timeouts)) {
169 return std::unexpected(std::make_error_code(std::errc::invalid_argument));
170 }
171 PurgeComm(handle, PURGE_RXCLEAR | PURGE_TXCLEAR);
172 return port;
173}

191std::expected<std::size_t, std::error_code> SerialPort::read(std::span<char> buffer, std::chrono::milliseconds timeout)
192{
193 const auto deadline = std::chrono::steady_clock::now() + timeout;
194 while (true) {
195 DWORD errors = 0;
196 COMSTAT status{};
197 if (!ClearCommError(m_handle, &errors, &status)) {
198 return std::unexpected(std::make_error_code(std::errc::no_such_device));
199 }
200 if (status.cbInQue > 0) {
201 DWORD got = 0;
202 const DWORD wanted = static_cast<DWORD>(std::min<std::size_t>(buffer.size(), status.cbInQue));
203 if (!ReadFile(m_handle, buffer.data(), wanted, &got, nullptr)) {
204 return std::unexpected(std::error_code(static_cast<int>(GetLastError()), std::system_category()));
205 }
206 return std::size_t{got};
207 }
208 if (std::chrono::steady_clock::now() >= deadline) {
209 return std::unexpected(std::make_error_code(std::errc::timed_out));
210 }
211 Sleep(1);
212 }
213}

217std::vector<std::string> mne::list_serial_ports()
218{
219 std::vector<std::string> ports;
220 std::array<char, 65536> devices{};
221 const DWORD length = QueryDosDeviceA(nullptr, devices.data(), static_cast<DWORD>(devices.size()));
222 for (const char* name = devices.data(); name < devices.data() + length && *name != '\0'; name += std::strlen(name) + 1) {
223 if (std::string_view(name).starts_with("COM")) {
224 ports.emplace_back(name);

227 std::ranges::sort(ports, [](const std::string& a, const std::string& b) { return a.size() != b.size() ? a.size() < b.size() : a < b; });

250std::expected<SerialPort, std::error_code> SerialPort::open(const std::string& name, int baud)
251{
252 const speed_t speed = speed_of(baud);
253 if (speed == B0) {
254 return std::unexpected(std::make_error_code(std::errc::invalid_argument));
255 }
256 SerialPort port;
257 port.m_fd = ::open(name.c_str(), O_RDWR | O_NOCTTY | O_NONBLOCK | O_CLOEXEC);
258 if (port.m_fd < 0) {
259 return std::unexpected(last_error());
260 }
261 termios tty{};
262 if (::tcgetattr(port.m_fd, &tty) != 0) {
263 return std::unexpected(std::make_error_code(std::errc::inappropriate_io_control_operation));
264 }
265 ::cfmakeraw(&tty);
266 tty.c_cflag |= CLOCAL | CREAD;
267 tty.c_cflag &= ~static_cast<tcflag_t>(CSTOPB | CRTSCTS);
268 tty.c_cc[VMIN] = 0;
269 tty.c_cc[VTIME] = 0;
270 if (::cfsetispeed(&tty, speed) != 0 || ::cfsetospeed(&tty, speed) != 0 || ::tcsetattr(port.m_fd, TCSANOW, &tty) != 0) {
271 return std::unexpected(last_error());
272 }
273 ::tcflush(port.m_fd, TCIOFLUSH);
274 return port;
275}

279std::error_code SerialPort::write(std::string_view data)
280{
281 while (!data.empty()) {
282 const ssize_t written = ::write(m_fd, data.data(), data.size());
283 if (written < 0) {
284 if (errno == EINTR) {
285 continue;
286 }
287 if (errno == EAGAIN) {
288 pollfd p{m_fd, POLLOUT, 0};
289 ::poll(&p, 1, 100);
290 continue;
291 }
292 return last_error();
293 }
294 data.remove_prefix(static_cast<std::size_t>(written));
295 }
296 return {};
297}

301std::expected<std::size_t, std::error_code> SerialPort::read(std::span<char> buffer, std::chrono::milliseconds timeout)
302{
303 const auto timed_out = std::unexpected(std::make_error_code(std::errc::timed_out));
304 const auto gone = std::unexpected(std::make_error_code(std::errc::no_such_device));
305 pollfd p{m_fd, POLLIN, 0};
306 const int ready = ::poll(&p, 1, static_cast<int>(timeout.count()));
307 if (ready < 0) {
308 return errno == EINTR ? timed_out : std::unexpected(last_error());
309 }
310 if (ready == 0) {
311 return timed_out;
312 }
313 if ((p.revents & (POLLERR | POLLNVAL)) != 0) {
314 return gone;
315 }
316 const ssize_t got = ::read(m_fd, buffer.data(), buffer.size());
317 if (got < 0) {
318 return errno == EAGAIN || errno == EINTR ? timed_out : std::unexpected(last_error());
319 }
320 if (got == 0) {
321 return gone;
322 }
323 return static_cast<std::size_t>(got);
324}

332 std::error_code error;

335#ifdef __APPLE__
336 const bool serial = name.starts_with("cu.") && name != "cu.Bluetooth-Incoming-Port";
337#else

340 if (serial) {
341 (rank(name) == 0 ? usb : other).push_back(entry.path().string());
342 }
343 }
344 std::ranges::sort(usb);
345 std::ranges::sort(other);
346 usb.insert(usb.end(), other.begin(), other.end());
347 return usb;
