45bool SensorStreamSession::startStreaming(const std::string& modality,
46 const SensorFieldMapper& fieldMapper,
47 const std::map<std::string, std::shared_ptr<BrainSurface>>& surfaces)
48{
49 if (m_isStreaming) {
50 mne::Log::debug("SensorStreamSession: Real-time sensor streaming already active");
51 return false;
52 }
53
54 // Require loaded evoked data and a built mapping matrix
55 if (!fieldMapper.isLoaded() || fieldMapper.evoked().empty()) {
56 mne::Log::warning("SensorStreamSession: Cannot start sensor streaming — no evoked data loaded");
57 return false;
58 }
59
60 // Select the mapping matrix and pick vector based on modality
61 std::shared_ptr<Eigen::MatrixXf> mappingMat;
62 Eigen::VectorXi pick;
63 std::string surfaceKey;
64
65 if (modality == "MEG") {
66 mappingMat = fieldMapper.megMapping();
67 pick = fieldMapper.megPick();
68 surfaceKey = fieldMapper.megSurfaceKey();
69 } else if (modality == "EEG") {
70 mappingMat = fieldMapper.eegMapping();
71 pick = fieldMapper.eegPick();
72 surfaceKey = fieldMapper.eegSurfaceKey();
73 } else {
74 mne::Log::warning("SensorStreamSession: Unknown modality: {}", modality);
75 return false;
76 }
77
78 if (!mappingMat || mappingMat->rows() == 0 || pick.size() == 0) {
79 mne::Log::warning("SensorStreamSession: Mapping matrix not built for {} — call loadSensorField() first", modality);
80 return false;
81 }
82
83 if (surfaceKey.empty() || !surfaces.contains(surfaceKey)) {
84 mne::Log::warning("SensorStreamSession: Target surface not found for {} streaming (key: {})", modality, surfaceKey);
85 return false;
86 }
87
88 m_modality = modality;
89
90 // Create controller on first use
91 if (!m_stream) {
92 m_stream = std::make_unique<RtSensorStream>();
93 m_stream->setNewSensorColorsAvailableCallback([this](const std::string& key, const std::vector<uint32_t>& colors) {
94 m_dispatcher([this, key, colors] {
95 if (m_colorsAvailableCallback) {
96 m_colorsAvailableCallback(key, colors);
97 }
98 });
99 });
100 }
101
102 // Propagate mapping matrix
103 m_stream->setMappingMatrix(surfaceKey, mappingMat);
104
105 // Propagate current visualization parameters
106 m_stream->setColormapType(fieldMapper.colormap());
107
108 // Compute sampling frequency from evoked data
109 double sFreq = 1000.0;
110 if (fieldMapper.evoked().times.size() > 1) {
111 double dt = fieldMapper.evoked().times(1) - fieldMapper.evoked().times(0);
112 if (dt > 0)
113 sFreq = 1.0 / dt;
114 }
115 m_stream->setSFreq(sFreq);
116
117 // Feed evoked time points as data into the queue
118 const int nTimePoints = static_cast<int>(fieldMapper.evoked().times.size());
119 mne::Log::debug("SensorStreamSession: Feeding {} {} sensor time points into real-time queue", nTimePoints, modality);
120 m_stream->clearData();
121
122 for (int t = 0; t < nTimePoints; ++t) {
123 Eigen::VectorXf meas(pick.size());
124 for (int i = 0; i < pick.size(); ++i) {
125 meas(i) = static_cast<float>(fieldMapper.evoked().data(pick(i), t));
126 }
127 m_stream->addData(meas);
128 }
129
130 // Start streaming
131 m_stream->setStreamingState(true);
132 m_isStreaming = true;
133
134 mne::Log::debug("SensorStreamSession: Real-time sensor streaming started for {}", modality);
135 return true;
136}
