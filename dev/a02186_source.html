244void Control3DView::onSceneColorPicker()
245{
246 QColorDialog dialog(m_colCurrentSceneColor, this);
247
248 //Update all connected View3D's scene colors
249 connect(&dialog, &QColorDialog::currentColorChanged,
250 this, &Control3DView::onSceneColorChanged);
251
252 if (dialog.exec() == QDialog::Accepted) {
253 m_colCurrentSceneColor = dialog.currentColor();
254 } else {
255 onSceneColorChanged(m_colCurrentSceneColor);
256 }
257
258 //Set color of button new new scene color
259 m_pUi->m_pushButton_sceneColorPicker->setStyleSheet(QString("background-color: rgb(%1, %2, %3);").arg(m_colCurrentSceneColor.red()).arg(m_colCurrentSceneColor.green()).arg(m_colCurrentSceneColor.blue()));

264void Control3DView::onCustomContextMenuRequested(QPoint pos)
265{
266 //create custom context menu and actions
267 QMenu* menu = new QMenu(this);
268 menu->setAttribute(Qt::WA_DeleteOnClose);
269
270 // Hide header
271 QAction* pHideHeader = menu->addAction(tr("Toggle header"));
272 connect(pHideHeader, &QAction::triggered,
273 this, &Control3DView::onTreeViewHeaderHide);
274
275 // Remove item
276 QAction* pRemoveItem = menu->addAction(tr("Remove"));
277 connect(pRemoveItem, &QAction::triggered, [=, this]() {
278 if (QMessageBox::question(this,
279 tr("Remove item"),
280 tr("Are you sure you want to delete the item?")) == QMessageBox::Yes) {
281 onTreeViewRemoveItem(m_pUi->m_treeView_loadedData->indexAt(pos));
282 }
283 });

287 // this, &Control3DView::onTreeViewDescriptionHide);
288
289 //show context menu
