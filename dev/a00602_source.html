178void ChannelWindow::initRhi()
179{
180 switch (m_api) {
181 case Api::Null: {
182 QRhiNullInitParams params;
183 m_pRhi.reset(QRhi::create(QRhi::Null, &params));
184 break;
185 }
186#if QT_CONFIG(opengl)
187 case Api::OpenGL: {
188 m_pFallbackSurface.reset(QRhiGles2InitParams::newFallbackSurface());
189 QRhiGles2InitParams params;
190 params.fallbackSurface = m_pFallbackSurface.get();
191 params.window = this;
192 m_pRhi.reset(QRhi::create(QRhi::OpenGLES2, &params));
193 break;
194 }
195#endif
196#if QT_CONFIG(vulkan) && __has_include(<vulkan/vulkan.h>)
197 case Api::Vulkan: {
198 if (!vulkanInstance()) {
199 mne::Log::warning("ChannelWindow: Vulkan needs a QVulkanInstance set with QWindow::setVulkanInstance()");
200 break;
201 }
202 QRhiVulkanInitParams params;
203 params.inst = vulkanInstance();
204 params.window = this;
205 m_pRhi.reset(QRhi::create(QRhi::Vulkan, &params));
206 break;
207 }
208#endif
209#if QT_CONFIG(metal)
210 case Api::Metal: {
211 QRhiMetalInitParams params;
212 m_pRhi.reset(QRhi::create(QRhi::Metal, &params));
213 break;
214 }
215#endif
216#if defined(Q_OS_WIN)
217 case Api::Direct3D11: {
218 QRhiD3D11InitParams params;
219 m_pRhi.reset(QRhi::create(QRhi::D3D11, &params));
220 break;
221 }
222 case Api::Direct3D12: {
223 QRhiD3D12InitParams params;
224 m_pRhi.reset(QRhi::create(QRhi::D3D12, &params));
225 break;
226 }
227#endif
228 default:
229 break;
230 }
231
232 if (!m_pRhi) {
233 mne::Log::warning("ChannelWindow: failed to create a QRhi for the selected graphics API");
234 return;
235 }
236
237 m_pSwapChain.reset(m_pRhi->newSwapChain());
238 m_pSwapChain->setWindow(this);
239 m_pRenderPass.reset(m_pSwapChain->newCompatibleRenderPassDescriptor());
240 m_pSwapChain->setRenderPassDescriptor(m_pRenderPass.get());
241 m_pRenderer->initializeResources();
242}

265void ChannelWindow::renderNow()
266{
267 if (!m_hasSwapChain || m_notExposed)
268 return;
269
270 if (m_pSwapChain->currentPixelSize() != m_pSwapChain->surfacePixelSize() || m_newlyExposed) {
271 resizeSwapChain();
272 if (!m_hasSwapChain)
273 return;
274 m_newlyExposed = false;
275 }
276
277 QRhi::FrameOpResult result = m_pRhi->beginFrame(m_pSwapChain.get());
278 if (result == QRhi::FrameOpSwapChainOutOfDate) {
279 resizeSwapChain();
280 if (!m_hasSwapChain)
281 return;
282 result = m_pRhi->beginFrame(m_pSwapChain.get());
283 }
284 if (result != QRhi::FrameOpSuccess) {
285 requestUpdate();
286 return;
287 }
288
289 m_pRenderer->renderFrame(m_pRhi.get(), m_pSwapChain->currentFrameRenderTarget(),
290 m_pSwapChain->currentFrameCommandBuffer());
291 m_pRhi->endFrame(m_pSwapChain.get());
292 ++m_frameCount;
293}

297void ChannelWindow::exposeEvent(QExposeEvent* event)
298{
299 Q_UNUSED(event);
300
301 if (isExposed() && !m_initialized) {
302 initRhi();
303 resizeSwapChain();
304 m_initialized = true;
305 }
306
307 const QSize surfaceSize = m_hasSwapChain ? m_pSwapChain->surfacePixelSize() : QSize();
308
309 // Stop rendering while the window is hidden or minimised
310 if ((!isExposed() || (m_hasSwapChain && surfaceSize.isEmpty())) && m_initialized && !m_notExposed)
311 m_notExposed = true;
312
313 // Resize the swap chain on the next frame after the window became visible again
314 if (isExposed() && m_initialized && m_notExposed && !surfaceSize.isEmpty()) {
315 m_notExposed = false;
316 m_newlyExposed = true;
317 }
318
319 if (isExposed() && !surfaceSize.isEmpty())
320 renderNow();
321}

325bool ChannelWindow::event(QEvent* event)
326{
327 switch (event->type()) {
328 case QEvent::UpdateRequest:
329 renderNow();
330 break;
331 case QEvent::PlatformSurface:
332 if (static_cast<QPlatformSurfaceEvent*>(event)->surfaceEventType() == QPlatformSurfaceEvent::SurfaceAboutToBeDestroyed)
333 releaseSwapChain();
334 break;
335 default:
336 break;
337 }
338 return QWindow::event(event);
339}
