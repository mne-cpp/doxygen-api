65std::vector<Network> mne::compute_connectivity(ConnectivitySettings& settings)
66{
67 using Method = Network (*)(ConnectivitySettings&);
68 static constexpr std::pair<std::string_view, Method> methods[] = {
69 {"WPLI", &WeightedPhaseLagIndex::calculate},
70 {"USPLI", &UnbiasedSquaredPhaseLagIndex::calculate},
71 {"COR", &Correlation::calculate},
72 {"XCOR", &CrossCorrelation::calculate},
73 {"PLI", &PhaseLagIndex::calculate},
74 {"COH", &Coherence::calculate},
75 {"IMAGCOH", &ImagCoherence::calculate},
76 {"PLV", &PhaseLockingValue::calculate},
77 {"DSWPLI", &DebiasedSquaredWeightedPhaseLagIndex::calculate},
78 {"GC", &GrangerCausality::calculate},
79 {"DTF", &DirectedTransferFunction::calculate},
80 {"PDC", &PartialDirectedCoherence::calculate},
81 };
82
83 const std::vector<std::string>& requested = settings.getConnectivityMethods();
84 std::vector<Network> networks;
85 networks.reserve(requested.size());
86 for (const auto& [name, method] : methods) {
87 if (std::ranges::find(requested, name) != requested.end()) {
88 networks.push_back(method(settings));
89 }
90 }
91 return networks;
92}
