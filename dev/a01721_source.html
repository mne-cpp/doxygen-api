96void AverageSelectionPanel::load(const SettingsStore& store, const std::string& path)
97{
98 if (path.empty()) {
99 return;
100 }
101 const std::string colorGroup = settings_key(path, kColorGroup);
102 for (const std::string& name : store.child_keys(colorGroup)) {
103 if (const std::optional<Color> color = parse_setting_color(store.read_string(child_key(colorGroup, name), {}))) {
104 m_colors[name] = *color;

106 }
107 const std::string activationGroup = settings_key(path, kActivationGroup);
108 for (const std::string& name : store.child_keys(activationGroup)) {
109 m_activation[name] = store.read_bool(child_key(activationGroup, name), true);
110 }
111 rebuild();

133void AverageSelectionPanel::rebuild()
134{
135 // The colours and activations arrive one after the other; the form waits until they match
136 if (m_colors.size() != m_activation.size()) {
137 return;
138 }
139 const bool updating = std::exchange(m_updating, true);
140 std::vector<std::string> keys;
141 keys.reserve(form().fields().size());
142 for (const SettingField& field : form().fields()) {
143 keys.push_back(field.key);
144 }
145 for (const std::string& key : keys) {
146 form().remove(key);
147 }
148 int count = 0;
149 for (const auto& [name, color] : m_colors) {
150 if (count++ >= kMaxAverages) {
151 break;
152 }
153 const auto active = m_activation.find(name);
154 SettingField show;
155 show.key = kActive + name;
156 show.kind = SettingField::Kind::Bool;
157 show.label = name;
158 show.default_value = active != m_activation.end() && active->second;
159 show.persistent = false;
160 form().add(std::move(show));
161 SettingField tint;
162 tint.key = kColor + name;
163 tint.kind = SettingField::Kind::Color;
164 tint.label = "Color:";
165 tint.default_value = setting_color(color);
166 tint.persistent = false;
167 form().add(std::move(tint));
168 }
169 m_updating = updating;
170}
