40RtSourceDataController::RtSourceDataController(QObject* parent)
41: QObject(parent)
42{
43 // Create data worker
44 // streamData() runs on the streaming thread; its results are re-emitted on this controller's thread.
45 m_pWorker = std::make_unique<mne::viz::RtSourceDataWorker>();
46
47 // Forward worker signals to controller signals
48 m_pWorker->setNewRtSmoothedDataCallback([this](const std::vector<uint32_t>& colorsLh, const std::vector<uint32_t>& colorsRh) {
49 QMetaObject::invokeMethod(this, [this, colorsLh, colorsRh] { emit newSmoothedDataAvailable(colorsLh, colorsRh); }, Qt::QueuedConnection);
50 });
51 m_pWorker->setNewRtRawDataCallback([this](const Eigen::VectorXd& dataLh, const Eigen::VectorXd& dataRh) {
52 QMetaObject::invokeMethod(this, [this, dataLh, dataRh] { emit newRawDataAvailable(dataLh, dataRh); }, Qt::QueuedConnection);
53 });
54
55 // Streaming thread driving the cadence
56 m_pStreamTask = std::make_unique<PeriodicTask>([worker = m_pWorker.get()] { worker->streamData(); });
57
58 // Create interpolation matrix worker
59 // One pool thread runs the recomputations in request order.
60 m_pInterpWorker = std::make_unique<mne::viz::RtSourceInterpolationMatWorker>();
61 m_pInterpPool = std::make_unique<mne::ThreadPool>(1);
62
63 // Forward interpolation results: auto-apply to data worker + re-emit
64 m_pInterpWorker->setNewInterpolationMatrixLeftAvailableCallback([this](const std::shared_ptr<Eigen::SparseMatrix<float>>& interpMat) {
65 QMetaObject::invokeMethod(this, [this, interpMat] { onNewInterpolationMatrixLeft(interpMat); }, Qt::QueuedConnection);
66 });
67 m_pInterpWorker->setNewInterpolationMatrixRightAvailableCallback([this](const std::shared_ptr<Eigen::SparseMatrix<float>>& interpMat) {
68 QMetaObject::invokeMethod(this, [this, interpMat] { onNewInterpolationMatrixRight(interpMat); }, Qt::QueuedConnection);
69 });
70
71 mne::Log::debug("RtSourceDataController: Initialized with {} ms interval", m_iTimeInterval);
72}

76RtSourceDataController::~RtSourceDataController()
77{
78 // Join the threads while the workers are still alive. Pending
79 // recomputations are skipped.
80 m_bShuttingDown.store(true, std::memory_order_release);
81 m_pStreamTask.reset();
82 m_pInterpPool.reset();
83}
