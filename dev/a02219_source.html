98MatrixX3i FsLabel::selectTris(const MatrixX3i& p_matTris)
99{
100 //check whether there are data to create the tris
101 if (this->vertices.size() == 0)
102 return MatrixX3i(0, 3);
103
104 MatrixX3i tris(p_matTris.rows(), 3);
105
106 const std::unordered_set<int> verts(this->vertices.data(), this->vertices.data() + this->vertices.size());
107
108 //
109 // Search for all the tris where is at least one corner part of the label
110 //
111 std::int32_t t_size = 0;
112 for (std::int32_t i = 0; i < p_matTris.rows(); ++i) {
113 if (verts.contains(p_matTris(i, 0)) || verts.contains(p_matTris(i, 1)) || verts.contains(p_matTris(i, 2))) {
114 tris.row(t_size) = p_matTris.row(i);
115 ++t_size;
116 }
117 }
118
119 tris.conservativeResize(t_size, 3);
120
121 return tris;
122}

126bool FsLabel::read(const std::filesystem::path& p_sFileName, FsLabel& p_Label)
127{
128 p_Label.clear();
129
130 const std::string fileName = mne::utf8_from_path(p_sFileName.filename());
131 if (!fileName.ends_with(".label")) {
132 mne::Log::warning("Given file ({}) is not a .label file!\n", mne::utf8_from_path(p_sFileName));
133 return false;
134 }
135
136 mne::Log::info("Reading label...");
137 std::ifstream file(p_sFileName);
138 if (!file) {
139 mne::Log::warning("\tError: Couldn't open the label file\n");
140 return false;
141 }
142
143 std::string comment;
144 std::string line;
145 std::getline(file, comment);
146 std::getline(file, line);
147 const int nv = mne::to_int(line).value_or(0);
148
149 // One row per vertex: vertex number, x, y, z (mm) and value
150 MatrixXd data = MatrixXd::Zero(nv, 5);
151 for (std::int32_t i = 0; i < nv && std::getline(file, line); ++i) {
152 std::istringstream fields(line);
153 for (std::int32_t j = 0; j < 5 && fields >> data(i, j); ++j) {
154 }
155 }
156
157 p_Label.comment = comment.empty() ? comment : comment.substr(1);
158 // As mne.read_label
159 p_Label.hemi = fileName.ends_with("lh.label") || fileName.starts_with("lh.") ? 0 : 1;
160 p_Label.vertices = data.col(0).cast<int>();
161 p_Label.pos = data.middleCols(1, 3).cast<float>().array() * 1e-3f;
162 p_Label.values = data.col(4);
163
164 // <name>-lh.label or lh.<name>.label
165 if (const std::size_t at = fileName.find("-lh.label"); at != std::string::npos)
166 p_Label.name = fileName.substr(0, at);
167 else if (const std::size_t rh = fileName.find("-rh.label"); rh != std::string::npos)
168 p_Label.name = fileName.substr(0, rh);
169 else if (fileName.size() > 3 + 6 && (fileName.starts_with("lh.") || fileName.starts_with("rh.")))
170 p_Label.name = fileName.substr(3, fileName.size() - 3 - 6);
171
172 mne::Log::info("[done]\n");
173
174 return true;
175}
