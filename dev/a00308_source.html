48FiffCoordTrans FiffCoordTransSet::readMniTransform(const QString& path)
49{
50 QFile file(path);
51 if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
52 return FiffCoordTrans();
53 }
54 // "Transform_Type = Linear; Linear_Transform = <3 x 4 matrix, translation in mm> ;"
55 const QString text = QTextStream(&file).readAll();
56 static const QRegularExpression linear(QStringLiteral("Transform_Type\\s*=\\s*Linear\\s*;\\s*Linear_Transform\\s*=([^;]*);"),
57 QRegularExpression::CaseInsensitiveOption);
58 const QRegularExpressionMatch match = linear.match(text);
59 const QStringList values = match.captured(1).split(QRegularExpression(QStringLiteral("\\s+")), Qt::SkipEmptyParts);
60 if (!match.hasMatch() || values.size() != 12) {
61 return FiffCoordTrans();
62 }
63 Matrix3f rot;
64 Vector3f move;
65 for (int row = 0; row < 3; ++row) {
66 for (int col = 0; col < 4; ++col) {
67 bool ok = false;
68 const float value = values[4 * row + col].toFloat(&ok);
69 if (!ok) {
70 return FiffCoordTrans();
71 }
72 if (col < 3) {
73 rot(row, col) = value;
74 } else {
75 move[row] = value / 1000.0f;
76 }
77 }
78 }
79 return FiffCoordTrans(FIFFV_MNE_COORD_RAS, FIFFV_MNE_COORD_MNI_TAL, rot, move);
80}

84bool FiffCoordTransSet::addTalairach(const QString& xfmPath)
85{
86 const FiffCoordTrans rasMni = readMniTransform(xfmPath);
87 if (rasMni.isEmpty()) {
88 return false;
89 }
90 // M. Brett's MNI -> Talairach approximation, separately above and below the AC-PC plane.
91 Matrix3f above;
92 above << 0.99f, 0.0f, 0.0f, 0.0f, 0.9688f, 0.046f, 0.0f, -0.0485f, 0.9189f;
93 Matrix3f below;
94 below << 0.99f, 0.0f, 0.0f, 0.0f, 0.9688f, 0.042f, 0.0f, -0.0485f, 0.839f;
95 RAS_MNI_tal_t = rasMni;
96 MNI_tal_tal_gtz_t = FiffCoordTrans(FIFFV_MNE_COORD_MNI_TAL, FIFFV_MNE_COORD_FS_TAL_GTZ, above, Vector3f::Zero());
97 MNI_tal_tal_ltz_t = FiffCoordTrans(FIFFV_MNE_COORD_MNI_TAL, FIFFV_MNE_COORD_FS_TAL_LTZ, below, Vector3f::Zero());

103int FiffCoordTransSet::read(const QString& path)
104{
105 QFile file(path);
106 FiffStream::SPtr stream(new FiffStream(&file));
107 if (!stream->open()) {
108 return -1;
109 }
110 const struct
111 {
112 int from;
113 int to;
114 FiffCoordTrans FiffCoordTransSet::* member;
115 } links[] = {
116 {FIFFV_COORD_HEAD, FIFFV_COORD_MRI, &FiffCoordTransSet::head_surf_RAS_t},
117 {FIFFV_COORD_MRI, FIFFV_MNE_COORD_RAS, &FiffCoordTransSet::surf_RAS_RAS_t},
118 {FIFFV_MNE_COORD_RAS, FIFFV_MNE_COORD_MNI_TAL, &FiffCoordTransSet::RAS_MNI_tal_t},
119 {FIFFV_MNE_COORD_MNI_TAL, FIFFV_MNE_COORD_FS_TAL_GTZ, &FiffCoordTransSet::MNI_tal_tal_gtz_t},
120 {FIFFV_MNE_COORD_MNI_TAL, FIFFV_MNE_COORD_FS_TAL_LTZ, &FiffCoordTransSet::MNI_tal_tal_ltz_t},
121 };
122 int count = 0;
123 FiffTag::UPtr tag;
124 for (const auto& entry : stream->dir()) {
125 if (entry->kind != FIFF_COORD_TRANS || !stream->read_tag(tag, entry->pos)) {
126 continue;
127 }
128 const FiffCoordTrans t = tag->toCoordTrans();
129 for (const auto& link : links) {
130 if (t.from == link.from && t.to == link.to) {
131 this->*link.member = t;
132 ++count;
133 } else if (t.from == link.to && t.to == link.from) {
134 this->*link.member = t.inverted();
135 ++count;
136 }
137 }
138 }
139 stream->close();
140 return count;
141}

157{
158 if (head_surf_RAS_t.isEmpty() || surf_RAS_RAS_t.isEmpty() || RAS_MNI_tal_t.isEmpty()) {

166MatrixX3f FiffCoordTransSet::mniToTalairach(const MatrixX3f& rr) const
167{
168 if (MNI_tal_tal_gtz_t.isEmpty() || MNI_tal_tal_ltz_t.isEmpty()) {
169 return MatrixX3f();
170 }
171 const MatrixX3f above = MNI_tal_tal_gtz_t.apply_trans(rr);
172 const MatrixX3f below = MNI_tal_tal_ltz_t.apply_trans(rr);
173 MatrixX3f result(rr.rows(), 3);
174 for (Index k = 0; k < rr.rows(); ++k) {
175 result.row(k) = rr(k, 2) > 0.0f ? above.row(k) : below.row(k);
176 }
177 return result;
178}
