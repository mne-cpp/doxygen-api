88mne::Expected<FiffCov> FiffCov::read(const std::filesystem::path& path)
89{
90 auto stream = FiffStream::open(path);
91 if (!stream) {
92 return std::unexpected(stream.error());

94 FiffCov cov;

119FiffCov FiffCov::pick_channels(const StringList& p_include, const StringList& p_exclude)
120{
121 RowVectorXi sel = FiffInfoBase::pick_channels(this->names, p_include, p_exclude);
122 FiffCov res; //No deep copy here - since almost everything else is adapted anyway
123
124 res.kind = this->kind;
125 res.diag = this->diag;
126 res.dim = sel.size();
127
128 for (std::int32_t k = 0; k < sel.size(); ++k)
129 res.names.push_back(this->names[sel(k)]);
130
131 res.data.resize(res.dim, res.dim);
132 for (std::int32_t i = 0; i < res.dim; ++i)
133 for (std::int32_t j = 0; j < res.dim; ++j)
134 res.data(i, j) = this->data(sel(i), sel(j));
135 res.projs = this->projs;
136
137 for (std::int32_t k = 0; k < static_cast<int>(this->bads.size()); ++k)
138 if (mne::has_channel(res.names, this->bads[k]))
139 res.bads.push_back(this->bads[k]);
140 res.nfree = this->nfree;
141
142 return res;
143}

147FiffCov FiffCov::prepare_noise_cov(const FiffInfo& p_Info, const StringList& p_ChNames) const
148{
149 FiffCov p_NoiseCov(*this);
150
151 VectorXi C_ch_idx = VectorXi::Zero(p_NoiseCov.names.size());
152 std::int32_t count = 0;
153 for (std::int32_t i = 0; i < static_cast<int>(p_ChNames.size()); ++i) {
154 std::int32_t idx = mne::channel_index(p_NoiseCov.names, p_ChNames[i]);
155 if (idx > -1) {
156 C_ch_idx[count] = idx;
157 ++count;
158 }
159 }
160 C_ch_idx.conservativeResize(count);
161
162 MatrixXd C(count, count);
163
164 if (!p_NoiseCov.diag)
165 for (std::int32_t i = 0; i < count; ++i)
166 for (std::int32_t j = 0; j < count; ++j)
167 C(i, j) = p_NoiseCov.data(C_ch_idx(i), C_ch_idx(j));
168 else {
169 Log::warning("Warning in FiffCov::prepare_noise_cov: This has to be debugged - not done before!");
170 C = MatrixXd::Zero(count, count);
171 for (std::int32_t i = 0; i < count; ++i)
172 C.diagonal()[i] = p_NoiseCov.data(C_ch_idx(i), 0);
173 }
174
175 MatrixXd proj;
176 std::int32_t ncomp = p_Info.make_projector(proj, p_ChNames);
177
178 //Create the projection operator
179 if (ncomp > 0 && proj.rows() == count) {
180 Log::info("Created an SSP operator (subspace dimension = {})\n", ncomp);
181 C = proj * (C * proj.transpose());
182 } else {
183 Log::warning("Warning in FiffCov::prepare_noise_cov: No projections applied since no projectors specified or projector dimensions do not match!");
184 }
185
186 RowVectorXi pick_meg = p_Info.pick_types(true, false, false, {}, p_Info.bads);
187 RowVectorXi pick_eeg = p_Info.pick_types(false, true, false, {}, p_Info.bads);
188
189 StringList meg_names, eeg_names;
190
191 for (std::int32_t i = 0; i < pick_meg.size(); ++i)
192 meg_names.push_back(p_Info.chs[pick_meg[i]].ch_name);
193 VectorXi C_meg_idx = VectorXi::Zero(p_NoiseCov.names.size());
194 count = 0;
195 for (std::int32_t k = 0; k < C.rows(); ++k) {
196 if (mne::channel_index(meg_names, p_ChNames[k]) > -1) {
197 C_meg_idx[count] = k;
198 ++count;
199 }
200 }
201 if (count > 0)
202 C_meg_idx.conservativeResize(count);
203 else
204 C_meg_idx = VectorXi();
205
206 //
207 for (std::int32_t i = 0; i < pick_eeg.size(); ++i)
208 eeg_names.push_back(p_Info.chs[pick_eeg(0, i)].ch_name);
209 VectorXi C_eeg_idx = VectorXi::Zero(p_NoiseCov.names.size());
210 count = 0;
211 for (std::int32_t k = 0; k < C.rows(); ++k) {
212 if (mne::channel_index(eeg_names, p_ChNames[k]) > -1) {
213 C_eeg_idx[count] = k;
214 ++count;
215 }
216 }
217
218 if (count > 0)
219 C_eeg_idx.conservativeResize(count);
220 else
221 C_eeg_idx = VectorXi();
222
223 bool has_meg = C_meg_idx.size() > 0;
224 bool has_eeg = C_eeg_idx.size() > 0;
225
226 MatrixXd C_meg, C_eeg;
227 VectorXd C_meg_eig, C_eeg_eig;
228 MatrixXd C_meg_eigvec, C_eeg_eigvec;
229 if (has_meg) {
230 count = C_meg_idx.rows();
231 C_meg = MatrixXd(count, count);
232 for (std::int32_t i = 0; i < count; ++i)
233 for (std::int32_t j = 0; j < count; ++j)
234 C_meg(i, j) = C(C_meg_idx(i), C_meg_idx(j));
235 Linalg::get_whitener(C_meg, false, "MEG", C_meg_eig, C_meg_eigvec);
236 }
237
238 if (has_eeg) {
239 count = C_eeg_idx.rows();
240 C_eeg = MatrixXd(count, count);
241 for (std::int32_t i = 0; i < count; ++i)
242 for (std::int32_t j = 0; j < count; ++j)
243 C_eeg(i, j) = C(C_eeg_idx(i), C_eeg_idx(j));
244 Linalg::get_whitener(C_eeg, false, "EEG", C_eeg_eig, C_eeg_eigvec);
245 }
246
247 std::int32_t n_chan = static_cast<int>(p_ChNames.size());
248 p_NoiseCov.eigvec = MatrixXd::Zero(n_chan, n_chan);
249 p_NoiseCov.eig = VectorXd::Zero(n_chan);
250
251 if (has_meg) {
252 for (std::int32_t i = 0; i < C_meg_idx.rows(); ++i)
253 for (std::int32_t j = 0; j < C_meg_idx.rows(); ++j)
254 p_NoiseCov.eigvec(C_meg_idx[i], C_meg_idx[j]) = C_meg_eigvec(i, j);
255 for (std::int32_t i = 0; i < C_meg_idx.rows(); ++i)
256 p_NoiseCov.eig(C_meg_idx[i]) = C_meg_eig[i];
257 }
258 if (has_eeg) {
259 for (std::int32_t i = 0; i < C_eeg_idx.rows(); ++i)
260 for (std::int32_t j = 0; j < C_eeg_idx.rows(); ++j)
261 p_NoiseCov.eigvec(C_eeg_idx[i], C_eeg_idx[j]) = C_eeg_eigvec(i, j);
262 for (std::int32_t i = 0; i < C_eeg_idx.rows(); ++i)
263 p_NoiseCov.eig(C_eeg_idx[i]) = C_eeg_eig[i];
264 }
265
266 if (C_meg_idx.size() + C_eeg_idx.size() != n_chan) {
267 Log::warning("FiffCov::prepare_noise_cov: {} MEG + {} EEG channels != {} total (unclassified channels present)", (long long)C_meg_idx.size(), (long long)C_eeg_idx.size(), n_chan);
268 }
269
270 p_NoiseCov.data = C;
271 p_NoiseCov.dim = static_cast<int>(p_ChNames.size());
272 p_NoiseCov.diag = false;
273 p_NoiseCov.names = p_ChNames;
274
275 return p_NoiseCov;
276}

280FiffCov FiffCov::regularize(const FiffInfo& p_info, double p_fRegMag, double p_fRegGrad, double p_fRegEeg, bool p_bProj, StringList p_exclude) const
281{
282 FiffCov cov(*this);
283
284 if (p_exclude.size() == 0) {
285 p_exclude = p_info.bads;
286 for (std::int32_t i = 0; i < static_cast<int>(cov.bads.size()); ++i)
287 if (!mne::has_channel(p_exclude, cov.bads[i]))
288 p_exclude.push_back(cov.bads[i]);
289 }
290
291 //Allways exclude all STI channels from covariance computation
292 for (int i = 0; i < std::ssize(p_info.chs); i++) {
293 if (p_info.chs[i].kind == FIFFV_STIM_CH) {
294 p_exclude.push_back(p_info.chs[i].ch_name);
295 }
296 }
297
298 RowVectorXi sel_eeg = p_info.pick_types(false, true, false, {}, p_exclude);
299 RowVectorXi sel_mag = p_info.pick_types(std::string("mag"), false, false, {}, p_exclude);
300 RowVectorXi sel_grad = p_info.pick_types(std::string("grad"), false, false, {}, p_exclude);
301
302 StringList info_ch_names = p_info.ch_names;
303 StringList ch_names_eeg, ch_names_mag, ch_names_grad;
304 for (std::int32_t i = 0; i < sel_eeg.size(); ++i)
305 ch_names_eeg.push_back(info_ch_names[sel_eeg(i)]);
306 for (std::int32_t i = 0; i < sel_mag.size(); ++i)
307 ch_names_mag.push_back(info_ch_names[sel_mag(i)]);
308 for (std::int32_t i = 0; i < sel_grad.size(); ++i)
309 ch_names_grad.push_back(info_ch_names[sel_grad(i)]);
310
311 // This actually removes bad channels from the cov, which is not backward
312 // compatible, so let's leave all channels in
313 FiffCov cov_good = cov.pick_channels(info_ch_names, p_exclude);
314 StringList ch_names = cov_good.names;
315
316 std::vector<std::int32_t> idx_eeg, idx_mag, idx_grad;
317 for (std::int32_t i = 0; i < static_cast<int>(ch_names.size()); ++i) {
318 if (mne::has_channel(ch_names_eeg, ch_names[i]))
319 idx_eeg.push_back(i);
320 else if (mne::has_channel(ch_names_mag, ch_names[i]))
321 idx_mag.push_back(i);
322 else if (mne::has_channel(ch_names_grad, ch_names[i]))
323 idx_grad.push_back(i);
324 }
325
326 MatrixXd C(cov_good.data);
327
328 //Check dimension consistency (channels not classified as EEG/MAG/GRAD, e.g. EOG/MISC, are expected)
329 if (static_cast<unsigned>(C.rows()) != idx_eeg.size() + idx_mag.size() + idx_grad.size()) {
330 Log::warning("FiffCov::regularize: {} channels in cov but only {} classified as EEG/MAG/GRAD (others will not be regularized)", C.rows(), idx_eeg.size() + idx_mag.size() + idx_grad.size());
331 }
332
333 std::vector<FiffProj> t_listProjs;
334 if (p_bProj) {
335 t_listProjs.reserve(p_info.projs.size() + cov_good.projs.size());
336 t_listProjs.insert(t_listProjs.end(), p_info.projs.begin(), p_info.projs.end());
337 t_listProjs.insert(t_listProjs.end(), cov_good.projs.begin(), cov_good.projs.end());
338 FiffProj::activate_projs(t_listProjs);
339 }
340
341 struct Regularization
342 {
343 std::string_view desc;
344 double reg;
345 const std::vector<std::int32_t>& idx;
346 };
347 const Regularization regData[] = {{"EEG", p_fRegEeg, idx_eeg}, {"GRAD", p_fRegGrad, idx_grad}, {"MAG", p_fRegMag, idx_mag}};
348
349 for (const auto& [desc, reg, idx] : regData) {
350 if (idx.size() == 0 || reg == 0.0)
351 Log::info("\tNothing to regularize within {} data.\n", desc);
352 else {
353 Log::info("\tRegularize {}: {:f}\n", desc, reg);
354 MatrixXd this_C(idx.size(), idx.size());
355 for (std::uint32_t i = 0; i < idx.size(); ++i)
356 for (std::uint32_t j = 0; j < idx.size(); ++j)
357 this_C(i, j) = cov_good.data(idx[i], idx[j]);
358
359 MatrixXd U;
360 // Only assigned when projecting, but read again below to decide
361 // whether the SSP operator has to be undone. Without projection
362 // there are no components to remove.
363 std::int32_t ncomp = 0;
364 if (p_bProj) {
365 StringList this_ch_names;
366 for (std::uint32_t k = 0; k < idx.size(); ++k)
367 this_ch_names.push_back(ch_names[idx[k]]);
368
369 MatrixXd P;
370 ncomp = FiffProj::make_projector(t_listProjs, this_ch_names, P); //ToDo: Synchronize with mne-python and debug
371
372 JacobiSVD<MatrixXd> svd(P, ComputeFullU);
373 //Sort singular values and singular vectors
374 VectorXd t_s = svd.singularValues();
375 MatrixXd t_U = svd.matrixU();
376 Linalg::sort<double>(t_s, t_U);
377
378 U = t_U.block(0, 0, t_U.rows(), t_U.cols() - ncomp);
379
380 if (ncomp > 0) {
381 Log::info("\tCreated an SSP operator for {} (dimension = {}).\n", desc, ncomp);
382 this_C = U.transpose() * (this_C * U);
383 }
384 }
385
386 double sigma = this_C.diagonal().mean();
387 this_C.diagonal() = this_C.diagonal().array() + reg * sigma; // modify diag inplace
388 if (p_bProj && ncomp > 0)
389 this_C = U * (this_C * U.transpose());
390
391 for (std::int32_t i = 0; i < this_C.rows(); ++i)
392 for (std::int32_t j = 0; j < this_C.cols(); ++j)
393 C(idx[i], idx[j]) = this_C(i, j);
394 }
395 }
396
397 // Put data back in correct locations
398 RowVectorXi idx = FiffInfo::pick_channels(cov.names, info_ch_names, p_exclude);
399 for (std::int32_t i = 0; i < idx.size(); ++i)
400 for (std::int32_t j = 0; j < idx.size(); ++j)
401 cov.data(idx[i], idx[j]) = C(i, j);
402
403 return cov;
404}

408FiffCov FiffCov::compute_from_epochs(const FiffRawData& raw,
409 const MatrixXi& events,
410 const std::vector<int>& eventCodes,
411 float tmin,
412 float tmax,
413 float bmin,
414 float bmax,
415 bool doBaseline,
416 bool removeMean,
417 unsigned int ignoreMask,
418 float delay,
419 const RejectionParams* rej)
420{
421 FiffCov cov;
422 float sfreq = raw.info.sfreq;
423 int nchan = raw.info.nchan;
424
425 int minSamp = static_cast<int>(std::round(tmin * sfreq));
426 int maxSamp = static_cast<int>(std::round(tmax * sfreq));
427 int ns = maxSamp - minSamp + 1;
428 int delaySamp = static_cast<int>(std::round(delay * sfreq));
429
430 if (ns <= 0) {
431 Log::warning("[FiffCov::compute_from_epochs] Invalid time window.");
432 return cov;
433 }
434
435 int bminSamp = 0, bmaxSamp = 0;
436 if (doBaseline) {
437 bminSamp = static_cast<int>(std::round(bmin * sfreq)) - minSamp;
438 bmaxSamp = static_cast<int>(std::round(bmax * sfreq)) - minSamp;
439 }
440
441 // The accepted epochs, grouped by event code
442 std::vector<std::vector<MatrixXd>> epochs(eventCodes.size());
443 for (int k = 0; k < events.rows(); ++k) {
444 int evFrom = events(k, 1) & ~static_cast<int>(ignoreMask);
445 int evTo = events(k, 2) & ~static_cast<int>(ignoreMask);
446
447 const auto codeIt = std::ranges::find(eventCodes, evTo);
448 const int iCode = evFrom == 0 && codeIt != eventCodes.end() ? static_cast<int>(codeIt - eventCodes.begin()) : -1;
449 if (iCode < 0)
450 continue;
451
452 int evSample = events(k, 0);
453 int epochStart = evSample + delaySamp + minSamp;
454 int epochEnd = evSample + delaySamp + maxSamp;
455
456 if (epochStart < raw.first_samp || epochEnd > raw.last_samp)
457 continue;
458
459 MatrixXd epochData;
460 MatrixXd epochTimes;
461 if (!raw.read_raw_segment(epochData, epochTimes, epochStart, epochEnd))
462 continue;
463 std::string reason;
464 if (rej && !FiffEvokedSet::checkArtifacts(epochData, raw.info, raw.info.bads, *rej, reason)) {
465 Log::info("[FiffCov::compute_from_epochs] Rejected epoch at {} {}", evSample, reason);
466 continue;
467 }
468
469 // Baseline subtraction
470 if (doBaseline) {
471 int bminIdx = std::max(0, bminSamp);
472 int bmaxIdx = std::min(static_cast<int>(epochData.cols()) - 1, bmaxSamp);
473 if (bmaxIdx >= bminIdx) {
474 int nBase = bmaxIdx - bminIdx + 1;
475 for (int c = 0; c < nchan; ++c) {
476 double baseVal = epochData.row(c).segment(bminIdx, nBase).mean();
477 epochData.row(c).array() -= baseVal;
478 }
479 }
480 }
481
482 epochs[iCode].push_back(epochData);
483 }
484
485 return compute_from_epochs(epochs, raw.info, removeMean);
486}

490FiffCov FiffCov::compute_from_epochs(const std::vector<std::vector<MatrixXd>>& epochs,
491 const FiffInfo& info,
492 bool removeMean)
493{
494 FiffCov cov;
495 const int nchan = info.nchan;
496 MatrixXd covAccum = MatrixXd::Zero(nchan, nchan);
497 int totalSamples = 0;
498 int nEpochs = 0;
499 double norm = 0.0;
500 for (const std::vector<MatrixXd>& codeEpochs : epochs) {
501 if (codeEpochs.empty())
502 continue;
503 MatrixXd sum = MatrixXd::Zero(nchan, codeEpochs.front().cols());
504 for (const MatrixXd& epoch : codeEpochs) {
505 covAccum += epoch * epoch.transpose();
506 sum += epoch;
507 totalSamples += static_cast<int>(epoch.cols());
508 }
509 nEpochs += static_cast<int>(codeEpochs.size());
510 if (removeMean) {
511 // As mne.compute_covariance(keep_sample_mean=False) and MNE-C compute_cov: subtract each code's evoked response
512 covAccum -= sum * sum.transpose() / static_cast<double>(codeEpochs.size());
513 norm += static_cast<double>(sum.cols()) * (codeEpochs.size() - 1);
514 }
515 }
516
517 if (totalSamples < 2) {
518 Log::warning("[FiffCov::compute_from_epochs] Not enough data.");
519 return cov;
520 }
521 if (removeMean && norm <= 0.0) {
522 Log::warning("[FiffCov::compute_from_epochs] Removing the sample mean needs two epochs of one event code.");
523 return cov;
524 }
525 cov.data = covAccum / (removeMean ? norm : static_cast<double>(totalSamples - 1));
526
527 cov.kind = FIFFV_MNE_NOISE_COV;
528 cov.dim = nchan;
529 cov.names = info.ch_names;
530 cov.nfree = totalSamples - 1;
531 cov.bads = info.bads;
532 cov.projs = info.projs;
533
534 Log::info("[FiffCov::compute_from_epochs] Computed: {} channels, {} epochs, {} total samples.", nchan, nEpochs, totalSamples);
535
536 return cov;
537}

560FiffCov FiffCov::compute_grand_average(const std::vector<FiffCov>& covs)
561{
562 FiffCov grandCov;
563
564 if (covs.empty()) {
565 Log::warning("[FiffCov::compute_grand_average] No covariance matrices provided.");
566 return grandCov;
567 }
568
569 grandCov = covs[0];
570 MatrixXd sumCov = grandCov.data * static_cast<double>(grandCov.nfree);
571 int totalNfree = grandCov.nfree;
572
573 for (int k = 1; k < std::ssize(covs); ++k) {
574 if (covs[k].dim != grandCov.dim) {
575 Log::warning("[FiffCov::compute_grand_average] Dimension mismatch.");
576 return FiffCov();
577 }
578 sumCov += covs[k].data * static_cast<double>(covs[k].nfree);
579 totalNfree += covs[k].nfree;
580 }
581
582 grandCov.data = sumCov / static_cast<double>(totalNfree);
583 grandCov.nfree = totalNfree;
584
585 return grandCov;
586}
