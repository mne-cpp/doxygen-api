90bool FiffChInfo::checkEegLocations(const std::vector<FiffChInfo>& chs, int nch)
91{
92 const float close = 0.02f;
93 for (int k = 0; k < nch; k++) {
94 if (chs.at(k).kind == FIFFV_EEG_CH) {
95 if (chs.at(k).chpos.r0.norm() < close) {
96 mne::Log::critical("Some EEG channels do not have locations assigned.");
97 return false;
98 }
99 }
100 }
101 return true;
102}
