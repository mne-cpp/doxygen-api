55std::string randomUuid()
56{
57 static std::mt19937_64 generator{std::random_device{}()};
58 static std::mutex mutex;
59 std::array<std::uint64_t, 2> words{};
60 {
61 const std::lock_guard lock(mutex);
62 words = {generator(), generator()};
63 }
64 words[0] = (words[0] & ~std::uint64_t{0xF000}) | std::uint64_t{0x4000}; // version 4
65 words[1] = (words[1] & ~(std::uint64_t{0xC} << 60)) | (std::uint64_t{0x8} << 60); // RFC 4122 variant
66 return std::format("{:08x}-{:04x}-{:04x}-{:04x}-{:012x}", words[0] >> 32, (words[0] >> 16) & 0xFFFF, words[0] & 0xFFFF,
67 words[1] >> 48, words[1] & 0xFFFFFFFFFFFFULL);
68}

70std::string escaped(const std::string& text)
71{
72 std::string out;
73 out.reserve(text.size());
74 for (const char c : text) {
75 switch (c) {
76 case '&':
77 out += "&amp;";
78 break;
79 case '<':
80 out += "&lt;";
81 break;
82 case '>':
83 out += "&gt;";
84 break;
85 case '"':
86 out += "&quot;";
87 break;
88 case '\'':
89 out += "&apos;";
90 break;
91 default:
92 out += c;
93 break;
94 }
95 }
96 return out;
97}

252std::string stream_info::to_string() const
253{
254 std::ostringstream xml;
255 xml << "<?xml version=\"1.0\"?>\n<info>\n"
256 << "\t<name>" << escaped(m_name) << "</name>\n"
257 << "\t<type>" << escaped(m_type) << "</type>\n"
258 << "\t<channel_count>" << m_channel_count << "</channel_count>\n"
259 << "\t<channel_format>" << formatName(m_channel_format) << "</channel_format>\n"
260 << "\t<source_id>" << escaped(m_source_id) << "</source_id>\n"
261 << "\t<nominal_srate>" << number(m_nominal_srate) << "</nominal_srate>\n"
262 << "\t<version>1.1</version>\n"
263 << "\t<created_at>" << number(m_created_at) << "</created_at>\n"
264 << "\t<uid>" << escaped(m_uid) << "</uid>\n"
265 << "\t<session_id>default</session_id>\n"
266 << "\t<hostname>" << escaped(m_hostname) << "</hostname>\n"
267 << "\t<v4address></v4address>\n"
268 << "\t<v4data_port>" << m_data_port << "</v4data_port>\n"
269 << "\t<v4service_port>" << m_service_port << "</v4service_port>\n"
270 << "\t<v6address></v6address>\n"
271 << "\t<v6data_port>0</v6data_port>\n"
272 << "\t<v6service_port>0</v6service_port>\n"
273 << "\t<desc />\n</info>\n";
274 return xml.str();
275}

279stream_info stream_info::from_string(const std::string& data)
280{
281 stream_info info;
282 const Expected<XmlElement> root = parse_xml(data);
283 if (!root || root->local_name() != "info") {
284 return {};
285 }
286 for (const XmlElement& element : root->children) {
287 const std::string_view key = element.local_name();
288 const std::string& value = element.text;
289 if (key == "name") {
290 info.m_name = value;
291 } else if (key == "type") {
292 info.m_type = value;
293 } else if (key == "channel_count") {
294 info.m_channel_count = to_int(value).value_or(0);
295 } else if (key == "channel_format") {
296 info.m_channel_format = formatFromName(value);
297 } else if (key == "source_id") {
298 info.m_source_id = value;
299 } else if (key == "nominal_srate") {
300 info.m_nominal_srate = to_double(value).value_or(0.0);
301 } else if (key == "created_at") {
302 info.m_created_at = to_double(value).value_or(0.0);
303 } else if (key == "uid") {
304 info.m_uid = value;
305 } else if (key == "hostname") {
306 info.m_hostname = value;
307 } else if (key == "v4data_port") {
308 info.m_data_port = to_int(value).value_or(0);
309 } else if (key == "v4service_port") {
310 info.m_service_port = to_int(value).value_or(0);
311 } else if (key == "v4address" && !value.empty()) {
312 info.m_data_host = value;
313 }
314 }
315 if (info.m_name.empty()) {
316 return {};
317 }
318 return info;
319}
