105std::string timestamp()
106{
107 const auto now = std::chrono::system_clock::now();
108 const std::time_t seconds = std::chrono::system_clock::to_time_t(now);
109 std::tm local{};
110#ifdef _WIN32
111 localtime_s(&local, &seconds);
112#else
113 localtime_r(&seconds, &local);
114#endif
115 std::array<char, 32> date{};
116 std::strftime(date.data(), date.size(), "%Y-%m-%d %H:%M:%S", &local);
117 const auto millis = std::chrono::duration_cast<std::chrono::milliseconds>(now.time_since_epoch()).count() % 1000;
118 return std::format("{}.{:03}", date.data(), millis);
119}

121void printToConsole(Log::Level level, std::string_view text)
122{
123 const auto index = static_cast<std::size_t>(level);
124 // Results go to stdout and diagnostics to stderr, so piped tool output stays parseable.
125 std::ostream& stream = level >= Log::Level::Warning ? std::cerr : std::cout;
126 if (!text.empty() && text.back() == '\n') {
127 text.remove_suffix(1);
128 }
129
130 Console& out = console();
131 std::lock_guard<std::mutex> lock(out.mutex);
132 if (out.file.is_open()) {
133 // Flushed per line, so an unattended session keeps everything up to a crash.
134 out.file << timestamp() << " [" << kTags[index] << "] " << trimmed(text) << std::endl;
135 }
136#ifdef MNE_LOG_MODE_FORMATTED
137 constexpr std::array<std::string_view, 4> colors{"\033[32;1m", "\033[32;1m", "\033[33;1m", "\033[31;1m"};
138 for (std::size_t start = 0, end = 0; end != std::string_view::npos; start = end + 1) {
139 end = text.find('\n', start);
140 stream << colors[index] << '[' << kTags[index] << "] \033[0m" << text.substr(start, end - start) << '\n';
141 }
142#else
143 stream << text << '\n';
144#endif
145 stream.flush();
146}

168void Log::write(Level level, std::string_view message)
169{
170 if (!enabled(level)) {
171 return;
172 }
173 std::shared_ptr<const Sink> sink;
174 {
175 std::lock_guard<std::mutex> lock(sinkSlot().mutex);
176 sink = sinkSlot().sink;
177 }
178 // Called outside the lock, so a sink may itself change the sink.
179 if (sink) {
180 (*sink)(level, message);
181 } else {
182 printToConsole(level, message);
183 }
184}

209Log::Sink Log::set_sink(Sink sink)
210{
211 std::shared_ptr<const Sink> next = sink ? std::make_shared<const Sink>(std::move(sink)) : nullptr;
212 std::lock_guard<std::mutex> lock(sinkSlot().mutex);
213 std::swap(sinkSlot().sink, next);
