70VectorXd computeRowPsd(const VectorXd& row, int nFft, double sfreq)
71{
72 // Zero-pad or truncate to nFft
73 VectorXd segment = VectorXd::Zero(nFft);
74 int copyLen = std::min(static_cast<int>(row.size()), nFft);
75 segment.head(copyLen) = row.head(copyLen);
76
77 // Apply Hann window
78 for (int i = 0; i < copyLen; ++i) {
79 double w = 0.5 * (1.0 - std::cos(2.0 * M_PI * i / (copyLen - 1)));
80 segment(i) *= w;
81 }
82
83 // Compute FFT via correlation (real FFT using DFT)
84 int nFreqs = nFft / 2 + 1;
85 VectorXd psd(nFreqs);
86
87 for (int f = 0; f < nFreqs; ++f) {
88 double re = 0.0, im = 0.0;
89 for (int t = 0; t < nFft; ++t) {
90 double angle = -2.0 * M_PI * f * t / nFft;
91 re += segment(t) * std::cos(angle);
92 im += segment(t) * std::sin(angle);
93 }
94 psd(f) = (re * re + im * im) / (sfreq * nFft);
95 // Double for non-DC/Nyquist bins (one-sided spectrum)
96 if (f > 0 && f < nFreqs - 1) psd(f) *= 2.0;
97 }
98
99 return psd;
100}

108QList<InvSourceEstimate> INVLIB::applyInverseEpochs(
109 const QList<MatrixXd>& epochs,
110 const MNEInverseOperator& inverse,
111 float lambda2,
112 const QString& method,
113 float tmin,
114 float tstep,
115 bool pickNormal)
116{
117 QList<InvSourceEstimate> results;
118
119 if (epochs.isEmpty()) {
120 qWarning() << "[applyInverseEpochs] No epochs provided.";
121 return results;
122 }
123
124 // Create the minimum norm estimator
125 InvMinimumNorm mn(inverse, lambda2, method);
126
127 // Setup once with nave=1 (per-epoch)
128 mn.doInverseSetup(1, pickNormal);
129
130 for (int i = 0; i < epochs.size(); ++i) {
131 InvSourceEstimate stc = mn.calculateInverse(epochs[i], tmin, tstep, pickNormal);
132 if (stc.isEmpty()) {
133 qWarning() << "[applyInverseEpochs] Epoch" << i << "produced empty source estimate.";
134 }
135 results.append(stc);
136 }
137
138 return results;
139}

143InvSourceEstimate INVLIB::applyInverseRaw(
144 const FiffRawData& raw,
145 const MNEInverseOperator& inverse,
146 float lambda2,
147 const QString& method,
148 int from,
149 int to,
150 bool pickNormal)
151{
152 // Default: use full range
153 if (from < 0) from = raw.first_samp;
154 if (to < 0) to = raw.last_samp;
155
156 // Pick channels matching the inverse operator
157 RowVectorXi picks = FiffInfo::pick_channels(raw.info.ch_names, inverse.noise_cov->names);
158 if (picks.size() == 0) {
159 qWarning() << "[applyInverseRaw] No channels match the inverse operator.";
160 return InvSourceEstimate();
161 }
162
163 // Read raw data
164 MatrixXd data, times;
165 raw.read_raw_segment(data, times, from, to, picks);
166
167 if (data.cols() == 0) {
168 qWarning() << "[applyInverseRaw] No data read from raw file.";
169 return InvSourceEstimate();
170 }
171
172 float tmin = static_cast<float>(from) / raw.info.sfreq;
173 float tstep = 1.0f / raw.info.sfreq;
174
175 // Apply inverse
176 InvMinimumNorm mn(inverse, lambda2, method);
177 mn.doInverseSetup(1, pickNormal);
178
179 return mn.calculateInverse(data, tmin, tstep, pickNormal);
180}

184QPair<VectorXd, RowVectorXf> INVLIB::estimateSnr(
185 const FiffEvoked& evoked,
186 const MNEInverseOperator& inverse,
187 const QString& method)
188{
189 float snr = 3.0f;
190 float lambda2 = 1.0f / (snr * snr);
191
192 // Apply inverse to get source estimate
193 InvMinimumNorm mn(inverse, lambda2, method);
194 InvSourceEstimate stc = mn.calculateInverse(evoked);
195
196 if (stc.isEmpty()) {
197 qWarning() << "[estimateSnr] Failed to compute source estimate.";
198 return QPair<VectorXd, RowVectorXf>();
199 }
200
201 // Compute SNR as sqrt(sum of squared source amplitudes per time point)
202 // This gives a time course of source-space SNR
203 const int nTimes = static_cast<int>(stc.data.cols());
204 VectorXd snrTimeCourse(nTimes);
205
206 for (int t = 0; t < nTimes; ++t) {
207 snrTimeCourse(t) = std::sqrt(stc.data.col(t).squaredNorm()
208 / static_cast<double>(stc.data.rows()));
209 }
210
211 return QPair<VectorXd, RowVectorXf>(snrTimeCourse, stc.times);
212}

216QPair<MatrixXd, int> INVLIB::computeWhitener(
217 const FiffCov& noiseCov,
218 int rank)
219{
220 const int dim = noiseCov.dim;
221
222 if (dim <= 0) {
223 qWarning() << "[computeWhitener] Empty noise covariance.";
224 return QPair<MatrixXd, int>(MatrixXd(), 0);
225 }
226
227 // Use pre-computed eigendecomposition if available
228 VectorXd eig;
229 MatrixXd eigvec;
230
231 if (noiseCov.eig.size() > 0 && noiseCov.eigvec.size() > 0) {
232 eig = noiseCov.eig;
233 eigvec = noiseCov.eigvec;
234 } else {
235 // Compute eigendecomposition
236 SelfAdjointEigenSolver<MatrixXd> solver(noiseCov.data);
237 eig = solver.eigenvalues();
238 eigvec = solver.eigenvectors();
239 }
240
241 // Auto-detect rank from eigenvalue spectrum
242 if (rank <= 0) {
243 double maxEig = eig.maxCoeff();
244 double threshold = maxEig * 1e-10;
245 rank = 0;
246 for (int i = 0; i < eig.size(); ++i) {
247 if (eig(i) > threshold) ++rank;
248 }
249 if (rank == 0) rank = 1;
250 }
251
252 // Build whitening matrix: W = diag(1/sqrt(eig)) @ V^T
253 // Only use the top 'rank' eigenvalues
254 VectorXd invSqrtEig = VectorXd::Zero(eig.size());
255 int effectiveRank = 0;
256
257 // Eigenvalues are in ascending order — use last 'rank' values
258 for (int i = eig.size() - 1; i >= 0 && effectiveRank < rank; --i) {
259 if (eig(i) > 1e-30) {
260 invSqrtEig(i) = 1.0 / std::sqrt(eig(i));
261 ++effectiveRank;
262 }
263 }
264
265 MatrixXd whitener = invSqrtEig.asDiagonal() * eigvec.transpose();
266
267 return QPair<MatrixXd, int>(whitener, effectiveRank);
268}

272QPair<MatrixXd, VectorXd> INVLIB::computeSourcePsd(
273 const InvSourceEstimate& stc,
274 float sfreq,
275 float fmin,
276 float fmax,
277 int nFft)
278{
279 if (stc.isEmpty()) {
280 qWarning() << "[computeSourcePsd] Empty source estimate.";
281 return QPair<MatrixXd, VectorXd>();
282 }
283
284 const int nSources = static_cast<int>(stc.data.rows());
285 const int nTimes = static_cast<int>(stc.data.cols());
286
287 if (nFft <= 0) nFft = nTimes;
288 if (fmax < 0) fmax = sfreq / 2.0f;
289
290 const int nFreqs = nFft / 2 + 1;
291
292 // Build frequency vector
293 VectorXd freqs(nFreqs);
294 for (int f = 0; f < nFreqs; ++f) {
295 freqs(f) = static_cast<double>(f) * sfreq / nFft;
296 }
297
298 // Find frequency range indices
299 int fminIdx = 0, fmaxIdx = nFreqs - 1;
300 for (int f = 0; f < nFreqs; ++f) {
301 if (freqs(f) >= fmin) { fminIdx = f; break; }
302 }
303 for (int f = nFreqs - 1; f >= 0; --f) {
304 if (freqs(f) <= fmax) { fmaxIdx = f; break; }
305 }
306
307 int nBandFreqs = fmaxIdx - fminIdx + 1;
308 if (nBandFreqs <= 0) {
309 qWarning() << "[computeSourcePsd] No frequencies in range.";
310 return QPair<MatrixXd, VectorXd>();
311 }
312
313 // Compute PSD for each source
314 MatrixXd psd(nSources, nBandFreqs);
315
316 for (int s = 0; s < nSources; ++s) {
317 VectorXd fullPsd = computeRowPsd(stc.data.row(s).transpose(), nFft, sfreq);
318 psd.row(s) = fullPsd.segment(fminIdx, nBandFreqs).transpose();
319 }
320
321 VectorXd bandFreqs = freqs.segment(fminIdx, nBandFreqs);
322
323 return QPair<MatrixXd, VectorXd>(psd, bandFreqs);
324}

328QMap<QString, VectorXd> INVLIB::computeSourceBandPower(
329 const InvSourceEstimate& stc,
330 float sfreq,
331 const QMap<QString, QPair<float, float>>& bands)
332{
333 QMap<QString, VectorXd> result;
334
335 if (stc.isEmpty() || bands.isEmpty()) {
336 return result;
337 }
338
339 // Compute full PSD
340 auto [psd, freqs] = computeSourcePsd(stc, sfreq);
341 if (psd.size() == 0) return result;
342
343 const int nSources = static_cast<int>(psd.rows());
344 const int nFreqs = static_cast<int>(freqs.size());
345 double df = (nFreqs > 1) ? (freqs(1) - freqs(0)) : 1.0;
346
347 for (auto it = bands.constBegin(); it != bands.constEnd(); ++it) {
348 float bfmin = it.value().first;
349 float bfmax = it.value().second;
350
351 VectorXd bandPower = VectorXd::Zero(nSources);
352
353 for (int f = 0; f < nFreqs; ++f) {
354 if (freqs(f) >= bfmin && freqs(f) <= bfmax) {
355 bandPower += psd.col(f) * df;
356 }
357 }
358
359 result[it.key()] = bandPower;
360 }
361
362 return result;
363}
