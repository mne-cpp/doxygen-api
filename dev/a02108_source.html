138JsonObject MnaProvenance::to_json() const
139{
140 JsonObject json;
141 if (!input_hashes.empty()) {
142 JsonObject hashes;
143 for (const auto& [port, hash] : input_hashes) {
144 hashes.insert(port, hash);
145 }
146 json["input_hashes"] = std::move(hashes);
147 }
148 if (!resolved_attributes.empty())
149 json["resolved_attributes"] = resolved_attributes;
150 if (!mne_cpp_version.empty())
151 json["mne_cpp_version"] = mne_cpp_version;
152 if (!compiler_info.empty())
153 json["compiler_info"] = compiler_info;
154 if (!os_info.empty())
155 json["os_info"] = os_info;
156 if (!host_name.empty())
157 json["host_name"] = host_name;
158 if (!external_tool_version.empty())
159 json["external_tool_version"] = external_tool_version;
160 if (started_at)
161 json["started_at"] = format_iso8601(*started_at, true);
162 if (finished_at)
163 json["finished_at"] = format_iso8601(*finished_at, true);
164 if (wall_time_ms > 0)
165 json["wall_time_ms"] = wall_time_ms;
166 if (peak_memory_bytes > 0)
167 json["peak_memory_bytes"] = peak_memory_bytes;
168 if (random_seed >= 0)
169 json["random_seed"] = random_seed;
170 return json;
171}

175MnaProvenance MnaProvenance::from_json(const JsonObject& json)
176{
177 MnaProvenance p;
178 for (const auto& [port, hash] : json["input_hashes"].to_object()) {
179 p.input_hashes.emplace(port, hash.to_std_string());
180 }
181 p.resolved_attributes = json["resolved_attributes"].to_object();
182 p.mne_cpp_version = json["mne_cpp_version"].to_std_string();
183 p.compiler_info = json["compiler_info"].to_std_string();
184 p.os_info = json["os_info"].to_std_string();
185 p.host_name = json["host_name"].to_std_string();
186 p.external_tool_version = json["external_tool_version"].to_std_string();
187 p.started_at = parse_iso8601(json["started_at"].to_string());
188 p.finished_at = parse_iso8601(json["finished_at"].to_string());
189 p.wall_time_ms = static_cast<std::int64_t>(json["wall_time_ms"].to_double(0));
190 p.peak_memory_bytes = static_cast<std::int64_t>(json["peak_memory_bytes"].to_double(0));
191 p.random_seed = static_cast<std::int64_t>(json["random_seed"].to_double(-1));
192 return p;
193}

213JsonObject MnaVerification::to_json() const
214{
215 JsonObject json;
216 if (!explanation.empty()) {
217 json["explanation"] = explanation;
218 }
219 if (!checks.empty()) {
220 json["checks"] = detail::to_json_array(checks);
221 }
222 if (!pre_results.empty()) {
223 json["pre_results"] = detail::to_json_array(pre_results);
224 }
225 if (!post_results.empty()) {
226 json["post_results"] = detail::to_json_array(post_results);
227 }
228 if (JsonObject provenanceJson = provenance.to_json(); !provenanceJson.empty()) {
229 json["provenance"] = std::move(provenanceJson);
230 }
231 return json;
232}

236MnaVerification MnaVerification::from_json(const JsonObject& json)
237{
238 MnaVerification v;

240 v.checks = detail::from_json_array<MnaVerificationCheck>(json["checks"]);
241 v.pre_results = detail::from_json_array<MnaVerificationResult>(json["pre_results"]);
242 v.post_results = detail::from_json_array<MnaVerificationResult>(json["post_results"]);
243 if (json.contains("provenance")) {
244 v.provenance = MnaProvenance::from_json(json["provenance"].to_object());
245 }
246 return v;
