199bool MNEForwardSolution::write(QIODevice& p_IODevice) const
200{
201 //
202 // Classify channels into MEG and EEG index sets
203 //
204 std::vector<int> megIdx, eegIdx;
205 for (int k = 0; k < info.chs.size(); ++k) {
206 fiff_int_t kind = info.chs[k].kind;
207 if (kind == FIFFV_MEG_CH || kind == FIFFV_REF_MEG_CH)
208 megIdx.push_back(k);
209 else if (kind == FIFFV_EEG_CH)
210 eegIdx.push_back(k);
211 }
212 int nmeg = megIdx.size();
213 int neeg = eegIdx.size();
214
215 //
216 // Compute the total number of active source vertices
217 //
218 int nvert = 0;
219 for (int k = 0; k < src.size(); ++k)
220 nvert += src[k].nuse;
221
222 //
223 // Open the file, create the directory
224 //
225 FiffStream::SPtr t_pStream = FiffStream::start_file(p_IODevice);
226 t_pStream->start_block(FIFFB_MNE);
227
228 //
229 // Information from the MRI file
230 //
231 {
232 t_pStream->start_block(FIFFB_MNE_PARENT_MRI_FILE);
233
234 if (!mri_filename.isEmpty())
235 t_pStream->write_string(FIFF_MNE_FILE_NAME, mri_filename);
236 if (!mri_id.isEmpty())
237 t_pStream->write_id(FIFF_PARENT_FILE_ID, mri_id);
238 t_pStream->write_coord_trans(mri_head_t);
239
240 t_pStream->end_block(FIFFB_MNE_PARENT_MRI_FILE);
241 }
242
243 //
244 // Information from the measurement file
245 //
246 {
247 t_pStream->start_block(FIFFB_MNE_PARENT_MEAS_FILE);
248
249 if (!info.filename.isEmpty())
250 t_pStream->write_string(FIFF_MNE_FILE_NAME, info.filename);
251 if (!info.meas_id.isEmpty())
252 t_pStream->write_id(FIFF_PARENT_BLOCK_ID, info.meas_id);
253 t_pStream->write_coord_trans(info.dev_head_t);
254
255 int totalChan = nmeg + neeg;
256 t_pStream->write_int(FIFF_NCHAN, &totalChan);
257
258 // Write channel infos with sequential scanNo
259 QList<FiffChInfo> allChs;
260 for (int k = 0; k < nmeg; ++k)
261 allChs.append(info.chs[megIdx[k]]);
262 for (int k = 0; k < neeg; ++k)
263 allChs.append(info.chs[eegIdx[k]]);
264 for (int p = 0; p < allChs.size(); ++p) {
265 allChs[p].scanNo = p + 1;
266 t_pStream->write_ch_info(allChs[p]);
267 }
268
269 t_pStream->write_bad_channels(info.bads);
270
271 t_pStream->end_block(FIFFB_MNE_PARENT_MEAS_FILE);
272 }
273
274 //
275 // Write the source spaces
276 //
277 for (int k = 0; k < src.size(); ++k) {
278 if (src[k].writeToStream(t_pStream, false) == FIFF_FAIL) {
279 t_pStream->close();
280 return false;
281 }
282 }
283
284 //
285 // Extract sub-matrices for MEG and EEG from the combined sol
286 //
287 auto extractRows = [](const FiffNamedMatrix& combined,
288 const std::vector<int>& rowIdx) -> FiffNamedMatrix
289 {
290 int nRows = rowIdx.size();
291 int nCols = combined.ncol;
292 MatrixXd data(nRows, nCols);
293 QStringList row_names;
294 for (int r = 0; r < nRows; ++r) {
295 data.row(r) = combined.data.row(rowIdx[r]);
296 row_names.append(combined.row_names[rowIdx[r]]);
297 }
298 FiffNamedMatrix sub;
299 sub.nrow = nRows;
300 sub.ncol = nCols;
301 sub.row_names = row_names;
302 sub.col_names = combined.col_names;
303 sub.data = data;
304 return sub;
305 };
306
307 int ori_val = (source_ori == FIFFV_MNE_FIXED_ORI) ? FIFFV_MNE_FIXED_ORI : FIFFV_MNE_FREE_ORI;
308 int frame = coord_frame;
309
310 //
311 // MEG forward solution
312 //
313 if (nmeg > 0) {
314 t_pStream->start_block(FIFFB_MNE_FORWARD_SOLUTION);
315
316 int val = FIFFV_MNE_MEG;
317 t_pStream->write_int(FIFF_MNE_INCLUDED_METHODS, &val);
318 t_pStream->write_int(FIFF_MNE_COORD_FRAME, &frame);
319 t_pStream->write_int(FIFF_MNE_SOURCE_ORIENTATION, &ori_val);
320 t_pStream->write_int(FIFF_MNE_SOURCE_SPACE_NPOINTS, &nvert);
321 t_pStream->write_int(FIFF_NCHAN, &nmeg);
322
323 FiffNamedMatrix megSol = extractRows(*sol.data(), megIdx);
324 megSol.transpose_named_matrix();
325 t_pStream->write_named_matrix(FIFF_MNE_FORWARD_SOLUTION, megSol);
326
327 if (!sol_grad->isEmpty()) {
328 FiffNamedMatrix megGrad = extractRows(*sol_grad.data(), megIdx);
329 megGrad.transpose_named_matrix();
330 t_pStream->write_named_matrix(FIFF_MNE_FORWARD_SOLUTION_GRAD, megGrad);
331 }
332 t_pStream->end_block(FIFFB_MNE_FORWARD_SOLUTION);
333 }
334
335 //
336 // EEG forward solution
337 //
338 if (neeg > 0) {
339 t_pStream->start_block(FIFFB_MNE_FORWARD_SOLUTION);
340
341 int val = FIFFV_MNE_EEG;
342 t_pStream->write_int(FIFF_MNE_INCLUDED_METHODS, &val);
343 t_pStream->write_int(FIFF_MNE_COORD_FRAME, &frame);
344 t_pStream->write_int(FIFF_MNE_SOURCE_ORIENTATION, &ori_val);
345 t_pStream->write_int(FIFF_NCHAN, &neeg);
346 t_pStream->write_int(FIFF_MNE_SOURCE_SPACE_NPOINTS, &nvert);
347
348 FiffNamedMatrix eegSol = extractRows(*sol.data(), eegIdx);
349 eegSol.transpose_named_matrix();
350 t_pStream->write_named_matrix(FIFF_MNE_FORWARD_SOLUTION, eegSol);
351
352 if (!sol_grad->isEmpty()) {
353 FiffNamedMatrix eegGrad = extractRows(*sol_grad.data(), eegIdx);
354 eegGrad.transpose_named_matrix();
355 t_pStream->write_named_matrix(FIFF_MNE_FORWARD_SOLUTION_GRAD, eegGrad);
356 }
357 t_pStream->end_block(FIFFB_MNE_FORWARD_SOLUTION);
358 }
359
360 t_pStream->end_block(FIFFB_MNE);
361 t_pStream->end_file();
362 t_pStream->close();
363 t_pStream.clear();
364
365 //
366 // Update the directory
367 //
368 if (auto* qf = dynamic_cast<QFile*>(&p_IODevice)) {
369 QFile fileIn(qf->fileName());
370 FiffStream::SPtr t_pStreamIn = FiffStream::open_update(fileIn);
371 if (t_pStreamIn) {
372 const auto& dir = t_pStreamIn->dir();
373 for (int i = 0; i < dir.size(); ++i) {
374 if (dir[i]->kind == FIFF_DIR_POINTER) {
375 fiff_int_t dirpos = (fiff_int_t)t_pStreamIn->write_dir_entries(dir);
376 if (dirpos >= 0)
377 t_pStreamIn->write_dir_pointer(dirpos, dir[i]->pos);
378 break;
379 }
380 }
381 t_pStreamIn->close();
382 }
383 }
384
385 return true;
386}

678 //
679 qint32 totalNumOfClust = 0;
680 for (qint32 h = 0; h < 2; ++h)
681 totalNumOfClust += p_fwdOut.src.hemisphereAt(h)->cluster_info.clusterVertnos.size();
682
683 if(this->isFixedOrient())
684 p_D = MatrixXd::Zero(this->sol->data.cols(), totalNumOfClust);
685 else
686 p_D = MatrixXd::Zero(this->sol->data.cols(), totalNumOfClust*3);
687
688 QList<VectorXi> t_vertnos = this->src.get_vertno();
689
690 qint32 currentCluster = 0;
691 for (qint32 h = 0; h < 2; ++h)
692 {
693 int hemiOffset = h == 0 ? 0 : t_vertnos[0].size();
694 for(qint32 i = 0; i < p_fwdOut.src.hemisphereAt(h)->cluster_info.clusterVertnos.size(); ++i)
695 {
696 VectorXi idx_sel;
697 MNEMath::intersect(t_vertnos[h], p_fwdOut.src.hemisphereAt(h)->cluster_info.clusterVertnos[i], idx_sel);
698
699 idx_sel.array() += hemiOffset;
700
701 double selectWeight = 1.0/idx_sel.size();
702 if(this->isFixedOrient())
703 {
704 for(qint32 j = 0; j < idx_sel.size(); ++j)
705 p_D.col(currentCluster)[idx_sel(j)] = selectWeight;
706 }
707 else
708 {
709 qint32 clustOffset = currentCluster*3;
710 for(qint32 j = 0; j < idx_sel.size(); ++j)
711 {
712 qint32 idx_sel_Offset = idx_sel(j)*3;
713 //x
714 p_D(idx_sel_Offset,clustOffset) = selectWeight;
715 //y
716 p_D(idx_sel_Offset+1, clustOffset+1) = selectWeight;
717 //z
718 p_D(idx_sel_Offset+2, clustOffset+2) = selectWeight;
719 }
720 }
721 ++currentCluster;
722 }
723 }
724
725 //
726 // Put it all together
727 //
728 p_fwdOut.sol->data = t_G_new;
729 p_fwdOut.sol->ncol = t_G_new.cols();
730
731 p_fwdOut.nsource = p_fwdOut.sol->ncol/3;
732
733 return p_fwdOut;
734}

737
738MNEForwardSolution MNEForwardSolution::reduce_forward_solution(qint32 p_iNumDipoles, MatrixXd& p_D) const
739{
740 MNEForwardSolution p_fwdOut = MNEForwardSolution(*this);
741
742 bool isFixed = p_fwdOut.isFixedOrient();
743 qint32 np = isFixed ? p_fwdOut.sol->data.cols() : p_fwdOut.sol->data.cols()/3;
744
745 if(p_iNumDipoles > np)
746 return p_fwdOut;
747
748 VectorXi sel(p_iNumDipoles);
749
750 float t_fStep = static_cast<float>(np) / static_cast<float>(p_iNumDipoles);
751
752 for(qint32 i = 0; i < p_iNumDipoles; ++i)
753 {
754 float t_fCurrent = static_cast<float>(i) * t_fStep;
755 sel[i] = (quint32)floor(t_fCurrent);
756 }
757
758 if(isFixed)
759 {
760 p_D = MatrixXd::Zero(p_fwdOut.sol->data.cols(), p_iNumDipoles);
761 for(qint32 i = 0; i < p_iNumDipoles; ++i)
762 p_D(sel[i], i) = 1;
763 }
764 else
765 {
766 p_D = MatrixXd::Zero(p_fwdOut.sol->data.cols(), p_iNumDipoles*3);
767 for(qint32 i = 0; i < p_iNumDipoles; ++i)
768 for(qint32 j = 0; j < 3; ++j)
769 p_D((sel[i]*3)+j, (i*3)+j) = 1;
770 }
771
772 // New gain matrix
773 p_fwdOut.sol->data = this->sol->data * p_D;
774
775 MatrixX3f rr(p_iNumDipoles,3);
776
777 MatrixX3f nn(p_iNumDipoles,3);
778
779 for(qint32 i = 0; i < p_iNumDipoles; ++i)
780 {
781 rr.row(i) = this->source_rr.row(sel(i));
782 nn.row(i) = this->source_nn.row(sel(i));
783 }
784
785 p_fwdOut.source_rr = rr;
786 p_fwdOut.source_nn = nn;
787
788 p_fwdOut.sol->ncol = p_fwdOut.sol->data.cols();
789
790 p_fwdOut.nsource = p_iNumDipoles;
791
792 return p_fwdOut;
793}

797FiffCov MNEForwardSolution::compute_depth_prior(const MatrixXd &Gain, const FiffInfo &gain_info, bool is_fixed_ori, double exp, double limit, const MatrixXd &patch_areas, bool limit_depth_chs)
798{
799 qInfo("\tCreating the depth weighting matrix...");
800
801 MatrixXd G(Gain);
802 // If possible, pick best depth-weighting channels
803 if(limit_depth_chs)
804 MNEForwardSolution::restrict_gain_matrix(G, gain_info);
805
806 VectorXd d;
807 // Compute the gain matrix
808 if(is_fixed_ori)
809 {
810 d = (G.array().square()).rowwise().sum();
811 }
812 else
813 {
814 qint32 n_pos = G.cols() / 3;
815 d = VectorXd::Zero(n_pos);
816 MatrixXd Gk;
817 for (qint32 k = 0; k < n_pos; ++k)
818 {
819 Gk = G.block(0,3*k, G.rows(), 3);
820 JacobiSVD<MatrixXd> svd(Gk.transpose()*Gk);
821 d[k] = svd.singularValues().maxCoeff();
822 }
823 }
824
825 // ToDo Currently the fwd solns never have "patch_areas" defined
826 if(patch_areas.cols() > 0)
827 {
828 qWarning("\tToDo!!!!! >>> Patch areas taken into account in the depth weighting");
829 }
830
831 qint32 n_limit;
832 VectorXd w = d.cwiseInverse();
833 VectorXd ws = w;
834 VectorXd wpp;
835 MNEMath::sort<double>(ws, false);
836 double weight_limit = pow(limit, 2);
837 if (!limit_depth_chs)
838 {
839 // match old mne-python behavor
840 qint32 ind = 0;
841 ws.minCoeff(&ind);
842 n_limit = ind;
843 limit = ws[ind] * weight_limit;
844 }
845 else
846 {
847 // match C code behavior
848 limit = ws[ws.size()-1];
849 qint32 ind = 0;
850 n_limit = d.size();
851 if (ws[ws.size()-1] > weight_limit * ws[0])
852 {
853 double th = weight_limit * ws[0];
854 for(qint32 i = 0; i < ws.size(); ++i)
855 {
856 if(ws[i] > th)
857 {
858 ind = i;
859 break;
860 }
861 }
862 limit = ws[ind];
863 n_limit = ind;
864 }
865 }
866
867 qInfo("\tlimit = %d/%ld = %f", n_limit + 1, d.size(), sqrt(limit / ws[0]));
868 double scale = 1.0 / limit;
869 qInfo("\tscale = %g exp = %g", scale, exp);
870
871 VectorXd t_w = w.array() / limit;
872 for(qint32 i = 0; i < t_w.size(); ++i)
873 t_w[i] = t_w[i] > 1 ? 1 : t_w[i];
874 wpp = t_w.array().pow(exp);
875
876 FiffCov depth_prior;
877 if(is_fixed_ori)
878 depth_prior.data = wpp;
879 else
880 {
881 depth_prior.data.resize(wpp.rows()*3, 1);
882 qint32 idx = 0;
883 double v;
884 for(qint32 i = 0; i < wpp.rows(); ++i)
885 {
886 idx = i*3;
887 v = wpp[i];
888 depth_prior.data(idx, 0) = v;
889 depth_prior.data(idx+1, 0) = v;
890 depth_prior.data(idx+2, 0) = v;
891 }
892 }
893
894 depth_prior.kind = FIFFV_MNE_DEPTH_PRIOR_COV;
895 depth_prior.diag = true;
896 depth_prior.dim = depth_prior.data.rows();
897 depth_prior.nfree = 1;
898
899 return depth_prior;
900}

904FiffCov MNEForwardSolution::compute_orient_prior(float loose)
905{
906 bool is_fixed_ori = this->isFixedOrient();
907 qint32 n_sources = this->sol->data.cols();
908
909 if (0 <= loose && loose <= 1)
910 {
911 qDebug() << "this->surf_ori" << this->surf_ori;
912 if(loose < 1 && !this->surf_ori)
913 {
914 qWarning("\tForward operator is not oriented in surface coordinates. loose parameter should be None not %f.", loose);
915 loose = 1;
916 qInfo("\tSetting loose to %f.", loose);
917 }
918
919 if(is_fixed_ori)
920 {
921 qInfo("\tIgnoring loose parameter with forward operator with fixed orientation.");
922 loose = 0.0;
923 }
924 }
925 else
926 {
927 if(loose < 0 || loose > 1)
928 {
929 qWarning("Warning: Loose value should be in interval [0,1] not %f.\n", loose);
930 loose = loose > 1 ? 1 : 0;
931 qInfo("Setting loose to %f.", loose);
932 }
933 }
934
935 FiffCov orient_prior;
936 orient_prior.data = VectorXd::Ones(n_sources);
937 if(!is_fixed_ori && (0 <= loose && loose <= 1))
938 {
939 qInfo("\tApplying loose dipole orientations. Loose value of %f.", loose);
940 for(qint32 i = 0; i < n_sources; i+=3)
941 orient_prior.data.block(i,0,2,1).array() *= loose;
942
943 orient_prior.kind = FIFFV_MNE_ORIENT_PRIOR_COV;
944 orient_prior.diag = true;
945 orient_prior.dim = orient_prior.data.size();
946 orient_prior.nfree = 1;
947 }
948 return orient_prior;
949}

1080void MNEForwardSolution::prepare_forward(const FiffInfo &p_info,
1081 const FiffCov &p_noise_cov,
1082 bool p_pca,
1083 FiffInfo &p_outFwdInfo,
1084 MatrixXd &gain,
1085 FiffCov &p_outNoiseCov,
1086 MatrixXd &p_outWhitener,
1087 qint32 &p_outNumNonZero) const
1088{
1089 QStringList fwd_ch_names, ch_names;
1090 for(qint32 i = 0; i < this->info.chs.size(); ++i)
1091 fwd_ch_names << this->info.chs[i].ch_name;
1092
1093 ch_names.clear();
1094 for(qint32 i = 0; i < p_info.chs.size(); ++i)
1095 if(!p_info.bads.contains(p_info.chs[i].ch_name)
1096 && !p_noise_cov.bads.contains(p_info.chs[i].ch_name)
1097 && p_noise_cov.names.contains(p_info.chs[i].ch_name)
1098 && fwd_ch_names.contains(p_info.chs[i].ch_name))
1099 ch_names << p_info.chs[i].ch_name;
1100
1101 qint32 n_chan = ch_names.size();
1102 qInfo("Computing inverse operator with %d channels.", n_chan);
1103
1104 //
1105 // Handle noise cov
1106 //
1107 p_outNoiseCov = p_noise_cov.prepare_noise_cov(p_info, ch_names);
1108
1109 // Omit the zeroes due to projection
1110 p_outNumNonZero = 0;
1111 VectorXi t_vecNonZero = VectorXi::Zero(n_chan);
1112 for(qint32 i = 0; i < p_outNoiseCov.eig.rows(); ++i)
1113 {
1114 if(p_outNoiseCov.eig[i] > 0)
1115 {
1116 t_vecNonZero[p_outNumNonZero] = i;
1117 ++p_outNumNonZero;
1118 }
1119 }
1120 if(p_outNumNonZero > 0)
1121 t_vecNonZero.conservativeResize(p_outNumNonZero);
1122
1123 if(p_outNumNonZero > 0)
1124 {
1125 if (p_pca)
1126 {
1127 qWarning("Warning in MNEForwardSolution::prepare_forward: if (p_pca) havent been debugged.");
1128 p_outWhitener = MatrixXd::Zero(n_chan, p_outNumNonZero);
1129 // Rows of eigvec are the eigenvectors
1130 for(qint32 i = 0; i < p_outNumNonZero; ++i)
1131 p_outWhitener.col(t_vecNonZero[i]) = p_outNoiseCov.eigvec.col(t_vecNonZero[i]).array() / sqrt(p_outNoiseCov.eig(t_vecNonZero[i]));
1132 qInfo("\tReducing data rank to %d.", p_outNumNonZero);
1133 }
1134 else
1135 {
1136 qInfo("Creating non pca whitener.");
1137 p_outWhitener = MatrixXd::Zero(n_chan, n_chan);
1138 for(qint32 i = 0; i < p_outNumNonZero; ++i)
1139 p_outWhitener(t_vecNonZero[i],t_vecNonZero[i]) = 1.0 / sqrt(p_outNoiseCov.eig(t_vecNonZero[i]));
1140 // Cols of eigvec are the eigenvectors
1141 p_outWhitener *= p_outNoiseCov.eigvec;
1142 }
1143 }
1144
1145 VectorXi fwd_idx = VectorXi::Zero(ch_names.size());
1146 VectorXi info_idx = VectorXi::Zero(ch_names.size());
1147 qint32 idx;
1148 qint32 count_fwd_idx = 0;
1149 qint32 count_info_idx = 0;
1150 for(qint32 i = 0; i < ch_names.size(); ++i)
1151 {
1152 idx = fwd_ch_names.indexOf(ch_names[i]);
1153 if(idx > -1)
1154 {
1155 fwd_idx[count_fwd_idx] = idx;
1156 ++count_fwd_idx;
1157 }
1158 idx = p_info.ch_names.indexOf(ch_names[i]);
1159 if(idx > -1)
1160 {
1161 info_idx[count_info_idx] = idx;
1162 ++count_info_idx;
1163 }
1164 }
1165 fwd_idx.conservativeResize(count_fwd_idx);
1166 info_idx.conservativeResize(count_info_idx);
1167
1168 gain.resize(count_fwd_idx, this->sol->data.cols());
1169 for(qint32 i = 0; i < count_fwd_idx; ++i)
1170 gain.row(i) = this->sol->data.row(fwd_idx[i]);
1171
1172 p_outFwdInfo = p_info.pick_info(info_idx);
1173
1174 qInfo("\tTotal rank is %d", p_outNumNonZero);
1175}

1178
1179bool MNEForwardSolution::read(QIODevice& p_IODevice,
1180 MNEForwardSolution& fwd,
1181 bool force_fixed,
1182 bool surf_ori,
1183 const QStringList& include,
1184 const QStringList& exclude,
1185 bool bExcludeBads)

1188
1189 qInfo("Reading forward solution from %s...", t_pStream->streamName().toUtf8().constData());
1190 if(!t_pStream->open())
1191 return false;
1192 //
1193 // Find all forward solutions
1194 //
1195 QList<FiffDirNode::SPtr> fwds = t_pStream->dirtree()->dir_tree_find(FIFFB_MNE_FORWARD_SOLUTION);
1196
1197 if (fwds.size() == 0)
1198 {
1199 t_pStream->close();
1200 qWarning("No forward solutions in %s", t_pStream->streamName().toUtf8().constData());
1201 return false;
1202 }
1203 //
1204 // Parent MRI data
1205 //
1206 QList<FiffDirNode::SPtr> parent_mri = t_pStream->dirtree()->dir_tree_find(FIFFB_MNE_PARENT_MRI_FILE);
1207 if (parent_mri.size() == 0)
1208 {
1209 t_pStream->close();
1210 qWarning("No parent MRI information in %s", t_pStream->streamName().toUtf8().constData());
1211 return false;
1212 }
1213
1214 MNELIB::MNESourceSpaces t_SourceSpace;
1215 if(!MNELIB::MNESourceSpaces::readFromStream(t_pStream, true, t_SourceSpace))
1216 {
1217 t_pStream->close();
1218 qWarning("Could not read the source spaces");
1219 //ToDo error(me,'Could not read the source spaces (%s)',mne_omit_first_line(lasterr));
1220 return false;
1221 }
1222
1223 for(qint32 k = 0; k < t_SourceSpace.size(); ++k)
1224 t_SourceSpace[k].id = t_SourceSpace[k].find_source_space_hemi();
1225
1226 //
1227 // Bad channel list
1228 //
1229 QStringList bads;
1230 if(bExcludeBads)
1231 {
1232 bads = t_pStream->read_bad_channels(t_pStream->dirtree());
1233 if(bads.size() > 0)
1234 {
1235 qInfo("\t%lld bad channels ( ", bads.size());
1236 for(qint32 i = 0; i < bads.size(); ++i)
1237 qInfo("\"%s\" ", bads[i].toUtf8().constData());
1238 qInfo(") read");
1239 }
1240 }
1241
1242 //
1243 // Locate and read the forward solutions
1244 //
1245 FiffTag::SPtr t_pTag;
1246 FiffDirNode::SPtr megnode;
1247 FiffDirNode::SPtr eegnode;
1248 for(qint32 k = 0; k < fwds.size(); ++k)
1249 {
1250 if(!fwds[k]->find_tag(t_pStream, FIFF_MNE_INCLUDED_METHODS, t_pTag))
1251 {
1252 t_pStream->close();
1253 qWarning("Methods not listed for one of the forward solutions");
1254 return false;
1255 }
1256 if (*t_pTag->toInt() == FIFFV_MNE_MEG)
1257 {
1258 qInfo("MEG solution found");
1259 megnode = fwds[k];
1260 }
1261 else if(*t_pTag->toInt() == FIFFV_MNE_EEG)
1262 {
1263 qInfo("EEG solution found");
1264 eegnode = fwds[k];
1265 }
1266 }
1267
1268 MNEForwardSolution megfwd;
1269 QString ori;
1270 if (read_one(t_pStream, megnode, megfwd))
1271 {
1272 if (megfwd.source_ori == FIFFV_MNE_FIXED_ORI)
1273 ori = QString("fixed");
1274 else
1275 ori = QString("free");
1276 qInfo("\tRead MEG forward solution (%d sources, %d channels, %s orientations)", megfwd.nsource,megfwd.nchan,ori.toUtf8().constData());
1277 }
1278 MNEForwardSolution eegfwd;
1279 if (read_one(t_pStream, eegnode, eegfwd))
1280 {
1281 if (eegfwd.source_ori == FIFFV_MNE_FIXED_ORI)
1282 ori = QString("fixed");
1283 else
1284 ori = QString("free");
1285 qInfo("\tRead EEG forward solution (%d sources, %d channels, %s orientations)", eegfwd.nsource,eegfwd.nchan,ori.toUtf8().constData());
1286 }
1287
1288 //
1289 // Merge the MEG and EEG solutions together
1290 //
1291 fwd.clear();
1292
1293 if (!megfwd.isEmpty() && !eegfwd.isEmpty())
1294 {
1295 if (megfwd.sol->data.cols() != eegfwd.sol->data.cols() ||
1296 megfwd.source_ori != eegfwd.source_ori ||
1297 megfwd.nsource != eegfwd.nsource ||
1298 megfwd.coord_frame != eegfwd.coord_frame)
1299 {
1300 t_pStream->close();
1301 qWarning("The MEG and EEG forward solutions do not match");
1302 return false;
1303 }
