256template<typename T>
257void Numerics::histcounts(const Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& matRawData,
258 bool bMakeSymmetrical,
259 int iClassAmount,
260 Eigen::VectorXd& vecResultClassLimits,
261 Eigen::VectorXi& vecResultFrequency,
262 double dGlobalMin,
263 double dGlobalMax)
264{
265 if (matRawData.rows() == 0 || matRawData.cols() == 0) {
266 return;
267 }
268
269 vecResultClassLimits.resize(iClassAmount + 1);
270 vecResultFrequency.resize(iClassAmount);
271
272 for (int count = 0; count < iClassAmount; ++count) {
273 vecResultFrequency(count) = 0;
274 }
275
276 double desiredMin,
277 desiredMax;
278 double rawMin(0.0),
279 rawMax(0.0),
280 localMin(0.0),
281 localMax(0.0);
282
283 rawMin = matRawData.minCoeff();
284 rawMax = matRawData.maxCoeff();
285
286 if (bMakeSymmetrical == true) {
287 if (std::fabs(rawMin) > rawMax) {
288 localMax = std::fabs(rawMin);
289 localMin = rawMin;
290 } else if (rawMax > std::fabs(rawMin)) {
291 localMin = -(rawMax);
292 localMax = rawMax;
293 } else {
294 localMin = rawMin;
295 localMax = rawMax;
296 }
297 } else {
298 localMin = rawMin;
299 localMax = rawMax;
300 }
301
302 if (dGlobalMin == 0.0 && dGlobalMax == 0.0) {
303 desiredMin = localMin;
304 desiredMax = localMax;
305 vecResultClassLimits[0] = desiredMin;
306 vecResultClassLimits[iClassAmount] = desiredMax;
307 } else {
308 desiredMin = dGlobalMin;
309 desiredMax = dGlobalMax;
310 vecResultClassLimits(0) = desiredMin;
311 vecResultClassLimits(iClassAmount) = desiredMax;
312 }
313
314 double range = (vecResultClassLimits(iClassAmount) - vecResultClassLimits(0)),
315 dynamicUpperClassLimit;
316
317 for (int kr = 0; kr < iClassAmount; ++kr) {
318 dynamicUpperClassLimit = (vecResultClassLimits(0) + (kr * (range / iClassAmount)));
319 vecResultClassLimits(kr) = dynamicUpperClassLimit;
320 }
321
322 for (int ir = 0; ir < matRawData.rows(); ++ir) {
323 for (int jr = 0; jr < matRawData.cols(); ++jr) {
324 if (matRawData(ir, jr) != 0.0) {
325 for (int kr = 0; kr < iClassAmount; ++kr) {
326 if (kr == iClassAmount - 1) {
327 if (matRawData(ir, jr) >= vecResultClassLimits(kr) && matRawData(ir, jr) <= vecResultClassLimits(kr + 1)) {
328 vecResultFrequency(kr) = vecResultFrequency(kr) + 1;
329 }
330 } else {
331 if (matRawData(ir, jr) >= vecResultClassLimits(kr) && matRawData(ir, jr) < vecResultClassLimits(kr + 1)) {
332 vecResultFrequency(kr) = vecResultFrequency(kr) + 1;
333 }
334 }
335 }
336 }
337 }
338 }
339}
