54void FilterPlot::setFilter(const FilterKernel& filter)
55{
56 const Eigen::RowVectorXcd& spectrum = filter.fft_coefficients();
57 if (filter.coefficients().cols() == 0 || spectrum.cols() == 0) {
58 return;
59 }
60 std::vector<double> magnitude(static_cast<std::size_t>(spectrum.cols()));
61 for (Eigen::Index i = 0; i < spectrum.cols(); ++i) {
62 magnitude[static_cast<std::size_t>(i)] = std::abs(spectrum(i));
63 }
64 const double nyquist = filter.sampling_frequency() / 2.0;
65 std::vector<double> cutoffs;
66 for (double hz : {filter.highpass_freq(), filter.lowpass_freq()}) {
67 if (hz > 0.0 && hz < nyquist) {
68 cutoffs.push_back(hz);
69 }
70 }
71 m_plot.set_response(std::move(magnitude), nyquist);
72 m_plot.set_cutoffs(std::move(cutoffs));
73 m_plot.set_title(filter.name());
74 update();
75}
