37Color Brush::color_at(double t) const
38{
39 if (stops.empty()) {
40 return color;
41 }
42 t = std::clamp(t, 0.0, 1.0);
43 if (t <= stops.front().position) {
44 return stops.front().color;
45 }
46 for (std::size_t i = 1; i < stops.size(); ++i) {
47 const GradientStop& a = stops[i - 1];
48 const GradientStop& b = stops[i];
49 if (t <= b.position) {
50 const double span = b.position - a.position;
51 const double f = span > 0.0 ? (t - a.position) / span : 1.0;
52 const auto mix = [f](std::uint8_t x, std::uint8_t y) {
53 return static_cast<std::uint8_t>(std::lround(x + (y - x) * f));
54 };
55 return {mix(a.color.r, b.color.r), mix(a.color.g, b.color.g), mix(a.color.b, b.color.b), mix(a.color.a, b.color.a)};
56 }
57 }
58 return stops.back().color;
59}
