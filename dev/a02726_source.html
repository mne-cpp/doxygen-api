109std::unique_ptr<MNESssData> MNESssData::read_from_node(FiffStream& stream, const std::shared_ptr<FiffDirNode>& start)
110{
111 auto s = std::make_unique<MNESssData>();
112 std::vector<std::shared_ptr<FiffDirNode>> sss;
113 std::shared_ptr<FiffDirNode> node;
114 std::optional<FiffTag> t_pTag;
115 const float* r0;
116 int j, p, q, n;
117 /*
118 * Locate the SSS information
119 */
120 sss = start->dir_tree_find(FIFFB_SSS_INFO);
121 if (sss.size() > 0) {
122 node = sss[0];
123 /*
124 * Read the SSS information, require all tags to be present
125 */
126 if (t_pTag = node->find_tag(stream, FIFF_SSS_JOB); !t_pTag) {
127 return nullptr;
128 }
129 s->job = *t_pTag->to_int();
130
131 if (t_pTag = node->find_tag(stream, FIFF_SSS_FRAME); !t_pTag) {
132 return nullptr;
133 }
134 s->coord_frame = *t_pTag->to_int();
135
136 if (t_pTag = node->find_tag(stream, FIFF_SSS_ORIGIN); !t_pTag) {
137 return nullptr;
138 }
139 r0 = t_pTag->to_float();
140 Eigen::Map<Eigen::Vector3f>(s->origin) = Eigen::Map<const Eigen::Vector3f>(r0);
141
142 if (t_pTag = node->find_tag(stream, FIFF_SSS_ORD_IN); !t_pTag) {
143 return nullptr;
144 }
145 s->in_order = *t_pTag->to_int();
146
147 if (t_pTag = node->find_tag(stream, FIFF_SSS_ORD_OUT); !t_pTag) {
148 return nullptr;
149 }
150 s->out_order = *t_pTag->to_int();
151
152 if (t_pTag = node->find_tag(stream, FIFF_SSS_NMAG); !t_pTag) {
153 return nullptr;
154 }
155 s->nchan = *t_pTag->to_int();
156
157 if (t_pTag = node->find_tag(stream, FIFF_SSS_COMPONENTS); !t_pTag) {
158 return nullptr;
159 }
160 {
161 int ncomp = static_cast<int>(t_pTag->size() / sizeof(fiff_int_t));
162 const int* raw = t_pTag->to_int();
163 s->comp_info = Eigen::VectorXi::Map(raw, ncomp);
164
165 if (ncomp != (s->in_order * (2 + s->in_order) + s->out_order * (2 + s->out_order))) {
166 mne::Log::critical("Number of SSS components does not match the expansion orders listed in the file");
167 return nullptr;
168 }
169 }
170 /*
171 * Count the components in use
172 */
173 for (j = 0, n = 3, p = 0; j < s->in_order; j++, n = n + 2) {
174 for (q = 0; q < n; q++, p++)
175 if (s->comp_info[p])
176 s->in_nuse++;
177 }
178 for (j = 0, n = 3; j < s->out_order; j++, n = n + 2) {
179 for (q = 0; q < n; q++, p++)
180 s->out_nuse++;
181 }
182 }
183 /*
184 * There it is!
185 */
186 return s;
187}

205{
206 static frameNameRec_1 frames[] = {
207 {FIFFV_COORD_UNKNOWN, "unknown"},
208 {FIFFV_COORD_DEVICE, "MEG device"},
209 {FIFFV_COORD_ISOTRAK, "isotrak"},
210 {FIFFV_COORD_HPI, "hpi"},
211 {FIFFV_COORD_HEAD, "head"},
212 {FIFFV_COORD_MRI, "MRI (surface RAS)"},
213 {FIFFV_MNE_COORD_MRI_VOXEL, "MRI voxel"},
214 {FIFFV_COORD_MRI_SLICE, "MRI slice"},
215 {FIFFV_COORD_MRI_DISPLAY, "MRI display"},
216 {FIFFV_MNE_COORD_CTF_DEVICE, "CTF MEG device"},
217 {FIFFV_MNE_COORD_CTF_HEAD, "CTF/4D/KIT head"},
218 {FIFFV_MNE_COORD_RAS, "RAS (non-zero origin)"},
219 {FIFFV_MNE_COORD_MNI_TAL, "MNI Talairach"},
220 {FIFFV_MNE_COORD_FS_TAL_GTZ, "Talairach (MNI z > 0)"},
221 {FIFFV_MNE_COORD_FS_TAL_LTZ, "Talairach (MNI z < 0)"},
222 {-1, "unknown"}};
223 int k;
224 for (k = 0; frames[k].frame != -1; k++) {
225 if (frame == frames[k].frame)
226 return frames[k].name;
227 }
228 return frames[k].name;

233void MNESssData::print(std::ostream& out) const
234{
235 int j, p, q, n;
236
237 out << "job : " << this->job << "\n";
238 out << "coord frame : " << mne_coord_frame_name_1(this->coord_frame) << "\n";
239 out << std::format("origin : {:6.1f} {:6.1f} {:6.1f} mm\n", 1000 * this->origin[0], 1000 * this->origin[1], 1000 * this->origin[2]);
240 out << "in order : " << this->in_order << "\n";
241 out << "out order : " << this->out_order << "\n";
242 out << "nchan : " << this->nchan << "\n";
243 out << "ncomp : " << this->comp_info.size() << "\n";
244 out << "in nuse : " << this->in_nuse << "\n";
245 out << "out nuse : " << this->out_nuse << "\n";
246 out << "comps : ";
247 /*
248 * This produces the same output as maxfilter
249 */
250 for (j = 0, n = 3, p = 0; j < this->in_order; j++, n = n + 2) {
251 if (j > 0)
252 out << ";";
253 for (q = 0; q < n; q++, p++)
254 out << this->comp_info[p];
255 }
256 out << "//";
257 for (j = 0, n = 3; j < this->out_order; j++, n = n + 2) {
258 if (j > 0)
259 out << ";";
260 for (q = 0; q < n; q++, p++)
261 out << this->comp_info[p];
262 }
263 out << "\n";
264 return;
265}
