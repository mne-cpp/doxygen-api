328template<typename T>
329Eigen::VectorXi Linalg::sort(Eigen::Matrix<T, Eigen::Dynamic, 1>& v,
330 bool desc)
331{
332 std::vector<std::pair<int, T>> t_vecIdxValue;
333 Eigen::VectorXi idx(v.size());
334
335 if (v.size() > 0) {
336 for (std::int32_t i = 0; i < v.size(); ++i)
337 t_vecIdxValue.push_back(std::pair<int, T>(i, v[i]));
338
339 if (desc)
340 std::sort(t_vecIdxValue.begin(), t_vecIdxValue.end(), Linalg::compareIdxValuePairBiggerThan<T>);
341 else
342 std::sort(t_vecIdxValue.begin(), t_vecIdxValue.end(), Linalg::compareIdxValuePairSmallerThan<T>);
343
344 for (std::int32_t i = 0; i < v.size(); ++i) {
345 idx[i] = t_vecIdxValue[i].first;
346 v[i] = t_vecIdxValue[i].second;
347 }
348 }
349
350 return idx;
351}

355template<typename T>
356Eigen::VectorXi Linalg::sort(Eigen::Matrix<T, Eigen::Dynamic, 1>& v_prime,
357 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& mat,
358 bool desc)
359{
360 Eigen::VectorXi idx = Linalg::sort<T>(v_prime, desc);
361
362 if (v_prime.size() > 0) {
363 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic> newMat(mat.rows(), mat.cols());
364 for (std::int32_t i = 0; i < idx.size(); ++i)
365 newMat.col(i) = mat.col(idx[i]);
366 mat = newMat;
367 }
368
369 return idx;
370}

374template<typename T>
375std::vector<Eigen::Triplet<T>> Linalg::sortrows(const std::vector<Eigen::Triplet<T>>& A,
376 std::int32_t column)
377{
378 std::vector<Eigen::Triplet<T>> p_ASorted;
379
380 for (std::uint32_t i = 0; i < A.size(); ++i)
381 p_ASorted.push_back(A[i]);
382
383 if (column == 0)
384 std::sort(p_ASorted.begin(), p_ASorted.end(), Linalg::compareTripletFirstEntry<T>);
385 if (column == 1)
386 std::sort(p_ASorted.begin(), p_ASorted.end(), Linalg::compareTripletSecondEntry<T>);
387
388 return p_ASorted;
389}

429template<typename T>
430Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic> Linalg::pinv(const Eigen::Matrix<T,
431 Eigen::Dynamic,
432 Eigen::Dynamic>& a)
433{
434 double epsilon = std::numeric_limits<double>::epsilon();
435 Eigen::JacobiSVD<Eigen::MatrixXd> svd(a, Eigen::ComputeThinU | Eigen::ComputeThinV);
436 double tolerance = epsilon * static_cast<double>(std::max(a.cols(), a.rows())) * svd.singularValues().array().abs()(0);
437 return svd.matrixV() * (svd.singularValues().array().abs() > tolerance).select(svd.singularValues().array().inverse(), 0).matrix().asDiagonal() * svd.matrixU().adjoint();
438}
