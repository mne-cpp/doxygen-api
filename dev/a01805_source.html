71AcqExpected<std::string> percentDecode(std::string_view uri, std::size_t begin, std::size_t end)
72{
73 std::string out;
74 out.reserve(end - begin);
75 for (std::size_t i = begin; i < end; ++i) {
76 if (uri[i] != '%') {
77 out.push_back(uri[i]);
78 continue;
79 }
80 const int hi = i + 2 < end ? hexValue(uri[i + 1]) : -1;
81 const int lo = i + 2 < end ? hexValue(uri[i + 2]) : -1;
82 if (hi < 0 || lo < 0) {
83 return invalid(uri, i, "invalid percent escape");
84 }
85 out.push_back(static_cast<char>(hi * 16 + lo));
86 i += 2;
87 }
88 return out;
89}

91void percentEncode(std::string& out, std::string_view text, std::string_view keep)
92{
93 static constexpr std::array<char, 16> kHex = {'0', '1', '2', '3', '4', '5', '6', '7',
94 '8', '9', 'A', 'B', 'C', 'D', 'E', 'F'};
95 for (const char c : text) {
96 if (isAlpha(c) || isDigit(c) || c == '-' || c == '.' || c == '_' || c == '~' ||
97 keep.find(c) != std::string_view::npos) {
98 out.push_back(c);
99 } else {
100 const auto byte = static_cast<unsigned char>(c);
101 out.push_back('%');
102 out.push_back(kHex[byte >> 4U]);
103 out.push_back(kHex[byte & 0xFU]);
104 }
105 }
106}

119AcqExpected<AcqUri> AcqUri::parse(std::string_view uri)
120{
121 const std::size_t separator = uri.find(kSeparator);
122 if (separator == std::string_view::npos) {
123 return invalid(uri, uri.size(), "missing '://'");
124 }
125 if (separator == 0) {
126 return invalid(uri, 0, "empty scheme");
127 }
128 AcqUri out;
129 for (std::size_t i = 0; i < separator; ++i) {
130 const char c = uri[i];
131 // RFC 3986 characters, and '_' as in BrainFlow's streaming_board://
132 const bool ok = isAlpha(c) || (i > 0 && (isDigit(c) || c == '+' || c == '-' || c == '.' || c == '_'));
133 if (!ok) {
134 return invalid(uri, i, std::string("invalid scheme character '") + c + "'");
135 }
136 out.scheme.push_back(c >= 'A' && c <= 'Z' ? static_cast<char>(c - 'A' + 'a') : c);
137 }
138
139 const std::size_t targetBegin = separator + kSeparator.size();
140 const std::size_t query = uri.find('?', targetBegin);
141 const std::size_t targetEnd = query == std::string_view::npos ? uri.size() : query;
142 auto target = percentDecode(uri, targetBegin, targetEnd);
143 if (!target) {
144 return target.error();
145 }
146 out.target = std::move(*target);
147 if (query == std::string_view::npos) {
148 return out;
149 }
150
151 std::size_t begin = query + 1;
152 while (true) {
153 const std::size_t amp = uri.find('&', begin);
154 const std::size_t end = amp == std::string_view::npos ? uri.size() : amp;
155 const std::size_t eq = uri.substr(begin, end - begin).find('=');
156 const std::size_t keyEnd = eq == std::string_view::npos ? end : begin + eq;
157 if (keyEnd == begin) {
158 return invalid(uri, begin, "empty parameter name");
159 }
160 auto key = percentDecode(uri, begin, keyEnd);
161 if (!key) {
162 return key.error();
163 }
164 auto value = percentDecode(uri, keyEnd == end ? end : keyEnd + 1, end);
165 if (!value) {
166 return value.error();
167 }
168 if (out.param(*key)) {
169 return invalid(uri, begin, "duplicate parameter '" + *key + "'");
170 }
171 out.params.emplace_back(std::move(*key), std::move(*value));
172 if (amp == std::string_view::npos) {
173 break;
174 }
175 begin = amp + 1;
176 }
177 return out;
178}

194std::string AcqUri::toString() const
195{
196 std::string out = scheme + std::string(kSeparator);
197 percentEncode(out, target, "/:@");
198 for (std::size_t i = 0; i < params.size(); ++i) {
199 out.push_back(i == 0 ? '?' : '&');
200 percentEncode(out, params[i].first, "");
201 out.push_back('=');
202 percentEncode(out, params[i].second, "");
203 }
204 return out;
205}

211AcqExpected<std::int64_t> mne::detail::paramInt(const AcqUri& uri, std::string_view key, std::int64_t fallback,
212 std::int64_t min, std::int64_t max)
213{
214 const auto text = uri.param(key);
215 if (!text) {
216 return fallback;
217 }
218 std::int64_t value = 0;
219 const auto [end, ec] = std::from_chars(text->data(), text->data() + text->size(), value);
220 if (ec != std::errc() || end != text->data() + text->size() || value < min || value > max) {
221 return badValue(uri, key, *text, "an integer in [" + std::to_string(min) + ", " + std::to_string(max) + "]");
222 }
223 return value;
224}

228AcqExpected<double> mne::detail::paramDouble(const AcqUri& uri, std::string_view key, double fallback, double min,
229 double max)
230{
231 const auto text = uri.param(key);
232 if (!text) {
233 return fallback;
234 }
235 // std::from_chars for double is unavailable on older macOS runtimes; the classic locale keeps '.'.
236 std::istringstream stream{std::string(*text)};
237 stream.imbue(std::locale::classic());
238 double value = 0.0;
239 stream >> value;
240 const bool consumed = !stream.fail() && stream.peek() == std::char_traits<char>::eof();
241 if (!consumed || !std::isfinite(value) || value < min || value > max) {
242 std::ostringstream range;
243 range.imbue(std::locale::classic());
244 range << "a number in [" << min << ", " << max << "]";
245 return badValue(uri, key, *text, range.str());
246 }
247 return value;
248}

269AcqExpected<AcqThreadPolicy> mne::detail::paramThreadPolicy(const AcqUri& uri)
270{
271 AcqThreadPolicy policy;
272 if (const auto cpus = uri.param("cpus")) {
273 const char* p = cpus->data();
274 const char* const end = p + cpus->size();
275 while (true) {
276 unsigned cpu = 0;
277 const auto [next, ec] = std::from_chars(p, end, cpu);
278 if (ec != std::errc() || (next != end && *next != ',')) {
279 return badValue(uri, "cpus", *cpus, "comma-separated CPU numbers, e.g. 2,3");
280 }
281 policy.cpus.push_back(cpu);
282 if (next == end) {
283 break;
284 }
285 p = next + 1;
286 }
287 }
288 if (const auto priority = uri.param("priority")) {
289 if (*priority == "high") {
290 policy.priority = AcqThreadPriority::High;
291 } else if (*priority == "realtime") {
292 policy.priority = AcqThreadPriority::Realtime;
293 } else if (*priority != "normal") {
294 return badValue(uri, "priority", *priority, "normal, high or realtime");
295 }
296 }
297 return policy;
298}
