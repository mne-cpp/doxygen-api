161[[nodiscard]] inline AcqExpected<std::uint64_t> mapSampleExact(const AcqTimebase& from, const AcqTimebase& to,
162 std::uint64_t sample, std::int64_t offsetTicks = 0,
163 AcqRounding rounding = AcqRounding::Exact) noexcept
164{
165 if (from.domain == 0 || from.domain != to.domain || from.lock != AcqLockState::PhaseLocked ||
166 to.lock != AcqLockState::PhaseLocked || to.ticksPerSample == 0 || to.sampleAlignment == 0) {
167 return AcqErrorCode::NotSynchronized;
168 }
169 const std::int64_t delta = from.tickOf(sample) + offsetTicks - to.originTick;
170 if (delta < 0) {
171 return AcqErrorCode::BeforeStart;
172 }
173 const std::uint64_t grid = to.ticksPerSample * to.sampleAlignment;
174 const auto ticks = static_cast<std::uint64_t>(delta);
175 std::uint64_t steps = ticks / grid;
176 if (ticks % grid != 0) {
177 if (rounding == AcqRounding::Exact) {
178 return AcqErrorCode::NotOnGrid;
179 }
180 steps += rounding == AcqRounding::Ceil ? 1U : 0U;
181 }
182 return steps * to.sampleAlignment;
183}

210[[nodiscard]] inline AcqExpected<AcqSampleRange> mapRangeExact(const AcqTimebase& from, const AcqTimebase& to,
211 std::uint64_t first, std::uint64_t end) noexcept
212{
213 if (from.domain == 0 || from.domain != to.domain || from.lock != AcqLockState::PhaseLocked ||
214 to.lock != AcqLockState::PhaseLocked || to.ticksPerSample == 0) {
215 return AcqErrorCode::NotSynchronized;
216 }
217 if (end < first) {
218 return AcqErrorCode::InvalidParameter;
219 }
220 // First sample of to at or after the instant of sample s of from.
221 const auto atOrAfter = [&](std::uint64_t s) -> std::uint64_t {
222 const std::int64_t delta = from.tickOf(s) - to.originTick;
223 if (delta <= 0) {
224 return 0;
225 }
226 const auto ticks = static_cast<std::uint64_t>(delta);
227 return ticks / to.ticksPerSample + (ticks % to.ticksPerSample != 0 ? 1U : 0U);
228 };
229 return AcqSampleRange{atOrAfter(first), atOrAfter(end)};
230}
