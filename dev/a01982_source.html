68QPen toQPen(const viz::Pen& pen)
69{
70 if (!pen.visible()) {
71 return QPen(Qt::NoPen);
72 }
73 Qt::PenStyle style = Qt::SolidLine;
74 switch (pen.style) {
75 case viz::LineStyle::Dash:
76 style = Qt::DashLine;
77 break;
78 case viz::LineStyle::Dot:
79 style = Qt::DotLine;
80 break;
81 case viz::LineStyle::DashDot:
82 style = Qt::DashDotLine;
83 break;
84 case viz::LineStyle::DashDotDot:
85 style = Qt::DashDotDotLine;
86 break;
87 default:
88 break;
89 }
90 const Qt::PenCapStyle cap = pen.cap == viz::LineCap::Flat ? Qt::FlatCap : pen.cap == viz::LineCap::Round ? Qt::RoundCap
91 : Qt::SquareCap;
92 const Qt::PenJoinStyle join = pen.join == viz::LineJoin::Miter ? Qt::MiterJoin : pen.join == viz::LineJoin::Round ? Qt::RoundJoin
93 : Qt::BevelJoin;
94 return QPen(QBrush(toQColor(pen.color)), pen.width, style, cap, join);
95}

114class Painter
115{
116public:
117 Painter(QPainter& painter, const QFont& baseFont)
118 : m_painter(painter)
119 , m_base(painter.worldTransform())
120 , m_baseOpacity(painter.opacity())
121 , m_baseFont(baseFont)
122 {
123 }
124
125 void draw(const viz::Command& command)
126 {
127 m_painter.save();
128 m_painter.setRenderHint(QPainter::Antialiasing, command.antialias);
129 m_painter.setOpacity(m_baseOpacity * command.opacity);
130 for (const viz::Clip& clip : command.clips) {
131 m_painter.setWorldTransform(toQTransform(clip.transform) * m_base);
132 m_painter.setClipRect(toQRect(clip.rect), Qt::IntersectClip);
133 }
134 m_painter.setWorldTransform(toQTransform(command.transform) * m_base);
135 std::visit([this](const auto& shape) { paint(shape); }, command.shape);
136 m_painter.restore();
137 }
138
139private:
140 void paint(const viz::LineShape& line)
141 {
142 m_painter.setPen(toQPen(line.pen));
143 m_painter.drawLine(toQPoint(line.from), toQPoint(line.to));
144 }
145
146 void paint(const viz::PathShape& path)
147 {
148 if (path.cubic) {
149 QPainterPath curve(toQPoint(path.points[0]));
150 for (std::size_t i = 1; i + 2 < path.points.size(); i += 3) {
151 curve.cubicTo(toQPoint(path.points[i]), toQPoint(path.points[i + 1]), toQPoint(path.points[i + 2]));
152 }
153 if (path.closed) {
154 curve.closeSubpath();
155 }
156 m_painter.setPen(toQPen(path.pen));
157 m_painter.setBrush(path.closed ? toQBrush(path.brush) : QBrush(Qt::NoBrush));
158 m_painter.drawPath(curve);
159 return;
160 }
161 QPolygonF polygon;
162 polygon.reserve(static_cast<qsizetype>(path.points.size()));
163 for (const viz::Point& p : path.points) {
164 polygon.append(toQPoint(p));
165 }
166 m_painter.setPen(toQPen(path.pen));
167 if (path.closed) {
168 m_painter.setBrush(toQBrush(path.brush));
169 m_painter.drawPolygon(polygon);
170 } else {
171 m_painter.setBrush(Qt::NoBrush);
172 m_painter.drawPolyline(polygon);
173 }
174 }
175
176 void paint(const viz::RectShape& rect)
177 {
178 m_painter.setPen(toQPen(rect.pen));
179 m_painter.setBrush(toQBrush(rect.brush));
180 if (rect.rx > 0.0 || rect.ry > 0.0) {
181 m_painter.drawRoundedRect(toQRect(rect.rect), rect.rx, rect.ry);
182 } else if (!rect.pen.visible()) {
183 m_painter.fillRect(toQRect(rect.rect), toQBrush(rect.brush));
184 } else {
185 m_painter.drawRect(toQRect(rect.rect));
186 }
187 }
188
189 void paint(const viz::EllipseShape& ellipse)
190 {
191 m_painter.setPen(toQPen(ellipse.pen));
192 m_painter.setBrush(toQBrush(ellipse.brush));
193 m_painter.drawEllipse(toQPoint(ellipse.center), ellipse.rx, ellipse.ry);
194 }
195
196 void paint(const viz::TextShape& text)
197 {
198 m_painter.setPen(toQColor(text.color));
199 m_painter.setFont(toQFont(text.font, m_baseFont));
200 const QString string = QString::fromStdString(text.text);
201 if (viz::has_flag(text.align, viz::Align::Baseline)) {
202 m_painter.drawText(QPointF(text.box.x, text.box.y), string);
203 } else {
204 m_painter.drawText(toQRect(text.box), static_cast<int>(text.align), string);
205 }
206 }
207
208 void paint(const viz::ImageShape& shape)
209 {
210 const viz::Image& image = *shape.image;
211 const QImage view(image.rgba.data(), image.width, image.height, image.width * 4, QImage::Format_RGBA8888);
212 m_painter.setRenderHint(QPainter::SmoothPixmapTransform, shape.smooth);
213 m_painter.drawImage(toQRect(shape.target), view);
214 }
215
216 QPainter& m_painter;
217 QTransform m_base;
218 qreal m_baseOpacity;
219 QFont m_baseFont;
220};

241QFont mne::viz::qt::toQFont(const viz::Font& font, const QFont& baseFont)
242{
243 QFont result = baseFont;
244 if (!font.family.empty()) {
245 result.setFamily(QString::fromStdString(font.family));
246 }
247 if (font.pixel_size > 0.0) {
248 result.setPixelSize(qMax(1, qRound(font.pixel_size)));
249 } else if (font.point_size > 0.0) {
250 result.setPointSizeF(font.point_size);
251 }
252 result.setBold(font.bold);
253 result.setItalic(font.italic);
254 return result;
255}
