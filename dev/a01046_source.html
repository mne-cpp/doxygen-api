72InvCmneResult InvCmne::compute(
73 const MatrixXd& matEvoked,
74 const MatrixXd& matGain,
75 const MatrixXd& matNoiseCov,
76 const MatrixXd& matSrcCov,
77 const InvCmneSettings& settings)
78{
79 InvCmneResult result;
80
81 // Step 1: Compute dSPM kernel
82 MatrixXd matKernelDspm = computeDspmKernel(matGain, matNoiseCov, matSrcCov, settings.lambda2);
83 result.matKernelDspm = matKernelDspm;
84
85 // Step 2: Apply kernel to evoked data -> dSPM source estimate
86 MatrixXd matDspmData = matKernelDspm * matEvoked; // n_sources x n_times
87
88 // Build dSPM source estimate
89 VectorXi vertices = VectorXi::LinSpaced(matDspmData.rows(), 0, matDspmData.rows() - 1);
90 result.stcDspm = InvSourceEstimate(matDspmData, vertices, 0.0f, 1.0f);
91
92 // Step 3: Z-score rectify
93 MatrixXd matZScored = zScoreRectify(matDspmData);
94
95 // Step 4: Apply LSTM correction if model available and enough time points
96 int nTimes = matDspmData.cols();
97 MatrixXd matCmneData;
98
99 if (!settings.onnxModelPath.isEmpty() && nTimes >= settings.lookBack) {
100 matCmneData = applyLstmCorrection(matZScored, settings.onnxModelPath, settings.lookBack);
101
102 // Store raw LSTM prediction for diagnostics
103 result.stcLstmPredict = InvSourceEstimate(matCmneData, vertices, 0.0f, 1.0f);
104 } else {
105 // No correction possible — CMNE falls back to dSPM
106 matCmneData = matDspmData;
107
108 if (settings.onnxModelPath.isEmpty()) {
109 qDebug() << "[InvCmne::compute] No ONNX model path provided, returning uncorrected dSPM.";
110 } else {
111 qDebug() << "[InvCmne::compute] Not enough time points for lookBack window"
112 << "(need" << settings.lookBack << ", have" << nTimes << ").";
113 }
114 }
115
116 // Build CMNE source estimate
117 result.stcCmne = InvSourceEstimate(matCmneData, vertices, 0.0f, 1.0f);
118
119 return result;
120}

124MatrixXd InvCmne::computeDspmKernel(
125 const MatrixXd& matGain,
126 const MatrixXd& matNoiseCov,
127 const MatrixXd& matSrcCov,
128 double lambda2)
129{
130 int nChannels = matGain.rows();
131 int nSources = matGain.cols();
132
133 // Step 1: Whiten noise covariance via eigendecomposition
134 // C_n = V * D * V^T -> C_n^{-1/2} = V * D^{-1/2} * V^T
135 SelfAdjointEigenSolver<MatrixXd> eigSolver(matNoiseCov);
136 VectorXd eigVals = eigSolver.eigenvalues();
137 MatrixXd eigVecs = eigSolver.eigenvectors();
138
139 // Regularize: clamp small eigenvalues
140 double maxEig = eigVals.maxCoeff();
141 double threshold = maxEig * 1e-10;
142 VectorXd eigValsInvSqrt(nChannels);
143 for (int i = 0; i < nChannels; ++i) {
144 eigValsInvSqrt(i) = (eigVals(i) > threshold) ? 1.0 / std::sqrt(eigVals(i)) : 0.0;
145 }
146
147 MatrixXd matWhitener = eigVecs * eigValsInvSqrt.asDiagonal() * eigVecs.transpose();
148
149 // Step 2: Whiten gain matrix
150 MatrixXd matGainWhitened = matWhitener * matGain; // n_channels x n_sources
151
152 // Step 3: MNE kernel
153 // K = C_R * G_tilde^T * (G_tilde * C_R * G_tilde^T + lambda2 * I)^{-1}
154 MatrixXd matGCR = matGainWhitened * matSrcCov; // n_channels x n_sources
155 MatrixXd matA = matGCR * matGainWhitened.transpose(); // n_channels x n_channels
156 matA.diagonal().array() += lambda2;
157
158 // Solve: K = C_R * G_tilde^T * A^{-1}
159 // Equivalent to: K^T = A^{-1} * G_tilde * C_R, then transpose
160 MatrixXd matKT = matA.ldlt().solve(matGCR); // n_channels x n_sources
161 MatrixXd matK = matSrcCov * matGainWhitened.transpose() * matA.ldlt().solve(MatrixXd::Identity(nChannels, nChannels));
162 // Simplification: K = (C_R * G_tilde^T) * A^{-1}
163 matK = (matSrcCov * matGainWhitened.transpose()) * matA.ldlt().solve(MatrixXd::Identity(nChannels, nChannels));
164
165 // Step 4: dSPM normalization
166 // noise_norm_i = sqrt((K * C_n * K^T)(i,i))
167 // K_dSPM(i,:) = K(i,:) / noise_norm_i
168 MatrixXd matKCn = matK * matNoiseCov; // n_sources x n_channels
169 for (int i = 0; i < nSources; ++i) {
170 double noiseNorm = std::sqrt(matKCn.row(i).dot(matK.row(i)));
171 if (noiseNorm > 1e-10) {
172 matK.row(i) /= noiseNorm;
173 }
174 }
175
176 return matK; // n_sources x n_channels (dSPM kernel)
177}

181MatrixXd InvCmne::zScoreRectify(const MatrixXd& matStcData)
182{
183 int nSources = matStcData.rows();
184 int nTimes = matStcData.cols();
185
186 MatrixXd matResult(nSources, nTimes);
187
188 for (int i = 0; i < nSources; ++i) {
189 // Absolute value
190 VectorXd absRow = matStcData.row(i).cwiseAbs();
191
192 // Mean and standard deviation across time
193 double mu = absRow.mean();
194 double variance = (absRow.array() - mu).square().mean();
195 double sigma = std::sqrt(variance);
196
197 // Z-score (guard against zero std)
198 double denom = std::max(sigma, 1e-10);
199 matResult.row(i) = (absRow.array() - mu) / denom;
200 }
201
202 return matResult;
203}

207MatrixXd InvCmne::applyLstmCorrection(
208 const MatrixXd& matDspmData,
209 const QString& onnxModelPath,
210 int lookBack)
211{
212 Q_UNUSED(onnxModelPath)
213
214 int nSources = matDspmData.rows();
215 int nTimes = matDspmData.cols();
216
217 MatrixXd result = matDspmData; // copy — for t < lookBack: identity (no correction)
218
219 // For t >= lookBack: apply temporal correction
220 for (int t = lookBack; t < nTimes; ++t) {
221 // Extract window: last lookBack columns before t
222 MatrixXd window = result.middleCols(t - lookBack, lookBack);
223
224 // TODO: Replace with actual ONNX inference when ml library is ready
225 // For now, use moving average as control estimate (from paper)
226 VectorXd prediction = window.rowwise().mean();
227
228 // Normalize prediction (Eq. 12)
229 double maxVal = prediction.cwiseAbs().maxCoeff();
230 if (maxVal > 1e-10) {
231 prediction = prediction.cwiseAbs() / maxVal;
232 }
233
234 // CMNE correction: element-wise product (Eq. 13)
235 result.col(t) = prediction.cwiseProduct(matDspmData.col(t));
236 }
237
238 return result;
239}

243UTILSLIB::PythonRunnerResult InvCmne::trainLstm(
244 const QString& fwdPath,
245 const QString& covPath,
246 const QString& epochsPath,
247 const QString& outOnnxPath,
248 const InvCmneSettings& settings,
249 const QString& gtStcPrefix,
250 int hiddenSize,
251 int numLayers,
252 int trainEpochs,
253 double learningRate,
254 int batchSize,
255 const QString& pythonExe)
256{
257 // Resolve training package directory (contains pyproject.toml + script)
258 // Expected layout: <app_dir>/../scripts/ml/training/cmne/
259 QString appDir = QCoreApplication::applicationDirPath();
260 QString cmneDir = QDir(appDir).absoluteFilePath(
261 QStringLiteral("../scripts/ml/training/cmne"));
262
263 // Fallback: source tree relative to working directory
264 if (!QFile::exists(QDir(cmneDir).absoluteFilePath(QStringLiteral("pyproject.toml")))) {
265 cmneDir = QStringLiteral("scripts/ml/training/cmne");
266 }
267
268 QString scriptPath = QDir(cmneDir).absoluteFilePath(QStringLiteral("train_cmne_lstm.py"));
269
270 if (!QFile::exists(scriptPath)) {
271 UTILSLIB::PythonRunnerResult result;
272 result.stdErr = QStringLiteral("Training script not found: ") + scriptPath;
273 qWarning() << "[InvCmne::trainLstm]" << result.stdErr;
274 return result;
275 }
276
277 qDebug() << "[InvCmne::trainLstm] Script:" << scriptPath;
278 qDebug() << "[InvCmne::trainLstm] Package dir:" << cmneDir;
279
280 // Map method integer to string
281 QString methodStr;
282 switch (settings.method) {
283 case 0: methodStr = QStringLiteral("MNE"); break;
284 case 1: methodStr = QStringLiteral("dSPM"); break;
285 case 2: methodStr = QStringLiteral("sLORETA"); break;
286 case 3: methodStr = QStringLiteral("eLORETA"); break;
287 default: methodStr = QStringLiteral("dSPM"); break;
288 }
289
290 double snr = 1.0 / std::sqrt(settings.lambda2);
291
292 // Build argument list matching train_cmne_lstm.py CLI
293 QStringList args;
294 args << QStringLiteral("--fwd") << fwdPath
295 << QStringLiteral("--cov") << covPath
296 << QStringLiteral("--epochs") << epochsPath
297 << QStringLiteral("--out") << outOnnxPath
298 << QStringLiteral("--look-back") << QString::number(settings.lookBack)
299 << QStringLiteral("--method") << methodStr
300 << QStringLiteral("--snr") << QString::number(snr, 'g', 6)
301 << QStringLiteral("--hidden") << QString::number(hiddenSize)
302 << QStringLiteral("--layers") << QString::number(numLayers)
303 << QStringLiteral("--train-epochs") << QString::number(trainEpochs)
304 << QStringLiteral("--lr") << QString::number(learningRate, 'g', 6)
305 << QStringLiteral("--batch") << QString::number(batchSize);
306
307 if (!gtStcPrefix.isEmpty()) {
308 args << QStringLiteral("--gt-stc") << gtStcPrefix;
309 }
310
311 // Configure PythonRunner with venv + pyproject.toml
312 // Venv lives inside the cmne package directory as .venv/
313 UTILSLIB::PythonRunnerConfig config;
314 config.pythonExe = pythonExe;
315 config.venvDir = QDir(cmneDir).absoluteFilePath(QStringLiteral(".venv"));
316 config.packageDir = cmneDir;
317
318 MLLIB::MlTrainer trainer(config);
319
320 return trainer.run(scriptPath, args);
321}
