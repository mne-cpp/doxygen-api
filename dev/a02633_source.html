35// Code points of UTF-8 text: every byte except continuation bytes starts one
36double codePoints(std::string_view text)
37{
38 double count = 0.0;
39 for (const char c : text) {
40 if ((static_cast<unsigned char>(c) & 0xC0) != 0x80) {
41 count += 1.0;
42 }
43 }
44 return count;
45}
