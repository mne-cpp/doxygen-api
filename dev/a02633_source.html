326 panes.push_back(&subView);

330 scene::BrainViewInteraction::Effects effects;
331 effects.redraw = [this] {
332 cameraChanged();
333 };
334 effects.layout_changed = [this] {
335 updateOverlayLayout();
336 cameraChanged();
337 };
338 effects.settings_changed = [this] {
339 saveMultiViewSettings();
340 };
341 effects.cursor_changed = [this](scene::CursorShape shape) {
342 setHostCursor(shape);
343 };
344 effects.viewport_selected = [this](int viewport) {
345 if (viewport != m_visualizationEditTarget) {
346 setVisualizationEditTarget(viewport);
347 }
348 };
349 // A hidden surface shows no labels, so its name labels take no clicks
350 effects.hits_viewport_label = [this](int viewport, scene::Point pos) {
351 const scene::BrainViewOverlay::Label* name = m_overlay.item(scene::BrainViewOverlay::Item::ViewportNameLabel, viewport);
352 return name && name->visible && m_host && m_host->isVisible() && toQRect(name->geometry).contains(toQPoint(pos));
353 };
354 effects.preset_menu_requested = [this](int viewport, scene::Point pos) {
355 if (m_host) {
356 m_host->requestViewportPresetMenu(viewport, toQPoint(pos));
357 }
358 };
359 effects.hovered = [this](scene::Point pos) {
360 castRay(toQPoint(pos));
361 };
362 effects.clicked = [this](scene::Point pos) {
363 castRay(toQPoint(pos));
364 if (const mne::IeegContact* contact = m_electrodes ? m_electrodes->contactAt(m_electrodes->hoveredContact()) : nullptr) {
365 emit electrodeContactClicked(QString::fromStdString(contact->name), QVector3D(contact->position.x(), contact->position.y(), contact->position.z()));
366 } else if (m_hasIntersection) {
367 emit surfacePointClicked(m_lastIntersectionPoint);
368 }
369 };
370 effects.double_clicked = [this](scene::Point pos) {
371 castRay(toQPoint(pos));
372 if (m_hasIntersection) {
373 emit surfacePointDoubleClicked(m_lastIntersectionPoint);
374 return true;
375 }
376 return false;
377 };
378 effects.snapshot_requested = [this] {
379 saveSnapshot();
380 };
381 m_interaction.set_effects(std::move(effects));
382}

403void BrainViewCore::setModel(BrainTreeModel* model)
404{
405 if (model == m_model)
406 return;
407
408 if (m_model) {
409 disconnect(m_model, nullptr, this, nullptr);
410 for (auto it = m_itemSurfaceMap.cbegin(); it != m_itemSurfaceMap.cend(); ++it) {
411 for (auto sit = m_surfaces.begin(); sit != m_surfaces.end();) {
412 sit = (sit->second == it->second) ? m_surfaces.erase(sit) : std::next(sit);
413 }
414 }
415 m_itemSurfaceMap.clear();
416 m_itemDipoleMap.clear();
417 m_hoveredItem = nullptr;
418 m_activeSurface.reset();
419 }
420
421 m_model = model;
422 if (!m_model) {
423 updateSceneBounds();
424 m_sceneDirty = true;
425 requestUpdate();
426 return;
427 }
428 connect(m_model, &BrainTreeModel::rowsInserted, this, &BrainViewCore::onRowsInserted);
429 connect(m_model, &BrainTreeModel::dataChanged, this, &BrainViewCore::onDataChanged);
430
431 if (m_model->rowCount() > 0) {
432 onRowsInserted(QModelIndex(), 0, m_model->rowCount() - 1);
433 } else {
434 updateSceneBounds();
435 m_sceneDirty = true;
436 requestUpdate();
437 }
438}

450void BrainViewCore::onRowsInserted(const QModelIndex& parent, int first, int last)
451{
452 if (!m_model)
453 return;
454
455 for (int i = first; i <= last; ++i) {
456 QModelIndex index = m_model->index(i, 0, parent);
457 QStandardItem* item = m_model->itemFromIndex(index);
458
459 AbstractTreeItem* absItem = dynamic_cast<AbstractTreeItem*>(item);
460
461 // Handle FsSurface Items
462 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::SurfaceItem)) {
463 SurfaceTreeItem* surfItem = static_cast<SurfaceTreeItem*>(absItem);
464 auto brainSurf = std::make_shared<BrainSurface>();
465
466 // Load geometry from item
467 brainSurf->fromSurface(surfItem->surfaceData());
468
469 // Determine Hemisphere from Parent
470 if (absItem->parent()) {
471 QString parentText = absItem->parent()->text();
472 if (parentText == "lh")
473 brainSurf->setHemi(0);
474 else if (parentText == "rh")
475 brainSurf->setHemi(1);
476 }
477
478 // Set properties
479 brainSurf->setVisible(surfItem->isVisible());
480
481 // Brain surfaces (pial, white, inflated, etc.) are brain tissue
482 brainSurf->setTissueType(BrainSurface::TissueBrain);
483
484 m_itemSurfaceMap[item] = brainSurf;
485
486 // Key generation: "hemi_type" e.g. "lh_pial"
487 std::string key;
488 if (absItem->parent()) {
489 key = (absItem->parent()->text() + "_" + surfItem->text()).toStdString();
490 } else {
491 key = surfItem->text().toStdString();
492 }
493 m_surfaces[key] = brainSurf;
494
495 // Check for annotations
496 if (!surfItem->annotationData().empty()) {
497 brainSurf->addAnnotation(surfItem->annotationData());
498 }
499
500 // Set active if first
501 if (!m_activeSurface) {
502 m_activeSurface = brainSurf;
503 m_activeSurfaceType = surfItem->text().toStdString();
504 }
505 }
506 // Check for BEM Item (using dynamic_cast for safety)
507 BemTreeItem* bemItem = dynamic_cast<BemTreeItem*>(absItem);
508 if (bemItem) {
509 const mne::MNEBemSurface& bemSurfData = bemItem->bemSurfaceData();
510
511 auto brainSurf = std::make_shared<BrainSurface>();
512
513 // Load BEM geometry with color from item
514 brainSurf->fromBemSurface(bemSurfData, itemColor(*bemItem));
515
516 brainSurf->setVisible(bemItem->isVisible());
517
518 // Set tissue type based on surface name
519 QString surfName = bemItem->text().toLower();
520 if (surfName.contains("head") || surfName.contains("skin") || surfName.contains("scalp")) {
521 brainSurf->setTissueType(BrainSurface::TissueSkin);
522 } else if (surfName.contains("outer") && surfName.contains("skull")) {
523 brainSurf->setTissueType(BrainSurface::TissueOuterSkull);
524 } else if (surfName.contains("inner") && surfName.contains("skull")) {
525 brainSurf->setTissueType(BrainSurface::TissueInnerSkull);
526 } else if (surfName.contains("skull")) {
527 brainSurf->setTissueType(BrainSurface::TissueOuterSkull); // Default skull to outer
528 } else if (surfName.contains("brain")) {
529 brainSurf->setTissueType(BrainSurface::TissueBrain);
530 }
531
532 m_itemSurfaceMap[item] = brainSurf;
533
534 // Legacy map support (Use item text e.g. "bem_head")
535 m_surfaces["bem_" + bemItem->text().toStdString()] = brainSurf;
536 }
537
538 // Handle Sensor Items
539 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::SensorItem)) {
540 SensorTreeItem* sensItem = static_cast<SensorTreeItem*>(absItem);
541
542 std::shared_ptr<BrainSurface> brainSurf;
543
544 QString parentText = "";
545 if (sensItem->parent())
546 parentText = sensItem->parent()->text();
547
548 if (parentText.contains("MEG/Grad") && sensItem->hasOrientation()) {
549 brainSurf = MeshFactory::createBarbell(sensItem->position(), sensItem->orientation(),
550 itemColor(*sensItem), sensItem->scale());
551 } else if (parentText.contains("MEG/Mag") && sensItem->hasOrientation()) {
552 brainSurf = MeshFactory::createPlate(sensItem->position(), sensItem->orientation(),
553 itemColor(*sensItem), sensItem->scale());
554 } else {
555 // EEG and other sensors: smooth icosphere
556 brainSurf = MeshFactory::createSphere(sensItem->position(), sensItem->scale(),
557 itemColor(*sensItem));
558 }
559
560 brainSurf->setVisible(sensItem->isVisible());
561 m_itemSurfaceMap[item] = brainSurf;
562
563
564 // Legacy map support
565 const std::string keyPrefix = mne::disp::surface_keys::sensorParentToKeyPrefix(parentText.toStdString());
566
567 const std::string key = std::format("{}{}_{}", keyPrefix, sensItem->text().toStdString(), reinterpret_cast<std::uintptr_t>(sensItem));
568 m_surfaces[key] = brainSurf;
569 }
570
571 // Handle Dipole Items
572 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::DipoleItem)) {
573 DipoleTreeItem* dipItem = static_cast<DipoleTreeItem*>(absItem);
574 auto dipObject = std::make_shared<DipoleObject>();
575 dipObject->load(dipItem->ecdSet());
576 dipObject->setVisible(dipItem->isVisible());
577 dipObject->applyTransform(itemTransform(item));
578
579 m_itemDipoleMap[item] = dipObject;
580 }
581
582 // Handle Source Space Items (one item per hemisphere, batched mesh)
583 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::SourceSpaceItem)) {
584 SourceSpaceTreeItem* srcItem = static_cast<SourceSpaceTreeItem*>(absItem);
585 const std::vector<QVector3D>& positions = srcItem->positions();
586 if (positions.empty())
587 continue;
588
589 auto brainSurf = MeshFactory::createBatchedSpheres(positions, srcItem->scale(),
590 itemColor(*srcItem));
591 brainSurf->setVisible(srcItem->isVisible());
592 m_itemSurfaceMap[item] = brainSurf;
593
594 const std::string key = "srcsp_" + srcItem->text().toStdString();
595 m_surfaces[key] = brainSurf;
596 }
597
598 // Handle Digitizer Items (batched sphere mesh per category)
599 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::DigitizerItem)) {
600 DigitizerTreeItem* digItem = static_cast<DigitizerTreeItem*>(absItem);
601 const std::vector<QVector3D>& positions = digItem->positions();
602 if (positions.empty())
603 continue;
604
605 auto brainSurf = MeshFactory::createBatchedSpheres(positions, digItem->scale(),
606 itemColor(*digItem));
607 brainSurf->setVisible(digItem->isVisible());
608
609 m_itemSurfaceMap[item] = brainSurf;
610
611 // Category name for surface-map key. A monotonic counter
612 // suffix guarantees uniqueness so that multiple items of the
613 // same PointKind each get their own entry in m_surfaces.
614 // shouldRenderSurface() matches by prefix ("dig_cardinal",
615 // "dig_hpi", …) so the suffix is transparent to visibility.
616 static int s_digKeyCounter = 0;
617 std::string catName;
618 switch (digItem->pointKind()) {
619 case DigitizerTreeItem::Cardinal:
620 catName = "cardinal";
621 break;
622 case DigitizerTreeItem::HPI:
623 catName = "hpi";
624 break;
625 case DigitizerTreeItem::EEG:
626 catName = "eeg";
627 break;
628 case DigitizerTreeItem::Extra:
629 catName = "extra";
630 break;
631 default:
632 catName = digItem->text().toLower().replace(' ', '_').toStdString();
633 break;
634 }
635 const std::string key = std::format("dig_{}_{}", catName, s_digKeyCounter++);
636 m_surfaces[key] = brainSurf;
637 }
638
639 if (m_itemSurfaceMap.contains(item))
640 m_itemSurfaceMap[item]->applyTransform(itemTransform(item));
641
642 // Check children recursively
643 if (m_model->hasChildren(index)) {
644 onRowsInserted(index, 0, m_model->rowCount(index) - 1);
645 }
646 }
647 updateInflatedSurfaceTransforms();
648 updateSceneBounds();
649 m_vertexCountDirty = true;
650 m_sceneDirty = true;
651 requestUpdate();
652}

654void BrainViewCore::onDataChanged(const QModelIndex& topLeft, const QModelIndex& bottomRight, const QList<int>& roles)
655{
656 bool transformChanged = false;
657 for (int i = topLeft.row(); i <= bottomRight.row(); ++i) {
658 QModelIndex index = m_model->index(i, 0, topLeft.parent());
659 QStandardItem* item = m_model->itemFromIndex(index);
660
661 if (m_itemSurfaceMap.contains(item)) {
662 auto surf = m_itemSurfaceMap[item];
663
664 AbstractTreeItem* absItem = dynamic_cast<AbstractTreeItem*>(item);
665 if (absItem) {
666 if (roles.contains(AbstractTreeItem::VisibleRole)) {
667 surf->setVisible(absItem->isVisible());
668 m_vertexCountDirty = true;
669 }
670 if ((roles.contains(AbstractTreeItem::ColorRole) || roles.contains(AbstractTreeItem::AlphaRole)) && absItem->color().isValid()) {
671 surf->setColor(itemColor(*absItem));
672 }
673 if (roles.contains(AbstractTreeItem::TransformRole)) {
674 surf->applyTransform(itemTransform(item));
675 transformChanged = true;
676 }
677 if (roles.contains(SurfaceTreeItem::AnnotationDataRole)) {
678 SurfaceTreeItem* sItem = static_cast<SurfaceTreeItem*>(absItem);
679 if (!sItem->annotationData().empty()) {
680 surf->addAnnotation(sItem->annotationData());
681 }
682 }
683 }
684 }
685
686 if (m_itemDipoleMap.contains(item)) {
687 auto* absItem = dynamic_cast<AbstractTreeItem*>(item);
688 if (absItem && roles.contains(AbstractTreeItem::VisibleRole))
689 m_itemDipoleMap[item]->setVisible(absItem->isVisible());
690 if (roles.contains(AbstractTreeItem::TransformRole))
691 m_itemDipoleMap[item]->applyTransform(itemTransform(item));
692 }
693 }
694 if (transformChanged) {
695 updateInflatedSurfaceTransforms();
696 m_vertexCountDirty = true;
697 }
698 updateSceneBounds();
699 m_sceneDirty = true;
700 requestUpdate();
701}

726void BrainViewCore::updateSceneBounds()
727{
728 QVector3D min(std::numeric_limits<float>::max(), std::numeric_limits<float>::max(), std::numeric_limits<float>::max());
729 QVector3D max(std::numeric_limits<float>::lowest(), std::numeric_limits<float>::lowest(), std::numeric_limits<float>::lowest());
730 bool hasContent = false;
731
732 // Iterate over all surfaces
733 for (auto it = m_surfaces.begin(); it != m_surfaces.end(); ++it) {
734 if (it->second->isVisible()) {
735 QVector3D sMin, sMax;
736 it->second->boundingBox(sMin, sMax);
737
738 min.setX(std::min(min.x(), sMin.x()));
739 min.setY(std::min(min.y(), sMin.y()));
740 min.setZ(std::min(min.z(), sMin.z()));
741
742 max.setX(std::max(max.x(), sMax.x()));
743 max.setY(std::max(max.y(), sMax.y()));
744 max.setZ(std::max(max.z(), sMax.z()));
745 hasContent = true;
746 }
747 }
748
749 for (auto it = m_itemDipoleMap.cbegin(); it != m_itemDipoleMap.cend(); ++it) {
750 QVector3D dMin, dMax;
751 if (it->second->isVisible() && it->second->boundingBox(dMin, dMax)) {
752 min = QVector3D(std::min(min.x(), dMin.x()), std::min(min.y(), dMin.y()), std::min(min.z(), dMin.z()));
753 max = QVector3D(std::max(max.x(), dMax.x()), std::max(max.y(), dMax.y()), std::max(max.z(), dMax.z()));
754 hasContent = true;
755 }
756 }
757
758 if (hasContent) {
759 m_sceneCenter = (min + max) * 0.5f;
760
761 QVector3D diag = max - min;
762 m_sceneSize = std::max(diag.x(), std::max(diag.y(), diag.z()));
763
764 // Ensure non-zero size
765 if (m_sceneSize < 0.01f)
766 m_sceneSize = 0.3f;
767
768 } else {
769 // Default
770 m_sceneCenter = QVector3D(0, 0, 0);
771 m_sceneSize = 0.3f;
772 }
773}

917void BrainViewCore::updateInflatedSurfaceTransforms()
918{
919 const bool needsInflated = (m_singleView.surfaceType == "inflated") || std::any_of(m_subViews.cbegin(), m_subViews.cend(), [](const SubView& sv) { return sv.surfaceType == "inflated"; });

937 const float lhMaxX = lhSurf->maxX();

940 const float gap = 0.005f;
941 const float lhOffset = -gap / 2.0f - lhMaxX;
942 const float rhOffset = gap / 2.0f - rhMinX;
943
944 lhSurf->translateX(lhOffset);
945 rhSurf->translateX(rhOffset);
946}

1208
1209bool BrainViewCore::savePng(const std::filesystem::path& path, int width, int height,
1210 const std::string& surfaceType)
1211{
1212 if (!m_host) {
1213 return false;
1214 }
1215
1216 // Set the surface type filter so the render loop matches the
1217 // loaded surface names (e.g. "white" vs default "pial").
1218 m_singleView.surfaceType = surfaceType;
1219
1220 // Let the host realise its surface off-screen (QRhiWidget with
1221 // Qt::WA_DontShowOnScreen, a hidden QWindow, ...) so that frames are
1222 // rendered without a window-manager surface.
1223 m_host->beginOffscreenCapture(QSize(width, height));
1224
1225 // Spin the event loop until the renderer exists (at least one frame
1226 // has been recorded) and the surface map is populated.
1227 const auto waitStart = std::chrono::steady_clock::now();
1228 while (std::chrono::steady_clock::now() - waitStart < std::chrono::seconds(5)) {
1229 QCoreApplication::processEvents(QEventLoop::AllEvents, 16);
1230 if (m_renderer && !m_surfaces.empty())
1231 break;
1232 }
1233
1234 if (!m_renderer) {
1235 m_host->endOffscreenCapture();
1236 return false;
1237 }
1238
1239 // Force a dirty scene so the next grab renders everything.
1240 m_sceneDirty = true;
1241 updateSceneBounds();
1242
1243 // Pump a few more frames so pipeline resources are fully created.
1244 for (int i = 0; i < 5; ++i)
1245 QCoreApplication::processEvents(QEventLoop::AllEvents, 16);
1246
1247 // The host renders a frame and reads the texture back.
1248 QImage img = grabSceneFrame();
1249
1250 m_host->endOffscreenCapture();
1251
1252 if (img.isNull())
1253 return false;
1254
1255 return img.save(QString::fromStdU16String(path.u16string()), "PNG");
1256}

1371
1372void BrainViewCore::loadMultiViewSettings()
1373{
1374 QSettings settings;
1375 settings.beginGroup(QStringLiteral("BrainView"));
1376
1377 m_interaction.layout().set_split_x(settings.value("multiSplitX", 0.5f).toFloat());
1378 m_interaction.layout().set_split_y(settings.value("multiSplitY", 0.5f).toFloat());
1379
1380 const int savedViewMode = settings.value("viewMode", static_cast<int>(SingleView)).toInt();
1381 m_viewCount = std::clamp(settings.value("viewCount", 1).toInt(), 1, static_cast<int>(m_subViews.size()));
1382 // showSingleView keeps the pane count for the next showMultiView; one pane is always the single view
1383 m_viewMode = (savedViewMode == static_cast<int>(MultiView) && m_viewCount > 1) ? MultiView : SingleView;
1384
1385 Eigen::Quaternionf cameraRotation = m_singleView.rotation;
1386 const bool hasCameraQuat = settings.contains("cameraRotW") && settings.contains("cameraRotX") && settings.contains("cameraRotY") && settings.contains("cameraRotZ");
1387 if (hasCameraQuat) {
1388 const float w = settings.value("cameraRotW", 1.0f).toFloat();
1389 const float x = settings.value("cameraRotX", 0.0f).toFloat();
1390 const float y = settings.value("cameraRotY", 0.0f).toFloat();
1391 const float z = settings.value("cameraRotZ", 0.0f).toFloat();
1392 cameraRotation = Eigen::Quaternionf(w, x, y, z);
1393 if (cameraRotation.squaredNorm() <= std::numeric_limits<float>::epsilon()) {
1394 cameraRotation = Eigen::Quaternionf::Identity();
1395 } else {
1396 cameraRotation.normalize();
1397 }
1398 }
1399
1400 // Reset per-index defaults, then load saved state on top
1401 for (int i = 0; i < std::ssize(m_subViews); ++i) {
1402 m_subViews[i] = SubView::defaultForIndex(i);
1403 m_subViews[i].enabled = (i < m_viewCount);
1404 }
1405
1406 // Delegate per-SubView serialization
1407 m_singleView.load(settings, "single_", cameraRotation);
1408 // The single view turns by the camera rotation, which is stored on its own
1409 m_singleView.rotation = cameraRotation;
1410 for (int i = 0; i < std::ssize(m_subViews); ++i)
1411 m_subViews[i].load(settings, QStringLiteral("multi%1_").arg(i), cameraRotation);
1412
1413 const int maxIdx = static_cast<int>(m_subViews.size()) - 1;
1414 m_visualizationEditTarget = normalizedVisualizationTarget(
1415 settings.value("visualizationEditTarget", -1).toInt(), maxIdx);
1416
1417 m_infoPanelVisible = settings.value("infoPanelVisible", true).toBool();
1418
1419 settings.endGroup();
1420
1421 setVisualizationEditTarget(m_visualizationEditTarget);
1422
1423 // Notify observers that persisted state has been restored. Defer to
1424 // the next event-loop tick so that callers constructing BrainView
1425 // and connecting to these signals immediately afterwards still
1426 // receive the initial state — direct emits from inside the ctor
1427 // chain would fire before any external connect() call.

1617
1618void BrainViewCore::renderFrame(QRhi* rhi, int sampleCount, QRhiCommandBuffer* cb)
1619{
1620 m_inRender = true;
1621 const QImage noOverlay;
1622 const auto overlayFor = [this, &noOverlay](const QSize& pixelSize) -> const QImage& {
1623 if (m_overlaySuppressed) {
1624 return noOverlay;
1625 }
1626 const qreal dpr = m_host ? m_host->devicePixelRatio() : 1.0;
1627 if (m_overlayDirty || m_overlayCache.size() != pixelSize || m_overlayCache.devicePixelRatio() != dpr) {
1628 m_overlayCache = overlayImage(pixelSize, dpr);
1629 m_overlayDirty = false;
1630 }
1631 return m_overlayCache;
1632 };
1633
1634 // Check if there is anything to render
1635 bool hasSurfaces = !m_surfaces.empty();
1636 bool hasDipoles = !m_itemDipoleMap.empty() || m_dipoles; // Check managed dipoles too
1637
1638 // If absolutely nothing is loaded, render black background
1639 if (!hasSurfaces && !hasDipoles && !m_electrodes) {
1640 // No surface loaded: render a black background (and the overlay) instead of leaving the target uninitialized
1641 m_renderer->initialize(rhi, m_renderer->rtClear()->renderPassDescriptor(), sampleCount);
1642 QRhiResourceUpdateBatch* updates = rhi->nextResourceUpdateBatch();
1643 m_renderer->prepareScreenOverlay(rhi, updates, overlayFor(m_renderer->rtClear()->pixelSize()));
1644 m_renderer->beginFrame(cb, updates);
1645 m_renderer->issueScreenOverlayDraw(cb);
1646 m_renderer->endPass(cb);
1647 m_inRender = false;
1648 return;
1649 }
1650
1651 // Ensure active surface pointer is valid if possible, otherwise just use first available for stats
1652 if (!m_activeSurface && !m_surfaces.empty()) {
1653 m_activeSurface = m_surfaces.begin()->second;
1654 }
1655
1656
1657 m_frameCount++;
1658 const std::chrono::duration<float> fpsElapsed = std::chrono::steady_clock::now() - m_fpsStart;
1659 if (fpsElapsed >= std::chrono::milliseconds(500)) {
1660 float fps = m_frameCount / fpsElapsed.count();
1661
1662 // Recount vertices only when surface list/visibility changed
1663 if (m_vertexCountDirty) {
1664 auto countVerticesForSubView = [this](const SubView& sv) -> qint64 {
1665 qint64 total = 0;
1666 for (const auto& [key, surface] : m_surfaces) {
1667 if (!surface)
1668 continue;
1669 if (!sv.shouldRenderSurface(key))
1670 continue;
1671 if (SubView::isBrainSurfaceKey(key)) {
1672 if (!sv.matchesSurfaceType(key))
1673 continue;
1674 } else {
1675 if (!surface->isVisible())
1676 continue;
1677 }
1678 total += surface->vertexCount();
1679 }
1680 return total;
1681 };
1682
1683 qint64 vCount = 0;
1684 if (m_viewMode == MultiView) {
1685 for (int vp : enabledViewportIndices()) {
1686 vCount += countVerticesForSubView(m_subViews[vp]);
1687 }
1688 } else {
1689 vCount = countVerticesForSubView(m_singleView);
1690 }
1691 m_cachedVertexCount = vCount;
1692 m_vertexCountDirty = false;
1693 }
1694
1695 m_overlay.set_text(scene::BrainViewOverlay::Item::FpsLabel, std::format("FPS: {:.1f}\nVertices: {}", fps, m_cachedVertexCount));
1696 updateOverlayLayout();
1697 m_frameCount = 0;
1698 m_fpsStart = std::chrono::steady_clock::now();
1699 }
1700
1701 // Initialize renderer
1702 m_renderer->initialize(rhi, m_renderer->rtClear()->renderPassDescriptor(), sampleCount);
1703
1704 // Determine viewport configuration
1705 QSize outputSize = m_renderer->rtClear()->pixelSize();
1706
1707 // Build list of enabled viewports
1708 const auto enabledViewports = enabledViewportIndices();
1709 int numEnabled = static_cast<int>(enabledViewports.size());
1710
1711 // ── Prepare phase ──────────────────────────────────────────────────
1712 // Every upload and uniform of the frame goes into one batch that
1713 // beginFrame() commits: QRhi forbids resource updates inside a pass.
1714 // The prepare calls upload only the streams that changed, and only for
1715 // objects that are drawn. Overlay modes and highlights are per-draw
1716 // uniforms, so panes with different overlays share the same buffers.
1717 m_renderer->newFrame();
1718 QRhiResourceUpdateBatch* updates = rhi->nextResourceUpdateBatch();
1719 if (m_videoOverlay && m_videoOverlay->isEnabled()) {
1720 m_renderer->prepareVideoOverlay(rhi, updates, m_videoOverlay.get());
1721 }
1722 for (int i = 0; i < kMaxSliceSlots; ++i) {
1723 m_renderer->prepareSlice(rhi, updates, (m_slices[i] && m_sliceVisible[i]) ? m_slices[i] : nullptr, i);
1724 }
1725
1726 BrainSurface* videoOverlayTargetSurface = nullptr;
1727 const bool hasVideoOverlay = m_videoOverlay && m_videoOverlay->isEnabled() && m_videoOverlay->hasFrame();
1728 if (hasVideoOverlay) {
1729 // Target is always the head surface (BEM head or TissueSkin).
1730 if (const auto head = m_surfaces.find("bem_head"); head != m_surfaces.end()) {
1731 videoOverlayTargetSurface = head->second.get();
1732 } else {
1733 for (auto it = m_surfaces.begin(); it != m_surfaces.end(); ++it) {
1734 if (it->second && it->second->tissueType() == BrainSurface::TissueSkin) {
1735 videoOverlayTargetSurface = it->second.get();
1736 break;
1737 }
1738 }
1739 }
1740 }
1741
1742 struct DrawItem
1743 {
1744 BrainSurface* surface;
1745 mne::disp::ShaderMode mode;
1746 float overlayMode;
1747 float distSq; // for transparent back-to-front sort
1748 int uniformOffset; // filled by prepareSurfaceDraw
1749 };
1750
1751 // What each pane draws, with the uniform slots prepared for it
1752 struct PaneDraws
1753 {
1754 QRhiViewport viewport;
1755 QRhiScissor scissor;
1756 std::vector<DrawItem> opaqueDraws;
1757 std::vector<DrawItem> transparentDraws;
1758 int sliceOffsets[kMaxSliceSlots] = {-1, -1, -1};
1759 int videoOverlayOffset = -1;
1760 int electrodeOffset = -1;
1761 std::vector<std::pair<DipoleObject*, int>> dipoleDraws;
1762 int networkOffset = -1;
1763 int headPathOffset = -1;
1764 int pointerOffset = -1;
1765 };
1766 std::vector<PaneDraws> panes(static_cast<size_t>(numEnabled));
1767
1768 for (int slot = 0; slot < numEnabled; ++slot) {
1769 PaneDraws& pane = panes[static_cast<size_t>(slot)];
1770 int vp = (m_viewMode == MultiView) ? enabledViewports[slot] : 0;
1771 const SubView& sv = (m_viewMode == MultiView) ? m_subViews[vp] : m_singleView;
1772
1773 // The scene leaves the separator lines to the right and below a pane free
1774 const scene::PaneLayout& layout = m_interaction.layout();
1775 const scene::PixelRect renderPixels = (m_viewMode == MultiView)
1776 ? layout.inset_for_separator(layout.slot_rect(slot, numEnabled, toPixelSize(outputSize)), slot, numEnabled)
1777 : scene::PixelRect{0, 0, outputSize.width(), outputSize.height()};
1778 const QRect renderRect = toQRect(renderPixels);
1779
1780 const int viewX = renderRect.x();
1781 const int viewY = outputSize.height() - (renderRect.y() + renderRect.height());
1782 const int viewW = std::max(1, renderRect.width());
1783 const int viewH = std::max(1, renderRect.height());
1784
1785 pane.viewport = QRhiViewport(viewX, viewY, viewW, viewH);
1786 pane.scissor = QRhiScissor(viewX, viewY, viewW, viewH);
1787 const float aspectRatio = float(viewW) / float(viewH);
1788
1789 // Calculate camera for this viewport
1790 m_camera.set_scene_center(toEigen(m_cameraFocusOverride ? m_cameraFocusCenter : m_sceneCenter));
1791 m_camera.set_scene_size(m_cameraFocusOverride ? m_cameraFocusSize : m_sceneSize);
1792 const scene::CameraMatrices cam = (m_viewMode == MultiView)
1793 ? m_camera.pane_view(sv, aspectRatio)
1794 : m_camera.single_view(m_singleView.rotation, m_singleView.zoom, aspectRatio);
1795 const Eigen::Matrix4f pvm = cam.pvm();
1796
1797 BrainRenderer::SceneData sceneData;
1798 sceneData.mvp = rhi->clipSpaceCorrMatrix() * toQMatrix(pvm);
1799
1800 sceneData.cameraPos = toQVector3D(cam.eye);
1801 sceneData.lightDir = sceneData.cameraPos.normalized();
1802 sceneData.lightingEnabled = m_lightingEnabled;
1803
1804 // Per-draw overlayMode uniform — the shader selects the vertex colour
1805 // channel (curvature / annotation) so no per-pane vertex buffer
1806 // re-uploads are needed.
1807 sceneData.overlayMode = static_cast<float>(sv.overlayMode);
1808
1809 // Pass 1: Opaque Surfaces (Brain surfaces)
1810 // Use viewport-specific shader from subview
1811 mne::disp::ShaderMode currentShader = sv.brainShader;
1812 const std::string overlayName = visualizationModeName(sv.overlayMode);
1813
1814 // Collect matched brain surface keys for this pane's info panel
1815 mne::StringList drawnKeys;
1816
1817 for (auto it = m_surfaces.begin(); it != m_surfaces.end(); ++it) {
1818 if (!sv.matchesSurfaceType(it->first))
1819 continue;
1820 if (!sv.shouldRenderSurface(it->first))
1821 continue;
1822 drawnKeys.push_back(it->first);
1823 BrainSurface* surf = it->second.get();
1824 pane.opaqueDraws.push_back({surf, currentShader, sceneData.overlayMode, 0.0f,
1825 m_renderer->prepareSurfaceDraw(updates, sceneData, surf)});
1826 }
1827
1828 // Info panel; relayout only on change
1829 std::string info = std::format("Shader: {}\nSurface: {}\nOverlay: {}\nDrawn: {}", shaderModeName(currentShader),
1830 sv.surfaceType, overlayName, drawnKeys.empty() ? std::string("none") : mne::join(drawnKeys, ", "));
1831 const bool infoChanged = (m_viewMode == MultiView)
1832 ? m_overlay.set_text(scene::BrainViewOverlay::Item::ViewportInfoLabel, std::move(info), vp)
1833 : m_overlay.set_text(scene::BrainViewOverlay::Item::SingleViewInfoLabel, std::move(info));
1834 if (infoChanged) {
1835 updateOverlayLayout();
1836 }
1837
1838 // Single pass over m_surfaces categorises non-brain items into opaque
1839 // and transparent draw lists.
1840
1841 // Determine per-viewport field-map visibility
1842 const bool megFieldVisible = sv.visibility.megFieldMap;
1843 const bool eegFieldVisible = sv.visibility.eegFieldMap;
1844 const mne::disp::ShaderMode currentBemShader = sv.bemShader;
1845 const std::string& megFieldKey = m_fieldMapper.megSurfaceKey();
1846 const std::string& eegFieldKey = m_fieldMapper.eegSurfaceKey();
1847
1848 // Brain surfaces were added above; the other opaque items follow them
1849 std::vector<DrawItem> opaqueDraws;
1850 std::vector<DrawItem> transparentDraws;
1851
1852 for (auto it = m_surfaces.begin(); it != m_surfaces.end(); ++it) {
1853 const std::string& key = it->first;
1854 BrainSurface* surf = it->second.get();
1855
1856 if (SubView::isBrainSurfaceKey(key)) {
1857 // Brain surfaces already rendered above
1858 continue;
1859 } else if (key.starts_with("srcsp_") || key.starts_with("dig_")) {
1860 if (!sv.shouldRenderSurface(key))
1861 continue;
1862 if (!surf->isVisible())
1863 continue;
1864 if (key.starts_with("dig_live_t_") || key.starts_with("dig_ray_") || key.starts_with("dig_probe_")) {
1865 QVector3D bmin, bmax;
1866 surf->boundingBox(bmin, bmax);
1867 QVector3D ctr = (bmin + bmax) * 0.5f;
1868 float dist = (sceneData.cameraPos - ctr).lengthSquared();
1869 transparentDraws.push_back({surf, mne::disp::Holographic,
1870 static_cast<float>(mne::disp::ModeScientific),
1871 dist, -1});
1872 } else {
1873 opaqueDraws.push_back({surf, currentShader,
1874 static_cast<float>(mne::disp::ModeScientific),
1875 0.0f, -1});
1876 }
1877 } else {
1878 bool isSensor = key.starts_with("sens_");
1879 bool isBem = key.starts_with("bem_");
1880 if (!isSensor && !isBem)
1881 continue;
1882 if (!sv.shouldRenderSurface(key))
1883 continue;
1884 if (!surf->isVisible())
1885 continue;
1886
1887 QVector3D bmin, bmax;
1888 surf->boundingBox(bmin, bmax);
1889 QVector3D center = (bmin + bmax) * 0.5f;
1890 float dist = (sceneData.cameraPos - center).lengthSquared();
1891
1892 auto mode = isBem ? currentBemShader : mne::disp::Holographic;
1893 float itemOverlay = static_cast<float>(mne::disp::ModeScientific);
1894 if (key == megFieldKey && !megFieldVisible)
1895 itemOverlay = static_cast<float>(mne::disp::ModeSurface);
1896 else if (key == eegFieldKey && !eegFieldVisible)
1897 itemOverlay = static_cast<float>(mne::disp::ModeSurface);
1898
1899 transparentDraws.push_back({surf, mode, itemOverlay, dist, -1});
1900 }
1901 }
1902
1903 // Sort transparent items back-to-front for correct alpha blending
1904 std::sort(transparentDraws.begin(), transparentDraws.end(),
1905 [](const DrawItem& a, const DrawItem& b) { return a.distSq > b.distSq; });
1906
1907 BrainRenderer::SceneData itemData = sceneData;
1908 for (auto& item : opaqueDraws) {
1909 itemData.overlayMode = item.overlayMode;
1910 item.uniformOffset = m_renderer->prepareSurfaceDraw(updates, itemData, item.surface);
1911 }
1912 pane.opaqueDraws.insert(pane.opaqueDraws.end(), opaqueDraws.begin(), opaqueDraws.end());
1913 for (auto& item : transparentDraws) {
1914 itemData.overlayMode = item.overlayMode;
1915 item.uniformOffset = m_renderer->prepareSurfaceDraw(updates, itemData, item.surface);
1916 }
1917 pane.transparentDraws = transparentDraws;
1918 pane.electrodeOffset = m_renderer->prepareElectrodeDraw(updates, sceneData, m_electrodes);
1919
1920 if (sv.visibility.mriSlices) {
1921 for (int i = 0; i < kMaxSliceSlots; ++i) {
1922 if (m_slices[i] && m_sliceVisible[i]) {
1923 pane.sliceOffsets[i] = m_renderer->prepareSliceDraw(updates, sceneData, i);
1924 }
1925 }
1926 }
1927
1928 if (hasVideoOverlay) {
1929 pane.videoOverlayOffset = m_renderer->prepareVideoOverlayDraw(updates, sceneData, m_videoOverlay.get(), videoOverlayTargetSurface);
1930 }
1931
1932 // The head movement path is not connectivity data, so it is not tied to the network
1933 // visibility flag.
1934 if (sv.visibility.dipoles) {
1935 for (auto it = m_itemDipoleMap.begin(); it != m_itemDipoleMap.end(); ++it) {
1936 const int offset = m_renderer->prepareDipoleDraw(updates, sceneData, it->second.get());
1937 if (offset >= 0)
1938 pane.dipoleDraws.push_back({it->second.get(), offset});
1939 }
1940 const int offset = m_renderer->prepareDipoleDraw(updates, sceneData, m_dipoles.get());
1941 if (offset >= 0)
1942 pane.dipoleDraws.push_back({m_dipoles.get(), offset});
1943 }
1944 if (sv.visibility.network) {
1945 pane.networkOffset = m_renderer->prepareNetworkDraw(updates, sceneData, m_network.get());
1946 }
1947 pane.headPathOffset = m_renderer->preparePolylineDraw(updates, sceneData, m_headPath.get());
1948
1949 // Intersection Pointer
1950 if (m_hasIntersection && m_debugPointerSurface) {
1951 BrainRenderer::SceneData debugSceneData = sceneData;
1952 debugSceneData.overlayMode = 0.0f; // pass-through for holographic shell
1953
1954 debugSceneData.mvp = rhi->clipSpaceCorrMatrix() * toQMatrix(pvm * scene::translation_matrix(toEigen(m_lastIntersectionPoint)));
1955
1956 pane.pointerOffset = m_renderer->prepareSurfaceDraw(updates, debugSceneData, m_debugPointerSurface.get());
1957 }
1958 } // End of viewport loop
1959
1960 // The 2-D overlay is painted after the panes updated their info labels
1961 m_renderer->prepareScreenOverlay(rhi, updates, overlayFor(outputSize));
1962
1963 // ── Draw phase: bind and draw only ──────────────────────────────────
1964 m_renderer->beginFrame(cb, updates);
1965
1966 for (const PaneDraws& pane : panes) {
1967 cb->setViewport(pane.viewport);
1968 cb->setScissor(pane.scissor);
1969
1970 for (const auto& item : pane.opaqueDraws)
1971 m_renderer->issueSurfaceDraw(cb, item.surface, item.mode, item.uniformOffset);
1972 m_renderer->issueElectrodeDraw(cb, m_electrodes, pane.electrodeOffset);
1973
1974 // Issue MRI slice draws after opaque surfaces but before the holographic
1975 // brain. Background voxels are discarded in the shader; remaining anatomy
1976 // alpha-blends into the framebuffer. The subsequent holographic additive
1977 // pass (SrcAlpha + One) adds its glow on top without being dimmed.
1978 // depthTest=true, depthWrite=false keeps slices behind opaque geometry.
1979 for (int i = 0; i < kMaxSliceSlots; ++i) {
1980 m_renderer->issueSliceDraw(cb, i, pane.sliceOffsets[i]);
1981 }
1982
1983 // The video overlay goes right after its target surface, or last
1984 bool videoOverlayDrawn = false;
1985 for (const auto& item : pane.transparentDraws) {
1986 m_renderer->issueSurfaceDraw(cb, item.surface, item.mode, item.uniformOffset);
1987 if (item.surface == videoOverlayTargetSurface && !videoOverlayDrawn) {
1988 m_renderer->issueVideoOverlayDraw(cb, m_videoOverlay.get(), videoOverlayTargetSurface, pane.videoOverlayOffset);
1989 videoOverlayDrawn = true;
1990 }
1991 }
1992 if (!videoOverlayDrawn) {
1993 m_renderer->issueVideoOverlayDraw(cb, m_videoOverlay.get(), videoOverlayTargetSurface, pane.videoOverlayOffset);
1994 }
1995
1996 for (const auto& [dipoles, offset] : pane.dipoleDraws)
1997 m_renderer->issueDipoleDraw(cb, dipoles, offset);
1998 m_renderer->issueNetworkDraw(cb, m_network.get(), pane.networkOffset);
1999 m_renderer->issuePolylineDraw(cb, m_headPath.get(), pane.headPathOffset);
2000
2001 m_renderer->issueSurfaceDraw(cb, m_debugPointerSurface.get(), mne::disp::Holographic, pane.pointerOffset);
2002 }
2003
2004 // Labels and pane separators on top of every pane
2005 m_renderer->issueScreenOverlayDraw(cb);
2006
2007 m_renderer->endPass(cb);
2008 m_inRender = false;
2009}

2836 m_itemSurfaceMap[m_hoveredItem]->setSelectedRegion(-1);
2837 }
2838 } else if (m_itemDipoleMap.contains(m_hoveredItem)) {

2880 for (auto it = m_itemSurfaceMap.begin(); it != m_itemSurfaceMap.end();) {
2881 const auto surface = std::ranges::find_if(m_surfaces, [&it](const auto& entry) { return entry.second == it->second; });

2884 ++it;

2889 if (item->parent())
2890 item->parent()->removeRow(item->row());
2891 else
2892 m_model->removeRow(item->row());
2893 }
2894 it = m_itemSurfaceMap.erase(it);
2895 }
2896 for (auto it = m_surfaces.begin(); it != m_surfaces.end();) {
2897 if (matches(it->first))
2898 it = m_surfaces.erase(it);
2899 else
2900 ++it;
2901 }

3447bool BrainViewCore::intersectWorldRay(const QVector3D& origin, const QVector3D& direction, QVector3D& hitPoint) const
3448{
3449 // Test against ALL surfaces (no SubView visibility filter) so
3450 // the optical ray always finds the head even if it's hidden in
3451 // the first SubView of a multi-view layout.
3452 float closestDist = std::numeric_limits<float>::max();
3453 bool found = false;
3454
3455 // Always test the BEM head surface first (even when hidden) — the
3456 // optical projection cares about the outermost physical shell.
3457 auto headIt = m_surfaces.find("bem_head");
3458 if (headIt != m_surfaces.cend() && headIt->second) {
3459 float dist = 0.0f;
3460 int vertexIdx = -1;
3461 if (headIt->second->intersects(origin, direction, dist, vertexIdx) && dist < closestDist) {
3462 closestDist = dist;
3463 hitPoint = origin + dist * direction;
3464 found = true;
3465 }
3466 }
3467
3468 // Fall back to any other visible surface if the head wasn't hit.
3469 if (!found) {
3470 for (auto it = m_surfaces.cbegin(); it != m_surfaces.cend(); ++it) {
3471 const auto& surf = it->second;
3472 if (!surf || !surf->isVisible())
3473 continue;
3474 float dist = 0.0f;
3475 int vertexIdx = -1;
3476 if (surf->intersects(origin, direction, dist, vertexIdx) && dist < closestDist) {
3477 closestDist = dist;
3478 hitPoint = origin + dist * direction;
3479 found = true;
3480 }
3481 }
3482 }
3483 return found;
3484}
