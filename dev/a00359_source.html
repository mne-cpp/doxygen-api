48// Every position of standard_1005.elc in metres, fiducials included, read as mne's _mgh_or_standard does.
49const std::vector<ElectrodePosition>& standard1005Table()
50{
51 static const std::vector<ElectrodePosition> table = [] {
52 std::vector<ElectrodePosition> positions;
53 const auto layout = LayoutLoader::parse_elc(standard1005Elc);
54 if (!layout) {
55 Log::warning("[StandardMontage] the built-in standard_1005 table is malformed");
56 return positions;
57 }
58 positions.reserve(layout->names.size());
59 for (std::size_t k = 0; k < layout->names.size(); ++k)
60 positions.push_back({layout->names[k], layout->positions_3d[k] / 1000.0});
61 return positions;
62 }();
63 return table;
64}

66// The positions of the named electrodes, in the order given.
67template<std::size_t N>
68std::vector<ElectrodePosition> pick(const std::array<std::string_view, N>& names)
69{
70 std::vector<ElectrodePosition> positions;
71 positions.reserve(N);
72 for (const std::string_view name : names) {
73 const auto& table = standard1005Table();
74 const auto it = std::ranges::find(table, name, &ElectrodePosition::name);
75 if (it != table.end())
76 positions.push_back(*it);
77 }
78 return positions;
79}

96std::vector<ElectrodePosition> StandardMontage::montage(System system)
97{
98 switch (system) {
99 case System::Standard_1020:
100 return pick(names1020);
101 case System::Standard_1010:
102 return pick(names1010);
103 case System::Standard_1005: {
104 // mne's standard_1005 montage: every electrode, the fiducials are not channels
105 std::vector<ElectrodePosition> positions = standard1005Table();
106 std::erase_if(positions, [](const ElectrodePosition& electrode) {
107 return electrode.name == "LPA" || electrode.name == "RPA" || electrode.name == "Nz";
108 });
109 return positions;
110 }
111 }
112 return {};
113}
