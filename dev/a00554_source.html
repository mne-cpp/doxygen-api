58void FiffDigitizerData::print() const
59{
60 mne::Log::info("Number of digitizer points: {}", points.size());
61
62 switch (coord_frame) {
63 case FIFFV_COORD_MRI:
64 mne::Log::info("Coord. Frame: FIFFV_COORD_MRI");
65 break;
66 case FIFFV_COORD_HEAD:
67 mne::Log::info("Coord. Frame: FIFFV_COORD_HEAD");
68 break;
69 }
70
71 for (auto& point : points) {
72 if (point.kind == FIFFV_POINT_HPI) {
73 mne::Log::info("HPI Point {} - {}, {}, {}", point.ident, point.r[0], point.r[1], point.r[2]);
74 }
75 }
76
77 mne::Log::info("Number of MRI fiducials: {}", nfids());
78}
