65class SphereGrid
66{
67public:
68 explicit SphereGrid(const MatrixX3d& points)
69 : m_points(points)
70 , m_n(std::max(1, static_cast<int>(std::cbrt(static_cast<double>(points.rows()) / 2.0))))
71 , m_cells(static_cast<std::size_t>(m_n) * m_n * m_n)
72 {
73 for (Index k = 0; k < points.rows(); ++k) {
74 m_cells[cellIndex(cellOf(points.row(k)))].push_back(static_cast<int>(k));
75 }
76 }
77
78 int nearest(const RowVector3d& r) const
79 {
80 const Vector3i c = cellOf(r);
81 const double size = 2.0 / m_n;
82 int best = -1;
83 double bestDist = std::numeric_limits<double>::max();
84 // Grow the searched cube until no cell outside it can hold a closer point.
85 for (int ring = 0; ring <= m_n; ++ring) {
86 for (int i = c.x() - ring; i <= c.x() + ring; ++i) {
87 for (int j = c.y() - ring; j <= c.y() + ring; ++j) {
88 for (int k = c.z() - ring; k <= c.z() + ring; ++k) {
89 const bool onShell = std::max({std::abs(i - c.x()), std::abs(j - c.y()), std::abs(k - c.z())}) == ring;
90 if (!onShell || i < 0 || j < 0 || k < 0 || i >= m_n || j >= m_n || k >= m_n) {
91 continue;
92 }
93 for (int p : m_cells[cellIndex(Vector3i(i, j, k))]) {
94 const double d = (m_points.row(p) - r).squaredNorm();
95 if (d < bestDist) {
96 bestDist = d;
97 best = p;
98 }
99 }
100 }
101 }
102 }
103 if (best >= 0 && std::sqrt(bestDist) <= ring * size) {
104 break;
105 }
106 }
107 return best;
108 }
109
110private:
111 Vector3i cellOf(const RowVector3d& r) const
112 {
113 Vector3i c;
114 for (int d = 0; d < 3; ++d) {
115 c[d] = std::clamp(static_cast<int>((r[d] + 1.0) / 2.0 * m_n), 0, m_n - 1);
116 }
117 return c;
118 }
119
120 std::size_t cellIndex(const Vector3i& c) const
121 {
122 return (static_cast<std::size_t>(c.x()) * m_n + c.y()) * m_n + c.z();
123 }
124
125 const MatrixX3d& m_points;
126 int m_n;
127 std::vector<std::vector<int>> m_cells;
128};

135struct TriangleProjection
136{
137 double p;
138 double q;
139 double dist;
140 double a;
141 double b;
142 double c;
143
144 TriangleProjection(const RowVector3d& r, const RowVector3d& r1, const RowVector3d& r2, const RowVector3d& r3)
145 {
146 const RowVector3d r12 = r2 - r1;
147 const RowVector3d r13 = r3 - r1;
148 const RowVector3d d = r - r1;
149 a = r12.squaredNorm();
150 b = r13.squaredNorm();
151 c = r12.dot(r13);
152 double det = a * b - c * c;
153 if (det == 0.0) {
154 det = 1.0;
155 }
156 const double v1 = d.dot(r12);
157 const double v2 = d.dot(r13);
158 p = (b * v1 - c * v2) / det;
159 q = (a * v2 - c * v1) / det;
160 dist = d.dot(r12.cross(r13).normalized());
161 }
162
163 bool inside() const
164 {
165 return p >= 0.0 && q >= 0.0 && p <= 1.0 && q <= 1.0 && p + q < 1.0;
166 }
167
169 double edgeDistance(double p0, double q0) const
170 {
171 const double dp = p - p0;
172 const double dq = q - q0;
173 return std::sqrt(dp * dp * a + dq * dq * b + dp * dq * c + dist * dist);
174 }
175
177 double nearestEdge(double& edgeP, double& edgeQ) const
178 {
179 const double sides[3][2] = {
180 {std::clamp(p + 0.5 * q * c / a, 0.0, 1.0), 0.0},
181 {0.0, std::clamp(0.5 * ((2.0 * a - c) * (1.0 - p) + (2.0 * b - c) * q) / (a + b - c), 0.0, 1.0)},
182 {0.0, std::clamp(q + 0.5 * p * c / b, 0.0, 1.0)},
183 };
184 double best = std::numeric_limits<double>::max();
185 for (int side = 0; side < 3; ++side) {
186 const double p0 = side == 1 ? 1.0 - sides[1][1] : sides[side][0];
187 const double q0 = sides[side][1];
188 const double d = edgeDistance(p0, q0);
189 if (d < best) {
190 best = d;
191 edgeP = p0;
192 edgeQ = q0;
193 }
194 }
195 return best;
196 }
197};

216MNEMorphMap MNEMorphMap::compute(const MatrixX3f& fromRr, const MatrixX3i& fromTris, const MatrixX3f& toRr)
217{
218 const MatrixX3d from = fromRr.cast<double>().rowwise().normalized();
219 const MatrixX3d to = toRr.cast<double>().rowwise().normalized();
220
221 std::vector<std::vector<int>> trisOfVertex(from.rows());
222 for (Index t = 0; t < fromTris.rows(); ++t) {
223 for (int v = 0; v < 3; ++v) {
224 trisOfVertex[fromTris(t, v)].push_back(static_cast<int>(t));
225 }
226 }
227
228 MNEMorphMap result;
229 result.best.resize(to.rows());
230 const SphereGrid grid(from);
231 std::vector<Triplet<float>> weights;
232 weights.reserve(3 * to.rows());
233 for (Index j = 0; j < to.rows(); ++j) {
234 const RowVector3d r = to.row(j);
235 const int nearest = grid.nearest(r);
236 result.best[j] = nearest;
237 // A triangle around the nearest vertex that contains the projected point wins (smallest plane
238 // distance); only if none does, the nearest point on their sides is used.
239 double bestDist = std::numeric_limits<double>::max();
240 int bestTri = -1;
241 double bestP = 0.0;
242 double bestQ = 0.0;
243 std::vector<std::pair<int, TriangleProjection>> outside;
244 for (int t : trisOfVertex[nearest]) {
245 const TriangleProjection proj(r, from.row(fromTris(t, 0)), from.row(fromTris(t, 1)), from.row(fromTris(t, 2)));
246 if (!proj.inside()) {
247 outside.emplace_back(t, proj);
248 } else if (std::abs(proj.dist) < bestDist) {
249 bestDist = std::abs(proj.dist);
250 bestTri = t;
251 bestP = proj.p;
252 bestQ = proj.q;
253 }
254 }
255 if (bestTri < 0) {
256 for (const auto& [t, proj] : outside) {
257 double p = 0.0;
258 double q = 0.0;
259 const double dist = proj.nearestEdge(p, q);
260 if (dist < bestDist) {
261 bestDist = dist;
262 bestTri = t;
263 bestP = p;
264 bestQ = q;
265 }
266 }
267 }
268 const int row = static_cast<int>(j);
269 weights.emplace_back(row, fromTris(bestTri, 0), static_cast<float>(1.0 - bestP - bestQ));
270 weights.emplace_back(row, fromTris(bestTri, 1), static_cast<float>(bestP));
271 weights.emplace_back(row, fromTris(bestTri, 2), static_cast<float>(bestQ));
272 }
273 SparseMatrix<float> matrix(to.rows(), from.rows());
274 matrix.setFromTriplets(weights.begin(), weights.end());
275 result.map = std::make_unique<FiffSparseMatrix>(std::move(matrix));
276 return result;
277}

281std::optional<MNEMorphMap> MNEMorphMap::read(const std::filesystem::path& path, const std::string& fromSubj, const std::string& toSubj, int hemi)
282{
283 FiffStream stream(path);
284 if (hemi < 0 || hemi > 1 || !stream.open()) {
285 return std::nullopt;
286 }
287 std::optional<FiffTag> tag;
288 for (const std::shared_ptr<FiffDirNode>& node : stream.dirtree()->dir_tree_find(FIFFB_MNE_MORPH_MAP)) {
289 const auto hemiTag = node->find_tag(stream, FIFF_MNE_HEMI);
290 if (!hemiTag || *hemiTag->to_int() != kHemiKind[hemi]) {
291 continue;
292 }
293 const auto fromTag = node->find_tag(stream, FIFF_MNE_MORPH_MAP_FROM);
294 if (!fromTag || fromTag->to_string() != fromSubj) {
295 continue;
296 }
297 const auto toTag = node->find_tag(stream, FIFF_MNE_MORPH_MAP_TO);
298 if (!toTag || toTag->to_string() != toSubj) {
299 continue;
300 }
301 if (tag = node->find_tag(stream, FIFF_MNE_MORPH_MAP); !tag) {
302 break;
303 }
304 MNEMorphMap result;
305 result.map = FiffSparseMatrix::fiff_get_float_sparse_matrix(*tag);
306 if (!result.map) {
307 break;
308 }
309 result.hemi = hemi;
310 result.from_subj = fromSubj;
311 result.to_subj = toSubj;
312 return result;
313 }
314 mne::Log::warning("MNEMorphMap::read - {} hemisphere morph map from {} to {} not found in {}", hemi == 0 ? "left" : "right", fromSubj, toSubj, mne::utf8_from_path(path));
315 return std::nullopt;
316}

320bool MNEMorphMap::write(const std::filesystem::path& path, const std::vector<const MNEMorphMap*>& maps)
321{
322 auto stream = FiffStream::start_file(path);
323 if (!stream) {
324 return false;
325 }
326 for (const MNEMorphMap* morph : maps) {
327 if (!morph || !morph->map || morph->hemi < 0 || morph->hemi > 1) {
328 mne::Log::warning("MNEMorphMap::write - Incomplete morph map");
329 return false;
330 }
331 stream->start_block(FIFFB_MNE_MORPH_MAP);
332 stream->write_string(FIFF_MNE_MORPH_MAP_FROM, morph->from_subj);
333 stream->write_string(FIFF_MNE_MORPH_MAP_TO, morph->to_subj);
334 stream->write_int(FIFF_MNE_HEMI, &kHemiKind[morph->hemi]);
335 stream->write_float_sparse_rcs(FIFF_MNE_MORPH_MAP, morph->map->eigen());
336 stream->end_block(FIFFB_MNE_MORPH_MAP);
337 }
338 stream->end_file();
339 return true;
340}
