62int mne_read_meg_comp_eeg_ch_info_32(const std::filesystem::path& name,
63 std::vector<mne::FiffChInfo>& megp, /* MEG channels */
64 int* nmegp,
65 std::vector<mne::FiffChInfo>& meg_compp,
66 int* nmeg_compp,
67 std::vector<mne::FiffChInfo>& eegp, /* EEG channels */
68 int* neegp,
69 FiffCoordTrans* meg_head_t)
70/*
71 * Read the channel information and split it into three arrays,
72 * one for MEG, one for MEG compensation channels, and one for EEG
73 */
74{
75 FiffStream stream(name);
76
77 std::vector<mne::FiffChInfo> chs;
78 int nchan = 0;
79 std::vector<mne::FiffChInfo> meg;
80 int nmeg = 0;
81 std::vector<mne::FiffChInfo> meg_comp;
82 int nmeg_comp = 0;
83 std::vector<mne::FiffChInfo> eeg;
84 int neeg = 0;
85 std::vector<std::shared_ptr<FiffDirNode>> nodes;
86 std::shared_ptr<FiffDirNode> info;
87 std::optional<FiffTag> t_pTag;
88 mne::FiffChInfo this_ch;
89 FiffCoordTrans t;
90 fiff_int_t kind, pos;
91 int j, k, to_find;
92
93 if (!stream.open()) {
94 stream.close();
95 return FIFF_FAIL;
96 }
97
98 nodes = stream.dirtree()->dir_tree_find(FIFFB_MNE_PARENT_MEAS_FILE);
99
100 if (nodes.size() == 0) {
101 nodes = stream.dirtree()->dir_tree_find(FIFFB_MEAS_INFO);
102 if (nodes.size() == 0) {
103 mne::Log::critical("Could not find the channel information.");
104 stream.close();
105 return FIFF_FAIL;
106 }
107 }
108 info = nodes[0];
109 to_find = 0;
110 for (k = 0; k < info->nent(); k++) {
111 kind = info->dir[k].kind;
112 pos = info->dir[k].pos;
113 switch (kind) {
114 case FIFF_NCHAN:
115 if (t_pTag = stream.read_tag(pos); !t_pTag) {
116 stream.close();
117 return FIFF_FAIL;
118 }
119 nchan = *t_pTag->to_int();
120
121 for (j = 0; j < nchan; j++) {
122 chs.push_back(FiffChInfo());
123 chs[j].scanNo = -1;
124 }
125 to_find = nchan;
126 break;
127
128 case FIFF_COORD_TRANS:
129 if (t_pTag = stream.read_tag(pos); !t_pTag) {
130 stream.close();
131 return FIFF_FAIL;
132 }
133 t = FiffCoordTrans::from_tag(*t_pTag);
134 if (t.from != FIFFV_COORD_DEVICE || t.to != FIFFV_COORD_HEAD)
135 t = FiffCoordTrans();
136 break;
137
138 case FIFF_CH_INFO: /* Information about one channel */
139 if (t_pTag = stream.read_tag(pos); !t_pTag) {
140 stream.close();
141 return FIFF_FAIL;
142 }
143
144 this_ch = t_pTag->to_ch_info();
145 if (this_ch.scanNo <= 0 || this_ch.scanNo > nchan) {
146 mne::Log::critical("FIFF_CH_INFO : scan # out of range {} ({})!", this_ch.scanNo, nchan);
147 stream.close();
148 return FIFF_FAIL;
149 } else
150 chs[this_ch.scanNo - 1] = this_ch;
151 to_find--;
152 break;
153 }
154 }
155 if (to_find != 0) {
156 mne::Log::critical("Some of the channel information was missing.");
157 stream.close();
158 return FIFF_FAIL;
159 }
160 if (t.empty() && meg_head_t != nullptr) {
161 /*
162 * Try again in a more general fashion
163 */
164 t = FiffCoordTrans::read(name, FIFFV_COORD_DEVICE, FIFFV_COORD_HEAD).value_or(FiffCoordTrans());
165 if (t.empty()) {
166 mne::Log::critical("MEG -> head coordinate transformation not found.");
167 stream.close();
168 return FIFF_FAIL;
169 }
170 }
171 /*
172 * Sort out the channels
173 */
174 for (k = 0; k < nchan; k++) {
175 if (chs[k].kind == FIFFV_MEG_CH) {
176 meg.push_back(chs[k]);
177 nmeg++;
178 } else if (chs[k].kind == FIFFV_REF_MEG_CH) {
179 meg_comp.push_back(chs[k]);
180 nmeg_comp++;
181 } else if (chs[k].kind == FIFFV_EEG_CH) {
182 eeg.push_back(chs[k]);
183 neeg++;
184 }
185 }
186 // fiff_close(in);
187 stream.close();
188
189 megp = meg;
190 if (nmegp) {
191 *nmegp = nmeg;
192 }
193
194 meg_compp = meg_comp;
195 if (nmeg_compp) {
196 *nmeg_compp = nmeg_comp;
197 }
198
199 eegp = eeg;
200 if (neegp) {
201 *neegp = neeg;
202 }
203
204 if (meg_head_t == nullptr) {
205 } else
206 *meg_head_t = t;
207
208 return FIFF_OK;
209}

233std::unique_ptr<FiffSparseMatrix> mne_convert_to_sparse(const Eigen::MatrixXf& dense, /* The dense matrix to be converted */
234 float small) /* How small elements should be ignored? */
235/*
236 * Convert a dense matrix to sparse using Eigen's sparseView.
237 */
238{
239 if (small < 0) { /* Automatic scaling */
240 float maxval = dense.cwiseAbs().maxCoeff();
241 if (maxval > 0)
242 small = maxval * std::fabs(small);
243 else
244 small = std::fabs(small);
245 }
246
247 Eigen::SparseMatrix<float> eigenSparse = dense.sparseView(small, 1.0f);
248 eigenSparse.makeCompressed();
249
250 if (eigenSparse.nonZeros() <= 0) {
251 mne::Log::warning("No nonzero elements found.");
252 return nullptr;
253 }
254
255 return std::make_unique<FiffSparseMatrix>(std::move(eigenSparse), FIFFTS_MC_RCS);
256}

258int mne_sparse_vec_mult2_32(FiffSparseMatrix* mat, /* The sparse matrix */
259 float* vector, /* Vector to be multiplied */
260 float* res) /* Result of the multiplication */
261/*
262 * Multiply a vector by a sparse matrix using Eigen.
263 */
264{
265 Eigen::Map<const Eigen::VectorXf> vecIn(vector, mat->cols());
266 Eigen::Map<Eigen::VectorXf> vecOut(res, mat->rows());
267 vecOut = mat->eigen() * vecIn;
268 return 0;
269}

315std::unique_ptr<MNECTFCompDataSet> MNECTFCompDataSet::read(const std::filesystem::path& name)
316/*
317 * Read all CTF compensation data from a given file
318 */
319{
320 FiffStream stream(name);
321
322 std::unique_ptr<MNECTFCompDataSet> set;
323 std::vector<std::shared_ptr<FiffDirNode>> nodes;
324 std::vector<std::shared_ptr<FiffDirNode>> comps;
325 int ncomp;
326 int kind, k;
327 std::optional<FiffTag> t_pTag;
328 std::vector<FiffChInfo> chs;
329 int nch = 0;
330 int calibrated;
331 /*
332 * Read the channel information
333 */
334 {
335 std::vector<FiffChInfo> comp_chs, temp;
336 int ncompch = 0;
337
338 if (mne_read_meg_comp_eeg_ch_info_32(name,
339 chs,
340 &nch,
341 comp_chs,
342 &ncompch,
343 temp,
344 nullptr,
345 nullptr) == FAIL)
346 return nullptr;
347 if (ncompch > 0) {
348 for (k = 0; k < ncompch; k++)
349 chs.push_back(comp_chs[k]);
350 nch = nch + ncompch;
351 }
352 }
353 /*
354 * Read the rest of the stuff
355 */
356 if (!stream.open()) {
357 stream.close();
358 return nullptr;
359 }
360 set = std::make_unique<MNECTFCompDataSet>();
361 /*
362 * Locate the compensation data sets
363 */
364 nodes = stream.dirtree()->dir_tree_find(FIFFB_MNE_CTF_COMP);
365 if (nodes.size() == 0) {
366 stream.close();
367 return set;
368 }
369 comps = nodes[0]->dir_tree_find(FIFFB_MNE_CTF_COMP_DATA);
370 if (comps.size() == 0) {
371 stream.close();
372 return set;
373 }
374 ncomp = static_cast<int>(comps.size());
375 /*
376 * Set the channel info
377 */
378 set->chs = chs;
379 set->nch = nch;
380 /*
381 * Read each data set
382 */
383 for (k = 0; k < ncomp; k++) {
384 auto mat = MNENamedMatrix::read(stream, comps[k], FIFF_MNE_CTF_COMP_DATA);
385 if (!mat) {
386 stream.close();
387 return nullptr;
388 }
389 t_pTag = comps[k]->find_tag(stream, FIFF_MNE_CTF_COMP_KIND);
390 if (t_pTag) {
391 kind = *t_pTag->to_int();
392 } else {
393 stream.close();
394 return nullptr;
395 }
396 t_pTag = comps[k]->find_tag(stream, FIFF_MNE_CTF_COMP_CALIBRATED);
397 if (t_pTag) {
398 calibrated = *t_pTag->to_int();
399 } else
400 calibrated = 0;
401 /*
402 * Add these data to the set
403 */
404 auto one = std::make_unique<MNECTFCompData>();
405 one->data = std::move(mat);
406 one->kind = kind;
407 one->mne_kind = mne_unmap_ctf_comp_kind(one->kind);
408 one->calibrated = calibrated;
409
410 if (one->calibrate(set->chs, set->nch, true) == FAIL) {
411 mne::Log::warning("Warning: Compensation data for '{}' omitted\n", explain_comp(one->kind));
412 } else {
413 set->comps.push_back(std::move(one));
414 set->ncomp++;
415 }
416 }
417#ifdef DEBUG
418 mne::Log::info("{} CTF compensation data sets read from {}\n", set->ncomp, name);
419#endif
420 stream.close();
421 return set;
422}

426int MNECTFCompDataSet::make_comp(const std::vector<FiffChInfo>& chList,
427 int nChan,
428 std::vector<FiffChInfo> compchs,
429 int nCompChan) /* How many of these */
430/*
431 * Make compensation data to apply to a set of channels to yield (or uncompensated) compensated data
432 */
433{
434 Eigen::VectorXi compGrades;
435 int need_comp;
436 int first_comp;
437 MNECTFCompData* this_comp;
438 Eigen::VectorXi comp_sel;
439 StringList names;
440 int j, k, p;
441
442 std::unique_ptr<FiffSparseMatrix> presel;
443 std::unique_ptr<FiffSparseMatrix> postsel;
444 std::unique_ptr<MNENamedMatrix> data;
445
446
447 if (compchs.empty()) {
448 compchs = chList;
449 nCompChan = nChan;
450 }
451 mne::Log::info("Setting up compensation data...\n");
452 if (nChan == 0)
453 return OK;
454 current.reset();
455 compGrades.resize(nChan);
456 for (k = 0, need_comp = 0, first_comp = MNE_CTFV_COMP_NONE; k < nChan; k++) {
457 if (chList[k].kind == FIFFV_MEG_CH) {
458 compGrades[k] = chList[k].chpos.coil_type >> 16;
459 if (compGrades[k] != MNE_CTFV_COMP_NONE) {
460 if (first_comp == MNE_CTFV_COMP_NONE)
461 first_comp = compGrades[k];
462 else {
463 if (compGrades[k] != first_comp) {
464 mne::Log::critical("We do not support nonuniform compensation yet.");
465 return FAIL;
466 }
467 }
468 need_comp++;
469 }
470 } else
471 compGrades[k] = MNE_CTFV_COMP_NONE;
472 }
473 if (need_comp == 0) {
474 mne::Log::info("\tNo compensation set. Nothing more to do.\n");
475 return OK;
476 }
477 mne::Log::info("\t{} out of {} channels have the compensation set.\n", need_comp, nChan);
478 /*
479 * Find the desired compensation data matrix
480 */
481 for (k = 0, this_comp = nullptr; k < this->ncomp; k++) {
482 if (this->comps[k]->mne_kind == first_comp) {
483 this_comp = this->comps[k].get();
484 break;
485 }
486 }
487 if (!this_comp) {
488 mne::Log::critical("Did not find the desired compensation data : {}", explain_comp(map_comp_kind(first_comp)));
489 return FAIL;
490 }
491 mne::Log::info("\tDesired compensation data ({}) found.\n", explain_comp(map_comp_kind(first_comp)));
492 /*
493 * Find the compensation channels
494 */
495 comp_sel.resize(this_comp->data->ncol());
496 for (k = 0; k < this_comp->data->ncol(); k++) {
497 comp_sel[k] = -1;
498 const std::string& name = this_comp->data->collist[k];
499 for (p = 0; p < nCompChan; p++)
500 if ((name == compchs[p].ch_name)) {
501 comp_sel[k] = p;
502 break;
503 }
504 if (comp_sel[k] < 0) {
505 mne::Log::critical("Compensation channel {} not found", name);
506 return FAIL;
507 }
508 }
509 mne::Log::info("\tAll compensation channels found.\n");
510 /*
511 * Create the preselector
512 */
513 {
514 Eigen::MatrixXf sel = Eigen::MatrixXf::Zero(this_comp->data->ncol(), nCompChan);
515 for (j = 0; j < this_comp->data->ncol(); j++)
516 sel(j, comp_sel[j]) = 1.0f;
517 presel = mne_convert_to_sparse(sel, 1e-30f);
518 if (!presel)
519 return FAIL;
520 mne::Log::info("\tPreselector created.\n");
521 }
522 /*
523 * Pick the desired channels
524 */
525 for (k = 0; k < nChan; k++) {
526 if (compGrades[k] != MNE_CTFV_COMP_NONE)
527 names.push_back(chList[k].ch_name);
528 }
529
530 {
531 auto d = this_comp->data->pick(names, {});
532 if (!d)
533 return FAIL;
534 data = std::move(d);
535 }
536 mne::Log::info("\tCompensation data matrix created.\n");
537 /*
538 * Create the postselector
539 */
540 {
541 Eigen::MatrixXf sel = Eigen::MatrixXf::Zero(nChan, data->nrow());
542 for (j = 0, p = 0; j < nChan; j++) {
543 if (compGrades[j] != MNE_CTFV_COMP_NONE)
544 sel(j, p++) = 1.0f;
545 }
546 postsel = mne_convert_to_sparse(sel, 1e-30f);
547 if (!postsel)
548 return FAIL;
549 mne::Log::info("\tPostselector created.\n");
550 }
551 current = std::make_unique<MNECTFCompData>();
552 current->kind = this_comp->kind;
553 current->mne_kind = this_comp->mne_kind;
554 current->data = std::move(data);
555 current->presel = std::move(presel);
556 current->postsel = std::move(postsel);
557
558 mne::Log::info("\tCompensation set up.\n");
559 return OK;
560}

592int MNECTFCompDataSet::apply(bool do_it, Eigen::Ref<Eigen::VectorXf> data, Eigen::Ref<const Eigen::VectorXf> compdata)
593/*
594 * Apply compensation or revert to uncompensated data
595 */
596{
597 MNECTFCompData* this_comp;
598 int ndata = static_cast<int>(data.size());
599 int ncompdata = static_cast<int>(compdata.size());
600
601 if (!current)
602 return OK;
603 this_comp = current.get();
604 /*
605 * Dimension checks
606 */
607 if (this_comp->presel) {
608 if (this_comp->presel->cols() != ncompdata) {
609 mne::Log::critical("Compensation data dimension mismatch. Expected {}, got {} channels.", this_comp->presel->cols(), ncompdata);
610 return FAIL;
611 }
612 } else if (this_comp->data->ncol() != ncompdata) {
613 mne::Log::critical("Compensation data dimension mismatch. Expected {}, got {} channels.", this_comp->data->ncol(), ncompdata);
614 return FAIL;
615 }
616 if (this_comp->postsel) {
617 if (this_comp->postsel->rows() != ndata) {
618 mne::Log::critical("Data dimension mismatch. Expected {}, got {} channels.", this_comp->postsel->rows(), ndata);
619 return FAIL;
620 }
621 } else if (this_comp->data->nrow() != ndata) {
622 mne::Log::critical("Data dimension mismatch. Expected {}, got {} channels.", this_comp->data->nrow(), ndata);
623 return FAIL;
624 }
625 /*
626 * Preselection is optional
627 */
628 const float* presel;
629 if (this_comp->presel) {
630 if (this_comp->presel_data.size() == 0)
631 this_comp->presel_data.resize(this_comp->presel->rows());
632 if (mne_sparse_vec_mult2_32(this_comp->presel.get(), const_cast<float*>(compdata.data()), this_comp->presel_data.data()) != OK)
633 return FAIL;
634 presel = this_comp->presel_data.data();
635 } else
636 presel = compdata.data();
637 /*
638 * This always happens
639 */
640 if (this_comp->comp_data.size() == 0)
641 this_comp->comp_data.resize(this_comp->data->nrow());
642 {
643 Eigen::Map<const Eigen::VectorXf> preselVec(presel, this_comp->data->ncol());
644 Eigen::Map<Eigen::VectorXf> compVec(this_comp->comp_data.data(), this_comp->data->nrow());
645 compVec = this_comp->data->data * preselVec;
646 }
647 /*
648 * Optional postselection
649 */
650 const float* comp;
651 if (!this_comp->postsel)
652 comp = this_comp->comp_data.data();
653 else {
654 if (this_comp->postsel_data.size() == 0)
655 this_comp->postsel_data.resize(this_comp->postsel->rows());
656 if (mne_sparse_vec_mult2_32(this_comp->postsel.get(), this_comp->comp_data.data(), this_comp->postsel_data.data()) != OK)
657 return FAIL;
658 comp = this_comp->postsel_data.data();
659 }
660 /*
661 * Compensate or revert compensation?
662 */
663 Eigen::Map<const Eigen::VectorXf> compVec(comp, ndata);
664 if (do_it)
665 data -= compVec;
666 else
667 data += compVec;
668 return OK;
669}

673int MNECTFCompDataSet::apply_transpose(bool do_it, Eigen::MatrixXf& data)
674/*
675 * Apply compensation or revert to uncompensated data
676 */
677{
678 MNECTFCompData* this_comp;
679 int ndata = static_cast<int>(data.rows());
680 int ncompdata = ndata;
681
682 if (!current)
683 return OK;
684 this_comp = current.get();
685 /*
686 * Dimension checks
687 */
688 if (this_comp->presel) {
689 if (this_comp->presel->cols() != ncompdata) {
690 mne::Log::critical("Compensation data dimension mismatch. Expected {}, got {} channels.", this_comp->presel->cols(), ncompdata);
691 return FAIL;
692 }
693 } else if (this_comp->data->ncol() != ncompdata) {
694 mne::Log::critical("Compensation data dimension mismatch. Expected {}, got {} channels.", this_comp->data->ncol(), ncompdata);
695 return FAIL;
696 }
697 if (this_comp->postsel) {
698 if (this_comp->postsel->rows() != ndata) {
699 mne::Log::critical("Data dimension mismatch. Expected {}, got {} channels.", this_comp->postsel->rows(), ndata);
700 return FAIL;
701 }
702 } else if (this_comp->data->nrow() != ndata) {
703 mne::Log::critical("Data dimension mismatch. Expected {}, got {} channels.", this_comp->data->nrow(), ndata);
704 return FAIL;
705 }
706 /*
707 * Preselection is optional — sparse matrix * data
708 */
709 Eigen::MatrixXf preselMat;
710 if (this_comp->presel) {
711 preselMat = this_comp->presel->eigen() * data;
712 } else {
713 preselMat = data;
714 }
715 /*
716 * Compensation: comp = data_matrix * preselMat
717 * data_matrix is float** (contiguous row-major via mne_cmatrix)
718 */
719 Eigen::MatrixXf comp = this_comp->data->data * preselMat;
720 /*
721 * Optional postselection — sparse matrix * comp
722 */
723 if (this_comp->postsel) {
724 comp = this_comp->postsel->eigen() * comp;
725 }
726 /*
727 * Compensate or revert compensation?
728 */
729 if (do_it)
730 data -= comp;
731 else
732 data += comp;
733 return OK;
734}

739{
740 int res = MNE_CTFV_NOGRAD;
741 int first_comp, comp;

800 std::vector<FiffChInfo>& chList,
801 int nchan,
802 std::vector<FiffChInfo> comp_chs,
803 int ncomp_chan) /* How many */
804/*
805 * Make data which has the third-order gradient compensation applied
806 */
807{
808 int k;
809 int have_comp_chs;
810 int comp_was = MNE_CTFV_COMP_UNKNOWN;
811
812 if (comp_chs.empty()) {
813 comp_chs = chList;
814 ncomp_chan = nchan;
815 }
816 undo.reset();
817 current.reset();
818 for (k = 0, have_comp_chs = 0; k < ncomp_chan; k++)
819 if (comp_chs[k].kind == FIFFV_REF_MEG_CH)
820 have_comp_chs++;
821 if (have_comp_chs == 0 && compensate_to != MNE_CTFV_NOGRAD) {
822 mne::Log::warning("No compensation channels in these data.");
823 return FAIL;
824 }
825 /*
826 * Update the 'current' field to reflect the compensation possibly present in the data now
827 */
828 if (make_comp(chList, nchan, comp_chs, ncomp_chan) == FAIL)
829 return FAIL;
830 /*
831 * Are we there already?
832 */
833 if (current && current->mne_kind == compensate_to) {
834 mne::Log::info("No further compensation necessary (comp = {})\n", explain_comp(current->kind));
835 current.reset();
836 return OK;
837 }
838 undo = std::move(current);
839 if (compensate_to == MNE_CTFV_NOGRAD) {
840 mne::Log::info("No compensation was requested.\n");
841 set_comp(chList, nchan, compensate_to);
842 return OK;
843 }
844 if (set_comp(chList, nchan, compensate_to) > 0) {
845 if (undo)
846 comp_was = undo->mne_kind;
847 else
848 comp_was = MNE_CTFV_NOGRAD;
849 if (make_comp(chList, nchan, comp_chs, ncomp_chan) == FAIL) {
850 if (comp_was != MNE_CTFV_COMP_UNKNOWN)
851 set_comp(chList, nchan, comp_was);
852 return FAIL;
853 }
854 mne::Log::info("Compensation set up as requested ({} -> {}).\n", explain_comp(map_comp_kind(comp_was)), explain_comp(current->kind));
855 }
856 return OK;
857}
