55static std::string readWord(std::istream& in)
56{
57 std::string word;
58 in >> std::ws;
59 if (in.peek() == '"') {
60 in.get();
61 std::getline(in, word, '"');
62 } else {
63 in >> word;
64 }

68FwdCoil* FwdCoilSet::fwd_add_coil_to_set(int type, int coil_class, int acc, int np, float size, float base, const std::string& desc)
69{
70 if (np <= 0) {
71 mne::Log::warning("Number of integration points should be positive (type = {} acc = {})", type, acc);
72 return nullptr;
73 }
74 if (!(acc == FWD_COIL_ACCURACY_POINT ||
75 acc == FWD_COIL_ACCURACY_NORMAL ||
76 acc == FWD_COIL_ACCURACY_ACCURATE)) {
77 mne::Log::warning("Illegal accuracy (type = {} acc = {})", type, acc);
78 return nullptr;
79 }
80 if (!(coil_class == FWD_COILC_MAG ||
81 coil_class == FWD_COILC_AXIAL_GRAD ||
82 coil_class == FWD_COILC_PLANAR_GRAD ||
83 coil_class == FWD_COILC_AXIAL_GRAD2)) {
84 mne::Log::warning("Illegal coil class (type = {} acc = {} class = {})", type, acc, coil_class);
85 return nullptr;
86 }
87
88 coils.push_back(std::make_unique<FwdCoil>(np));
89 FwdCoil* def = coils.back().get();
90
91 def->type = type;
92 def->coil_class = coil_class;
93 def->accuracy = acc;
94 def->np = np;
95 def->size = size;
96 def->base = base;
97 if (!desc.empty())
98 def->desc = desc;
99 return def;
100}

119std::unique_ptr<FwdCoil> FwdCoilSet::create_meg_coil(const FiffChInfo& ch, int acc, const FiffCoordTrans& t)
120{
121 if (ch.kind != FIFFV_MEG_CH && ch.kind != FIFFV_REF_MEG_CH) {
122 mne::Log::warning("{} is not a MEG channel. Cannot create a coil definition.", ch.ch_name);
123 return nullptr;
124 }
125 /*
126 * Simple linear search from the coil definitions
127 */
128 FwdCoil* def = nullptr;
129 for (int k = 0; k < this->ncoil(); k++) {
130 if ((this->coils[k]->type == (ch.chpos.coil_type & 0xFFFF)) &&
131 this->coils[k]->accuracy == acc) {
132 def = this->coils[k].get();
133 }
134 }
135 if (!def) {
136 mne::Log::warning("Desired coil definition not found (type = {} acc = {})", ch.chpos.coil_type, acc);
137 return nullptr;
138 }
139 /*
140 * Create the result
141 */
142 auto res = std::make_unique<FwdCoil>(def->np);
143
144 res->chname = ch.ch_name;
145 if (!def->desc.empty())
146 res->desc = def->desc;
147 res->coil_class = def->coil_class;
148 res->accuracy = def->accuracy;
149 res->base = def->base;
150 res->size = def->size;
151 res->type = ch.chpos.coil_type;
152
153 res->r0 = ch.chpos.r0;
154 res->ex = ch.chpos.ex;
155 res->ey = ch.chpos.ey;
156 res->ez = ch.chpos.ez;
157 /*
158 * Apply a coordinate transformation if so desired
159 */
160 if (!t.empty()) {
161 FiffCoordTrans::apply_trans(res->r0.data(), t, FIFFV_MOVE);
162 FiffCoordTrans::apply_trans(res->ex.data(), t, FIFFV_NO_MOVE);
163 FiffCoordTrans::apply_trans(res->ey.data(), t, FIFFV_NO_MOVE);
164 FiffCoordTrans::apply_trans(res->ez.data(), t, FIFFV_NO_MOVE);
165 res->coord_frame = t.to;
166 } else
167 res->coord_frame = FIFFV_COORD_DEVICE;
168
169 for (int p = 0; p < res->np; p++) {
170 res->w[p] = def->w[p];
171 res->rmag.row(p) = (res->r0 + def->rmag(p, 0) * res->ex + def->rmag(p, 1) * res->ey + def->rmag(p, 2) * res->ez).transpose();
172 res->cosmag.row(p) = (def->cosmag(p, 0) * res->ex + def->cosmag(p, 1) * res->ey + def->cosmag(p, 2) * res->ez).transpose();
173 }
174 return res;
175}

179std::unique_ptr<FwdCoilSet> FwdCoilSet::create_meg_coils(const std::vector<mne::FiffChInfo>& chs,
180 int nch,
181 int acc,
182 const FiffCoordTrans& t)
183{
184 auto res = std::make_unique<FwdCoilSet>();
185
186 for (int k = 0; k < nch; k++) {
187 auto next = this->create_meg_coil(chs.at(k), acc, t);
188 if (!next)
189 return nullptr;
190 res->coils.push_back(std::move(next));
191 }
192 if (!t.empty())
193 res->coord_frame = t.to;
194 return res;
195}

218std::unique_ptr<FwdCoilSet> FwdCoilSet::read_coil_defs(const std::filesystem::path& name)
219{
220 std::ifstream file(name);
221 if (!file) {
222 mne::Log::warning("FwdCoilSet::read_coil_defs - Cannot open {}", mne::utf8_from_path(name));
223 return nullptr;
224 }
225
226 // Read file content, stripping comments
227 std::string content;
228 for (std::string line; std::getline(file, line);) {
229 content.append(line, 0, line.find('#'));
230 content.push_back('\n');
231 }
232 std::istringstream in(content);
233 in.imbue(std::locale::classic());
234
235 auto res = std::make_unique<FwdCoilSet>();
236 while (in >> std::ws, !in.eof()) {
237 /*
238 * Read basic info
239 */
240 int coil_class;
241 if (!(in >> coil_class))
242 break;
243
244 int type, acc, np;
245 float size, base;
246
247 if (!(in >> type >> acc >> np >> size >> base)) {
248 mne::Log::warning("FwdCoilSet::read_coil_defs - Error reading coil header");
249 return nullptr;
250 }
251
252 const std::string desc = readWord(in);
253 if (desc.empty()) {
254 mne::Log::warning("FwdCoilSet::read_coil_defs - Missing coil description");
255 return nullptr;
256 }
257
258 FwdCoil* def = res->fwd_add_coil_to_set(type, coil_class, acc, np, size, base, desc);
259 if (!def)
260 return nullptr;
261
262 for (int p = 0; p < def->np; p++) {
263 /*
264 * Read and verify data for each integration point
265 */
266 if (!(in >> def->w[p] >> def->rmag(p, 0) >> def->rmag(p, 1) >> def->rmag(p, 2) >> def->cosmag(p, 0) >> def->cosmag(p, 1) >> def->cosmag(p, 2))) {
267 mne::Log::warning("FwdCoilSet::read_coil_defs - Error reading integration point {}", p);
268 return nullptr;
269 }
270
271 if (def->pos(p).norm() > BIG) {
272 mne::Log::warning("Unreasonable integration point: {:f} {:f} {:f} mm (coil type = {} acc = {})", 1000 * def->rmag(p, 0), 1000 * def->rmag(p, 1), 1000 * def->rmag(p, 2), def->type, def->accuracy);
273 return nullptr;
274 }
275 float cosmagNorm = def->dir(p).norm();
276 if (cosmagNorm <= 0) {
277 mne::Log::warning("Unreasonable normal: {:f} {:f} {:f} (coil type = {} acc = {})", def->cosmag(p, 0), def->cosmag(p, 1), def->cosmag(p, 2), def->type, def->accuracy);
278 return nullptr;
279 }
280 def->cosmag.row(p).normalize();
281 }
282 }
283
284 mne::Log::info("{} coil definitions read", res->ncoil());
285 return res;
286}

290std::unique_ptr<FwdCoilSet> FwdCoilSet::dup_coil_set(const FiffCoordTrans& t) const
291{
292 std::unique_ptr<FwdCoilSet> res;
293
294 if (!t.empty()) {
295 if (this->coord_frame != t.from) {
296 mne::Log::warning("Coordinate frame of the transformation does not match the coil set in fwd_dup_coil_set");
297 return nullptr;
298 }
299 }
300 res = std::make_unique<FwdCoilSet>();
301 if (!t.empty())
302 res->coord_frame = t.to;
303 else
304 res->coord_frame = this->coord_frame;
305
306 res->coils.reserve(this->ncoil());
307
308 for (int k = 0; k < this->ncoil(); k++) {
309 auto coil = std::make_unique<FwdCoil>(*(this->coils[k]));
310 /*
311 * Optional coordinate transformation
312 */
313 if (!t.empty()) {
314 FiffCoordTrans::apply_trans(coil->r0.data(), t, FIFFV_MOVE);
315 FiffCoordTrans::apply_trans(coil->ex.data(), t, FIFFV_NO_MOVE);
316 FiffCoordTrans::apply_trans(coil->ey.data(), t, FIFFV_NO_MOVE);
317 FiffCoordTrans::apply_trans(coil->ez.data(), t, FIFFV_NO_MOVE);
318
319 for (int p = 0; p < coil->np; p++) {
320 FiffCoordTrans::apply_trans(&coil->rmag(p, 0), t, FIFFV_MOVE);
321 FiffCoordTrans::apply_trans(&coil->cosmag(p, 0), t, FIFFV_NO_MOVE);
322 }
323 coil->coord_frame = t.to;
324 }
325 res->coils.push_back(std::move(coil));
326 }
327 return res;
328}
