64int FwdCompData::fwd_comp_field(const Eigen::Vector3f& rd, const Eigen::Vector3f& Q, FwdCoilSet& coils, Eigen::Ref<Eigen::VectorXf> res, void* client)
65{
66 FwdCompData* comp = static_cast<FwdCompData*>(client);
67
68 if (!comp->field) {
69 mne::Log::warning("Field computation function is missing in fwd_comp_field");
70 return FAIL;
71 }
72 /*
73 * First compute the field in the primary set of coils
74 */
75 if (comp->field(rd, Q, coils, res, comp->client) == FAIL)
76 return FAIL;
77 /*
78 * Compensation needed?
79 */
80 if (!comp->comp_coils || comp->comp_coils->ncoil() <= 0 || !comp->set || !comp->set->current)
81 return OK;
82 /*
83 * Workspace needed?
84 */
85 if (comp->work.size() == 0)
86 comp->work.resize(comp->comp_coils->ncoil());
87 /*
88 * Compute the field in the compensation coils
89 */
90 if (comp->field(rd, Q, *comp->comp_coils, comp->work, comp->client) == FAIL)
91 return FAIL;
92 /*
93 * Compute the compensated field
94 */
95 return comp->set->apply(true, res, comp->work);
96}

100int FwdCompData::fwd_make_ctf_comp_coils(MNECTFCompDataSet* set,
101 FwdCoilSet* coils,
102 FwdCoilSet* comp_coils)
103{
104 std::vector<FiffChInfo> chs;
105 std::vector<FiffChInfo> compchs;
106 int nchan = 0;
107 int ncomp = 0;
108 int k, res;
109
110 if (!set)
111 return OK;
112 if (!coils || coils->ncoil() <= 0) {
113 mne::Log::warning("Coil data missing in fwd_make_ctf_comp_coils");
114 return FAIL;
115 }
116
117 for (k = 0; k < coils->ncoil(); k++) {
118 chs.push_back(FiffChInfo());
119 FwdCoil* coil = coils->coils[k].get();
120 chs[k].ch_name = coil->chname;
121 chs[k].chpos.coil_type = coil->type;
122 chs[k].kind = (coil->coil_class == FWD_COILC_EEG) ? FIFFV_EEG_CH : FIFFV_MEG_CH;
123 }
124 nchan = coils->ncoil();
125 if (comp_coils && comp_coils->ncoil() > 0) {
126 for (k = 0; k < comp_coils->ncoil(); k++) {
127 compchs.push_back(FiffChInfo());
128 FwdCoil* coil = comp_coils->coils[k].get();
129 compchs[k].ch_name = coil->chname;
130 compchs[k].chpos.coil_type = coil->type;
131 compchs[k].kind = (coil->coil_class == FWD_COILC_EEG) ? FIFFV_EEG_CH : FIFFV_MEG_CH;
132 }
133 ncomp = comp_coils->ncoil();
134 }
135 res = set->make_comp(chs, nchan, compchs, ncomp);
136
137 return res;
138}

171int FwdCompData::fwd_comp_field_vec(const Eigen::Vector3f& rd, FwdCoilSet& coils, Eigen::Ref<Eigen::MatrixXf> res, void* client)
172{
173 FwdCompData* comp = static_cast<FwdCompData*>(client);
174
175 if (!comp->vec_field) {
176 mne::Log::warning("Field computation function is missing in fwd_comp_field_vec");
177 return FAIL;
178 }
179 /*
180 * First compute the field in the primary set of coils
181 */
182 if (comp->vec_field(rd, coils, res, comp->client) == FAIL)
183 return FAIL;
184 /*
185 * Compensation needed?
186 */
187 if (!comp->comp_coils || comp->comp_coils->ncoil() <= 0 || !comp->set || !comp->set->current)
188 return OK;
189 /*
190 * Need workspace?
191 */
192 if (comp->vec_work.size() == 0)
193 comp->vec_work.resize(3, comp->comp_coils->ncoil());
194 /*
195 * Compute the field at the compensation sensors
196 */
197 if (comp->vec_field(rd, *comp->comp_coils, comp->vec_work, comp->client) == FAIL)
198 return FAIL;
199 /*
200 * Compute the compensated field of three orthogonal dipoles
201 */
202 for (int k = 0; k < 3; k++) {
203 Eigen::VectorXf resRow = res.row(k).transpose();
204 Eigen::VectorXf workRow = comp->vec_work.row(k).transpose();
205 if (comp->set->apply(true, resRow, workRow) == FAIL)
206 return FAIL;
207 res.row(k) = resRow.transpose();
208 }
209 return OK;
210}

228 * Compensation needed?

240 * Compute the field in the compensation coils
241 */
242 Eigen::VectorXf vw0 = comp->vec_work.row(0).transpose();
243 Eigen::VectorXf vw1 = comp->vec_work.row(1).transpose();
244 Eigen::VectorXf vw2 = comp->vec_work.row(2).transpose();

248 comp->vec_work.row(1) = vw1.transpose();
