241 std::memcpy(cache.image.rgba.data() + static_cast<std::size_t>(y) * rowBytes, rgba.constScanLine(y), rowBytes);

243 cache.source = source;
244 cache.revision = ++imageRevision;
245 }
246 return {&cache.image, cache.revision};
247 }

398void BrainViewCore::setupInteraction()
399{
400 std::vector<viz::PaneCamera*> panes;
401 panes.reserve(m_subViews.size());
402 for (SubView& subView : m_subViews) {
403 panes.push_back(&subView);
404 }
405 m_interaction.attach_cameras(&m_singleView, std::move(panes));
406
407 viz::BrainViewInteraction::Effects effects;
408 effects.redraw = [this] {
409 cameraChanged();
410 };
411 effects.layout_changed = [this] {
412 updateOverlayLayout();
413 cameraChanged();
414 };
415 effects.settings_changed = [this] {
416 saveMultiViewSettings();
417 };
418 effects.cursor_changed = [this](viz::CursorShape shape) {
419 setHostCursor(shape);
420 };
421 effects.viewport_selected = [this](int viewport) {
422 if (viewport != m_visualizationEditTarget) {
423 setVisualizationEditTarget(viewport);
424 }
425 };
426 // A hidden surface shows no labels, so its name labels take no clicks
427 effects.hits_viewport_label = [this](int viewport, viz::Point pos) {
428 const viz::BrainViewOverlay::Label* name = m_overlay.item(viz::BrainViewOverlay::Item::ViewportNameLabel, viewport);
429 return name && name->visible && m_host && m_host->isVisible() && toQRect(name->geometry).contains(toQPoint(pos));
430 };
431 effects.preset_menu_requested = [this](int viewport, viz::Point pos) {
432 if (m_host) {
433 m_host->requestViewportPresetMenu(viewport, toQPoint(pos));
434 }
435 };
436 effects.hovered = [this](viz::Point pos) {
437 castRay(toQPoint(pos));
438 };
439 effects.clicked = [this](viz::Point pos) {
440 castRay(toQPoint(pos));
441 if (const mne::IeegContact* contact = m_electrodes ? m_electrodes->contactAt(m_electrodes->hoveredContact()) : nullptr) {
442 emit electrodeContactClicked(QString::fromStdString(contact->name), QVector3D(contact->position.x(), contact->position.y(), contact->position.z()));
443 } else if (m_hasIntersection) {
444 emit surfacePointClicked(m_lastIntersectionPoint);
445 }
446 };
447 effects.double_clicked = [this](viz::Point pos) {
448 castRay(toQPoint(pos));
449 if (m_hasIntersection) {
450 emit surfacePointDoubleClicked(m_lastIntersectionPoint);
451 return true;
452 }
453 return false;
454 };
455 effects.snapshot_requested = [this] {
456 saveSnapshot();
457 };
458 m_interaction.set_effects(std::move(effects));
459}

480void BrainViewCore::setModel(BrainTreeModel* model)
481{
482 if (model == m_model)
483 return;
484
485 if (m_model) {
486 disconnect(m_model, nullptr, this, nullptr);
487 for (auto it = m_itemSurfaceMap.cbegin(); it != m_itemSurfaceMap.cend(); ++it) {
488 for (auto sit = m_surfaces.begin(); sit != m_surfaces.end();) {
489 sit = (sit->second == it->second) ? m_surfaces.erase(sit) : std::next(sit);
490 }
491 }
492 m_itemSurfaceMap.clear();
493 m_itemDipoleMap.clear();
494 m_hoveredItem = nullptr;
495 m_activeSurface.reset();
496 }
497
498 m_model = model;
499 if (!m_model) {
500 updateSceneBounds();
501 m_sceneDirty = true;
502 requestUpdate();
503 return;
504 }
505 connect(m_model, &BrainTreeModel::rowsInserted, this, &BrainViewCore::onRowsInserted);
506 connect(m_model, &BrainTreeModel::dataChanged, this, &BrainViewCore::onDataChanged);
507
508 if (m_model->rowCount() > 0) {
509 onRowsInserted(QModelIndex(), 0, m_model->rowCount() - 1);
510 } else {
511 updateSceneBounds();
512 m_sceneDirty = true;
513 requestUpdate();
514 }
515}

527void BrainViewCore::onRowsInserted(const QModelIndex& parent, int first, int last)
528{
529 if (!m_model)
530 return;
531
532 for (int i = first; i <= last; ++i) {
533 QModelIndex index = m_model->index(i, 0, parent);
534 QStandardItem* item = m_model->itemFromIndex(index);
535
536 AbstractTreeItem* absItem = dynamic_cast<AbstractTreeItem*>(item);
537
538 // Handle FsSurface Items
539 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::SurfaceItem)) {
540 SurfaceTreeItem* surfItem = static_cast<SurfaceTreeItem*>(absItem);
541 auto brainSurf = std::make_shared<BrainSurface>();
542
543 // Load geometry from item
544 brainSurf->fromSurface(surfItem->surfaceData());
545
546 // Determine Hemisphere from Parent
547 if (absItem->parent()) {
548 QString parentText = absItem->parent()->text();
549 if (parentText == "lh")
550 brainSurf->setHemi(0);
551 else if (parentText == "rh")
552 brainSurf->setHemi(1);
553 }
554
555 // Set properties
556 brainSurf->setVisible(surfItem->isVisible());
557
558 // Brain surfaces (pial, white, inflated, etc.) are brain tissue
559 brainSurf->setTissueType(BrainSurface::TissueBrain);
560
561 m_itemSurfaceMap[item] = brainSurf;
562
563 // Key generation: "hemi_type" e.g. "lh_pial"
564 std::string key;
565 if (absItem->parent()) {
566 key = (absItem->parent()->text() + "_" + surfItem->text()).toStdString();
567 } else {
568 key = surfItem->text().toStdString();
569 }
570 m_surfaces[key] = brainSurf;
571
572 // Check for annotations
573 if (!surfItem->annotationData().empty()) {
574 brainSurf->addAnnotation(surfItem->annotationData());
575 }
576
577 // Set active if first
578 if (!m_activeSurface) {
579 m_activeSurface = brainSurf;
580 m_activeSurfaceType = surfItem->text().toStdString();
581 }
582 }
583 // Check for BEM Item (using dynamic_cast for safety)
584 BemTreeItem* bemItem = dynamic_cast<BemTreeItem*>(absItem);
585 if (bemItem) {
586 const mne::MNEBemSurface& bemSurfData = bemItem->bemSurfaceData();
587
588 auto brainSurf = std::make_shared<BrainSurface>();
589
590 // Load BEM geometry with color from item
591 brainSurf->fromBemSurface(bemSurfData, toColor(itemColor(*bemItem)));
592
593 brainSurf->setVisible(bemItem->isVisible());
594
595 // Set tissue type based on surface name
596 QString surfName = bemItem->text().toLower();
597 if (surfName.contains("head") || surfName.contains("skin") || surfName.contains("scalp")) {
598 brainSurf->setTissueType(BrainSurface::TissueSkin);
599 } else if (surfName.contains("outer") && surfName.contains("skull")) {
600 brainSurf->setTissueType(BrainSurface::TissueOuterSkull);
601 } else if (surfName.contains("inner") && surfName.contains("skull")) {
602 brainSurf->setTissueType(BrainSurface::TissueInnerSkull);
603 } else if (surfName.contains("skull")) {
604 brainSurf->setTissueType(BrainSurface::TissueOuterSkull); // Default skull to outer
605 } else if (surfName.contains("brain")) {
606 brainSurf->setTissueType(BrainSurface::TissueBrain);
607 }
608
609 m_itemSurfaceMap[item] = brainSurf;
610
611 // Legacy map support (Use item text e.g. "bem_head")
612 m_surfaces["bem_" + bemItem->text().toStdString()] = brainSurf;
613 }
614
615 // Handle Sensor Items
616 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::SensorItem)) {
617 SensorTreeItem* sensItem = static_cast<SensorTreeItem*>(absItem);
618
619 std::shared_ptr<BrainSurface> brainSurf;
620
621 QString parentText = "";
622 if (sensItem->parent())
623 parentText = sensItem->parent()->text();
624
625 if (parentText.contains("MEG/Grad") && sensItem->hasOrientation()) {
626 brainSurf = MeshFactory::createBarbell(toEigen(sensItem->position()), toEigen(sensItem->orientation()),
627 toColor(itemColor(*sensItem)), sensItem->scale());
628 } else if (parentText.contains("MEG/Mag") && sensItem->hasOrientation()) {
629 brainSurf = MeshFactory::createPlate(toEigen(sensItem->position()), toEigen(sensItem->orientation()),
630 toColor(itemColor(*sensItem)), sensItem->scale());
631 } else {
632 // EEG and other sensors: smooth icosphere
633 brainSurf = MeshFactory::createSphere(toEigen(sensItem->position()), sensItem->scale(),
634 toColor(itemColor(*sensItem)));
635 }
636
637 brainSurf->setVisible(sensItem->isVisible());
638 m_itemSurfaceMap[item] = brainSurf;
639
640
641 // Legacy map support
642 const std::string keyPrefix = mne::viz::surface_keys::sensorParentToKeyPrefix(parentText.toStdString());
643
644 const std::string key = std::format("{}{}_{}", keyPrefix, sensItem->text().toStdString(), reinterpret_cast<std::uintptr_t>(sensItem));
645 m_surfaces[key] = brainSurf;
646 }
647
648 // Handle Dipole Items
649 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::DipoleItem)) {
650 DipoleTreeItem* dipItem = static_cast<DipoleTreeItem*>(absItem);
651 auto dipObject = std::make_shared<DipoleObject>();
652 dipObject->load(dipItem->ecdSet());
653 dipObject->setVisible(dipItem->isVisible());
654 dipObject->applyTransform(toEigen(itemTransform(item)));
655
656 m_itemDipoleMap[item] = dipObject;
657 }
658
659 // Handle Source Space Items (one item per hemisphere, batched mesh)
660 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::SourceSpaceItem)) {
661 SourceSpaceTreeItem* srcItem = static_cast<SourceSpaceTreeItem*>(absItem);
662 const std::vector<QVector3D>& positions = srcItem->positions();
663 if (positions.empty())
664 continue;
665
666 auto brainSurf = MeshFactory::createBatchedSpheres(toEigen(positions), srcItem->scale(),
667 toColor(itemColor(*srcItem)));
668 brainSurf->setVisible(srcItem->isVisible());
669 m_itemSurfaceMap[item] = brainSurf;
670
671 const std::string key = "srcsp_" + srcItem->text().toStdString();
672 m_surfaces[key] = brainSurf;
673 }
674
675 // Handle Digitizer Items (batched sphere mesh per category)
676 if (absItem && absItem->type() == AbstractTreeItem::itemTypeId(AbstractTreeItem::DigitizerItem)) {
677 DigitizerTreeItem* digItem = static_cast<DigitizerTreeItem*>(absItem);
678 const std::vector<QVector3D>& positions = digItem->positions();
679 if (positions.empty())
680 continue;
681
682 auto brainSurf = MeshFactory::createBatchedSpheres(toEigen(positions), digItem->scale(),
683 toColor(itemColor(*digItem)));
684 brainSurf->setVisible(digItem->isVisible());
685
686 m_itemSurfaceMap[item] = brainSurf;
687
688 // Category name for surface-map key. A monotonic counter
689 // suffix guarantees uniqueness so that multiple items of the
690 // same PointKind each get their own entry in m_surfaces.
691 // shouldRenderSurface() matches by prefix ("dig_cardinal",
692 // "dig_hpi", …) so the suffix is transparent to visibility.
693 static int s_digKeyCounter = 0;
694 std::string catName;
695 switch (digItem->pointKind()) {
696 case DigitizerTreeItem::Cardinal:
697 catName = "cardinal";
698 break;
699 case DigitizerTreeItem::HPI:
700 catName = "hpi";
701 break;
702 case DigitizerTreeItem::EEG:
703 catName = "eeg";
704 break;
705 case DigitizerTreeItem::Extra:
706 catName = "extra";
707 break;
708 default:
709 catName = digItem->text().toLower().replace(' ', '_').toStdString();
710 break;
711 }
712 const std::string key = std::format("dig_{}_{}", catName, s_digKeyCounter++);
713 m_surfaces[key] = brainSurf;
714 }
715
716 if (m_itemSurfaceMap.contains(item))
717 m_itemSurfaceMap[item]->applyTransform(toEigen(itemTransform(item)));
718
719 // Check children recursively
720 if (m_model->hasChildren(index)) {
721 onRowsInserted(index, 0, m_model->rowCount(index) - 1);
722 }
723 }
724 updateInflatedSurfaceTransforms();
725 updateSceneBounds();

727 m_sceneDirty = true;

731void BrainViewCore::onDataChanged(const QModelIndex& topLeft, const QModelIndex& bottomRight, const QList<int>& roles)
732{
733 bool transformChanged = false;
734 for (int i = topLeft.row(); i <= bottomRight.row(); ++i) {
735 QModelIndex index = m_model->index(i, 0, topLeft.parent());
736 QStandardItem* item = m_model->itemFromIndex(index);
737
738 if (m_itemSurfaceMap.contains(item)) {
739 auto surf = m_itemSurfaceMap[item];
740
741 AbstractTreeItem* absItem = dynamic_cast<AbstractTreeItem*>(item);
742 if (absItem) {
743 if (roles.contains(AbstractTreeItem::VisibleRole)) {
744 surf->setVisible(absItem->isVisible());
745 m_vertexCountDirty = true;
746 }
747 if ((roles.contains(AbstractTreeItem::ColorRole) || roles.contains(AbstractTreeItem::AlphaRole)) && absItem->color().isValid()) {
748 surf->setColor(toColor(itemColor(*absItem)));
749 }
750 if (roles.contains(AbstractTreeItem::TransformRole)) {
751 surf->applyTransform(toEigen(itemTransform(item)));
752 transformChanged = true;
753 }
754 if (roles.contains(SurfaceTreeItem::AnnotationDataRole)) {
755 SurfaceTreeItem* sItem = static_cast<SurfaceTreeItem*>(absItem);
756 if (!sItem->annotationData().empty()) {
757 surf->addAnnotation(sItem->annotationData());
758 }
759 }
760 }
761 }
762
763 if (m_itemDipoleMap.contains(item)) {
764 auto* absItem = dynamic_cast<AbstractTreeItem*>(item);
765 if (absItem && roles.contains(AbstractTreeItem::VisibleRole))
766 m_itemDipoleMap[item]->setVisible(absItem->isVisible());
767 if (roles.contains(AbstractTreeItem::TransformRole))
768 m_itemDipoleMap[item]->applyTransform(toEigen(itemTransform(item)));
769 }
770 }
771 if (transformChanged) {
772 updateInflatedSurfaceTransforms();
773 m_vertexCountDirty = true;
774 }
775 updateSceneBounds();
776 m_sceneDirty = true;
777 requestUpdate();
778}

803void BrainViewCore::updateSceneBounds()
804{
805 QVector3D min(std::numeric_limits<float>::max(), std::numeric_limits<float>::max(), std::numeric_limits<float>::max());
806 QVector3D max(std::numeric_limits<float>::lowest(), std::numeric_limits<float>::lowest(), std::numeric_limits<float>::lowest());
807 bool hasContent = false;
808
809 // Iterate over all surfaces
810 for (auto it = m_surfaces.begin(); it != m_surfaces.end(); ++it) {
811 if (it->second->isVisible()) {
812 Eigen::Vector3f sMinEigen, sMaxEigen;
813 it->second->boundingBox(sMinEigen, sMaxEigen);
814 QVector3D sMin = toQVector3D(sMinEigen), sMax = toQVector3D(sMaxEigen);
815
816 min.setX(std::min(min.x(), sMin.x()));
817 min.setY(std::min(min.y(), sMin.y()));
818 min.setZ(std::min(min.z(), sMin.z()));
819
820 max.setX(std::max(max.x(), sMax.x()));
821 max.setY(std::max(max.y(), sMax.y()));
822 max.setZ(std::max(max.z(), sMax.z()));
823 hasContent = true;
824 }
825 }
826
827 for (auto it = m_itemDipoleMap.cbegin(); it != m_itemDipoleMap.cend(); ++it) {
828 Eigen::Vector3f dMin;
829 Eigen::Vector3f dMax;
830 if (it->second->isVisible() && it->second->boundingBox(dMin, dMax)) {
831 min = QVector3D(std::min(min.x(), dMin.x()), std::min(min.y(), dMin.y()), std::min(min.z(), dMin.z()));
832 max = QVector3D(std::max(max.x(), dMax.x()), std::max(max.y(), dMax.y()), std::max(max.z(), dMax.z()));
833 hasContent = true;
834 }
835 }
836
837 if (hasContent) {
838 m_sceneCenter = (min + max) * 0.5f;
839
840 QVector3D diag = max - min;
841 m_sceneSize = std::max(diag.x(), std::max(diag.y(), diag.z()));
842
843 // Ensure non-zero size
844 if (m_sceneSize < 0.01f)
845 m_sceneSize = 0.3f;
846
847 } else {
848 // Default
849 m_sceneCenter = QVector3D(0, 0, 0);
850 m_sceneSize = 0.3f;
851 }
852}

996void BrainViewCore::updateInflatedSurfaceTransforms()
997{
998 const bool needsInflated = (m_singleView.surfaceType == "inflated") || std::any_of(m_subViews.cbegin(), m_subViews.cend(), [](const SubView& sv) { return sv.surfaceType == "inflated"; });
999
1000 const auto lhIt = m_surfaces.find("lh_inflated");
1001 const auto rhIt = m_surfaces.find("rh_inflated");
1002 if (lhIt == m_surfaces.end() || rhIt == m_surfaces.end()) {
1003 return;
1004 }
1005
1006 auto lhSurf = lhIt->second;
1007 auto rhSurf = rhIt->second;
1008
1009 lhSurf->applyTransform(toEigen(itemTransform(itemForSurface(lhSurf))));
1010 rhSurf->applyTransform(toEigen(itemTransform(itemForSurface(rhSurf))));
1011
1012 if (!needsInflated) {
1013 return;
1014 }
1015
1016 const float lhMaxX = lhSurf->maxX();
1017 const float rhMinX = rhSurf->minX();
1018
1019 const float gap = 0.005f;
1020 const float lhOffset = -gap / 2.0f - lhMaxX;
1021 const float rhOffset = gap / 2.0f - rhMinX;
1022
1023 lhSurf->translateX(lhOffset);
1024 rhSurf->translateX(rhOffset);
1025}

1286//=============================================================================================================
1287
1288bool BrainViewCore::savePng(const std::filesystem::path& path, int width, int height,
1289 const std::string& surfaceType)
1290{
1291 if (!m_host) {
1292 return false;
1293 }
1294
1295 // Set the surface type filter so the render loop matches the
1296 // loaded surface names (e.g. "white" vs default "pial").
1297 m_singleView.surfaceType = surfaceType;
1298
1299 // Let the host realise its surface off-screen (QRhiWidget with
1300 // Qt::WA_DontShowOnScreen, a hidden QWindow, ...) so that frames are
1301 // rendered without a window-manager surface.
1302 m_host->beginOffscreenCapture(QSize(width, height));
1303
1304 // Spin the event loop until the renderer exists (at least one frame
1305 // has been recorded) and the surface map is populated.
1306 const auto waitStart = std::chrono::steady_clock::now();
1307 while (std::chrono::steady_clock::now() - waitStart < std::chrono::seconds(5)) {
1308 QCoreApplication::processEvents(QEventLoop::AllEvents, 16);
1309 if (m_frame && !m_surfaces.empty())
1310 break;
1311 }
1312
1313 if (!m_frame) {
1314 m_host->endOffscreenCapture();
1315 return false;
1316 }
1317
1318 // Force a dirty scene so the next grab renders everything.
1319 m_sceneDirty = true;
1320 updateSceneBounds();
1321
1322 // Pump a few more frames so pipeline resources are fully created.
1323 for (int i = 0; i < 5; ++i)
1324 QCoreApplication::processEvents(QEventLoop::AllEvents, 16);
1325
1326 // The host renders a frame and reads the texture back.
1327 QImage img = grabSceneFrame();
1328
1329 m_host->endOffscreenCapture();
1330
1331 if (img.isNull())
1332 return false;
1333
1334 return img.save(QString::fromStdU16String(path.u16string()), "PNG");
1335}

1430{
1431 const std::vector<int> viewports = enabledViewportIndices();

1433 presets.reserve(m_subViews.size());
1434 for (const SubView& subView : m_subViews) {

1451void BrainViewCore::loadMultiViewSettings()
1452{
1453 QSettings settings;
1454 settings.beginGroup(QStringLiteral("BrainView"));
1455
1456 m_interaction.layout().set_split_x(settings.value("multiSplitX", 0.5f).toFloat());
1457 m_interaction.layout().set_split_y(settings.value("multiSplitY", 0.5f).toFloat());
1458
1459 const int savedViewMode = settings.value("viewMode", static_cast<int>(SingleView)).toInt();
1460 m_viewCount = std::clamp(settings.value("viewCount", 1).toInt(), 1, static_cast<int>(m_subViews.size()));
1461 // showSingleView keeps the pane count for the next showMultiView; one pane is always the single view
1462 m_viewMode = (savedViewMode == static_cast<int>(MultiView) && m_viewCount > 1) ? MultiView : SingleView;
1463
1464 Eigen::Quaternionf cameraRotation = m_singleView.rotation;
1465 const bool hasCameraQuat = settings.contains("cameraRotW") && settings.contains("cameraRotX") && settings.contains("cameraRotY") && settings.contains("cameraRotZ");
1466 if (hasCameraQuat) {
1467 const float w = settings.value("cameraRotW", 1.0f).toFloat();
1468 const float x = settings.value("cameraRotX", 0.0f).toFloat();
1469 const float y = settings.value("cameraRotY", 0.0f).toFloat();
1470 const float z = settings.value("cameraRotZ", 0.0f).toFloat();
1471 cameraRotation = Eigen::Quaternionf(w, x, y, z);
1472 if (cameraRotation.squaredNorm() <= std::numeric_limits<float>::epsilon()) {
1473 cameraRotation = Eigen::Quaternionf::Identity();
1474 } else {
1475 cameraRotation.normalize();
1476 }
1477 }
1478
1479 // Reset per-index defaults, then load saved state on top
1480 for (int i = 0; i < std::ssize(m_subViews); ++i) {
1481 m_subViews[i] = SubView::defaultForIndex(i);
1482 m_subViews[i].enabled = (i < m_viewCount);
1483 }
1484
1485 // Delegate per-SubView serialization
1486 const QtSettingsStore store(settings);
1487 m_singleView.load(store, "single_", cameraRotation);
1488 // The single view turns by the camera rotation, which is stored on its own
1489 m_singleView.rotation = cameraRotation;
1490 for (int i = 0; i < std::ssize(m_subViews); ++i)
1491 m_subViews[i].load(store, std::format("multi{}_", i), cameraRotation);
1492
1493 const int maxIdx = static_cast<int>(m_subViews.size()) - 1;
1494 m_visualizationEditTarget = normalizedVisualizationTarget(
1495 settings.value("visualizationEditTarget", -1).toInt(), maxIdx);
1496
1497 m_infoPanelVisible = settings.value("infoPanelVisible", true).toBool();
1498
1499 settings.endGroup();
1500
1501 setVisualizationEditTarget(m_visualizationEditTarget);
1502
1503 // Notify observers that persisted state has been restored. Defer to
1504 // the next event-loop tick so that callers constructing BrainView
1505 // and connecting to these signals immediately afterwards still
1506 // receive the initial state — direct emits from inside the ctor
1507 // chain would fire before any external connect() call.
1508 QTimer::singleShot(0, this, [this]() {
1509 emit viewCountChanged(m_viewCount);
1510 emit shaderModeChanged(QString::fromStdString(shaderModeName(m_brainShaderMode)));
1511 });
1512}

1693void BrainViewCore::renderFrame(QRhi* rhi, QRhiRenderTarget* target, QRhiCommandBuffer* cb)
1694{
1695 m_inRender = true;
1696 const qreal dpr = m_host ? m_host->devicePixelRatio() : 1.0;
1697 const QSize outputSize = target->pixelSize();
1698 viz::BrainFrameComposer& composer = m_frame->composer;
1699 composer.begin(RhiFrameRenderer::frameTarget(rhi, target, dpr), viz::Color::black());
1700
1701 // Without anything loaded the frame shows the overlay on black
1702 if (!m_surfaces.empty() || !m_itemDipoleMap.empty() || m_dipoles || m_electrodes) {
1703 composeScene(outputSize);
1704 }
1705
1706 // The overlay is laid out after the panes updated their info labels, and uploaded only when it changed
1707 if (!m_overlaySuppressed) {
1708 const bool resized = m_frame->overlaySize != outputSize || m_frame->overlayRatio != dpr;
1709 if (m_overlayDirty || resized || !composer.redraw_overlay()) {
1710 const QtTextMetrics metrics(overlayBaseFont());
1711 composer.draw_overlay(m_overlay.draw(qCeil(outputSize.width() / dpr), qCeil(outputSize.height() / dpr), metrics), dpr);
1712 m_frame->overlaySize = outputSize;
1713 m_frame->overlayRatio = dpr;
1714 m_overlayDirty = false;
1715 }
1716 }
1717
1718 const viz::GpuFrame frame = composer.finish();
1719 m_frame->renderer.render(rhi, target, cb, frame, composer.resources());
1720 m_inRender = false;
1721}

1723//=============================================================================================================
1724
1725void BrainViewCore::composeScene(QSize outputSize)
1726{
1727 // Ensure active surface pointer is valid if possible, otherwise just use first available for stats
1728 if (!m_activeSurface && !m_surfaces.empty()) {
1729 m_activeSurface = m_surfaces.begin()->second;
1730 }
1731
1732 m_frameCount++;
1733 const std::chrono::duration<float> fpsElapsed = std::chrono::steady_clock::now() - m_fpsStart;
1734 if (fpsElapsed >= std::chrono::milliseconds(500)) {
1735 float fps = m_frameCount / fpsElapsed.count();
1736
1737 // Recount vertices only when surface list/visibility changed
1738 if (m_vertexCountDirty) {
1739 auto countVerticesForSubView = [this](const SubView& sv) -> qint64 {
1740 qint64 total = 0;
1741 for (const auto& [key, surface] : m_surfaces) {
1742 if (!surface)
1743 continue;
1744 if (!sv.shouldRenderSurface(key))
1745 continue;
1746 if (SubView::isBrainSurfaceKey(key)) {
1747 if (!sv.matchesSurfaceType(key))
1748 continue;
1749 } else {
1750 if (!surface->isVisible())
1751 continue;
1752 }
1753 total += surface->vertexCount();
1754 }
1755 return total;
1756 };
1757
1758 qint64 vCount = 0;
1759 if (m_viewMode == MultiView) {
1760 for (int vp : enabledViewportIndices()) {
1761 vCount += countVerticesForSubView(m_subViews[vp]);
1762 }
1763 } else {
1764 vCount = countVerticesForSubView(m_singleView);
1765 }
1766 m_cachedVertexCount = vCount;
1767 m_vertexCountDirty = false;
1768 }
1769
1770 m_overlay.set_text(viz::BrainViewOverlay::Item::FpsLabel, std::format("FPS: {:.1f}\nVertices: {}", fps, m_cachedVertexCount));
1771 updateOverlayLayout();
1772 m_frameCount = 0;
1773 m_fpsStart = std::chrono::steady_clock::now();
1774 }
1775
1776 // What every pane draws
1777 viz::BrainScene scene;
1778 scene.surfaces = &m_surfaces;
1779 scene.meg_field_key = m_fieldMapper.megSurfaceKey();
1780 scene.eeg_field_key = m_fieldMapper.eegSurfaceKey();
1781 for (const auto& [item, dipoles] : m_itemDipoleMap) {
1782 scene.dipoles.push_back(dipoles.get());
1783 }
1784 if (m_dipoles) {
1785 scene.dipoles.push_back(m_dipoles.get());
1786 }
1787 scene.electrodes = m_electrodes;
1788 scene.network = m_network.get();
1789 scene.head_path = m_headPath.get();
1790 if (m_hasIntersection && m_debugPointerSurface) {
1791 scene.pointer = m_debugPointerSurface.get();
1792 scene.pointer_position = toEigen(m_lastIntersectionPoint);
1793 }
1794 scene.lighting = m_lightingEnabled;
1795
1796 // Slice and video pixels are converted only when they change
1797 for (int i = 0; i < kMaxSliceSlots; ++i) {
1798 if (m_slices[i] && m_sliceVisible[i] && !m_slices[i]->image().isNull()) {
1799 const QImage& image = m_slices[i]->image();
1800 viz::BrainSliceLayer slice;
1801 slice.image = m_frame->cachedImage(m_frame->sliceImages[static_cast<std::size_t>(i)], image, static_cast<std::uint64_t>(image.cacheKey()));
1802 std::vector<float> quad;
1803 m_slices[i]->generateQuadVertices(quad);
1804 std::copy_n(quad.begin(), std::min(quad.size(), slice.quad.size()), slice.quad.begin());
1805 slice.opacity = m_slices[i]->opacity();
1806 slice.window_center = m_slices[i]->windowCenter();
1807 slice.window_width = m_slices[i]->windowWidth();
1808 scene.slices.push_back(slice);
1809 }
1810 }
1811 if (m_videoOverlay && m_videoOverlay->isEnabled() && m_videoOverlay->hasFrame()) {
1812 viz::BrainVideoLayer video;
1813 video.frame = m_frame->cachedImage(m_frame->videoFrame, m_videoOverlay->frame(), m_videoOverlay->frameGeneration());
1814 if (m_videoOverlay->isDepthEnabled() && m_videoOverlay->hasDepthFrame()) {
1815 video.depth = m_frame->cachedImage(m_frame->videoDepth, m_videoOverlay->depthFrame(), m_videoOverlay->depthFrameGeneration());
1816 }
1817 video.focus = toEigen(m_videoOverlay->focusPosition());
1818 video.size = m_videoOverlay->sizeMeters();
1819 video.up_hint = toEigen(m_videoOverlay->upHint());
1820 video.opacity = m_videoOverlay->opacity();
1821 video.depth_scale = m_videoOverlay->depthScale();
1822 video.depth_steps = m_videoOverlay->depthSteps();
1823 scene.video = video;
1824 }
1825
1826 // The panes, leaving the separator lines to the right and below each pane free
1827 const auto enabledViewports = enabledViewportIndices();
1828 const int numEnabled = static_cast<int>(enabledViewports.size());
1829 m_camera.set_scene_center(toEigen(m_cameraFocusOverride ? m_cameraFocusCenter : m_sceneCenter));
1830 m_camera.set_scene_size(m_cameraFocusOverride ? m_cameraFocusSize : m_sceneSize);
1831 std::vector<viz::BrainScenePane> panes;
1832 panes.reserve(static_cast<std::size_t>(numEnabled));
1833 for (int slot = 0; slot < numEnabled; ++slot) {
1834 const int vp = (m_viewMode == MultiView) ? enabledViewports[slot] : 0;
1835 const SubView& sv = (m_viewMode == MultiView) ? m_subViews[vp] : m_singleView;
1836 const viz::PaneLayout& layout = m_interaction.layout();
1837 const viz::PixelRect pixels = (m_viewMode == MultiView)
1838 ? layout.inset_for_separator(layout.slot_rect(slot, numEnabled, toPixelSize(outputSize)), slot, numEnabled)
1839 : viz::PixelRect{0, 0, outputSize.width(), outputSize.height()};
1840 const viz::GpuRect viewport{pixels.x, pixels.y, std::max(1, pixels.width), std::max(1, pixels.height)};
1841 const float aspectRatio = static_cast<float>(viewport.width) / static_cast<float>(viewport.height);
1842 const viz::CameraMatrices camera = (m_viewMode == MultiView) ? m_camera.pane_view(sv, aspectRatio)
1843 : m_camera.single_view(m_singleView.rotation, m_singleView.zoom, aspectRatio);
1844 panes.push_back({viewport, camera, &sv});
1845 }
1846 const std::vector<std::vector<std::string>> drawn = m_frame->composer.draw(scene, panes);
1847
1848 // Per-pane info labels; relayout only on change
1849 for (int slot = 0; slot < numEnabled; ++slot) {
1850 const int vp = (m_viewMode == MultiView) ? enabledViewports[slot] : 0;
1851 const SubView& sv = *panes[static_cast<std::size_t>(slot)].view;
1852 const std::vector<std::string>& keys = drawn[static_cast<std::size_t>(slot)];
1853 std::string info = std::format("Shader: {}\nSurface: {}\nOverlay: {}\nDrawn: {}", shaderModeName(sv.brainShader), sv.surfaceType,
1854 visualizationModeName(sv.overlayMode), keys.empty() ? std::string("none") : mne::join(keys, ", "));
1855 const bool infoChanged = (m_viewMode == MultiView)
1856 ? m_overlay.set_text(viz::BrainViewOverlay::Item::ViewportInfoLabel, std::move(info), vp)
1857 : m_overlay.set_text(viz::BrainViewOverlay::Item::SingleViewInfoLabel, std::move(info));
1858 if (infoChanged) {
1859 updateOverlayLayout();
1860 }
1861 }
1862}

2036
2037bool BrainViewCore::loadSensorField(const std::filesystem::path& evokedPath, int aveIndex)
2038{
2039 auto evoked = DataLoader::loadEvoked(evokedPath, aveIndex);
2040 if (evoked.empty())
2041 return false;
2042
2043 // Preserve the current time point when switching between evoked sets
2044 // that share the same sensor configuration (same file, different condition).
2045 const int previousTimePoint = m_fieldMapper.timePoint();
2046 const bool canReuse = m_fieldMapper.hasMappingFor(evoked);
2047
2048 m_fieldMapper.setEvoked(evoked);
2049
2050 if (!canReuse) {
2051 // Sensor config changed — full rebuild required (also precomputes global range)
2052 if (!m_fieldMapper.buildMapping(m_surfaces, m_headToMriTrans, m_applySensorTrans)) {
2053 m_fieldMapper.setEvoked(mne::FiffEvoked()); // Clear state on failure
2054 return false;
2055 }
2056 } else {
2057 // Mapping reused — recompute normalization for new evoked data
2058 m_fieldMapper.computeNormRange();
2059 }
2060
2061 // Clamp preserved time point to the range of the new evoked data
2062 const int numTimes = static_cast<int>(m_fieldMapper.evoked().times.size());
2063 const int tp = std::clamp(previousTimePoint, 0, std::max(0, numTimes - 1));
2064
2065 emit sensorFieldLoaded(numTimes, tp);
2066 setSensorFieldTimePoint(tp);

3314bool BrainViewCore::intersectWorldRay(const QVector3D& origin, const QVector3D& direction, QVector3D& hitPoint) const
3315{
3316 // Test against ALL surfaces (no SubView visibility filter) so
3317 // the optical ray always finds the head even if it's hidden in
3318 // the first SubView of a multi-view layout.
3319 float closestDist = std::numeric_limits<float>::max();
3320 bool found = false;
3321
3322 // Always test the BEM head surface first (even when hidden) — the
3323 // optical projection cares about the outermost physical shell.
3324 auto headIt = m_surfaces.find("bem_head");
3325 if (headIt != m_surfaces.cend() && headIt->second) {
3326 float dist = 0.0f;
3327 int vertexIdx = -1;
3328 if (headIt->second->intersects(toEigen(origin), toEigen(direction), dist, vertexIdx) && dist < closestDist) {
3329 closestDist = dist;
3330 hitPoint = origin + dist * direction;
3331 found = true;
3332 }
3333 }
3334
3335 // Fall back to any other visible surface if the head wasn't hit.
3336 if (!found) {
3337 for (auto it = m_surfaces.cbegin(); it != m_surfaces.cend(); ++it) {
3338 const auto& surf = it->second;
3339 if (!surf || !surf->isVisible())
3340 continue;
3341 float dist = 0.0f;
3342 int vertexIdx = -1;
3343 if (surf->intersects(toEigen(origin), toEigen(direction), dist, vertexIdx) && dist < closestDist) {
3344 closestDist = dist;
3345 hitPoint = origin + dist * direction;
3346 found = true;
3347 }
3348 }
3349 }
3350 return found;
3351}
