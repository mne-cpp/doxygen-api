97AcqExpected<Config> parseConfig(const AcqUri& uri)
98{
99 Config c;
100 const auto simulate = paramBool(uri, "simulate", false);
101 const auto timeout = paramDouble(uri, "timeout", 10.0, 0.1, 3600.0);
102 const auto meg = paramInt(uri, "meg", 306, 0, 4096);
103 const auto eeg = paramInt(uri, "eeg", 0, 0, 4096);
104 const auto rate = paramDouble(uri, "rate", 1000.0, 1.0, 1e5);
105 const auto block = paramInt(uri, "block", 50, 1, 100000);
106 const auto seed = paramInt(uri, "seed", 0, 0, INT64_MAX);
107 const auto realtime = paramBool(uri, "realtime", true);
108 const auto duration = paramDouble(uri, "duration", 0.0, 0.0, 1e7);
109 auto thread = paramThreadPolicy(uri);
110 for (const AcqError& error : {simulate.error(), timeout.error(), meg.error(), eeg.error(), rate.error(), block.error(), seed.error(),
111 realtime.error(), duration.error(), thread.error()}) {
112 if (!error.ok()) {
113 return error;
114 }
115 }
116 c.simulate = *simulate;
117 if (!uri.target.empty()) {
118 if (c.simulate) {
119 return invalid("target '" + uri.target + "': the simulator takes no server");
120 }
121 const std::size_t colon = uri.target.rfind(':');
122 c.host = uri.target.substr(0, colon);
123 if (colon != std::string::npos) {
124 const std::string_view port = std::string_view(uri.target).substr(colon + 1);
125 unsigned value = 0;
126 const auto [end, ec] = std::from_chars(port.data(), port.data() + port.size(), value);
127 if (ec != std::errc() || end != port.data() + port.size() || value == 0 || value > 65535) {
128 return invalid("target '" + uri.target + "': host:port with a port 1..65535");
129 }
130 c.port = static_cast<std::uint16_t>(value);
131 }
132 if (c.host.empty()) {
133 return invalid("target '" + uri.target + "': host:port");
134 }
135 }
136 if (*meg + *eeg == 0) {
137 return invalid("meg=0&eeg=0: the simulated system needs channels");
138 }
139 c.timeout = std::chrono::milliseconds(static_cast<std::int64_t>(*timeout * 1000.0));
140 c.simulation = {static_cast<std::size_t>(*meg), static_cast<std::size_t>(*eeg), *rate, static_cast<std::size_t>(*block),
141 static_cast<std::uint64_t>(*seed), *realtime, *duration};
142 c.thread = std::move(*thread);
143 return c;
144}

185std::optional<std::int32_t> event_code(std::uint32_t type, std::span<const std::byte> value)
186{
187 const auto integral = [](double v) -> std::optional<std::int32_t> {
188 if (!std::isfinite(v) || v != std::trunc(v) || std::abs(v) > 2147483647.0) {
189 return std::nullopt;
190 }
191 return static_cast<std::int32_t>(v);
192 };
193 const auto need = [&](std::size_t bytes) {
194 return value.size() >= bytes;
195 };
196 switch (type) {
197 case ft::kTypeUint8:
198 return need(1) ? std::optional(static_cast<std::int32_t>(get<std::uint8_t>(value, 0))) : std::nullopt;
199 case ft::kTypeInt8:
200 return need(1) ? std::optional(static_cast<std::int32_t>(get<std::int8_t>(value, 0))) : std::nullopt;
201 case ft::kTypeUint16:
202 return need(2) ? std::optional(static_cast<std::int32_t>(get<std::uint16_t>(value, 0))) : std::nullopt;
203 case ft::kTypeInt16:
204 return need(2) ? std::optional(static_cast<std::int32_t>(get<std::int16_t>(value, 0))) : std::nullopt;
205 case ft::kTypeUint32:
206 return need(4) ? integral(static_cast<double>(get<std::uint32_t>(value, 0))) : std::nullopt;
207 case ft::kTypeInt32:
208 return need(4) ? std::optional(get<std::int32_t>(value, 0)) : std::nullopt;
209 case ft::kTypeUint64:
210 return need(8) ? integral(static_cast<double>(get<std::uint64_t>(value, 0))) : std::nullopt;
211 case ft::kTypeInt64:
212 return need(8) ? integral(static_cast<double>(get<std::int64_t>(value, 0))) : std::nullopt;
213 case ft::kTypeFloat32:
214 return need(4) ? integral(get<float>(value, 0)) : std::nullopt;
215 case ft::kTypeFloat64:
216 return need(8) ? integral(get<double>(value, 0)) : std::nullopt;
217 default:
218 return std::nullopt; // strings
219 }
220}

230struct Header
231{
232 std::uint32_t nchans = 0;
233 std::uint32_t nsamples = 0;
234 std::uint32_t nevents = 0;
235 float fsample = 0.0F;
236 std::uint32_t data_type = 0;
237 std::vector<std::pair<std::uint32_t, std::span<const std::byte>>> chunks;
238 std::vector<std::byte> bytes; // owns the chunk data
239
240 [[nodiscard]] std::span<const std::byte> chunk(std::uint32_t type) const
241 {
242 const auto it = std::find_if(chunks.begin(), chunks.end(), [type](const auto& c) { return c.first == type; });
243 return it == chunks.end() ? std::span<const std::byte>() : it->second;
244 }
245};

247AcqExpected<Header> parse_header(std::vector<std::byte> payload)
248{
249 if (payload.size() < ft::kHeaderDefBytes) {
250 return AcqError(AcqErrorCode::DeviceIo, "the FieldTrip buffer sent a truncated header");
251 }
252 Header h;
253 h.nchans = get<std::uint32_t>(payload, 0);
254 h.nsamples = get<std::uint32_t>(payload, 4);
255 h.nevents = get<std::uint32_t>(payload, 8);
256 h.fsample = get<float>(payload, 12);
257 h.data_type = get<std::uint32_t>(payload, 16);
258 h.bytes = std::move(payload);
259 const std::span<const std::byte> all(h.bytes);
260 const std::size_t size = std::min<std::size_t>(get<std::uint32_t>(all, 20), all.size() - ft::kHeaderDefBytes);
261 std::span<const std::byte> rest = all.subspan(ft::kHeaderDefBytes, size);
262 while (rest.size() >= 8) {
263 const auto type = get<std::uint32_t>(rest, 0);
264 const std::size_t length = std::min<std::size_t>(get<std::uint32_t>(rest, 4), rest.size() - 8);
265 h.chunks.emplace_back(type, rest.subspan(8, length));
266 rest = rest.subspan(8 + length);
267 }
268 return h;
269}

272AcqExpected<AcqStreamInfo> stream_of(const Header& h, const std::string& source)
273{
274 const auto format = format_of(h.data_type);
275 if (!format) {
276 return AcqError(AcqErrorCode::UnsupportedFormat, std::format("FieldTrip data type {} (int16, int32, float32 and float64 are read)", h.data_type));
277 }
278 if (h.nchans == 0 || !(h.fsample > 0.0F)) {
279 return AcqError(AcqErrorCode::UnsupportedFormat, std::format("the FieldTrip header has {} channels at {} Hz", h.nchans, h.fsample));
280 }
281 if (const auto neuromag = h.chunk(ft::kChunkNeuromagHeader); !neuromag.empty()) {
282 // neuromag2ft writes the acquisition's tags in its own byte order, the isotrak file as a FIFF file
283 std::vector<NeuromagTag> meas_info = block_of(parse_tags(neuromag, false), fiff::kBlockMeasInfo);
284 if (const auto isotrak = h.chunk(ft::kChunkNeuromagIsotrak); !isotrak.empty() && !meas_info.empty()) {
285 const std::vector<NeuromagTag> dig = block_of(parse_tags(isotrak, true), fiff::kBlockIsotrak);
286 meas_info.insert(meas_info.end() - 1, dig.begin(), dig.end());
287 }
288 auto info = stream_info(meas_info, source, *format);
289 if (!info) {
290 return info.error();
291 }
292 if (info->channelCount() != h.nchans) {
293 return AcqError(AcqErrorCode::UnsupportedFormat,
294 std::format("the Neuromag header describes {} channels, the FieldTrip header {}", info->channelCount(), h.nchans));
295 }
296 info->sfreq = h.fsample;
297 return info;
298 }
299 AcqStreamInfo info;
300 info.name = "raw";
301 info.sourceId = source;
302 info.format = *format;
303 info.layout = AcqLayout::Interleaved;
304 info.sfreq = h.fsample;
305 std::span<const std::byte> names = h.chunk(ft::kChunkChannelNames);
306 for (std::uint32_t c = 0; c < h.nchans; ++c) {
307 AcqChannelInfo& channel = info.channels.emplace_back();
308 const auto* begin = reinterpret_cast<const char*>(names.data());
309 const auto length = static_cast<std::size_t>(std::find(names.begin(), names.end(), std::byte{0}) - names.begin());
310 channel.name = length > 0 ? std::string(begin, length) : std::format("CH {:03}", c + 1);
311 names = names.subspan(std::min(names.size(), length + 1));
312 channel.kind = AcqChannelKind::Misc;
313 }
314 return info;
315}

320 static AcqExpected<std::unique_ptr<AcqBackend>> open(const AcqUri& uri)
321 {
322 auto config = parseConfig(uri);
323 if (!config) {
324 return config.error();
325 }
326 std::unique_ptr<FtBufferTransport> transport;
327 if (config->simulate) {
328 transport = make_ftbuffer_simulator(config->simulation);
329 } else {
330 auto client = connect_ftbuffer(config->host, config->port, std::min(config->timeout, std::chrono::milliseconds(5000)));
331 if (!client) {
332 return client.error();
333 }
334 transport = std::move(*client);
335 }
336 auto header = read_header(*transport, config->timeout);
337 if (!header) {
338 return header.error();
339 }
340 const bool neuromag = !header->chunk(ft::kChunkNeuromagHeader).empty();
341 auto info = stream_of(*header, transport->description());
342 if (!info) {
343 return info.error();
344 }
345
346 std::unique_ptr<FtBufferBackend> backend(new FtBufferBackend(std::move(*config), std::move(transport), uri.toString()));
347 backend->m_format = info->format;
348 backend->m_width = info->channelCount();
349 backend->m_trigger_channel = neuromag ? trigger_channel(*info) : std::nullopt;
350 AcqStreamOptions options;
351 options.capacitySamples = static_cast<std::size_t>(std::max(4.0 * info->sfreq, 4096.0));
352 backend->m_chunk = options.capacitySamples / 4;
353 backend->m_rate = info->sfreq;
354 auto writer = backend->addInputStream(std::move(*info), options);
355 if (!writer) {
356 return writer.error();
357 }
358 backend->m_writer = *writer;
359 return std::unique_ptr<AcqBackend>(std::move(backend));
360 }

422 static AcqExpected<Header> read_header(FtBufferTransport& transport, std::chrono::milliseconds timeout)
423 {
424 const auto deadline = std::chrono::steady_clock::now() + timeout;
425 while (true) {
426 auto answer = transport.request(ft::kGetHdr, {}, kAnswer);
427 if (!answer) {
428 return answer.error();
429 }
430 if (answer->command == ft::kGetOk) {
431 return parse_header(std::move(answer->payload));
432 }
433 if (std::chrono::steady_clock::now() >= deadline) {
434 return AcqError(AcqErrorCode::Timeout, std::format("{} has no header; is the producer running?", transport.description()));
435 }
436 std::this_thread::sleep_for(kWait);
437 }
438 }

441 AcqExpected<std::pair<std::uint32_t, std::uint32_t>> wait(std::pair<std::uint32_t, std::uint32_t> threshold, std::chrono::milliseconds timeout)
442 {
443 std::vector<std::byte> payload;
444 put(payload, threshold.first);
445 put(payload, threshold.second);
446 put(payload, static_cast<std::uint32_t>(timeout.count()));
447 auto answer = m_transport->request(ft::kWaitDat, payload, timeout + kAnswer);
448 if (!answer) {
449 return answer.error();
450 }
451 if (answer->command != ft::kWaitOk || answer->payload.size() < 8) {
452 return AcqError(AcqErrorCode::DeviceIo, "the FieldTrip buffer refuses to wait for data; has its header gone?");
453 }
454 return std::pair{get<std::uint32_t>(answer->payload, 0), get<std::uint32_t>(answer->payload, 4)};
455 }

458 AcqError fetch_events(std::uint32_t count)
459 {
460 std::vector<std::byte> payload;
461 put(payload, static_cast<std::int32_t>(m_next_event));
462 put(payload, static_cast<std::int32_t>(count - 1));
463 auto answer = m_transport->request(ft::kGetEvt, payload, kAnswer);
464 if (!answer) {
465 return answer.error();
466 }
467 m_next_event = count;
468 if (answer->command != ft::kGetOk) {
469 return {}; // overwritten before they were read
470 }
471 std::span<const std::byte> rest(answer->payload);
472 while (rest.size() >= ft::kEventDefBytes) {
473 const auto type_type = get<std::uint32_t>(rest, 0);
474 const auto type_numel = get<std::uint32_t>(rest, 4);
475 const auto value_type = get<std::uint32_t>(rest, 8);
476 const auto sample = get<std::int32_t>(rest, 16);
477 const std::size_t size = std::min<std::size_t>(get<std::uint32_t>(rest, 28), rest.size() - ft::kEventDefBytes);
478 const std::size_t skip = std::min(size, std::size_t{type_numel} * type_size(type_type));
479 const auto code = event_code(value_type, rest.subspan(ft::kEventDefBytes + skip, size - skip));
480 if (code && sample >= 0 && static_cast<std::uint64_t>(sample) >= m_offset) {
481 m_pending.push_back({static_cast<std::uint64_t>(sample) - m_offset, 0, *code});
482 }
483 rest = rest.subspan(ft::kEventDefBytes + size);
484 }
485 std::stable_sort(m_pending.begin(), m_pending.end(), [](const AcqEvent& a, const AcqEvent& b) { return a.sample < b.sample; });
486 return {};
487 }

490 AcqError push(const AcqEvent& event)
491 {
492 if (event.sample < m_last_event) {
493 return {};
494 }
495 m_last_event = event.sample;
496 return m_writer->pushEvent(event);
497 }

500 AcqError fetch_samples(std::uint32_t count)
501 {
502 std::vector<std::byte> payload;
503 put(payload, m_next);
504 put(payload, m_next + count - 1);
505 auto answer = m_transport->request(ft::kGetDat, payload, kAnswer);
506 if (!answer) {
507 return answer.error();
508 }
509 const std::size_t value = width_of(m_format);
510 const std::size_t frame = m_width * value;
511 const std::span<const std::byte> data(answer->payload);
512 if (answer->command != ft::kGetOk) {
513 return {AcqErrorCode::DeviceIo, "the FieldTrip buffer overwrote samples before they were read; read faster or enlarge it"};
514 }
515 if (data.size() < ft::kDataDefBytes + count * frame || get<std::uint32_t>(data, 0) != m_width || get<std::uint32_t>(data, 4) != count ||
516 format_of(get<std::uint32_t>(data, 8)) != m_format) {
517 return {AcqErrorCode::DeviceIo, "the FieldTrip buffer sent samples that do not match its header"};
518 }
519 AcqExpected<AcqWriteView> view = AcqError(AcqErrorCode::Timeout);
520 while (!view && view.error() == AcqErrorCode::Timeout) {
521 if (m_stop.load(std::memory_order_acquire)) {
522 return {};
523 }
524 view = m_writer->beginWrite(count, kPoll); // a gating reader may not have freed the space yet
525 }
526 if (!view) {
527 return view.error();
528 }
529 const std::uint64_t first = m_writer->writePosition();
530 const std::byte* source = data.data() + ft::kDataDefBytes;
531 std::size_t s = 0;
532 for (std::size_t p = 0; p < view->partCount(); ++p) {
533 const AcqWritePart& part = view->part(p);
534 for (std::size_t i = 0; i < part.samples; ++i, ++s) {
535 const std::byte* values = source + s * frame;
536 std::memcpy(part.at(0, i), values, frame);
537 const std::uint64_t at = first + s;
538 while (!m_pending.empty() && m_pending.front().sample <= at) {
539 if (AcqError e = push(m_pending.front()); !e.ok()) {
540 return e;
541 }
542 m_pending.pop_front();
543 }
544 if (m_trigger_channel) {
545 const auto code = static_cast<std::int32_t>(std::lround(value_at(values + *m_trigger_channel * value, m_format)));
546 if (m_trigger.changed(code)) {
547 if (AcqError e = push({at, 0, code}); !e.ok()) {
548 return e;
549 }
550 }
551 }
552 }
553 }
554 m_writer->commit(count);
555 m_next += count;
556 return {};
557 }
558
559 void run()
560 {
561 while (!m_stop.load(std::memory_order_acquire)) {
562 auto counts = wait({m_next, m_next_event}, kWait);
563 if (!counts) {
564 if (counts.error() == AcqErrorCode::EndOfStream) {
565 m_writer->finish();
566 } else {
567 reportError(counts.error());
568 }
569 return;
570 }
571 const auto [samples, events] = *counts;
572 if (samples < m_next || events < m_next_event) {
573 m_writer->finish(); // a new header: the measurement this stream describes is over
574 return;
575 }
576 if (events > m_next_event) {
577 if (AcqError e = fetch_events(events); !e.ok()) {
578 reportError(e);
579 return;
580 }
581 }
582 while (m_next < samples && !m_stop.load(std::memory_order_acquire)) {
583 if (AcqError e = fetch_samples(static_cast<std::uint32_t>(std::min<std::size_t>(samples - m_next, m_chunk))); !e.ok()) {
584 reportError(e);
585 return;
586 }
587 }
588 }
589 }

594 AcqWriter* m_writer = nullptr;
595 AcqSampleFormat m_format = AcqSampleFormat::Float32;
596 std::size_t m_width = 0;
597 std::size_t m_chunk = 1024; // samples per data request
598 double m_rate = 0.0;
