171void BrainRenderer::Impl::createResources(QRhi *rhi, QRhiRenderPassDescriptor *rp, int sampleCount)
172{
173 uniformBufferOffsetAlignment = rhi->ubufAlignment();
174
175 // Create Uniform Buffer
176 if (!uniformBuffer) {
177 // Size for 8192 slots with alignment — enough for 4 viewports × ~1000 surfaces
178 uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, kUniformSlotCount * uniformBufferOffsetAlignment));
179 uniformBuffer->create();
180 }
181
182 // Create SRB
183 if (!srb) {
184 srb.reset(rhi->newShaderResourceBindings());
185 srb->setBindings({
186 // Use dynamic offset for the uniform buffer.
187 // The size of one uniform block in the shader is ~104 bytes,
188 // but we use uniformBufferOffsetAlignment for the stride.
189 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage, uniformBuffer.get(),kUniformBlockSize)
190 });
191 srb->create();
192 }
193
194 // Shader Loader
195 auto getShader = [](const QString &name) {
196 QFile f(name);
197 return f.open(QIODevice::ReadOnly) ? QShader::fromSerialized(f.readAll()) : QShader();
198 };
199
200 // List of modes to initialize
201 QList<ShaderMode> modes = {Standard, Holographic, Anatomical, Dipole, XRay, ShowNormals};
202
203 for (ShaderMode mode : modes) {
204 QString vert = (mode == Holographic || mode == XRay) ? ":/holographic.vert.qsb" :
205 (mode == Anatomical) ? ":/anatomical.vert.qsb" :
206 (mode == Dipole) ? ":/dipole.vert.qsb" :
207 (mode == ShowNormals) ? ":/shownormals.vert.qsb" : ":/standard.vert.qsb";
208
209 QString frag = (mode == Holographic || mode == XRay) ? ":/holographic.frag.qsb" :
210 (mode == Anatomical) ? ":/anatomical.frag.qsb" :
211 (mode == Dipole) ? ":/dipole.frag.qsb" :
212 (mode == ShowNormals) ? ":/shownormals.frag.qsb" : ":/standard.frag.qsb";
213
214 QShader vS = getShader(vert);
215 QShader fS = getShader(frag);
216
217 if (!vS.isValid() || !fS.isValid()) {
218 qWarning() << "BrainRenderer: Could not load shaders for mode" << mode << vert << frag;
219 continue;
220 }
221
222 // Setup Pipeline
223 auto pipeline = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
224
225 QRhiGraphicsPipeline::TargetBlend blend;
226 if (mode == Holographic || mode == XRay) {
227 blend.enable = true;
228 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
229 blend.dstColor = QRhiGraphicsPipeline::One;
230 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
231 blend.dstAlpha = QRhiGraphicsPipeline::One;
232 } else if (mode == Dipole) {
233 blend.enable = true;
234 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
235 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
236 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
237 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
238 }
239
240 auto setup = [&](QRhiGraphicsPipeline* p, QRhiGraphicsPipeline::CullMode cull) {
241 p->setShaderStages({{ QRhiShaderStage::Vertex, vS }, { QRhiShaderStage::Fragment, fS }});
242
243 QRhiVertexInputLayout il;
244
245 if (mode == Dipole) {
246 il.setBindings({
247 { 6 * sizeof(float) }, // Binding 0: Vertex Data (Pos + Normal) -> stride 6 floats
248 { 21 * sizeof(float), QRhiVertexInputBinding::PerInstance } // Binding 1: Instance Data (Mat4 + Color + Selected) -> stride 21 floats
249 });
250
251 il.setAttributes({
252 // Vertex Buffer (Binding 0)
253 { 0, 0, QRhiVertexInputAttribute::Float3, 0 }, // Pos
254 { 0, 1, QRhiVertexInputAttribute::Float3, 3 * sizeof(float) }, // Normal
255
256 // Instance Buffer (Binding 1)
257 // Model Matrix (4 x vec4)
258 { 1, 2, QRhiVertexInputAttribute::Float4, 0 },
259 { 1, 3, QRhiVertexInputAttribute::Float4, 4 * sizeof(float) },
260 { 1, 4, QRhiVertexInputAttribute::Float4, 8 * sizeof(float) },
261 { 1, 5, QRhiVertexInputAttribute::Float4, 12 * sizeof(float) },
262 // Color
263 { 1, 6, QRhiVertexInputAttribute::Float4, 16 * sizeof(float) },
264 // isSelected
265 { 1, 7, QRhiVertexInputAttribute::Float, 20 * sizeof(float) }
266 });
267 } else {
268 il.setBindings({{ 36 }}); // sizeof(VertexData) = 36 with surfaceId
269 il.setAttributes({{ 0, 0, QRhiVertexInputAttribute::Float3, 0 },
270 { 0, 1, QRhiVertexInputAttribute::Float3, 12 },
271 { 0, 2, QRhiVertexInputAttribute::UNormByte4, 24 },
272 { 0, 3, QRhiVertexInputAttribute::UNormByte4, 28 },
273 { 0, 4, QRhiVertexInputAttribute::Float, 32 }}); // surfaceId
274 }
275
276 p->setVertexInputLayout(il);
277 p->setShaderResourceBindings(srb.get());
278 p->setRenderPassDescriptor(rp);
279 p->setSampleCount(sampleCount);
280 p->setCullMode(cull);
281 if (mode == Holographic) {
282 p->setTargetBlends({blend});
283 p->setDepthTest(true);
284 p->setDepthWrite(false);
285 } else if (mode == XRay) {
286 p->setTargetBlends({blend});
287 p->setDepthTest(false); // Disable Depth Test to see through head
288 p->setDepthWrite(false);
289 } else if (mode == Dipole) {
290 p->setTargetBlends({blend});
291 p->setCullMode(QRhiGraphicsPipeline::None);
292 p->setDepthTest(true);
293 p->setDepthWrite(false);
294 } else {
295 p->setDepthTest(true);
296 p->setDepthWrite(true);
297 }
298 p->setFlags(QRhiGraphicsPipeline::UsesScissor);
299 p->create();
300 };
301
302 if (mode == Holographic || mode == XRay) { // Handle XRay back-faces same as Holographic
303 auto pipelineBack = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
304 setup(pipelineBack.get(), QRhiGraphicsPipeline::Front);
305 pipelinesBackColor[mode] = std::move(pipelineBack);
306 setup(pipeline.get(), QRhiGraphicsPipeline::Back); // Front faces
307 } else {
308 // Culling: None (Double-sided) to be safe for FreeSurfer meshes
309 setup(pipeline.get(), QRhiGraphicsPipeline::None);
310 }
311 pipelines[mode] = std::move(pipeline);
312 }
313
314 resourcesDirty = false;
315}

321void BrainRenderer::ensureRenderTargets(QRhi *rhi, QRhiTexture *colorTex, const QSize &pixelSize)
322{
323 // Rebuild when size changes OR when the backing texture changes
324 // (QRhiWidget may return a different colorTexture() each frame).
325 if (d->rtClear && d->rtSize == pixelSize && d->rtColorTex == colorTex)
326 return;
327
328 d->rtSize = pixelSize;
329 d->rtColorTex = colorTex;
330
331 // Shared depth-stencil buffer
332 d->dsBuffer.reset(rhi->newRenderBuffer(QRhiRenderBuffer::DepthStencil, pixelSize));
333 d->dsBuffer->create();
334
335 QRhiColorAttachment colorAtt(colorTex);
336 QRhiTextureRenderTargetDescription desc(colorAtt);
337 desc.setDepthStencilBuffer(d->dsBuffer.get());
338
339 // RT 1: Clearing (no preserve flags) — used for the first pass of each frame
340 d->rtClear.reset(rhi->newTextureRenderTarget(desc));
341 d->rpClear.reset(d->rtClear->newCompatibleRenderPassDescriptor());
342 d->rtClear->setRenderPassDescriptor(d->rpClear.get());
343 d->rtClear->create();
344
345 // RT 2: Preserving (load previous contents) — used for passes 2+
346 d->rtPreserve.reset(rhi->newTextureRenderTarget(desc,
347 QRhiTextureRenderTarget::PreserveColorContents
348 | QRhiTextureRenderTarget::PreserveDepthStencilContents));
349 d->rpPreserve.reset(d->rtPreserve->newCompatibleRenderPassDescriptor());
350 d->rtPreserve->setRenderPassDescriptor(d->rpPreserve.get());
351 d->rtPreserve->create();
352}

408void BrainRenderer::renderSurface(QRhiCommandBuffer *cb, QRhi *rhi, const SceneData &data, BrainSurface *surface, ShaderMode mode)
409{
410 if (!surface || !surface->isVisible()) return;
411
412 auto *pipeline = d->pipelines[mode].get();
413 if (!pipeline) return;
414
415 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
416 surface->updateBuffers(rhi, u);
417
418 // Dynamic slot update
419 int offset = d->currentUniformOffset;
420 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
421 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
422 qWarning("BrainRenderer: uniform buffer overflow (%d / %d bytes) — too many surfaces. Some draws will be skipped.",
423 d->currentUniformOffset, (int)d->uniformBuffer->size());
424 return; // Skip this draw rather than silently corrupt earlier viewport data
425 }
426
427 // On desktop, when a specific annotation region or vertex range is
428 // selected the CPU vertex-color gold tint (in updateVertexColors)
429 // provides per-region feedback. Suppress the shader's whole-surface
430 // gold glow so it doesn't drown out the region highlight.
431 float selected = surface->isSelected() ? 1.0f : 0.0f;
432#ifndef __EMSCRIPTEN__
433 if (surface->isSelected()
434 && (surface->selectedRegionId() != -1 || surface->selectedVertexStart() >= 0)) {
435 selected = 0.0f;
436 }
437#endif
438
439 // Pack ALL uniforms into a contiguous block for a single upload
440 struct {
441 float mvp[16]; // 0..63
442 float cameraPos[3]; // 64..75
443 float isSelected; // 76..79
444 float lightDir[3]; // 80..91
445 float tissueType; // 92..95
446 float lightingEnabled; // 96..99
447 float overlayMode; // 100..103
448 float selectedSurfaceId;// 104..107
449 } ub;
450 memcpy(ub.mvp, data.mvp.constData(), 64);
451 memcpy(ub.cameraPos, &data.cameraPos, 12);
452 ub.isSelected = selected;
453 memcpy(ub.lightDir, &data.lightDir, 12);
454 ub.tissueType = static_cast<float>(surface->tissueType());
455 ub.lightingEnabled = data.lightingEnabled ? 1.0f : 0.0f;
456 ub.overlayMode = data.overlayMode;
457 ub.selectedSurfaceId = -1.0f; // Per-surface path: surfaceId selection disabled
458
459 u->updateDynamicBuffer(d->uniformBuffer.get(), offset, sizeof(ub), &ub);
460
461 cb->resourceUpdate(u);
462
463 // Re-assert the per-pane viewport and scissor after resourceUpdate.
464 // The scissor provides a hard pixel clip that guarantees no cross-pane
465 // bleeding, regardless of Metal render-encoder restarts.
466 cb->setViewport(toViewport(data));
467 cb->setScissor(toScissor(data));
468
469 auto draw = [&](QRhiGraphicsPipeline *p) {
470 cb->setGraphicsPipeline(p);
471
472 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
473 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
474 const QRhiCommandBuffer::VertexInput vbuf(surface->vertexBuffer(), 0);
475 cb->setVertexInput(0, 1, &vbuf, surface->indexBuffer(), 0, QRhiCommandBuffer::IndexUInt32);
476 cb->drawIndexed(surface->indexCount());
477 };
478
479 if (mode == Holographic && d->pipelinesBackColor[Holographic]) {
480 draw(d->pipelinesBackColor[Holographic].get());
481 }
482
483 draw(pipeline);
484}

560void BrainRenderer::renderDipoles(QRhiCommandBuffer *cb, QRhi *rhi, const SceneData &data, DipoleObject *dipoles)
561{

563
564 auto *pipeline = d->pipelines[Dipole].get();
565 if (!pipeline) return;
566
567 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
568 dipoles->updateBuffers(rhi, u);
569
570 // Dynamic slot update
571 int offset = d->currentUniformOffset;
572 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
573 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
574 qWarning("BrainRenderer: uniform buffer overflow in renderDipoles");
575 return;
576 }
577
578 // Pack all uniforms into a single contiguous upload
579 struct {
580 float mvp[16]; // 0..63
581 float cameraPos[3]; // 64..75
582 float _pad0; // 76..79
583 float lightDir[3]; // 80..91
584 float _pad1; // 92..95
585 float lightingEnabled; // 96..99

588 memcpy(dub.cameraPos, &data.cameraPos, 12);
589 dub._pad0 = 0.0f;
590 memcpy(dub.lightDir, &data.lightDir, 12);
591 dub._pad1 = 0.0f;
592 dub.lightingEnabled = data.lightingEnabled ? 1.0f : 0.0f;
593 u->updateDynamicBuffer(d->uniformBuffer.get(), offset, sizeof(dub), &dub);
594
595 cb->resourceUpdate(u);
596
597 // Re-assert the per-pane viewport and scissor.
598 cb->setViewport(toViewport(data));
599 cb->setScissor(toScissor(data));
600
601 cb->setGraphicsPipeline(pipeline);
602
603 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
604 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
605
606 const QRhiCommandBuffer::VertexInput bindings[2] = {
607 QRhiCommandBuffer::VertexInput(dipoles->vertexBuffer(), 0),
608 QRhiCommandBuffer::VertexInput(dipoles->instanceBuffer(), 0)
609 };
610
611 cb->setVertexInput(0, 2, bindings, dipoles->indexBuffer(), 0, QRhiCommandBuffer::IndexUInt32);
612
613 cb->drawIndexed(dipoles->indexCount(), dipoles->instanceCount());
614}

618void BrainRenderer::renderNetwork(QRhiCommandBuffer *cb, QRhi *rhi, const SceneData &data, NetworkObject *network)
619{
620 if (!network || !network->isVisible() || !network->hasData()) return;
621
622 auto *pipeline = d->pipelines[Dipole].get();
623 if (!pipeline) return;
624
625 // --- Render Nodes (instanced spheres) ---
626 if (network->nodeInstanceCount() > 0) {
627 QRhiResourceUpdateBatch *uNodes = rhi->nextResourceUpdateBatch();
628 network->updateNodeBuffers(rhi, uNodes);
629
630 int offset = d->currentUniformOffset;
631 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
632 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
633 qWarning("BrainRenderer: uniform buffer overflow in renderNetwork (nodes)");
634 return;
635 }
636
637 struct { float mvp[16]; float cp[3]; float _p0; float ld[3]; float _p1; float le; } nub;
638 memcpy(nub.mvp, data.mvp.constData(), 64);
639 memcpy(nub.cp, &data.cameraPos, 12); nub._p0 = 0.0f;
640 memcpy(nub.ld, &data.lightDir, 12); nub._p1 = 0.0f;
641 nub.le = data.lightingEnabled ? 1.0f : 0.0f;
642 uNodes->updateDynamicBuffer(d->uniformBuffer.get(), offset, sizeof(nub), &nub);
643
644 cb->resourceUpdate(uNodes);
645 cb->setViewport(toViewport(data));
646 cb->setScissor(toScissor(data));
647
648 cb->setGraphicsPipeline(pipeline);
649
650 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
651 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
652
653 const QRhiCommandBuffer::VertexInput nodeBindings[2] = {
654 QRhiCommandBuffer::VertexInput(network->nodeVertexBuffer(), 0),
655 QRhiCommandBuffer::VertexInput(network->nodeInstanceBuffer(), 0)
656 };
657
658 cb->setVertexInput(0, 2, nodeBindings, network->nodeIndexBuffer(), 0, QRhiCommandBuffer::IndexUInt32);
659 cb->drawIndexed(network->nodeIndexCount(), network->nodeInstanceCount());
660 }
661
662 // --- Render Edges (instanced cylinders) ---
663 if (network->edgeInstanceCount() > 0) {
664 QRhiResourceUpdateBatch *uEdges = rhi->nextResourceUpdateBatch();
665 network->updateEdgeBuffers(rhi, uEdges);
666
667 int offset = d->currentUniformOffset;
668 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
669 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
670 qWarning("BrainRenderer: uniform buffer overflow in renderNetwork (edges)");
671 return;
672 }
673
674 struct { float mvp[16]; float cp[3]; float _p0; float ld[3]; float _p1; float le; } eub;
675 memcpy(eub.mvp, data.mvp.constData(), 64);
676 memcpy(eub.cp, &data.cameraPos, 12); eub._p0 = 0.0f;
677 memcpy(eub.ld, &data.lightDir, 12); eub._p1 = 0.0f;

700//=============================================================================================================
701// WORKAROUND(QRhi-GLES2): Merged single-drawIndexed rendering.
702// The Qt QRhi GLES2/WebGL backend has a bug where only the first
703// drawIndexed() per render pass produces visible output. These two
704// methods merge all surfaces (brain, BEM, sensors, digitizers,
705// source-space) into a single VBO/IBO so that all geometry is drawn
706// in one call.
707//
708// Remove when upstream Qt fixes the issue.
709//=============================================================================================================
710
711void BrainRenderer::prepareMergedSurfaces(QRhi *rhi, QRhiResourceUpdateBatch * /*u*/,
712 const QVector<BrainSurface*> &surfaces,
713 const QString &groupName)
714{
715 auto &group = d->mergedGroups[groupName];
716
717 // Check if surface list changed (different count or different pointers)
718 if (!group.dirty) {
719 if (group.surfaces.size() != surfaces.size()) {
720 group.dirty = true;
721 } else {
722 for (int i = 0; i < surfaces.size(); ++i) {
723 if (group.surfaces[i] != surfaces[i]) {
724 group.dirty = true;
725 break;
726 }
727 }
728 }
729 }
730
731 // If geometry hasn't changed, check if any surface vertex data actually changed
732 // (STC animation changes vertex colors but not topology)
733 if (!group.dirty && group.indexCount > 0) {
734 // Compare per-surface vertex generation counters
735 bool anyChanged = false;
736 if (group.surfaceGenerations.size() != surfaces.size()) {
737 anyChanged = true;
738 } else {
739 for (int i = 0; i < surfaces.size(); ++i) {
740 if (surfaces[i] && surfaces[i]->vertexGeneration() != group.surfaceGenerations[i]) {
741 anyChanged = true;
742 break;
743 }
744 }
745 }
746
747 if (!anyChanged) {
748 // Nothing changed — skip vertex rebuild entirely
749 return;
750 }
751
752 // Re-merge vertex data directly into vertexRaw (no temp allocation)
753 // Safety: verify vertex count hasn't changed since the full rebuild.
754 // If it has, fall through to the full rebuild path to update indices.
755 int totalVerts = 0;
756 for (int si = 0; si < surfaces.size(); ++si)
757 if (surfaces[si]) totalVerts += surfaces[si]->vertexDataRef().size();
758 if (totalVerts != group.totalVertexCount) {
759 group.dirty = true;
760 // Fall through to full rebuild below
761 } else {
762 float brainId = 0.0f;
763 float nonBrainId = 100.0f; // offset so shaders can distinguish
764 group.surfaceGenerations.resize(surfaces.size());
765 VertexData *dst = reinterpret_cast<VertexData*>(group.vertexRaw.data());
766 for (int si = 0; si < surfaces.size(); ++si) {
767 BrainSurface *surf = surfaces[si];
768 if (!surf) { brainId += 1.0f; nonBrainId += 1.0f; continue; }
769 const bool isBrain = (surf->tissueType() == BrainSurface::TissueBrain);
770 const float id = isBrain ? brainId : nonBrainId;
771 const auto &srcVerts = surf->vertexDataRef();
772 const int n = srcVerts.size();
773 memcpy(dst, srcVerts.constData(), n * sizeof(VertexData));
774 for (int j = 0; j < n; ++j)
775 dst[j].surfaceId = id;
776 dst += n;
777 group.surfaceGenerations[si] = surf->vertexGeneration();
778 brainId += 1.0f;
779 nonBrainId += 1.0f;
780 }
781 group.gpuVertexDirty = true;
782 return;
783 }
784 }
785
786 // Full rebuild: topology or surface list changed
787 group.surfaces = surfaces;
788 group.indexCount = 0;
789 group.totalVertexCount = 0;
790 group.dirty = false;
791
792 // Build merged vertex + index arrays
793 // Pre-calculate total sizes for single allocation
794 int totalVerts = 0;
795 int totalIndices = 0;
796 for (int si = 0; si < surfaces.size(); ++si) {
797 if (!surfaces[si]) continue;
798 totalVerts += surfaces[si]->vertexDataRef().size();
799 totalIndices += surfaces[si]->indexDataRef().size();
800 }
801
802 group.indexCount = totalIndices;
803 if (group.indexCount == 0) return;
804
805 const quint32 vbufSize = totalVerts * sizeof(VertexData);
806 const quint32 ibufSize = totalIndices * sizeof(uint32_t);
807
808 // (Re-)create Dynamic buffers when they don't exist or are too small
809 if (!group.vertexBuffer || group.vertexBuffer->size() < vbufSize) {
810 group.vertexBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
811 QRhiBuffer::VertexBuffer, vbufSize));
812 group.vertexBuffer->create();
813 }
814 if (!group.indexBuffer || group.indexBuffer->size() < ibufSize) {
815 group.indexBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
816 QRhiBuffer::IndexBuffer, ibufSize));
817 group.indexBuffer->create();
818 }
819
820 // Write directly into QByteArrays — no temp QVector intermediaries
821 group.vertexRaw.resize(vbufSize);
822 group.indexRaw.resize(ibufSize);
823 group.surfaceGenerations.resize(surfaces.size());
824
825 VertexData *vDst = reinterpret_cast<VertexData*>(group.vertexRaw.data());
826 uint32_t *iDst = reinterpret_cast<uint32_t*>(group.indexRaw.data());
827 float brainId = 0.0f;
828 float nonBrainId = 100.0f; // offset so shaders can distinguish
829 uint32_t vertexOffset = 0;
830 for (int si = 0; si < surfaces.size(); ++si) {
831 BrainSurface *surf = surfaces[si];
832 if (!surf) { brainId += 1.0f; nonBrainId += 1.0f; continue; }
833
834 const bool isBrain = (surf->tissueType() == BrainSurface::TissueBrain);
835 const float id = isBrain ? brainId : nonBrainId;
836 const auto &srcVerts = surf->vertexDataRef();
837 const auto &srcIdx = surf->indexDataRef();
838 const int nv = srcVerts.size();
839 const int ni = srcIdx.size();
840
841 // Bulk copy vertices + stamp surfaceId
842 memcpy(vDst, srcVerts.constData(), nv * sizeof(VertexData));
843 for (int j = 0; j < nv; ++j)
844 vDst[j].surfaceId = id;
845 vDst += nv;
846
847 // Copy indices with global vertex offset
848 const uint32_t *srcI = srcIdx.constData();
849 for (int j = 0; j < ni; ++j)
850 iDst[j] = srcI[j] + vertexOffset;
851 iDst += ni;
852
853 vertexOffset += nv;
854 group.surfaceGenerations[si] = surf->vertexGeneration();
855 brainId += 1.0f;
856 nonBrainId += 1.0f;
857 }
858 group.totalVertexCount = totalVerts;
859 group.gpuVertexDirty = true;
860 group.gpuIndexDirty = true;
861}

882
883void BrainRenderer::drawMergedSurfaces(QRhiCommandBuffer *cb, QRhi *rhi,
884 const SceneData &data, ShaderMode mode,
885 const QString &groupName)
886{
887 auto it = d->mergedGroups.find(groupName);
888 if (it == d->mergedGroups.end()) return;
889 auto &group = it->second;
890
891 if (group.indexCount == 0) return;
892
893 auto *pipeline = d->pipelines[mode].get();
894 if (!pipeline) return;
895
896 // Determine which merged surface (if any) is selected.
897 // surfaceId encoding: brain surfaces get ids 0,1,2...; non-brain get 100,101,102...
898 float selectedSurfaceId = -1.0f;
899 for (int i = 0; i < group.surfaces.size(); ++i) {
900 if (group.surfaces[i] && group.surfaces[i]->isSelected()) {
901 if (group.surfaces[i]->selectedRegionId() == -1
902 && group.surfaces[i]->selectedVertexStart() < 0) {
903 const bool isBrain = (group.surfaces[i]->tissueType() == BrainSurface::TissueBrain);
904 selectedSurfaceId = static_cast<float>(isBrain ? i : 100 + i);
905 }
906 break;
907 }
908 }
909
910 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
911
912 // Re-upload merged geometry only when data actually changed.
913 // Split VBO / IBO uploads: the fast-update path (STC color changes)
914 // only modifies vertices; re-uploading the IBO via glBufferSubData on
915 // WebGL can corrupt the VAO's element-buffer binding.
916 if (group.gpuVertexDirty) {
917 u->updateDynamicBuffer(group.vertexBuffer.get(), 0, group.vertexRaw.size(), group.vertexRaw.constData());
918 group.gpuVertexDirty = false;
919 }
920 if (group.gpuIndexDirty) {
921 u->updateDynamicBuffer(group.indexBuffer.get(), 0, group.indexRaw.size(), group.indexRaw.constData());
922 group.gpuIndexDirty = false;
923 }
924
925 int offset = d->currentUniformOffset;
926 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
927 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
928 qWarning("BrainRenderer: uniform buffer overflow in drawMergedSurfaces");
929 return;
930 }
931
932 // Pack all uniforms into a contiguous block for a single upload
933 // Layout must match the shader's UniformBlock (std140).
934 struct {
935 float mvp[16]; // 0..63
936 float cameraPos[3]; // 64..75
937 float isSelected; // 76..79
938 float lightDir[3]; // 80..91
939 float tissueType; // 92..95
940 float lightingEnabled; // 96..99
941 float overlayMode; // 100..103
942 float selectedSurfaceId;// 104..107
943 } ub;
944 memcpy(ub.mvp, data.mvp.constData(), 64);
945 memcpy(ub.cameraPos, &data.cameraPos, 12);
946 ub.isSelected = 0.0f;
947 memcpy(ub.lightDir, &data.lightDir, 12);
948 ub.tissueType = (!group.surfaces.isEmpty() && group.surfaces.first())
949 ? static_cast<float>(group.surfaces.first()->tissueType()) : 0.0f;
950 ub.lightingEnabled = data.lightingEnabled ? 1.0f : 0.0f;
951 ub.overlayMode = data.overlayMode;
952 ub.selectedSurfaceId = selectedSurfaceId;
953
954 u->updateDynamicBuffer(d->uniformBuffer.get(), offset, sizeof(ub), &ub);
955
956 cb->resourceUpdate(u);
957
958 cb->setViewport(toViewport(data));
959 cb->setScissor(toScissor(data));
960
961 auto draw = [&](QRhiGraphicsPipeline *p) {
962 cb->setGraphicsPipeline(p);
963 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
964 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
965 const QRhiCommandBuffer::VertexInput vbuf(group.vertexBuffer.get(), 0);
966 cb->setVertexInput(0, 1, &vbuf, group.indexBuffer.get(), 0, QRhiCommandBuffer::IndexUInt32);
967 cb->drawIndexed(group.indexCount);
968 };
969
970 // WORKAROUND(QRhi-GLES2): On WebGL, only one drawIndexed() per pass.
971 // For Holographic mode, the back-face pass must happen in a separate
972 // render pass. The caller is responsible for wrapping each call in
973 // its own beginPreservingPass/endPass on WASM.
974#ifdef __EMSCRIPTEN__
975 draw(pipeline);
976#else
977 if (mode == Holographic && d->pipelinesBackColor[Holographic]) {
978 draw(d->pipelinesBackColor[Holographic].get());
979 }
980
981 draw(pipeline);
982#endif
983}
