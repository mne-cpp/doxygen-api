108QRhiGraphicsPipeline::TargetBlend toRhi(GpuBlend blend, bool colorWrite)
109{
110 QRhiGraphicsPipeline::TargetBlend target;
111 if (!colorWrite) {
112 target.colorWrite = {};
113 }
114 switch (blend) {
115 case GpuBlend::None:
116 break;
117 case GpuBlend::Alpha:
118 target.enable = true;
119 target.srcColor = QRhiGraphicsPipeline::SrcAlpha;
120 target.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
121 target.srcAlpha = QRhiGraphicsPipeline::One;
122 target.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
123 break;
124 case GpuBlend::Additive:
125 target.enable = true;
126 target.srcColor = QRhiGraphicsPipeline::SrcAlpha;
127 target.dstColor = QRhiGraphicsPipeline::One;
128 target.srcAlpha = QRhiGraphicsPipeline::One;
129 target.dstAlpha = QRhiGraphicsPipeline::One;
130 break;
131 }
132 return target;
133}

223QRhiBuffer* RhiFrameRenderer::Impl::buffer(GpuHandle handle, const GpuResources& resources, QRhiResourceUpdateBatch* batch)
224{
225 const GpuBuffer* source = resources.buffer(handle);
226 if (!source || source->data.empty()) {
227 return nullptr;
228 }
229 Buffer& cached = buffers[handle];
230 if (cached.buffer && cached.revision == source->revision) {
231 return cached.buffer.get();
232 }
233 const auto size = static_cast<quint32>(source->data.size());
234 const QRhiBuffer::UsageFlags usage = source->usage == GpuBufferUsage::Vertex ? QRhiBuffer::VertexBuffer : QRhiBuffer::IndexBuffer;
235 // Buffers that change after their first upload become dynamic, e.g. streamed trace vertices
236 const QRhiBuffer::Type type = cached.uploads > 0 ? QRhiBuffer::Dynamic : QRhiBuffer::Static;
237 if (!cached.buffer || cached.buffer->size() < size || cached.buffer->type() != type) {
238 cached.buffer.reset(rhi->newBuffer(type, usage, size));
239 if (!cached.buffer->create()) {
240 mne::Log::warning("RhiFrameRenderer: cannot create a buffer of {} bytes", size);
241 cached.buffer.reset();
242 return nullptr;
243 }
244 }
245 if (type == QRhiBuffer::Dynamic) {
246 batch->updateDynamicBuffer(cached.buffer.get(), 0, size, source->data.data());
247 } else {
248 batch->uploadStaticBuffer(cached.buffer.get(), 0, size, source->data.data());
249 }
250 cached.revision = source->revision;
251 ++cached.uploads;
252 return cached.buffer.get();
253}

257QRhiTexture* RhiFrameRenderer::Impl::texture(GpuHandle handle, const GpuResources& resources, QRhiResourceUpdateBatch* batch)
258{
259 const GpuTexture* source = resources.texture(handle);
260 if (!source) {
261 return nullptr;
262 }
263 Texture& cached = textures[handle];
264 if (cached.texture && cached.revision == source->revision) {
265 return cached.texture.get();
266 }
267 const QSize size(std::max(source->pixel_width(), 1), std::max(source->pixel_height(), 1));
268 QImage image;
269 if (const Image* pixels = std::get_if<Image>(&source->content)) {
270 image = QImage(size, QImage::Format_RGBA8888);
271 image.fill(Qt::transparent);
272 if (pixels->width > 0 && pixels->height > 0) {
273 for (int y = 0; y < pixels->height; ++y) {
274 std::memcpy(image.scanLine(y), pixels->rgba.data() + static_cast<std::size_t>(y) * static_cast<std::size_t>(pixels->width) * 4,
275 static_cast<std::size_t>(pixels->width) * 4);
276 }
277 }
278 } else {
279 image = QImage(size, QImage::Format_RGBA8888);
280 image.setDevicePixelRatio(source->scale);
281 image.fill(Qt::transparent);
282 QPainter painter(&image);
283 painter.setRenderHint(QPainter::Antialiasing, false);
284 paintScene(painter, std::get<DisplayList>(source->content), font);
285 }
286 if (!cached.texture || cached.size != size) {
287 cached.texture.reset(rhi->newTexture(QRhiTexture::RGBA8, size));
288 if (!cached.texture->create()) {
289 cached.texture.reset();
290 return nullptr;
291 }
292 cached.size = size;
293 cached.serial = ++serial;
294 }
295 batch->uploadTexture(cached.texture.get(), QRhiTextureUploadEntry(0, 0, QRhiTextureSubresourceUploadDescription(image)));
296 cached.revision = source->revision;
297 cached.image = std::move(image);
298 return cached.texture.get();
299}

303const std::pair<QShader, QShader>* RhiFrameRenderer::Impl::shaderPair(const std::string& id)
304{
305 const auto found = shaders.find(id);
306 if (found != shaders.end()) {
307 return found->second.first.isValid() ? &found->second : nullptr;
308 }
309 std::pair<QShader, QShader> pair;
310 for (const std::string& prefix : prefixes) {
311 pair = {loadShader(QString::fromStdString(prefix + id + ".vert.qsb")), loadShader(QString::fromStdString(prefix + id + ".frag.qsb"))};
312 if (pair.first.isValid() && pair.second.isValid()) {
313 break;
314 }
315 }
316 if (!pair.first.isValid() || !pair.second.isValid()) {
317 mne::Log::warning("RhiFrameRenderer: no shaders for '{}'", id);
318 pair = {};
319 }
320 const auto& stored = shaders[id] = pair;
321 return stored.first.isValid() ? &stored : nullptr;
322}

326QRhiShaderResourceBindings* RhiFrameRenderer::Impl::shaderBindings(const GpuDraw& draw, const std::vector<quint32>& blockSizes,
327 const GpuResources& resources, QRhiResourceUpdateBatch* batch)
328{
329 std::string key = std::to_string(draw.pipeline) + "|b" + std::to_string(uniformsSerial);
330 std::vector<QRhiShaderResourceBinding> list;
331 for (std::size_t i = 0; i < draw.uniforms.size(); ++i) {
332 key += "|u" + std::to_string(draw.uniforms[i].binding) + ":" + std::to_string(blockSizes[i]);
333 list.push_back(QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(draw.uniforms[i].binding, kAllStages, uniforms.get(), blockSizes[i]));
334 }
335 for (const GpuTextureBinding& binding : draw.textures) {
336 QRhiTexture* tex = texture(binding.texture, resources, batch);
337 const GpuTexture* source = resources.texture(binding.texture);
338 if (!tex || !source) {
339 return nullptr;
340 }
341 key += "|t" + std::to_string(binding.binding) + ":" + std::to_string(textures[binding.texture].serial);
342 list.push_back(QRhiShaderResourceBinding::sampledTexture(binding.binding, kAllStages, tex,
343 source->linear_filter ? linearSampler.get() : nearestSampler.get()));
344 }
345 Bindings& entry = bindings[key];
346 entry.used = true;
347 std::unique_ptr<QRhiShaderResourceBindings>& srb = entry.srb;
348 if (!srb) {
349 srb.reset(rhi->newShaderResourceBindings());
350 srb->setBindings(list.begin(), list.end());
351 if (!srb->create()) {
352 srb.reset();
353 return nullptr;
354 }
355 }
356 return srb.get();
357}

361QRhiGraphicsPipeline* RhiFrameRenderer::Impl::pipeline(GpuHandle handle, const GpuResources& resources, QRhiRenderTarget* target,
362 QRhiShaderResourceBindings* layout)
363{
364 const GpuPipeline* source = resources.pipeline(handle);
365 if (!source) {
366 return nullptr;
367 }
368 Pipeline& cached = pipelines[handle];
369 const QVector<quint32> serialized = target->renderPassDescriptor()->serializedFormat();
370 const std::vector<std::uint32_t> format(serialized.begin(), serialized.end());
371 if (cached.pipeline && cached.renderPassFormat == format && cached.sampleCount == target->sampleCount()) {
372 return cached.pipeline.get();
373 }
374 const std::pair<QShader, QShader>* stages = shaderPair(source->shader);
375 if (!stages) {
376 return nullptr;
377 }
378
379 std::unique_ptr<QRhiGraphicsPipeline> created(rhi->newGraphicsPipeline());
380 created->setShaderStages({{QRhiShaderStage::Vertex, stages->first}, {QRhiShaderStage::Fragment, stages->second}});
381 QRhiVertexInputLayout inputLayout;
382 std::vector<QRhiVertexInputBinding> inputBindings;
383 inputBindings.reserve(source->bindings.size());
384 for (const GpuVertexBinding& binding : source->bindings) {
385 inputBindings.emplace_back(binding.stride, binding.per_instance ? QRhiVertexInputBinding::PerInstance : QRhiVertexInputBinding::PerVertex);
386 }
387 std::vector<QRhiVertexInputAttribute> attributes;
388 attributes.reserve(source->attributes.size());
389 for (const GpuVertexAttribute& attribute : source->attributes) {
390 attributes.emplace_back(attribute.binding, attribute.location, toRhi(attribute.format), attribute.offset);
391 }
392 inputLayout.setBindings(inputBindings.begin(), inputBindings.end());
393 inputLayout.setAttributes(attributes.begin(), attributes.end());
394 created->setVertexInputLayout(inputLayout);
395 created->setTopology(toRhi(source->topology));
396 created->setTargetBlends({toRhi(source->blend, source->color_write)});
397 created->setCullMode(source->cull == GpuCull::Back ? QRhiGraphicsPipeline::Back
398 : source->cull == GpuCull::Front ? QRhiGraphicsPipeline::Front
399 : QRhiGraphicsPipeline::None);
400 created->setFrontFace(source->front_counter_clockwise ? QRhiGraphicsPipeline::CCW : QRhiGraphicsPipeline::CW);
401 created->setDepthTest(source->depth_test);
402 created->setDepthWrite(source->depth_write);
403 created->setDepthOp(toRhi(source->depth_compare));
404 created->setFlags(QRhiGraphicsPipeline::UsesScissor);
405 created->setSampleCount(target->sampleCount());
406 created->setShaderResourceBindings(layout);
407 created->setRenderPassDescriptor(target->renderPassDescriptor());
408 if (!created->create()) {
409 mne::Log::warning("RhiFrameRenderer: cannot create the pipeline of '{}'", source->shader);
410 return nullptr;
411 }
412 cached.pipeline = std::move(created);
413 cached.renderPassFormat = format;
414 cached.sampleCount = target->sampleCount();
415 return cached.pipeline.get();
416}

476void RhiFrameRenderer::render(QRhi* rhi, QRhiRenderTarget* target, QRhiCommandBuffer* cb, const GpuFrame& frame, const GpuResources& resources)
477{
478 Impl& d = *m_pImpl;
479 if (d.rhi != rhi) {
480 d.release();
481 d.rhi = rhi;
482 }
483 if (!d.linearSampler) {
484 d.linearSampler.reset(rhi->newSampler(QRhiSampler::Linear, QRhiSampler::Linear, QRhiSampler::None, QRhiSampler::ClampToEdge,
485 QRhiSampler::ClampToEdge));
486 d.linearSampler->create();
487 d.nearestSampler.reset(rhi->newSampler(QRhiSampler::Nearest, QRhiSampler::Nearest, QRhiSampler::None, QRhiSampler::ClampToEdge,
488 QRhiSampler::ClampToEdge));
489 d.nearestSampler->create();
490 }
491 QRhiResourceUpdateBatch* batch = rhi->nextResourceUpdateBatch();
492
493 // The uniform blocks of all draws in one dynamic buffer, each at an aligned offset
494 const auto align = [rhi](quint32 size) {
495 const auto alignment = static_cast<quint32>(rhi->ubufAlignment());
496 return (size + alignment - 1) & ~(alignment - 1);
497 };
498 quint32 uniformBytes = 0;
499 for (const GpuPass& pass : frame.passes) {
500 for (const GpuDraw& draw : pass.draws) {
501 for (const GpuUniformBlock& block : draw.uniforms) {
502 uniformBytes += align(std::max<quint32>(static_cast<quint32>(block.data.size()), 16));
503 }
504 }
505 }
506 if (uniformBytes > 0 && (!d.uniforms || d.uniforms->size() < uniformBytes)) {
507 d.uniforms.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, std::max(uniformBytes * 2, 4096u)));
508 if (!d.uniforms->create()) {
509 d.uniforms.reset();
510 }
511 d.uniformsSerial = ++d.serial;
512 }
513
514 // Shader resource bindings unused in the last frame refer to replaced buffers or textures
515 std::erase_if(d.bindings, [](const auto& entry) {
516 return !entry.second.used;
517 });
518 for (auto& [key, entry] : d.bindings) {
519 entry.used = false;
520 }
521
522 // Upload everything and resolve the QRhi objects of the draws before the passes start
523 struct Prepared
524 {
525 const GpuDraw* draw = nullptr;
526 QRhiGraphicsPipeline* pipeline = nullptr;
527 QRhiShaderResourceBindings* srb = nullptr;
528 std::vector<QRhiCommandBuffer::DynamicOffset> offsets;
529 std::vector<QRhiCommandBuffer::VertexInput> inputs;
530 QRhiBuffer* indices = nullptr;
531 };
532 std::vector<std::vector<Prepared>> passes(frame.passes.size());
533 quint32 uniformOffset = 0;
534 for (std::size_t p = 0; p < frame.passes.size(); ++p) {
535 for (const GpuDraw& draw : frame.passes[p].draws) {
536 Prepared prepared;
537 prepared.draw = &draw;
538 std::vector<quint32> blockSizes;
539 for (const GpuUniformBlock& block : draw.uniforms) {
540 const auto size = static_cast<quint32>(block.data.size());
541 blockSizes.push_back(std::max<quint32>(size, 16));
542 if (d.uniforms && size > 0) {
543 batch->updateDynamicBuffer(d.uniforms.get(), uniformOffset, size, block.data.data());
544 }
545 prepared.offsets.push_back({block.binding, uniformOffset});
546 uniformOffset += align(std::max<quint32>(size, 16));
547 }
548 bool complete = !draw.uniforms.empty() ? d.uniforms != nullptr : true;
549 for (const GpuHandle handle : draw.vertex_buffers) {
550 QRhiBuffer* vertexBuffer = d.buffer(handle, resources, batch);
551 complete = complete && vertexBuffer;
552 prepared.inputs.emplace_back(vertexBuffer, 0);
553 }
554 if (draw.index_buffer != 0) {
555 prepared.indices = d.buffer(draw.index_buffer, resources, batch);
556 complete = complete && prepared.indices;
557 }
558 prepared.srb = complete ? d.shaderBindings(draw, blockSizes, resources, batch) : nullptr;
559 prepared.pipeline = prepared.srb ? d.pipeline(draw.pipeline, resources, target, prepared.srb) : nullptr;
560 if (prepared.pipeline) {
561 passes[p].push_back(std::move(prepared));
562 }
563 }
564 }
565
566 const QSize size = target->pixelSize();
567 bool inPass = false;
568 for (std::size_t p = 0; p < frame.passes.size(); ++p) {
569 const GpuPass& pass = frame.passes[p];
570 if (!inPass || pass.clear_color) {
571 if (inPass) {
572 cb->endPass();
573 }
574 cb->beginPass(target, toQColor(pass.clear_color.value_or(Color::transparent())), {1.f, 0}, batch);
575 batch = nullptr;
576 inPass = true;
577 }
578 for (const Prepared& prepared : passes[p]) {
579 const GpuDraw& draw = *prepared.draw;
580 cb->setGraphicsPipeline(prepared.pipeline);
581 const GpuRect viewport = draw.viewport.value_or(GpuRect{0, 0, size.width(), size.height()});
582 // QRhi viewports and scissors have their origin at the bottom left
583 cb->setViewport(QRhiViewport(static_cast<float>(viewport.x), static_cast<float>(size.height() - viewport.y - viewport.height),
584 static_cast<float>(viewport.width), static_cast<float>(viewport.height)));
585 const GpuRect scissor = draw.scissor.value_or(GpuRect{0, 0, size.width(), size.height()});
586 cb->setScissor(QRhiScissor(scissor.x, size.height() - scissor.y - scissor.height, scissor.width, scissor.height));
587 cb->setShaderResources(prepared.srb, static_cast<int>(prepared.offsets.size()), prepared.offsets.data());
588 if (prepared.indices) {
589 const GpuBuffer* indices = resources.buffer(draw.index_buffer);
590 cb->setVertexInput(0, static_cast<int>(prepared.inputs.size()), prepared.inputs.data(), prepared.indices, 0,
591 indices->usage == GpuBufferUsage::Index16 ? QRhiCommandBuffer::IndexUInt16 : QRhiCommandBuffer::IndexUInt32);
592 cb->drawIndexed(draw.count, draw.instance_count, draw.first);
593 } else {
594 cb->setVertexInput(0, static_cast<int>(prepared.inputs.size()), prepared.inputs.data());
595 cb->draw(draw.count, draw.instance_count, draw.first);
596 }
597 }
598 }
599 if (!inPass) {
600 cb->beginPass(target, toQColor(Color::transparent()), {1.f, 0}, batch);
601 }
602 cb->endPass();
603}

607std::vector<std::string> RhiFrameRenderer::checkShaders(const GpuFrame& frame, const GpuResources& resources) const
608{
609 std::vector<std::string> problems;
610 for (const GpuPass& pass : frame.passes) {
611 for (const GpuDraw& draw : pass.draws) {
612 const GpuPipeline* pipeline = resources.pipeline(draw.pipeline);
613 if (!pipeline) {
614 problems.emplace_back("draw without pipeline");
615 continue;
616 }
617 const std::string& id = pipeline->shader;
618 const std::pair<QShader, QShader>* stages = m_pImpl->shaderPair(pipeline->shader);
619 if (!stages) {
620 problems.push_back(std::format("{}: no shaders", id));
621 continue;
622 }
623 const QShaderDescription vertex = stages->first.description();
624 const QShaderDescription fragment = stages->second.description();
625
626 // Uniform blocks of both stages by binding
627 std::map<int, int> blocks;
628 std::map<int, int> samplers;
629 for (const QShaderDescription& description : {vertex, fragment}) {
630 for (const QShaderDescription::UniformBlock& block : description.uniformBlocks()) {
631 blocks[block.binding] = block.size;
632 }
633 for (const QShaderDescription::InOutVariable& sampler : description.combinedImageSamplers()) {
634 samplers[sampler.binding] = 1;
635 }
636 }
637 for (const GpuUniformBlock& block : draw.uniforms) {
638 const auto found = blocks.find(block.binding);
639 if (found == blocks.end()) {
640 problems.push_back(std::format("{}: no uniform block at binding {}", id, block.binding));
641 } else if (found->second != static_cast<int>(block.data.size())) {
642 problems.push_back(std::format("{}: uniform block {} has {} bytes, the shader {}", id, block.binding, block.data.size(), found->second));
643 }
644 blocks.erase(block.binding);
645 }
646 for (const auto& [binding, size] : blocks) {
647 problems.push_back(std::format("{}: uniform block {} not set", id, binding));
648 }
649 for (const GpuTextureBinding& texture : draw.textures) {
650 if (samplers.erase(texture.binding) == 0) {
651 problems.push_back(std::format("{}: no sampler at binding {}", id, texture.binding));
652 }
653 }
654 for (const auto& [binding, unused] : samplers) {
655 problems.push_back(std::format("{}: sampler {} not set", id, binding));
656 }
657
658 // Vertex inputs
659 std::map<int, int> inputs;
660 for (const QShaderDescription::InOutVariable& input : vertex.inputVariables()) {
661 // BATCHABLE shaders get an input for Qt Quick's batching that only Qt Quick feeds
662 if (input.name.startsWith("_qt_")) {
663 continue;
664 }
665 const int components = input.type == QShaderDescription::Float ? 1
666 : input.type == QShaderDescription::Vec2 ? 2
667 : input.type == QShaderDescription::Vec3 ? 3
668 : input.type == QShaderDescription::Vec4 ? 4
669 : 0;
670 inputs[input.location] = components;
671 }
672 for (const GpuVertexAttribute& attribute : pipeline->attributes) {
673 const int components = attribute.format == GpuVertexFormat::Float ? 1
674 : attribute.format == GpuVertexFormat::Float2 ? 2
675 : attribute.format == GpuVertexFormat::Float3 ? 3
676 : 4;
677 const auto found = inputs.find(attribute.location);
678 if (found == inputs.end()) {
679 problems.push_back(std::format("{}: no vertex input at location {}", id, attribute.location));
680 } else if (found->second != components) {
681 problems.push_back(std::format("{}: vertex input {} has {} components, the attribute {}", id, attribute.location, found->second, components));
682 }
683 inputs.erase(attribute.location);
684 }
685 for (const auto& [location, components] : inputs) {
686 problems.push_back(std::format("{}: vertex input {} not fed", id, location));
687 }
688 }
689 }
690 std::sort(problems.begin(), problems.end());
691 problems.erase(std::unique(problems.begin(), problems.end()), problems.end());
692 return problems;
693}
