46const char* modality_of(const mne::FiffChInfo& channel)
47{
48 if (channel.unit == FIFF_UNIT_T) {
49 return "MAG";
50 }
51 if (channel.unit == FIFF_UNIT_T_M) {
52 return "GRAD";
53 }
54 switch (channel.kind) {
55 case FIFFV_EEG_CH:
56 return "EEG";
57 case FIFFV_EOG_CH:
58 return "EOG";
59 case FIFFV_STIM_CH:
60 return "STIM";
61 case FIFFV_MISC_CH:
62 return "MISC";
63 default:
64 return nullptr;

74ModalitySelectionPanel::ModalitySelectionPanel(const std::vector<FiffChInfo>& channels)
75: Panel("ModalitySelectionView", "Modality Selection")
76{
77 set_width(330);
78 for (const FiffChInfo& channel : channels) {
79 const char* modality = modality_of(channel);
80 if (modality && std::ranges::find(m_present, modality) == m_present.end()) {
81 m_present.emplace_back(modality);
82 }
83 }
84 for (const char* modality : kModalities) {
85 m_modalities[modality] = true;
86 }
87 m_listener = form().add_listener([this](const std::string& key, const std::optional<SettingValue>& value) {
88 if (m_updating || !value || !key.starts_with(kPrefix)) {
89 return;
90 }
91 m_modalities[key.substr(std::char_traits<char>::length(kPrefix))] = form().get<bool>(key);
92 if (m_callback) {
93 m_callback(m_modalities);
94 }
95 });
96 rebuild();
97}

116void ModalitySelectionPanel::load(const SettingsStore& store, const std::string& path)
117{
118 if (path.empty()) {
119 return;
120 }
121 for (const char* modality : kModalities) {
122 const std::string key = settings_key(path, std::string(kPrefix) + modality);
123 m_modalities[modality] = store.read_bool(key, m_modalities.contains(modality) ? m_modalities.at(modality) : true);

142void ModalitySelectionPanel::rebuild()
143{
144 const bool updating = std::exchange(m_updating, true);
145 std::vector<std::string> keys;
146 keys.reserve(form().fields().size());
147 for (const SettingField& field : form().fields()) {
148 keys.push_back(field.key);
149 }
150 for (const std::string& key : keys) {
151 form().remove(key);
152 }
153 for (const auto& [modality, on] : m_modalities) {
154 const bool present = std::ranges::any_of(m_present, [&modality](const std::string& type) { return mne::iequals(type, modality); });
155 if (present) {
156 SettingField field;
157 field.key = kPrefix + modality;
158 field.kind = SettingField::Kind::Bool;
159 field.label = modality;
160 field.default_value = on;
161 field.persistent = false;
162 form().add(std::move(field));
163 }
164 }
165 m_updating = updating;
166}
