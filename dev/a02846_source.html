184std::string BIDSPath::basename() const
185{
186 const std::array<std::pair<std::string_view, const std::string*>, 10> entities{{
187 {"sub-", &m_sSubject},
188 {"ses-", &m_sSession},
189 {"task-", &m_sTask},
190 {"acq-", &m_sAcquisition},
191 {"run-", &m_sRun},
192 {"proc-", &m_sProcessing},
193 {"space-", &m_sSpace},
194 {"recording-", &m_sRecording},
195 {"split-", &m_sSplit},
196 {"desc-", &m_sDescription},
197 }};
198
199 std::string name;
200 const auto append = [&name](std::string_view part) {
201 if (!name.empty())
202 name += '_';
203 name += part;
204 };
205 for (const auto& [key, value] : entities) {
206 if (!value->empty())
207 append(std::string(key) + *value);
208 }
209 if (!m_sSuffix.empty())
210 append(m_sSuffix);
211
212 return name + m_sExtension;
213}

218{
219 std::filesystem::path dir = m_sRoot;
220
221 if (!m_sSubject.empty())
222 dir /= mne::path_from_utf8("sub-" + m_sSubject);
223
224 if (!m_sSession.empty())
225 dir /= mne::path_from_utf8("ses-" + m_sSession);
226
227 if (!m_sDatatype.empty())
228 dir /= mne::path_from_utf8(m_sDatatype);
229
230 return dir;

315std::vector<BIDSPath> BIDSPath::match() const
316{
317 // Like mne_bids.BIDSPath.match: every set entity must equal the file's, unset ones match anything
318 std::vector<BIDSPath> results;
319 std::error_code error;
320 if (m_sRoot.empty() || !std::filesystem::is_directory(m_sRoot, error))
321 return results;
322
323 for (auto it = std::filesystem::recursive_directory_iterator(m_sRoot, error);
324 it != std::filesystem::recursive_directory_iterator();
325 it.increment(error)) {
326 const std::string name = mne::utf8_from_path(it->path().filename());
327 if (name.starts_with('.')) {
328 if (it->is_directory(error))
329 it.disable_recursion_pending();
330 continue;
331 }
332 if (!it->is_regular_file(error))
333 continue;
334 const std::size_t dot = name.find('.');
335 if (dot == std::string::npos)
336 continue;
337
338 BIDSPath p;
339 p.m_sRoot = m_sRoot;
340 p.m_sExtension = name.substr(dot);
341 const std::string dirName = mne::utf8_from_path(it->path().parent_path().filename());
342 if (!dirName.starts_with("sub-") && !dirName.starts_with("ses-"))
343 p.m_sDatatype = dirName;
344
345 bool valid = true;
346 const mne::StringList parts = mne::split(std::string_view(name).substr(0, dot), '_');
347 for (std::size_t i = 0; i < parts.size() && valid; ++i) {
348 const std::size_t dash = parts[i].find('-');
349 if (dash == std::string::npos) {
350 // Only the last part may be the suffix
351 valid = (i == parts.size() - 1);
352 p.m_sSuffix = parts[i];
353 continue;
354 }
355 const std::string_view key = std::string_view(parts[i]).substr(0, dash);
356 std::string val = parts[i].substr(dash + 1);
357 // Labels are kept as on disk (the run and split setters would pad), so the path names the file found
358 if (key == "sub")
359 p.m_sSubject = std::move(val);
360 else if (key == "ses")
361 p.m_sSession = std::move(val);
362 else if (key == "task")
363 p.m_sTask = std::move(val);
364 else if (key == "acq")
365 p.m_sAcquisition = std::move(val);
366 else if (key == "run")
367 p.m_sRun = std::move(val);
368 else if (key == "proc")
369 p.m_sProcessing = std::move(val);
370 else if (key == "space")
371 p.m_sSpace = std::move(val);
372 else if (key == "recording")
373 p.m_sRecording = std::move(val);
374 else if (key == "split")
375 p.m_sSplit = std::move(val);
376 else if (key == "desc")
377 p.m_sDescription = std::move(val);
378 }
379 if (!valid || p.m_sSubject.empty())
380 continue;
381
382 const std::array<std::pair<const std::string*, const std::string*>, 13> wanted{{
383 {&m_sSubject, &p.m_sSubject},
384 {&m_sSession, &p.m_sSession},
385 {&m_sTask, &p.m_sTask},
386 {&m_sAcquisition, &p.m_sAcquisition},
387 {&m_sRun, &p.m_sRun},
388 {&m_sProcessing, &p.m_sProcessing},
389 {&m_sSpace, &p.m_sSpace},
390 {&m_sRecording, &p.m_sRecording},
391 {&m_sSplit, &p.m_sSplit},
392 {&m_sDescription, &p.m_sDescription},
393 {&m_sDatatype, &p.m_sDatatype},
394 {&m_sSuffix, &p.m_sSuffix},
395 {&m_sExtension, &p.m_sExtension},
396 }};
397 if (std::ranges::all_of(wanted, [](const auto& w) { return w.first->empty() || *w.first == *w.second; }))
398 results.push_back(std::move(p));
399 }
400
401 std::ranges::sort(results, {}, &BIDSPath::file_path);
402 return results;
403}

409bool BIDSPath::is_valid_entity_value(std::string_view sValue)
410{
411 // BIDS labels are alphanumeric ASCII
412 return std::ranges::all_of(sValue, [](char c) {
413 return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9');
414 });
415}
