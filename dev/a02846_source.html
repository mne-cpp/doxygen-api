44// Full-target quad as a strip: (x, y) in clip space, (u, v) with v = 0 at the top of the texture
45constexpr std::array<float, 16> kQuad = {-1.f, 1.f, 0.f, 0.f, -1.f, -1.f, 0.f, 1.f, 1.f, 1.f, 1.f, 0.f, 1.f, -1.f, 1.f, 1.f};

48constexpr double kWheelZoom = 1.15;
49// Angle of one wheel notch, in eighths of a degree
50constexpr double kWheelNotch = 120.0;

52constexpr double kScrollPerNotch = 48.0;
53// Gap around fitted contents
54constexpr double kFitMargin = 2.0;

69FigureView::FigureView(SceneFigure& figure, Theme theme)
70: m_figure(figure)
71, m_theme(std::move(theme))
72{
73 m_listener = m_figure.add_listener([this](SceneFigure::Change change) {
74 figure_changed(change);

76
77 GpuPipeline overlay;

79 overlay.bindings = {{4 * sizeof(float), false}};
80 overlay.attributes = {{0, 0, GpuVertexFormat::Float2, 0}, {1, 0, GpuVertexFormat::Float2, 2 * sizeof(float)}};
81 overlay.topology = GpuTopology::TriangleStrip;
82 overlay.blend = GpuBlend::Alpha;

85 m_resources.set_buffer_values(m_quad, std::span<const float>(kQuad));

158{
159 if (!(factor > 0.0) || !m_figure.zoomable()) {

163 m_zoom *= factor;
164 m_offset = {anchor.x - fixed.x * m_zoom, anchor.y - fixed.y * m_zoom};
165 m_fitted = false;

216DisplayList FigureView::draw() const
217{
218 Canvas canvas(m_width, m_height, m_background);
219 canvas.save();
220 canvas.translate(m_offset.x, m_offset.y);
221 canvas.scale(m_zoom, m_zoom);
222 m_figure.draw(canvas, *m_metrics, m_font);
223 canvas.restore();
224
225 if (const std::optional<Rect> rect = band()) {
226 Color fill = m_theme.accent;
227 fill.a = kBandAlpha;
228 canvas.set_pen(Pen{m_theme.accent, 1.0});
229 canvas.set_brush(Brush::solid(fill));
230 canvas.draw_rect(*rect);
231 }
232 return canvas.take();
233}

237DisplayList FigureView::picture(double width, double height) const
238{
239 Canvas canvas(width, height, m_background);
240 if (!m_figure.zoomable()) {
241 m_figure.resize(width, height);
242 m_figure.draw(canvas, *m_metrics, m_font);
243 m_figure.resize(m_width, m_height);
244 return canvas.take();
245 }
246 const Rect bounds = m_figure.bounds();
247 if (!bounds.empty() && width > 2 * kFitMargin && height > 2 * kFitMargin) {
248 const double zoom = std::min((width - 2 * kFitMargin) / bounds.width, (height - 2 * kFitMargin) / bounds.height);
249 canvas.translate(width / 2.0 - bounds.center().x * zoom, height / 2.0 - bounds.center().y * zoom);
250 canvas.scale(zoom, zoom);
251 m_figure.draw(canvas, *m_metrics, m_font);
252 }
253 return canvas.take();
254}

265bool FigureView::handle_pointer(const PointerEvent& event)
266{
267 const Point at{event.x, event.y};
268 switch (event.type) {
269 case PointerEvent::Type::Press:
270 if (event.button == MouseButton::Right) {
271 m_drag = Drag::Pan;
272 m_last = at;
273 return true;
274 }
275 if (event.button == MouseButton::Left) {
276 if (m_figure.press(map_to_scene(at), event.modifiers)) {
277 m_drag = Drag::Item;
278 } else if (m_figure.band_selection_enabled()) {
279 m_drag = Drag::Band;
280 m_bandStart = m_bandEnd = at;
281 m_bandModifiers = event.modifiers;
282 update();
283 }
284 return true;
285 }
286 return false;
287 case PointerEvent::Type::Move:
288 if (m_drag == Drag::Pan) {
289 pan_by(at.x - m_last.x, at.y - m_last.y);
290 m_last = at;
291 } else if (m_drag == Drag::Band) {
292 m_bandEnd = at;
293 update();
294 } else if (m_drag == Drag::None) {
295 m_figure.hover(map_to_scene(at));
296 }
297 return true;
298 case PointerEvent::Type::Release: {
299 const Drag drag = std::exchange(m_drag, Drag::None);
300 if (drag == Drag::Item) {
301 m_figure.release(map_to_scene(at));
302 } else if (drag == Drag::Band) {
303 m_bandEnd = at;
304 const Point a = map_to_scene(m_bandStart);
305 const Point b = map_to_scene(m_bandEnd);
306 m_figure.band_selected(normalized(a, b), m_bandModifiers);
307 update();
308 }
309 return drag != Drag::None;
310 }
311 case PointerEvent::Type::DoubleClick:
312 if (event.button != MouseButton::Left) {
313 return false;
314 }
315 if (!m_figure.double_click(map_to_scene(at)) && m_figure.zoomable()) {
316 fit_to_contents();
317 }
318 return true;
319 }
320 return false;
321}

325bool FigureView::handle_wheel(const WheelEvent& event)
326{
327 if (event.angle_delta_y == 0.0) {
328 return false;
329 }
330 if (m_figure.zoomable()) {
331 zoom_by(event.angle_delta_y > 0.0 ? kWheelZoom : 1.0 / kWheelZoom, {event.x, event.y});
332 return true;
333 }
334 const double before = m_offset.y;
335 pan_by(0.0, event.angle_delta_y / kWheelNotch * kScrollPerNotch);
336 return m_offset.y != before;
337}

341GpuFrame FigureView::frame(const GpuFrameTarget& target)
342{
343 const double ratio = target.device_pixel_ratio > 0.0 ? target.device_pixel_ratio : 1.0;
344 if (m_width <= 0.0 || m_height <= 0.0) {
345 set_view_size(target.pixel_width / ratio, target.pixel_height / ratio);
346 }
347
348 GpuFrame frame;
349 frame.pixel_width = target.pixel_width;
350 frame.pixel_height = target.pixel_height;
351 frame.device_pixel_ratio = ratio;
352 GpuPass pass;
353 pass.clear_color = m_background;
354
355 DisplayList list = draw();
356 if (m_texture == 0) {
357 GpuTexture texture;
358 texture.content = std::move(list);
359 texture.scale = ratio;
360 texture.linear_filter = false;
361 m_texture = m_resources.create_texture(std::move(texture));
362 } else {
363 m_resources.set_texture(m_texture, std::move(list), ratio);
364 }
365 GpuDraw quad;
366 quad.pipeline = m_pipeline;
367 quad.vertex_buffers = {m_quad};
368 quad.count = 4;
369 quad.textures = {{0, m_texture}};
370 pass.draws.push_back(std::move(quad));
371 frame.passes.push_back(std::move(pass));
372 return frame;
373}

384void FigureView::apply_fit()
385{
386 const Rect bounds = m_figure.bounds();
387 if (bounds.empty() || m_width <= 2 * kFitMargin || m_height <= 2 * kFitMargin) {
388 return;
389 }
390 m_zoom = std::min((m_width - 2 * kFitMargin) / bounds.width, (m_height - 2 * kFitMargin) / bounds.height);
391 m_offset = {m_width / 2.0 - bounds.center().x * m_zoom, m_height / 2.0 - bounds.center().y * m_zoom};
392}

396void FigureView::clamp_scroll()
397{
398 // Figures that are not zoomable draw in view units and only scroll vertically
399 m_zoom = 1.0;
400 m_offset.x = 0.0;
401 const Rect bounds = m_figure.bounds();
402 const double overflow = bounds.empty() ? 0.0 : std::max(0.0, bounds.bottom() - m_height);
403 m_offset.y = std::clamp(m_offset.y, -overflow, 0.0);
404}
