101double SplinePlot::data_to_pixel_x(double x, double width) const
102{
103 const double plotW = width - kLeftMargin - kRightMargin;
104 if (m_maxX == m_minX) {
105 return kLeftMargin;
106 }
107 return kLeftMargin + (x - m_minX) / (m_maxX - m_minX) * plotW;
108}

112double SplinePlot::data_to_pixel_y(double y, double height) const
113{
114 const double plotH = height - kTopMargin - kBottomMargin;
115 // Headroom for the curve, which overshoots between the class marks
116 const double count = max_count();
117 const double maxY = count > 0 ? 1.1 * count : 1.0;
118 return kTopMargin + plotH - (y / maxY) * plotH;
119}

143void SplinePlot::draw(Canvas& canvas, const TextMetrics& /*metrics*/, const Font& font) const
144{
145 const double w = canvas.width();
146 const double h = canvas.height();
147 const double mLeft = kLeftMargin;
148 const double mTop = kTopMargin;
149 const double plotW = w - kLeftMargin - kRightMargin;
150 const double plotH = h - kTopMargin - kBottomMargin;
151 if (plotW <= 0 || plotH <= 0) {
152 return;
153 }
154 canvas.save();
155 canvas.set_antialiasing(true);
156 canvas.set_font(font);
157
158 // Colour map between the thresholds
159 if (m_hasThresholds && !m_colormap.empty() && m_maxX > m_minX) {
160 const double leftNorm = (m_left - m_minX) / (m_maxX - m_minX);
161 const double middleNorm = (m_middle - m_minX) / (m_maxX - m_minX);
162 const double rightNorm = (m_right - m_minX) / (m_maxX - m_minX);
163 std::vector<GradientStop> stops;
164 const auto stop = [&](double position, double value) {
165 const double at = std::clamp(position, 0.0, 1.0);
166 // A later stop at the same position replaces the earlier one, as in QGradient::setColorAt
167 std::erase_if(stops, [at](const GradientStop& s) { return s.position == at; });
168 stops.push_back({at, map_color(m_colormap, value)});
169 };
170 const int steps = 25;
171 const double stepLeftMiddle = (middleNorm - leftNorm) / steps;
172 const double stepMiddleRight = (rightNorm - middleNorm) / steps;
173 stop(leftNorm, 0.0);
174 for (int i = 1; i < steps; ++i) {
175 stop(leftNorm + stepLeftMiddle * i, i * (0.5 / steps));
176 stop(middleNorm + stepMiddleRight * i, 0.5 + i * (0.5 / steps));
177 }
178 stop(rightNorm, 1.0);
179 std::stable_sort(stops.begin(), stops.end(), [](const GradientStop& a, const GradientStop& b) { return a.position < b.position; });
180 canvas.fill_rect({mLeft, mTop, plotW, plotH}, Brush::linear_gradient({mLeft, 0}, {mLeft + plotW, 0}, std::move(stops)));
181 }
182
183 // Axes
184 canvas.set_pen(Pen{Color::black(), 1});
185 canvas.draw_line({mLeft, mTop}, {mLeft, mTop + plotH});
186 canvas.draw_line({mLeft, mTop + plotH}, {mLeft + plotW, mTop + plotH});
187
188 const double count = max_count();
189 for (double value : nice_ticks(0.0, count > 0 ? 1.1 * count : 1.0, 5, 1.0)) {
190 const double y = data_to_pixel_y(value, h);
191 canvas.draw_line({mLeft - 5, y}, {mLeft, y});
192 canvas.draw_text(Rect{0, y - 10, mLeft - 8, 20}, Align::Right | Align::VCenter, tick_label(value));
193 }
194 for (double value : nice_ticks(m_minX, m_maxX, 6)) {
195 const double x = data_to_pixel_x(value, w);
196 canvas.draw_line({x, mTop + plotH}, {x, mTop + plotH + 5});
197 canvas.draw_text(Rect{x - 40, mTop + plotH + 6, 80, 20}, Align::Center, tick_label(value));
198 }
199
200 if (!m_title.empty()) {
201 Font titleFont = font;
202 titleFont.bold = true;
203 canvas.set_font(titleFont);
204 canvas.draw_text(Rect{0, 0, w, mTop}, Align::Center, m_title);
205 canvas.set_font(font);
206 }
207 if (!m_xLabel.empty()) {
208 canvas.draw_text(Rect{mLeft, h - 22, plotW, 20}, Align::Center, m_xLabel);
209 }
210 if (!m_yLabel.empty()) {
211 canvas.save();
212 canvas.translate(4, mTop + plotH / 2.0);
213 canvas.rotate(-90);
214 canvas.draw_text(Rect{-plotH / 2.0, 0, plotH, 20}, Align::Center, m_yLabel);
215 canvas.restore();
216 }
217
218 if (m_marks.empty()) {
219 canvas.restore();
220 return;
221 }
222 const auto pixel = [&](std::ptrdiff_t index) {
223 index = std::clamp<std::ptrdiff_t>(index, 0, std::ssize(m_marks) - 1);
224 const Point& mark = m_marks[static_cast<std::size_t>(index)];
225 return Point{data_to_pixel_x(mark.x, w), data_to_pixel_y(mark.y, h)};
226 };
227 if (m_marks.size() == 1) {
228 canvas.set_brush(Brush::solid({70, 130, 180}));
229 canvas.draw_ellipse(pixel(0), 3, 3);
230 } else {
231 // Catmull-Rom spline as cubic Bezier segments
232 std::vector<Point> curve{pixel(0)};
233 for (std::ptrdiff_t i = 0; i + 1 < std::ssize(m_marks); ++i) {
234 const Point p0 = pixel(i - 1);
235 const Point p1 = pixel(i);
236 const Point p2 = pixel(i + 1);
237 const Point p3 = pixel(i + 2);
238 curve.push_back(p1 + (p2 - p0) * (1.0 / 6.0));
239 curve.push_back(p2 - (p3 - p1) * (1.0 / 6.0));
240 curve.push_back(p2);
241 }
242 canvas.set_pen(Pen{{70, 130, 180}, 2});
243 canvas.set_brush(Brush{});
244 canvas.draw_bezier(curve);
245 }
246
247 if (m_hasThresholds) {
248 const auto thresholdLine = [&](double x, Color color) {
249 if (x >= m_minX && x <= m_maxX) {
250 const double px = data_to_pixel_x(x, w);
251 canvas.set_pen(Pen{color, 2});
252 canvas.draw_line({px, mTop}, {px, mTop + plotH});
253 }
254 };
255 thresholdLine(m_left, {255, 0, 0});
256 thresholdLine(m_middle, {0, 255, 0});
257 thresholdLine(m_right, {0, 0, 255});
258 }
259 canvas.restore();
260}
