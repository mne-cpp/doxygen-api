108static QImage tightlyPackedRgba(const QImage &source)
109{
110 if (source.isNull())
111 return {};
112
113 QImage rgba = source.convertToFormat(QImage::Format_RGBA8888);
114 const qsizetype tightStride = qsizetype(rgba.width()) * 4;
115 if (rgba.bytesPerLine() == tightStride)
116 return rgba.copy();
117
118 QImage packed(rgba.size(), QImage::Format_RGBA8888);
119 for (int y = 0; y < rgba.height(); ++y) {
120 memcpy(packed.scanLine(y), rgba.constScanLine(y), size_t(tightStride));
121 }
122 return packed;
123}

170void BrainRenderer::Impl::createResources(QRhi *rhi, QRhiRenderPassDescriptor *rp, int sampleCount)
171{
172 uniformBufferOffsetAlignment = rhi->ubufAlignment();
173
174 // Create Uniform Buffer
175 if (!uniformBuffer) {
176 // Size for 8192 slots with alignment — enough for 4 viewports × ~1000 surfaces
177 uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, kUniformSlotCount * uniformBufferOffsetAlignment));
178 uniformBuffer->create();
179 }
180
181 // Create SRB
182 if (!srb) {
183 srb.reset(rhi->newShaderResourceBindings());
184 srb->setBindings({
185 // Use dynamic offset for the uniform buffer.
186 // The size of one uniform block in the shader is ~104 bytes,
187 // but we use uniformBufferOffsetAlignment for the stride.
188 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage, uniformBuffer.get(),kUniformBlockSize)
189 });
190 srb->create();
191 }
192
193 // Shader Loader
194 auto getShader = [](const QString &name) {
195 QFile f(name);
196 return f.open(QIODevice::ReadOnly) ? QShader::fromSerialized(f.readAll()) : QShader();
197 };
198
199 // List of modes to initialize
200 QList<ShaderMode> modes = {Standard, Holographic, Anatomical, Dipole, XRay, ShowNormals};
201
202 for (ShaderMode mode : modes) {
203 QString vert = (mode == Holographic || mode == XRay) ? ":/holographic.vert.qsb" :
204 (mode == Anatomical) ? ":/anatomical.vert.qsb" :
205 (mode == Dipole) ? ":/dipole.vert.qsb" :
206 (mode == ShowNormals) ? ":/shownormals.vert.qsb" : ":/standard.vert.qsb";
207
208 QString frag = (mode == Holographic || mode == XRay) ? ":/holographic.frag.qsb" :
209 (mode == Anatomical) ? ":/anatomical.frag.qsb" :
210 (mode == Dipole) ? ":/dipole.frag.qsb" :
211 (mode == ShowNormals) ? ":/shownormals.frag.qsb" : ":/standard.frag.qsb";
212
213 QShader vS = getShader(vert);
214 QShader fS = getShader(frag);
215
216 if (!vS.isValid() || !fS.isValid()) {
217 qWarning() << "BrainRenderer: Could not load shaders for mode" << mode << vert << frag;
218 continue;
219 }
220
221 // Setup Pipeline
222 auto pipeline = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
223
224 QRhiGraphicsPipeline::TargetBlend blend;
225 if (mode == Holographic || mode == XRay) {
226 blend.enable = true;
227 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
228 blend.dstColor = QRhiGraphicsPipeline::One;
229 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
230 blend.dstAlpha = QRhiGraphicsPipeline::One;
231 } else if (mode == Dipole) {
232 blend.enable = true;
233 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
234 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
235 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
236 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
237 }
238
239 auto setup = [&](QRhiGraphicsPipeline* p, QRhiGraphicsPipeline::CullMode cull) {
240 p->setShaderStages({{ QRhiShaderStage::Vertex, vS }, { QRhiShaderStage::Fragment, fS }});
241
242 QRhiVertexInputLayout il;
243
244 if (mode == Dipole) {
245 il.setBindings({
246 { 6 * sizeof(float) }, // Binding 0: Vertex Data (Pos + Normal) -> stride 6 floats
247 { 21 * sizeof(float), QRhiVertexInputBinding::PerInstance } // Binding 1: Instance Data (Mat4 + Color + Selected) -> stride 21 floats
248 });
249
250 il.setAttributes({
251 // Vertex Buffer (Binding 0)
252 { 0, 0, QRhiVertexInputAttribute::Float3, 0 }, // Pos
253 { 0, 1, QRhiVertexInputAttribute::Float3, 3 * sizeof(float) }, // Normal
254
255 // Instance Buffer (Binding 1)
256 // Model Matrix (4 x vec4)
257 { 1, 2, QRhiVertexInputAttribute::Float4, 0 },
258 { 1, 3, QRhiVertexInputAttribute::Float4, 4 * sizeof(float) },
259 { 1, 4, QRhiVertexInputAttribute::Float4, 8 * sizeof(float) },
260 { 1, 5, QRhiVertexInputAttribute::Float4, 12 * sizeof(float) },
261 // Color
262 { 1, 6, QRhiVertexInputAttribute::Float4, 16 * sizeof(float) },
263 // isSelected
264 { 1, 7, QRhiVertexInputAttribute::Float, 20 * sizeof(float) }
265 });
266 } else {
267 il.setBindings({{ 32 }}); // sizeof(VertexData) = 32
268 il.setAttributes({{ 0, 0, QRhiVertexInputAttribute::Float3, 0 },
269 { 0, 1, QRhiVertexInputAttribute::Float3, 12 },
270 { 0, 2, QRhiVertexInputAttribute::UNormByte4, 24 },
271 { 0, 3, QRhiVertexInputAttribute::UNormByte4, 28 }});
272 }
273
274 p->setVertexInputLayout(il);
275 p->setShaderResourceBindings(srb.get());
276 p->setRenderPassDescriptor(rp);
277 p->setSampleCount(sampleCount);
278 p->setCullMode(cull);
279 if (mode == Holographic) {
280 p->setTargetBlends({blend});
281 p->setDepthTest(true);
282 p->setDepthWrite(false);
283 } else if (mode == XRay) {
284 p->setTargetBlends({blend});
285 p->setDepthTest(false); // Disable Depth Test to see through head
286 p->setDepthWrite(false);
287 } else if (mode == Dipole) {
288 p->setTargetBlends({blend});
289 p->setCullMode(QRhiGraphicsPipeline::None);
290 p->setDepthTest(true);
291 p->setDepthWrite(false);
292 } else {
293 p->setDepthTest(true);
294 p->setDepthWrite(true);
295 }
296 p->setFlags(QRhiGraphicsPipeline::UsesScissor);
297 p->create();
298 };
299
300 if (mode == Holographic || mode == XRay) { // Handle XRay back-faces same as Holographic
301 auto pipelineBack = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
302 setup(pipelineBack.get(), QRhiGraphicsPipeline::Front);
303 pipelinesBackColor[mode] = std::move(pipelineBack);
304 setup(pipeline.get(), QRhiGraphicsPipeline::Back); // Front faces
305 } else {
306 // Culling: None (Double-sided) to be safe for FreeSurfer meshes
307 setup(pipeline.get(), QRhiGraphicsPipeline::None);
308 }
309 pipelines[mode] = std::move(pipeline);
310 }
311
312 resourcesDirty = false;
313}

319void BrainRenderer::ensureRenderTargets(QRhi *rhi, QRhiTexture *colorTex, const QSize &pixelSize)
320{
321 // Rebuild when size changes OR when the backing texture changes
322 // (QRhiWidget may return a different colorTexture() each frame).
323 if (d->rtClear && d->rtSize == pixelSize && d->rtColorTex == colorTex)
324 return;
325
326 d->rtSize = pixelSize;
327 d->rtColorTex = colorTex;
328
329 // Shared depth-stencil buffer
330 d->dsBuffer.reset(rhi->newRenderBuffer(QRhiRenderBuffer::DepthStencil, pixelSize));
331 d->dsBuffer->create();
332
333 QRhiColorAttachment colorAtt(colorTex);
334 QRhiTextureRenderTargetDescription desc(colorAtt);
335 desc.setDepthStencilBuffer(d->dsBuffer.get());
336
337 // RT 1: Clearing (no preserve flags) — used for the first pass of each frame
338 d->rtClear.reset(rhi->newTextureRenderTarget(desc));
339 d->rpClear.reset(d->rtClear->newCompatibleRenderPassDescriptor());
340 d->rtClear->setRenderPassDescriptor(d->rpClear.get());
341 d->rtClear->create();
342
343 // RT 2: Preserving (load previous contents) — used for passes 2+
344 d->rtPreserve.reset(rhi->newTextureRenderTarget(desc,
345 QRhiTextureRenderTarget::PreserveColorContents
346 | QRhiTextureRenderTarget::PreserveDepthStencilContents));
347 d->rpPreserve.reset(d->rtPreserve->newCompatibleRenderPassDescriptor());
348 d->rtPreserve->setRenderPassDescriptor(d->rpPreserve.get());
349 d->rtPreserve->create();
350}

408void BrainRenderer::prepareVideoOverlay(QRhi *rhi,
409 QRhiResourceUpdateBatch *u,
410 VideoOverlay *overlay)
411{
412 if (!overlay || !overlay->isEnabled()) return;
413 if (!overlay->hasFrame()) return;
414 if (!rhi || !u) return;
415
416 QImage frame = tightlyPackedRgba(overlay->frame());
417 if (frame.isNull()) return;
418
419 auto &k = d->videoOverlay;
420 k.currentUniformOffset = 0;
421
422 // ── Lazy resource creation ──────────────────────────────────────
423 if (!k.initialized) {
424 // Shaders
425 QFile vFile(QStringLiteral(":/video_overlay.vert.qsb"));
426 QFile fFile(QStringLiteral(":/video_overlay.frag.qsb"));
427 if (!vFile.open(QIODevice::ReadOnly) || !fFile.open(QIODevice::ReadOnly)) {
428 qWarning() << "BrainRenderer: failed to open video overlay shaders";
429 return;
430 }
431 QShader vShader = QShader::fromSerialized(vFile.readAll());
432 QShader fShader = QShader::fromSerialized(fFile.readAll());
433 if (!vShader.isValid() || !fShader.isValid()) {
434 qWarning() << "BrainRenderer: invalid video overlay shaders";
435 return;
436 }
437
438 // Uniform buffer (mat4 + vec4 + 4 floats = 96 bytes, pad to 256)
439 constexpr int kUbSize = 256;
440 k.uniformBufferOffsetAlignment = rhi->ubufAlignment();
441 k.uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
442 QRhiBuffer::UniformBuffer,
443 64 * k.uniformBufferOffsetAlignment));
444 k.uniformBuffer->create();
445
446 // Vertex buffer: 4 vertices × (3 pos + 2 uv) floats — refilled each frame
447 constexpr int kVbSize = 4 * 5 * sizeof(float);
448 k.vertexBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
449 QRhiBuffer::VertexBuffer, kVbSize));
450 k.vertexBuffer->create();
451
452 // Index buffer: 2 triangles, 6 indices — uploaded once
453 constexpr int kIbSize = 6 * sizeof(quint32);
454 k.indexBuffer.reset(rhi->newBuffer(QRhiBuffer::Immutable,
455 QRhiBuffer::IndexBuffer, kIbSize));
456 k.indexBuffer->create();
457
458 // Sampler (linear, clamp)
459 k.sampler.reset(rhi->newSampler(QRhiSampler::Linear, QRhiSampler::Linear,
460 QRhiSampler::None,
461 QRhiSampler::ClampToEdge,
462 QRhiSampler::ClampToEdge));
463 k.sampler->create();
464
465 // Initial texture uses the first real frame size.
466 k.texture.reset(rhi->newTexture(QRhiTexture::RGBA8, frame.size()));
467 k.texture->create();
468 k.textureSize = frame.size();
469
470 // SRB binding: uniform @ 0, sampled image @ 1
471 k.srb.reset(rhi->newShaderResourceBindings());
472 k.srb->setBindings({
473 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
474 QRhiShaderResourceBinding::VertexStage |
475 QRhiShaderResourceBinding::FragmentStage,
476 k.uniformBuffer.get(), kUniformBlockSize),
477 QRhiShaderResourceBinding::sampledTexture(1,
478 QRhiShaderResourceBinding::FragmentStage,
479 k.texture.get(), k.sampler.get())
480 });
481 k.srb->create();
482
483 // Pipeline (alpha-blended, depth-test off so it sits on top)
484 QRhiGraphicsPipeline::TargetBlend blend;
485 blend.enable = true;
486 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
487 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
488 blend.srcAlpha = QRhiGraphicsPipeline::One;
489 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
490
491 k.pipeline.reset(rhi->newGraphicsPipeline());
492 k.pipeline->setShaderStages({
493 { QRhiShaderStage::Vertex, vShader },
494 { QRhiShaderStage::Fragment, fShader }
495 });
496 QRhiVertexInputLayout il;
497 il.setBindings({{ 5 * sizeof(float) }});
498 il.setAttributes({
499 { 0, 0, QRhiVertexInputAttribute::Float3, 0 },
500 { 0, 1, QRhiVertexInputAttribute::Float2, 3 * sizeof(float) }
501 });
502 k.pipeline->setVertexInputLayout(il);
503 k.pipeline->setShaderResourceBindings(k.srb.get());
504 k.pipeline->setRenderPassDescriptor(d->rtClear->renderPassDescriptor());
505 k.pipeline->setSampleCount(d->rtClear->sampleCount());
506 k.pipeline->setCullMode(QRhiGraphicsPipeline::None);
507 k.pipeline->setTargetBlends({blend});
508 k.pipeline->setDepthTest(false);
509 k.pipeline->setDepthWrite(false);
510 k.pipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);
511 k.pipeline->create();
512
513 QFile dvFile(QStringLiteral(":/video_decal.vert.qsb"));
514 QFile dfFile(QStringLiteral(":/video_decal.frag.qsb"));
515 if (!dvFile.open(QIODevice::ReadOnly) || !dfFile.open(QIODevice::ReadOnly)) {
516 qWarning() << "BrainRenderer: failed to open video decal shaders";
517 return;
518 }
519 QShader dvShader = QShader::fromSerialized(dvFile.readAll());
520 QShader dfShader = QShader::fromSerialized(dfFile.readAll());
521 if (!dvShader.isValid() || !dfShader.isValid()) {
522 qWarning() << "BrainRenderer: invalid video decal shaders";
523 return;
524 }
525
526 QRhiGraphicsPipeline::TargetBlend decalBlend;
527 decalBlend.enable = true;
528 decalBlend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
529 decalBlend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
530 decalBlend.srcAlpha = QRhiGraphicsPipeline::One;
531 decalBlend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
532
533 k.surfacePipeline.reset(rhi->newGraphicsPipeline());
534 k.surfacePipeline->setShaderStages({
535 { QRhiShaderStage::Vertex, dvShader },
536 { QRhiShaderStage::Fragment, dfShader }
537 });
538 QRhiVertexInputLayout dil;
539 dil.setBindings({{ 36 }});
540 dil.setAttributes({
541 { 0, 0, QRhiVertexInputAttribute::Float3, 0 },
542 { 0, 1, QRhiVertexInputAttribute::Float3, 12 },
543 { 0, 2, QRhiVertexInputAttribute::UNormByte4, 24 },
544 { 0, 3, QRhiVertexInputAttribute::UNormByte4, 28 },
545 { 0, 4, QRhiVertexInputAttribute::Float, 32 }
546 });
547 k.surfacePipeline->setVertexInputLayout(dil);
548 k.surfacePipeline->setShaderResourceBindings(k.srb.get());
549 k.surfacePipeline->setRenderPassDescriptor(d->rtClear->renderPassDescriptor());
550 k.surfacePipeline->setSampleCount(d->rtClear->sampleCount());
551 k.surfacePipeline->setCullMode(QRhiGraphicsPipeline::None);
552 k.surfacePipeline->setTargetBlends({decalBlend});
553 k.surfacePipeline->setDepthTest(true);
554 k.surfacePipeline->setDepthWrite(false);
555 k.surfacePipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);
556 k.surfacePipeline->create();
557
558 k.initialized = true;
559 }
560
561 // ── Index buffer (one-shot upload) ──────────────────────────────
562 if (!k.indexUploaded) {
563 const quint32 idx[6] = { 0, 1, 2, 2, 1, 3 };
564 u->uploadStaticBuffer(k.indexBuffer.get(), idx);
565 k.indexUploaded = true;
566 }
567
568 // ── Texture (re-create on size change, re-upload on new frame) ──
569 if (frame.size() != k.textureSize) {
570 k.texture.reset(rhi->newTexture(QRhiTexture::RGBA8, frame.size()));
571 k.texture->create();
572 k.textureSize = frame.size();
573 // Rebuild SRB to point at the new texture
574 k.srb->setBindings({
575 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
576 QRhiShaderResourceBinding::VertexStage |
577 QRhiShaderResourceBinding::FragmentStage,
578 k.uniformBuffer.get(), kUniformBlockSize),
579 QRhiShaderResourceBinding::sampledTexture(1,
580 QRhiShaderResourceBinding::FragmentStage,
581 k.texture.get(), k.sampler.get())
582 });
583 k.srb->create();
584 k.uploadedFrameGen = std::numeric_limits<quint64>::max();
585 }
586 if (overlay->frameGeneration() != k.uploadedFrameGen) {
587 QRhiTextureSubresourceUploadDescription sub(frame);
588 QRhiTextureUploadDescription desc({ 0, 0, sub });
589 u->uploadTexture(k.texture.get(), desc);
590 k.uploadedFrameGen = overlay->frameGeneration();
591 }
592}

594void BrainRenderer::renderVideoOverlay(QRhiCommandBuffer *cb, QRhi *rhi,
595 const SceneData &data,
596 VideoOverlay *overlay)
597{
598 if (!overlay || !overlay->isEnabled()) return;
599 if (!overlay->hasFrame()) return;
600
601 auto &k = d->videoOverlay;
602 if (!k.initialized) return;
603 if (k.uniformBufferOffsetAlignment <= 0) return;
604
605 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
606 const int uniformOffset = k.currentUniformOffset;
607 k.currentUniformOffset += k.uniformBufferOffsetAlignment;
608 if (uniformOffset + kUniformBlockSize > k.uniformBuffer->size()) return;
609
610 // ── Billboard the quad to face the camera ──────────────────────
611 const QVector3D centre = overlay->focusPosition();
612 const float halfSize = 0.5f * overlay->sizeMeters();
613
614 QVector3D viewDir = centre - data.cameraPos;
615 if (viewDir.lengthSquared() < 1e-12f) viewDir = QVector3D(0, 0, -1);
616 viewDir.normalize();
617
618 QVector3D worldUp(0.0f, 0.0f, 1.0f);
619 if (std::abs(QVector3D::dotProduct(viewDir, worldUp)) > 0.95f) {
620 worldUp = QVector3D(0.0f, 1.0f, 0.0f);
621 }
622 QVector3D right = QVector3D::crossProduct(viewDir, worldUp).normalized();
623 QVector3D up = QVector3D::crossProduct(right, viewDir).normalized();
624
625 const QVector3D c00 = centre - right * halfSize - up * halfSize;
626 const QVector3D c10 = centre + right * halfSize - up * halfSize;
627 const QVector3D c01 = centre - right * halfSize + up * halfSize;
628 const QVector3D c11 = centre + right * halfSize + up * halfSize;
629
630 const float verts[4 * 5] = {
631 c00.x(), c00.y(), c00.z(), 0.0f, 1.0f, // image y is flipped vs uv
632 c10.x(), c10.y(), c10.z(), 1.0f, 1.0f,
633 c01.x(), c01.y(), c01.z(), 0.0f, 0.0f,
634 c11.x(), c11.y(), c11.z(), 1.0f, 0.0f
635 };
636 u->updateDynamicBuffer(k.vertexBuffer.get(), 0, sizeof(verts), verts);
637
638 // ── Uniforms (mat4 mvp, vec4 borderColor, float opacity, 3 pad) ─
639 struct {

750void BrainRenderer::renderSurface(QRhiCommandBuffer *cb, QRhi *rhi, const SceneData &data, BrainSurface *surface, ShaderMode mode)
751{
752 if (!surface || !surface->isVisible()) return;
753
754 auto *pipeline = d->pipelines[mode].get();
755 if (!pipeline) return;
756
757 // NOTE: Buffer uploads are handled in the pre-render phase
758 // (BrainView::render pre-upload loop). Do not call
759 // surface->updateBuffers() here — it would allocate a redundant
760 // QRhiResourceUpdateBatch per surface.
761
762 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
763
764 // Dynamic slot update
765 int offset = d->currentUniformOffset;
766 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
767 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
768 qWarning("BrainRenderer: uniform buffer overflow (%d / %d bytes) — too many surfaces. Some draws will be skipped.",
769 d->currentUniformOffset, (int)d->uniformBuffer->size());
770 return; // Skip this draw rather than silently corrupt earlier viewport data
771 }
772
773 // On desktop, when a specific annotation region or vertex range is
774 // selected the CPU vertex-color gold tint (in updateVertexColors)
775 // provides per-region feedback. Suppress the shader's whole-surface
776 // gold glow so it doesn't drown out the region highlight.
777 float selected = surface->isSelected() ? 1.0f : 0.0f;
778#ifndef __EMSCRIPTEN__
779 if (surface->isSelected()
780 && (surface->selectedRegionId() != -1 || surface->selectedVertexStart() >= 0)) {
781 selected = 0.0f;
782 }
783#endif
784
785 // Pack ALL uniforms into a contiguous block for a single upload
786 struct {
787 float mvp[16]; // 0..63
788 float cameraPos[3]; // 64..75
789 float isSelected; // 76..79
790 float lightDir[3]; // 80..91
791 float tissueType; // 92..95
792 float lightingEnabled; // 96..99
793 float overlayMode; // 100..103
794 } ub;
795 memcpy(ub.mvp, data.mvp.constData(), 64);
796 memcpy(ub.cameraPos, &data.cameraPos, 12);
797 ub.isSelected = selected;
798 memcpy(ub.lightDir, &data.lightDir, 12);
799 ub.tissueType = static_cast<float>(surface->tissueType());
800 ub.lightingEnabled = data.lightingEnabled ? 1.0f : 0.0f;
801 ub.overlayMode = data.overlayMode;
802
803 u->updateDynamicBuffer(d->uniformBuffer.get(), offset, sizeof(ub), &ub);
804
805 cb->resourceUpdate(u);
806
807 // Re-assert the per-pane viewport and scissor after resourceUpdate.
808 // The scissor provides a hard pixel clip that guarantees no cross-pane
809 // bleeding, regardless of Metal render-encoder restarts.
810 cb->setViewport(toViewport(data));
811 cb->setScissor(toScissor(data));
812
813 auto draw = [&](QRhiGraphicsPipeline *p) {
814 cb->setGraphicsPipeline(p);
815
816 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
817 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
818 const QRhiCommandBuffer::VertexInput vbuf(surface->vertexBuffer(), 0);
819 cb->setVertexInput(0, 1, &vbuf, surface->indexBuffer(), 0, QRhiCommandBuffer::IndexUInt32);
820 cb->drawIndexed(surface->indexCount());
821 };
822
823 if (mode == Holographic && d->pipelinesBackColor[Holographic]) {
824 draw(d->pipelinesBackColor[Holographic].get());
825 }
826
827 draw(pipeline);
828}
