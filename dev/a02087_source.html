68TFplot::TFplot(const MatrixXd& tf_matrix,
69 qreal sample_rate,
70 qreal lower_frq,
71 qreal upper_frq,
72 ColorMaps cmap,
73 qreal tmin)
74{
75 if (tf_matrix.rows() == 0 || tf_matrix.cols() == 0 || sample_rate <= 0) {
76 return;
77 }
78
79 // Row k is bin k of an FFT over all N columns
80 const qreal binWidth = sample_rate / tf_matrix.cols();
81 const Index last = tf_matrix.rows() - 1;
82 const Index first = std::clamp<Index>(static_cast<Index>(std::ceil(std::min(lower_frq, upper_frq) / binWidth)), 0, last);
83 const Index top = std::clamp<Index>(static_cast<Index>(std::floor(std::max(lower_frq, upper_frq) / binWidth)), first, last);
84
85 // Highest frequency in the first row, drawn at the top
86 MatrixXd power = tf_matrix.middleRows(first, top - first + 1).colwise().reverse().cwiseAbs();
87 const double maxPower = power.maxCoeff();
88 if (maxPower > 0.0) {
89 power /= maxPower;
90 }
91
92 setColorMap(colorMapName(cmap));
93 setColorRange(0.0, 1.0);
94 updateData(power);
95 setExtent(tmin, tmin + (tf_matrix.cols() - 1) / sample_rate, first * binWidth, top * binWidth);
96 setXLabel(QStringLiteral("Time [s]"));
97 setYLabel(QStringLiteral("Frequency [Hz]"));
98 setColorbarLabel(QStringLiteral("Power (normalized)"));
99}
