115bool BrainViewInteraction::handle_pointer(const PointerEvent& event)
116{
117 const Point pos = pixel_of(event.x, event.y);
118 switch (event.type) {
119 case PointerEvent::Type::Press:
120 press(pos, event.button);
121 return true;
122 case PointerEvent::Type::Move:
123 move(pos, has_flag(event.buttons, MouseButtons::Left));
124 return true;
125 case PointerEvent::Type::Release:
126 release(pos, event.button);
127 return true;
128 case PointerEvent::Type::DoubleClick:
129 return event.button == MouseButton::Left && m_effects.double_clicked && m_effects.double_clicked(pos);
130 }
131 return false;
132}

136void BrainViewInteraction::press(Point pos, MouseButton button)
137{
138 m_last_pos = pos;
139 if (button != MouseButton::Left) {
140 return;
141 }
142 m_dragged_since_press = false;
143 if (!m_multi_view) {
144 return;
145 }
146
147 const int viewport = viewport_at(pos);
148 if (viewport >= 0 && m_effects.hits_viewport_label && m_effects.hits_viewport_label(viewport, pos)) {
149 notify(m_effects.viewport_selected, viewport);
150 notify(m_effects.preset_menu_requested, viewport, pos);
151 return;
152 }
153
154 const SplitterHit hit = m_layout.hit_test_splitter(pos, static_cast<int>(m_viewports.size()), m_size);
155 if (hit != SplitterHit::None) {
156 m_dragging_splitter = true;
157 m_active_splitter = hit;
158 update_cursor(pos);
159 return;
160 }
161
162 if (viewport >= 0) {
163 notify(m_effects.viewport_selected, viewport);
164 }
165}

169void BrainViewInteraction::move(Point pos, bool left_held)
170{
171 if (m_dragging_splitter && left_held) {
172 m_layout.drag_splitter(pos, m_active_splitter, m_size);
173 m_last_pos = pos;
174 notify(m_effects.layout_changed);
175 return;
176 }
177
178 if (!left_held) {
179 if (m_multi_view) {
180 update_cursor(pos);
181 } else {
182 notify(m_effects.cursor_changed, CursorShape::Default);
183 }
184 notify(m_effects.hovered, pos);
185 return;
186 }
187
188 const double dx = pos.x - m_last_pos.x;
189 const double dy = pos.y - m_last_pos.y;
190 m_last_pos = pos;
191
192 if (!m_multi_view) {
193 if (m_single) {
194 m_single->rotation = OrbitCamera::orbit(m_single->rotation, dx, dy);
195 }
196 m_dragged_since_press = true;
197 notify(m_effects.redraw);
198 return;
199 }
200
201 PaneCamera* camera = pane_camera(viewport_at(pos));
202 if (!camera) {
203 return;
204 }
205 // The perspective pane orbits, the planar panes pan along their view plane
206 if (view_preset_is_perspective(clamp_view_preset(camera->preset))) {
207 camera->rotation = OrbitCamera::orbit(camera->rotation, dx, dy);
208 } else {
209 camera->pan = OrbitCamera::panned(camera->pan, dx, dy, m_scene_size);
210 }
211 m_dragged_since_press = true;
212 notify(m_effects.redraw);
213}

217void BrainViewInteraction::release(Point pos, MouseButton button)
218{
219 if (button == MouseButton::Left) {
220 if (m_dragging_splitter) {
221 m_dragging_splitter = false;
222 m_active_splitter = SplitterHit::None;
223 notify(m_effects.settings_changed);
224 update_cursor(pos);
225 return;
226 }
227 // Rotations and pans of the panes are persisted when the drag ends
228 if (m_multi_view) {
229 notify(m_effects.settings_changed);
230 }
231 if (!m_dragged_since_press) {
232 notify(m_effects.clicked, pos);
233 }
234 }
235
236 if (m_multi_view) {
237 update_cursor(pos);
238 } else {
239 notify(m_effects.cursor_changed, CursorShape::Default);
240 }
241}

245void BrainViewInteraction::handle_wheel(const WheelEvent& event)
246{
247 const auto delta = static_cast<float>(event.angle_delta_y / 120.0);
248 PaneCamera* camera = m_multi_view ? pane_camera(viewport_at(pixel_of(event.x, event.y))) : m_single;
249 if (camera) {

251 notify(m_effects.settings_changed);
