46BadChannelsMaxwellResult mne::findBadChannelsMaxwell(
47 const MatrixXd& matData,
48 const FiffInfo& info,
49 const BadChannelsMaxwellParams& params)
50{
51 BadChannelsMaxwellResult result;
52
53 // Build SSS basis
54 SSSParams sssParams;
55 sssParams.iOrderIn = params.iOrderIn;
56 sssParams.iOrderOut = params.iOrderOut;
57 sssParams.origin = params.origin;
58
59 SSS::Basis basis = SSS::computeBasis(info, sssParams);
60
61 if (basis.megChannelIdx.empty()) {
62 Log::warning("[findBadChannelsMaxwell] No MEG channels found.");
63 return result;
64 }
65
66 const int nMeg = static_cast<int>(basis.megChannelIdx.size());
67
68 // Extract MEG data
69 int nTimes = static_cast<int>(matData.cols());
70 MatrixXd megData(nMeg, nTimes);
71 for (int i = 0; i < nMeg; ++i) {
72 megData.row(i) = matData.row(basis.megChannelIdx[i]);
73 }
74
75 // Apply SSS reconstruction
76 MatrixXd reconData = SSS::apply(matData, basis);
77
78 // Extract reconstructed MEG channels
79 MatrixXd megRecon(nMeg, nTimes);
80 for (int i = 0; i < nMeg; ++i) {
81 megRecon.row(i) = reconData.row(basis.megChannelIdx[i]);
82 }
83
84 // Compute per-channel residual (RMS of difference)
85 result.residuals.resize(nMeg);
86 for (int i = 0; i < nMeg; ++i) {
87 VectorXd diff = megData.row(i) - megRecon.row(i);
88 result.residuals(i) = std::sqrt(diff.squaredNorm() / nTimes);
89 }
90
91 // Compute z-scores
92 // Use median/MAD for robustness
93 VectorXd sorted = result.residuals;
94 std::sort(sorted.data(), sorted.data() + sorted.size());
95
96 double median = sorted(nMeg / 2);
97 if (nMeg % 2 == 0 && nMeg > 1) {
98 median = (sorted(nMeg / 2 - 1) + sorted(nMeg / 2)) / 2.0;
99 }
100
101 VectorXd absDev(nMeg);
102 for (int i = 0; i < nMeg; ++i) {
103 absDev(i) = std::abs(result.residuals(i) - median);
104 }
105 std::sort(absDev.data(), absDev.data() + absDev.size());
106
107 double mad = absDev(nMeg / 2);
108 if (nMeg % 2 == 0 && nMeg > 1) {
109 mad = (absDev(nMeg / 2 - 1) + absDev(nMeg / 2)) / 2.0;
110 }
111
112 // Robust scale: MAD * 1.4826 (consistency constant for normal distribution)
113 double scale = mad * 1.4826;
114 if (scale < 1e-30)
115 scale = 1e-30;
116
117 result.zScores.resize(nMeg);
118 for (int i = 0; i < nMeg; ++i) {
119 result.zScores(i) = (result.residuals(i) - median) / scale;
120 }
121
122 // Flag channels exceeding threshold
123 for (int i = 0; i < nMeg; ++i) {
124 if (result.zScores(i) > params.dZThreshold) {
125 int origIdx = basis.megChannelIdx[i];
126 result.badIndices.push_back(origIdx);
127 if (origIdx < static_cast<int>(info.ch_names.size())) {
128 result.badChannels.push_back(info.ch_names[origIdx]);
129 }
130 }
131 }
132
133 Log::info("[findBadChannelsMaxwell] Detected {} bad channel(s) out of {} MEG channels.", static_cast<long long>(result.badChannels.size()), nMeg);
134
135 return result;
136}
