76AcqExpected<Config> parseConfig(const AcqUri& uri)
77{
78 if (!uri.target.empty()) {
79 return invalid("target '" + uri.target + "': neuromag:// reads the acquisition of this machine and takes no target");
80 }
81 const auto simulate = paramBool(uri, "simulate", false);
82 const auto port = paramInt(uri, "port", 11122, 1, 65535);
83 const auto client = paramInt(uri, "client", 13014, 1, 1 << 20);
84 const auto buffer = paramInt(uri, "buffer", 0, 0, 100000);
85 const auto autostart = paramBool(uri, "autostart", true);
86 const auto timeout = paramDouble(uri, "timeout", 10.0, 0.0, 3600.0);
87 const auto meg = paramInt(uri, "meg", 306, 0, 1024);
88 const auto eeg = paramInt(uri, "eeg", 0, 0, 512);
89 const auto rate = paramDouble(uri, "rate", 1000.0, 1.0, 1e5);
90 const auto seed = paramInt(uri, "seed", 0, 0, INT64_MAX);
91 const auto realtime = paramBool(uri, "realtime", true);
92 const auto duration = paramDouble(uri, "duration", 0.0, 0.0, 1e7);
93 const auto gain = paramDouble(uri, "gain", 1.0, 0.0, 1e6);
94 auto thread = paramThreadPolicy(uri);
95 for (const AcqError& error : {simulate.error(), port.error(), client.error(), buffer.error(), autostart.error(), timeout.error(),
96 meg.error(), eeg.error(), rate.error(), seed.error(), realtime.error(), duration.error(), gain.error(),
97 thread.error()}) {
98 if (!error.ok()) {
99 return error;
100 }
101 }
102 Config c;
103 c.simulate = *simulate;
104 c.dacq.root = std::string(uri.param("root").value_or(c.dacq.root));
105 c.dacq.host = std::string(uri.param("host").value_or(c.dacq.host));
106 c.dacq.port = static_cast<int>(*port);
107 c.dacq.client = static_cast<int>(*client);
108 c.dacq.buffer = static_cast<int>(*buffer);
109 c.dacq.autostart = *autostart;
110 c.simulation.meg = static_cast<std::size_t>(*meg);
111 c.simulation.eeg = static_cast<std::size_t>(*eeg);
112 c.simulation.rate = *rate;
113 c.simulation.buffer = *buffer > 0 ? static_cast<std::size_t>(*buffer) : 100;
114 c.simulation.seed = static_cast<std::uint64_t>(*seed);
115 c.simulation.realtime = *realtime;
116 c.simulation.duration = *duration;
117 c.simulation.gain = *gain;
118 c.timeout = std::chrono::milliseconds(static_cast<std::int64_t>(*timeout * 1000.0));
119 c.thread = std::move(*thread);
120 return c;
121}

129AcqExpected<std::vector<NeuromagTag>> read_meas_info(NeuromagTransport& transport, std::chrono::milliseconds timeout)
130{
131 const auto deadline = std::chrono::steady_clock::now() + timeout;
132 std::vector<NeuromagTag> tags;
133 bool inside = false;
134 while (true) {
135 const auto left = std::chrono::duration_cast<std::chrono::milliseconds>(deadline - std::chrono::steady_clock::now());
136 if (left.count() <= 0) {
137 return AcqError(AcqErrorCode::Timeout, "the Neuromag acquisition sent no measurement info");
138 }
139 auto tag = transport.next(std::min(left, kWait));
140 if (!tag) {
141 if (tag.error() == AcqErrorCode::Timeout) {
142 continue;
143 }
144 return tag.error();
145 }
146 const NeuromagTag& t = **tag;
147 if (t.kind == fiff::kErrorMessage) {
148 return AcqError(AcqErrorCode::DeviceIo,
149 "the Neuromag acquisition reports: " + std::string(reinterpret_cast<const char*>(t.data.data()), t.data.size()));
150 }
151 inside = inside || is_block(t, fiff::kBlockStart, fiff::kBlockMeasInfo);
152 if (inside) {
153 tags.push_back(t);
154 if (is_block(t, fiff::kBlockEnd, fiff::kBlockMeasInfo)) {
155 return tags;
156 }
157 }
158 }

164 static AcqExpected<std::unique_ptr<AcqBackend>> open(const AcqUri& uri)
165 {
166 auto config = parseConfig(uri);
167 if (!config) {
168 return config.error();
169 }
170 std::unique_ptr<NeuromagTransport> transport;
171 if (config->simulate) {
172 transport = make_neuromag_simulator(config->simulation);
173 } else {
174 auto client = make_neuromag_client(config->dacq);
175 if (!client) {
176 return client.error();
177 }
178 transport = std::move(*client);
179 }
180
181 // The measurement info comes at the start of a measurement
182 if (AcqError e = transport->start(); !e.ok()) {
183 return e;
184 }
185 auto meas_info = read_meas_info(*transport, config->timeout);
186 transport->stop();
187 if (!meas_info) {
188 return meas_info.error();
189 }
190 auto info = stream_info(*meas_info, transport->description(), AcqSampleFormat::Int32);
191 if (!info) {
192 return info.error();
193 }
194 const std::optional<std::size_t> trigger = trigger_channel(*info);
195 const std::size_t width = info->channelCount();
196 const double rate = info->sfreq;
197
198 std::unique_ptr<NeuromagBackend> backend(new NeuromagBackend(*config, std::move(transport), uri.toString()));
199 backend->m_width = width;
200 backend->m_rate = rate;
201 backend->m_trigger_channel = trigger;
202 AcqStreamOptions options;
203 options.capacitySamples = static_cast<std::size_t>(std::max(4.0 * rate, 4096.0));
204 auto writer = backend->addInputStream(std::move(*info), options);
205 if (!writer) {
206 return writer.error();
207 }
208 backend->m_writer = *writer;
209 return std::unique_ptr<AcqBackend>(std::move(backend));
210 }

225 AcqError start() override
226 {
227 if (AcqError e = m_transport->start(); !e.ok()) {
228 return e;
229 }
230 // A new measurement repeats its info; its channels must be those of the stream
231 auto meas_info = read_meas_info(*m_transport, m_config.timeout);
232 if (meas_info) {
233 if (auto info = stream_info(*meas_info, m_transport->description(), AcqSampleFormat::Int32); !info || info->channelCount() != m_width) {
234 meas_info = AcqError(AcqErrorCode::InvalidState, "the Neuromag measurement has other channels than when it was opened");
235 }
236 }
237 if (!meas_info) {
238 m_transport->stop();
239 return meas_info.error();
240 }
241 m_trigger.reset();
242 const auto now = std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now().time_since_epoch()).count();
243 AcqClockModel clock;
244 clock.sfreq = m_rate;
245 clock.anchorSample = m_writer->writePosition();
246 clock.anchorHostNs = now;
247 clock.utcOffsetNs =
248 std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::system_clock::now().time_since_epoch()).count() - now;
249 clock.source = AcqClockSource::Nominal;
250 m_writer->setClock(clock);
251 m_stop.store(false, std::memory_order_relaxed);
252 auto thread = start_thread(m_config.thread, [this] { run(); });
253 if (!thread) {
254 m_transport->stop();
255 return thread.error();
256 }
257 m_thread = std::move(*thread);
258 return {};
259 }
260
261 void stop() override
262 {
263 m_stop.store(true, std::memory_order_release);
264 if (m_thread.joinable()) {
265 m_thread.join();
266 m_transport->stop();
267 }
268 }

278 void run()
279 {
280 while (!m_stop.load(std::memory_order_acquire)) {
281 auto tag = m_transport->next(kWait);
282 if (!tag) {
283 if (tag.error() == AcqErrorCode::Timeout) {
284 continue;
285 }
286 if (tag.error() == AcqErrorCode::EndOfStream) {
287 m_writer->finish();
288 } else {
289 reportError(tag.error());
290 }
291 return;
292 }
293 const NeuromagTag& t = **tag;
294 if (t.kind == fiff::kCloseFile) {
295 m_writer->finish(); // the measurement was stopped at the console
296 return;
297 }
298 if (t.kind == fiff::kErrorMessage) {
299 reportError({AcqErrorCode::DeviceIo,
300 "the Neuromag acquisition reports: " + std::string(reinterpret_cast<const char*>(t.data.data()), t.data.size())});
301 return;
302 }
303 if (t.kind == fiff::kDataBuffer && !write(t.data)) {
304 return;
305 }
306 }
307 }

310 bool write(std::span<const std::byte> data)
311 {
312 const std::size_t frame = m_width * sizeof(std::int32_t);
313 if (data.empty() || data.size() % frame != 0) {
314 reportError({AcqErrorCode::DeviceIo, std::format("a Neuromag data buffer of {} bytes holds no whole samples of {} channels", data.size(), m_width)});
315 return false;
316 }
317 const std::size_t samples = data.size() / frame;
318 AcqExpected<AcqWriteView> view = AcqError(AcqErrorCode::Timeout);
319 while (!view && view.error() == AcqErrorCode::Timeout) {
320 if (m_stop.load(std::memory_order_acquire)) {
321 return false;
322 }
323 view = m_writer->beginWrite(samples, kPoll); // a gating reader may not have freed the space yet
324 }
325 if (!view) {
326 reportError(view.error());
327 return false;
328 }
329 std::size_t done = 0;
330 for (std::size_t p = 0; p < view->partCount(); ++p) {
331 const AcqWritePart& part = view->part(p);
332 std::memcpy(part.at(0, 0), data.data() + done * frame, part.samples * frame);
333 done += part.samples;
334 }
335 if (m_trigger_channel) {
336 const std::uint64_t first = m_writer->writePosition();
337 for (std::size_t s = 0; s < samples; ++s) {
338 std::int32_t word = 0;
339 std::memcpy(&word, data.data() + s * frame + *m_trigger_channel * sizeof(word), sizeof(word));
340 if (m_trigger.changed(word)) {
341 if (AcqError e = m_writer->pushEvent({first + s, 0, word}); !e.ok()) {
342 reportError(e);
343 return false;
344 }
345 }
346 }
347 }
348 m_writer->commit(samples);
349 return true;
350 }

352 Config m_config;
353 std::unique_ptr<NeuromagTransport> m_transport;
354 std::string m_uri;
355 AcqWriter* m_writer = nullptr;
356 std::size_t m_width = 0;
357 double m_rate = 0.0;
358 std::optional<std::size_t> m_trigger_channel;
359 TriggerWord m_trigger; // owned by the acquisition thread while it runs
360 std::thread m_thread;
361 std::atomic<bool> m_stop{false};
362};
