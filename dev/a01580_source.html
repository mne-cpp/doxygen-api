78 AcqError start() override
79 {
80 // The session restarts at sample 0, as the amplifier's buffer does
81 std::string uri = "synthetic://?sfreq=5000&stim=1&eeg=" + std::to_string(m_setup.channels.size()) +
82 "&block=" + std::to_string(m_setup.points) + "&seed=" + std::to_string(m_simulation.seed) +
83 "&realtime=" + (m_simulation.realtime ? "1" : "0");
84 if (m_simulation.duration > 0.0) {
85 uri += "&duration=" + std::to_string(m_simulation.duration);
86 }
87 auto session = AcqDevice::open(uri);
88 if (!session) {
89 return session.error();
90 }
91 AcqReaderOptions gating;
92 gating.gating = true; // the session waits for this reader instead of overwriting unread samples
93 auto reader = session->stream().reader(AcqReader::ReadMode::Lossless, gating);
94 if (!reader) {
95 return reader.error();
96 }
97 if (AcqError e = session->start(); !e.ok()) {
98 return e;
99 }
100 m_session = std::move(*session);
101 m_reader = std::move(*reader);
102 m_filled = 0;
103 return {};

112 AcqExpected<bool> read(std::span<std::int16_t> block) override
113 {
114 const std::size_t width = m_setup.channels.size() + 1;
115 const double bitsPerVolt = m_simulation.gain / brainAmpVoltsPerBit(m_setup.resolution);
116 while (m_filled < m_setup.points) {
117 auto view = m_reader.acquire(1, std::chrono::nanoseconds(0), m_setup.points - m_filled);
118 if (!view) {
119 if (view.error() == AcqErrorCode::Timeout) {
120 return false;
121 }
122 return view.error();
123 }
124 for (std::size_t p = 0; p < view->partCount(); ++p) {
125 const AcqBlockPart& part = view->part(p);
126 for (std::size_t s = 0; s < part.samples; ++s, ++m_filled) {
127 std::int16_t* frame = m_block.data() + m_filled * width;
128 for (std::size_t c = 0; c < width; ++c) {
129 float v = 0.0F;
130 std::memcpy(&v, part.at(c, s), sizeof(v));
131 frame[c] = c + 1 < width ? quantize(v * bitsPerVolt) : static_cast<std::int16_t>(v);
132 }
133 }
134 }
135 if (AcqError e = m_reader.release(*view); !e.ok()) {
136 return e;
137 }
138 }
139 std::copy(m_block.begin(), m_block.end(), block.begin());
140 m_filled = 0;
141 return true;
142 }

146 {
147 constexpr double kLow = std::numeric_limits<std::int16_t>::min();
148 constexpr double kHigh = std::numeric_limits<std::int16_t>::max();
149 return static_cast<std::int16_t>(std::clamp(std::nearbyint(bits), kLow, kHigh));
150 }
