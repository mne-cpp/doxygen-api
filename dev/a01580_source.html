47std::string normalized(std::string_view family)
48{
49 std::string out;
50 for (const char c : family) {
51 if (c != ' ' || (!out.empty() && out.back() != ' ')) {
52 out += (c >= 'A' && c <= 'Z') ? static_cast<char>(c - 'A' + 'a') : c;
53 }
54 }
55 while (!out.empty() && out.back() == ' ') {
56 out.pop_back();
57 }
58 return out;
59}

119std::shared_ptr<const FontFace> FontLibrary::face(const Font& font) const
120{
121 const std::string family = normalized(font.family);
122 if (family.empty() || isOneOf(family, {"source sans 3", "source sans pro", "sans", "sans-serif", "sans serif", "system-ui", "ui-sans-serif"})) {
123 return sans(font.bold);
124 }
125 if (isOneOf(family, {"monospace", "mono", "ui-monospace", "source code pro"})) {
126 return mono();
127 }
128 if (m_system) {
129 if (auto installed = m_system->find(font.family, font.bold, font.italic)) {
130 return installed;
131 }
132 }
133 return isMonoName(family) ? mono() : sans(font.bold);
134}
