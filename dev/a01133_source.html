102struct RegionMT
103{
104 Eigen::MatrixXd matRoiMT;
105 Eigen::MatrixXd matRoiMTOrig;
106 qint32 nClusters;
107 Eigen::VectorXi idcs;
108 qint32 iLabelIdxIn;
109 QString sDistMeasure;
110
115 RegionMTOut cluster() const
116 {
117 const QString distMeasure = sDistMeasure.isEmpty()
118 ? QStringLiteral("cityblock")
119 : sDistMeasure;
120
121 RegionMTOut out;
122 UTILSLIB::KMeans kMeans(distMeasure, QStringLiteral("sample"), 5);
123 kMeans.calculate(matRoiMT, nClusters,
124 out.roiIdx, out.ctrs, out.sumd, out.D);
125 out.iLabelIdxOut = iLabelIdxIn;
126 return out;
127 }
