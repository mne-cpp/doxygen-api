136static std::optional<FiffCoordTrans> make_voxel_ras_trans(float *r0,
137 float *x_ras,
138 float *y_ras,
139 float *z_ras,
140 float *voxel_size)
141
142{
143 float rot[3][3],move[3];
144 int j,k;
145
146 VEC_COPY_17(move,r0);
147
148 for (j = 0; j < 3; j++) {
149 rot[j][0] = x_ras[j];
150 rot[j][1] = y_ras[j];
151 rot[j][2] = z_ras[j];
152 }
153
154 for (j = 0; j < 3; j++)
155 for (k = 0; k < 3; k++)
156 rot[j][k] = voxel_size[k]*rot[j][k];
157
158 return FiffCoordTrans(FIFFV_MNE_COORD_MRI_VOXEL,FIFFV_COORD_MRI,
159 Eigen::Map<Eigen::Matrix3f>(&rot[0][0]),
160 Eigen::Map<Eigen::Vector3f>(move));
161}

237double MNESurfaceOrVolume::solid_angle(const Eigen::Vector3f& from, const MNETriangle& tri) /* ...to this triangle */
238/*
239 * Compute the solid angle according to van Oosterom's
240 * formula
241 */
242{
243 double v1[3],v2[3],v3[3];
244 double l1,l2,l3,s,triple;
245 double cross[3];
246
247 VEC_DIFF_17 (from,tri.r1,v1);
248 VEC_DIFF_17 (from,tri.r2,v2);
249 VEC_DIFF_17 (from,tri.r3,v3);
250
251 CROSS_PRODUCT_17(v1,v2,cross);
252 triple = VEC_DOT_17(cross,v3);
253
254 l1 = VEC_LEN_17(v1);
255 l2 = VEC_LEN_17(v2);
256 l3 = VEC_LEN_17(v3);
257 s = (l1*l2*l3+VEC_DOT_17(v1,v2)*l3+VEC_DOT_17(v1,v3)*l2+VEC_DOT_17(v2,v3)*l1);
258
259 return (2.0*atan2(triple,s));
260}

265/*
266 * Add the triangle data structures
267 */
268{
269 int k;
270 MNETriangle* tri;
271
272 if (type != MNE_SOURCE_SPACE_SURFACE)

275 tris.clear();
276 use_tris.clear();
277 /*
278 * Add information for the complete triangulation
279 */
280 if (itris.rows() > 0 && ntri > 0) {
281 tris.resize(ntri);
282 tot_area = 0.0;
283 for (k = 0, tri = tris.data(); k < ntri; k++, tri++) {
284 tri->vert = &itris(k,0);
285 tri->r1 = rr.row(tri->vert[0]).transpose();
286 tri->r2 = rr.row(tri->vert[1]).transpose();
287 tri->r3 = rr.row(tri->vert[2]).transpose();
288 tri->compute_data();
289 tot_area += tri->area;
290 }
291#ifdef TRIANGLE_SIZE_WARNING
292 for (k = 0, tri = tris.data(); k < ntri; k++, tri++)
293 if (tri->area < 1e-5*tot_area/ntri)
294 printf("Warning: Triangle area is only %g um^2 (%.5f %% of expected average)\n",
295 1e12*tri->area,100*ntri*tri->area/tot_area);
296#endif
297 }
298#ifdef DEBUG
299 printf("\ttotal area = %-.1f cm^2\n",1e4*tot_area);
300#endif
301 /*
302 * Add information for the selected subset if applicable
303 */
304 if (use_itris.rows() > 0 && nuse_tri > 0) {
305 use_tris.resize(nuse_tri);
306 for (k = 0, tri = use_tris.data(); k < nuse_tri; k++, tri++) {
307 tri->vert = &use_itris(k,0);
308 tri->r1 = rr.row(tri->vert[0]).transpose();
309 tri->r2 = rr.row(tri->vert[1]).transpose();
310 tri->r3 = rr.row(tri->vert[2]).transpose();

319void MNESurfaceOrVolume::compute_cm(const MNESurfaceOrVolume::PointsT& rr, int np, float (&cm)[3])
320/*
321 * Compute the center of mass of a set of points
322 */
323{
324 int q;
325 cm[0] = cm[1] = cm[2] = 0.0;
326 for (q = 0; q < np; q++) {
327 cm[0] += rr(q,0);
328 cm[1] += rr(q,1);
329 cm[2] += rr(q,2);
330 }
331 if (np > 0) {
332 cm[0] = cm[0]/np;
333 cm[1] = cm[1]/np;
334 cm[2] = cm[2]/np;
335 }
336 return;
337}

355 float diff[3];

358 if (neighbor_vert.empty() || nneighbor_vert.size() == 0)

361 vert_dist.clear();

363 printf("\tDistances between neighboring vertices...");
364 for (k = 0, ndist = 0; k < np; k++) {
365 nneigh = nneighbor_vert[k];
366 vert_dist[k] = Eigen::VectorXf(nneigh);
367 const Eigen::VectorXi& neigh = neighbor_vert[k];
368 for (p = 0; p < nneigh; p++) {
369 if (neigh[p] >= 0) {
370 VEC_DIFF_17(&rr(k,0),&rr(neigh[p],0),diff);
371 vert_dist[k][p] = VEC_LEN_17(diff);
372 }
373 else
374 vert_dist[k][p] = -1.0;
375 ndist++;
376 }
377 }
378 printf("[%d distances done]\n",ndist);
379 return;

384int MNESurfaceOrVolume::add_vertex_normals()
385{
386 int k,c,p;
387 int *ii;
388 float w,size;
389 MNETriangle* tri;
390
391 if (type != MNE_SOURCE_SPACE_SURFACE)
392 return OK;
393 /*
394 * Reallocate the stuff and initialize
395 */
396 nn = MNESurfaceOrVolume::NormalsT::Zero(np,3);
397
398 for (k = 0; k < np; k++) {
399 nn(k,X_17) = nn(k,Y_17) = nn(k,Z_17) = 0.0;
400 }
401 /*
402 * One pass through the triangles will do it
403 */
404 add_triangle_data();
405 for (p = 0, tri = tris.data(); p < ntri; p++, tri++) {
406 ii = tri->vert;
407 w = 1.0; /* This should be related to the triangle size */
408 /*
409 * Then the vertex normals
410 */
411 for (k = 0; k < 3; k++)
412 for (c = 0; c < 3; c++)
413 nn(ii[k],c) += w*tri->nn[c];
414 }
415 for (k = 0; k < np; k++) {
416 size = VEC_LEN_17(&nn(k,0));
417 if (size > 0.0)
418 for (c = 0; c < 3; c++)
419 nn(k,c) = nn(k,c)/size;
420 }
421 compute_surface_cm();
422 return OK;
423}

427int MNESurfaceOrVolume::add_geometry_info(bool do_normals, bool check_too_many_neighbors)
428/*
429 * Add vertex normals and neighbourhood information
430 */
431{
432 int k,c,p,q;
433 int vert;
434 int *ii;
435 int nneighbors;
436 float w,size;
437 int found;
438 int nfix_distinct,nfix_no_neighbors,nfix_defect;
439 MNETriangle* tri;
440
441 if (type == MNE_SOURCE_SPACE_VOLUME) {
442 calculate_vertex_distances();
443 return OK;
444 }
445 if (type != MNE_SOURCE_SPACE_SURFACE)
446 return OK;
447 /*
448 * Reallocate the stuff and initialize
449 */
450 if (do_normals) {
451 nn = MNESurfaceOrVolume::NormalsT::Zero(np,3);
452 }
453 neighbor_tri.clear();
454 neighbor_tri.resize(np);
455 nneighbor_tri = Eigen::VectorXi::Zero(np);
456
457 for (k = 0; k < np; k++) {
458 if (do_normals)
459 nn(k,X_17) = nn(k,Y_17) = nn(k,Z_17) = 0.0;
460 }
461 /*
462 * One pass through the triangles will do it
463 */
464 add_triangle_data();
465 for (p = 0, tri = tris.data(); p < ntri; p++, tri++)
466 if (tri->area == 0)
467 printf("\tWarning : zero size triangle # %d\n",p);
468 printf("\tTriangle ");
469 if (do_normals)
470 printf("and vertex ");
471 printf("normals and neighboring triangles...");
472 for (p = 0, tri = tris.data(); p < ntri; p++, tri++) {
473 ii = tri->vert;
474 w = 1.0; /* This should be related to the triangle size */
475 for (k = 0; k < 3; k++) {
476 /*
477 * Then the vertex normals
478 */
479 if (do_normals)
480 for (c = 0; c < 3; c++)
481 nn(ii[k],c) += w*tri->nn[c];
482 /*
483 * Add to the list of neighbors
484 */
485 neighbor_tri[ii[k]].conservativeResize(nneighbor_tri[ii[k]]+1);
486 neighbor_tri[ii[k]][nneighbor_tri[ii[k]]] = p;
487 nneighbor_tri[ii[k]]++;
488 }
489 }
490 nfix_no_neighbors = 0;
491 nfix_defect = 0;
492 for (k = 0; k < np; k++) {
493 if (nneighbor_tri[k] <= 0) {
494#ifdef STRICT_ERROR
495 err_printf_set_error("Vertex %d does not have any neighboring triangles!",k);
496 return FAIL;
497#else
498#ifdef REPORT_WARNINGS
499 printf("Warning: Vertex %d does not have any neighboring triangles!\n",k);
500#endif
501#endif
502 nfix_no_neighbors++;
503 }
504 else if (nneighbor_tri[k] < 3) {
505#ifdef REPORT_WARNINGS
506 printf("\n\tTopological defect: Vertex %d has only %d neighboring triangle%s Vertex omitted.\n\t",
507 k,nneighbor_tri[k],nneighbor_tri[k] > 1 ? "s." : ".");
508#endif
509 nfix_defect++;
510 nneighbor_tri[k] = 0;
511 neighbor_tri[k].resize(0);
512 }
513 }
514 /*
515 * Scale the vertex normals to unit length
516 */
517 for (k = 0; k < np; k++)
518 if (nneighbor_tri[k] > 0) {
519 size = VEC_LEN_17(&nn(k,0));
520 if (size > 0.0)
521 for (c = 0; c < 3; c++)
522 nn(k,c) = nn(k,c)/size;
523 }
524 printf("[done]\n");
525 /*
526 * Determine the neighboring vertices
527 */
528 printf("\tVertex neighbors...");
529 neighbor_vert.clear();
530 neighbor_vert.resize(np);
531 nneighbor_vert = VectorXi::Zero(np);
532 /*
533 * We know the number of neighbors beforehand
534 */
535 for (k = 0; k < np; k++) {
536 if (nneighbor_tri[k] > 0) {
537 neighbor_vert[k] = VectorXi(nneighbor_tri[k]);
538 nneighbor_vert[k] = nneighbor_tri[k];
539 }
540 else {
541 nneighbor_vert[k] = 0;
542 }
543 }
544 nfix_distinct = 0;
545 for (k = 0; k < np; k++) {
546 Eigen::VectorXi& neighbors = neighbor_vert[k];
547 nneighbors = 0;
548 for (p = 0; p < nneighbor_tri[k]; p++) {
549 /*
550 * Fit in the other vertices of the neighboring triangle
551 */
552 for (c = 0; c < 3; c++) {
553 vert = tris[neighbor_tri[k][p]].vert[c];
554 if (vert != k) {
555 for (q = 0, found = false; q < nneighbors; q++) {
556 if (neighbors[q] == vert) {
557 found = true;
558 break;
559 }
560 }
561 if (!found) {
562 if (nneighbors < nneighbor_vert[k])
563 neighbors[nneighbors++] = vert;
564 else {
565 if (check_too_many_neighbors) {
566 printf("Too many neighbors for vertex %d.",k);
567 return FAIL;
568 }
569 else
570 printf("\tWarning: Too many neighbors for vertex %d\n",k);
571 }
572 }
573 }
574 }
575 }
576 if (nneighbors != nneighbor_vert[k]) {
577#ifdef REPORT_WARNINGS
578 printf("\n\tIncorrect number of distinct neighbors for vertex %d (%d instead of %d) [fixed].",
579 k,nneighbors,nneighbor_vert[k]);
580#endif
581 nfix_distinct++;
582 nneighbor_vert[k] = nneighbors;
583 }
584 }
585 printf("[done]\n");
586 /*
587 * Distance calculation follows
588 */
589 calculate_vertex_distances();
590 compute_surface_cm();
591 /*
592 * Summarize the defects
593 */
594 if (nfix_defect > 0)
595 printf("\tWarning: %d topological defects were fixed.\n",nfix_defect);
596 if (nfix_distinct > 0)
597 printf("\tWarning: %d vertices had incorrect number of distinct neighbors (fixed).\n",nfix_distinct);
598 if (nfix_no_neighbors > 0)
599 printf("\tWarning: %d vertices did not have any neighboring triangles (fixed)\n",nfix_no_neighbors);
600#ifdef DEBUG
601 for (k = 0; k < np; k++) {
602 if (nneighbor_vert[k] <= 0)
603 printf("No neighbors for vertex %d\n",k);
604 if (nneighbor_tri[k] <= 0)
605 printf("No neighbor tris for vertex %d\n",k);
606 }
607#endif
608 return OK;
609}
