48SoftwareShader channeldata_shader()
49{
50 SoftwareShader shader;
51 shader.varying_count = 1;
52 shader.vertex = [](const SoftwareAttributes& in, const SoftwareBindings& bindings, SoftwareVaryings& out) {
53 const auto u = bindings.uniforms<TraceUniforms>(0);
54 const float sample = u.first_sample + in[0].x();
55 const float xPixel = (sample - u.scroll_sample) / std::max(u.samples_per_pixel, 1e-6f);
56 const float x = (xPixel / u.view_width) * 2.f - 1.f;
57 const float norm = u.amplitude_max > 0.f ? std::clamp(in[0].y() / u.amplitude_max, -2.f, 2.f) : 0.f;
58 out[0] = norm;
59 return Eigen::Vector4f(x, u.channel_y_center + norm * (u.channel_y_range * 0.45f), 0.f, 1.f);
60 };
61 shader.fragment = [](const SoftwareVaryings& in, const SoftwareFragment&, const SoftwareBindings& bindings) -> std::optional<Eigen::Vector4f> {
62 const auto u = bindings.uniforms<TraceUniforms>(0);
63 if (u.show_clipping > 0.5f && std::abs(in[0]) >= 0.95f) {
64 return Eigen::Vector4f(1.f, 0.f, 0.f, 1.f);
65 }
66 return Eigen::Vector4f(u.color[0], u.color[1], u.color[2], u.color[3]);
67 };
68 return shader;
69}

72SoftwareShader overlay_shader()
73{
74 SoftwareShader shader;
75 shader.varying_count = 2;
76 shader.vertex = [](const SoftwareAttributes& in, const SoftwareBindings&, SoftwareVaryings& out) {
77 out[0] = in[1].x();
78 out[1] = in[1].y();
79 return Eigen::Vector4f(in[0].x(), in[0].y(), 0.f, 1.f);
80 };
81 shader.fragment = [](const SoftwareVaryings& in, const SoftwareFragment&, const SoftwareBindings& bindings) -> std::optional<Eigen::Vector4f> {
82 const auto u = bindings.uniforms<TraceOverlayUniforms>(2);
83 const float screenSample = u.scroll_sample + in[0] * u.view_width * u.samples_per_pixel;
84 const float overlayU = std::clamp((screenSample - u.overlay_first_sample) / std::max(u.overlay_total_samples, 1.f), 0.f, 1.f);
85 Eigen::Vector4f color = bindings.sample(1, Eigen::Vector2f(overlayU, in[1]));
86 if (u.grid_enabled > 0.5f && u.sfreq > 0.f) {
87 const float second = std::floor((screenSample - u.first_file_sample) / u.sfreq);
88 if (std::fmod(std::abs(second), 2.f) > 0.5f) {
89 color.head<3>() *= 1.f - 0.08f;
90 color.w() = std::max(color.w(), 0.08f);
91 }
92 }
93 return color;
94 };
95 return shader;
96}

136Vec4 surface_vertex(const SoftwareAttributes& in, const SoftwareBindings& bindings, SoftwareVaryings& out)
137{
138 const auto u = bindings.uniforms<BrainSurfaceUniforms>(0);
139 const Vec3 position = in[0].head<3>();
140 const Vec4& annotation = in[3];
141 const Vec4 selected(u.selected_label[0], u.selected_label[1], u.selected_label[2], u.selected_label[3]);
142 set_varying3(out, kWorldPos, position);
143 set_varying3(out, kNormal, in[1].head<3>().normalized());
144 set_varying3(out, kColor, in[2].head<3>());
145 set_varying3(out, kAnnotation, annotation.head<3>());
146 out[kCurvature] = in[2].w();
147 out[kHighlight] = u.label_active > 0.5f && ((annotation - selected).cwiseAbs().array() < 0.5f / 255.f).all() ? 1.f : 0.f;
148 set_varying3(out, kViewDir, (Vec3(u.camera_pos[0], u.camera_pos[1], u.camera_pos[2]) - position).normalized());
149 return Eigen::Map<const Eigen::Matrix4f>(u.mvp) * Vec4(position.x(), position.y(), position.z(), 1.f);
150}

165std::optional<Vec4> standard_fragment(const SoftwareVaryings& in, const BrainSurfaceUniforms& u)
166{
167 const Vec3 n = varying3(in, kNormal).normalized();
168 const Vec3 l = Vec3(u.light_dir[0], u.light_dir[1], u.light_dir[2]).normalized();
169 const Vec3 v = (Vec3(u.camera_pos[0], u.camera_pos[1], u.camera_pos[2]) - varying3(in, kWorldPos)).normalized();
170 const Vec3 h = (l + v).normalized();
171 const float diff = std::max(n.dot(l), 0.f);
172 const float spec = std::pow(std::max(n.dot(h), 0.f), 32.f) * 0.3f;
173
174 const Vec3 annotation = varying3(in, kAnnotation);
175 Vec3 base;
176 if (u.overlay_mode < 0.5f) {
177 base = Vec3::Constant(in[kCurvature]);
178 } else if (u.overlay_mode < 1.5f) {
179 base = annotation.squaredNorm() > 0.001f ? annotation : Vec3(1.f, 0.97f, 0.94f);
180 } else {
181 base = varying3(in, kColor);
182 }
183 if (in[kHighlight] > 0.5f) {
184 base = Vec3(1.f, 0.784f, 0.235f);
185 }
186 base = base.cwiseProduct(Vec3(1.f, 0.97f, 0.94f));
187 const float fill = std::max(n.dot(Vec3(0.f, -0.5f, 0.3f).normalized()), 0.f) * 0.15f;
188 Vec3 color = base * (0.35f + diff * 0.65f + fill) + Vec3(0.95f, 0.95f, 1.f) * spec;
189 color = color.cwiseMax(0.f).cwiseMin(1.f);
190 if (u.is_selected > 0.5f) {
191 const float fresnel = std::pow(1.f - std::clamp(n.dot(v), 0.f, 1.f), 2.5f);
192 const Vec3 glow(1.f, 0.85f, 0.3f);
193 color = mix3(color, glow, 0.25f) + glow * fresnel * 0.6f;
194 }
195 return Vec4(color.x(), color.y(), color.z(), 1.f);
196}

199std::optional<Vec4> holographic_fragment(const SoftwareVaryings& in, const BrainSurfaceUniforms& u)
200{
201 const Vec3 n = varying3(in, kNormal).normalized();
202 const Vec3 v = varying3(in, kViewDir).normalized();
203 const Vec3 annotation = varying3(in, kAnnotation);
204 Vec3 color;
205 if (u.overlay_mode < 0.5f) {
206 color = Vec3::Constant(in[kCurvature]);
207 } else if (u.overlay_mode < 1.5f) {
208 color = annotation.squaredNorm() > 0.001f ? annotation : Vec3::Ones();
209 } else {
210 color = varying3(in, kColor);
211 }
212 if (in[kHighlight] > 0.5f) {
213 color = Vec3(1.f, 0.784f, 0.235f);
214 }
215 const float gray = color.dot(Vec3(0.299f, 0.587f, 0.114f));
216 const float isData = smoothstep(0.0025f, 0.04f, (color - Vec3::Constant(gray)).squaredNorm());
217 const float nDotV = std::abs(n.dot(v));
218 const float fresnel = std::pow(1.f - nDotV, 2.5f);
219 const float rim = std::pow(1.f - nDotV, 3.f);
220 const Vec3 shell = Vec3(0.f, 0.4f, 0.8f) * (fresnel * 1.5f + rim * 0.5f);
221 const Vec3 data = color * (1.f + fresnel * 2.f);
222 Vec3 rgb = mix3(shell, data, isData);
223 float alpha = (0.1f + 0.7f * fresnel) + isData * ((0.4f + 0.5f * fresnel) - (0.1f + 0.7f * fresnel));
224 if (u.is_selected > 0.5f) {
225 const Vec3 white(1.f, 1.1f, 1.2f);
226 rgb += white * 0.2f + white * std::pow(1.f - nDotV, 3.f);
227 alpha = std::clamp(alpha + 0.1f + std::pow(1.f - nDotV, 3.f) * 0.5f, 0.f, 1.f);
228 }
229 rgb *= alpha;
230 return Vec4(rgb.x(), rgb.y(), rgb.z(), alpha);
231}

234std::optional<Vec4> anatomical_fragment(const SoftwareVaryings& in, const BrainSurfaceUniforms& u)
235{
236 const Vec3 n = varying3(in, kNormal).normalized();
237 const Vec3 v = varying3(in, kViewDir).normalized();
238 const Vec3 l = Vec3(u.light_dir[0], u.light_dir[1], u.light_dir[2]).normalized();
239 const Vec3 h = (l + v).normalized();
240 const auto tissue = std::lround(u.tissue_type);
241 const float curvature = in[kCurvature];
242 const float selection = std::max(u.is_selected, in[kHighlight]);
243
244 Vec3 base;
245 float roughness = 0.5f;
246 float specIntensity = 0.3f;
247 if (tissue == 1) {
248 base = mix3(Vec3(0.85f, 0.75f, 0.72f), Vec3(0.55f, 0.40f, 0.38f), std::clamp(curvature * 2.f + 0.5f, 0.f, 1.f));
249 roughness = 0.7f;
250 specIntensity = 0.15f;
251 } else if (tissue == 2) {
252 base = Vec3(0.92f, 0.78f, 0.68f) + Vec3(1.f, 0.4f, 0.3f) * std::max(n.dot(l) * 0.5f + 0.5f, 0.f) * 0.15f;
253 roughness = 0.4f;
254 specIntensity = 0.25f;
255 } else if (tissue == 3) {
256 base = Vec3(0.95f, 0.90f, 0.82f);
257 roughness = 0.6f;
258 specIntensity = 0.2f;
259 } else if (tissue == 4) {
260 base = Vec3(0.88f, 0.82f, 0.75f);
261 roughness = 0.65f;
262 specIntensity = 0.15f;
263 } else {
264 base = varying3(in, kColor);
265 }
266 const Vec3 annotation = varying3(in, kAnnotation);
267 if (u.overlay_mode > 0.5f && u.overlay_mode < 1.5f) {
268 if (annotation.squaredNorm() > 0.001f) {
269 base = annotation;
270 }
271 } else if (u.overlay_mode > 2.5f) {
272 base = varying3(in, kColor);
273 }
274 if (in[kHighlight] > 0.5f && ((u.overlay_mode > 0.5f && u.overlay_mode < 1.5f) || u.overlay_mode > 2.5f)) {
275 base = Vec3(1.f, 0.784f, 0.235f);
276 }
277
278 const float ao = 0.7f + 0.3f * std::clamp(1.f - curvature * 0.5f, 0.f, 1.f);
279 const float nDotH = std::max(n.dot(h), 0.f);
280 const float specPower = 8.f + 56.f * (1.f - roughness);
281 const float fill = std::max(n.dot(Vec3(0.2f, -0.6f, 0.3f).normalized()), 0.f) * 0.12f;
282 Vec3 color = base * 0.3f * ao + base * std::max(n.dot(l), 0.f) * 0.6f + Vec3(0.95f, 0.95f, 1.f) * std::pow(nDotH, specPower) * specIntensity + base * fill;
283 if (selection > 0.1f) {
284 const float fresnel = std::pow(1.f - std::clamp(n.dot(v), 0.f, 1.f), 2.5f);
285 color = mix3(color, Vec3(0.95f, 0.55f, 0.50f) * 0.9f, 0.25f * selection);
286 color += Vec3(0.08f, 0.03f, 0.02f) * selection;
287 color += Vec3(0.90f, 0.40f, 0.35f) * fresnel * 0.3f * selection;
288 color += Vec3(0.95f, 0.90f, 0.90f) * std::pow(nDotH, 32.f) * 0.15f * selection;
289 }
290 color = color.cwiseMax(0.f).cwiseMin(1.f);
291 return Vec4(color.x(), color.y(), color.z(), 1.f);
292}

312Vec4 instanced_vertex(const SoftwareAttributes& in, const SoftwareBindings& bindings, SoftwareVaryings& out)
313{
314 const auto u = bindings.uniforms<BrainInstanceUniforms>(0);
315 Eigen::Matrix4f model;
316 model << in[2], in[3], in[4], in[5];
317 const Vec4 world = model * Vec4(in[0].x(), in[0].y(), in[0].z(), 1.f);
318 set_varying3(out, 0, world.head<3>());
319 set_varying3(out, 3, model.topLeftCorner<3, 3>() * in[1].head<3>());
320 for (std::size_t i = 0; i < 4; ++i) {
321 out[6 + i] = in[6][static_cast<Eigen::Index>(i)];
322 }
323 out[10] = in[7].x();
324 return Vec4(Eigen::Map<const Eigen::Matrix4f>(u.mvp) * world);
325}

328std::optional<Vec4> dipole_fragment(const SoftwareVaryings& in, const SoftwareFragment&, const SoftwareBindings& bindings)
329{
330 const auto u = bindings.uniforms<BrainInstanceUniforms>(0);
331 const Vec3 n = varying3(in, 3).normalized();
332 const Vec3 v = (Vec3(u.camera_pos[0], u.camera_pos[1], u.camera_pos[2]) - varying3(in, 0)).normalized();

334 const float fresnel = std::pow(1.f - nDotV, 2.5f);

337 if (in[10] > 0.5f) {
338 const float rim = std::pow(1.f - nDotV, 3.f);
339 color += Vec3::Constant(0.15f + rim);
340 alpha = std::clamp(alpha + 0.1f + rim * 0.5f, 0.f, 1.f);
341 }
342 return Vec4(color.x(), color.y(), color.z(), alpha);

347{
348 const auto u = bindings.uniforms<BrainInstanceUniforms>(0);
349 const Vec3 base = in[10] > 0.5f ? Vec3(1.f, 0.78f, 0.24f) : Vec3(in[6], in[7], in[8]);
