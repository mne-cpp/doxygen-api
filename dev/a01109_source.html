83void InvHpiDataUpdater::updateChannels(const std::shared_ptr<FiffInfo>& pFiffInfo)
84{
85 // Get the indices of inner layer channels and exclude bad channels and create channellist
86 const int iNumCh = pFiffInfo->nchan;
87 m_lChannels.clear();
88 m_vecInnerind.clear();
89 for (int i = 0; i < iNumCh; ++i) {
90 if (pFiffInfo->chs[i].chpos.coil_type == FIFFV_COIL_BABY_MAG ||
91 pFiffInfo->chs[i].chpos.coil_type == FIFFV_COIL_VV_PLANAR_T1 ||
92 pFiffInfo->chs[i].chpos.coil_type == FIFFV_COIL_VV_PLANAR_T2 ||
93 pFiffInfo->chs[i].chpos.coil_type == FIFFV_COIL_VV_PLANAR_T3) {
94 // Check if the sensor is bad, if not append to innerind
95
96 if (!(mne::has_channel(pFiffInfo->bads, pFiffInfo->ch_names.at(i)))) {
97 m_vecInnerind.push_back(i);
98 m_lChannels.push_back(pFiffInfo->chs[i]);
99 }
100 }
101 }
102}

114void InvHpiDataUpdater::updateHpiDigitizer(const FiffDigPointSet& dig)
115{
116 const FiffDigPointSet coils = dig.pick_types({FIFFV_POINT_HPI});
117 if (coils.empty()) {
118 mne::Log::warning("InvHpiFit::updateHpiDigitizer - No HPI coils digitized. Returning.");
119 return;
120 }
121 m_matHpiDigitizer = MatrixXd(static_cast<Eigen::Index>(coils.size()), 3);
122 for (std::size_t i = 0; i < coils.size(); ++i) {
123 m_matHpiDigitizer.row(static_cast<Eigen::Index>(i)) << coils[i].r[0], coils[i].r[1], coils[i].r[2];
124 }
125}

174void InvHpiDataUpdater::prepareProjectors(const Eigen::MatrixXd& matProjectors)
175{
176 // check if m_vecInnerInd is alreadz initialized
177 if (m_vecInnerind.empty()) {
178 mne::Log::warning("InvHpiFit::updateProjectors - No channels. Returning.");
179 return;
180 }
181
182 //Create new projector based on the excluded channels, first exclude the rows then the columns
183 MatrixXd matProjectorsRows(std::ssize(m_vecInnerind), matProjectors.cols());
184 MatrixXd matProjectorsInnerind(std::ssize(m_vecInnerind), std::ssize(m_vecInnerind));
185
186 for (int i = 0; i < matProjectorsRows.rows(); ++i) {
187 matProjectorsRows.row(i) = matProjectors.row(m_vecInnerind[static_cast<std::size_t>(i)]);
188 }
189
190 for (int i = 0; i < matProjectorsInnerind.cols(); ++i) {
191 matProjectorsInnerind.col(i) = matProjectorsRows.col(m_vecInnerind[static_cast<std::size_t>(i)]);
192 }
193 m_matProjectors = matProjectorsInnerind;
194 return;
195}
