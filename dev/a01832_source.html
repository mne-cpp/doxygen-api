157void CrossCorrelation::compute(ConnectivitySettings::IntermediateTrialData& inputData,
158 MatrixXd& matDist,
159 QMutex& mutex,
160 int iNfft,
161 const QPair<MatrixXd, VectorXd>& tapers)
162{
163// QElapsedTimer timer;
164// qint64 iTime = 0;
165// timer.start();
166
167 // Calculate tapered spectra if not available already
168 RowVectorXd vecInputFFT, rowData;
169 RowVectorXcd vecResultFreq;
170
171 FFT<double> fft;
172 fft.SetFlag(fft.HalfSpectrum);
173
174 int i, j;
175 int iNRows = inputData.matData.rows();
176
177 // Calculate tapered spectra if not available already
178 // This code was copied and changed modified Utils/Spectra since we do not want to call the function due to time loss.
179 if(inputData.vecTapSpectra.isEmpty()) {
180 int iNFreqs = int(floor(iNfft / 2.0)) + 1;
181 MatrixXcd matTapSpectrum(tapers.first.rows(), iNFreqs);
182
183 for (i = 0; i < iNRows; ++i) {
184 // Substract mean
185 rowData.array() = inputData.matData.row(i).array() - inputData.matData.row(i).mean();
186
187 // Calculate tapered spectra
188 for(j = 0; j < tapers.first.rows(); j++) {
189 // Zero padd if necessary. The zero padding in Eigen's FFT is only working for column vectors.
190 if (rowData.cols() < iNfft) {
191 vecInputFFT.setZero(iNfft);
192 vecInputFFT.block(0,0,1,rowData.cols()) = rowData.cwiseProduct(tapers.first.row(j));;
193 } else {
194 vecInputFFT = rowData.cwiseProduct(tapers.first.row(j));
195 }
196
197 // FFT for freq domain returning the half spectrum and multiply taper weights
198 fft.fwd(vecResultFreq, vecInputFFT, iNfft);
199 matTapSpectrum.row(j) = vecResultFreq * tapers.second(j);
200 }
201
202 inputData.vecTapSpectra.append(matTapSpectrum);
203 }
204 }
205
206// iTime = timer.elapsed();
207// qDebug() << QThread::currentThreadId() << "CrossCorrelation::compute timer - Tapered spectra:" << iTime;
208// timer.restart();
209
210 // Perform multiplication and transform back to time domain to find max XCOR coefficient
211 // Note that the result in time domain is mirrored around the center of the data (compared to Matlab)
212 MatrixXd matDistTrial = MatrixXd::Zero(iNRows, iNRows);
213 RowVectorXcd vecResultXCor;
214 int idx = 0;
215 double denom = tapers.second.sum();
216
217 for(i = 0; i < inputData.vecTapSpectra.size(); ++i) {
218 vecResultFreq = inputData.vecTapSpectra.at(i).colwise().sum() / denom;
219
220 for(j = i; j < inputData.vecTapSpectra.size(); ++j) {
221 vecResultXCor = vecResultFreq.cwiseProduct(inputData.vecTapSpectra.at(j).colwise().sum() / denom);
222
223 fft.inv(vecInputFFT, vecResultXCor, iNfft);
224
225 vecInputFFT.maxCoeff(&idx);
226
227 matDistTrial(i,j) = vecInputFFT(idx);
228 }
229 }
230
231// iTime = timer.elapsed();
232// qDebug() << QThread::currentThreadId() << "CrossCorrelation::compute timer - Multiplication and inv FFT:" << iTime;
233// timer.restart();
234
235 // Sum up weights
236 mutex.lock();
237
238 if(matDist.rows() != matDistTrial.rows() || matDist.cols() != matDistTrial.cols()) {
239 matDist.resize(matDistTrial.rows(), matDistTrial.cols());
240 matDist.setZero();
241 }
242
243 matDist += matDistTrial;
244
245 mutex.unlock();
246
247// iTime = timer.elapsed();
248// qDebug() << QThread::currentThreadId() << "CrossCorrelation::compute timer - Summing up matDist:" << iTime;
249// timer.restart();
250
251 if(!m_bStorageModeIsActive) {
252 inputData.vecTapSpectra.clear();
253 }
254}
