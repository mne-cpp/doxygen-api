87struct RegionData
88{
89 Eigen::MatrixXd matRoiG;
90 Eigen::MatrixXd matRoiGWhitened;
91 bool bUseWhitened;
92
93 Eigen::MatrixXd matRoiGOrig;
94
95 qint32 nClusters;
96
97 Eigen::VectorXi idcs;
98 qint32 iLabelIdxIn;
99 QString sDistMeasure;
100
101 RegionDataOut cluster() const
102 {
103 QString t_sDistMeasure;
104 if(sDistMeasure.isEmpty())
105 t_sDistMeasure = QString("cityblock");
106 else
107 t_sDistMeasure = sDistMeasure;
108
109 // Kmeans Reduction
110 RegionDataOut p_RegionDataOut;
111
112 UTILSLIB::KMeans t_kMeans(t_sDistMeasure, QString("sample"), 5);
113
114 if(bUseWhitened)
115 {
116 t_kMeans.calculate(this->matRoiGWhitened, this->nClusters, p_RegionDataOut.roiIdx, p_RegionDataOut.ctrs, p_RegionDataOut.sumd, p_RegionDataOut.D);
117
118 Eigen::MatrixXd newCtrs = Eigen::MatrixXd::Zero(p_RegionDataOut.ctrs.rows(), p_RegionDataOut.ctrs.cols());
119 for(qint32 c = 0; c < p_RegionDataOut.ctrs.rows(); ++c)
120 {
121 qint32 num = 0;
122
123 for(qint32 idx = 0; idx < p_RegionDataOut.roiIdx.size(); ++idx)
124 {
125 if(c == p_RegionDataOut.roiIdx[idx])
126 {
127 newCtrs.row(c) += this->matRoiG.row(idx); //just take whitened to get indeces calculate centroids using the original matrix
128 ++num;
129 }
130 }
131
132 if(num > 0)
133 newCtrs.row(c) /= num;
134 }
135 p_RegionDataOut.ctrs = newCtrs; //Replace whitened with original
136 }
137 else
138 t_kMeans.calculate(this->matRoiG, this->nClusters, p_RegionDataOut.roiIdx, p_RegionDataOut.ctrs, p_RegionDataOut.sumd, p_RegionDataOut.D);
139
140 p_RegionDataOut.iLabelIdxOut = this->iLabelIdxIn;
141
142 return p_RegionDataOut;
143 }
144};
