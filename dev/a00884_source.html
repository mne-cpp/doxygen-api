158void BrainTreeModel::addSourceSpace(const MNELIB::MNESourceSpaces &srcSpace)
159{
160 QStandardItem* parentItem = new QStandardItem("Source Space");
161 parentItem->setCheckable(true);
162 parentItem->setCheckState(Qt::Checked);
163
164 // Color: pinkish-red matching disp3D's source space rendering
165 QColor srcColor(212, 28, 92);
166
167 for (int h = 0; h < srcSpace.size(); ++h) {
168 const auto &hemi = srcSpace[h];
169 QString hemiLabel = (h == 0) ? "LH" : "RH";
170
171 // Collect all source point positions for this hemisphere
172 QVector<QVector3D> positions;
173 positions.reserve(hemi.vertno.size());
174 for (int i = 0; i < hemi.vertno.size(); ++i) {
175 int vIdx = hemi.vertno(i);
176 if (vIdx < 0 || vIdx >= hemi.rr.rows()) continue;
177 positions.append(QVector3D(hemi.rr(vIdx, 0), hemi.rr(vIdx, 1), hemi.rr(vIdx, 2)));
178 }
179
180 // One item per hemisphere with all positions batched (0.00075 = 0.75mm radius, same as disp3D)
181 parentItem->appendRow(new SourceSpaceTreeItem(hemiLabel, positions, srcColor, 0.00075f));
182 qDebug() << "BrainTreeModel: Source space" << hemiLabel
183 << "- points:" << positions.size()
184 << "coord_frame:" << hemi.coord_frame;
185 if (!positions.isEmpty()) {
186 qDebug() << " First point:" << positions.first()
187 << " Last point:" << positions.last();
188 }
189 }
190
191 this->appendRow(parentItem);
192}
