257// Compute sub-sampling stride
258int stride(int total, int max)
259{
260 if (max <= 0 || max >= total)
261 return 1;
262 return (total + max - 1) / max; // ceiling, so at most max entries are kept
263}

274std::vector<InvToken> tokenize(const InvSourceEstimate& estimate, const InvTokenizeOptions& options)
275{
276 std::vector<InvToken> tokens;
277
278 // Rough capacity estimate to avoid excessive reallocations
279 int nSrc = static_cast<int>(estimate.data.rows());
280 int nTimes = static_cast<int>(estimate.data.cols());
281 int srcStride = stride(nSrc, options.maxSources);
282 int timeStride = stride(nTimes, options.maxTimePoints);
283 int effSrc = (nSrc + srcStride - 1) / srcStride;
284 int effTime = (nTimes + timeStride - 1) / timeStride;
285
286 size_t est = 10; // structural overhead
287 if (options.includeGridData && estimate.has_grid_data())
288 est += static_cast<size_t>(effSrc) * (1 + effTime) + effSrc + 6;
289 est += estimate.focalDipoles.size() * 16;
290 est += estimate.couplings.size() * 20;
291 est += estimate.connectivity.size() * 10;
292 if (estimate.has_positions())
293 est += static_cast<size_t>(estimate.positions.rows()) * 3 + 2;
294 tokens.reserve(est);
295
296 // --- BOS ---
297 tokens.emplace_back(InvTokenId::Bos);
298
299 // --- Metadata ---
300 tokens.emplace_back(InvTokenId::MetaBegin);
301 tokens.emplace_back(methodToTokenId(estimate.method));
302 tokens.emplace_back(spaceToTokenId(estimate.sourceSpaceType));
303 tokens.emplace_back(orientToTokenId(estimate.orientationType));
304 tokens.emplace_back(InvTokenId::MetaEnd);
305
306 // --- Grid data ---
307 if (options.includeGridData && estimate.has_grid_data()) {
308 tokens.emplace_back(InvTokenId::GridBegin);
309 tokens.emplace_back(InvTokenId::NSources, static_cast<float>(effSrc));
310 tokens.emplace_back(InvTokenId::NTimes, static_cast<float>(effTime));
311 tokens.emplace_back(InvTokenId::TimeVal, estimate.tmin);
312 tokens.emplace_back(InvTokenId::TStep, estimate.tstep * static_cast<float>(timeStride));
313
314 // Vertex indices
315 for (int s = 0; s < nSrc; s += srcStride)
316 tokens.emplace_back(InvTokenId::Vertex, static_cast<float>(estimate.vertices[s]));
317
318 // Amplitude data row by row
319 for (int s = 0; s < nSrc; s += srcStride) {
320 tokens.emplace_back(InvTokenId::GridRow);
321 for (int t = 0; t < nTimes; t += timeStride)
322 tokens.emplace_back(InvTokenId::Amplitude, static_cast<float>(estimate.data(s, t)));
323 }
324 tokens.emplace_back(InvTokenId::GridEnd);
325 }
326
327 // --- Positions ---
328 if (options.includePositions && estimate.has_positions()) {
329 tokens.emplace_back(InvTokenId::PosBegin);
330 int nPos = static_cast<int>(estimate.positions.rows());
331 for (int i = 0; i < nPos; i += srcStride) {
332 tokens.emplace_back(InvTokenId::PosX, estimate.positions(i, 0));
333 tokens.emplace_back(InvTokenId::PosY, estimate.positions(i, 1));
334 tokens.emplace_back(InvTokenId::PosZ, estimate.positions(i, 2));
335 }
336 tokens.emplace_back(InvTokenId::PosEnd);
337 }
338
339 // --- Couplings ---
340 if (options.includeCouplings && estimate.has_couplings()) {
341 tokens.emplace_back(InvTokenId::CouplingBegin);
342 tokens.emplace_back(InvTokenId::NGroups, static_cast<float>(estimate.couplings.size()));
343
344 for (const auto& grp : estimate.couplings) {
345 tokens.emplace_back(InvTokenId::GroupBegin);
346 tokens.emplace_back(InvTokenId::TimeVal, grp.tmin);
347 tokens.emplace_back(InvTokenId::TimeVal, grp.tmax);
348 tokens.emplace_back(InvTokenId::NIndices, static_cast<float>(grp.gridIndices.size()));
349
350 for (size_t k = 0; k < grp.gridIndices.size(); ++k) {
351 tokens.emplace_back(InvTokenId::GridIndex, static_cast<float>(grp.gridIndices[k]));
352 if (k < grp.moments.size()) {
353 tokens.emplace_back(InvTokenId::MomX, static_cast<float>(grp.moments[k].x()));
354 tokens.emplace_back(InvTokenId::MomY, static_cast<float>(grp.moments[k].y()));
355 tokens.emplace_back(InvTokenId::MomZ, static_cast<float>(grp.moments[k].z()));
356 }
357 }
358
359 // Upper-triangle of N×N correlation matrix
360 int n = static_cast<int>(grp.gridIndices.size());
361 for (int r = 0; r < n && r < grp.correlations.rows(); ++r)
362 for (int c = r; c < n && c < grp.correlations.cols(); ++c)
363 tokens.emplace_back(InvTokenId::Correlation, static_cast<float>(grp.correlations(r, c)));
364
365 tokens.emplace_back(InvTokenId::GroupEnd);
366 }
367 tokens.emplace_back(InvTokenId::CouplingEnd);
368 }
369
370 // --- Focal dipoles ---
371 if (options.includeFocalDipoles && estimate.has_focal_dipoles()) {
372 tokens.emplace_back(InvTokenId::FocalBegin);
373 tokens.emplace_back(InvTokenId::NDipoles, static_cast<float>(estimate.focalDipoles.size()));
374
375 for (const auto& dip : estimate.focalDipoles) {
376 tokens.emplace_back(InvTokenId::DipoleBegin);
377 tokens.emplace_back(InvTokenId::TimeVal, dip.tmin);
378 tokens.emplace_back(InvTokenId::TimeVal, dip.tmax);
379 tokens.emplace_back(InvTokenId::PosX, dip.position.x());
380 tokens.emplace_back(InvTokenId::PosY, dip.position.y());
381 tokens.emplace_back(InvTokenId::PosZ, dip.position.z());
382 tokens.emplace_back(InvTokenId::MomX, dip.moment.x());
383 tokens.emplace_back(InvTokenId::MomY, dip.moment.y());
384 tokens.emplace_back(InvTokenId::MomZ, dip.moment.z());
385 tokens.emplace_back(InvTokenId::GridIndex, static_cast<float>(dip.gridIndex));
386 tokens.emplace_back(InvTokenId::Goodness, dip.goodness);
387 tokens.emplace_back(InvTokenId::ChiSquared, dip.khi2);
388 tokens.emplace_back(InvTokenId::NFreeDof, static_cast<float>(dip.nfree));
389 tokens.emplace_back(dip.valid ? InvTokenId::ValidTrue : InvTokenId::ValidFalse);
390 tokens.emplace_back(InvTokenId::DipoleEnd);
391 }
392 tokens.emplace_back(InvTokenId::FocalEnd);
393 }
394
395 // --- Connectivity ---
396 if (options.includeConnectivity && estimate.has_connectivity()) {
397 tokens.emplace_back(InvTokenId::ConnBegin);
398 tokens.emplace_back(InvTokenId::NMeasures, static_cast<float>(estimate.connectivity.size()));
399
400 for (const auto& conn : estimate.connectivity) {
401 tokens.emplace_back(InvTokenId::ConnEntryBegin);
402 tokens.emplace_back(measureToTokenId(conn.measure));
403 tokens.emplace_back(conn.directed ? InvTokenId::DirectedTrue : InvTokenId::DirectedFalse);
404 tokens.emplace_back(InvTokenId::FreqVal, conn.fmin);
405 tokens.emplace_back(InvTokenId::FreqVal, conn.fmax);
406 tokens.emplace_back(InvTokenId::TimeVal, conn.tmin);
407 tokens.emplace_back(InvTokenId::TimeVal, conn.tmax);
408
409 int n = static_cast<int>(conn.matrix.rows());
410 tokens.emplace_back(InvTokenId::NSources, static_cast<float>(n));
411 for (int r = 0; r < n; ++r)
412 for (int c = 0; c < n; ++c)
413 tokens.emplace_back(InvTokenId::ConnValue, static_cast<float>(conn.matrix(r, c)));
414
415 tokens.emplace_back(InvTokenId::ConnEntryEnd);
416 }
417 tokens.emplace_back(InvTokenId::ConnEnd);
418 }
419
420 // --- EOS ---
421 tokens.emplace_back(InvTokenId::Eos);
422 return tokens;
423}

429InvSourceEstimate fromTokens(const std::vector<InvToken>& tokens)
430{
431 InvSourceEstimate est;
432 size_t pos = 0;
433 const size_t len = tokens.size();
434
435 auto advance = [&]() -> const InvToken& {
436 return tokens[pos++];
437 };
438
439 auto peek = [&]() -> InvTokenId {
440 return (pos < len) ? tokens[pos].id : InvTokenId::Eos;
441 };
442
443 // Skip BOS
444 if (pos < len && tokens[pos].id == InvTokenId::Bos)
445 ++pos;
446
447 while (pos < len && tokens[pos].id != InvTokenId::Eos) {
448 const InvToken& tok = tokens[pos];
449
450 // --- Metadata ---
451 if (tok.id == InvTokenId::MetaBegin) {
452 ++pos;
453 while (pos < len && peek() != InvTokenId::MetaEnd) {
454 if (isMethodToken(peek()))
455 est.method = tokenIdToMethod(advance().id);
456 else if (isSpaceToken(peek()))
457 est.sourceSpaceType = tokenIdToSpace(advance().id);
458 else if (isOrientToken(peek()))
459 est.orientationType = tokenIdToOrient(advance().id);
460 else
461 ++pos;
462 }
463 if (pos < len)
464 ++pos; // skip MetaEnd
465 }
466
467 // --- Grid data ---
468 else if (tok.id == InvTokenId::GridBegin) {
469 ++pos;
470 std::vector<int> verts;
471 std::vector<std::vector<float>> rows;
472
473 while (pos < len && peek() != InvTokenId::GridEnd) {
474 const InvToken& g = tokens[pos];
475
476 if (g.id == InvTokenId::NSources) {
477 ++pos;
478 } else if (g.id == InvTokenId::NTimes) {
479 ++pos;
480 } else if (g.id == InvTokenId::TimeVal && est.tmin == 0 && est.tstep == -1) {
481 est.tmin = g.value;
482 ++pos;
483 } else if (g.id == InvTokenId::TStep) {
484 est.tstep = g.value;
485 ++pos;
486 } else if (g.id == InvTokenId::TimeVal) {
487 // Second TimeVal inside grid is still tmin if tstep wasn't set yet
488 ++pos;
489 } else if (g.id == InvTokenId::Vertex) {
490 verts.push_back(static_cast<int>(g.value));
491 ++pos;
492 } else if (g.id == InvTokenId::GridRow) {
493 ++pos;
494 std::vector<float> row;
495 while (pos < len && tokens[pos].id == InvTokenId::Amplitude) {
496 row.push_back(tokens[pos].value);
497 ++pos;
498 }
499 rows.push_back(std::move(row));
500 } else {
501 ++pos;
502 }
503 }
504 if (pos < len)
505 ++pos; // skip GridEnd
506
507 // Fill Eigen structures
508 int effSrc = static_cast<int>(rows.size());
509 int effTime = effSrc > 0 ? static_cast<int>(rows[0].size()) : 0;
510 if (effSrc > 0 && effTime > 0) {
511 est.data = MatrixXd(effSrc, effTime);
512 for (int s = 0; s < effSrc; ++s)
513 for (int t = 0; t < effTime && t < static_cast<int>(rows[s].size()); ++t)
514 est.data(s, t) = static_cast<double>(rows[s][t]);
515 }
516 est.vertices = VectorXi(static_cast<int>(verts.size()));
517 for (int i = 0; i < static_cast<int>(verts.size()); ++i)
518 est.vertices[i] = verts[i];
519 }
520
521 // --- Positions ---
522 else if (tok.id == InvTokenId::PosBegin) {
523 ++pos;
524 std::vector<Vector3f> posVec;
525 while (pos + 2 < len && peek() != InvTokenId::PosEnd) {
526 if (tokens[pos].id == InvTokenId::PosX) {
527 Vector3f p;
528 p.x() = tokens[pos].value;
529 p.y() = tokens[pos + 1].value;
530 p.z() = tokens[pos + 2].value;
531 posVec.push_back(p);
532 pos += 3;
533 } else {
534 ++pos;
535 }
536 }
537 if (pos < len)
538 ++pos; // skip PosEnd
539
540 est.positions = MatrixX3f(static_cast<int>(posVec.size()), 3);
541 for (int i = 0; i < static_cast<int>(posVec.size()); ++i)
542 est.positions.row(i) = posVec[i].transpose();
543 }
544
545 // --- Couplings ---
546 else if (tok.id == InvTokenId::CouplingBegin) {
547 ++pos;
548 if (pos < len && tokens[pos].id == InvTokenId::NGroups)
549 ++pos; // skip count
550
551 while (pos < len && peek() != InvTokenId::CouplingEnd) {
552 if (peek() == InvTokenId::GroupBegin) {
553 ++pos;
554 InvSourceCoupling grp;
555 int timeIdx = 0;
556
557 while (pos < len && peek() != InvTokenId::GroupEnd) {
558 const InvToken& ct = tokens[pos];
559 if (ct.id == InvTokenId::TimeVal) {
560 if (timeIdx++ == 0)
561 grp.tmin = ct.value;
562 else
563 grp.tmax = ct.value;
564 ++pos;
565 } else if (ct.id == InvTokenId::NIndices) {
566 ++pos;
567 } else if (ct.id == InvTokenId::GridIndex) {
568 grp.gridIndices.push_back(static_cast<int>(ct.value));
569 ++pos;
570 } else if (ct.id == InvTokenId::MomX && pos + 2 < len) {
571 Vector3d m;
572 m.x() = static_cast<double>(tokens[pos].value);
573 m.y() = static_cast<double>(tokens[pos + 1].value);
574 m.z() = static_cast<double>(tokens[pos + 2].value);
575 grp.moments.push_back(m);
576 pos += 3;
577 } else if (ct.id == InvTokenId::Correlation) {
578 // Reconstruct upper-triangle
579 int n = static_cast<int>(grp.gridIndices.size());
580 if (n > 0 && grp.correlations.size() == 0) {
581 grp.correlations = MatrixXd::Zero(n, n);
582 for (int r = 0; r < n; ++r) {
583 for (int c = r; c < n && pos < len && tokens[pos].id == InvTokenId::Correlation; ++c) {
584 double v = static_cast<double>(tokens[pos].value);
585 grp.correlations(r, c) = v;
586 grp.correlations(c, r) = v;
587 ++pos;
588 }
589 }
590 } else {
591 ++pos;
592 }
593 } else {
594 ++pos;
595 }
596 }
597 if (pos < len)
598 ++pos; // skip GroupEnd
599 est.couplings.push_back(std::move(grp));
600 } else {
601 ++pos;
602 }
603 }
604 if (pos < len)
605 ++pos; // skip CouplingEnd
606 }
607
608 // --- Focal dipoles ---
609 else if (tok.id == InvTokenId::FocalBegin) {
610 ++pos;
611 if (pos < len && tokens[pos].id == InvTokenId::NDipoles)
612 ++pos;
613
614 while (pos < len && peek() != InvTokenId::FocalEnd) {
615 if (peek() == InvTokenId::DipoleBegin) {
616 ++pos;
617 InvFocalDipole dip;
618 int timeIdx = 0;
619
620 while (pos < len && peek() != InvTokenId::DipoleEnd) {
621 const InvToken& dt = tokens[pos];
622 if (dt.id == InvTokenId::TimeVal) {
623 if (timeIdx == 0)
624 dip.tmin = dt.value;
625 else
626 dip.tmax = dt.value;
627 ++timeIdx;
628 ++pos;
629 } else if (dt.id == InvTokenId::PosX) {
630 dip.position.x() = dt.value;
631 ++pos;
632 } else if (dt.id == InvTokenId::PosY) {
633 dip.position.y() = dt.value;
634 ++pos;
635 } else if (dt.id == InvTokenId::PosZ) {
636 dip.position.z() = dt.value;
637 ++pos;
638 } else if (dt.id == InvTokenId::MomX) {
639 dip.moment.x() = dt.value;
640 ++pos;
641 } else if (dt.id == InvTokenId::MomY) {
642 dip.moment.y() = dt.value;
643 ++pos;
644 } else if (dt.id == InvTokenId::MomZ) {
645 dip.moment.z() = dt.value;
646 ++pos;
647 } else if (dt.id == InvTokenId::GridIndex) {
648 dip.gridIndex = static_cast<int>(dt.value);
649 ++pos;
650 } else if (dt.id == InvTokenId::Goodness) {
651 dip.goodness = dt.value;
652 ++pos;
653 } else if (dt.id == InvTokenId::ChiSquared) {
654 dip.khi2 = dt.value;
655 ++pos;
656 } else if (dt.id == InvTokenId::NFreeDof) {
657 dip.nfree = static_cast<int>(dt.value);
658 ++pos;
659 } else if (dt.id == InvTokenId::ValidTrue) {
660 dip.valid = true;
661 ++pos;
662 } else if (dt.id == InvTokenId::ValidFalse) {
663 dip.valid = false;
664 ++pos;
665 } else {
666 ++pos;
667 }
668 }
669 if (pos < len)
670 ++pos; // skip DipoleEnd
671 est.focalDipoles.push_back(dip);
672 } else {
673 ++pos;
674 }
675 }
676 if (pos < len)
677 ++pos; // skip FocalEnd
678 }
679
680 // --- Connectivity ---
681 else if (tok.id == InvTokenId::ConnBegin) {
682 ++pos;
683 if (pos < len && tokens[pos].id == InvTokenId::NMeasures)
684 ++pos;
685
686 while (pos < len && peek() != InvTokenId::ConnEnd) {
687 if (peek() == InvTokenId::ConnEntryBegin) {
688 ++pos;
689 InvConnectivity conn;
690 int n = 0;
691 int freqIdx = 0, timeIdx = 0;
692
693 while (pos < len && peek() != InvTokenId::ConnEntryEnd) {
694 const InvToken& ct = tokens[pos];
695 if (isMeasureToken(ct.id)) {
696 conn.measure = tokenIdToMeasure(ct.id);
697 ++pos;
698 } else if (ct.id == InvTokenId::DirectedTrue) {
699 conn.directed = true;
700 ++pos;
701 } else if (ct.id == InvTokenId::DirectedFalse) {
702 conn.directed = false;
703 ++pos;
704 } else if (ct.id == InvTokenId::FreqVal) {
705 if (freqIdx == 0)
706 conn.fmin = ct.value;
707 else
708 conn.fmax = ct.value;
709 ++freqIdx;
710 ++pos;
711 } else if (ct.id == InvTokenId::TimeVal) {
712 if (timeIdx == 0)
713 conn.tmin = ct.value;
714 else
715 conn.tmax = ct.value;
716 ++timeIdx;
717 ++pos;
718 } else if (ct.id == InvTokenId::NSources) {
719 n = static_cast<int>(ct.value);
720 ++pos;
721 } else if (ct.id == InvTokenId::ConnValue && n > 0) {
722 conn.matrix = MatrixXd(n, n);
723 for (int r = 0; r < n; ++r)
724 for (int c = 0; c < n && pos < len && tokens[pos].id == InvTokenId::ConnValue; ++c, ++pos)
725 conn.matrix(r, c) = static_cast<double>(tokens[pos].value);
726 } else {
727 ++pos;
728 }
729 }
730 if (pos < len)
731 ++pos; // skip ConnEntryEnd
732 est.connectivity.push_back(std::move(conn));
733 } else {
734 ++pos;
735 }
736 }
737 if (pos < len)
738 ++pos; // skip ConnEnd
739 }
740
741 else {
742 ++pos; // skip unrecognised tokens
743 }
744 }
745
746 // Rebuild times vector
747 if (est.data.cols() > 0 && est.tstep > 0) {
748 est.times = RowVectorXf(est.data.cols());
749 est.times[0] = est.tmin;
750 for (int i = 1; i < est.times.size(); ++i)
751 est.times[i] = est.times[i - 1] + est.tstep;
752 }
753
754 return est;
755}
