96void Spline::mousePressEvent(QMouseEvent* event)
97{
98 if (m_plot.class_marks().empty()) {
99 mne::Log::debug("Data set not found.");
100 return;
101 }
102 const double dataX = pixelToDataX(event->position().x());
103 if (dataX < m_plot.min_x() || dataX > m_plot.max_x()) {
104 return;
105 }
106 double left = m_plot.left_threshold();
107 double middle = m_plot.middle_threshold();
108 double right = m_plot.right_threshold();
109 bool changed = false;
110 if (event->buttons() == Qt::LeftButton) {
111 if (dataX < middle && dataX < right) {
112 left = dataX;
113 changed = true;
114 }
115 } else if (event->buttons() == Qt::MiddleButton) {
116 if (dataX > left && dataX < right) {
117 middle = dataX;
118 changed = true;
119 }
120 } else if (event->buttons() == Qt::RightButton) {
121 if (dataX > left && dataX > middle) {
122 right = dataX;
123 changed = true;
124 }
125 }
126 if (changed) {
127 m_plot.set_thresholds(left, middle, right);
128 }
129 emit borderChanged(left, middle, right);
130 QWidget::update();
131}

135void Spline::setThreshold(const QVector3D& vecThresholdValues)
136{
137 if (m_plot.class_marks().empty()) {
138 mne::Log::debug("Data set not found.");
139 return;
140 }
141 const double minX = m_plot.min_x();
142 const double maxX = m_plot.max_x();
143 double values[3] = {static_cast<double>(vecThresholdValues.x()), static_cast<double>(vecThresholdValues.y()),
144 static_cast<double>(vecThresholdValues.z())};
145 if (std::any_of(std::begin(values), std::end(values), [&](double v) { return v < minX || v > maxX; })) {
146 mne::Log::debug("One or more of the values given are out of the minimum and maximum range. Changed to default thresholds.");
147 // 1 % inside each end of the span; scaling the limits themselves leaves the axis for negative values
148 const double span = maxX - minX;
149 m_plot.set_thresholds(minX + 0.01 * span, minX + 0.5 * span, maxX - 0.01 * span);
150 } else {
151 // Sort the three values into left < middle < right
152 std::sort(std::begin(values), std::end(values));
153 m_plot.set_thresholds(values[0], values[1], values[2]);
154 }
155 QWidget::update();
156}
