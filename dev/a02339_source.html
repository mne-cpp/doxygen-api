87 void requestViewportPresetMenu(int viewport, const QPoint& pos) override
88 {
89 BrainViewCore* core = m_view.core();
90 if (viewport < 0 || viewport > 3) {
91 return;
92 }
93
94 static constexpr std::array<const char*, 7> kPresetNames = {"Top", "Perspective", "Front", "Left", "Bottom", "Back", "Right"};
95 QMenu menu;
96 const int currentPreset = std::clamp(core->viewportCameraPreset(viewport), 0, 6);
97 std::array<QAction*, kPresetNames.size()> actions{};
98 for (int preset = 0; preset < std::ssize(kPresetNames); ++preset) {
99 if (preset == 4) {
100 menu.addSeparator();
101 }
102 actions[preset] = menu.addAction(QString::fromLatin1(kPresetNames[preset]));
103 actions[preset]->setCheckable(true);
104 actions[preset]->setChecked(preset == currentPreset);
105 }
106
107 QAction* selected = menu.exec(m_view.mapToGlobal(pos));
108 const auto chosen = std::ranges::find(actions, selected);
109 if (selected && chosen != actions.end()) {
110 core->setViewportCameraPreset(viewport, static_cast<int>(chosen - actions.begin()));
111 }
112 }
113
114 bool beginOffscreenCapture(const QSize& size) override
115 {
116 // Realise the widget off-screen: QRhi init still runs because the widget is
117 // technically "shown", but no window-manager surface is created.
118 m_view.setAttribute(Qt::WA_DontShowOnScreen, true);
119 m_view.resize(size);
120 m_view.show();
121 return true;
122 }
123
124 void endOffscreenCapture() override
125 {
126 m_view.hide();
127 m_view.setAttribute(Qt::WA_DontShowOnScreen, false);
128 }
129
130private:
131 BrainView& m_view;
132};

138BrainView::BrainView(QWidget* parent)
139: QRhiWidget(parent)
140, m_pHost(std::make_unique<Host>(*this))
141{
142 setMinimumSize(800, 600);
143 setSampleCount(1);
144 setAutoRenderTarget(false); // The renderer manages its own dual render targets
145
146#if defined(WASMBUILD) || defined(__EMSCRIPTEN__)
147 setApi(Api::OpenGL); // WebGL 2 (OpenGL ES 3.0) on WASM
148#elif defined(Q_OS_MACOS) || defined(Q_OS_IOS)
149 setApi(Api::Metal);
150#elif defined(Q_OS_WIN)
151 setApi(Api::Direct3D11);
152#else
153 setApi(Api::OpenGL);
154#endif
155
156 setMouseTracking(true); // Enable hover events
157
158 m_pCore = std::make_unique<BrainViewCore>(m_pHost.get());
159
160 // Re-emit the core's signals so connections to BrainView keep working
161 BrainViewCore* core = m_pCore.get();
162 connect(core, &BrainViewCore::timePointChanged, this, &BrainView::timePointChanged);
163 connect(core, &BrainViewCore::sourceEstimateLoaded, this, &BrainView::sourceEstimateLoaded);
164 connect(core, &BrainViewCore::sensorFieldLoaded, this, &BrainView::sensorFieldLoaded);
165 connect(core, &BrainViewCore::sourceThresholdsUpdated, this, &BrainView::sourceThresholdsUpdated);
166 connect(core, &BrainViewCore::stcLoadingProgress, this, &BrainView::stcLoadingProgress);
167 connect(core, &BrainViewCore::sensorFieldTimePointChanged, this, &BrainView::sensorFieldTimePointChanged);
168 connect(core, &BrainViewCore::hoveredRegionChanged, this, &BrainView::hoveredRegionChanged);
169 connect(core, &BrainViewCore::electrodeContactClicked, this, &BrainView::electrodeContactClicked);
170 connect(core, &BrainViewCore::visualizationEditTargetChanged, this, &BrainView::visualizationEditTargetChanged);
171 connect(core, &BrainViewCore::viewCountChanged, this, &BrainView::viewCountChanged);
172 connect(core, &BrainViewCore::shaderModeChanged, this, &BrainView::shaderModeChanged);
173 connect(core, &BrainViewCore::surfacePointClicked, this, &BrainView::surfacePointClicked);
174 connect(core, &BrainViewCore::surfacePointDoubleClicked, this, &BrainView::surfacePointDoubleClicked);
175}
