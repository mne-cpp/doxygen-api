144AcqExpected<std::shared_ptr<const Api>> loadApi()
145{
146 // Only from the system directory, where the driver installs it, so no other copy can be planted
147 char system[MAX_PATH] = {};
148 const UINT length = GetSystemDirectoryA(system, MAX_PATH);
149 if (length == 0 || length >= MAX_PATH) {
150 return AcqError(AcqErrorCode::BackendUnavailable, "the Windows system directory is unknown");
151 }
152 auto library = AcqSharedLibrary::load({std::string(system) + "\\gUSBamp.dll"});
153 if (!library) {
154 return AcqError(AcqErrorCode::BackendUnavailable,
155 "the g.USBamp API is not installed; install the amplifier's driver (" + library.error().text() + ")");
156 }
157 auto api = std::make_shared<Api>();
158 api->library = std::move(*library);
159 const AcqSharedLibrary& l = api->library;
160 for (const AcqError& e :
161 {resolve(l, api->openDevice, "GT_OpenDevice"), resolve(l, api->openDeviceEx, "GT_OpenDeviceEx"),
162 resolve(l, api->closeDevice, "GT_CloseDevice"), resolve(l, api->getData, "GT_GetData"),
163 resolve(l, api->setBufferSize, "GT_SetBufferSize"), resolve(l, api->setSampleRate, "GT_SetSampleRate"),
164 resolve(l, api->start, "GT_Start"), resolve(l, api->stop, "GT_Stop"), resolve(l, api->setChannels, "GT_SetChannels"),
165 resolve(l, api->resetTransfer, "GT_ResetTransfer"), resolve(l, api->getSerial, "GT_GetSerial"),
166 resolve(l, api->enableTriggerLine, "GT_EnableTriggerLine"), resolve(l, api->getImpedance, "GT_GetImpedance"),
167 resolve(l, api->setBandPass, "GT_SetBandPass"), resolve(l, api->setNotch, "GT_SetNotch"), resolve(l, api->setMode, "GT_SetMode"),
168 resolve(l, api->setGround, "GT_SetGround"), resolve(l, api->setReference, "GT_SetReference"),

170 resolve(l, api->getLastError, "GT_GetLastError")}) {

206 AcqError open(std::span<const std::string> serials)
207 {
208 for (const std::string& serial : serials) {
209 std::string name = serial; // the API takes a mutable string
210 HANDLE handle = m_api->openDeviceEx(name.data());
211 if (handle == nullptr) {
212 return {AcqErrorCode::DeviceNotFound, std::format("no g.USBamp {} is connected", serial)};
213 }
214 Unit& unit = m_units.emplace_back();
215 unit.serial = serial;
216 unit.handle = handle;
217 }
218 return {};

222 {
223 std::vector<std::string> out;
224 for (const Unit& unit : m_units) {
225 out.push_back(unit.serial);
226 }
227 return out;

230 std::string description() const override
231 {
232 std::string text = "g.USBamp";
233 for (std::size_t u = 0; u < m_units.size(); ++u) {
234 text += (u == 0 ? " " : ", ") + m_units[u].serial + (u == 0 && m_units.size() > 1 ? " (master)" : "");
235 }
236 return text;

239 AcqError start(const GusbampSetup& setup) override
240 {
241 m_setup = setup;
242 for (std::size_t u = 0; u < m_units.size(); ++u) {
243 if (AcqError e = configure(m_units[u], u); !e.ok()) {
244 return e;
245 }
246 }
247 // The slaves start first, so the master's clock reaches every unit; then the transfers are queued
248 for (std::size_t n = 0; n < m_units.size(); ++n) {
249 const std::size_t u = (n + 1) % m_units.size();
250 Unit& unit = m_units[u];
251 for (std::size_t q = 0; q < kQueue; ++q) {
252 unit.buffers[q].assign(setup.transfer_bytes(u), std::byte{0});
253 unit.overlapped[q] = {};
254 unit.overlapped[q].hEvent = CreateEventA(nullptr, FALSE, FALSE, nullptr);
255 if (unit.overlapped[q].hEvent == nullptr) {
256 stop();
257 return {AcqErrorCode::OutOfMemory, "no event for a g.USBamp transfer"};
258 }
259 }
260 if (!m_api->start(unit.handle)) {
261 stop();
262 return m_api->error("g.USBamp " + unit.serial + " did not start");
263 }
264 m_started = true;
265 for (std::size_t q = 0; q < kQueue; ++q) {
266 if (!queue(unit, q)) {
267 stop();
268 return m_api->error("g.USBamp " + unit.serial + " refused a transfer");
269 }
270 }
271 }
272 m_next = 0;
273 m_held = false;
274 m_transfers.assign(m_units.size(), {});
275 return {};
276 }
277
278 void stop() override
279 {
280 if (!m_started) {
281 return;
282 }
283 for (Unit& unit : m_units) {
284 for (OVERLAPPED& overlapped : unit.overlapped) {
285 if (overlapped.hEvent != nullptr) {
286 WaitForSingleObject(overlapped.hEvent, kStopWaitMs);
287 CloseHandle(overlapped.hEvent);
288 overlapped.hEvent = nullptr;
289 }
290 }
291 m_api->stop(unit.handle);
292 m_api->resetTransfer(unit.handle);
293 }
294 m_started = false;
295 }

297 AcqExpected<std::span<const std::span<const std::byte>>> read(std::chrono::milliseconds timeout) override
298 {
299 if (m_held) {
300 // The previous transfer has been copied: queue its buffers again
301 for (Unit& unit : m_units) {
302 if (!queue(unit, m_next)) {
303 return m_api->error("g.USBamp " + unit.serial + " refused a transfer");
304 }
305 }
306 m_next = (m_next + 1) % kQueue;
307 m_held = false;
308 }
309 for (std::size_t u = 0; u < m_units.size(); ++u) {
310 Unit& unit = m_units[u];
311 const DWORD waited = WaitForSingleObject(unit.overlapped[m_next].hEvent, static_cast<DWORD>(timeout.count()));
312 if (waited == WAIT_TIMEOUT) {
313 return AcqError(AcqErrorCode::Timeout);
314 }
315 DWORD bytes = 0;
316 if (waited != WAIT_OBJECT_0 || !GetOverlappedResult(unit.handle, &unit.overlapped[m_next], &bytes, FALSE)) {
317 return AcqError(AcqErrorCode::DeviceIo, std::format("the transfer of g.USBamp {} failed (error {})", unit.serial, GetLastError()));
318 }
319 if (bytes != unit.buffers[m_next].size()) {
320 return AcqError(AcqErrorCode::DeviceIo,
321 std::format("g.USBamp {} delivered {} of {} bytes", unit.serial, bytes, unit.buffers[m_next].size()));
322 }
323 m_transfers[u] = unit.buffers[m_next];
324 }
325 m_held = true;
326 return std::span<const std::span<const std::byte>>(m_transfers);
327 }
328
329 AcqExpected<std::vector<double>> impedances(std::span<const std::uint8_t> inputs) override
330 {
331 if (m_started) {
332 return AcqError(AcqErrorCode::InvalidState, "the g.USBamp measures impedances only while stopped");
333 }
334 std::vector<double> ohms;
335 ohms.reserve(m_units.size() * inputs.size());
336 for (const Unit& unit : m_units) {
337 for (const std::uint8_t input : inputs) {
338 double value = 0.0;
339 if (!m_api->getImpedance(unit.handle, input, &value)) {
340 return m_api->error(std::format("g.USBamp {} did not measure input {}", unit.serial, input));
341 }
342 ohms.push_back(value);
343 }
344 }
345 return ohms;
346 }

348private:
350 AcqError configure(const Unit& unit, std::size_t index)
351 {
352 std::vector<UCHAR> inputs(m_setup.inputs.begin(), m_setup.inputs.end());
353 const bool master = index == 0;
354 const std::string name = "g.USBamp " + unit.serial;
355 if (!m_api->setMode(unit.handle, kModeNormal) || !m_api->setSlave(unit.handle, master ? FALSE : TRUE) ||
356 !m_api->setSampleRate(unit.handle, m_setup.rate) || !m_api->setBufferSize(unit.handle, m_setup.scans) ||
357 !m_api->setChannels(unit.handle, inputs.data(), static_cast<UCHAR>(inputs.size())) ||
358 !m_api->enableTriggerLine(unit.handle, m_setup.trigger && master ? TRUE : FALSE) || !m_api->enableSc(unit.handle, FALSE) ||
359 !m_api->setBipolar(unit.handle, CHANNEL{}) || !m_api->setReference(unit.handle, REF{}) || !m_api->setGround(unit.handle, GND{})) {
360 return m_api->error(name + " refused the setup");
361 }
362 for (const UCHAR input : inputs) {
363 if (!m_api->setBandPass(unit.handle, input, -1) || !m_api->setNotch(unit.handle, input, -1)) {
364 return m_api->error(name + " refused to turn off its filters");
365 }
366 }
367 return {};
368 }

370 bool queue(Unit& unit, std::size_t slot)
371 {
372 std::vector<std::byte>& buffer = unit.buffers[slot];
373 return m_api->getData(unit.handle, reinterpret_cast<BYTE*>(buffer.data()), static_cast<DWORD>(buffer.size()), &unit.overlapped[slot]) !=
374 FALSE;
375 }
376
377 std::shared_ptr<const Api> m_api;
378 std::vector<Unit> m_units; // master first
379 GusbampSetup m_setup;
380 bool m_started = false;
381 std::size_t m_next = 0; // queue slot of the next transfer
382 bool m_held = false; // the slot's buffers were handed out by read() and are not queued
383 std::vector<std::span<const std::byte>> m_transfers;
384};

392AcqExpected<std::vector<std::string>> mne::detail::list_gusbamp_units()
393{
394 auto api = loadApi();
395 if (!api) {
396 return api.error();
397 }
398 std::vector<std::string> serials;
399 for (int port = 0; port < kPorts; ++port) {
400 HANDLE handle = (*api)->openDevice(port);
401 if (handle == nullptr) {
402 continue;
403 }
404 std::array<char, 16> serial{};
405 if ((*api)->getSerial(handle, serial.data(), static_cast<UINT>(serial.size()))) {
406 serials.emplace_back(serial.data());
407 }
408 (*api)->closeDevice(&handle);
409 }
410 return serials;
411}

415AcqExpected<std::unique_ptr<GusbampTransport>> mne::detail::open_gusbamp_driver(std::span<const std::string> serials)
416{
417 auto api = loadApi();
418 if (!api) {
419 return api.error();
420 }
421 std::vector<std::string> wanted(serials.begin(), serials.end());
422 if (wanted.empty()) {
423 auto units = list_gusbamp_units();
424 if (!units) {
425 return units.error();
426 }
427 if (units->empty()) {
428 return AcqError(AcqErrorCode::DeviceNotFound, "no g.USBamp is connected");
429 }
430 wanted.push_back(units->front());
431 }
432 auto driver = std::make_unique<GusbampDriver>(std::move(*api));
433 if (AcqError e = driver->open(wanted); !e.ok()) {
434 return e;
435 }
436 return std::unique_ptr<GusbampTransport>(std::move(driver));
