37std::vector<IeegBipolarPair> IeegReference::bipolarPairs(const IeegImplantation& implantation)
38{
39 const std::vector<IeegContact>& contacts = implantation.contacts();
40 std::vector<IeegBipolarPair> pairs;
41 auto add = [&](int anode, int cathode) {
42 const Eigen::Vector3f& a = contacts[std::size_t(anode)].position;
43 const Eigen::Vector3f& c = contacts[std::size_t(cathode)].position;
44 if ((a - c).norm() > kMaxNeighbourDistance)
45 return;
46 pairs.push_back({anode, cathode, contacts[std::size_t(anode)].name + '-' + contacts[std::size_t(cathode)].name, (a + c) * 0.5f});
47 };
48
49 for (const IeegLead& lead : implantation.leads()) {
50 const std::vector<int>& order = lead.contacts;
51 const int count = static_cast<int>(order.size());
52 const bool grid = lead.type == IeegLeadType::Grid && std::int64_t{lead.rows} * lead.cols == count && lead.cols > 0;
53 const int cols = grid ? lead.cols : count;
54 // Along each row (the whole lead for depth leads and strips)
55 for (int k = 0; k + 1 < count; ++k) {
56 if ((k + 1) % cols != 0)
57 add(order[std::size_t(k)], order[std::size_t(k) + 1]);
58 }
59 // Down each column of a grid
60 if (grid) {
61 for (int k = 0; k + cols < count; ++k)
62 add(order[std::size_t(k)], order[std::size_t(k) + std::size_t(cols)]);
63 }
64 }
65 return pairs;
66}

70Eigen::SparseMatrix<double> IeegReference::bipolarMatrix(const std::vector<IeegBipolarPair>& pairs, int contactCount)
71{
72 std::vector<Eigen::Triplet<double>> entries;
73 entries.reserve(pairs.size() * 2);
74 for (std::size_t p = 0; p < pairs.size(); ++p) {
75 entries.emplace_back(int(p), pairs[p].anode, 1.0);
76 entries.emplace_back(int(p), pairs[p].cathode, -1.0);
77 }
78 Eigen::SparseMatrix<double> montage(Eigen::Index(pairs.size()), contactCount);
79 montage.setFromTriplets(entries.begin(), entries.end());
80 return montage;
81}
