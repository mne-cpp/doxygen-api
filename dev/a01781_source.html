64QList<FiffEpochData> FiffEpochs::makeFixedLengthEpochs(const MatrixXd& matData,
65 double dSFreq,
66 double dDuration,
67 double dOverlap,
68 bool bDropLast)
69{
70 QList<FiffEpochData> epochs;
71
72 if (matData.size() == 0 || dSFreq <= 0.0 || dDuration <= 0.0) {
73 qWarning() << "[FiffEpochs::makeFixedLengthEpochs] Invalid input parameters.";
74 return epochs;
75 }
76
77 if (dOverlap >= dDuration) {
78 qWarning() << "[FiffEpochs::makeFixedLengthEpochs] Overlap must be less than duration.";
79 return epochs;
80 }
81
82 const int nCh = static_cast<int>(matData.rows());
83 const int nTimes = static_cast<int>(matData.cols());
84 const int epochSamples = static_cast<int>(std::round(dDuration * dSFreq));
85 const int stepSamples = static_cast<int>(std::round((dDuration - dOverlap) * dSFreq));
86
87 if (epochSamples <= 0 || stepSamples <= 0)
88 return epochs;
89
90 int start = 0;
91 while (start + epochSamples <= nTimes) {
92 FiffEpochData epoch;
93 epoch.data = matData.block(0, start, nCh, epochSamples);
94 epoch.tmin = static_cast<double>(start) / dSFreq;
95 epoch.tmax = static_cast<double>(start + epochSamples - 1) / dSFreq;
96 epochs.append(epoch);
97 start += stepSamples;
98 }
99
100 // Handle remaining samples if not dropping last
101 if (!bDropLast && start < nTimes) {
102 int remaining = nTimes - start;
103 FiffEpochData epoch;
104 epoch.data = matData.block(0, start, nCh, remaining);
105 epoch.tmin = static_cast<double>(start) / dSFreq;
106 epoch.tmax = static_cast<double>(nTimes - 1) / dSFreq;
107 epochs.append(epoch);
108 }
109
110 return epochs;
111}

127FiffEvoked FiffEpochs::averageEpochs(const QList<FiffEpochData>& epochs,
128 double dSFreq,
129 const QString& comment)
130{
131 FiffEvoked evoked;
132
133 if (epochs.isEmpty()) {
134 qWarning() << "[FiffEpochs::averageEpochs] Empty epoch list.";
135 return evoked;
136 }
137
138 const int nCh = static_cast<int>(epochs[0].data.rows());
139 const int nTimes = static_cast<int>(epochs[0].data.cols());
140 const int nEpochs = epochs.size();
141
142 MatrixXd avg = MatrixXd::Zero(nCh, nTimes);
143
144 for (const auto& epoch : epochs) {
145 if (epoch.data.rows() != nCh || epoch.data.cols() != nTimes) {
146 qWarning() << "[FiffEpochs::averageEpochs] Inconsistent epoch dimensions; skipping.";
147 continue;
148 }
149 avg += epoch.data;
150 }
151
152 avg /= static_cast<double>(nEpochs);
153
154 // Populate the evoked
155 evoked.data = avg;
156 evoked.nave = nEpochs;
157 evoked.aspect_kind = FIFFV_ASPECT_AVERAGE;
158 evoked.comment = comment;
159
160 // Compute times vector from the first epoch
161 const double tmin = epochs[0].tmin;
162 evoked.first = static_cast<fiff_int_t>(std::round(tmin * dSFreq));
163 evoked.last = evoked.first + nTimes - 1;
164
165 RowVectorXf times(nTimes);
166 for (int i = 0; i < nTimes; ++i)
167 times(i) = static_cast<float>(tmin + static_cast<double>(i) / dSFreq);
168 evoked.times = times;
169
170 return evoked;
171}
