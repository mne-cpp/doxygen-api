181void DipoleFitPanel::changed(const std::string& key)
182{
183 if (key == kFit) {
184 if (m_callbacks.fit) {
185 m_callbacks.fit(fit_name());
186 }
187 } else if (key == kSet) {
188 if (m_callbacks.set_changed) {
189 m_callbacks.set_changed(form().get<int>(kSet));
190 }
191 } else if (key == kMeg || key == kEeg) {
192 report_modality();
193 } else if (key == kTmin || key == kTmax || key == kTstep || key == kTint) {
194 report_time();
195 } else if (key == kBmin || key == kBmax) {
196 report_baseline();
197 } else if (key == kMinDistance || key == kGuessRadius) {
198 report_fitting();
199 } else if (key == kGradNoise || key == kMagNoise || key == kEegNoise) {
200 report_noise();
201 } else if (key == kGradReg || key == kMagReg || key == kEegReg) {
202 report_regularization();
203 } else if (key == kOriginX || key == kOriginY || key == kOriginZ || key == kRadius) {
204 report_sphere();
205 } else {
206 for (const Model kind : {Model::Measurement, Model::Bem, Model::Mri, Model::Noise}) {
207 if (key == model_key(kind)) {
208 if (kind == Model::Measurement) {
209 name_fit();
210 }
211 if (m_callbacks.model_changed) {
212 m_callbacks.model_changed(kind, model(kind));
213 }
214 }
215 }
216 }
217}

221void DipoleFitPanel::name_fit()
222{
223 const std::string name = measurement_name(model(Model::Measurement));
224 std::string fitName;
225 if (!name.empty()) {
226 const std::time_t seconds = std::time(nullptr);
227 std::tm local{};
228#ifdef _WIN32
229 localtime_s(&local, &seconds);
230#else
231 localtime_r(&seconds, &local);
232#endif
233 std::array<char, 64> month{};
234 std::array<char, 64> rest{};
235 std::strftime(month.data(), month.size(), "%B", &local);
236 std::strftime(rest.data(), rest.size(), "%Y %H:%M:%S", &local);
237 fitName = "Dipole Fit - " + name + " - " + month.data() + " " + std::to_string(local.tm_mday) + " " + rest.data();
238 }
239 const bool updating = std::exchange(m_updating, true);
240 form().set_value(kName, fitName);
241 m_updating = updating;
242}

246std::string DipoleFitPanel::measurement_name(const std::string& measurement)
247{
248 // The first part between dots that is not empty
249 std::size_t start = measurement.find_first_not_of('.');
250 if (start == std::string::npos) {
251 return {};
252 }
253 std::string name = measurement.substr(start, measurement.find('.', start) - start);
254 for (const char* suffix : {"-ave", "_ave", "-raw", "_raw"}) {
255 if (name.ends_with(suffix)) {
256 name.resize(name.size() - 4);
257 break;
258 }
259 }
260 return name;
261}

277void DipoleFitPanel::remove_model(Model kind, const std::string& name)
278{
279 const char* key = model_key(kind);
280 std::vector<SettingChoice> choices = form().field(key)->choices;
281 const auto it = std::ranges::find(choices, name, &SettingChoice::value);
282 if (it == choices.end() || name == kNone) {
283 return;
284 }
285 choices.erase(it);
286 const bool updating = std::exchange(m_updating, true);
287 form().set_choices(key, std::move(choices));
288 m_updating = updating;
289}
