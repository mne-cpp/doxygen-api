112QVariant ChannelInfoModel::data(const QModelIndex& index, int role) const
113{
114 using Column = viz::ChannelTable::Column;
115
116 if (!index.isValid() || index.row() >= m_table.row_count() || index.column() >= viz::ChannelTable::column_count) {
117 return QVariant();
118 }
119
120 if (role == Qt::TextAlignmentRole) {
121 return QVariant(Qt::Alignment(Qt::AlignHCenter | Qt::AlignVCenter));
122 }
123
124 const int row = index.row();
125 const auto column = static_cast<Column>(index.column());
126 const FiffChInfo& ch = m_table.info()->chs.at(static_cast<std::size_t>(row));
127 const auto text = [&] {
128 return QVariant(QString::fromStdString(m_table.text(row, column)));
129 };
130
131 switch (column) {
132 case Column::Number:
133 return role == Qt::DisplayRole || role == ChannelInfoModelRoles::GetChNumber ? QVariant(row) : QVariant();
134 case Column::OriginalName:
135 return role == Qt::DisplayRole || role == ChannelInfoModelRoles::GetOrigChName ? text() : QVariant();
136 case Column::Alias:
137 return role == Qt::DisplayRole || role == ChannelInfoModelRoles::GetChAlias ? text() : QVariant();
138 case Column::MappedLayoutName:
139 return role == Qt::DisplayRole || role == ChannelInfoModelRoles::GetMappedLayoutChName ? text() : QVariant();
140 case Column::Kind:
141 if (role == ChannelInfoModelRoles::GetChKind) {
142 return QVariant(ch.kind);
143 }
144 return role == Qt::DisplayRole ? text() : QVariant();
145 case Column::MegType:
146 return role == Qt::DisplayRole || role == ChannelInfoModelRoles::GetMEGType ? text() : QVariant();
147 case Column::Unit:
148 if (role == ChannelInfoModelRoles::GetChUnit) {
149 return QVariant(ch.unit);
150 }
151 return role == Qt::DisplayRole ? text() : QVariant();
152 case Column::Position:
153 if (role == ChannelInfoModelRoles::GetChPosition) {
154 const std::array<double, 2> point = m_table.position(row);
155 return QVariant(QPointF(point[0], point[1]));
156 }
157 return role == Qt::DisplayRole ? text() : QVariant();
158 case Column::Digitizer:
159 if (role == ChannelInfoModelRoles::GetChDigitizer) {
160 const Eigen::Vector3f point = m_table.digitizer_cm(row);
161 return QVariant(QVector3D(point.x(), point.y(), point.z()));
162 }
163 return role == Qt::DisplayRole ? text() : QVariant();
164 case Column::ActiveFilter:
165 return QVariant();
166 case Column::CoilType:
167 if (role == ChannelInfoModelRoles::GetChCoilType) {
168 return QVariant(ch.chpos.coil_type);
169 }
170 return role == Qt::DisplayRole ? text() : QVariant();
171 case Column::Bad:
172 return role == Qt::DisplayRole || role == ChannelInfoModelRoles::GetIsBad ? QVariant(m_table.is_bad(row)) : QVariant();
173 case Column::Compensators:
174 return role == Qt::DisplayRole ? QVariant(static_cast<int>(m_table.info()->comps.size())) : QVariant();
175 }
176
177 return QVariant();
178}
