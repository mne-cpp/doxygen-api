216void AverageLayoutView::updateData()
217{
218 if (!m_pFiffInfo && (!m_pEvokedSetModel || !m_pChannelInfoModel)) {
219 mne::Log::debug("AverageLayoutView::updateData - m_pEvokedSetModel or m_pChannelInfoModel are NULL. Returning. ");
220 return;
221 }
222
223 // Column 2 of the evoked set model holds the averages, each a column-major channels x samples matrix
224 std::vector<AvrTypeRowVectorPair> averageData;
225 if (m_pEvokedSetModel) {
226 averageData = m_pEvokedSetModel->data(0, 2, EvokedSetModelRoles::GetAverageData).value<std::vector<AvrTypeRowVectorPair>>();
227 }
228
229 for (AverageLayoutWidget::Channel& channel : m_pAverageLayoutWidget->channels()) {
230 channel.averages.clear();
231 if (!m_pEvokedSetModel) {
232 continue;
233 }
234
235 // With a FiffInfo the selection carries the channel numbers, otherwise the names are looked up
236 if (!m_pFiffInfo) {
237 channel.number = m_pChannelInfoModel->getIndexFromMappedChName(channel.name);
238 }
239 const int channelNumber = channel.number;
240 if (channelNumber == -1) {
241 continue;
242 }
243
244 if (m_pFiffInfo) {
245 channel.kind = m_pFiffInfo->chs.at(channelNumber).kind;
246 channel.unit = m_pFiffInfo->chs.at(channelNumber).unit;
247 } else {
248 channel.kind = m_pChannelInfoModel->data(m_pChannelInfoModel->index(channelNumber, 4), ChannelInfoModelRoles::GetChKind).toInt();
249 channel.unit = m_pChannelInfoModel->data(m_pChannelInfoModel->index(channelNumber, 6), ChannelInfoModelRoles::GetChUnit).toInt();
250 }
251 channel.firstSample = -m_pEvokedSetModel->getNumPreStimSamples();
252 channel.bad = m_pEvokedSetModel->getIsChannelBad(channelNumber);
253 const int channelCount = m_pEvokedSetModel->rowCount();
254 for (const auto& [comment, average] : averageData) {
255 channel.averages.push_back({comment, average.first + channelNumber, average.second, channelCount});
256 }
257 }
258
259 m_pAverageLayoutWidget->update();
260}
