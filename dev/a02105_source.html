194MnaValues runScript(const MnaScript& script, const MnaValues& inputs, const JsonObject& attributes)
195{
196 const auto language = std::ranges::find(kLanguages, std::string_view(script.language), &Language::name);
197 const std::string_view extension = language != kLanguages.end() ? language->extension : ".txt";
198 std::string interpreter = script.interpreter;
199 if (interpreter.empty() && language != kLanguages.end()) {
200 interpreter = language->interpreter;
201 }
202
203 std::random_device random;
204 std::error_code ec;
205 const std::filesystem::path file =
206 std::filesystem::temp_directory_path(ec) / std::format("mna_script_{:08x}{:08x}{}", random(), random(), extension);
207 std::ofstream(file, std::ios::binary) << substitute(script.code, inputs, attributes);
208 if (!std::filesystem::exists(file, ec)) {
209 return {{"stderr", std::string("Failed to create temporary script file")}, {"exit_code", -1}};
210 }
211
212 StringList args = script.interpreter_args;
213 args.push_back(utf8_from_path(file));
214 MnaValues outputs = runProgram(interpreter, args);
215 if (!script.keep_temp_file) {
216 std::filesystem::remove(file, ec);
217 }
218 return outputs;
219}

227MnaGraphExecutor::Context MnaGraphExecutor::execute(MnaGraph& graph, const MnaValues& graph_inputs)
228{
229 Context context;
230 context.graph_inputs = graph_inputs;
231 for (const auto& [port, value] : graph_inputs) {
232 context.results.emplace("graph::" + port, value);
233 }
234
235 graph.param_tree.evaluate(scalars(context.results));
236 apply_param_tree(graph);
237
238 const StringList order = graph.topological_sort();
239 const int total = static_cast<int>(order.size());
240 for (int i = 0; i < total; ++i) {
241 const std::string& id = order[static_cast<std::size_t>(i)];
242 if (progressCallback()) {
243 progressCallback()(id, i + 1, total);
244 }
245 if (!run_node(graph, graph.node(id), context)) {
246 break;
247 }
248 }
249
250 graph.param_tree.evaluate(scalars(context.results)); // on_change bindings see the new results
251 return context;
252}

256void MnaGraphExecutor::execute_incremental(MnaGraph& graph, Context& context)
257{
258 std::set<std::string, std::less<>> pending;
259 for (const std::string& id : graph.dirty_nodes()) {
260 pending.insert(id);
261 for (std::string& downstream : graph.downstream_nodes(id)) {
262 pending.insert(std::move(downstream));
263 }
264 }
265
266 apply_param_tree(graph);
267
268 StringList order;
269 for (std::string& id : graph.topological_sort()) {
270 if (pending.contains(id)) {
271 order.push_back(std::move(id));
272 }
273 }
274 const int total = static_cast<int>(order.size());
275 for (int i = 0; i < total; ++i) {
276 const std::string& id = order[static_cast<std::size_t>(i)];
277 if (progressCallback()) {
278 progressCallback()(id, i + 1, total);
279 }
280 if (!run_node(graph, graph.node(id), context)) {
281 break;
282 }
283 }
284
285 graph.param_tree.evaluate(scalars(context.results));
286}

305bool MnaGraphExecutor::run_node(MnaGraph& graph, MnaNode& node, Context& context)
306{
307 MnaValues inputs;
308 for (const MnaPort& p : node.inputs) {
309 if (p.source_node_id.empty()) {
310 continue;
311 }
312 const auto it = context.results.find(p.source_node_id + "::" + p.source_port_name);
313 inputs[p.name] = it != context.results.end() ? it->second : std::any();
314 }
315 JsonObject scope = node.attributes; // checks see attributes and inputs by name, post checks the outputs too
316 for (auto& [key, value] : scalars(inputs)) {
317 scope.insert(key, value);
318 }
319
320 MnaVerification& verification = node.verification;
321 MnaProvenance& provenance = verification.provenance;
322 provenance = MnaProvenance();
323 provenance.mne_cpp_version = MNE_CPP_VERSION;
324 provenance.os_info = os_pretty_name() + ' ' + std::string(cpu_architecture());
325 provenance.host_name = host_name();
326 provenance.resolved_attributes = node.attributes;
327 for (const auto& [port, value] : inputs) {
328 provenance.input_hashes.emplace(port, to_hex(Sha256::hash(text(value))));
329 }
330
331 const auto failsHard = [](const std::vector<MnaVerificationResult>& results) {
332 return std::ranges::any_of(results, [](const MnaVerificationResult& r) {
333 return !r.passed && r.severity == "error";
334 });
335 };
336 verification.pre_results = run_checks(graph, node, "pre", scope);
337 verification.post_results.clear();
338 if (failsHard(verification.pre_results)) {
339 context.aborted_node = node.id;
340 return false;
341 }
342
343 provenance.started_at = now();
344 const auto start = std::chrono::steady_clock::now();
345 MnaValues outputs = execute_node(node, inputs);
346 provenance.wall_time_ms = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::steady_clock::now() - start).count();
347 provenance.finished_at = now();
348
349 for (auto& [key, value] : scalars(outputs)) {
350 scope.insert(key, value);
351 }
352 for (auto& [port, value] : outputs) {
353 context.results.insert_or_assign(node.id + "::" + port, std::move(value));
354 }
355 node.dirty = false;
356 node.executed_at = provenance.finished_at;
357
358 verification.post_results = run_checks(graph, node, "post", scope);
359 if (failsHard(verification.post_results)) {
360 context.aborted_node = node.id;
361 return false;
362 }
363 return true;
364}

368std::vector<MnaVerificationResult> MnaGraphExecutor::run_checks(const MnaGraph& graph, const MnaNode& node, std::string_view phase, const JsonObject& scope)
369{
370 std::vector<MnaVerificationResult> results;
371 for (const MnaVerificationCheck& check : node.verification.checks) {
372 if (check.phase != phase) {
373 continue;
374 }
375 MnaVerificationResult result;
376 result.check_id = check.id;
377 result.severity = check.severity;
378 result.evaluated_at = now();
379 if (!check.script.code.empty()) {
380 MnaValues values;
381 for (const auto& [key, value] : scope) {
382 values.emplace(key, value);
383 }
384 const MnaValues out = runScript(check.script, values, node.attributes);
385 const int exitCode = std::any_cast<int>(out.at("exit_code"));
386 result.actual_value = exitCode;
387 result.passed = exitCode == 0;
388 } else {
389 result.actual_value = graph.param_tree.evaluate_expression(check.expression, scope);
390 result.passed = result.actual_value.is_bool() && result.actual_value.to_bool();
391 }
392 result.message = result.passed ? std::format("PASS: {}", check.description) : std::format("FAIL [{}]: {}", check.severity, check.description);
393 if (!result.passed && !check.on_fail.empty()) {
394 result.message += std::format(" ({})", check.on_fail);
395 }
396 results.push_back(std::move(result));
397 }
398 return results;
399}

403MnaValues MnaGraphExecutor::execute_node(const MnaNode& node, const MnaValues& inputs)
404{
405 if (node.exec_mode == MnaNodeExecMode::Script) {
406 return runScript(node.script, inputs, node.attributes);
407 }
408 if (node.exec_mode == MnaNodeExecMode::Ipc) {
409 StringList args;
410 args.reserve(node.ipc_args.size());
411 for (const std::string& arg : node.ipc_args) {
412 args.push_back(substitute(arg, inputs, node.attributes));
413 }
414 MnaValues outputs = runProgram(node.ipc_command, args, node.ipc_work_dir);
415 for (const MnaPort& p : node.outputs) {
416 if (!p.cached_result_path.empty()) {
417 outputs[p.name] = p.cached_result_path;
418 }
419 }
420 return outputs;
421 }
422 if (const MnaOpRegistry::OpFunc& func = MnaOpRegistry::instance().op_func(node.op_type)) {
423 return func(inputs, node.attributes);
424 }
425 return {}; // no implementation registered
426}

437MnaGraphExecutor::StreamContext MnaGraphExecutor::start_stream(MnaGraph& graph, const PluginFactory& factory)
438{
439 StreamContext context;
440 context.graph = &graph;
441
442 if (const StringList errors = graph.validate(); !errors.empty()) {
443 Log::warning("MnaGraphExecutor::start_stream - graph validation failed: {}", join(errors, ", "));
444 return context;
445 }
446 context.execution_order = graph.topological_sort();
447 apply_param_tree(graph);
448
449 for (const std::string& id : context.execution_order) {
450 const MnaNode& n = graph.node(id);
451 std::shared_ptr<void> plugin = factory(n.op_type);
452 if (!plugin) {
453 Log::warning("MnaGraphExecutor::start_stream - factory returned nullptr for op type: {}", n.op_type);
454 context.live_plugins.clear();
455 return context;
456 }
457 context.live_plugins[id] = std::move(plugin);
458 }
459 context.running = true; // wiring the plugins together is up to the host application
460 return context;
461}

465void MnaGraphExecutor::stop_stream(StreamContext& context)
466{
467 if (!context.running) {
468 return;
469 }
470 context.running = false;
471 for (auto it = context.execution_order.rbegin(); it != context.execution_order.rend(); ++it) {
472 context.live_plugins.erase(*it);
473 }
474 context.live_plugins.clear();
475 context.execution_order.clear();
476 context.graph = nullptr;
