147void TriggerDetectionView::saveSettings()
148{
149 if (m_sSettingsPath.isEmpty()) {
150 return;
151 }
152
153 QSettings settings("MNECPP");
154
155 settings.setValue(m_sSettingsPath + QString("/TriggerDetectionView/Activated"), m_pUi->m_checkBox_activateTriggerDetection->isChecked());
156 settings.setValue(m_sSettingsPath + QString("/TriggerDetectionView/ChannelIndex"), m_pUi->m_comboBox_triggerChannels->currentIndex());
157 settings.setValue(m_sSettingsPath + QString("/TriggerDetectionView/FirstThresholdValue"), m_pUi->m_doubleSpinBox_detectionThresholdFirst->value());
158 settings.setValue(m_sSettingsPath + QString("/TriggerDetectionView/SecondThresholdValue"), m_pUi->m_spinBox_detectionThresholdSecond->value());
159
160 settings.beginGroup(m_sSettingsPath + QString("/TriggerDetectionView/Colors"));
161 QMap<double, QColor>::const_iterator i = m_qMapTriggerColor.constBegin();
162 while (i != m_qMapTriggerColor.constEnd()) {
163 settings.setValue(QString::number(i.key()), i.value());
164 ++i;
165 }
166 settings.endGroup();
167}

171void TriggerDetectionView::loadSettings()
172{
173 if (m_sSettingsPath.isEmpty()) {
174 return;
175 }
176
177 QSettings settings("MNECPP");
178
179 m_pUi->m_checkBox_activateTriggerDetection->setChecked(settings.value(m_sSettingsPath + QString("/TriggerDetectionView/Activated"), false).toBool());
180 m_pUi->m_comboBox_triggerChannels->setCurrentIndex(settings.value(m_sSettingsPath + QString("/TriggerDetectionView/ChannelIndex"), 0).toInt());
181 m_pUi->m_doubleSpinBox_detectionThresholdFirst->setValue(settings.value(m_sSettingsPath + QString("/TriggerDetectionView/FirstThresholdValue"), 0.1).toDouble());
182 m_pUi->m_spinBox_detectionThresholdSecond->setValue(settings.value(m_sSettingsPath + QString("/TriggerDetectionView/SecondThresholdValue"), -1).toInt());
183
184 settings.beginGroup(m_sSettingsPath + QString("/TriggerDetectionView/Colors"));
185 QStringList keys = settings.childKeys();
186 foreach (QString key, keys) {
187 double dKey = key.toDouble();
188 m_qMapTriggerColor[dKey] = settings.value(key).value<QColor>();
189 }
190 settings.endGroup();
191
192 onTriggerInfoChanged();
193}
