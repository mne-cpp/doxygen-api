47uint32_t withAlpha(uint32_t color, uint32_t alpha)
48{
49 return (color & 0x00FFFFFFu) | ((alpha & 0xFFu) << 24);
50}

52uint32_t curvatureGray(const std::vector<float>& curvature, int index)
53{
54 return (index >= 0 && index < std::ssize(curvature) && curvature[index] > 0.0f)
55 ? 0x40u
56 : 0xAAu;
57}
58
59// Overlay alpha is blended into RGB here because the shader keeps the curvature grey in the alpha channel.
60uint32_t overCortex(uint32_t overlay, uint32_t gray)
61{
62 const uint32_t a = overlay >> 24;
63 uint32_t rgb = 0;
64 for (int shift = 0; shift < 24; shift += 8) {
65 const uint32_t c = (overlay >> shift) & 0xFFu;
66 rgb |= ((c * a + gray * (255u - a) + 127u) / 255u) << shift;
67 }
68 return withAlpha(rgb, gray);
69}

135void BrainSurface::fromSurface(const mne::FsSurface& surf)
136{
137 const Eigen::MatrixXf& rr = surf.rr();
138 const Eigen::MatrixXf& nn = surf.nn();
139 const Eigen::MatrixXi& tris = surf.tris();
140 const Eigen::VectorXf& curv = surf.curv();
141 m_curvature.resize(curv.size());
142 for (int i = 0; i < curv.size(); ++i)
143 m_curvature[i] = curv[i];
144
145 std::vector<VertexData> vertices(rr.rows());
146 for (int i = 0; i < rr.rows(); ++i)
147 vertices[i] = {Eigen::Vector3f(rr(i, 0), rr(i, 1), rr(i, 2)), Eigen::Vector3f(nn(i, 0), nn(i, 1), nn(i, 2))};
148 std::vector<uint32_t> indices(tris.rows() * 3);
149 for (int i = 0; i < tris.rows(); ++i)
150 for (int j = 0; j < 3; ++j)
151 indices[3 * i + j] = tris(i, j);
152 setGeometry(std::move(vertices), std::move(indices), Color::white());
153}

234void BrainSurface::addAnnotation(const mne::FsAnnotation& annotation)
235{
236 m_annotation = annotation;
237 m_hasAnnotation = true;
238
239 // Per-vertex label ids and colours, so hover lookups and drawing need no search
240 const Eigen::VectorXi& vertices = m_annotation.getVertices();
241 const Eigen::VectorXi& labelIds = m_annotation.getLabelIds();
242 const mne::FsColortable& ct = m_annotation.getColortable();
243 std::unordered_map<int, uint32_t> labelColors;
244 for (int c = 0; c < ct.numEntries; ++c)
245 labelColors.insert_or_assign(ct.table(c, 4), packABGR(ct.table(c, 0), ct.table(c, 1), ct.table(c, 2)));
246 m_vertexLabelIds.assign(m_vertexData.size(), -1);
247 m_annotationColors.assign(m_vertexData.size(), 0u);
248 for (int i = 0; i < labelIds.rows(); ++i) {
249 const int v = vertices(i);
250 if (v < 0 || v >= std::ssize(m_vertexData))
251 continue;
252 m_vertexLabelIds[v] = labelIds(i);
253 const auto color = labelColors.find(labelIds(i));
254 m_annotationColors[v] = color == labelColors.end() ? 0u : color->second;
255 }
256 ++m_annotationGeneration;
257}

330void BrainSurface::updateVertexColors()
331{
332 // Cortex: curvature grey; other surfaces (BEM, sensors, ...): the base colour
333 if (!m_curvature.empty() && m_curvature.size() == m_colors.size()) {
334 for (int i = 0; i < std::ssize(m_colors); ++i) {
335 const uint32_t val = curvatureGray(m_curvature, i);
336 m_colors[i] = packABGR(val, val, val, val);
337 }
338 } else {
339 std::ranges::fill(m_colors, packABGR(m_baseColor.r, m_baseColor.g, m_baseColor.b, m_baseColor.a));
340 }
341
342 // Source-estimate colours stay in the colour stream: each view selects its overlay by uniform
343 const int n = static_cast<int>(std::min(m_stcColors.size(), m_colors.size()));
344 for (int i = 0; i < n; ++i) {
345 m_colors[i] = overCortex(m_stcColors[i], curvatureGray(m_curvature, i));
346 }
347 ++m_colorGeneration;
348}

380void BrainSurface::transform(const Eigen::Matrix4f& m)
381{
382 // Normals transform with the inverse transpose of the upper-left 3x3, so non-uniform scales keep them normal
383 const Eigen::Matrix3f normalMat = m.topLeftCorner<3, 3>().inverse().transpose();
384
385 for (auto& v : m_vertexData) {
386 const Eigen::Vector4f p = m * v.pos.homogeneous();
387 v.pos = p.head<3>() / p.w();
388 v.norm = (normalMat * v.norm).normalized();
389 }
390 markGeometryDirty();
391}

407std::vector<Eigen::VectorXi> BrainSurface::computeNeighbors() const
408{
409 // Use temporary std::vector<std::set<int>> for dedup during construction
410 std::vector<std::set<int>> tempNeighbors(m_vertexData.size());
411
412 // Triangles are stored in m_indexData as triplets
413 for (int i = 0; i + 2 < std::ssize(m_indexData); i += 3) {
414 int v0 = m_indexData[i];
415 int v1 = m_indexData[i + 1];
416 int v2 = m_indexData[i + 2];
417
418 // Add bidirectional edges (set handles dedup)
419 tempNeighbors[v0].insert(v1);
420 tempNeighbors[v0].insert(v2);
421 tempNeighbors[v1].insert(v0);
422 tempNeighbors[v1].insert(v2);
423 tempNeighbors[v2].insert(v0);
424 tempNeighbors[v2].insert(v1);
425 }
426
427 // Convert to std::vector<VectorXi>
428 std::vector<Eigen::VectorXi> neighbors(tempNeighbors.size());
429 for (size_t k = 0; k < tempNeighbors.size(); ++k) {
430 const auto& s = tempNeighbors[k];
431 neighbors[k].resize(static_cast<Eigen::Index>(s.size()));
432 Eigen::Index idx = 0;
433 for (int val : s) {
434 neighbors[k][idx++] = val;
435 }
436 }
437
438 return neighbors;
439}

457void BrainSurface::boundingBox(Eigen::Vector3f& min, Eigen::Vector3f& max) const
458{
459 if (!m_bAABBDirty) {
460 min = m_aabbMin;
461 max = m_aabbMax;
462 return;
463 }
464
465 if (m_vertexData.empty()) {
466 min = Eigen::Vector3f(0, 0, 0);
467 max = Eigen::Vector3f(0, 0, 0);
468 return;
469 }
470
471 min = m_vertexData[0].pos;
472 max = m_vertexData[0].pos;
473
474 for (const auto& v : m_vertexData) {
475 min = min.cwiseMin(v.pos);
476 max = max.cwiseMax(v.pos);
477 }
478
479 m_aabbMin = min;
480 m_aabbMax = max;
481 m_bAABBDirty = false;
482}

484struct BrainSurface::Bvh
485{
486 struct Node
487 {
488 float min[3];
489 float max[3];
490 int first; // first triangle (leaf) or left child (inner node)
491 int count; // triangles in a leaf, 0 for an inner node
492 int right; // right child of an inner node
493 };
494 std::vector<Node> nodes;
495 std::vector<int> triangles; // triangle indices, grouped by leaf
496 std::uint64_t geometryGeneration = 0;
497 std::uint64_t indexGeneration = 0;
498};

500namespace
501{
502constexpr int kBvhLeafSize = 4;
503// Pads node bounds so a ray grazing a flat, axis-aligned box still enters it
504constexpr float kBvhPad = 1e-6f;
505
506void growBox(float* lo, float* hi, const Eigen::Vector3f& p)
507{
508 for (int a = 0; a < 3; ++a) {
509 lo[a] = std::min(lo[a], p[a]);
510 hi[a] = std::max(hi[a], p[a]);
511 }
512}
513
514// Ray parameter where the ray enters the box, or +inf when it misses it before maxT
515float enterBox(const float* lo, const float* hi, const float* origin, const float* invDir, float maxT)
516{
517 float tNear = 0.0f;
518 float tFar = maxT;
519 for (int a = 0; a < 3; ++a) {
520 float t0 = (lo[a] - origin[a]) * invDir[a];
521 float t1 = (hi[a] - origin[a]) * invDir[a];
522 if (t0 > t1)
523 std::swap(t0, t1);
524 // NaN (origin on a slab plane of a ray parallel to it) keeps the previous bound
525 tNear = t0 > tNear ? t0 : tNear;
526 tFar = t1 < tFar ? t1 : tFar;
527 if (tNear > tFar)
528 return std::numeric_limits<float>::infinity();
529 }
530 return tNear;
531}
532} // namespace

536const BrainSurface::Bvh& BrainSurface::bvh() const
537{
538 if (m_bvh && m_bvh->geometryGeneration == m_geometryGeneration && m_bvh->indexGeneration == m_indexGeneration)
539 return *m_bvh;
540
541 auto tree = std::make_shared<Bvh>();
542 tree->geometryGeneration = m_geometryGeneration;
543 tree->indexGeneration = m_indexGeneration;
544 const int triangleCount = int(m_indexData.size() / 3);
545
546 // Triangle boxes in one contiguous array that the build partitions in place
547 struct Item
548 {
549 float lo[3];
550 int triangle;
551 float hi[3];
552 float pad;
553 };
554 std::vector<Item> items(static_cast<size_t>(triangleCount));
555 for (int t = 0; t < triangleCount; ++t) {
556 Item& item = items[size_t(t)];
557 item.triangle = t;
558 for (int a = 0; a < 3; ++a) {
559 item.lo[a] = std::numeric_limits<float>::max();
560 item.hi[a] = std::numeric_limits<float>::lowest();
561 }
562 for (int k = 0; k < 3; ++k)
563 growBox(item.lo, item.hi, m_vertexData[m_indexData[3 * t + k]].pos);
564 }
565 tree->nodes.reserve(size_t(std::max(1, 2 * triangleCount / kBvhLeafSize + 1)));
566
567 // Top-down build: split each node at the median box centre along its widest centre axis
568 struct Task
569 {
570 int node;
571 int begin;
572 int end;
573 };
574 std::vector<Task> tasks;
575 tree->nodes.push_back({});
576 tasks.push_back({0, 0, triangleCount});
577 while (!tasks.empty()) {
578 const Task task = tasks.back();
579 tasks.pop_back();
580 Bvh::Node node{};
581 float cLo[3];
582 float cHi[3];
583 for (int a = 0; a < 3; ++a) {
584 node.min[a] = cLo[a] = std::numeric_limits<float>::max();
585 node.max[a] = cHi[a] = std::numeric_limits<float>::lowest();
586 }
587 for (int i = task.begin; i < task.end; ++i) {
588 const Item& item = items[size_t(i)];
589 for (int a = 0; a < 3; ++a) {
590 node.min[a] = std::min(node.min[a], item.lo[a]);
591 node.max[a] = std::max(node.max[a], item.hi[a]);
592 const float c = item.lo[a] + item.hi[a];
593 cLo[a] = std::min(cLo[a], c);
594 cHi[a] = std::max(cHi[a], c);
595 }
596 }
597 for (int a = 0; a < 3; ++a) {
598 node.min[a] -= kBvhPad;
599 node.max[a] += kBvhPad;
600 }
601
602 const int axis = (cHi[1] - cLo[1] > cHi[0] - cLo[0]) ? ((cHi[2] - cLo[2] > cHi[1] - cLo[1]) ? 2 : 1)
603 : ((cHi[2] - cLo[2] > cHi[0] - cLo[0]) ? 2 : 0);
604 if (task.end - task.begin <= kBvhLeafSize || cHi[axis] <= cLo[axis]) {
605 node.first = task.begin;
606 node.count = task.end - task.begin;
607 tree->nodes[size_t(task.node)] = node;
608 continue;
609 }
610 const int mid = (task.begin + task.end) / 2;
611 std::nth_element(items.begin() + task.begin, items.begin() + mid, items.begin() + task.end,
612 [axis](const Item& l, const Item& r) { return l.lo[axis] + l.hi[axis] < r.lo[axis] + r.hi[axis]; });
613 const int left = int(tree->nodes.size());
614 const int right = left + 1;
615 tree->nodes.resize(tree->nodes.size() + 2);
616 node.first = left;
617 node.count = 0;
618 node.right = right;
619 tree->nodes[size_t(task.node)] = node;
620 tasks.push_back({right, mid, task.end});
621 tasks.push_back({left, task.begin, mid});
622 }
623
624 tree->triangles.resize(size_t(triangleCount));
625 for (int i = 0; i < triangleCount; ++i)
626 tree->triangles[size_t(i)] = items[size_t(i)].triangle;
627
628 m_bvh = std::move(tree);
629 return *m_bvh;
630}

634bool BrainSurface::intersects(const Eigen::Vector3f& rayOrigin, const Eigen::Vector3f& rayDir, float& dist, int& vertexIdx) const
635{
636 vertexIdx = -1;
637 if (m_vertexData.empty() || m_indexData.size() < 3)
638 return false;
639
640 const Bvh& tree = bvh();
641 const float origin[3] = {rayOrigin.x(), rayOrigin.y(), rayOrigin.z()};
642 const float invDir[3] = {1.0f / rayDir.x(), 1.0f / rayDir.y(), 1.0f / rayDir.z()};
643 const viz::Ray ray{{rayOrigin.x(), rayOrigin.y(), rayOrigin.z()}, {rayDir.x(), rayDir.y(), rayDir.z()}};
644
645 double closestDist = std::numeric_limits<double>::max();
646 int closestTriangle = -1;
647
648 // Möller–Trumbore (viz::intersect_ray_triangle) on the triangles of the leaves the ray enters, nearest
649 // first. Median splits keep the depth near log2(triangles / leaf size), far below the stack size.
650 int stack[64];
651 int top = 0;
652 stack[top++] = 0;
653 while (top > 0) {
654 const Bvh::Node& node = tree.nodes[size_t(stack[--top])];
655 const float limit = closestDist < double(std::numeric_limits<float>::max()) ? float(closestDist) : std::numeric_limits<float>::max();
656 if (enterBox(node.min, node.max, origin, invDir, limit) == std::numeric_limits<float>::infinity())
657 continue;
658 if (node.count == 0) {
659 const Bvh::Node& l = tree.nodes[size_t(node.first)];
660 const Bvh::Node& r = tree.nodes[size_t(node.right)];
661 const float tl = enterBox(l.min, l.max, origin, invDir, limit);
662 const float tr = enterBox(r.min, r.max, origin, invDir, limit);
663 constexpr float miss = std::numeric_limits<float>::infinity();
664 // Push the farther child first so the nearer one is visited next
665 const bool leftFirst = tl <= tr;
666 const int nearChild = leftFirst ? node.first : node.right;
667 const int farChild = leftFirst ? node.right : node.first;
668 if ((leftFirst ? tr : tl) != miss)
669 stack[top++] = farChild;
670 if ((leftFirst ? tl : tr) != miss)
671 stack[top++] = nearChild;
672 continue;
673 }
674
675 for (int i = node.first; i < node.first + node.count; ++i) {
676 const int t = tree.triangles[size_t(i)];
677 const Eigen::Vector3f& v0 = m_vertexData[m_indexData[3 * t]].pos;
678 const Eigen::Vector3f& v1 = m_vertexData[m_indexData[3 * t + 1]].pos;
679 const Eigen::Vector3f& v2 = m_vertexData[m_indexData[3 * t + 2]].pos;
680 const std::optional<double> hitT = viz::intersect_ray_triangle(ray, {v0.x(), v0.y(), v0.z()}, {v1.x(), v1.y(), v1.z()},
681 {v2.x(), v2.y(), v2.z()});
682 if (hitT && *hitT < closestDist) {
683 closestDist = *hitT;
684 closestTriangle = t;
685 }
686 }
687 }
688
689 if (closestTriangle < 0)
690 return false;
691
692 // The vertex of the hit triangle nearest to the hit point, used for region lookup
693 const Eigen::Vector3f hit = rayOrigin + static_cast<float>(closestDist) * rayDir;
694 float best = std::numeric_limits<float>::max();
695 for (int k = 0; k < 3; ++k) {
696 const int v = int(m_indexData[3 * closestTriangle + k]);
697 const float d = (m_vertexData[v].pos - hit).squaredNorm();
698 if (d < best) {
699 best = d;
700 vertexIdx = v;
701 }
702 }
703 dist = float(closestDist);
704 return true;
705}

709std::string BrainSurface::getAnnotationLabel(int vertexIdx) const
710{
711 if (!m_hasAnnotation || vertexIdx < 0 || vertexIdx >= std::ssize(m_vertexLabelIds)) {
712 return "";
713 }
714 const int labelId = m_vertexLabelIds[vertexIdx];
715 if (labelId == -1)
716 return "Unknown";
717
718 const mne::FsColortable& ct = m_annotation.getColortable();
719 for (int i = 0; i < ct.numEntries; ++i) {
720 if (ct.table(i, 4) == labelId) {
721 std::string name = ct.struct_names[static_cast<std::size_t>(i)];
722 // Remove null characters and trailing whitespace that might cause "strange signs"
723 while (!name.empty() && (name.back() == '\0' || name.back() == ' ')) {
724 name.pop_back();
725 }
726 return name;
727 }
728 }
729
730 return "Unknown";
731}
