66static QVector<int> findGoodEegIndices(const FiffInfo& info)
67{
68 QVector<int> indices;
69 const QSet<QString> badSet(info.bads.begin(), info.bads.end());
70 for (int i = 0; i < info.nchan; ++i) {
71 if (info.chs[i].kind == FIFFV_EEG_CH && !badSet.contains(info.ch_names[i])) {
72 indices.append(i);

74 }

97void UTILSLIB::setEegReference(MatrixXd& data,
98 const FiffInfo& info,
99 const QStringList& refChannels,
100 bool projection)
101{
102 if (projection) {
103 qWarning("setEegReference: projection mode is not yet implemented. "
104 "Falling back to direct data modification.");
105 }
106
107 if (data.rows() != info.nchan) {
108 qWarning("setEegReference: data row count (%lld) does not match info.nchan (%d).",
109 static_cast<long long>(data.rows()), info.nchan);
110 return;
111 }
112
113 if (data.cols() == 0 || data.rows() == 0) {
114 return;
115 }
116
117 // Determine which EEG channels to subtract from
118 const QVector<int> allEegIdx = findAllEegIndices(info);
119 if (allEegIdx.isEmpty()) {
120 qWarning("setEegReference: no EEG channels found in info.");
121 return;
122 }
123
124 // Compute reference signal
125 RowVectorXd refSignal;
126
127 const bool useAverage = refChannels.isEmpty()
128 || (refChannels.size() == 1
129 && refChannels.first().compare(QLatin1String("average"), Qt::CaseInsensitive) == 0);
130
131 if (useAverage) {
132 // Average reference: mean of all good EEG channels
133 const QVector<int> goodEegIdx = findGoodEegIndices(info);
134 if (goodEegIdx.isEmpty()) {
135 qWarning("setEegReference: all EEG channels are marked as bad.");
136 return;
137 }
138 refSignal = RowVectorXd::Zero(data.cols());
139 for (int idx : goodEegIdx) {
140 refSignal += data.row(idx);
141 }
142 refSignal /= static_cast<double>(goodEegIdx.size());
143 } else {
144 // Specific channel(s): mean of named channels
145 QVector<int> refIdx;
146 for (const QString& name : refChannels) {
147 int idx = info.ch_names.indexOf(name);
148 if (idx < 0) {
149 qWarning("setEegReference: reference channel '%s' not found.", qPrintable(name));
150 return;
151 }
152 refIdx.append(idx);
153 }
154 refSignal = RowVectorXd::Zero(data.cols());
155 for (int idx : refIdx) {
156 refSignal += data.row(idx);
157 }
158 refSignal /= static_cast<double>(refIdx.size());
159 }
160
161 // Subtract reference from all EEG channels
162 for (int idx : allEegIdx) {
163 data.row(idx) -= refSignal;
164 }
165}

169void UTILSLIB::addReferenceChannels(MatrixXd& data,
170 FiffInfo& info,
171 const QStringList& chNames)
172{
173 if (chNames.isEmpty()) {
174 return;
175 }
176
177 const Eigen::Index nTimes = data.cols();
178 const Eigen::Index nOldCh = data.rows();
179 const int nNew = chNames.size();
180
181 // Expand data matrix with zero rows
182 MatrixXd newData(nOldCh + nNew, nTimes);
183 if (nOldCh > 0 && nTimes > 0) {
184 newData.topRows(nOldCh) = data;
185 }
186 newData.bottomRows(nNew).setZero();
187 data = newData;
188
189 // Find a template EEG channel for defaults
190 FiffChInfo templateCh;
191 bool haveTemplate = false;
192 for (int i = 0; i < info.nchan; ++i) {
193 if (info.chs[i].kind == FIFFV_EEG_CH) {
194 templateCh = info.chs[i];
195 haveTemplate = true;
196 break;
197 }
198 }
199
200 for (int n = 0; n < nNew; ++n) {
201 FiffChInfo ch;
202 if (haveTemplate) {
203 ch = templateCh;
204 }
205 ch.ch_name = chNames[n];
206 ch.kind = FIFFV_EEG_CH;
207 ch.scanNo = info.nchan + n + 1;
208 ch.logNo = ch.scanNo;
209
210 info.chs.append(ch);
211 info.ch_names.append(chNames[n]);
212 }
213 info.nchan += nNew;
214}

218void UTILSLIB::setBipolarReference(MatrixXd& data,
219 FiffInfo& info,
220 const QStringList& anodes,
221 const QStringList& cathodes,
222 bool dropOriginals)
223{
224 if (anodes.size() != cathodes.size()) {
225 qWarning("setBipolarReference: anodes and cathodes must have the same length "
226 "(%lld vs %lld).", static_cast<long long>(anodes.size()), static_cast<long long>(cathodes.size()));
227 return;
228 }
229
230 if (anodes.isEmpty()) {
231 return;
232 }
233
234 if (data.rows() != info.nchan) {
235 qWarning("setBipolarReference: data row count (%lld) does not match info.nchan (%d).",
236 static_cast<long long>(data.rows()), info.nchan);
237 return;
238 }
239
240 const int nPairs = anodes.size();
241 const Eigen::Index nTimes = data.cols();
242
243 // Resolve anode and cathode indices
244 QVector<int> anodeIdx(nPairs), cathodeIdx(nPairs);
245 for (int i = 0; i < nPairs; ++i) {
246 anodeIdx[i] = info.ch_names.indexOf(anodes[i]);
247 if (anodeIdx[i] < 0) {
248 qWarning("setBipolarReference: anode channel '%s' not found.", qPrintable(anodes[i]));
249 return;
250 }
251 cathodeIdx[i] = info.ch_names.indexOf(cathodes[i]);
252 if (cathodeIdx[i] < 0) {
253 qWarning("setBipolarReference: cathode channel '%s' not found.", qPrintable(cathodes[i]));
254 return;
255 }
256 }
257
258 // Collect the set of original channels involved
259 QSet<int> bipolarOriginals;
260 for (int i = 0; i < nPairs; ++i) {
261 bipolarOriginals.insert(anodeIdx[i]);
262 bipolarOriginals.insert(cathodeIdx[i]);
263 }
264
265 // Build bipolar data rows
266 MatrixXd bipolarData(nPairs, nTimes);
267 for (int i = 0; i < nPairs; ++i) {
268 bipolarData.row(i) = data.row(anodeIdx[i]) - data.row(cathodeIdx[i]);
269 }
270
271 // Build bipolar channel info
272 QList<FiffChInfo> bipolarChs;
273 QStringList bipolarNames;
274 for (int i = 0; i < nPairs; ++i) {
275 FiffChInfo ch = info.chs[anodeIdx[i]];
276 ch.ch_name = anodes[i] + "-" + cathodes[i];
277 ch.logNo = i + 1;
278 bipolarChs.append(ch);
279 bipolarNames.append(ch.ch_name);
280 }
281
282 if (dropOriginals) {
283 // Keep only non-bipolar-original rows + append bipolar rows
284 QVector<int> keepIdx;
285 for (int i = 0; i < static_cast<int>(data.rows()); ++i) {
286 if (!bipolarOriginals.contains(i)) {
287 keepIdx.append(i);
288 }
289 }
290
291 const int nKeep = keepIdx.size();
292 MatrixXd newData(nKeep + nPairs, nTimes);
293 QList<FiffChInfo> newChs;
294 QStringList newNames;
295
296 for (int i = 0; i < nKeep; ++i) {
297 newData.row(i) = data.row(keepIdx[i]);
298 newChs.append(info.chs[keepIdx[i]]);
299 newNames.append(info.ch_names[keepIdx[i]]);
300 }
301 for (int i = 0; i < nPairs; ++i) {
302 newData.row(nKeep + i) = bipolarData.row(i);
303 newChs.append(bipolarChs[i]);
304 newNames.append(bipolarNames[i]);
305 }
306
307 // Update scan numbers
308 for (int i = 0; i < newChs.size(); ++i) {
309 newChs[i].scanNo = i + 1;
310 }
311
312 data = newData;
313 info.chs = newChs;
314 info.ch_names = newNames;
315 info.nchan = static_cast<int>(newData.rows());
316 } else {
317 // Keep all original rows, append bipolar rows at the end
318 MatrixXd newData(data.rows() + nPairs, nTimes);
319 newData.topRows(data.rows()) = data;
320 newData.bottomRows(nPairs) = bipolarData;
321
322 for (int i = 0; i < nPairs; ++i) {
323 bipolarChs[i].scanNo = info.nchan + i + 1;
324 }
325
326 info.chs.append(bipolarChs);
327 info.ch_names.append(bipolarNames);
328 info.nchan += nPairs;
329 data = newData;
330 }
331}
