105ArtifactSettingsView::ArtifactSettingsView(const QString& sSettingsPath,
106 const std::vector<FiffChInfo>& fiffChInfoList,
107 QWidget* parent)
108: AbstractView(parent)
109, m_pForm(std::make_shared<viz::SettingsForm>("Artifact Rejection Settings"))
110, m_fiffChInfoList(fiffChInfoList)
111{
112 m_sSettingsPath = sSettingsPath;
113 qRegisterMetaType<std::map<std::string, double>>("std::map<std::string, double>");
114
115 this->setWindowTitle("Artifact Rejection Settings");
116 this->setMinimumWidth(330);
117 this->setMaximumWidth(330);
118
119 // One row per channel type, shown while channels of the type are present
120 m_pForm->add_flag("hasChannelTypes");
121 viz::SettingField active;
122 active.key = "doArtifactThresholdReduction";
123 active.kind = viz::SettingField::Kind::Bool;
124 active.label = "Activate artifact rejection";
125 active.default_value = false;
126 active.visible_if = "hasChannelTypes";
127 m_pForm->add(active);
128 for (const std::string type : kChannelTypes) {
129 m_pForm->add_flag(presentKey(type));
130
131 viz::SettingField mantissa;
132 mantissa.key = mantissaKey(type);
133 mantissa.kind = viz::SettingField::Kind::Double;
134 mantissa.label = type;
135 mantissa.default_value = 1.0;
136 mantissa.maximum = 100000.0;
137 mantissa.prefix = "+/-";
138 mantissa.visible_if = presentKey(type);
139 m_pForm->add(mantissa);
140
141 viz::SettingField exponent;
142 exponent.key = exponentKey(type);
143 exponent.kind = viz::SettingField::Kind::Int;
144 exponent.default_value = -1;
145 exponent.minimum = -10000;
146 exponent.maximum = 0;
147 exponent.decimals = 0;
148 exponent.prefix = "e";
149 exponent.visible_if = presentKey(type);
150 m_pForm->add(exponent);
151 }
152
153 m_listener = m_pForm->add_listener([this](const std::string& key, const std::optional<viz::SettingValue>&) {
154 if (!m_bSilent && (key == "doArtifactThresholdReduction" || key.starts_with("artifactThresholds"))) {
155 onChangeArtifactThreshold();
156 }
157 });
158
159 auto* layout = new QVBoxLayout(this);
160 layout->setContentsMargins(0, 0, 0, 0);
161 m_pPanel = new SettingsPanel(m_pForm, this);
162 layout->addWidget(m_pPanel);
163
164 loadSettings();
165 redrawGUI();
166 updateThresholdMap();
167}

260void ArtifactSettingsView::redrawGUI()
261{
262 std::vector<std::string> present;
263 present.reserve(kChannelTypes.size());
264 for (const FiffChInfo& channel : m_fiffChInfoList) {
265 if (const char* type = channelType(channel)) {
266 present.emplace_back(type);
267 }
268 }
269
270 const detail::ScopedFlag silent(m_bSilent);
271 for (const std::string type : kChannelTypes) {
272 m_pForm->set_value(presentKey(type), std::ranges::contains(present, type));
273 }
274 m_pForm->set_value("hasChannelTypes", !present.empty());
275}

279void ArtifactSettingsView::updateThresholdMap()
280{
281 m_mapThresholds.clear();
282 m_mapThresholds["Active"] = getDoArtifactThresholdRejection() ? 1.0 : 0.0;
283 for (const std::string type : kChannelTypes) {
284 if (m_pForm->get<bool>(presentKey(type))) {
285 m_mapThresholds[type] = m_pForm->get<double>(mantissaKey(type)) * std::pow(10, m_pForm->get<int>(exponentKey(type)));
286 }
287 }
288}
