108AcqError AcqDevice::start()
109{
110 if (!m_pImpl) {
111 return AcqErrorCode::InvalidState;
112 }
113 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
114 const AcqDeviceState state = m_pImpl->effectiveState();
115 if (state != AcqDeviceState::Opened && state != AcqDeviceState::Stopped) {
116 return AcqErrorCode::InvalidState;
117 }
118 AcqError error = m_pImpl->backend->start();
119 if (!error.ok()) {
120 m_pImpl->lastError = error;
121 return error;
122 }
123 m_pImpl->backendRunning = true;
124 m_pImpl->state = AcqDeviceState::Running;
125 return {};
126}

160void AcqDevice::close()
161{
162 if (!m_pImpl) {
163 return;
164 }
165 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
166 if (!m_pImpl->backend) {
167 return;
168 }
169 m_pImpl->stopBackend();
170 m_pImpl->lastError = m_pImpl->error();
171 m_pImpl->backend->close();
172 // Destroying the backend finishes its input streams: readers drain what was committed, then see EndOfStream.
173 m_pImpl->backend.reset();
174 m_pImpl->state = AcqDeviceState::Closed;
175}
