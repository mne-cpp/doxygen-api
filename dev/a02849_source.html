156 template<Swappable T>
157 bool write_big_endian(const T* values, std::size_t count)
158 {
159 if constexpr (std::endian::native == std::endian::big) {
160 return write(values, count * sizeof(T)) == count * sizeof(T);
161 } else {
162 constexpr std::size_t kChunk = 4096;
163 T swapped[kChunk];
164 for (std::size_t done = 0; done < count;) {
165 const std::size_t n = count - done < kChunk ? count - done : kChunk;
166 for (std::size_t i = 0; i < n; ++i) {
167 swapped[i] = byte_swap(values[done + i]);
168 }
169 if (write(swapped, n * sizeof(T)) != n * sizeof(T)) {
170 return false;
171 }
172 done += n;
173 }
174 return true;
175 }
176 }
