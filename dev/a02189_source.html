65using Adjacency = std::map<std::string, std::set<std::string>, std::less<>>;

67Adjacency forwardEdges(const std::vector<MnaNode>& nodes)
68{
69 Adjacency adj;
70 for (const MnaNode& n : nodes) {
71 adj[n.id];
72 }
73 for (const MnaNode& n : nodes) {
74 for (const MnaPort& p : n.inputs) {
75 if (const auto it = adj.find(p.source_node_id); !p.source_node_id.empty() && it != adj.end()) {
76 it->second.insert(n.id);
77 }
78 }
79 }
80 return adj;
81}

84StringList kahn(const Adjacency& adj)
85{
86 std::map<std::string, int, std::less<>> inDegree;
87 for (const auto& [id, next] : adj) {
88 inDegree[id];
89 for (const std::string& n : next) {
90 ++inDegree[n];
91 }
92 }
93 std::deque<std::string> queue;
94 for (const auto& [id, degree] : inDegree) {
95 if (degree == 0) {
96 queue.push_back(id);
97 }
98 }
99 StringList sorted;
100 while (!queue.empty()) {
101 std::string current = std::move(queue.front());
102 queue.pop_front();
103 for (const std::string& n : adj.find(current)->second) {
104 if (--inDegree[n] == 0) {
105 queue.push_back(n);
106 }
107 }
108 sorted.push_back(std::move(current));
109 }
110 return sorted;
111}

114template<class Neighbours>
115StringList walk(std::string_view start, Neighbours neighbours)
116{
117 StringList order;
118 std::set<std::string, std::less<>> visited;
119 std::deque<std::string> queue;
120 const auto visit = [&](const std::string& id) {
121 if (!id.empty() && visited.insert(id).second) {
122 queue.push_back(id);
123 }
124 };
125 neighbours(start, visit);
126 while (!queue.empty()) {
127 std::string current = std::move(queue.front());
128 queue.pop_front();
129 neighbours(current, visit);
130 order.push_back(std::move(current));
131 }
132 return order;
133}

214bool MnaGraph::connect(std::string_view src_node_id, std::string_view src_port_name, std::string_view dst_node_id, std::string_view dst_port_name)
215{
216 const MnaNode* src = find_node(src_node_id);
217 MnaNode* dst = find_node(dst_node_id);
218 if (!src || !dst || std::ranges::find(src->outputs, src_port_name, &MnaPort::name) == src->outputs.end()) {
219 return false;
220 }
221 const auto port = std::ranges::find(dst->inputs, dst_port_name, &MnaPort::name);
222 if (port == dst->inputs.end()) {
223 return false;
224 }
225 port->source_node_id = src_node_id;
226 port->source_port_name = src_port_name;
227 return true;
228}

232StringList MnaGraph::validate() const
233{
234 StringList errors;
235
236 for (const MnaNode& n : m_nodes) {
237 for (const MnaPort& p : n.inputs) {
238 if (!p.source_node_id.empty() && !has_node(p.source_node_id)) {
239 errors.push_back(std::format("Node '{}' input port '{}' references unknown source node '{}'", n.id, p.name, p.source_node_id));
240 }
241 }
242 }
243 if (kahn(forwardEdges(m_nodes)).size() != m_nodes.size()) {
244 errors.emplace_back("Graph contains a cycle");
245 }
246
247 const MnaOpRegistry& registry = MnaOpRegistry::instance();
248 for (const MnaNode& n : m_nodes) {
249 if (!registry.has_op(n.op_type)) {
250 errors.push_back(std::format("Node '{}' has unregistered op type '{}'", n.id, n.op_type));
251 continue;
252 }
253 const MnaOpSchema& schema = registry.schema(n.op_type);
254 for (const std::string& e : schema.validate(n)) {
255 errors.push_back(std::format("Node '{}': {}", n.id, e));
256 }
257 for (const MnaOpSchemaPort& sp : schema.input_ports) {
258 if (!sp.required) {
259 continue;
260 }
261 for (const MnaPort& np : n.inputs) {
262 if (np.name == sp.name && np.source_node_id.empty()) {
263 errors.push_back(std::format("Node '{}': required input port '{}' is not connected", n.id, sp.name));
264 }
265 }
266 }
267 }
268
269 for (const MnaNode& n : m_nodes) {
270 for (const MnaPort& in : n.inputs) {
271 const MnaNode* src = in.source_port_name.empty() ? nullptr : find_node(in.source_node_id);
272 if (!src) {
273 continue;
274 }
275 for (const MnaPort& out : src->outputs) {
276 if (out.name == in.source_port_name && in.data_kind != MnaDataKind::Custom && out.data_kind != MnaDataKind::Custom &&
277 in.data_kind != out.data_kind) {
278 errors.push_back(std::format("Edge {}.{} -> {}.{}: data kind mismatch ({} != {})", src->id, out.name, n.id, in.name,
279 static_cast<int>(out.data_kind), static_cast<int>(in.data_kind)));
280 }
281 }
282 }
283 }
284 return errors;
285}
