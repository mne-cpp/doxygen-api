101void AverageSceneItem::paint(QPainter* painter, const QStyleOptionGraphicsItem* option, QWidget* widget)
102{
103 Q_UNUSED(option);
104 Q_UNUSED(widget);
105
106 scene::AverageTile tile;
107 tile.box = {m_rectBoundingRect.x(), m_rectBoundingRect.y(), m_rectBoundingRect.width(), m_rectBoundingRect.height()};
108 tile.name = m_sChannelName.toStdString();
109 tile.name_size = m_iFontTextSize;
110 tile.bad = m_bIsBad;
111 tile.max_value = mne::disp::getScalingValue(m_scaleMap, m_iChannelKind, m_iChannelUnit);
112 tile.stim_sample = m_firstLastSample.first;
113 tile.columns = m_lAverageData.empty() ? 0 : m_lAverageData.front().second.second;
114 for (const auto& [comment, data] : m_lAverageData) {
115 // Default to active when no activation map has been configured
116 if (m_qMapAverageActivation.contains(comment) && !m_qMapAverageActivation.at(comment)) {
117 continue;
118 }
119 scene::AverageTile::Trace trace;
120 trace.color = toSceneColor(m_qMapAverageColor.contains(comment) ? m_qMapAverageColor.at(comment) : m_colorDefault);
121 // The evoked matrix is stored column by column
122 for (int i = 0; i < data.second; ++i) {
123 trace.samples.push_back(*(data.first + (i * m_iTotalNumberChannels) + m_iChannelNumber));
124 }
125 tile.traces.push_back(std::move(trace));
126 }
127
128 scene::Canvas canvas(m_rectBoundingRect.width(), m_rectBoundingRect.height());
129 tile.draw(canvas, toSceneFont(painter->font()));
130 paintScene(*painter, canvas.display_list(), painter->font());
131}
