118FiffEvoked FiffEvoked::pick_channels(const StringList& include,
119 const StringList& exclude) const
120{
121 if (include.size() == 0 && exclude.size() == 0)
122 return FiffEvoked(*this);
123
124 RowVectorXi sel = FiffInfo::pick_channels(this->info.ch_names, include, exclude);
125 if (sel.cols() == 0) {
126 Log::warning("Warning : No channels match the selection.\n");
127 return FiffEvoked(*this);
128 }
129
130 FiffEvoked res(*this);
131 //
132 // Modify the measurement info
133 //
134 res.info = FiffInfo(res.info.pick_info(sel));
135 //
136 // Create the reduced data set
137 //
138 MatrixXd selBlock = MatrixXd::Zero(sel.cols(), res.data.cols());
139 for (std::int32_t l = 0; l < sel.cols(); ++l) {
140 if (sel(0, l) < res.data.rows()) {
141 selBlock.block(l, 0, 1, selBlock.cols()) = res.data.block(sel(0, l), 0, 1, selBlock.cols());
142 } else {
143 Log::warning("FiffEvoked::pick_channels - Warning : Selected channel index out of bound.\n");
144 }
145 }
146 res.data.resize(sel.cols(), res.data.cols());
147 res.data = selBlock;
148
149 return res;
150}

173Expected<FiffEvokedFile> FiffEvokedFile::open(const std::filesystem::path& path)
174{
175 auto opened = FiffStream::open(path);
176 if (!opened) {
177 return std::unexpected(opened.error());
178 }
179 FiffEvokedFile file(std::move(*opened), utf8_from_path(path));
180 FiffStream& stream = file.m_stream;
181 Log::info("Reading {} ...\n", file.m_file);
182
183 std::shared_ptr<FiffDirNode> meas;
184 if (stream.dirtree()->dir_tree_find(FIFFB_MEAS).empty()) {
185 return fail(ErrorCode::NotFound, "No measurement in {}", file.m_file);
186 }
187 if (!stream.read_meas_info(stream.dirtree(), file.info, meas)) {
188 return fail(ErrorCode::Format, "No measurement info in {}", file.m_file);
189 }
190 file.info.filename = stream.stream_name();
191 if (meas->dir_tree_find(FIFFB_PROCESSED_DATA).empty()) {
192 return fail(ErrorCode::NotFound, "No processed data in {}", file.m_file);
193 }
194 file.m_sets = meas->dir_tree_find(FIFFB_EVOKED);
195 if (file.m_sets.empty()) {
196 return fail(ErrorCode::NotFound, "No evoked data in {}", file.m_file);
197 }
198 std::string listing;
199 stream.get_evoked_entries(file.m_sets, file.m_comments, file.m_kinds, listing);
200 return file;
201}

205Expected<int> FiffEvokedFile::find(const FiffEvoked::Condition& condition, fiff_int_t kind) const
206{
207 if (const int* index = std::get_if<int>(&condition)) {
208 if (*index >= 0 && *index < std::ssize(m_sets)) {

212 const std::string& wanted = std::get<std::string>(condition);
213 for (std::size_t i = 0; i < m_comments.size(); ++i) {

215 return static_cast<int>(i);
216 }

218 }
219 return fail(ErrorCode::NotFound, "No evoked data set {} of kind {} in {}",
220 std::visit([](const auto& c) { return std::format("{}", c); }, condition), kind, m_file);

225Expected<FiffEvoked> FiffEvokedFile::read(int set, const FiffEvoked::Baseline& baseline, bool proj)
226{
227 FiffInfo evokedInfo = this->info;
228 const std::string& file = m_file;
229 FiffEvoked p_FiffEvoked;
230 std::shared_ptr<FiffDirNode> my_evoked = m_sets[set];
231
232 //
233 // Identify the aspects
234 //
235 std::vector<std::shared_ptr<FiffDirNode>> aspects = my_evoked->dir_tree_find(FIFFB_ASPECT);
236 if (aspects.empty()) {
237 return fail(ErrorCode::Format, "Evoked data set {} without data in {}", set, file);
238 }
239
240 if (aspects.size() > 1)
241 Log::info("\tMultiple ({}) aspects found. Taking first one.\n", aspects.size());
242
243 const std::shared_ptr<FiffDirNode>& my_aspect = aspects[0];
244
245 //
246 // Now find the data in the evoked block
247 //
248 fiff_int_t nchan = 0;
249 float sfreq = -1.0f;
250 std::vector<FiffChInfo> chs;
251 fiff_int_t tag_kind, pos, first = 0, last = 0;
252 std::optional<FiffTag> t_pTag;
253 std::string comment;
254 std::int32_t k;
255 for (k = 0; k < my_evoked->nent(); ++k) {
256 tag_kind = my_evoked->dir[k].kind;
257 pos = my_evoked->dir[k].pos;
258 switch (tag_kind) {
259 case FIFF_COMMENT:
260 t_pTag = m_stream.read_tag(pos);
261 comment = t_pTag->to_string();
262 break;
263 case FIFF_FIRST_SAMPLE:
264 t_pTag = m_stream.read_tag(pos);
265 first = *t_pTag->to_int();
266 break;
267 case FIFF_LAST_SAMPLE:
268 t_pTag = m_stream.read_tag(pos);
269 last = *t_pTag->to_int();
270 break;
271 case FIFF_NCHAN:
272 t_pTag = m_stream.read_tag(pos);
273 nchan = *t_pTag->to_int();
274 break;
275 case FIFF_SFREQ:
276 t_pTag = m_stream.read_tag(pos);
277 sfreq = *t_pTag->to_float();
278 break;
279 case FIFF_CH_INFO:
280 t_pTag = m_stream.read_tag(pos);
281 chs.push_back(t_pTag->to_ch_info());
282 break;
283 }
284 }
285 if (comment.empty())
286 comment = "No comment";
287
288 //
289 // Local channel information?
290 //
291 if (nchan > 0) {
292 if (chs.size() == 0) {
293 return fail(ErrorCode::Format, "Local channel information missing in {}", file);
294 }
295 if (std::ssize(chs) != nchan) {
296 return fail(ErrorCode::Format, "{} channels but {} channel definitions in {}", nchan, chs.size(), file);
297 }
298 evokedInfo.chs = chs;
299 evokedInfo.nchan = nchan;
300 Log::info("\tFound channel information in evoked data. nchan = {}\n", nchan);
301 if (sfreq > 0.0f)
302 evokedInfo.sfreq = sfreq;
303 }
304 std::int32_t nsamp = last - first + 1;
305 Log::info("\tFound the data of interest:\n");
306 Log::info("\t\tt = {:10.2f} ... {:10.2f} ms ({})\n", 1000 * static_cast<float>(first) / evokedInfo.sfreq, 1000 * static_cast<float>(last) / evokedInfo.sfreq, comment);
307 if (evokedInfo.comps.size() > 0)
308 Log::info("\t\t{} CTF compensation matrices available\n", evokedInfo.comps.size());
309
310 //
311 // Read the data in the aspect block
312 //
313 fiff_int_t aspect_kind = -1;
314 fiff_int_t nave = -1;
315 std::vector<FiffTag> epoch;
316 for (k = 0; k < my_aspect->nent(); ++k) {
317 tag_kind = my_aspect->dir[k].kind;
318 pos = my_aspect->dir[k].pos;
319
320 switch (tag_kind) {
321 case FIFF_COMMENT:
322 t_pTag = m_stream.read_tag(pos);
323 comment = t_pTag->to_string();
324 break;
325 case FIFF_ASPECT_KIND:
326 t_pTag = m_stream.read_tag(pos);
327 aspect_kind = *t_pTag->to_int();
328 break;
329 case FIFF_NAVE:
330 t_pTag = m_stream.read_tag(pos);
331 nave = *t_pTag->to_int();
332 break;
333 case FIFF_EPOCH:
334 t_pTag = m_stream.read_tag(pos);
335 epoch.push_back(std::move(*t_pTag));
336 break;
337 }
338 }
339 if (nave == -1)
340 nave = 1;
341 Log::info("\t\tnave = {} - aspect type = {}\n", nave, aspect_kind);
342
343 std::int32_t nepoch = static_cast<int>(epoch.size());
344 if (nepoch != 1 && nepoch != evokedInfo.nchan) {
345 return fail(ErrorCode::Format, "{} epoch tags for {} channels in {}", nepoch, evokedInfo.nchan, file);
346 }
347 MatrixXd all_data;
348 if (nepoch == 1) {
349 //
350 // Only one epoch
351 //
352 all_data = epoch[0].to_float_matrix().cast<double>();
353 all_data.transposeInPlace();
354 //
355 // May need a transpose if the number of channels is one
356 //
357 if (all_data.cols() == 1 && evokedInfo.nchan == 1)
358 all_data.transposeInPlace();
359 } else {
360 //
361 // Put the old style epochs together: one channel per tag
362 //
363 all_data.resize(nepoch, static_cast<int>(epoch[0].size() / sizeof(float)));
364 for (k = 0; k < nepoch; ++k) {
365 const int n = static_cast<int>(epoch[k].size() / sizeof(float));
366 if (n != all_data.cols()) {
367 return fail(ErrorCode::Format, "Old style epochs of lengths {} and {} in {}", n, all_data.cols(), file);
368 }
369 all_data.row(k) = Map<const RowVectorXf>(epoch[k].to_float(), n).cast<double>();
370 }
371 }
372 if (all_data.cols() != nsamp) {
373 return fail(ErrorCode::Format, "{} samples instead of {} in {}", all_data.cols(), nsamp, file);
374 }
375
376 //
377 // Calibrate
378 //
379 Log::info("\n\tPreprocessing...\n");
380 Log::info("\t{} channels remain after picking\n", evokedInfo.nchan);
381
382 using T = Eigen::Triplet<double>;
383 std::vector<T> tripletList;
384 tripletList.reserve(evokedInfo.nchan);
385 for (k = 0; k < evokedInfo.nchan; ++k)
386 tripletList.push_back(T(k, k, evokedInfo.chs[k].cal));
387 SparseMatrix<double> cals(evokedInfo.nchan, evokedInfo.nchan);
388 cals.setFromTriplets(tripletList.begin(), tripletList.end());
389
390 all_data = cals * all_data;
391
392 RowVectorXf times = RowVectorXf(last - first + 1);
393 for (k = 0; k < times.size(); ++k)
394 times[k] = static_cast<float>(first + k) / evokedInfo.sfreq;
395
396 //
397 // Set up projection
398 //
399 if (evokedInfo.projs.size() == 0 || !proj) {
400 Log::info("\tNo projector specified for these data.\n");
401 p_FiffEvoked.proj = MatrixXd();
402 } else {
403 // Create the projector
404 MatrixXd projection;
405 std::int32_t nproj = evokedInfo.make_projector(projection);
406 if (nproj == 0) {
407 Log::warning("\tThe projection vectors do not apply to these channels\n");
408 p_FiffEvoked.proj = MatrixXd();
409 } else {
410 Log::info("\tCreated an SSP operator (subspace dimension = {})\n", nproj);
411 p_FiffEvoked.proj = projection;
412 }
413
414 // The projection items have been activated
415 FiffProj::activate_projs(evokedInfo.projs);
416 }
417
418 if (p_FiffEvoked.proj.rows() > 0) {
419 all_data = p_FiffEvoked.proj * all_data;
420 Log::info("\tSSP projectors applied to the evoked data\n");
421 }
422
423 // Put it all together
424 p_FiffEvoked.info = evokedInfo;
425 p_FiffEvoked.nave = nave;
426 p_FiffEvoked.aspect_kind = aspect_kind;
427 p_FiffEvoked.first = first;
428 p_FiffEvoked.last = last;
429 p_FiffEvoked.comment = std::move(comment);
430 p_FiffEvoked.times = times;
431 p_FiffEvoked.data = all_data;
432
433 if (baseline && baseline->first != baseline->second) {
434 std::pair<float, float> window(baseline->first, baseline->second);
435 p_FiffEvoked.applyBaselineCorrection(window);
436 } else {
437 Log::info("\tNo baseline correction applied\n");
438 }
439
440 return p_FiffEvoked;
441}

474FiffEvoked& FiffEvoked::operator+=(const MatrixXd& newData)
475{
476 //Init matrix if necessary
477 if (nave == -1 || nave == 0) {
478 data = MatrixXd::Zero(newData.rows(), newData.cols());
479 }
480
481 if (data.cols() == newData.cols() && data.rows() == newData.rows()) {
482 //Revert old averaging
483 data = data * nave;
484
485 //Do new averaging
486 data += newData;
487 if (nave <= 0) {
488 nave = 1;
489 } else {
490 nave++;
491 }
492
493 data /= nave;
494 }
495
496 return *this;
497}

501FiffEvoked FiffEvoked::combine(const std::vector<FiffEvoked>& evokeds, CombineWeights weights)
502{
503 VectorXd w(evokeds.size());
504 double naveSum = 0.0;
505 for (const FiffEvoked& e : evokeds) {
506 naveSum += e.nave;
507 }
508 for (int i = 0; i < std::ssize(evokeds); ++i) {
509 w(i) = weights == CombineWeights::Nave ? evokeds[i].nave / naveSum
510 : 1.0 / static_cast<double>(evokeds.size());
511 }
512 return combine(evokeds, w);
513}

517FiffEvoked FiffEvoked::combine(const std::vector<FiffEvoked>& evokeds, const VectorXd& weights)
518{
519 if (evokeds.empty() || weights.size() != std::ssize(evokeds)) {
520 Log::warning("[FiffEvoked::combine] weights must be the same size as the evoked list.");
521 return FiffEvoked();
522 }
523 const FiffEvoked& ref = evokeds.front();
524 for (const FiffEvoked& e : evokeds) {
525 if (e.info.ch_names != ref.info.ch_names || e.times.size() != ref.times.size() || (e.times - ref.times).cwiseAbs().maxCoeff() > 1e-6f) {
526 Log::warning("[FiffEvoked::combine] All evoked data must have the same channels and times.");
527 return FiffEvoked();
528 }
529 }
530
531 FiffEvoked res(ref);
532 res.data = MatrixXd::Zero(ref.data.rows(), ref.data.cols());
533 double invNave = 0.0;
534 std::string comment;
535 for (int i = 0; i < std::ssize(evokeds); ++i) {
536 const FiffEvoked& e = evokeds[i];
537 const double w = weights(i);
538 res.data += w * e.data;
539 invNave += w * w / e.nave;
540 for (const std::string& bad : e.info.bads) {
541 if (!mne::has_channel(res.info.bads, bad)) {
542 res.info.bads.push_back(bad);
543 }
544 }
545
546 const bool unitWeight = std::abs(std::abs(w) - 1.0) <= 1e-8 + 1e-5;
547 const std::string weight = unitWeight ? std::string() : std::format("{:.3f} \u00d7 ", std::abs(w));
548 const std::string_view sign = w >= 0.0 ? (i == 0 ? "" : " + ") : (i == 0 ? "-" : " - ");
549 if (e.comment.contains(" + ")) {
550 std::format_to(std::back_inserter(comment), "{}{}({})", sign, weight, e.comment);
551 } else {
552 std::format_to(std::back_inserter(comment), "{}{}{}", sign, weight, e.comment.empty() ? std::string_view("unknown") : std::string_view(e.comment));
553 }
554 }
555 for (std::size_t at = comment.find(" - - "); at != std::string::npos; at = comment.find(" - - ", at + 3)) {
556 comment.replace(at, 5, " + ");
557 }
558 res.comment = std::move(comment);
559 res.nave = static_cast<fiff_int_t>(std::lround(1.0 / invNave));
560 return res;
561}
