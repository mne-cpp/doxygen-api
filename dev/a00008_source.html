140void ImageSc::normalize()
141{
142 m_matCentNormData = m_matData;
143 m_matCentNormData.array() -= m_dMinValue;
144 // A constant matrix has no range: keep it at 0 instead of dividing 0 by 0
145 if (m_dMaxValue > m_dMinValue) {
146 m_matCentNormData.array() /= (m_dMaxValue - m_dMinValue);
147 }
148 m_matCentNormData = m_matCentNormData.cwiseMax(0.0).cwiseMin(1.0);
149}

153void ImageSc::updateMaps()
154{
155 m_pixmapData = QPixmap();
156 m_pixmapColorbar = QPixmap();
157
158 if (m_matCentNormData.rows() > 0 && m_matCentNormData.cols() > 0) {
159 const qint32 cols = m_matCentNormData.cols();
160 const qint32 rows = m_matCentNormData.rows();
161 QImage imageData(cols, rows, QImage::Format_RGB32);
162 for (qint32 i = 0; i < cols; ++i) {
163 for (qint32 j = 0; j < rows; ++j) {
164 imageData.setPixel(i, j, pColorMapper(m_matCentNormData(j, i), m_sColorMap));
165 }
166 }
167 m_pixmapData = QPixmap::fromImage(imageData);
168
169 QImage imageColorbar(1, m_iColorbarGradSteps, QImage::Format_RGB32);
170 for (qint32 j = 0; j < m_iColorbarGradSteps; ++j) {
171 imageColorbar.setPixel(0, j, pColorMapper(static_cast<double>(m_iColorbarGradSteps - 1 - j) / (m_iColorbarGradSteps - 1), m_sColorMap));
172 }
173 m_pixmapColorbar = QPixmap::fromImage(imageColorbar);
174 }
175 update();
176}

222void ImageSc::paintEvent(QPaintEvent* event)
223{
224 Q_UNUSED(event);
225
226 if (m_pixmapData.isNull()) {
227 return;
228 }
229
230 QPainter painter(this);
231 painter.setRenderHint(QPainter::TextAntialiasing);
232 const QColor text(51, 51, 51);
233 const QFontMetrics metrics(font());
234 const int line = metrics.height();
235
236 // Column and row centres in data units: without an extent, column and row indices
237 const int cols = m_matCentNormData.cols();
238 const int rows = m_matCentNormData.rows();
239 const double left = m_bExtentSet ? m_dLeft : 0.0;
240 const double right = m_bExtentSet ? m_dRight : cols - 1;
241 const double top = m_bExtentSet ? m_dTop : 0.0;
242 const double bottom = m_bExtentSet ? m_dBottom : rows - 1;
243 // Half a cell beyond the first and last centre, so each cell is centred on its value
244 const double halfX = cols > 1 ? (right - left) / (cols - 1) / 2.0 : 0.5;
245 const double halfY = rows > 1 ? (bottom - top) / (rows - 1) / 2.0 : 0.5;
246 const double x0 = left - halfX;
247 const double x1 = right + halfX;
248 const double y0 = top - halfY;
249 const double y1 = bottom + halfY;
250
251 // No ticks between cell centres: whole numbers for indices, at least one cell apart in data units
252 const QVector<double> xTicks = niceTicks(std::min(x0, x1), std::max(x0, x1), 8, std::fabs(2.0 * halfX));
253 const QVector<double> yTicks = niceTicks(std::min(y0, y1), std::max(y0, y1), 6, std::fabs(2.0 * halfY));
254 const QVector<double> cTicks = niceTicks(m_dMinValue, m_dMaxValue, 6);
255
256 int yTickWidth = 0;
257 for (double v : yTicks) {
258 yTickWidth = std::max(yTickWidth, metrics.horizontalAdvance(tickLabel(v)));
259 }
260 int cTickWidth = 0;
261 for (double v : cTicks) {
262 cTickWidth = std::max(cTickWidth, metrics.horizontalAdvance(tickLabel(v)));
263 }
264
265 const int mLeft = yTickWidth + 14 + (m_sYLabel.isEmpty() ? 0 : line + 6);
266 const int mRight = 16 + (m_bColorbar ? m_iColorbarWidth + 14 + cTickWidth + (m_sColorbarLabel.isEmpty() ? 0 : line + 6) : 0);
267 const int mTop = m_sTitle.isEmpty() ? 12 : 2 * line + 8;
268 const int mBottom = line + 14 + (m_sXLabel.isEmpty() ? 0 : line + 6);
269 const QRect area(mLeft, mTop, width() - mLeft - mRight, height() - mTop - mBottom);
270 if (area.width() <= 0 || area.height() <= 0) {
271 return;
272 }
273
274 const auto toPixelX = [&](double x) {
275 return area.left() + (x - x0) / (x1 - x0) * area.width();
276 };
277 const auto toPixelY = [&](double y) {
278 return area.top() + (y - y0) / (y1 - y0) * area.height();
279 };
280
281 painter.drawPixmap(area, m_pixmapData);
282 painter.setPen(QPen(QColor(136, 136, 136), 1));
283 painter.setBrush(Qt::NoBrush);
284 painter.drawRect(area.adjusted(0, 0, -1, -1));
285
286 painter.setPen(text);
287 for (double v : xTicks) {
288 const double px = toPixelX(v);
289 painter.drawLine(QPointF(px, area.bottom()), QPointF(px, area.bottom() + 4));
290 painter.drawText(QRectF(px - 50, area.bottom() + 6, 100, line), Qt::AlignHCenter | Qt::AlignTop, tickLabel(v));
291 }
292 for (double v : yTicks) {
293 const double py = toPixelY(v);
294 painter.drawLine(QPointF(area.left() - 4, py), QPointF(area.left(), py));
295 painter.drawText(QRectF(0, py - line, area.left() - 8, 2 * line), Qt::AlignRight | Qt::AlignVCenter, tickLabel(v));
296 }
297
298 if (m_bColorbar && !m_pixmapColorbar.isNull()) {
299 const QRect bar(area.right() + 16, area.top(), m_iColorbarWidth, area.height());
300 painter.drawPixmap(bar, m_pixmapColorbar);
301 painter.setPen(QPen(QColor(136, 136, 136), 1));
302 painter.drawRect(bar.adjusted(0, 0, -1, -1));
303 painter.setPen(text);
304 if (m_dMaxValue > m_dMinValue) {
305 for (double v : cTicks) {
306 const double py = bar.bottom() - (v - m_dMinValue) / (m_dMaxValue - m_dMinValue) * bar.height();
307 painter.drawLine(QPointF(bar.right(), py), QPointF(bar.right() + 4, py));
308 painter.drawText(QRectF(bar.right() + 7, py - line, cTickWidth + 4, 2 * line), Qt::AlignLeft | Qt::AlignVCenter, tickLabel(v));
309 }
310 }
311 if (!m_sColorbarLabel.isEmpty()) {
312 painter.save();
313 painter.translate(bar.right() + 13 + cTickWidth, bar.center().y());
314 painter.rotate(-90);
315 painter.drawText(QRectF(-bar.height() / 2.0, 0, bar.height(), line), Qt::AlignCenter, m_sColorbarLabel);
316 painter.restore();
317 }
318 }
319
320 if (!m_sTitle.isEmpty()) {
321 QFont titleFont = font();
322 titleFont.setBold(true);
323 titleFont.setPointSizeF(titleFont.pointSizeF() * 1.25);
324 painter.setFont(titleFont);
325 painter.drawText(QRectF(0, 4, width(), mTop - 4), Qt::AlignCenter, m_sTitle);
326 painter.setFont(font());
327 }
328 if (!m_sXLabel.isEmpty()) {
329 painter.drawText(QRectF(area.left(), height() - line - 6, area.width(), line), Qt::AlignCenter, m_sXLabel);
330 }
331 if (!m_sYLabel.isEmpty()) {
332 painter.save();
333 painter.translate(6, area.center().y());
334 painter.rotate(-90);
335 painter.drawText(QRectF(-area.height() / 2.0, 0, area.height(), line), Qt::AlignCenter, m_sYLabel);
336 painter.restore();
337 }
338}
