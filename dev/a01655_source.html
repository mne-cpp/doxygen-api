84std::vector<std::string_view> split(std::string_view text)
85{
86 std::vector<std::string_view> tokens;
87 std::size_t begin = 0;
88 while (true) {
89 const std::size_t end = text.find(',', begin);
90 tokens.push_back(text.substr(begin, end - begin));
91 if (end == std::string_view::npos) {
92 return tokens;
93 }
94 begin = end + 1;
95 }

99AcqExpected<std::vector<std::uint8_t>> parseInputs(std::string_view text)
100{
101 std::array<bool, kGusbampInputs + 1> chosen{};
102 for (const std::string_view token : split(text)) {
103 const std::size_t dash = token.find('-');
104 const std::string_view low = token.substr(0, dash);
105 const std::string_view high = dash == std::string_view::npos ? low : token.substr(dash + 1);
106 unsigned first = 0;
107 unsigned last = 0;
108 const auto a = std::from_chars(low.data(), low.data() + low.size(), first);
109 const auto b = std::from_chars(high.data(), high.data() + high.size(), last);
110 if (a.ec != std::errc() || a.ptr != low.data() + low.size() || b.ec != std::errc() || b.ptr != high.data() + high.size() ||
111 first < 1 || last > kGusbampInputs || first > last) {
112 return invalid(std::format("inputs={}: a list of inputs 1..16 such as 1-8,11", text));
113 }
114 for (unsigned i = first; i <= last; ++i) {
115 chosen[i] = true;
116 }
117 }
118 std::vector<std::uint8_t> inputs;
119 for (unsigned i = 1; i <= kGusbampInputs; ++i) {
120 if (chosen[i]) {
121 inputs.push_back(static_cast<std::uint8_t>(i));
122 }
123 }
124 return inputs;
125}

127AcqExpected<Config> parseConfig(const AcqUri& uri)
128{
129 Config c;
130 if (!uri.target.empty()) {
131 for (const std::string_view serial : split(uri.target)) {
132 if (serial.empty() || std::find(c.serials.begin(), c.serials.end(), serial) != c.serials.end()) {
133 return invalid("target '" + uri.target + "': distinct serial numbers, master first, separated by commas");
134 }
135 c.serials.emplace_back(serial);
136 }
137 if (c.serials.size() > kGusbampMaxUnits) {
138 return invalid(std::format("target '{}': at most {} units run synchronised", uri.target, kGusbampMaxUnits));
139 }
140 }
141 auto inputs = parseInputs(uri.param("inputs").value_or("1-16"));
142 const auto simulate = paramBool(uri, "simulate", false);
143 const auto rate = paramInt(uri, "rate", 1200, 1, 38400);
144 const auto trigger = paramBool(uri, "trigger", false);
145 const auto seed = paramInt(uri, "seed", 0, 0, INT64_MAX);
146 const auto realtime = paramBool(uri, "realtime", true);
147 const auto duration = paramDouble(uri, "duration", 0.0, 0.0, 1e7);
148 const auto gain = paramDouble(uri, "gain", 1.0, 0.0, 1e6);
149 auto thread = paramThreadPolicy(uri);
150 for (const AcqError& error : {inputs.error(), simulate.error(), rate.error(), trigger.error(), seed.error(), realtime.error(),
151 duration.error(), gain.error(), thread.error()}) {
152 if (!error.ok()) {
153 return error;
154 }
155 }
156 const auto supported = std::find_if(kRates.begin(), kRates.end(), [&](const auto& r) { return r.first == *rate; });
157 if (supported == kRates.end()) {
158 return invalid(std::format("rate={}: one of 32, 64, 128, 256, 512, 600, 1200, 2400, 4800, 9600, 19200 or 38400", *rate));
159 }
160 c.simulate = *simulate;
161 c.setup.inputs = std::move(*inputs);
162 c.setup.rate = supported->first;
163 c.setup.scans = supported->second;
164 c.setup.trigger = *trigger;
165 c.simulation.seed = static_cast<std::uint64_t>(*seed);
166 c.simulation.realtime = *realtime;
167 c.simulation.duration = *duration;
168 c.simulation.gain = *gain;
169 c.thread = std::move(*thread);
170 return c;
171}

176 static AcqExpected<std::unique_ptr<AcqBackend>> open(const AcqUri& uri)
177 {
178 auto config = parseConfig(uri);
179 if (!config) {
180 return config.error();
181 }
182 std::unique_ptr<GusbampTransport> transport;
183 if (config->simulate) {
184 transport = make_gusbamp_simulator(config->serials.empty() ? std::vector<std::string>{"UB-0000.00.01"} : config->serials,
185 config->simulation);
186 } else {
187 auto driver = open_gusbamp_driver(config->serials);
188 if (!driver) {
189 return driver.error();
190 }
191 transport = std::move(*driver);
192 }
193 const std::size_t units = transport->serials().size();
194
195 AcqStreamInfo info;
196 info.name = "eeg";
197 info.sourceId = transport->description();
198 info.format = AcqSampleFormat::Float32;
199 info.layout = AcqLayout::Interleaved;
200 info.sfreq = config->setup.rate;
201 for (std::size_t c = 0; c < units * config->setup.inputs.size(); ++c) {
202 AcqChannelInfo& channel = info.channels.emplace_back();
203 channel.name = std::format("EEG {:03}", c + 1);
204 channel.kind = AcqChannelKind::Eeg;
205 channel.unit = AcqUnit::Volt;
206 channel.scale = 1e-6;
207 }
208 if (config->setup.trigger) {
209 AcqChannelInfo& channel = info.channels.emplace_back();
210 channel.name = "STI 014";
211 channel.kind = AcqChannelKind::Stim;
212 channel.unit = AcqUnit::None;
213 }
214
215 std::unique_ptr<GusbampBackend> backend(new GusbampBackend(std::move(*config), std::move(transport), uri.toString()));
216 AcqStreamOptions options;
217 options.capacitySamples = std::max<std::size_t>(16 * std::size_t{backend->m_config.setup.scans}, backend->m_config.setup.rate);
218 auto writer = backend->addInputStream(std::move(info), options);
219 if (!writer) {
220 return writer.error();
221 }
222 backend->m_writer = *writer;
223 return std::unique_ptr<AcqBackend>(std::move(backend));
224 }

239 AcqError start() override
240 {
241 if (AcqError e = m_transport->start(m_config.setup); !e.ok()) {
242 return e;
243 }
244 m_trigger.reset();
245 const auto now = std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now().time_since_epoch()).count();
246 AcqClockModel clock;
247 clock.sfreq = m_config.setup.rate;
248 clock.anchorSample = m_writer->writePosition();
249 clock.anchorHostNs = now;
250 clock.utcOffsetNs =
251 std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::system_clock::now().time_since_epoch()).count() - now;
252 clock.source = AcqClockSource::Nominal;
253 m_writer->setClock(clock);
254 m_stop.store(false, std::memory_order_relaxed);
255 auto thread = start_thread(m_config.thread, [this] { run(); });
256 if (!thread) {
257 m_transport->stop();
258 return thread.error();
259 }
260 m_thread = std::move(*thread);
261 return {};
262 }

295 void run()
296 {
297 const GusbampSetup& setup = m_config.setup;
298 const std::size_t inputs = setup.inputs.size();
299 while (!m_stop.load(std::memory_order_acquire)) {
300 auto transfers = m_transport->read(kWait);
301 if (!transfers) {
302 if (transfers.error() == AcqErrorCode::Timeout) {
303 continue;
304 }
305 if (transfers.error() == AcqErrorCode::EndOfStream) {
306 m_writer->finish();
307 } else {
308 reportError(transfers.error());
309 }
310 return;
311 }
312 AcqExpected<AcqWriteView> view = AcqError(AcqErrorCode::Timeout);
313 while (!view && view.error() == AcqErrorCode::Timeout) {
314 if (m_stop.load(std::memory_order_acquire)) {
315 return;
316 }
317 view = m_writer->beginWrite(setup.scans, kPoll); // a gating reader may not have freed the space yet
318 }
319 if (!view) {
320 reportError(view.error());
321 return;
322 }
323 // Scan after scan: the inputs of every unit, master first, then the master's trigger line
324 const std::uint64_t first = m_writer->writePosition();
325 std::size_t scan = 0;
326 for (std::size_t p = 0; p < view->partCount(); ++p) {
327 const AcqWritePart& part = view->part(p);
328 for (std::size_t i = 0; i < part.samples; ++i, ++scan) {
329 auto* frame = static_cast<std::byte*>(part.at(0, i));
330 for (std::size_t u = 0; u < transfers->size(); ++u) {
331 const std::byte* values = (*transfers)[u].data() + kGusbampHeaderBytes + scan * setup.width(u) * sizeof(float);
332 std::memcpy(frame + u * inputs * sizeof(float), values, inputs * sizeof(float));
333 if (setup.width(u) > inputs) {
334 std::memcpy(frame + transfers->size() * inputs * sizeof(float), values + inputs * sizeof(float), sizeof(float));
335 float level = 0.0F;
336 std::memcpy(&level, values + inputs * sizeof(float), sizeof(level));
337 if (const auto code = static_cast<std::int32_t>(std::lround(level)); m_trigger.changed(code)) {
338 if (AcqError e = m_writer->pushEvent({first + scan, 0, code}); !e.ok()) {
339 reportError(e);
340 return;
341 }
342 }
343 }
344 }
345 }
346 }
347 m_writer->commit(setup.scans);
348 }
349 }
