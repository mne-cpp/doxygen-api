67// "LA 12" -> ("LA", 12): the trailing number and the name before it without separators
68NameParts splitName(const std::string& name)
69{
70 std::size_t digits = name.size();
71 while (digits > 0 && isDigit(name[digits - 1]))
72 --digits;
73 if (digits == name.size())
74 return {};
75 std::size_t prefixEnd = digits;
76 while (prefixEnd > 0 && (isSpace(name[prefixEnd - 1]) || name[prefixEnd - 1] == '_' || name[prefixEnd - 1] == '-'))
77 --prefixEnd;
78 return {true, name.substr(0, prefixEnd), mne::to_int(std::string_view(name).substr(digits)).value_or(0)};
79}

81// "[4x8]", "4 x 8" -> (4, 8)
82std::optional<std::pair<int, int>> parseDimension(std::string_view text)
83{
84 std::size_t i = 0;
85 auto skipSpace = [&] {
86 while (i < text.size() && isSpace(text[i]))
87 ++i;
88 };
89 auto number = [&]() -> std::optional<int> {
90 const std::size_t start = i;
91 while (i < text.size() && isDigit(text[i]))
92 ++i;
93 if (i == start)
94 return std::nullopt;
95 return mne::to_int(text.substr(start, i - start)).value_or(0);
96 };
97 if (i < text.size() && text[i] == '[')
98 ++i;
99 skipSpace();
100 const std::optional<int> rows = number();
101 skipSpace();
102 if (!rows || i >= text.size() || text[i] != 'x')
103 return std::nullopt;
104 ++i;
105 skipSpace();
106 const std::optional<int> cols = number();
107 skipSpace();
108 if (cols && i < text.size() && text[i] == ']')
109 ++i;
110 if (!cols || i != text.size())
111 return std::nullopt;
112 return std::make_pair(*rows, *cols);
113}

140IeegImplantation::IeegImplantation(const std::vector<IeegContact>& contacts, const std::string& coordinateSystem)
141: m_contacts(contacts)
142, m_coordinateSystem(coordinateSystem)
143{
144 std::unordered_map<std::string, std::size_t> leadIndex;
145 std::vector<std::vector<std::pair<int, int>>> members; // per lead: (number along the lead, contact index)
146 for (std::size_t i = 0; i < m_contacts.size(); ++i) {
147 IeegContact& contact = m_contacts[i];
148 const NameParts parts = splitName(contact.name);
149 if (contact.lead.empty())
150 contact.lead = (parts.numbered && !parts.prefix.empty()) ? parts.prefix : contact.name;
151 auto [it, inserted] = leadIndex.try_emplace(contact.lead, m_leads.size());
152 if (inserted) {
153 IeegLead lead;
154 lead.name = contact.lead;
155 m_leads.push_back(lead);
156 members.emplace_back();
157 }
158 members[it->second].emplace_back(parts.number, static_cast<int>(i));
159 }
160
161 for (std::size_t l = 0; l < m_leads.size(); ++l) {
162 std::vector<std::pair<int, int>>& lead = members[l];
163 std::stable_sort(lead.begin(), lead.end(), [](const auto& a, const auto& b) { return a.first < b.first; });
164 float spacing = std::numeric_limits<float>::max();
165 m_leads[l].contacts.reserve(lead.size());
166 for (std::size_t k = 0; k < lead.size(); ++k) {
167 m_leads[l].contacts.push_back(lead[k].second);
168 if (k > 0)
169 spacing = std::min(spacing, (m_contacts[std::size_t(lead[k].second)].position - m_contacts[std::size_t(lead[k - 1].second)].position).norm());
170 }
171 m_leads[l].type = spacing < kMaxDepthSpacing ? IeegLeadType::Depth : IeegLeadType::Strip;
172 }
173}

177IeegImplantation IeegImplantation::readBids(const std::string& electrodesTsv, std::string* error)
178{
179 const std::filesystem::path tsvPath = mne::path_from_utf8(electrodesTsv);
180 const std::string tsvName = mne::utf8_from_path(tsvPath.filename());
181 // The coordsystem sidecar sharing the longest name prefix, so sessions are never mixed
182 std::vector<std::pair<std::string, std::string>> candidates; // (lower-case name, name), sorted like QDir
183 std::error_code ec;
184 for (const auto& entry : std::filesystem::directory_iterator(tsvPath.parent_path().empty() ? "." : tsvPath.parent_path(), ec)) {
185 std::error_code typeError;
186 const std::string name = mne::utf8_from_path(entry.path().filename());
187 if (entry.is_regular_file(typeError) && endsWithIgnoreCase(name, "_coordsystem.json"))
188 candidates.emplace_back(mne::to_lower(name), name);
189 }
190 std::sort(candidates.begin(), candidates.end());
191 std::string coordPath;
192 std::ptrdiff_t bestPrefix = -1;
193 for (const auto& [lower, candidate] : candidates) {
194 const auto prefix = std::mismatch(candidate.begin(), candidate.end(), tsvName.begin(), tsvName.end()).first - candidate.begin();
195 if (prefix > bestPrefix) {
196 bestPrefix = prefix;
197 coordPath = mne::utf8_from_path(tsvPath.parent_path() / mne::path_from_utf8(candidate));
198 }
199 }
200 if (coordPath.empty()) {
201 setError(error, "no *_coordsystem.json next to " + electrodesTsv + ", so the units are unknown");
202 return {};
203 }
204 const mne::BidsCoordinateSystem coordSystem = mne::BidsCoordinateSystem::read_json(mne::path_from_utf8(coordPath));
205 const std::string& units = coordSystem.units;
206 const float scale = units == "m" ? 1.0f
207 : units == "cm" ? 0.01f
208 : units == "mm" ? 0.001f
209 : 0.0f;
210 if (scale == 0.0f) {
211 setError(error, "unsupported iEEGCoordinateUnits \"" + units + "\" in " + coordPath);
212 return {};
213 }
214
215 std::vector<IeegContact> contacts;
216 std::unordered_map<std::string, std::string> declaredTypes;
217 std::unordered_map<std::string, std::string> declaredDimensions;
218 for (const mne::BidsElectrode& e : mne::BidsElectrode::read_tsv(mne::path_from_utf8(electrodesTsv))) {
219 const std::optional<float> x = toCoordinate(e.x);
220 const std::optional<float> y = toCoordinate(e.y);
221 const std::optional<float> z = toCoordinate(e.z);
222 if (!x || !y || !z)
223 continue;
224 IeegContact contact;
225 contact.name = e.name;
226 contact.position = Eigen::Vector3f(*x, *y, *z) * scale;
227 contact.lead = e.group;
228 // Declarations are per lead; contacts without a group take their lead from the name later
229 const std::string key = contact.lead.empty() ? contact.name : contact.lead;
230 if (!e.type.empty())
231 declaredTypes.insert_or_assign(key, mne::to_lower(e.type));
232 if (!e.dimension.empty())
233 declaredDimensions.insert_or_assign(key, e.dimension);
234 contacts.push_back(std::move(contact));
235 }
236 if (contacts.empty()) {
237 setError(error, "no contacts with coordinates in " + electrodesTsv);
238 return {};
239 }
240
241 auto lookup = [](const std::unordered_map<std::string, std::string>& map, const std::string& key) {
242 const auto it = map.find(key);
243 return it == map.end() ? std::string() : it->second;
244 };
245 IeegImplantation implantation(contacts, coordSystem.system);
246 for (std::size_t l = 0; l < implantation.m_leads.size(); ++l) {
247 const IeegLead lead = implantation.m_leads[l];
248 // A declaration on any contact of the lead applies to the lead
249 std::string type = lookup(declaredTypes, lead.name);
250 std::string shape = lookup(declaredDimensions, lead.name);
251 for (const int c : lead.contacts) {
252 const std::string& name = implantation.m_contacts[std::size_t(c)].name;
253 if (type.empty())
254 type = lookup(declaredTypes, name);
255 if (shape.empty())
256 shape = lookup(declaredDimensions, name);
257 }
258 const std::optional<std::pair<int, int>> dimension = parseDimension(shape);
259 const int rows = dimension ? dimension->first : 0;
260 const int cols = dimension ? dimension->second : 0;
261 IeegLeadType leadType = (rows > 1 && cols > 1 && std::int64_t{rows} * cols == std::int64_t(lead.contacts.size())) ? IeegLeadType::Grid : lead.type;
262 if (type.find("depth") != std::string::npos || type.find("seeg") != std::string::npos)
263 leadType = IeegLeadType::Depth;
264 else if (type.find("strip") != std::string::npos)
265 leadType = IeegLeadType::Strip;
266 else if (type.find("grid") != std::string::npos)
267 leadType = IeegLeadType::Grid;
268 implantation.setLeadType(lead.name, leadType, rows, cols);
269 }
270 return implantation;
271}

275bool IeegImplantation::setLeadType(const std::string& lead, IeegLeadType type, int rows, int cols)
276{
277 for (IeegLead& l : m_leads) {
278 if (l.name != lead)
279 continue;
280 l.type = type;
281 const bool shapeFits = rows > 0 && cols > 0 && std::int64_t{rows} * cols == std::int64_t(l.contacts.size());
282 l.rows = shapeFits ? rows : 0;
283 l.cols = shapeFits ? cols : 0;
284 return true;
285 }
286 return false;
287}

307{
308 IeegImplantation result = *this;
309 for (IeegContact& contact : result.m_contacts)
310 contact.position = trans.topLeftCorner<3, 3>() * contact.position + trans.topRightCorner<3, 1>();
311 result.m_coordinateSystem = frame;
