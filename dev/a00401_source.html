108void unitOf(const std::string& unit, AcqChannelInfo& channel)
109{
110 const std::string u = lower(unit);
111 struct Entry
112 {
113 const char* name;
114 AcqUnit unit;
115 int exponent;
116 };
117 static constexpr Entry kUnits[] = {
118 {"microvolts", AcqUnit::Volt, -6},
119 {"microvolt", AcqUnit::Volt, -6},
120 {"uv", AcqUnit::Volt, -6},
121 {"\xc2\xb5v", AcqUnit::Volt, -6},
122 {"millivolts", AcqUnit::Volt, -3},
123 {"mv", AcqUnit::Volt, -3},
124 {"volts", AcqUnit::Volt, 0},
125 {"volt", AcqUnit::Volt, 0},
126 {"v", AcqUnit::Volt, 0},
127 {"femtotesla", AcqUnit::Tesla, -15},
128 {"ft", AcqUnit::Tesla, -15},
129 {"tesla", AcqUnit::Tesla, 0},
130 {"t", AcqUnit::Tesla, 0},
131 };
132 for (const Entry& entry : kUnits) {
133 if (u == entry.name) {
134 channel.unit = entry.unit;
135 channel.unitExponent = entry.exponent;
136 return;
137 }
138 }
139 if (channel.kind == AcqChannelKind::Stim) {
140 channel.unit = AcqUnit::None;
141 }

157AcqExpected<Config> parseConfig(const AcqUri& uri)
158{
159 Config c;
160 c.name = uri.target;
161 c.type = std::string(uri.param("type").value_or(""));
162 c.sourceId = std::string(uri.param("source_id").value_or(""));
163 const auto timeout = paramDouble(uri, "timeout", 5.0, 0.0, 3600.0);
164 auto thread = paramThreadPolicy(uri);
165 const auto simulate = paramBool(uri, "simulate", false);
166 const auto channels = paramInt(uri, "channels", 8, 1, 1024);
167 const auto sfreq = paramDouble(uri, "sfreq", 250.0, 1.0, 100000.0);
168 const auto seed = paramInt(uri, "seed", 0, 0, std::numeric_limits<std::int32_t>::max());
169 for (const AcqError& error : {timeout.error(), thread.error(), simulate.error(), channels.error(), sfreq.error(), seed.error()}) {
170 if (!error.ok()) {
171 return error;
172 }
173 }
174 c.simulate = *simulate;
175 c.channels = static_cast<std::size_t>(*channels);
176 c.sfreq = *sfreq;
177 c.seed = *seed;
178 if (c.simulate) {
179 // Resolved by a source id of its own, so a real stream of the same name is never read instead
180 static std::atomic<unsigned> count{0};
181 c.name = c.name.empty() ? "MNE-CPP simulator" : c.name;
182 c.type = "EEG";
183 c.sourceId = std::format("mne-acq-simulator-{}-{}", Clock::now().time_since_epoch().count(), count++);
184 }
185 if (c.name.empty() && c.type.empty() && c.sourceId.empty()) {
186 return AcqError(AcqErrorCode::InvalidParameter, "lsl:// needs a stream name, type= or source_id=, e.g. lsl://BioSemi");
187 }
188 c.timeout = *timeout;
189 c.thread = std::move(*thread);
190 return c;
191}

195{
196 if (c.simulate) {
197 return {"source_id", c.sourceId};

199 if (!c.name.empty()) {
200 return {"name", c.name};
201 }
202 if (!c.sourceId.empty()) {
203 return {"source_id", c.sourceId};
204 }
205 return {"type", c.type};

239 std::size_t pull(std::byte* out, std::size_t frames, double& lastTimestamp)
240 {
241 if (m_pending.empty()) {
242 if (m_inlet.samples_available()) {
243 m_pending = m_inlet.pull_chunk_double(m_stamps);
244 m_next = 0;
245 }
246 if (m_pending.empty()) {
247 std::this_thread::sleep_for(std::chrono::milliseconds(1));
248 return 0;
249 }
250 }
251 const std::size_t size = bytesPerSample(m_format);
252 std::size_t n = 0;
253 for (; n < frames && m_next < m_pending.size(); ++n, ++m_next) {
254 const std::vector<double>& sample = m_pending[m_next];
255 for (std::size_t c = 0; c < m_channels; ++c) {
256 store(out + (n * m_channels + c) * size, c < sample.size() ? sample[c] : 0.0);
257 }
258 }
259 lastTimestamp = m_stamps[m_next - 1];
260 if (m_next == m_pending.size()) {
261 m_pending.clear();
262 }
263 return n;
264 }
265
267 double timeCorrection()
268 {
269 const double correction = m_inlet.time_correction(2.0);
270 if (correction == correction) {
271 m_correction = correction;
272 }
273 return m_correction;
274 }

278 static void storeAs(std::byte* at, double value)
279 {
280 const auto v = static_cast<T>(value);
281 std::memcpy(at, &v, sizeof(T));
282 }

284 void store(std::byte* at, double value) const
285 {
286 switch (m_format) {
287 case AcqSampleFormat::Float64:
288 storeAs<double>(at, value);
289 break;
290 case AcqSampleFormat::Int32:
291 storeAs<std::int32_t>(at, value);
292 break;
293 case AcqSampleFormat::Int16:
294 storeAs<std::int16_t>(at, value);
295 break;
296 default:
297 storeAs<float>(at, value);
298 break;
299 }
300 }

365 LslSimulator(const Config& c, std::size_t block, AcqDevice session, AcqReader reader)
366 : m_outlet(outletInfo(c, session))
367 , m_session(std::move(session))
368 , m_reader(std::move(reader))
369 , m_chunk(block, std::vector<float>(c.channels))
370 {
371 }
372
373 void run()
374 {
375 while (!m_stop.load(std::memory_order_acquire)) {
376 auto view = m_reader.acquire(m_chunk.size(), std::chrono::milliseconds(200), m_chunk.size());
377 if (!view) {
378 if (view.error().code() == AcqErrorCode::Timeout) {
379 continue;
380 }
381 return;
382 }
383 std::size_t s = 0;
384 for (std::size_t p = 0; p < view->partCount(); ++p) {
385 const AcqBlockPart& part = view->part(p);
386 for (std::size_t i = 0; i < part.samples; ++i, ++s) {
387 for (std::size_t c = 0; c < m_chunk[s].size(); ++c) {
388 std::memcpy(&m_chunk[s][c], part.at(c, i), sizeof(float));
389 }
390 }
391 }
392 const bool whole = s == m_chunk.size();
393 if (!m_reader.release(*view).ok() || !whole) {
394 return;
395 }
396 m_outlet.push_chunk(m_chunk);
397 }
398 }

401 AcqDevice m_session;
402 AcqReader m_reader;
403 std::vector<std::vector<float>> m_chunk;
404 std::thread m_thread;
405 std::atomic<bool> m_stop{false};

493 AcqError describe(const mne::stream_info& resolved, AcqStreamInfo& stream)
494 {
495 switch (resolved.channel_format()) {
496 case mne::ChannelFormat::Float32:
497 stream.format = AcqSampleFormat::Float32;
498 break;
499 case mne::ChannelFormat::Double64:
500 case mne::ChannelFormat::Int64:
501 stream.format = AcqSampleFormat::Float64;
502 break;
503 case mne::ChannelFormat::Int32:
504 stream.format = AcqSampleFormat::Int32;
505 break;
506 case mne::ChannelFormat::Int16:
507 case mne::ChannelFormat::Int8:
508 stream.format = AcqSampleFormat::Int16;
509 break;
510 case mne::ChannelFormat::String:
511 return {AcqErrorCode::UnsupportedFormat, "string (marker) LSL streams are not supported yet"};
512 default:
513 return {AcqErrorCode::UnsupportedFormat, "LSL channel format " + std::to_string(static_cast<int>(resolved.channel_format()))};
514 }
515 if (resolved.nominal_srate() <= 0.0) {
516 return {AcqErrorCode::UnsupportedFormat, "irregular-rate LSL streams are not supported yet"};
517 }
518 mne::stream_info info;
519 try {
520 mne::stream_inlet inlet(resolved);
521 info = inlet.info(std::max(m_config.timeout, 1.0));
522 } catch (const std::exception& e) {
523 return {AcqErrorCode::DeviceIo, std::string("cannot read the LSL stream description: ") + e.what()};
524 }
525 stream.name = info.name();
526 stream.sourceId = info.source_id();
527 stream.sfreq = info.nominal_srate();
528 stream.clockDomain = "lsl:" + info.source_id();
529 m_device = info.name() + " (" + info.type() + ") on " + info.hostname();
530 const std::vector<mne::channel_description>& described = info.channels();
531 for (std::size_t c = 0; c < static_cast<std::size_t>(std::max(0, info.channel_count())); ++c) {
532 const mne::channel_description none;
533 const mne::channel_description& d = c < described.size() ? described[c] : none;
534 AcqChannelInfo ch;
535 ch.name = d.label.empty() ? info.type() + " " + std::to_string(c + 1) : d.label;
536 ch.kind = kindOf(d.type.empty() ? info.type() : d.type);
537 unitOf(d.unit, ch);
538 stream.channels.push_back(std::move(ch));
539 }
540 return {};
541 }

563 std::pair<std::int64_t, double> sameInstant() const
564 {
565 std::pair<std::int64_t, double> best{0, std::numeric_limits<double>::quiet_NaN()};
566 double width = std::numeric_limits<double>::infinity();
567 for (int attempt = 0; attempt < 3; ++attempt) {
568 const double before = mne::local_clock();
569 const auto steady = std::chrono::duration_cast<std::chrono::nanoseconds>(Clock::now().time_since_epoch()).count();
570 const double after = mne::local_clock();
571 if (!(after - before < width)) {
572 continue;
573 }
574 width = after - before;
575 best = {steady, 0.5 * (before + after)};
576 }
577 if (width == std::numeric_limits<double>::infinity()) { // no LSL clock
578 best.first = std::chrono::duration_cast<std::chrono::nanoseconds>(Clock::now().time_since_epoch()).count();
579 }
580 return best;
581 }

583 void publishClock(std::uint64_t sample, double timestamp, bool refresh)
584 {
585 const auto [steady, local] = sameInstant();
586 AcqClockModel clock;
587 clock.sfreq = m_sfreq;
588 clock.anchorSample = sample;
589 clock.utcOffsetNs =
590 std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::system_clock::now().time_since_epoch())
591 .count() -
592 steady;
593 if (timestamp == timestamp && local == local) {
594 if (refresh) {
595 m_correction = m_link->timeCorrection();
596 }
597 // LSL time to steady_clock: both are monotonic, but need not share an epoch.
598 clock.anchorHostNs = steady + static_cast<std::int64_t>((timestamp + m_correction - local) * 1e9);
599 clock.source = AcqClockSource::Lsl;
600 } else {
601 clock.anchorHostNs = steady;
602 clock.source = AcqClockSource::Host;
603 }
604 m_writer->setClock(clock);
605 }
606
607 void run()
608 {
609 std::uint64_t next = 0;
610 Clock::time_point lastClock;
611 while (!m_stop.load(std::memory_order_acquire)) {
612 double timestamp = std::numeric_limits<double>::quiet_NaN();
613 const std::size_t n = m_link->pull(m_scratch.data(), m_block, timestamp);
614 if (n == 0) {
615 continue;
616 }
617 // A gating reader may hold the ring; the samples already pulled wait, later ones stay in LSL's buffer.
618 AcqError space = m_writer->waitForSpace(n, std::chrono::milliseconds(200));
619 while (space.code() == AcqErrorCode::Timeout && !m_stop.load(std::memory_order_acquire)) {
620 space = m_writer->waitForSpace(n, std::chrono::milliseconds(200));
621 }
622 if (!space.ok()) {
623 return;
624 }
625 auto view = m_writer->beginWrite(n);
626 if (!view) {
627 reportError(view.error());
628 return;
629 }
630 std::size_t done = 0;
631 for (std::size_t p = 0; p < view->partCount(); ++p) {
632 const AcqWritePart& part = view->part(p);
633 std::memcpy(part.at(0, 0), m_scratch.data() + done * m_frameBytes, part.samples * m_frameBytes);
634 done += part.samples;
635 }
636 m_writer->commit(n);
637 next += n;
638 const Clock::time_point now = Clock::now();
639 const bool refresh = lastClock == Clock::time_point{} || now - lastClock >= std::chrono::seconds(1);
640 if (refresh) {
641 publishClock(next - 1, timestamp, true);
642 lastClock = now;
643 }
644 }
645 }
