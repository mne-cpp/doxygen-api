93void MNESurface::triangle_coords(const Eigen::Vector3f& r, int tri, float& x, float& y, float& z) const
94{
95 double a, b, c, v1, v2, det;
96 const MNETriangle* this_tri;
97
98 this_tri = &tris[tri];
99
100 Eigen::Vector3d rDiff = (r - this_tri->r1).cast<double>();
101 z = rDiff.dot(this_tri->nn.cast<double>());
102
103 a = this_tri->r12.cast<double>().squaredNorm();
104 b = this_tri->r13.cast<double>().squaredNorm();
105 c = this_tri->r12.cast<double>().dot(this_tri->r13.cast<double>());
106
107 v1 = rDiff.dot(this_tri->r12.cast<double>());
108 v2 = rDiff.dot(this_tri->r13.cast<double>());
109
110 det = a * b - c * c;
111
112 x = (b * v1 - c * v2) / det;
113 y = (a * v2 - c * v1) / det;
114}

118int MNESurface::nearest_triangle_point(const Eigen::Vector3f& r, const MNEProjData* user, int tri, float& x, float& y, float& z) const
119{
120 double p, q, p0, q0, t0;
121 double a, b, c, v1, v2, det;
122 double best, distance, dist0;
123 const MNEProjData* pd = user;
124 const MNETriangle* this_tri;
125
126 this_tri = &tris[tri];
127 Eigen::Vector3d rDiff = (r - this_tri->r1).cast<double>();
128 distance = rDiff.dot(this_tri->nn.cast<double>());
129
130 if (pd) {
131 if (!pd->act[tri])
132 return false;
133 a = pd->a[tri];
134 b = pd->b[tri];
135 c = pd->c[tri];
136 } else {
137 a = this_tri->r12.cast<double>().squaredNorm();
138 b = this_tri->r13.cast<double>().squaredNorm();
139 c = this_tri->r12.cast<double>().dot(this_tri->r13.cast<double>());
140 }
141
142 v1 = rDiff.dot(this_tri->r12.cast<double>());
143 v2 = rDiff.dot(this_tri->r13.cast<double>());
144
145 det = a * b - c * c;
146
147 p = (b * v1 - c * v2) / det;
148 q = (a * v2 - c * v1) / det;
149
150 if (p >= 0.0 && p <= 1.0 &&
151 q >= 0.0 && q <= 1.0 &&
152 q <= 1.0 - p) {
153 x = p;
154 y = q;
155 z = distance;
156 return true;
157 }
158 /*
159 * Side 1 -> 2
160 */
161 p0 = p + 0.5 * (q * c) / a;
162 if (p0 < 0.0)
163 p0 = 0.0;
164 else if (p0 > 1.0)
165 p0 = 1.0;
166 q0 = 0.0;
167 dist0 = sqrt((p - p0) * (p - p0) * a +
168 (q - q0) * (q - q0) * b +
169 (p - p0) * (q - q0) * c +
170 distance * distance);
171 best = dist0;
172 x = p0;
173 y = q0;
174 z = dist0;
175 /*
176 * Side 2 -> 3
177 */
178 t0 = 0.5 * ((2.0 * a - c) * (1.0 - p) + (2.0 * b - c) * q) / (a + b - c);
179 if (t0 < 0.0)
180 t0 = 0.0;
181 else if (t0 > 1.0)
182 t0 = 1.0;
183 p0 = 1.0 - t0;
184 q0 = t0;
185 dist0 = sqrt((p - p0) * (p - p0) * a +
186 (q - q0) * (q - q0) * b +
187 (p - p0) * (q - q0) * c +
188 distance * distance);
189 if (dist0 < best) {
190 best = dist0;
191 x = p0;
192 y = q0;
193 z = dist0;
194 }
195 /*
196 * Side 1 -> 3
197 */
198 p0 = 0.0;
199 q0 = q + 0.5 * (p * c) / b;
200 if (q0 < 0.0)
201 q0 = 0.0;
202 else if (q0 > 1.0)
203 q0 = 1.0;
204 dist0 = sqrt((p - p0) * (p - p0) * a +
205 (q - q0) * (q - q0) * b +
206 (p - p0) * (q - q0) * c +
207 distance * distance);
208 if (dist0 < best) {
209 best = dist0;
210 x = p0;
211 y = q0;
212 z = dist0;
213 }
214 return true;
215}

245int MNESurface::project_to_surface(const MNEProjData* proj_data, const Eigen::Vector3f& r, float& distp) const
246{
247 float distance;
248 float p, q;
249 float dist0;
250 int best;
251 int k;
252
253 dist0 = 0.0;
254 for (best = -1, k = 0; k < ntri; k++) {
255 if (nearest_triangle_point(r, proj_data, k, p, q, distance)) {
256 if (best < 0 || std::fabs(distance) < std::fabs(dist0)) {
257 dist0 = distance;
258 best = k;
259 }
260 }
261 }
262 distp = dist0;
263 return best;
264}

268void MNESurface::find_closest_on_surface_approx(const PointsT& r, int np_points,
269 Eigen::VectorXi& nearest_tri,
270 Eigen::VectorXf& distances, int nstep) const
271{
272 auto p = std::make_unique<MNEProjData>(this);
273 int k;
274
275 mne::Log::info("{} for {} points {} steps...", nearest_tri[0] < 0 ? "Closest" : "Approx closest", np_points, nstep);
276
277 for (k = 0; k < np_points; k++) {
278 Eigen::Vector3f pt = Eigen::Map<const Eigen::Vector3f>(r.row(k).data());
279 decide_search_restriction(*p, nearest_tri[k], nstep, pt);
280 nearest_tri[k] = project_to_surface(p.get(), pt, distances[k]);
281 if (nearest_tri[k] < 0) {
282 decide_search_restriction(*p, -1, nstep, pt);
283 nearest_tri[k] = project_to_surface(p.get(), pt, distances[k]);
284 }
285 }
286
287 mne::Log::info("[done]\n");
288}

292void MNESurface::decide_search_restriction(MNEProjData& p,
293 int approx_best,
294 int nstep,
295 const Eigen::Vector3f& r) const
296{
297 int k;
298 Eigen::Vector3f diff_vec;
299 float dist_val, mindist;
300 int minvert;
301
302 for (k = 0; k < ntri; k++)
303 p.act[k] = 0;
304
305 if (approx_best < 0) {
306 mindist = 1000.0;
307 minvert = 0;
308 for (k = 0; k < np; k++) {
309 diff_vec = rr.row(k).transpose() - r;
310 dist_val = diff_vec.norm();
311 if (dist_val < mindist && nneighbor_tri[k] > 0) {
312 mindist = dist_val;
313 minvert = k;
314 }
315 }
316 } else {
317 const MNETriangle* this_tri = &tris[approx_best];
318 diff_vec = this_tri->r1 - r;
319 mindist = diff_vec.norm();
320 minvert = this_tri->vert[0];
321
322 diff_vec = this_tri->r2 - r;
323 dist_val = diff_vec.norm();
324 if (dist_val < mindist) {
325 mindist = dist_val;
326 minvert = this_tri->vert[1];
327 }
328 diff_vec = this_tri->r3 - r;
329 dist_val = diff_vec.norm();
330 if (dist_val < mindist) {
331 mindist = dist_val;
332 minvert = this_tri->vert[2];
333 }
334 }
335
336 activate_neighbors(minvert, p.act, nstep);
337
338 for (k = 0, p.nactive = 0; k < ntri; k++)
339 if (p.act[k])
340 p.nactive++;
341}

345void MNESurface::activate_neighbors(int start, Eigen::VectorXi& act, int nstep) const
346{
347 int k;
348
349 if (nstep == 0)
350 return;
351
352 for (k = 0; k < nneighbor_tri[start]; k++)
353 act[neighbor_tri[start][k]] = 1;
354 for (k = 0; k < nneighbor_vert[start]; k++)
355 activate_neighbors(neighbor_vert[start][k], act, nstep - 1);
356}

385std::unique_ptr<MNESurface> MNESurface::read_bem_surface(const std::filesystem::path& name, int which, bool add_geometry, float& sigma, bool check_too_many_neighbors)
386{
387 FiffStream stream(name);
388
389 std::vector<std::shared_ptr<FiffDirNode>> surfs;
390 std::vector<std::shared_ptr<FiffDirNode>> bems;
391 std::shared_ptr<FiffDirNode> node;
392 std::optional<FiffTag> t_pTag;
393
394 int id = -1;
395 int nnode, ntri_count;
396 MNESurface* s = nullptr;
397 std::unique_ptr<MNESurface> s_ptr;
398 int k;
399 MatrixXf tmp_node_normals;
400 int coord_frame = FIFFV_COORD_MRI;
401 float sigmaLocal = -1.0;
402 MatrixXf tmp_nodes;
403 MatrixXi tmp_triangles;
404
405 if (!stream.open())
406 return nullptr;
407
408 bems = stream.dirtree()->dir_tree_find(FIFFB_BEM);
409 if (bems.size() > 0) {
410 node = bems[0];
411 if (t_pTag = node->find_tag(stream, FIFF_BEM_COORD_FRAME); t_pTag) {
412 coord_frame = *t_pTag->to_int();
413 }
414 }
415 surfs = stream.dirtree()->dir_tree_find(FIFFB_BEM_SURF);
416 if (surfs.size() == 0) {
417 mne::Log::critical("No BEM surfaces found in {}", mne::utf8_from_path(name));
418 stream.close();
419 return nullptr;
420 }
421 if (which >= 0) {
422 for (k = 0; k < std::ssize(surfs); ++k) {
423 node = surfs[k];
424 if (t_pTag = node->find_tag(stream, FIFF_BEM_SURF_ID); t_pTag) {
425 id = *t_pTag->to_int();
426 if (id == which)
427 break;
428 }
429 }
430 if (id != which) {
431 mne::Log::critical("Desired surface not found in {}", mne::utf8_from_path(name));
432 stream.close();
433 return nullptr;
434 }
435 } else
436 node = surfs[0];
437
438 if (t_pTag = node->find_tag(stream, FIFF_BEM_SURF_NNODE); !t_pTag) {
439 stream.close();
440 return nullptr;
441 }
442 nnode = *t_pTag->to_int();
443
444 if (t_pTag = node->find_tag(stream, FIFF_BEM_SURF_NTRI); !t_pTag) {
445 stream.close();
446 return nullptr;
447 }
448 ntri_count = *t_pTag->to_int();
449
450 if (t_pTag = node->find_tag(stream, FIFF_BEM_SURF_NODES); !t_pTag) {
451 stream.close();
452 return nullptr;
453 }
454 tmp_nodes = t_pTag->to_float_matrix().transpose();
455
456 if (t_pTag = node->find_tag(stream, FIFF_BEM_SURF_NORMALS); t_pTag) {
457 tmp_node_normals = t_pTag->to_float_matrix().transpose();
458 }
459
460 if (t_pTag = node->find_tag(stream, FIFF_BEM_SURF_TRIANGLES); !t_pTag) {
461 stream.close();
462 return nullptr;
463 }
464 tmp_triangles = t_pTag->to_int_matrix().transpose();
465
466 if (t_pTag = node->find_tag(stream, FIFF_MNE_COORD_FRAME); t_pTag) {
467 coord_frame = *t_pTag->to_int();
468 } else if (t_pTag = node->find_tag(stream, FIFF_BEM_COORD_FRAME); t_pTag) {
469 coord_frame = *t_pTag->to_int();
470 }
471 if (t_pTag = node->find_tag(stream, FIFF_BEM_SIGMA); t_pTag) {
472 sigmaLocal = *t_pTag->to_float();
473 }
474
475 stream.close();
476
477 s_ptr = std::make_unique<MNESurface>();
478 s = s_ptr.get();
479 tmp_triangles.array() -= 1;
480 s->itris = tmp_triangles;
481 s->id = which;
482 s->sigma = sigmaLocal;
483 s->coord_frame = coord_frame;
484 s->rr = tmp_nodes;
485 if (tmp_node_normals.rows() > 0)
486 s->nn = tmp_node_normals;
487 s->ntri = ntri_count;
488 s->np = nnode;
489 s->type = FIFFV_MNE_SPACE_SURFACE;
490 s->nuse_tri = 0;
491 s->tot_area = 0.0;
492 s->dist_limit = -1.0;
493 s->vol_dims[0] = s->vol_dims[1] = s->vol_dims[2] = 0;
494 s->MRI_vol_dims[0] = s->MRI_vol_dims[1] = s->MRI_vol_dims[2] = 0;
495 s->cm[0] = s->cm[1] = s->cm[2] = 0.0;
496
497 if (add_geometry) {
498 if (check_too_many_neighbors) {
499 if (s->add_geometry_info(s->nn.rows() == 0) != OK) {
500 return nullptr;
501 }
502 } else {
503 if (s->add_geometry_info2(s->nn.rows() == 0) != OK) {
504 return nullptr;
505 }
506 }
507 } else if (s->nn.rows() == 0) {
508 if (s->add_vertex_normals() != OK) {
509 return nullptr;
510 }
511 } else
512 s->add_triangle_data();
513
514 s->nuse = s->np;
515 s->inuse = Eigen::VectorXi::Ones(s->np);
516 s->vertno = Eigen::VectorXi::LinSpaced(s->np, 0, s->np - 1);
517 sigma = sigmaLocal;
518
519 return s_ptr;
520}
