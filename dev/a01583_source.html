44static QString execModeToString(MnaNodeExecMode mode)
45{
46 switch (mode) {
47 case MnaNodeExecMode::Batch: return QStringLiteral("batch");
48 case MnaNodeExecMode::Stream: return QStringLiteral("stream");
49 case MnaNodeExecMode::Ipc: return QStringLiteral("ipc");
50 case MnaNodeExecMode::Script: return QStringLiteral("script");
51 }
52 return QStringLiteral("batch");
53}

57static MnaNodeExecMode execModeFromString(const QString& str)
58{
59 if (str == QLatin1String("stream")) return MnaNodeExecMode::Stream;
60 if (str == QLatin1String("ipc")) return MnaNodeExecMode::Ipc;
61 if (str == QLatin1String("script")) return MnaNodeExecMode::Script;
62 return MnaNodeExecMode::Batch;

103 json[QLatin1String("script")] = script.toJson();

105

107 QJsonObject verObj = verification.toJson();
108 if (!verObj.isEmpty()) {
109 json[QLatin1String("verification")] = verObj;

114 json[QLatin1String("tool_version")] = toolVersion;
115 if (executedAt.isValid())

117 json[QLatin1String("dirty")] = dirty;
