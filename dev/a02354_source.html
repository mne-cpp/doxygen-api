533inline std::int32_t FiffCoordTrans::storage_size()
534{
535 // On-disk layout: from, to, rot[3][3], move[3], invrot[3][3], invmove[3]
536 // (C++ class uses Matrix4f but on-disk stores 3x3 rot + 3-vec separately)
537 return 2 * sizeof(fiff_int_t) + 2 * 12 * sizeof(fiff_float_t);
538}
