56std::string randomUuid()
57{
58 static std::mt19937_64 generator{std::random_device{}()};
59 static std::mutex mutex;
60 std::array<std::uint64_t, 2> words{};
61 {
62 const std::lock_guard lock(mutex);
63 words = {generator(), generator()};
64 }
65 words[0] = (words[0] & ~std::uint64_t{0xF000}) | std::uint64_t{0x4000}; // version 4
66 words[1] = (words[1] & ~(std::uint64_t{0xC} << 60)) | (std::uint64_t{0x8} << 60); // RFC 4122 variant
67 return std::format("{:08x}-{:04x}-{:04x}-{:04x}-{:012x}", words[0] >> 32, (words[0] >> 16) & 0xFFFF, words[0] & 0xFFFF,
68 words[1] >> 48, words[1] & 0xFFFFFFFFFFFFULL);
69}

71std::string escaped(const std::string& text)
72{
73 std::string out;
74 out.reserve(text.size());
75 for (const char c : text) {
76 switch (c) {
77 case '&':
78 out += "&amp;";
79 break;
80 case '<':
81 out += "&lt;";
82 break;
83 case '>':
84 out += "&gt;";
85 break;
86 case '"':
87 out += "&quot;";
88 break;
89 case '\'':
90 out += "&apos;";
91 break;
92 default:
93 out += c;
94 break;
95 }
96 }
97 return out;
98}

281std::string stream_info::xml(bool full) const
282{
283 std::ostringstream xml;
284 xml << "<?xml version=\"1.0\"?>\n<info>\n"
285 << "\t<name>" << escaped(m_name) << "</name>\n"
286 << "\t<type>" << escaped(m_type) << "</type>\n"
287 << "\t<channel_count>" << m_channel_count << "</channel_count>\n"
288 << "\t<channel_format>" << formatName(m_channel_format) << "</channel_format>\n"
289 << "\t<source_id>" << escaped(m_source_id) << "</source_id>\n"
290 << "\t<nominal_srate>" << number(m_nominal_srate) << "</nominal_srate>\n"
291 << "\t<version>1.1</version>\n"
292 << "\t<created_at>" << number(m_created_at) << "</created_at>\n"
293 << "\t<uid>" << escaped(m_uid) << "</uid>\n"
294 << "\t<session_id>default</session_id>\n"
295 << "\t<hostname>" << escaped(m_hostname) << "</hostname>\n"
296 << "\t<v4address></v4address>\n"
297 << "\t<v4data_port>" << m_data_port << "</v4data_port>\n"
298 << "\t<v4service_port>" << m_service_port << "</v4service_port>\n"
299 << "\t<v6address></v6address>\n"
300 << "\t<v6data_port>0</v6data_port>\n"
301 << "\t<v6service_port>0</v6service_port>\n";
302 if (!full || m_channels.empty()) {
303 xml << "\t<desc />\n</info>\n";
304 return xml.str();
305 }
306 xml << "\t<desc>\n\t\t<channels>\n";
307 for (const channel_description& channel : m_channels) {
308 xml << "\t\t\t<channel>\n"
309 << "\t\t\t\t<label>" << escaped(channel.label) << "</label>\n"
310 << "\t\t\t\t<type>" << escaped(channel.type) << "</type>\n"
311 << "\t\t\t\t<unit>" << escaped(channel.unit) << "</unit>\n"
312 << "\t\t\t</channel>\n";
313 }
314 xml << "\t\t</channels>\n\t</desc>\n</info>\n";
315 return xml.str();
316}

320stream_info stream_info::from_string(const std::string& data)
321{
322 stream_info info;
323 const Expected<XmlElement> root = parse_xml(data);
324 if (!root || root->local_name() != "info") {
325 return {};
326 }
327 for (const XmlElement& element : root->children) {
328 const std::string_view key = element.local_name();
329 const std::string& value = element.text;
330 if (key == "name") {
331 info.m_name = value;
332 } else if (key == "type") {
333 info.m_type = value;
334 } else if (key == "channel_count") {
335 info.m_channel_count = to_int(value).value_or(0);
336 } else if (key == "channel_format") {
337 info.m_channel_format = formatFromName(value);
338 } else if (key == "source_id") {
339 info.m_source_id = value;
340 } else if (key == "nominal_srate") {
341 info.m_nominal_srate = to_double(value).value_or(0.0);
342 } else if (key == "created_at") {
343 info.m_created_at = to_double(value).value_or(0.0);
344 } else if (key == "uid") {
345 info.m_uid = value;
346 } else if (key == "hostname") {
347 info.m_hostname = value;
348 } else if (key == "v4data_port") {
349 info.m_data_port = to_int(value).value_or(0);
350 } else if (key == "v4service_port") {
351 info.m_service_port = to_int(value).value_or(0);
352 } else if (key == "v4address" && !value.empty()) {
353 info.m_data_host = value;
354 } else if (key == "desc") {
355 if (const XmlElement* channels = element.child("channels")) {
356 for (const XmlElement& channel : channels->children) {
357 if (channel.local_name() != "channel") {
358 continue;
359 }
360 channel_description description;
361 for (const XmlElement& property : channel.children) {
362 const std::string_view tag = property.local_name();
363 if (tag == "label") {
364 description.label = property.text;
365 } else if (tag == "type") {
366 description.type = property.text;
367 } else if (tag == "unit") {
368 description.unit = property.text;
369 }
370 }
371 info.m_channels.push_back(std::move(description));
372 }
373 }
374 }
375 }
376 if (info.m_name.empty()) {
377 return {};
378 }
379 return info;
380}
