44Expected<FineCalibration> FineCalibration::read(const std::filesystem::path& path)
45{
46 const Expected<std::string> text = read_text_file(path);
47 if (!text) {
48 return std::unexpected(text.error());
49 }
50 FineCalibration cal;
51 std::string_view rest = *text;
52 while (!rest.empty()) {
53 const std::size_t end = rest.find('\n');
54 const std::string_view line = rest.substr(0, end);
55 rest = end == std::string_view::npos ? std::string_view() : rest.substr(end + 1);
56 const std::vector<std::string_view> parts = split_whitespace(line);
57 if (parts.empty() || parts.front().starts_with('#')) {
58 continue;
59 }
60 std::string_view name = parts.front();
61 if (name.starts_with("MEG")) {
62 name.remove_prefix(3);
63 }
64 FineCalEntry entry;
65 VectorXd values(std::ssize(parts) - 1);
66 std::optional<int> number = to_int(name);
67 bool ok = number.has_value() && (parts.size() == 14 || parts.size() == 16);
68 for (Index k = 0; ok && k < values.size(); ++k) {
69 const std::optional<double> value = to_double(parts[static_cast<std::size_t>(k) + 1]);
70 ok = value.has_value();
71 values(k) = value.value_or(0.0);
72 }
73 if (!ok) {
74 return fail(ErrorCode::Format, "{} has no fine calibration of 14 or 16 values in line \"{}\"",
75 utf8_from_path(path), line);
76 }
77 entry.chNumber = *number;
78 entry.position = values.head<3>();
79 entry.orientation = Map<const Matrix<double, 3, 3, RowMajor>>(values.data() + 3);
80 entry.imbalance = values.tail(values.size() - 12);
81 cal.push_back(entry);
82 }
83 return cal;
84}

88Expected<void> FineCalibration::write(const std::filesystem::path& path) const
89{
90 std::string text;
91 for (const FineCalEntry& entry : m_entries) {
92 std::format_to(std::back_inserter(text), "{}", entry.chNumber);
93 const Matrix<double, 3, 3, RowMajor> axes = entry.orientation;
94 const auto put = [&text](double value) {
95 std::format_to(std::back_inserter(text), " {:.6f}", value);
96 };
97 for (Index k = 0; k < 3; ++k) {
98 put(entry.position(k));
99 }
100 for (Index k = 0; k < 9; ++k) {
101 put(axes.data()[k]);
102 }
103 for (Index k = 0; k < entry.imbalance.size(); ++k) {
104 put(entry.imbalance(k));
105 }
106 text += '\n';
107 }
108 return write_text_file(path, text);
109}
