64 return run;
65 }

67 const double primaryScale = pixel_size / primary.units_per_em();
68 run.ascent = std::round(primary.ascender() * primaryScale);
69 run.descent = std::round(-primary.descender() * primaryScale);
70
71 double pen = 0.0;
72 const FontFace* previousFace = nullptr;
73 std::uint32_t previous = 0;
74 std::shared_ptr<const FontFace> lastFallback;
75 char32_t lastFallbackFor = 0;

78 std::uint32_t glyph = primary.glyph_index(c);

81 lastFallback = fonts.fallback_face(font, c);
82 lastFallbackFor = c;
83 }

86 face = lastFallback;
87 glyph = other;
88 }
89 }
90 }
