300template<typename T>
301void Spline::updatePlot(const Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& matClassLimitData,
302 const Eigen::VectorXi& matClassFrequencyData)
303{
304 int iClassAmount = std::min(static_cast<int>(matClassFrequencyData.rows()),
305 static_cast<int>(matClassLimitData.rows()) - 1);
306 if (iClassAmount < 1)
307 return;
308 m_dMinAxisX = static_cast<double>(matClassLimitData(0));
309 m_dMaxAxisX = static_cast<double>(matClassLimitData(iClassAmount));
310
311 m_seriesData.clear();
312 m_iMaximumFrequency = 0;
313
314 for (int ir = 0; ir < iClassAmount; ++ir) {
315 double classMark = (static_cast<double>(matClassLimitData(ir)) + static_cast<double>(matClassLimitData(ir + 1))) / 2.0;
316 m_seriesData.append(QPointF(classMark, matClassFrequencyData(ir)));
317 if (matClassFrequencyData(ir) > m_iMaximumFrequency) {
318 m_iMaximumFrequency = matClassFrequencyData(ir);
319 }
320 }
321
322 m_bHasData = true;
323 m_dLeftThreshold = m_dMinAxisX;
324 m_dMiddleThreshold = m_dMaxAxisX;
325 m_dRightThreshold = m_dMaxAxisX;
326 m_bHasThresholds = false;
327
328 QWidget::update();
329}
