100void ConnectivityPanel::set_trigger_types(const std::vector<std::string>& types)
101{
102 std::vector<SettingChoice> choices = form().field(kTrigger)->choices;
103 const std::size_t count = choices.size();
104 for (const std::string& type : types) {
105 if (std::ranges::none_of(choices, [&type](const SettingChoice& choice) { return choice.value == type; })) {
106 choices.push_back({type, type});
107 }
108 }
109 if (choices.size() != count) {
110 form().set_choices(kTrigger, std::move(choices));
111 }
112}

186void ConnectivityPanel::changed(const std::string& key)
187{
188 if (key == kMetric) {
189 if (m_callbacks.metric_changed) {
190 m_callbacks.metric_changed(metric());
191 }
192 } else if (key == kWindow) {
193 if (m_callbacks.window_type_changed) {
194 m_callbacks.window_type_changed(window_type());
195 }
196 } else if (key == kTrials) {
197 if (number_trials() != m_trials) {
198 m_trials = number_trials();
199 if (m_callbacks.number_trials_changed) {
200 m_callbacks.number_trials_changed(m_trials);
201 }
202 }
203 } else if (key == kTrigger) {
204 if (m_callbacks.trigger_type_changed) {
205 m_callbacks.trigger_type_changed(trigger_type());
206 }
207 } else if (key == kLow || key == kHigh) {
208 if (m_callbacks.frequency_band_changed) {
209 m_callbacks.frequency_band_changed(static_cast<float>(lower_frequency()), static_cast<float>(upper_frequency()));
210 }
211 }
