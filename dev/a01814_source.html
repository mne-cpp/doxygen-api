95ArtifactPanel::ArtifactPanel(const std::vector<mne::FiffChInfo>& channels)
96: Panel("ArtifactSettingsView", "Artifact Rejection Settings")
97{
98 set_width(330);
99 SettingsForm& f = form();
100 f.add_flag(kHasTypes);
101 SettingField activeField;
102 activeField.key = kActive;
103 activeField.kind = SettingField::Kind::Bool;
104 activeField.label = "Activate artifact rejection";
105 activeField.default_value = false;
106 activeField.visible_if = kHasTypes;
107 f.add(std::move(activeField));
108 for (const std::string type : kChannelTypes) {
109 f.add_flag(present_key(type));
110 SettingField mantissa;
111 mantissa.key = mantissa_key(type);
112 mantissa.kind = SettingField::Kind::Double;
113 mantissa.label = type;
114 mantissa.default_value = 1.0;
115 mantissa.maximum = 100000.0;
116 mantissa.prefix = "+/-";
117 mantissa.visible_if = present_key(type);
118 f.add(std::move(mantissa));
119 SettingField exponent;
120 exponent.key = exponent_key(type);
121 exponent.kind = SettingField::Kind::Int;
122 exponent.default_value = -1;
123 exponent.minimum = -10000;
124 exponent.maximum = 0;
125 exponent.decimals = 0;
126 exponent.prefix = "e";
127 exponent.visible_if = present_key(type);
128 f.add(std::move(exponent));
129 }
130 m_listener = f.add_listener([this](const std::string& key, const std::optional<SettingValue>&) {
131 if (!m_updating && (key == kActive || key.starts_with("artifactThresholds"))) {
132 report();
133 }
134 });
135 const bool updating = std::exchange(m_updating, true);
136 set_channels(channels);
137 m_updating = updating;

149void ArtifactPanel::set_channels(const std::vector<mne::FiffChInfo>& channels)
150{
151 std::vector<std::string> present;
152 present.reserve(kChannelTypes.size());
153 for (const mne::FiffChInfo& channel : channels) {
154 if (const char* type = channel_type(channel)) {
155 present.emplace_back(type);
156 }
157 }
158 {
159 const bool updating = std::exchange(m_updating, true);
160 for (const std::string type : kChannelTypes) {
161 form().set_value(present_key(type), std::ranges::find(present, type) != present.end());
162 }
163 form().set_value(kHasTypes, !present.empty());
164 m_updating = updating;
165 }
166 if (!m_updating) {
167 report();
168 }
169}

173std::map<std::string, double> ArtifactPanel::thresholds() const
174{
175 std::map<std::string, double> result;
176 result["Active"] = rejection_active() ? 1.0 : 0.0;
177 for (const std::string type : kChannelTypes) {
178 if (form().get<bool>(present_key(type))) {
179 result[type] = form().get<double>(mantissa_key(type)) * std::pow(10.0, form().get<int>(exponent_key(type)));
180 }
181 }
182 return result;
183}
