236void unitOf(const std::string& unit, AcqChannelInfo& channel)
237{
238 const std::string u = lower(unit);
239 struct Entry
240 {
241 const char* name;
242 AcqUnit unit;
243 int exponent;
244 };
245 static constexpr Entry kUnits[] = {
246 {"microvolts", AcqUnit::Volt, -6},
247 {"microvolt", AcqUnit::Volt, -6},
248 {"uv", AcqUnit::Volt, -6},
249 {"\xc2\xb5v", AcqUnit::Volt, -6},
250 {"millivolts", AcqUnit::Volt, -3},
251 {"mv", AcqUnit::Volt, -3},
252 {"volts", AcqUnit::Volt, 0},
253 {"volt", AcqUnit::Volt, 0},
254 {"v", AcqUnit::Volt, 0},
255 {"femtotesla", AcqUnit::Tesla, -15},
256 {"ft", AcqUnit::Tesla, -15},
257 {"tesla", AcqUnit::Tesla, 0},
258 {"t", AcqUnit::Tesla, 0},
259 };
260 for (const Entry& entry : kUnits) {
261 if (u == entry.name) {
262 channel.unit = entry.unit;
263 channel.unitExponent = entry.exponent;
264 return;
265 }
266 }
267 if (channel.kind == AcqChannelKind::Stim) {
268 channel.unit = AcqUnit::None;

286AcqExpected<Config> parseConfig(const AcqUri& uri)
287{
288 Config c;
289 c.name = uri.target;
290 c.type = std::string(uri.param("type").value_or(""));
291 c.sourceId = std::string(uri.param("source_id").value_or(""));
292 c.library = std::string(uri.param("library").value_or("auto"));
293 const auto timeout = paramDouble(uri, "timeout", 5.0, 0.0, 3600.0);
294 auto thread = paramThreadPolicy(uri);
295 const auto simulate = paramBool(uri, "simulate", false);
296 const auto channels = paramInt(uri, "channels", 8, 1, 1024);
297 const auto sfreq = paramDouble(uri, "sfreq", 250.0, 1.0, 100000.0);
298 const auto seed = paramInt(uri, "seed", 0, 0, std::numeric_limits<std::int32_t>::max());
299 for (const AcqError& error : {timeout.error(), thread.error(), simulate.error(), channels.error(), sfreq.error(), seed.error()}) {
300 if (!error.ok()) {
301 return error;
302 }
303 }
304 if (c.library != "auto" && c.library != "liblsl" && c.library != "intree") {
305 return AcqError(AcqErrorCode::InvalidParameter, "library=" + c.library + ": use auto, liblsl or intree");
306 }
307 c.simulate = *simulate;
308 c.channels = static_cast<std::size_t>(*channels);
309 c.sfreq = *sfreq;
310 c.seed = *seed;
311 if (c.simulate) {
312 // Resolved by a source id of its own, so a real stream of the same name is never read instead
313 static std::atomic<unsigned> count{0};
314 c.name = c.name.empty() ? "MNE-CPP simulator" : c.name;
315 c.type = "EEG";
316 c.sourceId = std::format("mne-acq-simulator-{}-{}", Clock::now().time_since_epoch().count(), count++);
317 }
318 if (c.name.empty() && c.type.empty() && c.sourceId.empty()) {
319 return AcqError(AcqErrorCode::InvalidParameter, "lsl:// needs a stream name, type= or source_id=, e.g. lsl://BioSemi");
320 }
321 c.timeout = *timeout;
322 c.thread = std::move(*thread);
323 return c;
324}

327std::pair<std::string, std::string> selector(const Config& c)
328{
329 if (c.simulate) {
330 return {"source_id", c.sourceId};
331 }
332 if (!c.name.empty()) {
333 return {"name", c.name};
334 }
335 if (!c.sourceId.empty()) {
336 return {"source_id", c.sourceId};
337 }
338 return {"type", c.type};
339}

356public:
357 LiblslLink(const Liblsl& lsl, Liblsl::Inlet inlet, int format, std::size_t channels)
358 : m_lsl(lsl)
359 , m_inlet(inlet)
360 , m_format(format)
361 , m_channels(channels)
362 {
363 }
364
365 ~LiblslLink() override
366 {
367 m_lsl.destroyInlet(m_inlet);
368 }
369
370 LiblslLink(const LiblslLink&) = delete;
371 LiblslLink& operator=(const LiblslLink&) = delete;
372
373 AcqExpected<std::size_t> pull(std::byte* out, std::size_t frames, double& lastTimestamp) override
374 {
375 m_stamps.resize(frames);
376 const unsigned long elements = static_cast<unsigned long>(frames * m_channels);
377 int ec = 0;
378 unsigned long got = 0;
379 switch (m_format) {
380 case kFloat32:
381 got = m_lsl.pullF(m_inlet, reinterpret_cast<float*>(out), m_stamps.data(), elements, static_cast<unsigned long>(frames), 0.0, &ec);
382 break;
383 case kDouble64:
384 case kInt64:
385 got = m_lsl.pullD(m_inlet, reinterpret_cast<double*>(out), m_stamps.data(), elements, static_cast<unsigned long>(frames), 0.0, &ec);
386 break;
387 case kInt32:
388 got = m_lsl.pullI(m_inlet, reinterpret_cast<std::int32_t*>(out), m_stamps.data(), elements, static_cast<unsigned long>(frames), 0.0, &ec);
389 break;
390 default:
391 got = m_lsl.pullS(m_inlet, reinterpret_cast<std::int16_t*>(out), m_stamps.data(), elements, static_cast<unsigned long>(frames), 0.0, &ec);
392 break;
393 }
394 if (ec == kLostError) {
395 return AcqError(AcqErrorCode::DeviceIo, "the LSL stream was lost");
396 }
397 const std::size_t n = got / m_channels;
398 if (n > 0) {
399 lastTimestamp = m_stamps[n - 1];
400 return n;
401 }
402 std::this_thread::sleep_for(std::chrono::milliseconds(1)); // nothing buffered yet
403 return std::size_t(0);
404 }
405
406 double timeCorrection() override
407 {
408 int ec = 0;
409 const double correction = m_lsl.timeCorrection(m_inlet, 2.0, &ec);
410 if (ec == 0) {
411 m_correction = correction;
412 }
413 return m_correction;
414 }
415
416 double localClock() const override
417 {
418 return m_lsl.localClock();
419 }
420
421private:
422 const Liblsl& m_lsl;
423 Liblsl::Inlet m_inlet;
424 int m_format;
425 std::size_t m_channels;
426 std::vector<double> m_stamps;
427 double m_correction = 0.0;
428};

459 {
460 if (m_pending.empty()) {
461 if (m_inlet.samples_available()) {
462 m_pending = m_inlet.pull_chunk_float(m_stamps);
463 m_next = 0;
464 }
465 if (m_pending.empty()) {
466 std::this_thread::sleep_for(std::chrono::milliseconds(1));
467 return std::size_t(0);
468 }
469 }
470 std::size_t n = 0;
471 for (; n < frames && m_next < m_pending.size(); ++n, ++m_next) {
472 const std::vector<float>& sample = m_pending[m_next];
473 for (std::size_t c = 0; c < m_channels; ++c) {
474 const float v = c < sample.size() ? sample[c] : 0.0F;
475 std::memcpy(out + (n * m_channels + c) * sizeof(float), &v, sizeof(float));
476 }
477 }
478 lastTimestamp = m_stamps[m_next - 1];
479 if (m_next == m_pending.size()) {
480 m_pending.clear();
481 }
482 return n;
483 }
484
485 double timeCorrection() override
486 {
487 const double correction = m_inlet.time_correction(2.0);
488 if (correction == correction) {
489 m_correction = correction;
490 }
491 return m_correction;
492 }
493
494 double localClock() const override
495 {
496 return mne::local_clock();
497 }
498
499private:

501 std::size_t m_channels;
502 std::vector<std::vector<float>> m_pending;
503 std::vector<double> m_stamps;
504 std::size_t m_next = 0;
505 double m_correction = 0.0;

511class LslSimulator
512{
513public:
514 static AcqExpected<std::unique_ptr<LslSimulator>> start(const Config& c)
515 {
516 const auto block = static_cast<std::size_t>(std::max(1.0, std::round(c.sfreq / 100.0)));
517 auto session = AcqDevice::open(
518 std::format("synthetic://?sfreq={}&eeg={}&stim=0&block={}&seed={}&realtime=1", c.sfreq, c.channels, block, c.seed));
519 if (!session) {
520 return session.error();
521 }
522 AcqReaderOptions gating;
523 gating.gating = true;
524 auto reader = session->stream().reader(AcqReader::ReadMode::Lossless, gating);
525 if (!reader) {
526 return reader.error();
527 }
528 std::unique_ptr<LslSimulator> simulator(new LslSimulator(
529 c, block, std::move(*session), std::move(*reader)));
530 if (AcqError e = simulator->m_session.start(); !e.ok()) {
531 return e;
532 }
533 simulator->m_thread = std::thread([s = simulator.get()] { s->run(); });
534 return simulator;
535 }
536
537 ~LslSimulator()
538 {
539 m_stop.store(true, std::memory_order_release);
540 if (m_thread.joinable()) {
541 m_thread.join();
542 }
543 m_session.close();
544 }
545
546 LslSimulator(const LslSimulator&) = delete;
547 LslSimulator& operator=(const LslSimulator&) = delete;
548
549private:
550 LslSimulator(const Config& c, std::size_t block, AcqDevice session, AcqReader reader)
551 : m_outlet(mne::stream_info(c.name, c.type, static_cast<int>(c.channels), c.sfreq, mne::ChannelFormat::Float32, c.sourceId))
552 , m_session(std::move(session))
553 , m_reader(std::move(reader))
554 , m_chunk(block, std::vector<float>(c.channels))
555 {
556 }
557
558 void run()
559 {
560 while (!m_stop.load(std::memory_order_acquire)) {
561 auto view = m_reader.acquire(m_chunk.size(), std::chrono::milliseconds(200), m_chunk.size());
562 if (!view) {
563 if (view.error().code() == AcqErrorCode::Timeout) {
564 continue;
565 }
566 return;
567 }
568 std::size_t s = 0;
569 for (std::size_t p = 0; p < view->partCount(); ++p) {
570 const AcqBlockPart& part = view->part(p);
571 for (std::size_t i = 0; i < part.samples; ++i, ++s) {
572 for (std::size_t c = 0; c < m_chunk[s].size(); ++c) {
573 std::memcpy(&m_chunk[s][c], part.at(c, i), sizeof(float));
574 }
575 }
576 }
577 const bool whole = s == m_chunk.size();
578 if (!m_reader.release(*view).ok() || !whole) {
579 return;
580 }
581 m_outlet.push_chunk(m_chunk);
582 }
583 }
584
585 mne::stream_outlet m_outlet;
586 AcqDevice m_session;
587 AcqReader m_reader;
588 std::vector<std::vector<float>> m_chunk;
589 std::thread m_thread;
590 std::atomic<bool> m_stop{false};
591};

663 AcqError connect(const Liblsl& lsl)
664 {
665 const auto [property, value] = selector(m_config);
666 Liblsl::Info found = nullptr;
667 if (lsl.resolveByProp(&found, 1, property.c_str(), value.c_str(), 1, m_config.timeout) < 1 || found == nullptr) {
668 return {AcqErrorCode::DeviceNotFound, "no LSL stream with " + property + "=" + value + " within " + std::to_string(m_config.timeout) + " s"};
669 }
670 Liblsl::Inlet inlet = lsl.createInlet(found, 360, 0, 1);
671 lsl.destroyInfo(found);
672 if (inlet == nullptr) {
673 return {AcqErrorCode::DeviceIo, "cannot create an LSL inlet"};
674 }
675 int ec = 0;
676 Liblsl::Info info = lsl.fullInfo(inlet, std::max(m_config.timeout, 1.0), &ec);
677 if (info == nullptr || ec != 0) {
678 lsl.destroyInlet(inlet);
679 return {AcqErrorCode::DeviceIo, "cannot read the LSL stream description"};
680 }
681 AcqStreamInfo stream;
682 AcqError described = describe(lsl, info, stream);
683 const int format = lsl.channelFormat(info);
684 lsl.destroyInfo(info);
685 if (!described.ok()) {
686 lsl.destroyInlet(inlet);
687 return described;
688 }
689 lsl.openStream(inlet, std::max(m_config.timeout, 1.0), &ec);
690 if (ec != 0) {

699 AcqError describe(const Liblsl& lsl, Liblsl::Info info, AcqStreamInfo& stream)
700 {
701 const int format = lsl.channelFormat(info);
702 switch (format) {
703 case kFloat32:
704 stream.format = AcqSampleFormat::Float32;
705 break;
706 case kDouble64:
707 case kInt64:
708 stream.format = AcqSampleFormat::Float64;
709 break;
710 case kInt32:
711 stream.format = AcqSampleFormat::Int32;
712 break;
713 case kInt16:
714 case kInt8:
715 stream.format = AcqSampleFormat::Int16;
716 break;
717 case kString:
718 return {AcqErrorCode::UnsupportedFormat, "string (marker) LSL streams are not supported yet"};
719 default:
720 return {AcqErrorCode::UnsupportedFormat, "LSL channel format " + std::to_string(format)};
721 }
722 stream.sfreq = lsl.nominalSrate(info);
723 if (stream.sfreq <= 0.0) {
724 return {AcqErrorCode::UnsupportedFormat, "irregular-rate LSL streams are not supported yet"};
725 }
726 stream.name = lsl.name(info);
727 stream.sourceId = lsl.sourceId(info);
728 m_device = stream.name + " (" + lsl.type(info) + ") on " + lsl.hostname(info);
729 stream.clockDomain = std::string("lsl:") + lsl.sourceId(info);
730 const std::string streamType = lsl.type(info);
731 const auto count = static_cast<std::size_t>(std::max(0, lsl.channelCount(info)));
732 Liblsl::Xml channel = lsl.child(lsl.child(lsl.desc(info), "channels"), "channel");
733 for (std::size_t c = 0; c < count; ++c) {
734 AcqChannelInfo ch;
735 const bool described = lsl.empty(channel) == 0;
736 const std::string label = described ? lsl.childValue(channel, "label") : "";
737 const std::string type = described ? lsl.childValue(channel, "type") : "";
738 ch.name = label.empty() ? streamType + " " + std::to_string(c + 1) : label;
739 ch.kind = kindOf(type.empty() ? streamType : type);
740 unitOf(described ? lsl.childValue(channel, "unit") : "", ch);
741 stream.channels.push_back(std::move(ch));
742 if (described) {
743 channel = lsl.nextSiblingNamed(channel, "channel");
744 }
745 }
746 return {};
747 }

810 std::pair<std::int64_t, double> sameInstant() const
811 {
812 std::pair<std::int64_t, double> best{0, std::numeric_limits<double>::quiet_NaN()};
813 double width = std::numeric_limits<double>::infinity();
814 for (int attempt = 0; attempt < 3; ++attempt) {
815 const double before = m_link->localClock();
816 const auto steady = std::chrono::duration_cast<std::chrono::nanoseconds>(Clock::now().time_since_epoch()).count();
817 const double after = m_link->localClock();
818 if (!(after - before < width)) {
819 continue;
820 }
821 width = after - before;
822 best = {steady, 0.5 * (before + after)};
823 }
824 if (width == std::numeric_limits<double>::infinity()) { // no LSL clock
825 best.first = std::chrono::duration_cast<std::chrono::nanoseconds>(Clock::now().time_since_epoch()).count();
826 }
827 return best;
828 }
829
830 void publishClock(std::uint64_t sample, double timestamp, bool refresh)
831 {
832 const auto [steady, local] = sameInstant();
833 AcqClockModel clock;
834 clock.sfreq = m_sfreq;
835 clock.anchorSample = sample;
836 clock.utcOffsetNs =
837 std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::system_clock::now().time_since_epoch())
838 .count() -
839 steady;
840 if (timestamp == timestamp && local == local) {
841 if (refresh) {
842 m_correction = m_link->timeCorrection();
843 }
844 // LSL time to steady_clock: both are monotonic, but need not share an epoch.
845 clock.anchorHostNs = steady + static_cast<std::int64_t>((timestamp + m_correction - local) * 1e9);
846 clock.source = AcqClockSource::Lsl;
847 } else {
848 clock.anchorHostNs = steady;
849 clock.source = AcqClockSource::Host;
850 }
851 m_writer->setClock(clock);
852 }
853
854 void run()
855 {
856 std::uint64_t next = 0;
857 Clock::time_point lastClock;
858 while (!m_stop.load(std::memory_order_acquire)) {
859 double timestamp = std::numeric_limits<double>::quiet_NaN();
860 auto got = m_link->pull(m_scratch.data(), m_block, timestamp);
861 if (!got) {
862 reportError(got.error());
863 return;
864 }
865 const std::size_t n = *got;
866 if (n == 0) {
867 continue;
868 }
869 // A gating reader may hold the ring; the samples already pulled wait, later ones stay in LSL's buffer.
870 AcqError space = m_writer->waitForSpace(n, std::chrono::milliseconds(200));
871 while (space.code() == AcqErrorCode::Timeout && !m_stop.load(std::memory_order_acquire)) {
872 space = m_writer->waitForSpace(n, std::chrono::milliseconds(200));
873 }
874 if (!space.ok()) {
875 return;
876 }
877 auto view = m_writer->beginWrite(n);
878 if (!view) {
879 reportError(view.error());
880 return;
881 }
882 std::size_t done = 0;
883 for (std::size_t p = 0; p < view->partCount(); ++p) {
884 const AcqWritePart& part = view->part(p);
885 std::memcpy(part.at(0, 0), m_scratch.data() + done * m_frameBytes, part.samples * m_frameBytes);
886 done += part.samples;
887 }
888 m_writer->commit(n);
889 next += n;
890 const Clock::time_point now = Clock::now();
891 const bool refresh = lastClock == Clock::time_point{} || now - lastClock >= std::chrono::seconds(1);
892 if (refresh) {
893 publishClock(next - 1, timestamp, true);
894 lastClock = now;
895 }
896 }
897 }
