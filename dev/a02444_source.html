81bool fits(const AcqControlInfo& info, AcqValueType type, const AcqValue& value)
82{
83 switch (type) {
84 case AcqValueType::None:
85 return std::holds_alternative<std::monostate>(value);
86 case AcqValueType::Bool:
87 return std::holds_alternative<bool>(value);
88 case AcqValueType::Int:
89 return std::holds_alternative<std::int64_t>(value) &&
90 static_cast<double>(std::get<std::int64_t>(value)) >= info.minimum &&
91 static_cast<double>(std::get<std::int64_t>(value)) <= info.maximum;
92 case AcqValueType::Double: {
93 if (!std::holds_alternative<double>(value) && !std::holds_alternative<std::int64_t>(value)) {
94 return false;
95 }
96 const double number = std::holds_alternative<double>(value) ? std::get<double>(value)
97 : static_cast<double>(std::get<std::int64_t>(value));
98 return number >= info.minimum && number <= info.maximum; // NaN fails
99 }
100 case AcqValueType::String:
101 return std::holds_alternative<std::string>(value);
102 case AcqValueType::Enum:
103 return std::holds_alternative<std::string>(value) &&
104 std::find(info.options.begin(), info.options.end(), std::get<std::string>(value)) != info.options.end();
105 }
106 return false;
107}
108
110AcqValue normalized(AcqValueType type, const AcqValue& value)
111{
112 if (type == AcqValueType::Double && std::holds_alternative<std::int64_t>(value)) {
113 return static_cast<double>(std::get<std::int64_t>(value));
114 }
115 return value;
116}

119bool allowed(std::uint8_t when, AcqDeviceState state)
120{
121 switch (state) {
122 case AcqDeviceState::Opened:
123 return (when & AcqWhenOpened) != 0;
124 case AcqDeviceState::Running:
125 return (when & AcqWhenRunning) != 0;
126 case AcqDeviceState::Stopped:
127 return (when & AcqWhenStopped) != 0;
128 default:
129 return false;
130 }

174AcqError AcqDevice::start()
175{
176 if (!m_pImpl) {
177 return AcqErrorCode::InvalidState;
178 }
179 std::lock_guard<std::mutex> control(m_pImpl->controlMutex); // a control call never sees the state change under it
180 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
181 const AcqDeviceState state = m_pImpl->effectiveState();
182 if (state != AcqDeviceState::Opened && state != AcqDeviceState::Stopped) {
183 return AcqErrorCode::InvalidState;
184 }
185 AcqError error = m_pImpl->backend->start();
186 if (!error.ok()) {
187 m_pImpl->lastError = error;
188 return error;
189 }
190 m_pImpl->backendRunning = true;
191 m_pImpl->state = AcqDeviceState::Running;
192 return {};
193}

213{

215 return {};

228void AcqDevice::close()
229{
230 if (!m_pImpl) {
231 return;
232 }
233 std::lock_guard<std::mutex> control(m_pImpl->controlMutex); // no control call is inside the backend
234 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
235 if (!m_pImpl->backend) {
236 return;
237 }
238 m_pImpl->stopBackend();
239 m_pImpl->lastError = m_pImpl->error();
240 m_pImpl->backend->close();
241 // Destroying the backend finishes its input streams: readers drain what was committed, then see EndOfStream.
242 m_pImpl->backend.reset();
243 m_pImpl->state = AcqDeviceState::Closed;
244}

344AcqExpected<AcqValue> AcqDevice::read(std::string_view id)
345{
346 const auto index = findControl(id);
347 if (!index) {
348 return AcqError(AcqErrorCode::InvalidParameter, "no control " + std::string(id));
349 }
350 if (m_pImpl->controls[*index].kind == AcqControlKind::Command) {
351 return AcqError(AcqErrorCode::InvalidParameter, std::string(id) + " is a command; invoke it");
352 }
353 std::lock_guard<std::mutex> control(m_pImpl->controlMutex);
354 {
355 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
356 const AcqDeviceState state = m_pImpl->effectiveState();
357 if (!m_pImpl->backend || state == AcqDeviceState::Failed) {
358 return AcqErrorCode::InvalidState;
359 }
360 }
361 return m_pImpl->backend->readControl(*index);
362}

366AcqError AcqDevice::write(std::string_view id, const AcqValue& value)
367{
368 const auto index = findControl(id);
369 if (!index) {
370 return {AcqErrorCode::InvalidParameter, "no control " + std::string(id)};
371 }
372 const AcqControlInfo& info = m_pImpl->controls[*index];
373 if (info.kind != AcqControlKind::Parameter || !info.writable) {
374 return {AcqErrorCode::InvalidParameter, std::string(id) + " cannot be set"};
375 }
376 if (!fits(info, info.type, value)) {
377 return {AcqErrorCode::InvalidParameter, std::string(id) + ": the value has the wrong type or is out of range"};
378 }
379 std::lock_guard<std::mutex> control(m_pImpl->controlMutex);
380 {
381 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
382 if (!m_pImpl->backend || !allowed(info.when, m_pImpl->effectiveState())) {
383 return {AcqErrorCode::InvalidState, std::string(id) + " cannot be set in this state"};
384 }
385 }
386 return m_pImpl->backend->writeControl(*index, normalized(info.type, value));
387}

391AcqExpected<AcqValue> AcqDevice::invoke(std::string_view id, std::span<const AcqValue> arguments,
392 std::chrono::milliseconds timeout)
393{
394 const auto index = findControl(id);
395 if (!index) {
396 return AcqError(AcqErrorCode::InvalidParameter, "no control " + std::string(id));
397 }
398 const AcqControlInfo& info = m_pImpl->controls[*index];
399 if (info.kind != AcqControlKind::Command) {
400 return AcqError(AcqErrorCode::InvalidParameter, std::string(id) + " is not a command");
401 }
402 if (arguments.size() != info.arguments.size()) {
403 return AcqError(AcqErrorCode::InvalidParameter,
404 std::string(id) + " takes " + std::to_string(info.arguments.size()) + " arguments");
405 }
406 std::vector<AcqValue> checked;
407 checked.reserve(arguments.size());
408 for (std::size_t i = 0; i < arguments.size(); ++i) {
409 if (!fits(info, info.arguments[i], arguments[i])) {
410 return AcqError(AcqErrorCode::InvalidParameter, std::string(id) + ": argument " + std::to_string(i + 1) + " has the wrong type or is out of range");
411 }
412 checked.push_back(normalized(info.arguments[i], arguments[i]));
413 }
414 std::lock_guard<std::mutex> control(m_pImpl->controlMutex);
415 {
416 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
417 if (!m_pImpl->backend || !allowed(info.when, m_pImpl->effectiveState())) {
418 return AcqError(AcqErrorCode::InvalidState, std::string(id) + " cannot run in this state");
419 }
420 }
421 return m_pImpl->backend->invokeControl(*index, checked, timeout);
422}

426std::size_t AcqDevice::pollControlEvents(std::span<AcqControlEvent> out)
427{
428 if (!m_pImpl) {
429 return 0;
430 }
431 std::lock_guard<std::mutex> lock(m_pImpl->mutex);
432 if (!m_pImpl->backend) {
433 return 0;
434 }
435 auto& queue = detail::BackendAccess::state(*m_pImpl->backend).controlEvents;
436 std::size_t n = 0;
437 while (n < out.size() && queue.pop(out[n])) {
438 ++n;
439 }
440 return n;
441}
