33std::unique_ptr<MNESparseNamedMatrix> MNESparseNamedMatrix::read(FiffStream::SPtr& stream,
34 const FiffDirNode::SPtr& node,
35 int kind)
36{
37 FiffDirNode::SPtr block;
38 FiffTag::UPtr tag;
39 if (node->type == FIFFB_MNE_NAMED_MATRIX) {
40 if (node->find_tag(stream, kind, tag)) {
41 block = node;
42 }
43 } else {
44 for (const FiffDirNode::SPtr& child : node->children) {
45 if (child->type == FIFFB_MNE_NAMED_MATRIX && child->find_tag(stream, kind, tag)) {
46 block = child;
47 break;
48 }
49 }
50 }
51 if (!block) {
52 return nullptr;
53 }
54 FiffSparseMatrix::UPtr data = FiffSparseMatrix::fiff_get_float_sparse_matrix(tag);
55 if (!data) {
56 return nullptr;
57 }
58 auto matrix = std::make_unique<MNESparseNamedMatrix>();
59 matrix->nrow = data->rows();
60 matrix->ncol = data->cols();
61 if (block->find_tag(stream, FIFF_MNE_NROW, tag) && *tag->toInt() != matrix->nrow) {
62 qCritical("MNESparseNamedMatrix::read - FIFF_MNE_NROW conflicts with the matrix data.");
63 return nullptr;
64 }
65 if (block->find_tag(stream, FIFF_MNE_NCOL, tag) && *tag->toInt() != matrix->ncol) {
66 qCritical("MNESparseNamedMatrix::read - FIFF_MNE_NCOL conflicts with the matrix data.");
67 return nullptr;
68 }
69 if (block->find_tag(stream, FIFF_MNE_ROW_NAMES, tag)) {
70 matrix->rowlist = FiffStream::split_name_list(tag->toString());
71 if (matrix->rowlist.size() != matrix->nrow) {
72 qCritical("MNESparseNamedMatrix::read - Incorrect number of entries in the row name list.");
73 return nullptr;
74 }
75 }
76 if (block->find_tag(stream, FIFF_MNE_COL_NAMES, tag)) {
77 matrix->collist = FiffStream::split_name_list(tag->toString());
78 if (matrix->collist.size() != matrix->ncol) {
79 qCritical("MNESparseNamedMatrix::read - Incorrect number of entries in the column name list.");
80 return nullptr;
81 }
82 }
83 matrix->data = std::move(data);
84 return matrix;
85}

89void MNESparseNamedMatrix::write(FiffStream& stream, int kind) const
90{
91 stream.start_block(FIFFB_MNE_NAMED_MATRIX);
92 stream.write_int(FIFF_MNE_NROW, &nrow);
93 stream.write_int(FIFF_MNE_NCOL, &ncol);
94 if (!rowlist.isEmpty()) {

96 }
97 if (!collist.isEmpty()) {
98 stream.write_name_list(FIFF_MNE_COL_NAMES, collist);
99 }
100 stream.write_float_sparse_rcs(kind, data->eigen());
101 stream.end_block(FIFFB_MNE_NAMED_MATRIX);
102}
