151class ChannelRhiRenderer::TraceSourceAdapter : public scene::TraceSource
152{
153public:
154 explicit TraceSourceAdapter(const ChannelRhiRenderer& renderer)
155 : m_renderer(renderer)
156 {
157 }
158
159 int channel_count() const override
160 {
161 return m_renderer.totalLogicalChannels();
162 }
163
164 std::optional<Channel> channel(int index) const override
165 {
166 const int ch = m_renderer.actualChannelAt(index);
167 if (!m_renderer.m_model || ch < 0)
168 return std::nullopt;
169 ChannelDisplayInfo info = m_renderer.m_model->channelInfo(ch);
170 return Channel{std::move(info.name), std::move(info.typeLabel), info.amplitudeMax, toSceneColor(info.color), info.bad};
171 }
172
173 float value_at(int index, int sample) const override
174 {
175 const int ch = m_renderer.actualChannelAt(index);
176 return (m_renderer.m_model && ch >= 0) ? m_renderer.m_model->sampleValueAt(ch, sample) : 0.f;
177 }
178
179 int first_sample() const override
180 {
181 const ChannelDataModel* model = m_renderer.m_model.data();
182 return (model && model->totalSamples() > 0) ? model->firstSample() : 0;
183 }
184
185 int sample_count() const override
186 {
187 return m_renderer.m_model ? m_renderer.m_model->totalSamples() : 0;
188 }
189
190 scene::TraceVertices vertices(int index, int first, int last, int pixelWidth) const override
191 {
192 scene::TraceVertices result{first, {}};
193 const int ch = m_renderer.actualChannelAt(index);
194 if (m_renderer.m_model && ch >= 0)
195 result.xy = m_renderer.m_model->decimatedVertices(ch, first, last, pixelWidth, result.first_sample);
196 return result;
197 }
198
199private:
200 const ChannelRhiRenderer& m_renderer;
201};

227void ChannelRhiRenderer::connectView()
228{
229 scene::TraceView::Effects effects;
230 effects.scroll_changed = [this](double) {
231 const float sample = scrollSample();
232 const float visible = static_cast<float>(viewWidth()) * samplesPerPixel();
233
234 // Overlay prefetch: only rebuild when scroll exceeds the cached sample range.
235 // The shader handles bands via uniforms, so we only need to rebuild when
236 // annotations/events would be outside the cached texture.
237 if (m_overlayTotalSamples <= 0.f || sample < m_overlayFirstSample ||
238 sample + visible > m_overlayFirstSample + m_overlayTotalSamples) {
239 m_overlayDirty = true;
240 }
241
242 emit scrollSampleChanged(sample);
243 requestUpdate();
244 };
245 effects.zoom_changed = [this](double) {
246 m_geometry.invalidate(); // zoom change → decimation changes
247 m_overlayDirty = true;
248 emit samplesPerPixelChanged(samplesPerPixel());
249 requestUpdate();
250 };
251 effects.channel_offset_changed = [this](int firstChannel) {
252 m_geometry.invalidate();
253 m_pipelineDirty = true;
254 emit channelOffsetChanged(firstChannel);
255 requestUpdate();
256 };
257 effects.animate_scroll = [this](const scene::TraceView::ScrollAnimation& animation) {
258 m_animator.start_scroll(animation, monotonicMs());
259 startAnimationTimer();
260 };
261 effects.animate_zoom = [this](const scene::TraceView::ZoomAnimation& animation) {
262 m_animator.start_zoom(animation, monotonicMs());
263 startAnimationTimer();
264 };
265 effects.stop_animations = [this] {
266 m_animator.stop_scroll();
267 };
268 effects.sample_clicked = [this](int sample) {
269 emit sampleClicked(sample);
270 };
271 effects.range_selected = [this](int startSample, int endSample) {
272 emit sampleRangeSelected(startSample, endSample);
273 };
274 effects.annotation_boundary_moved = [this](int index, bool isStart, int sample) {
275 emit annotationBoundaryMoved(index, isStart, sample);
276 };
277 effects.annotations_changed = [this] {
278 m_overlayDirty = true;
279 requestUpdate();
280 };
281 effects.channel_double_clicked = [this](int logicalChannel) {
282 const int ch = actualChannelAt(logicalChannel);
283 if (m_model && ch >= 0) {
284 m_model->setChannelBad(ch, !m_model->channelInfo(ch).bad);
285 }
286 };
287 effects.cursor_changed = [this](scene::CursorShape shape) {
288 if (!m_pHost)
289 return;
290 if (shape == scene::CursorShape::SizeHorizontal)
291 m_pHost->setCursorShape(Qt::SizeHorCursor);
292 else
293 m_pHost->unsetCursorShape();
294 };
295 effects.cursor_data_changed = [this](const scene::TraceView::CursorData& data) {
296 emit cursorDataChanged(data.time_sec, data.amplitude, QString::fromStdString(data.channel_name),
297 QString::fromStdString(data.unit_label));
298 };
299 effects.redraw = [this] {
300 requestUpdate();
301 };
302 m_view.set_effects(std::move(effects));
303}

794void ChannelRhiRenderer::ensurePipeline(QRhi* rhi, QRhiRenderTarget* target)
795{
796 if (!m_pipelineDirty)
797 return;
798
799 m_uboStride = static_cast<int>(
800 (static_cast<int>(sizeof(scene::TraceUniforms)) + rhi->ubufAlignment() - 1) & ~(rhi->ubufAlignment() - 1));
801
802 // One UBO slot per drawn trace: the visible rows, or every channel in butterfly mode
803 const int nCh = scene::TraceGeometry::slot_count(view(), *m_traceSource);
804
805 // ── Uniform buffer ──────────────────────────────────────────────────
806 bool uboRecreated = false;
807 if (!m_ubo || m_ubo->size() < static_cast<quint32>(nCh * m_uboStride)) {
808 m_ubo.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
809 QRhiBuffer::UniformBuffer,
810 nCh * m_uboStride));
811 m_ubo->create();
812 uboRecreated = true;
813 }
814
815 // ── Shader resource bindings ────────────────────────────────────────
816 // Recreate if the UBO pointer changed — SRB holds a raw pointer to the UBO.
817 if (!m_srb || uboRecreated) {
818 m_srb.reset(rhi->newShaderResourceBindings());
819 m_srb->setBindings({QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(
820 0,
821 QRhiShaderResourceBinding::VertexStage |
822 QRhiShaderResourceBinding::FragmentStage,
823 m_ubo.get(),
824 sizeof(scene::TraceUniforms) // visible block size for the shader
825 )});
826 m_srb->create();
827 }
828
829 // ── Shaders ─────────────────────────────────────────────────────────
830 // Resource path matches qt_add_shaders PREFIX + file path (including subdirectory).
831 QShader vs = loadShader(QStringLiteral(":/disp/shaders/viewers/helpers/shaders/channeldata.vert.qsb"));
832 QShader fs = loadShader(QStringLiteral(":/disp/shaders/viewers/helpers/shaders/channeldata.frag.qsb"));
833
834 if (!vs.isValid() || !fs.isValid()) {
835 mne::Log::warning("ChannelRhiView: shaders not found. Ensure qt_add_shaders is configured in CMakeLists.");
836 return;
837 }
838
839 // ── Graphics pipeline ───────────────────────────────────────────────
840 // Destroy any existing pipeline before creating a new one.
841 m_pipeline.reset();
842 m_pipeline.reset(rhi->newGraphicsPipeline());
843 m_pipeline->setShaderStages({{QRhiShaderStage::Vertex, vs},
844 {QRhiShaderStage::Fragment, fs}});
845
846 QRhiVertexInputLayout il;
847 il.setBindings({{2 * sizeof(float)}}); // stride = vec2
848 il.setAttributes({{0, 0, QRhiVertexInputAttribute::Float2, 0}}); // location 0 = vec2
849
850 m_pipeline->setVertexInputLayout(il);

884void ChannelRhiRenderer::rebuildVBOs(QRhi* rhi, QRhiResourceUpdateBatch* batch)
885{
886 if (!m_model || !m_geometry.stream(view(), *m_traceSource, m_bZScoreMode))
887 return;
888
889 // VBOs are indexed by logical (filtered) channel index, not model channel index
890 const std::vector<scene::TraceVertices>& channels = m_geometry.channels();
891 m_gpuChannels.resize(channels.size());
892 for (std::size_t logCh = 0; logCh < channels.size(); ++logCh) {
893 const std::vector<float>& verts = channels[logCh].xy;
894 ChannelGpuData& gd = m_gpuChannels[logCh];
895 gd.vertexCount = 0;
896 if (verts.empty())
897 continue;
898
899 const quint32 byteSize = static_cast<quint32>(verts.size() * sizeof(float));
900 if (!gd.vbo || static_cast<quint32>(gd.vbo->size()) < byteSize) {
901 gd.vbo.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::VertexBuffer, byteSize));
902 if (!gd.vbo->create()) {
903 mne::Log::warning("ChannelRhiView: VBO create failed for channel {}", logCh);
904 continue;
905 }
906 }
907 batch->updateDynamicBuffer(gd.vbo.get(), 0, byteSize, verts.data());
908 gd.vertexCount = channels[logCh].vertex_count();
909 }

914void ChannelRhiRenderer::updateUBO(QRhiResourceUpdateBatch* batch)

918 return;

934// Overlay texture — annotations, events, and epoch markers baked into a QImage.
935// Alternating per-second bands are now computed in the fragment shader, so this
936// image only needs rebuilding when annotations/events change or when the scroll
937// exceeds the prefetch window.
938//
939// The overlay covers a wider sample range than the viewport (controlled by
940// kOverlayPrefetchFactor). The fragment shader maps screen UVs into this
941// wider texture via the OverlayParams UBO.

944void ChannelRhiRenderer::rebuildOverlayImage(int logicalWidth, int logicalHeight, qreal devicePixelRatio)
945{
946 const qreal dpr = qMax(devicePixelRatio, 1.0);
947 const int pixelWidth = qMax(1, qRound(logicalWidth * dpr));
948 const int pixelHeight = qMax(1, qRound(logicalHeight * dpr));
949
950 m_overlayImage = QImage(pixelWidth, pixelHeight, QImage::Format_RGBA8888);
951 m_overlayImage.setDevicePixelRatio(dpr);
952 m_overlayImage.fill(Qt::transparent);
953 m_overlayDirty = false;
954
955 if (logicalWidth <= 0 || logicalHeight <= 0 || m_view.sfreq() <= 0.0 || m_view.samples_per_pixel() <= 0.0 || m_overlayTotalSamples <= 0.f)
956 return;
957
958 // The overlay covers m_overlayFirstSample .. m_overlayFirstSample + m_overlayTotalSamples; per-second bands
959 // are computed in the fragment shader
960 const QFont font = QGuiApplication::font();
961 const QtTextMetrics metrics(font);
962 scene::Canvas canvas(logicalWidth, logicalHeight);
963 m_markers.draw(canvas, metrics, scene::Font{}, m_view.annotations(), m_overlayFirstSample,
964 static_cast<double>(logicalWidth) / static_cast<double>(m_overlayTotalSamples));
965 QPainter p(&m_overlayImage);
966 paintScene(p, canvas.display_list(), font);
967}

971void ChannelRhiRenderer::ensureOverlayPipeline(QRhi* rhi, QRhiRenderTarget* target)
972{
973 if (m_overlayPipeline)
974 return;
975
976 // Static full-screen quad (screen-space NDC coordinates); uploaded once in ensureOverlayResources()
977 static constexpr int kQuadBytes = sizeof(kQuadVerts);
978 m_overlayVbo.reset(rhi->newBuffer(QRhiBuffer::Immutable,
979 QRhiBuffer::VertexBuffer,
980 kQuadBytes));
981 if (!m_overlayVbo->create()) {
982 m_overlayVbo.reset();
983 return;
984 }
985
986 // Texture — placeholder 1×1; resized lazily in renderFrame() when pw/ph are known.
987 m_overlayTex.reset(rhi->newTexture(QRhiTexture::RGBA8, QSize(1, 1)));
988 m_overlayTex->create();
989 m_overlayTexSize = QSize();
990
991 m_overlaySampler.reset(rhi->newSampler(
992 QRhiSampler::Linear, QRhiSampler::Linear,
993 QRhiSampler::None,
994 QRhiSampler::ClampToEdge, QRhiSampler::ClampToEdge));
995 m_overlaySampler->create();
996
997 // UBO for per-frame overlay parameters (binding 2)
998 m_overlayUbo.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
999 QRhiBuffer::UniformBuffer,
1000 kOverlayUboSize));
1001 if (!m_overlayUbo->create()) {
1002 m_overlayVbo.reset();
1003 m_overlayUbo.reset();
1004 return;
1005 }
1006
1007 // SRB: binding 1 = combined image sampler, binding 2 = overlay UBO
1008 m_overlaySrb.reset(rhi->newShaderResourceBindings());
1009 m_overlaySrb->setBindings({QRhiShaderResourceBinding::sampledTexture(
1010 1, QRhiShaderResourceBinding::FragmentStage,
1011 m_overlayTex.get(), m_overlaySampler.get()),
1012 QRhiShaderResourceBinding::uniformBuffer(
1013 2, QRhiShaderResourceBinding::FragmentStage,
1014 m_overlayUbo.get())});
1015 m_overlaySrb->create();
1016
1017 QShader vs = loadShader(QStringLiteral(":/disp/shaders/viewers/helpers/shaders/overlay.vert.qsb"));
1018 QShader fs = loadShader(QStringLiteral(":/disp/shaders/viewers/helpers/shaders/overlay.frag.qsb"));
1019 if (!vs.isValid() || !fs.isValid()) {
1020 m_overlayVbo.reset();
1021 m_overlayUbo.reset();
1022 m_overlaySrb.reset();
1023 m_overlaySampler.reset();
1024 m_overlayTex.reset();
1025 return;
1026 }
1027
1028 m_overlayPipeline.reset(rhi->newGraphicsPipeline());
1029 QRhiGraphicsPipeline::TargetBlend blend;
1030 blend.enable = true;
1031 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
1032 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
1033 blend.srcAlpha = QRhiGraphicsPipeline::One;
1034 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
1035 m_overlayPipeline->setTargetBlends({blend});
1036 m_overlayPipeline->setTopology(QRhiGraphicsPipeline::TriangleStrip);
1037 m_overlayPipeline->setDepthTest(false);
1038 m_overlayPipeline->setDepthWrite(false);
1039 m_overlayPipeline->setShaderStages({
1040 {QRhiShaderStage::Vertex, vs},
1041 {QRhiShaderStage::Fragment, fs},
1042 });
1043
1044 QRhiVertexInputLayout inputLayout;
1045 inputLayout.setBindings({QRhiVertexInputBinding(4 * sizeof(float))});
1046 inputLayout.setAttributes({QRhiVertexInputAttribute(0, 0, QRhiVertexInputAttribute::Float2, 0),
1047 QRhiVertexInputAttribute(0, 1, QRhiVertexInputAttribute::Float2, 2 * sizeof(float))});
1048 m_overlayPipeline->setVertexInputLayout(inputLayout);
1049 m_overlayPipeline->setShaderResourceBindings(m_overlaySrb.get());
1050 m_overlayPipeline->setRenderPassDescriptor(target->renderPassDescriptor());
1051
1052 if (!m_overlayPipeline->create()) {
1053 mne::Log::warning("ChannelRhiView: overlay pipeline create failed");
1054 m_overlayPipeline.reset();
1055 return;
1056 }
1057
1058 // The static VBO needs an initial upload in the frame's resource batch.
1059 m_overlayVboNeedsUpload = true;
1060}

1163
1164void ChannelRhiRenderer::renderFrame(QRhi* rhi, QRhiRenderTarget* target, QRhiCommandBuffer* cb)
1165{
1166 tickAnimations();
1167 const bool hasTraces = m_model && totalLogicalChannels() > 0;
1168
1169 // ── Ensure GPU resources ─────────────────────────────────────────────
1170 if (hasTraces) {
1171 ensurePipeline(rhi, target);
1172 if (!m_pipeline) {
1173 // Pipeline not ready: show RED background so the failure is visible
1174 QRhiResourceUpdateBatch* u = rhi->nextResourceUpdateBatch();
1175 cb->beginPass(target, QColor(220, 0, 0), {1.f, 0}, u);
1176 cb->endPass();
1177 return;
1178 }
1179 }
1180
1181 QSize ps = target->pixelSize();
1182 const int pw = ps.width();
1183 const int ph = ps.height();
1184 const int logicalW = viewWidth();
1185 const int logicalH = viewHeight();
1186 const qreal overlayDpr = (logicalW > 0) ? (static_cast<qreal>(pw) / static_cast<qreal>(logicalW))
1187 : (m_pHost ? m_pHost->surfaceDevicePixelRatio() : 1.0);
1188
1189 QRhiResourceUpdateBatch* batch = rhi->nextResourceUpdateBatch();
1190
1191 bool overlayReady = false;
1192 if (hasTraces) {
1193 if (isVboDirty())
1194 rebuildVBOs(rhi, batch);
1195
1196 updateUBO(batch);
1197
1198 // ── Overlay image (annotations + events; bands computed in shader) ──
1199 if (ensureOverlayResources(rhi, target, batch) && pw > 0 && ph > 0 && logicalW > 0 && logicalH > 0) {
1200 // Compute the overlay prefetch window (wider than the viewport)
1201 const float visibleSamples = static_cast<float>(logicalW) * samplesPerPixel();
1202 const float extraSamples = kOverlayPrefetchFactor * visibleSamples;
1203 const float overlayFirstSample = scrollSample() - extraSamples;
1204 const float overlayTotalSamples = visibleSamples + 2.0f * extraSamples;
1205 // Width of overlay texture in logical pixels = 1 + 2*prefetch factor times viewport
1206 const int overlayLogicalW = static_cast<int>(std::ceil(
1207 (1.0f + 2.0f * kOverlayPrefetchFactor) * static_cast<float>(logicalW)));
1208
1209 // Rebuild the overlay QImage if dirty or resized
1210 const QSize requiredTexSize(qRound(overlayLogicalW * overlayDpr),
1211 qRound(logicalH * overlayDpr));
1212 if (m_overlayDirty || requiredTexSize != m_overlayTexSize) {
1213 // Store the sample range covered by this overlay build
1214 m_overlayFirstSample = overlayFirstSample;
1215 m_overlayTotalSamples = overlayTotalSamples;
1216
1217 rebuildOverlayImage(overlayLogicalW, logicalH, overlayDpr);
1218
1219 // (Re)create texture at the correct pixel size
1220 if (requiredTexSize != m_overlayTexSize) {
1221 m_overlayTex.reset(rhi->newTexture(QRhiTexture::RGBA8, requiredTexSize));
1222 m_overlayTex->create();
1223 // Re-create SRB because it references the texture
1224 m_overlaySrb->setBindings({QRhiShaderResourceBinding::sampledTexture(
1225 1, QRhiShaderResourceBinding::FragmentStage,
1226 m_overlayTex.get(), m_overlaySampler.get()),
1227 QRhiShaderResourceBinding::uniformBuffer(
1228 2, QRhiShaderResourceBinding::FragmentStage,
1229 m_overlayUbo.get())});
1230 m_overlaySrb->create();
1231 m_overlayTexSize = requiredTexSize;
1232 }
1233 QRhiTextureUploadEntry entry(0, 0, QRhiTextureSubresourceUploadDescription(m_overlayImage));
1234 batch->uploadTexture(m_overlayTex.get(), entry);
1235 }
1236
1237 // Upload per-frame overlay UBO (scroll params for shader-computed bands + UV mapping)
1238 OverlayParams params;
1239 params.scrollSample = scrollSample();
1240 params.samplesPerPixel = samplesPerPixel();
1241 params.viewWidth = static_cast<float>(logicalW);
1242 params.sfreq = static_cast<float>(m_view.sfreq());
1243 params.firstFileSample = static_cast<float>(m_view.first_file_sample());
1244 params.gridEnabled = m_gridVisible ? 1.0f : 0.0f;
1245 params.overlayFirstSample = m_overlayFirstSample;
1246 params.overlayTotalSamples = m_overlayTotalSamples;
1247 batch->updateDynamicBuffer(m_overlayUbo.get(), 0,
1248 static_cast<quint32>(sizeof(params)), &params);
1249
1250 overlayReady = true;
1251 }
1252 }
1253
1254 // ── HUD image (crosshair, scalebars, ruler, annotation selection) ──
1255 const bool hudReady = hudVisible() && pw > 0 && ph > 0 && ensureOverlayResources(rhi, target, batch) && prepareHud(rhi, batch, logicalW, logicalH, overlayDpr);
1256
1257 // ── Render pass ──────────────────────────────────────────────────────
1258 QColor bg = m_bgColor;
1259 cb->beginPass(target, bg, {1.f, 0}, batch);
1260
1261 cb->setViewport(QRhiViewport(0.f, 0.f, static_cast<float>(pw),
1262 static_cast<float>(ph)));
1263
1264 if (hasTraces) {
1265 // ── Waveform traces ──────────────────────────────────────────────────
1266 cb->setGraphicsPipeline(m_pipeline.get());
1267
1268 // One draw per trace; its uniforms are in the UBO slot of the same index
1269 for (int slot = 0; slot < std::ssize(m_traceDraws); ++slot) {
1270 const int logCh = m_traceDraws[static_cast<std::size_t>(slot)].channel;
1271 if (logCh >= std::ssize(m_gpuChannels))
1272 continue;
1273 const ChannelGpuData& gd = m_gpuChannels[static_cast<std::size_t>(logCh)];
1274 if (!gd.vbo || gd.vertexCount < 2)
1275 continue;
1276
1277 QRhiCommandBuffer::DynamicOffset dynOff{0, static_cast<quint32>(slot * m_uboStride)};
1278 cb->setShaderResources(m_srb.get(), 1, &dynOff);
1279 QRhiCommandBuffer::VertexInput vi(gd.vbo.get(), 0);
1280 cb->setVertexInput(0, 1, &vi);
1281 cb->draw(static_cast<quint32>(gd.vertexCount));
1282 }
1283 }
1284
1285 QRhiCommandBuffer::VertexInput overlayVi(m_overlayVbo.get(), 0);
1286
1287 // ── Overlay blit (bands + event lines) — drawn after waveforms ───────
1288 if (overlayReady) {
1289 cb->setGraphicsPipeline(m_overlayPipeline.get());
1290 cb->setShaderResources(m_overlaySrb.get());
1291 cb->setVertexInput(0, 1, &overlayVi);
1292 cb->draw(4); // TriangleStrip: 4 vertices = 2 triangles = full-screen quad
1293 }
1294
1295 // ── HUD blit (crosshair, scalebars, ruler, annotation selection) — on top of everything ──
1296 if (hudReady) {
1297 cb->setGraphicsPipeline(m_overlayPipeline.get());
1298 cb->setShaderResources(m_hudSrb.get());
1299 cb->setVertexInput(0, 1, &overlayVi);
1300 cb->draw(4);
1301 }
1302
1303 cb->endPass();
1304}
