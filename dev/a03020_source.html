80std::optional<ImageFormat> formatOfExtension(const std::filesystem::path& path)
81{
82 const std::string extension = to_lower(utf8_from_path(path.extension()));
83 if (extension == ".png") {
84 return ImageFormat::Png;
85 }
86 if (extension == ".jpg" || extension == ".jpeg") {
87 return ImageFormat::Jpeg;
88 }
89 if (extension == ".svg") {
90 return ImageFormat::Svg;
91 }
92 return std::nullopt;

107std::string mne::html_escape(std::string_view text)
108{
109 std::string escaped;
110 escaped.reserve(text.size());
111 for (const char c : text) {
112 switch (c) {
113 case '&':
114 escaped += "&amp;";
115 break;
116 case '<':
117 escaped += "&lt;";
118 break;
119 case '>':
120 escaped += "&gt;";
121 break;
122 case '"':
123 escaped += "&quot;";
124 break;
125 case '\'':
126 escaped += "&#39;";
127 break;
128 default:
129 escaped += c;
130 }
131 }
132 return escaped;
133}

137std::string mne::base64_encode(std::string_view data)
138{
139 constexpr std::string_view alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
140 std::string encoded;
141 encoded.reserve((data.size() + 2) / 3 * 4);
142 std::size_t i = 0;
143 for (; i + 2 < data.size(); i += 3) {
144 const auto chunk = static_cast<std::uint32_t>(static_cast<unsigned char>(data[i]) << 16U |
145 static_cast<unsigned char>(data[i + 1]) << 8U |
146 static_cast<unsigned char>(data[i + 2]));
147 encoded += alphabet[chunk >> 18U & 0x3FU];
148 encoded += alphabet[chunk >> 12U & 0x3FU];
149 encoded += alphabet[chunk >> 6U & 0x3FU];
150 encoded += alphabet[chunk & 0x3FU];
151 }
152 if (const std::size_t rest = data.size() - i; rest > 0) {
153 std::uint32_t chunk = static_cast<std::uint32_t>(static_cast<unsigned char>(data[i])) << 16U;
154 if (rest == 2) {
155 chunk |= static_cast<std::uint32_t>(static_cast<unsigned char>(data[i + 1])) << 8U;
156 }
157 encoded += alphabet[chunk >> 18U & 0x3FU];
158 encoded += alphabet[chunk >> 12U & 0x3FU];
159 encoded += rest == 2 ? alphabet[chunk >> 6U & 0x3FU] : '=';
160 encoded += '=';
161 }
162 return encoded;
163}

276std::string Report::to_html() const
277{
278 std::string html = "<!DOCTYPE html>\n<html lang=\"en\">\n<head>\n"
279 "<meta charset=\"UTF-8\">\n"
280 "<meta name=\"viewport\" content=\"width=device-width, initial-scale=1.0\">\n";
281 appendElement(html, "title", m_title);
282 html += "<style>\n";
283 html += kStyle;
284 html += "</style>\n</head>\n<body>\n";
285 appendElement(html, "h1", m_title);
286 if (m_sections.size() > 1) {
287 html += "<nav class=\"toc\">\n<ul>\n";
288 for (std::size_t i = 0; i < m_sections.size(); ++i) {
289 std::format_to(std::back_inserter(html), "<li><a href=\"#section-{}\">{}</a></li>\n", i + 1,
290 html_escape(m_sections[i].title));
291 }
292 html += "</ul>\n</nav>\n";
293 }
294 for (std::size_t i = 0; i < m_sections.size(); ++i) {
295 std::format_to(std::back_inserter(html), "<h2 id=\"section-{}\">{}</h2>\n", i + 1, html_escape(m_sections[i].title));
296 html += m_sections[i].html;
297 html += '\n';
298 }
299 const auto now = std::chrono::floor<std::chrono::seconds>(std::chrono::system_clock::now());
300 std::format_to(std::back_inserter(html), "<div class=\"footer\">\n<p>Generated by MNE-CPP on {:%FT%TZ}</p>\n</div>\n", now);
301 html += "</body>\n</html>\n";
302 return html;
303}
