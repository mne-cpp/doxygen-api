71bool isDecimal(std::string_view text)
72{
73 std::size_t i = 0;
74 const auto digits = [&] {
75 const std::size_t start = i;
76 while (i < text.size() && text[i] >= '0' && text[i] <= '9') {
77 ++i;
78 }
79 return i - start;
80 };
81 if (i < text.size() && (text[i] == '+' || text[i] == '-')) {
82 ++i;
83 }
84 std::size_t mantissa = digits();
85 if (i < text.size() && text[i] == '.') {
86 ++i;
87 mantissa += digits();
88 }
89 if (mantissa == 0) {
90 return false;
91 }
92 if (i < text.size() && (text[i] == 'e' || text[i] == 'E')) {
93 ++i;
94 if (i < text.size() && (text[i] == '+' || text[i] == '-')) {
95 ++i;
96 }
97 if (digits() == 0) {
98 return false;
99 }
100 }
101 return i == text.size();
102}

205std::optional<double> mne::to_double(std::string_view text)
206{
207 text = stripPlus(trimmed(text));
208 const std::string lowered = to_lower(text);
209 const bool negative = !lowered.empty() && lowered.front() == '-';
210 const std::string_view word = negative ? std::string_view(lowered).substr(1) : std::string_view(lowered);
211 if (word == "inf") {
212 return negative ? -std::numeric_limits<double>::infinity() : std::numeric_limits<double>::infinity();
213 }
214 if (word == "nan" && !negative) {
215 return std::numeric_limits<double>::quiet_NaN();
216 }
217 if (!isDecimal(text)) {
218 return std::nullopt;
219 }
220 // Floating-point std::from_chars needs macOS 26 with libc++, and streams disagree on denormals across
221 // standard libraries, so the C library parses in the "C" locale whatever the application set.
222 const std::string copy(text);
223 const double value = strtodC(copy.c_str());
224 const bool nonZeroDigit = std::any_of(text.begin(), std::find_if(text.begin(), text.end(), [](char c) { return c == 'e' || c == 'E'; }),
225 [](char c) { return c >= '1' && c <= '9'; });
226 if (std::isinf(value) || (value == 0.0 && nonZeroDigit)) {
227 return std::nullopt; // overflow or underflow to zero; QString rejects both and keeps denormals
228 }
229 return value;
230}

280bool mne::is_utf8(std::string_view text)
281{
282 const auto byte = [&](std::size_t i) {
283 return static_cast<unsigned char>(text[i]);
284 };
285 for (std::size_t i = 0; i < text.size();) {
286 const unsigned char lead = byte(i);
287 if (lead < 0x80) {
288 ++i;
289 continue;
290 }
291 // Second-byte range per lead byte (Unicode table 3-7), which rules out overlongs, surrogates and > U+10FFFF
292 std::size_t length = 0;
293 unsigned char low = 0x80;
294 unsigned char high = 0xBF;
295 if (lead >= 0xC2 && lead <= 0xDF) {
296 length = 2;
297 } else if (lead >= 0xE0 && lead <= 0xEF) {
298 length = 3;
299 low = lead == 0xE0 ? 0xA0 : 0x80;
300 high = lead == 0xED ? 0x9F : 0xBF;
301 } else if (lead >= 0xF0 && lead <= 0xF4) {
302 length = 4;
303 low = lead == 0xF0 ? 0x90 : 0x80;
304 high = lead == 0xF4 ? 0x8F : 0xBF;
305 } else {
306 return false;
307 }
308 if (i + length > text.size() || byte(i + 1) < low || byte(i + 1) > high) {
309 return false;
310 }
311 for (std::size_t k = 2; k < length; ++k) {
312 if ((byte(i + k) & 0xC0) != 0x80) {
313 return false;
314 }
315 }
316 i += length;
317 }
318 return true;
319}

337double mne::tie_away_from_zero(double value, int decimals)
338{
339 if (!std::isfinite(value)) {
340 return value;
341 }
342 bool tie = false;
343 if (decimals >= 0) {
344 // A tie at decimals is an odd multiple of 2^-(decimals + 1)
345 const double scaled = std::ldexp(value, decimals + 1);
346 tie = std::isfinite(scaled) && scaled == std::trunc(scaled) && std::fmod(scaled, 2.0) != 0.0;
347 } else if (std::abs(value) < 0x1p53) {
348 const double unit = std::pow(10.0, -decimals);
349 tie = std::fmod(std::abs(value), unit) == unit / 2;
350 }
351 return tie ? std::nextafter(value, value < 0 ? -std::numeric_limits<double>::infinity() : std::numeric_limits<double>::infinity()) : value;
352}

356std::string mne::format_number(double value, char format, int precision)
357{
358 if (std::isnan(value))
359 return "nan";
360 if (std::isinf(value))
361 return value < 0 ? "-inf" : "inf";
362 const double magnitude = std::abs(value);
363 // Decimal digits d0 d1 d2 ... of magnitude = d0.d1d2... * 10^exponent
364 const auto expand = [magnitude](int decimals, std::string& digits, int& exponent) {
365 const std::string text = std::format("{:.{}e}", magnitude, decimals);
366 const std::size_t ePos = text.find('e');
367 digits = text.substr(0, 1) + text.substr(2, ePos - 2);
368 exponent = std::stoi(text.substr(ePos + 1));
369 };
370 std::string digits;
371 int exponent = 0;
372 expand(17, digits, exponent);
373 const int significant = format == 'e' ? precision + 1 : format == 'g' ? std::max(precision, 1)
374 : exponent + 1 + precision;
375 expand(std::max(significant, 0) + 40, digits, exponent);
376 const std::string_view tail = std::string_view(digits).substr(static_cast<std::size_t>(std::max(significant, 0)) + 1);
377 if (tail.find_first_not_of('0') == std::string_view::npos || tail.find_first_not_of('9') == std::string_view::npos)
378 expand(std::max(significant, 0) + 1100, digits, exponent); // exact for every double
379
380 std::string rounded;
381 if (significant >= 0) {
382 rounded = digits.substr(0, static_cast<std::size_t>(significant));
383 if (digits[static_cast<std::size_t>(significant)] >= '5') {
384 int i = significant - 1;
385 while (i >= 0 && rounded[static_cast<std::size_t>(i)] == '9')
386 rounded[static_cast<std::size_t>(i--)] = '0';
387 if (i >= 0) {
388 ++rounded[static_cast<std::size_t>(i)];
389 } else {
390 rounded.insert(rounded.begin(), '1');
391 ++exponent;
392 if (format != 'f')
393 rounded.pop_back();
394 }
395 }
396 }
397 const bool zero = rounded.find_first_not_of('0') == std::string::npos;
398 if (zero)
399 exponent = 0;
400
401 const auto fixed = [&](int decimals) {
402 std::string intPart;
403 std::string fraction;
404 if (zero) {
405 intPart = "0";
406 fraction.assign(static_cast<std::size_t>(decimals), '0');
407 } else if (exponent + 1 <= 0) {
408 intPart = "0";
409 fraction = std::string(static_cast<std::size_t>(-(exponent + 1)), '0') + rounded;
410 } else {
411 const std::size_t intDigits = static_cast<std::size_t>(exponent) + 1;
412 intPart = rounded.substr(0, intDigits);
413 fraction = rounded.substr(intDigits);
414 }
415 fraction.resize(static_cast<std::size_t>(decimals), '0');
416 return decimals > 0 ? intPart + "." + fraction : intPart;
417 };
418 const auto scientific = [&](std::string mantissaDigits) {
419 if (mantissaDigits.empty())
420 mantissaDigits = "0";
421 std::string text(1, mantissaDigits[0]);
422 if (mantissaDigits.size() > 1)
423 text += "." + mantissaDigits.substr(1);
424 return text + std::format("e{}{:02}", exponent < 0 ? '-' : '+', std::abs(exponent));
425 };
426
427 std::string text;
428 if (format == 'f') {
429 text = fixed(precision);
430 } else if (format == 'e') {
431 text = scientific(zero ? std::string(static_cast<std::size_t>(precision + 1), '0') : rounded);
432 } else {
433 const int digitsWanted = std::max(precision, 1);
434 if (!zero && (exponent < -4 || exponent >= digitsWanted)) {
435 std::string mantissa = rounded;
436 mantissa.erase(mantissa.find_last_not_of('0') + 1);
437 text = scientific(mantissa);
438 } else {
439 text = fixed(std::max(digitsWanted - 1 - exponent, 0));
440 if (text.find('.') != std::string::npos) {
441 text.erase(text.find_last_not_of('0') + 1);
442 if (text.back() == '.')
443 text.pop_back();
444 }
445 }
446 }
447 return value < 0 ? "-" + text : text;
448}

498std::string mne::to_base64(std::span<const std::byte> data, bool url)
499{
500 const std::string_view alphabet = url ? "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_"
501 : "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
502 const auto at = [&](unsigned bits) {
503 return alphabet[bits & 63U];
504 };
505 std::string text;
506 text.reserve((data.size() + 2) / 3 * 4);
507 std::size_t i = 0;
508 for (; i + 2 < data.size(); i += 3) {
509 const unsigned bits = (std::to_integer<unsigned>(data[i]) << 16) | (std::to_integer<unsigned>(data[i + 1]) << 8) | std::to_integer<unsigned>(data[i + 2]);
510 text += {at(bits >> 18), at(bits >> 12), at(bits >> 6), at(bits)};
511 }
512 if (i < data.size()) {
513 const bool two = i + 1 < data.size();
514 const unsigned bits = (std::to_integer<unsigned>(data[i]) << 16) | (two ? std::to_integer<unsigned>(data[i + 1]) << 8 : 0U);
515 text += {at(bits >> 18), at(bits >> 12)};
516 if (two) {
517 text += at(bits >> 6);
518 }
519 if (!url) {
520 text.append(two ? 1 : 2, '=');
521 }
522 }
523 return text;
524}

528std::optional<std::vector<std::byte>> mne::from_base64(std::string_view text)
529{
530 const auto sextet = [](char c) -> int {
531 if (c >= 'A' && c <= 'Z')
532 return c - 'A';
533 if (c >= 'a' && c <= 'z')
534 return c - 'a' + 26;
535 if (c >= '0' && c <= '9')
536 return c - '0' + 52;
537 if (c == '+' || c == '-')
538 return 62;
539 if (c == '/' || c == '_')
540 return 63;
541 return -1;
542 };
543 std::vector<std::byte> data;
544 data.reserve(text.size() / 4 * 3);
545 unsigned bits = 0;
546 int count = 0;
547 bool padding = false;
548 for (const char c : text) {
549 if (c == ' ' || c == '\t' || c == '\n' || c == '\r') {
550 continue;
551 }
552 if (c == '=') {
553 padding = true;
554 continue;
555 }
556 const int value = sextet(c);
557 if (value < 0 || padding) {
558 return std::nullopt;
559 }
560 bits = (bits << 6) | static_cast<unsigned>(value);
561 if (++count == 4) {
562 data.insert(data.end(), {std::byte(bits >> 16), std::byte(bits >> 8), std::byte(bits)});
563 bits = 0;
564 count = 0;
565 }
566 }
567 if (count == 1) {
568 return std::nullopt;
569 }
570 if (count == 2) {
571 data.push_back(std::byte(bits >> 4));
572 } else if (count == 3) {
573 data.insert(data.end(), {std::byte(bits >> 10), std::byte(bits >> 2)});
574 }
575 return data;
576}

592std::optional<mne::UtcTime> mne::parse_iso8601(std::string_view text)
593{
594 using namespace std::chrono;
595 std::size_t pos = 0;
596 const auto number = [&](std::size_t digits, int& value) {
597 if (pos + digits > text.size()) {
598 return false;
599 }
600 const char* first = text.data() + pos;
601 const auto [end, ec] = std::from_chars(first, first + digits, value);
602 if (ec != std::errc{} || end != first + digits || *first == '-' || *first == '+') {
603 return false;
604 }
605 pos += digits;
606 return true;
607 };
608 const auto expect = [&](char c) {
609 if (pos >= text.size() || text[pos] != c) {
610 return false;
611 }
612 ++pos;
613 return true;
614 };
615
616 int y = 0;
617 int mo = 0;
618 int d = 0;
619 if (!number(4, y) || !expect('-') || !number(2, mo) || !expect('-') || !number(2, d)) {
620 return std::nullopt;
621 }
622 const year_month_day date{year{y}, month{static_cast<unsigned>(mo)}, day{static_cast<unsigned>(d)}};
623 if (!date.ok()) {
624 return std::nullopt;
625 }
626 UtcTime time = sys_days(date);
627 if (pos == text.size()) {
628 return time;
629 }
630 if (text[pos] != 'T' && text[pos] != ' ') {
631 return std::nullopt;
632 }
633 ++pos;
634
635 int h = 0;
636 int mi = 0;
637 int s = 0;
638 if (!number(2, h) || !expect(':') || !number(2, mi) || h > 23 || mi > 59) {
639 return std::nullopt;
640 }
641 if (expect(':')) {
642 if (!number(2, s) || s > 59) {
643 return std::nullopt;
644 }
645 if (expect('.')) {
646 std::int64_t nanos = 0;
647 std::int64_t scale = 100'000'000;
648 const std::size_t first = pos;
649 for (; pos < text.size() && text[pos] >= '0' && text[pos] <= '9'; ++pos, scale /= 10) {
650 nanos += (text[pos] - '0') * scale;
651 }
652 if (pos == first) {
653 return std::nullopt;
654 }
655 time += std::chrono::milliseconds((nanos + 500'000) / 1'000'000); // rounded, as Qt does
656 }
657 }
658 time += hours(h) + minutes(mi) + seconds(s);
659
660 if (pos == text.size() || (text[pos] == 'Z' && pos + 1 == text.size())) {
661 return time;
662 }
663 if (text[pos] != '+' && text[pos] != '-') {
664 return std::nullopt;
665 }
666 const bool east = text[pos++] == '+';
667 int oh = 0;
668 int om = 0;
669 if (!number(2, oh) || oh > 23) {
670 return std::nullopt;
671 }
672 expect(':');
673 if (!number(2, om) || om > 59 || pos != text.size()) {
674 return std::nullopt;
675 }
676 const minutes offset = hours(oh) + minutes(om);
677 return east ? time - offset : time + offset;
678}
