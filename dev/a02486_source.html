58std::unique_ptr<MNENamedMatrix> MNENamedMatrix::build(const StringList& rowlist,
59 const StringList& collist,
60 const Eigen::MatrixXf& data)
61{
62 if ((!rowlist.empty() && std::ssize(rowlist) != data.rows()) || (!collist.empty() && std::ssize(collist) != data.cols())) {
63 mne::Log::critical("MNENamedMatrix::build - {} row and {} column names do not fit a {} x {} matrix.",
64 rowlist.size(), collist.size(), data.rows(), data.cols());
65 return nullptr;
66 }
67 auto mat = std::make_unique<MNENamedMatrix>();
68 mat->rowlist = rowlist;
69 mat->collist = collist;
70 mat->data = data;
71 return mat;
72}

79// Indices of the picked names in the original list; all of them when nothing is picked.
80std::optional<std::vector<Eigen::Index>> pick_indices(std::span<const std::string> picks,
81 const StringList& names,
82 Eigen::Index count,
83 std::string_view what)
84{
85 std::vector<Eigen::Index> indices;
86 if (picks.empty()) {
87 indices.resize(static_cast<std::size_t>(count));
88 std::iota(indices.begin(), indices.end(), Eigen::Index{0});
89 return indices;
90 }
91 if (names.empty()) {
92 mne::Log::critical("MNENamedMatrix::pick - Cannot pick {}s: no {} names in original matrix.", what, what);
93 return std::nullopt;
94 }
95 indices.reserve(picks.size());
96 for (const auto& name : picks) {
97 const auto it = std::ranges::find(names, name);
98 if (it == names.end()) {
99 mne::Log::critical("MNENamedMatrix::pick - {} '{}' not found in original matrix.", what, name);
100 return std::nullopt;
101 }
102 indices.push_back(std::distance(names.begin(), it));
103 }
104 return indices;
105}

109std::unique_ptr<MNENamedMatrix> MNENamedMatrix::pick(std::span<const std::string> pickrowlist,
110 std::span<const std::string> pickcollist) const
111{
112 const auto rows = pick_indices(pickrowlist, rowlist, data.rows(), "row");
113 const auto cols = pick_indices(pickcollist, collist, data.cols(), "column");
114 if (!rows || !cols)
115 return nullptr;
116
117 auto mat = std::make_unique<MNENamedMatrix>();
118 mat->rowlist = pickrowlist.empty() ? rowlist : StringList(pickrowlist.begin(), pickrowlist.end());
119 mat->collist = pickcollist.empty() ? collist : StringList(pickcollist.begin(), pickcollist.end());
120 mat->data = data(*rows, *cols);
121 return mat;
122}

126std::unique_ptr<MNENamedMatrix> MNENamedMatrix::read(FiffStream& stream,
127 const std::shared_ptr<FiffDirNode>& node,
128 int kind)
129{
130 StringList colnames;
131 StringList rownames;
132 int ncol = 0;
133 int nrow = 0;
134 std::int32_t ndim;
135 std::vector<std::int32_t> dims;
136 MatrixXf data;
137 std::optional<FiffTag> t_pTag;
138 bool dataFound = false;
139
140 std::shared_ptr<FiffDirNode> tmp_node = node;
141
142 /*
143 * If the node itself is a FIFFB_MNE_NAMED_MATRIX block, read from it.
144 * Otherwise look in its first-generation children for such a block.
145 */
146 if (tmp_node->type == FIFFB_MNE_NAMED_MATRIX) {
147 if (t_pTag = tmp_node->find_tag(stream, kind); !t_pTag)
148 return nullptr;
149
150 t_pTag->matrix_dimensions(ndim, dims);
151 if (ndim != 2) {
152 mne::Log::critical("MNENamedMatrix::read - Only two-dimensional matrices are supported.");
153 return nullptr;
154 }
155
156 data = t_pTag->to_float_matrix().transpose();
157 dataFound = true;
158 } else {
159 for (int k = 0; k < tmp_node->nchild(); ++k) {
160 if (tmp_node->children[k]->type == FIFFB_MNE_NAMED_MATRIX) {
161 if (t_pTag = tmp_node->children[k]->find_tag(stream, kind); t_pTag) {
162 t_pTag->matrix_dimensions(ndim, dims);
163 if (ndim != 2) {
164 mne::Log::critical("MNENamedMatrix::read - Only two-dimensional matrices are supported.");
165 return nullptr;
166 }
167
168 data = t_pTag->to_float_matrix().transpose();
169 dataFound = true;
170 tmp_node = tmp_node->children[k];
171 break;
172 }
173 }
174 }
175 if (!dataFound)
176 return nullptr;
177 }
178
179 /*
180 * Read optional FIFF_MNE_NROW / FIFF_MNE_NCOL dimension tags and
181 * cross-check them against the matrix data.
182 */
183 if (t_pTag = tmp_node->find_tag(stream, FIFF_MNE_NROW); !t_pTag) {
184 nrow = static_cast<int>(data.rows());
185 } else {
186 nrow = *t_pTag->to_int();
187 if (nrow != data.rows()) {
188 mne::Log::critical("MNENamedMatrix::read - FIFF_MNE_NROW tag ({}) conflicts with matrix data ({}).", nrow, data.rows());
189 return nullptr;
190 }
191 }
192
193 if (t_pTag = tmp_node->find_tag(stream, FIFF_MNE_NCOL); !t_pTag) {
194 ncol = static_cast<int>(data.cols());
195 } else {
196 ncol = *t_pTag->to_int();
197 if (ncol != data.cols()) {
198 mne::Log::critical("MNENamedMatrix::read - FIFF_MNE_NCOL tag ({}) conflicts with matrix data ({}).", ncol, data.cols());
199 return nullptr;
200 }
201 }
202
203 /*
204 * Read optional row and column name lists.
205 */
206 if (t_pTag = tmp_node->find_tag(stream, FIFF_MNE_ROW_NAMES); t_pTag) {
207 const std::string s = t_pTag->to_string();
208 rownames = FiffStream::split_name_list(s);
209 if (static_cast<int>(rownames.size()) != nrow) {
210 mne::Log::critical("MNENamedMatrix::read - Row name count ({}) does not match nrow ({}).", rownames.size(), nrow);
211 return nullptr;
212 }
213 }
214
215 if (t_pTag = tmp_node->find_tag(stream, FIFF_MNE_COL_NAMES); t_pTag) {
216 const std::string s = t_pTag->to_string();
217 colnames = FiffStream::split_name_list(s);
218 if (static_cast<int>(colnames.size()) != ncol) {
219 mne::Log::critical("MNENamedMatrix::read - Column name count ({}) does not match ncol ({}).", colnames.size(), ncol);
220 return nullptr;
221 }
222 }
223
224 return build(rownames, colnames, data);
225}
