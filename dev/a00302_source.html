73void DecodingCsp::fit(const std::vector<MatrixXd>& epochs,
74 const VectorXi& y)
75{
76 if (epochs.empty()) {
77 throw std::invalid_argument("DecodingCsp::fit: epochs must be non-empty");
78 }
79 if (static_cast<int>(epochs.size()) != y.size()) {
80 throw std::invalid_argument(
81 "DecodingCsp::fit: epochs and y must have the same length");
82 }
83
84 // Identify unique classes
85 std::set<int> classSet(y.data(), y.data() + y.size());
86 if (classSet.size() != 2) {
87 throw std::invalid_argument(
88 "DecodingCsp::fit: y must contain exactly 2 unique class labels");
89 }
90
91 auto it = classSet.begin();
92 int label1 = *it++;
93
94 // Split epochs by class
95 std::vector<MatrixXd> epochs1, epochs2;
96 for (int i = 0; i < y.size(); ++i) {
97 if (y(i) == label1) {
98 epochs1.push_back(epochs[static_cast<size_t>(i)]);
99 } else {
100 epochs2.push_back(epochs[static_cast<size_t>(i)]);
101 }
102 }
103
104 const auto n_ch = epochs1[0].rows();
105
106 // Average covariance for each class
107 MatrixXd cov1 = MatrixXd::Zero(n_ch, n_ch);
108 for (const auto& epoch : epochs1) {
109 MatrixXd centered = epoch.colwise() - epoch.rowwise().mean();
110 cov1 += centered * centered.transpose()
111 / static_cast<double>(epoch.cols() - 1);
112 }
113 cov1 /= static_cast<double>(epochs1.size());
114
115 MatrixXd cov2 = MatrixXd::Zero(n_ch, n_ch);
116 for (const auto& epoch : epochs2) {
117 MatrixXd centered = epoch.colwise() - epoch.rowwise().mean();
118 cov2 += centered * centered.transpose()
119 / static_cast<double>(epoch.cols() - 1);
120 }
121 cov2 /= static_cast<double>(epochs2.size());
122
123 // Composite covariance
124 MatrixXd cov_comp = cov1 + cov2;
125
126 // Whitening: W = D^{-1/2} U^T
127 SelfAdjointEigenSolver<MatrixXd> eig_comp(cov_comp);
128 VectorXd d = eig_comp.eigenvalues();
129 MatrixXd U = eig_comp.eigenvectors();
130
131 const double d_min = d.maxCoeff() * 1e-10;
132 for (Index i = 0; i < d.size(); ++i) {
133 if (d(i) < d_min) d(i) = d_min;
134 }
135
136 MatrixXd W = d.array().sqrt().inverse().matrix().asDiagonal()
137 * U.transpose();
138
139 // Whiten class-1 covariance and eigendecompose
140 MatrixXd S1 = W * cov1 * W.transpose();
141 SelfAdjointEigenSolver<MatrixXd> eig_s1(S1);
142 VectorXd lambdas = eig_s1.eigenvalues();
143 MatrixXd B = eig_s1.eigenvectors();
144
145 // All filters in eigenvalue order (ascending)
146 MatrixXd all_filters = B.transpose() * W;
147
148 // Select first and last components
149 int n_per_class = m_nComponents / 2;
150 int n_total = std::min(m_nComponents, static_cast<int>(n_ch));
151 n_per_class = std::min(n_per_class, static_cast<int>(n_ch) / 2);
152
153 m_filters = MatrixXd(n_total, n_ch);
154 VectorXd eigenvalues(n_total);
155
156 // Bottom n_per_class (maximize class-2 variance)
157 for (int i = 0; i < n_per_class; ++i) {
158 m_filters.row(i) = all_filters.row(i);
159 eigenvalues(i) = lambdas(i);
160 }
161 // Top (n_total - n_per_class) (maximize class-1 variance)
162 for (int i = 0; i < n_total - n_per_class; ++i) {
163 m_filters.row(n_per_class + i) =
164 all_filters.row(static_cast<int>(n_ch) - 1 - i);
165 eigenvalues(n_per_class + i) =
166 lambdas(static_cast<int>(n_ch) - 1 - i);
167 }
168
169 // Patterns = pinv(filters) — shape (n_ch × n_comp)
170 auto svd = m_filters.bdcSvd<ComputeThinU | ComputeThinV>();
171 m_patterns = svd.solve(MatrixXd::Identity(n_total, n_total));
172
173 // Compute mean band power for z-score normalisation
174 MatrixXd powerFeatures = computePowerFeatures(epochs);
175 m_mean = powerFeatures.colwise().mean();
176
177 VectorXd centered = VectorXd(powerFeatures.rows());
178 m_std = VectorXd(powerFeatures.cols());
179 for (int c = 0; c < powerFeatures.cols(); ++c) {
180 centered = powerFeatures.col(c).array() - m_mean(c);
181 m_std(c) = std::sqrt(centered.squaredNorm()
182 / static_cast<double>(centered.size()));
183 }
184
185 m_fitted = true;
186}

190MatrixXd DecodingCsp::transform(const std::vector<MatrixXd>& epochs) const
191{
192 if (!m_fitted) {
193 throw std::runtime_error("DecodingCsp::transform: not fitted");
194 }
195
196 if (m_transformInto == TransformMode::CspSpace) {
197 // Return data in CSP space: (n_epochs * n_components, n_times)
198 const int nEpochs = static_cast<int>(epochs.size());
199 const int nComp = static_cast<int>(m_filters.rows());
200 const int nTimes = static_cast<int>(epochs[0].cols());
201
202 MatrixXd result(nEpochs * nComp, nTimes);
203 for (int e = 0; e < nEpochs; ++e) {
204 result.middleRows(static_cast<Eigen::Index>(e) * nComp, nComp) = m_filters * epochs[static_cast<size_t>(e)];
205 }
206 return result;
207 }
208
209 // AveragePower mode
210 MatrixXd X = computePowerFeatures(epochs);
211
212 if (m_useLog) {
213 X = X.array().max(1e-30).log().matrix();
214 } else {
215 // z-score
216 for (int c = 0; c < X.cols(); ++c) {
217 double s = m_std(c);
218 if (s < 1e-15) s = 1.0;
219 X.col(c) = (X.col(c).array() - m_mean(c)) / s;
220 }
221 }
222
223 return X;
224}

237MatrixXd DecodingCsp::inverseTransform(const MatrixXd& X) const
238{
239 if (!m_fitted) {
240 throw std::runtime_error("DecodingCsp::inverseTransform: not fitted");
241 }
242 if (m_transformInto != TransformMode::AveragePower) {
243 throw std::runtime_error(
244 "DecodingCsp::inverseTransform: only valid for AveragePower mode");
245 }
246
247 const int nComp = static_cast<int>(m_filters.rows());
248 if (X.cols() != nComp) {
249 throw std::invalid_argument(
250 "DecodingCsp::inverseTransform: X must have n_components columns");
251 }
252
253 // patterns_ is (n_channels × n_components), X is (n_epochs × n_components)
254 // Result: (n_epochs × n_channels)
255 return X * m_patterns.transpose();
256}

307MatrixXd DecodingCsp::computePowerFeatures(
308 const std::vector<MatrixXd>& epochs) const
309{
310 const int nEpochs = static_cast<int>(epochs.size());
311 const int nComp = static_cast<int>(m_filters.rows());
312
313 MatrixXd features(nEpochs, nComp);
314 for (int e = 0; e < nEpochs; ++e) {
315 MatrixXd filtered = m_filters * epochs[static_cast<size_t>(e)];
316 for (int c = 0; c < nComp; ++c) {
317 features(e, c) = filtered.row(c).squaredNorm()
318 / static_cast<double>(filtered.cols());
319 }
320 }
321
322 return features;
323}
