98AverageLayout AverageLayoutFigure::average_layout() const
99{
100 std::vector<AverageTile> tiles;
101 tiles.reserve(m_channels.size());
102 for (std::size_t i = 0; i < m_channels.size(); ++i) {
103 const Channel& channel = m_channels[i];
104 AverageTile tile;
105 tile.box = AverageLayout::box_at(channel.position);
106 tile.name = channel.name;
107 tile.name_size = static_cast<int>(i) == m_pressedTile ? kPressedNameSize : kNameSize;
108 tile.bad = channel.bad;
109 tile.max_value = channel_scale(m_scales, channel.kind, channel.unit);
110 tile.stim_sample = channel.first_sample;
111 tile.columns = channel.averages.empty() ? 0 : channel.averages.front().samples;
112 for (const Average& average : channel.averages) {
113 // Averages missing from the activation map are drawn
114 if (const auto active = m_activation.find(average.comment); active != m_activation.end() && !active->second) {
115 continue;
116 }
117 AverageTile::Trace trace;
118 const auto color = m_colors.find(average.comment);
119 trace.color = color != m_colors.end() ? color->second : m_signalColor;
120 if (average.data) {
121 trace.samples.reserve(static_cast<std::size_t>(average.samples));
122 for (int s = 0; s < average.samples; ++s) {
123 trace.samples.push_back(average.data[static_cast<std::ptrdiff_t>(s) * average.stride]);
124 }
125 }
126 tile.traces.push_back(std::move(trace));
127 }
128 tiles.push_back(std::move(tile));
129 }
130 AverageLayout layout;
131 layout.set_tiles(std::move(tiles));
132 return layout;
133}

138{
