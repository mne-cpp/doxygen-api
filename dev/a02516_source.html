59int MNEDeriv::validate(const QList<FiffChInfo>& chInfo)
60{
61 const int nrow = deriv_data ? deriv_data->nrow : 0;
62 valid = Eigen::VectorXi::Zero(nrow);
63 chs.clear();
64 for (int j = 0; j < nrow; ++j) {
65 chs.append(FiffChInfo());
66 }
67 if (chInfo.isEmpty() || nrow == 0) {
68 return 0;
69 }
70 QStringList names;
71 for (const FiffChInfo& ch : chInfo) {
72 names << ch.ch_name;
73 }
74 // Row-major view: the inputs of derived channel j are the non-zeros of row j.
75 const Eigen::SparseMatrix<float, Eigen::RowMajor> rows = deriv_data->data->eigen();
76 int nvalid = 0;
77 for (int j = 0; j < nrow; ++j) {
78 const FiffChInfo* first = nullptr;
79 bool ok = true;
80 for (Eigen::SparseMatrix<float, Eigen::RowMajor>::InnerIterator it(rows, j); it && ok; ++it) {
81 const int pick = static_cast<int>(names.indexOf(deriv_data->collist.value(static_cast<int>(it.col()))));
82 if (pick < 0) {
83 ok = false;
84 } else if (!first) {
85 first = &chInfo[pick];
86 } else {
87 const FiffChInfo& ch = chInfo[pick];
88 ok = ch.unit == first->unit && ch.unit_mul == first->unit_mul && ch.kind == first->kind;
89 }
90 }
91 if (ok && first) {
92 valid[j] = 1;
93 chs[j] = *first;
94 ++nvalid;
95 }
96 }
97 return nvalid;
98}
