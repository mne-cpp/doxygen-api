46// A number with at most three decimals, without trailing zeros
47std::string num(double value)
48{
49 if (!std::isfinite(value)) {
50 return "0";
51 }
52 std::string text = std::format("{:.3f}", value);
53 while (text.back() == '0') {
54 text.pop_back();
55 }
56 if (text.back() == '.') {
57 text.pop_back();
58 }
59 return text == "-0" ? "0" : text;
60}

62std::string escape(std::string_view text)
63{
64 std::string out;
65 out.reserve(text.size());
66 for (const char c : text) {
67 switch (c) {
68 case '&':
69 out += "&amp;";
70 break;
71 case '<':
72 out += "&lt;";
73 break;
74 case '>':
75 out += "&gt;";
76 break;
77 case '"':
78 out += "&quot;";
79 break;
80 default:
81 out += c;
82 }
83 }
84 return out;
85}

98std::vector<double> dashPattern(LineStyle style)
99{
100 switch (style) {

123 std::string write(const DisplayList& list)
124 {
125 m_out += std::format("<svg xmlns=\"http://www.w3.org/2000/svg\" xmlns:xlink=\"http://www.w3.org/1999/xlink\" "
126 "width=\"{0}\" height=\"{1}\" viewBox=\"0 0 {0} {1}\">\n",
127 num(list.width), num(list.height));
128 if (!m_options.title.empty()) {
129 m_out += std::format("<title>{}</title>\n", escape(m_options.title));
130 }
131 if (list.background) {
132 m_out += std::format("<rect x=\"0\" y=\"0\" width=\"{}\" height=\"{}\"{}/>\n", num(list.width), num(list.height),
133 fill(Brush::solid(*list.background)));
134 }
135 for (const Command& command : list.commands) {
136 setClips(command.clips);
137 std::visit([&](const auto& shape) { draw(shape, command); }, command.shape);
138 }
139 setClips({});
140 m_out += "</svg>\n";
141 return std::move(m_out);
142 }

144private:
145 // Clip groups stay open while consecutive commands share their clips
146 void setClips(const std::vector<Clip>& clips)
147 {
148 if (clips == m_openClips) {
149 return;
150 }
151 for (std::size_t i = 0; i < m_openClips.size(); ++i) {
152 m_out += "</g>\n";
153 }
154 m_openClips = clips;
155 for (const Clip& clip : clips) {
156 auto it = std::find(m_clipDefs.begin(), m_clipDefs.end(), clip);
157 if (it == m_clipDefs.end()) {
158 m_clipDefs.push_back(clip);
159 it = m_clipDefs.end() - 1;
160 m_out += std::format("<defs><clipPath id=\"c{}\"><rect x=\"{}\" y=\"{}\" width=\"{}\" height=\"{}\"{}/></clipPath></defs>\n",
161 it - m_clipDefs.begin(), num(clip.rect.x), num(clip.rect.y), num(clip.rect.width),
162 num(clip.rect.height), transformAttribute(clip.transform));
163 }
164 m_out += std::format("<g clip-path=\"url(#c{})\">\n", it - m_clipDefs.begin());
165 }
166 }

183 std::string stroke(const Pen& pen) const
184 {
185 if (!pen.visible()) {
186 return " stroke=\"none\"";
187 }
188 const double width = pen.width > 0.0 ? pen.width : 1.0;
189 std::string out = std::format(" stroke=\"{}\"", hex(pen.color));
190 if (pen.color.a < 255) {
191 out += std::format(" stroke-opacity=\"{}\"", num(pen.color.a / 255.0));
192 }
193 out += std::format(" stroke-width=\"{}\"", num(width));
194 if (pen.width <= 0.0) {
195 out += " vector-effect=\"non-scaling-stroke\"";
196 }
197 out += pen.cap == LineCap::Flat ? " stroke-linecap=\"butt\"" : pen.cap == LineCap::Round ? " stroke-linecap=\"round\""
198 : " stroke-linecap=\"square\"";
199 out += pen.join == LineJoin::Miter ? " stroke-linejoin=\"miter\"" : pen.join == LineJoin::Round ? " stroke-linejoin=\"round\""
200 : " stroke-linejoin=\"bevel\"";
201 const std::vector<double> dashes = dashPattern(pen.style);
202 if (!dashes.empty()) {
203 out += " stroke-dasharray=\"";
204 for (std::size_t i = 0; i < dashes.size(); ++i) {
205 out += (i ? " " : "") + num(dashes[i] * width);
206 }
207 out += "\"";
208 }
209 return out;
210 }

212 std::string fill(const Brush& brush)
213 {
214 if (!brush.visible()) {
215 return " fill=\"none\"";
216 }
217 if (brush.style == Brush::Style::Solid) {
218 std::string out = std::format(" fill=\"{}\"", hex(brush.color));
219 if (brush.color.a < 255) {
220 out += std::format(" fill-opacity=\"{}\"", num(brush.color.a / 255.0));
221 }
222 return out;
223 }
224 const int id = m_gradients++;
225 m_out += std::format("<defs><linearGradient id=\"g{}\" gradientUnits=\"userSpaceOnUse\" x1=\"{}\" y1=\"{}\" x2=\"{}\" y2=\"{}\">", id,
226 num(brush.start.x), num(brush.start.y), num(brush.end.x), num(brush.end.y));
227 for (const GradientStop& stop : brush.stops) {
228 m_out += std::format("<stop offset=\"{}\" stop-color=\"{}\"", num(std::clamp(stop.position, 0.0, 1.0)), hex(stop.color));
229 if (stop.color.a < 255) {
230 m_out += std::format(" stop-opacity=\"{}\"", num(stop.color.a / 255.0));
231 }
232 m_out += "/>";
233 }
234 m_out += "</linearGradient></defs>\n";
235 return std::format(" fill=\"url(#g{})\"", id);
236 }

244 void draw(const PathShape& path, const Command& command)
245 {
246 if (path.cubic) {
247 std::string d = "M" + num(path.points[0].x) + "," + num(path.points[0].y);
248 for (std::size_t i = 1; i + 2 < path.points.size(); i += 3) {
249 d += " C";
250 for (std::size_t k = i; k < i + 3; ++k) {
251 d += (k == i ? "" : " ") + num(path.points[k].x) + "," + num(path.points[k].y);
252 }
253 }
254 if (path.closed) {
255 d += " Z";
256 }
257 const std::string paint = path.closed ? fill(path.brush) : std::string(" fill=\"none\"");
258 m_out += std::format("<path d=\"{}\"{}{}{}/>\n", d, paint, stroke(path.pen), transformAttribute(command));
259 return;
260 }
261 std::string points;
262 for (const Point& p : path.points) {
263 points += (points.empty() ? "" : " ") + num(p.x) + "," + num(p.y);
264 }
265 const std::string paint = path.closed ? fill(path.brush) : std::string(" fill=\"none\"");
266 m_out += std::format("<{} points=\"{}\"{}{}{}/>\n", path.closed ? "polygon" : "polyline", points, paint, stroke(path.pen),
267 transformAttribute(command));
268 }

289 void draw(const TextShape& text, const Command& command)
290 {
291 const Font& font = text.font;
292 // The x position comes from the anchor, so only the vertical placement uses the metrics
293 const Point origin = detail::text_origin(text, 0.0, m_metrics.ascent(font), m_metrics.descent(font));
294 std::string_view anchor = "start";
295 double x = text.box.x;
296 if (!has_flag(text.align, Align::Baseline)) {
297 if (has_flag(text.align, Align::Right)) {
298 anchor = "end";
299 x = text.box.right();
300 } else if (has_flag(text.align, Align::HCenter)) {
301 anchor = "middle";
302 x = text.box.center().x;
303 }
304 }
305 std::string attributes = std::format(" font-family=\"{}\" font-size=\"{}\"", escape(font.family.empty() ? m_options.font_family : font.family),
306 num(m_metrics.pixel_size(font)));
307 if (font.bold) {
308 attributes += " font-weight=\"bold\"";
309 }
310 if (font.italic) {
311 attributes += " font-style=\"italic\"";
312 }
313 if (anchor != "start") {
314 attributes += std::format(" text-anchor=\"{}\"", anchor);
315 }
316 attributes += std::format(" fill=\"{}\"", hex(text.color));
317 if (text.color.a < 255) {
318 attributes += std::format(" fill-opacity=\"{}\"", num(text.color.a / 255.0));
319 }
320 m_out += std::format("<text x=\"{}\" y=\"{}\"{}{} xml:space=\"preserve\">{}</text>\n", num(x), num(origin.y), attributes,
321 transformAttribute(command), escape(text.text));
322 }

324 void draw(const ImageShape& image, const Command& command)
325 {
326 RasterImage raster(image.image->width, image.image->height);
327 for (int y = 0; y < raster.height(); ++y) {
328 for (int x = 0; x < raster.width(); ++x) {
329 const Color c = image.image->pixel(x, y);
330 raster.set_pixel(x, y, {c.r, c.g, c.b});
331 }
332 }
333 const std::vector<std::uint8_t> png = raster.encode_png();
334 const std::string data = to_base64(std::as_bytes(std::span(png)));
335 m_out += std::format("<image x=\"{}\" y=\"{}\" width=\"{}\" height=\"{}\" preserveAspectRatio=\"none\"{}{} xlink:href=\"data:image/png;base64,{}\"/>\n",
336 num(image.target.x), num(image.target.y), num(image.target.width), num(image.target.height),
337 image.smooth ? "" : " style=\"image-rendering:pixelated\"", transformAttribute(command), data);
338 }
