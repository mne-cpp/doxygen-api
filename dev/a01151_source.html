95void LineChart::add_series(const std::vector<double>& x, const std::vector<double>& y, std::string label, std::optional<Color> color,
96 LineStyle style)
97{
98 const std::size_t n = std::min(x.size(), y.size());
99 const bool empty = std::all_of(m_series.cbegin(), m_series.cend(), [](const Series& s) { return s.x.empty(); });
100 for (std::size_t i = 0; i < n; ++i) {
101 const bool first = empty && i == 0;
102 m_minX = first ? x[i] : std::min(m_minX, x[i]);
103 m_maxX = first ? x[i] : std::max(m_maxX, x[i]);
104 m_minY = first ? y[i] : std::min(m_minY, y[i]);
105 m_maxY = first ? y[i] : std::max(m_maxY, y[i]);
106 }
107 Color lineColor = kPalette[m_series.size() % kPalette.size()];
108 if (color) {
109 lineColor = *color;
110 } else {
111 for (const Color& candidate : kPalette) {
112 if (std::none_of(m_series.cbegin(), m_series.cend(), [&](const Series& s) { return s.color == candidate; })) {
113 lineColor = candidate;
114 break;
115 }
116 }
117 }
118 m_series.push_back({std::vector<double>(x.begin(), x.begin() + static_cast<std::ptrdiff_t>(n)),
119 std::vector<double>(y.begin(), y.begin() + static_cast<std::ptrdiff_t>(n)), std::move(label), lineColor, style});
120}

141void LineChart::draw(Canvas& canvas, const TextMetrics& metrics, const Font& font) const
142{
143 const double width = canvas.width();
144 const double height = canvas.height();
145 canvas.save();
146 canvas.set_antialiasing(true);
147 const Color text{51, 51, 51};
148 const Color grid{230, 230, 230};
149
150 double minY = m_minY;
151 double maxY = m_maxY;
152 const double padY = (maxY - minY) * 0.05;
153 minY -= padY == 0.0 ? 1.0 : padY;
154 maxY += padY == 0.0 ? 1.0 : padY;
155 const double minX = m_minX;
156 const double maxX = m_maxX > m_minX ? m_maxX : m_minX + 1.0;
157
158 const std::vector<double> yTicks = nice_ticks(minY, maxY, 6);
159 const std::vector<double> xTicks = nice_ticks(minX, maxX, 8);
160 double tickWidth = 0.0;
161 for (double v : yTicks) {
162 tickWidth = std::max(tickWidth, metrics.advance(font, tick_label(v)));
163 }
164 const double line = metrics.height(font);
165 const double mLeft = tickWidth + 14 + (m_yLabel.empty() ? 0 : line + 6);
166 const double mRight = 24;
167 const double mTop = m_title.empty() ? 16 : 2 * line + 8;
168 const double mBottom = line + 14 + (m_xLabel.empty() ? 0 : line + 6);
169 Rect area{mLeft, mTop, width - mLeft - mRight, height - mTop - mBottom};
170 if (area.empty()) {
171 canvas.restore();
172 return;
173 }
174
175 const auto toPixelX = [&](double dx) {
176 return area.left() + (dx - minX) / (maxX - minX) * area.width;
177 };
178 const auto toPixelY = [&](double dy) {
179 return area.bottom() - (dy - minY) / (maxY - minY) * area.height;
180 };
181 // Length of the lines inside a rectangle, sampled every 2 pixels
182 const auto hiddenLength = [&](const Rect& rect) {
183 double hidden = 0.0;
184 for (const Series& series : m_series) {
185 for (std::size_t i = 1; i < series.x.size(); ++i) {
186 const Point a{toPixelX(series.x[i - 1]), toPixelY(series.y[i - 1])};
187 const Point b{toPixelX(series.x[i]), toPixelY(series.y[i])};
188 const double length = std::hypot(b.x - a.x, b.y - a.y);
189 const int steps = std::max(1, static_cast<int>(length / 2.0));
190 for (int s = 0; s <= steps; ++s) {
191 hidden += contains(rect, a + (b - a) * (static_cast<double>(s) / steps)) ? length / (steps + 1) : 0.0;
192 }
193 }
194 }
195 return hidden;
196 };
197
198 // Legend: in the corner of the axes that hides the least of the lines, else to the right of them
199 double labelWidth = 0.0;
200 int labelCount = 0;
201 for (const Series& series : m_series) {
202 if (!series.label.empty()) {
203 labelWidth = std::max(labelWidth, metrics.advance(font, series.label));
204 ++labelCount;
205 }
206 }
207 const double legendWidth = labelWidth + 38;
208 const double legendHeight = labelCount * (line + 2) + 10;
209 Rect legend;
210 if (labelCount > 0) {
211 const std::array<Rect, 4> corners = {Rect{area.right() - legendWidth - 8, area.top() + 8, legendWidth, legendHeight},
212 Rect{area.left() + 8, area.top() + 8, legendWidth, legendHeight},
213 Rect{area.right() - legendWidth - 8, area.bottom() - legendHeight - 8, legendWidth, legendHeight},
214 Rect{area.left() + 8, area.bottom() - legendHeight - 8, legendWidth, legendHeight}};
215 double fewest = std::numeric_limits<double>::max();
216 for (const Rect& corner : corners) {
217 const double hidden = hiddenLength(corner);
218 if (hidden < fewest) {
219 fewest = hidden;
220 legend = corner;
221 }
222 }
223 if (fewest > 0.0 && area.width > 2 * legendWidth) {
224 area.width -= legendWidth + 12;
225 legend = Rect{area.right() + 12, area.top(), legendWidth, legendHeight};
226 }
227 }
228
229 if (!m_title.empty()) {
230 Font titleFont = scaled_font(font, 1.25, metrics);
231 titleFont.bold = true;
232 canvas.set_font(titleFont);
233 canvas.set_pen(text);
234 canvas.draw_text(Rect{0, 4, width, mTop - 4}, Align::Center, m_title);
235 }
236
237 canvas.set_font(font);
238 for (double v : yTicks) {
239 const double py = toPixelY(v);
240 canvas.set_pen(Pen{grid, 1});
241 canvas.draw_line({area.left(), py}, {area.right(), py});
242 canvas.set_pen(text);
243 canvas.draw_text(Rect{0, py - line, area.left() - 8, 2 * line}, Align::Right | Align::VCenter, tick_label(v));
244 }
245 for (double v : xTicks) {
246 const double px = toPixelX(v);
247 canvas.set_pen(Pen{grid, 1});
248 canvas.draw_line({px, area.top()}, {px, area.bottom()});
249 canvas.set_pen(text);
250 canvas.draw_text(Rect{px - 50, area.bottom() + 6, 100, line}, Align::HCenter | Align::Top, tick_label(v));
251 }
252
253 if (!m_xLabel.empty()) {
254 canvas.draw_text(Rect{area.left(), height - line - 6, area.width, line}, Align::Center, m_xLabel);
255 }
256 if (!m_yLabel.empty()) {
257 canvas.save();
258 canvas.translate(6, area.center().y);
259 canvas.rotate(-90);
260 canvas.draw_text(Rect{-area.height / 2, 0, area.height, line}, Align::Center, m_yLabel);
261 canvas.restore();
262 }
263
264 canvas.save();
265 canvas.clip_rect(area);
266 const double penWidth = m_series.size() > 20 ? 1.0 : 1.75;
267 for (const Series& series : m_series) {
268 if (series.x.size() < 2) {
269 continue;
270 }
271 std::vector<Point> points;
272 points.reserve(series.x.size());
273 for (std::size_t i = 0; i < series.x.size(); ++i) {
274 points.push_back({toPixelX(series.x[i]), toPixelY(series.y[i])});
275 }
276 canvas.set_pen(Pen{series.color, penWidth, series.style, series.style == LineStyle::Solid ? LineCap::Round : LineCap::Flat, LineJoin::Round});
277 canvas.draw_polyline(points);
278 }
279 canvas.restore();
280
281 canvas.set_pen(Pen{{136, 136, 136}, 1});
282 canvas.set_brush(Brush{});
283 canvas.draw_rect(area);
284
285 if (labelCount > 0) {
286 canvas.set_pen(Pen{{204, 204, 204}, 1});
287 canvas.set_brush(Brush::solid({255, 255, 255, 225}));
288 canvas.draw_rounded_rect(legend, 4, 4);
289 int row = 0;
290 for (const Series& series : m_series) {
291 if (series.label.empty()) {
292 continue;
293 }
294 const double y = legend.top() + 5 + row++ * (line + 2) + line / 2.0;
295 canvas.set_pen(Pen{series.color, 2.5, series.style, series.style == LineStyle::Solid ? LineCap::Round : LineCap::Flat});
296 canvas.draw_line({legend.left() + 8, y}, {legend.left() + 26, y});
297 canvas.set_pen(text);
298 canvas.draw_text(Rect{legend.left() + 32, y - line / 2.0, labelWidth + 4, line}, Align::Left | Align::VCenter, series.label);
299 }
300 }
301 canvas.restore();
302}
