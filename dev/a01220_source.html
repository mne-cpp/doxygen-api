56std::uint32_t get_big(std::span<const std::byte> data, std::size_t offset, std::size_t bytes)
57{
58 std::uint32_t value = 0;
59 for (std::size_t i = 0; i < bytes; ++i) {
60 value = (value << 8) | std::to_integer<std::uint32_t>(data[offset + i]);
61 }
62 return value;
63}

66std::vector<std::byte> utf16(std::string_view text)
67{
68 std::vector<std::byte> out;
69 out.reserve(2 * text.size());
70 for (std::size_t i = 0; i < text.size();) {
71 const auto lead = static_cast<unsigned char>(text[i]);
72 const std::size_t length = lead < 0x80 ? 1 : lead < 0xE0 ? 2
73 : lead < 0xF0 ? 3
74 : 4;
75 std::uint32_t code = length == 1 ? lead : lead & (0x7FU >> length);
76 for (std::size_t k = 1; k < length && i + k < text.size(); ++k) {
77 code = (code << 6) | (static_cast<unsigned char>(text[i + k]) & 0x3FU);
78 }
79 i += length;
80 if (code >= 0x10000) {
81 code -= 0x10000;
82 put_big(out, 0xD800 + (code >> 10), 2);
83 put_big(out, 0xDC00 + (code & 0x3FF), 2);
84 } else {
85 put_big(out, code, 2);
86 }
87 }
88 return out;
89}

92std::string utf8(std::span<const std::byte> text)
93{
94 std::string out;
95 out.reserve(text.size() / 2);
96 for (std::size_t i = 0; i + 1 < text.size(); i += 2) {
97 std::uint32_t code = get_big(text, i, 2);
98 if (code >= 0xD800 && code < 0xDC00 && i + 3 < text.size()) {
99 code = 0x10000 + ((code - 0xD800) << 10) + (get_big(text, i + 2, 2) - 0xDC00);
100 i += 2;
101 }
102 const auto byte = [](std::uint32_t v) {
103 return static_cast<char>(v);
104 };
105 if (code < 0x80) {
106 out += byte(code);
107 } else if (code < 0x800) {
108 out += {byte(0xC0 | (code >> 6)), byte(0x80 | (code & 0x3F))};
109 } else if (code < 0x10000) {
110 out += {byte(0xE0 | (code >> 12)), byte(0x80 | ((code >> 6) & 0x3F)), byte(0x80 | (code & 0x3F))};
111 } else {
112 out += {byte(0xF0 | (code >> 18)), byte(0x80 | ((code >> 12) & 0x3F)), byte(0x80 | ((code >> 6) & 0x3F)), byte(0x80 | (code & 0x3F))};
113 }
114 }
115 return out;

133 AcqExpected<std::string> command(std::string_view line, std::chrono::milliseconds timeout) override
134 {
135 // Replies to earlier commands, e.g. a second one to "start", belong to those
136 while (true) {
137 auto stale = reply(std::chrono::milliseconds(0));
138 if (!stale) {
139 if (stale.error() != AcqErrorCode::Timeout) {
140 return stale.error();
141 }
142 break;
143 }
144 }
145 const std::vector<std::byte> text = utf16(line);
146 std::vector<std::byte> frame;
147 put_big(frame, static_cast<std::uint32_t>(4 + text.size()), 2);
148 put_big(frame, static_cast<std::uint32_t>(text.size()), 4);
149 frame.insert(frame.end(), text.begin(), text.end());
150 if (const std::error_code e = m_commands.write(frame)) {
151 return transport_error(e, AcqErrorCode::DeviceIo, std::format("cannot send to {}", m_name));
152 }
153 return reply(timeout);
154 }

156 AcqError send(std::int32_t command, std::string_view text) override
157 {
158 std::vector<std::byte> tag;
159 put_big(tag, static_cast<std::uint32_t>(rt::kCommand), 4);
160 put_big(tag, static_cast<std::uint32_t>(fiff::kTypeVoid), 4);
161 put_big(tag, static_cast<std::uint32_t>(4 + text.size()), 4);
162 put_big(tag, 0, 4);
163 put_big(tag, static_cast<std::uint32_t>(command), 4);
164 const auto* chars = reinterpret_cast<const std::byte*>(text.data());
165 tag.insert(tag.end(), chars, chars + text.size());
166 return transport_error(m_data.write(tag), AcqErrorCode::DeviceIo, std::format("cannot send to {}", m_name));
167 }

169 AcqExpected<const NeuromagTag*> next(std::chrono::milliseconds timeout) override
170 {
171 std::array<std::byte, kTagHeader> header{};
172 if (const std::error_code e = m_data.read(header, timeout)) {
173 return transport_error(e, AcqErrorCode::DeviceIo, std::format("{} failed", m_name));
174 }
175 const std::uint32_t size = get_big(header, 8, 4);
176 if (size > kMaxTag) {
177 return AcqError(AcqErrorCode::DeviceIo, std::format("{} sent a tag of {} bytes", m_name, size));
178 }
179 m_tag.kind = static_cast<int>(get_big(header, 0, 4));
180 m_tag.type = static_cast<int>(get_big(header, 4, 4));
181 m_tag.data.resize(size);
182 if (m_data.read(m_tag.data, std::chrono::milliseconds(5000))) {
183 return AcqError(AcqErrorCode::DeviceIo, std::format("{} sent an incomplete tag", m_name));
184 }
185 big_to_native(m_tag.type, m_tag.data);
186 return &m_tag;
187 }

189private:
191 AcqExpected<std::string> reply(std::chrono::milliseconds timeout)
192 {
193 std::array<std::byte, 2> size{};
194 if (const std::error_code e = m_commands.read(size, timeout)) {
195 return transport_error(e, AcqErrorCode::DeviceIo, std::format("{} failed", m_name));
196 }
197 std::vector<std::byte> frame(get_big(size, 0, 2));
198 if (m_commands.read(frame, std::chrono::milliseconds(5000))) {
199 return AcqError(AcqErrorCode::DeviceIo, std::format("{} sent an incomplete reply", m_name));
200 }
201 if (frame.size() < 4) {
202 return std::string();
203 }
204 const std::uint32_t length = get_big(frame, 0, 4); // 0xFFFFFFFF for a null string
205 return utf8(std::span<const std::byte>(frame).subspan(4, std::min<std::size_t>(length, frame.size() - 4)));
206 }

220AcqExpected<std::unique_ptr<RtServerTransport>> mne::detail::connect_rtserver(const std::string& host, std::uint16_t port,
221 std::chrono::milliseconds timeout)
222{
223 if (port == 65535) {
224 return AcqError(AcqErrorCode::InvalidParameter, "the data port follows the command port; 65535 leaves none");
225 }
226 auto commands = TcpSocket::connect(host, port, timeout);
227 if (!commands) {
228 return AcqError(AcqErrorCode::DeviceNotFound, std::format("no mne_rt_server: cannot connect to {}:{} ({})", host, port, commands.error().message()));
229 }
230 auto data = TcpSocket::connect(host, static_cast<std::uint16_t>(port + 1), timeout);
231 if (!data) {
232 return AcqError(AcqErrorCode::DeviceNotFound,
233 std::format("no mne_rt_server data port: cannot connect to {}:{} ({})", host, port + 1, data.error().message()));
234 }
235 return std::unique_ptr<RtServerTransport>(std::make_unique<RtServerClient>(std::move(*commands), std::move(*data), std::format("{}:{}", host, port)));
