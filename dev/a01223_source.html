306template<typename T>
307void Spline::updatePlot(const Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& matClassLimitData,
308 const Eigen::VectorXi& matClassFrequencyData)
309{
310 int iClassAmount = std::min(static_cast<int>(matClassFrequencyData.rows()),
311 static_cast<int>(matClassLimitData.rows()) - 1);
312 if (iClassAmount < 1)
313 return;
314 m_dMinAxisX = static_cast<double>(matClassLimitData(0));
315 m_dMaxAxisX = static_cast<double>(matClassLimitData(iClassAmount));
316
317 m_seriesData.clear();
318 m_iMaximumFrequency = 0;
319
320 for (int ir = 0; ir < iClassAmount; ++ir) {
321 double classMark = (static_cast<double>(matClassLimitData(ir)) + static_cast<double>(matClassLimitData(ir + 1))) / 2.0;
322 m_seriesData.push_back(QPointF(classMark, matClassFrequencyData(ir)));
323 if (matClassFrequencyData(ir) > m_iMaximumFrequency) {
324 m_iMaximumFrequency = matClassFrequencyData(ir);
325 }
326 }
327
328 m_bHasData = true;
329 m_dLeftThreshold = m_dMinAxisX;
330 m_dMiddleThreshold = m_dMaxAxisX;
331 m_dRightThreshold = m_dMaxAxisX;
332 m_bHasThresholds = false;
333
334 QWidget::update();
335}
