79AcqExpected<Config> parseConfig(const AcqUri& uri)
80{
81 Config c;
82 const auto simulate = paramBool(uri, "simulate", false);
83 const auto seed = paramInt(uri, "seed", 0, 0, INT32_MAX);
84 auto thread = paramThreadPolicy(uri);
85 for (const AcqError& error : {simulate.error(), seed.error(), thread.error()}) {
86 if (!error.ok()) {
87 return error;
88 }
89 }
90 c.simulate = *simulate;
91 c.seed = *seed;
92 c.thread = std::move(*thread);
93 if (const auto board = uri.param("board")) {
94 c.board = std::string(*board);
95 } else if (!c.simulate) {
96 return invalid("streaming_board:// needs board=, the BrainFlow board behind the stream, e.g. board=synthetic");
97 }
98 if (c.simulate) {
99 if (!uri.target.empty()) {
100 return invalid("target '" + uri.target + "': the simulator streams to itself");
101 }
102 return c;
103 }
104 const std::size_t colon = uri.target.rfind(':');
105 int port = 0;
106 if (colon != std::string::npos) {
107 const char* first = uri.target.data() + colon + 1;
108 const char* last = uri.target.data() + uri.target.size();
109 const auto [end, error] = std::from_chars(first, last, port);
110 if (error != std::errc() || end != last) {
111 port = 0;
112 }
113 }
114 if (colon == std::string::npos || port <= 0 || port > 65535) {
115 return invalid("streaming_board:// needs <address>:<port> as BrainFlow's add_streamer has it, e.g. "
116 "streaming_board://225.1.1.1:6677?board=synthetic");
117 }
118 c.address = uri.target.substr(0, colon);
119 c.port = static_cast<std::uint16_t>(port);
120 return c;
121}

127 static AcqExpected<std::unique_ptr<Simulator>> start(const BrainFlowDescription& description, std::uint16_t port, std::int64_t seed)
128 {
129 auto streamer = BrainFlowStreamer::open(description, "127.0.0.1", port, kSimulatedBatch);
130 if (!streamer) {
131 return streamer.error();
132 }
133 const std::size_t eeg = streamer->eeg_rows().size();
134 auto session = AcqDevice::open(std::format("synthetic://?sfreq={}&eeg={}&stim=1&seed={}&realtime=1", description.sampling_rate,
135 std::max<std::size_t>(eeg, 1), seed));
136 if (!session) {
137 return session.error();
138 }
139 AcqReaderOptions gating;
140 gating.gating = true;
141 auto reader = session->stream().reader(AcqReader::ReadMode::Lossless, gating);
142 if (!reader) {
143 return reader.error();
144 }
145 std::unique_ptr<Simulator> simulator(new Simulator(std::move(*streamer), std::move(*session), std::move(*reader), description.sampling_rate));
146 if (AcqError e = simulator->m_session.start(); !e.ok()) {
147 return e;
148 }
149 simulator->m_thread = std::thread([s = simulator.get()] { s->run(); });
150 return simulator;
151 }

174 void run()
175 {
176 const std::size_t channels = m_reader.stream().info()->channelCount(); // EEG, then the trigger code
177 std::vector<double> eeg(channels - 1);
178 const double t0 = std::chrono::duration<double>(std::chrono::system_clock::now().time_since_epoch()).count();
179 std::uint64_t sample = 0;
180 float previous = 0.0F;
181 while (!m_stop.load(std::memory_order_acquire)) {
182 auto view = m_reader.acquire(1, kWait);
183 if (!view) {
184 if (view.error() == AcqErrorCode::Timeout) {
185 continue;
186 }
187 return;
188 }
189 for (std::size_t p = 0; p < view->partCount(); ++p) {
190 const AcqBlockPart& part = view->part(p);
191 for (std::size_t s = 0; s < part.samples; ++s, ++sample) {
192 for (std::size_t c = 0; c + 1 < channels; ++c) {
193 float volts = 0.0F;
194 std::memcpy(&volts, part.at(c, s), sizeof(volts));
195 eeg[c] = static_cast<double>(volts) * 1e6;
196 }
197 float code = 0.0F;
198 std::memcpy(&code, part.at(channels - 1, s), sizeof(code));
199 const double marker = previous == 0.0F ? code : 0.0; // a marker at the onset of the trigger pulse
200 previous = code;
201 if (!m_streamer.push(eeg, marker, t0 + static_cast<double>(sample) / m_sfreq).ok()) {
202 static_cast<void>(m_reader.release(*view));
203 return;
204 }
205 }
206 }
207 if (!m_reader.release(*view).ok()) {
208 return;
209 }
210 }
211 }

215 AcqReader m_reader;
216 double m_sfreq = 0.0;
217 std::thread m_thread;
218 std::atomic<bool> m_stop{false};

221class StreamingBoardBackend final : public AcqBackend
222{
223public:
224 static AcqExpected<std::unique_ptr<AcqBackend>> open(const AcqUri& uri)
225 {
226 auto config = parseConfig(uri);
227 if (!config) {
228 return config.error();
229 }
230 auto board = brainflow_board(config->board);
231 if (!board) {
232 return board.error();
233 }
234 auto& [id, description] = *board;
235 if (description.sampling_rate <= 0 || description.rows == 0) {
236 return AcqError(AcqErrorCode::InvalidParameter, std::format("BrainFlow board {} describes no samples", id));
237 }
238 auto socket = mne::UdpSocket::bind(config->address, config->port);
239 if (!socket) {
240 const AcqErrorCode code = socket.error() == std::errc::invalid_argument ? AcqErrorCode::InvalidParameter : AcqErrorCode::DeviceIo;
241 return AcqError(code, std::format("cannot receive on {}:{}: {}", config->address, config->port, socket.error().message()));
242 }
243 std::unique_ptr<StreamingBoardBackend> backend(new StreamingBoardBackend(std::move(*config), std::move(*socket), id, description, uri.toString()));
244 AcqStreamOptions options;
245 options.capacitySamples = static_cast<std::size_t>(std::max(4 * description.sampling_rate, 1024));
246 auto writer = backend->addInputStream(brainflow_stream_info(description, std::format("streaming_board:{}", id)), options);
247 if (!writer) {
248 return writer.error();
249 }
250 backend->m_writer = BrainFlowRowWriter(**writer, description);
251 return std::unique_ptr<AcqBackend>(std::move(backend));
252 }

268 AcqError start() override
269 {
270 m_stop.store(false, std::memory_order_relaxed);
271 m_writer.restart();
272 auto thread = start_thread(m_config.thread, [this] { run(); });
273 if (!thread) {
274 return thread.error();
275 }
276 m_thread = std::move(*thread);

284 }

308 void run()
309 {
310 const std::size_t rows = m_description.rows;
311 const std::size_t frame = rows * sizeof(double);
312 const auto stopping = [this] {
313 return m_stop.load(std::memory_order_acquire);
314 };
315 while (!stopping()) {
316 const auto got = m_socket.receive(std::as_writable_bytes(std::span(m_datagram)), kWait);
317 if (!got) {
318 if (got.error() == std::errc::timed_out) {
319 continue;
320 }
321 reportError({AcqErrorCode::DeviceIo, "receiving: " + got.error().message()});
322 return;
323 }
324 if (*got == 0 || *got % frame != 0) {
325 continue; // not whole samples of this board, as BrainFlow's STREAMING_BOARD drops them
326 }
327 const AcqError written = m_writer.write(m_datagram.data(), *got / frame, rows, 1, stopping);
328 if (written.code() == AcqErrorCode::DeviceStopped) {
329 return;
330 }
331 if (!written.ok()) {
332 reportError(written);
333 return;
334 }
335 }
336 }
