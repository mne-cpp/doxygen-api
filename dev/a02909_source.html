70std::unique_ptr<FwdThreadArg> FwdThreadArg::create_eeg_multi_thread_duplicate(FwdThreadArg& one, bool bem_model)
71{
72 auto res = std::make_unique<FwdThreadArg>();
73
74 *res = one;
75 res->client_free = nullptr; /* Don't copy the source's deleter */
76 if (bem_model) {
77 auto bem = std::make_shared<FwdBemModel>();
78 *bem = *static_cast<FwdBemModel*>(res->client);
79 bem->v0.resize(0);
80 res->client = bem.get();
81 res->client_free = [bem]() {
82 }; /* shared_ptr releases FwdBemModel on destruction */
83 }
84 return res;
85}

89std::unique_ptr<FwdThreadArg> FwdThreadArg::create_meg_multi_thread_duplicate(FwdThreadArg& one, bool bem_model)
90{
91 auto res = std::make_unique<FwdThreadArg>();
92 FwdCompData* orig = static_cast<FwdCompData*>(one.client);
93
94 *res = one;
95 res->client_free = nullptr; /* Don't copy the source's deleter */
96
97 // Each thread gets its own compensation set (it is mutated while applying)
98 // and work buffers; the compensation coils are read-only and shared.
99 auto comp = std::make_shared<FwdCompData>(*orig);
100 comp->set = orig->set ? std::make_shared<MNECTFCompDataSet>(*orig->set) : nullptr;
101 comp->work.resize(0);
102 comp->vec_work.resize(0, 0);
103
104 res->client = comp.get();
105
106 if (bem_model) {
107 auto bem = std::make_shared<FwdBemModel>();
108 *bem = *static_cast<FwdBemModel*>(comp->client);
109 bem->v0.resize(0);
110 comp->client = bem.get();
111 /* shared_ptrs release their objects when client_free is destroyed */
112 res->client_free = [comp, bem]() {
113 };
114 } else {
115 res->client_free = [comp]() {
116 };
117 }
118 return res;
119}
