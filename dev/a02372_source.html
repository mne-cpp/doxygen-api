41MatrixXd MNECorticalMap::makeCorticalMap(
42 const MNEForwardSolution& fwd,
43 const MNEInverseOperator& inv,
44 [[maybe_unused]] const FiffInfo& info)
45{
46 // Get the inverse kernel (nSources x nChannels)
47 const MatrixXd& kernel = const_cast<MNEInverseOperator&>(inv).getKernel();
48 if (kernel.rows() == 0 || kernel.cols() == 0) {
49 mne::Log::warning("MNECorticalMap::makeCorticalMap - Inverse kernel is empty. Make sure the inverse operator has been prepared (assemble_kernel).");
50 return MatrixXd();
51 }
52
53 if (fwd.sol.data.rows() == 0) {
54 mne::Log::warning("MNECorticalMap::makeCorticalMap - Forward solution is empty.");
55 return MatrixXd();
56 }
57 // A loose inverse kernel is in surface coordinates; rotate a Cartesian free forward to match
58 // (mne-python _convert_forward_match_inv).
59 MNEForwardSolution matched = fwd;
60 const bool looseInverse = inv.orient_prior.data.size() > 0 && (inv.orient_prior.data.array() < 1.0).any();
61 if (looseInverse && !matched.isFixedOrient())
62 matched.convert_to_surf_ori();
63
64 // The kernel's columns are the inverse operator's channels; pick those rows of the gain by name, as
65 // mne-python's make_inverse_resolution_matrix does.
66 MatrixXd gain = matched.sol.data;
67 const StringList& invChannels = inv.eigen_fields.col_names;
68 if (static_cast<int>(invChannels.size()) == kernel.cols() && static_cast<int>(matched.sol.row_names.size()) == gain.rows()) {
69 MatrixXd picked(kernel.cols(), gain.cols());
70 for (int c = 0; c < static_cast<int>(invChannels.size()); ++c) {
71 const int row = static_cast<int>(mne::channel_index(matched.sol.row_names, invChannels[c]));
72 if (row < 0) {
73 mne::Log::warning("MNECorticalMap::makeCorticalMap - Channel {} is not in the forward solution.", invChannels[c]);
74 return MatrixXd();
75 }
76 picked.row(c) = gain.row(row);
77 }
78 gain = picked;
79 }
80
81 // Verify dimension compatibility
82 if (kernel.cols() != gain.rows()) {
83 mne::Log::warning("MNECorticalMap::makeCorticalMap - Dimension mismatch: kernel is {} x {}, gain is {} x {}", kernel.rows(), kernel.cols(), gain.rows(), gain.cols());
84 return MatrixXd();
85 }
86
87 return kernel * gain;
88}
