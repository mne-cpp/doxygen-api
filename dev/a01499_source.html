98template<typename T>
99void storeRecord(T& dst, const T& src) noexcept
100{
101 static_assert(std::is_trivially_copyable_v<T> && sizeof(T) % 8 == 0 && alignof(T) >= 8);
102 std::array<std::uint64_t, sizeof(T) / 8> words{};
103 std::memcpy(words.data(), &src, sizeof(T));
104 auto* raw = reinterpret_cast<std::uint64_t*>(&dst);
105 for (std::size_t k = 0; k < words.size(); ++k) {
106 std::atomic_ref<std::uint64_t>(raw[k]).store(words[k], std::memory_order_relaxed);
107 }
108}

123template<typename T>
124class SeqlockValue
125{
126 static_assert(std::is_trivially_copyable_v<T>);
127 static constexpr std::size_t kWords = (sizeof(T) + 7) / 8;
128
129public:
130 static constexpr std::uint64_t kSeqNone = ~std::uint64_t{0};
131
132 explicit SeqlockValue(const T& value = T()) noexcept
133 {
134 store(value);
135 }
136
137 void store(const T& value) noexcept
138 {
139 std::array<std::uint64_t, kWords> words{};
140 std::memcpy(words.data(), &value, sizeof(T));
141 const std::uint64_t seq = m_seq.load(std::memory_order_relaxed);
142 m_seq.store(seq + 1, std::memory_order_relaxed);
143 std::atomic_thread_fence(std::memory_order_release);
144 for (std::size_t i = 0; i < kWords; ++i) {
145 m_words[i].store(words[i], std::memory_order_relaxed);
146 }
147 m_seq.store(seq + 2, std::memory_order_release);
148 }
149
150 [[nodiscard]] T load() const noexcept
151 {
152 T value;
153 std::uint64_t seen = kSeqNone;
154 refresh(value, seen);
155 return value;
156 }
157
162 void refresh(T& cache, std::uint64_t& seen) const noexcept
163 {
164 if (m_seq.load(std::memory_order_acquire) == seen) {
165 return;
166 }
167 std::array<std::uint64_t, kWords> words{};
168 std::uint64_t before = 0;
169 std::uint64_t after = 0;
170 do {
171 before = m_seq.load(std::memory_order_acquire);
172 for (std::size_t i = 0; i < kWords; ++i) {
173 words[i] = m_words[i].load(std::memory_order_relaxed);
174 }
175 std::atomic_thread_fence(std::memory_order_acquire);
176 after = m_seq.load(std::memory_order_relaxed);
177 } while (before != after || (before & 1U) != 0);
178 std::memcpy(static_cast<void*>(&cache), words.data(), sizeof(T)); // T is trivially copyable (asserted)
179 seen = before;
180 }
181
182private:
183 std::atomic<std::uint64_t> m_seq{0};
184 std::array<std::atomic<std::uint64_t>, kWords> m_words{};
185};

387 template<typename Byte>
388 std::size_t buildParts(std::uint64_t first, std::uint64_t count, std::array<AcqPart<Byte>, 2>& parts) const noexcept
389 {
390 parts = {};
391 if (count == 0) {
392 return 0;
393 }
394 const std::uint64_t index = first & mask;
395 const std::uint64_t head = ownedMemory.mirrored ? count : std::min<std::uint64_t>(count, capacity - index);
396 const std::ptrdiff_t sampleStride = layout == AcqLayout::Interleaved
397 ? static_cast<std::ptrdiff_t>(frameBytes)
398 : static_cast<std::ptrdiff_t>(bytesPerSample);
399 const std::ptrdiff_t channelStride =
400 layout == AcqLayout::Interleaved ? static_cast<std::ptrdiff_t>(bytesPerSample) : rowStride;
401 parts[0] = {sampleAddress(first), static_cast<std::size_t>(head), sampleStride, channelStride};
402 if (head == count) {
403 return 1;
404 }
405 parts[1] = {base, static_cast<std::size_t>(count - head), sampleStride, channelStride};
406 return 2;
407 }

495template<typename Ready>
496bool waitFor(Ready&& ready, AcqWait wait, std::chrono::nanoseconds spinFor,
497 std::chrono::steady_clock::time_point deadline, std::atomic<bool>& flag, std::binary_semaphore& semaphore,
498 std::atomic<std::uint32_t>* counter)
499{
500 using Clock = std::chrono::steady_clock;
501 if (wait != AcqWait::Block) {
502 const auto spinEnd = wait == AcqWait::Spin ? deadline : std::min(deadline, Clock::now() + spinFor);
503 for (std::uint32_t i = 0;; ++i) {
504 if (ready()) {
505 return true;
506 }
507 if ((i & 63U) == 0 && Clock::now() >= spinEnd) {
508 break;
509 }
510 cpuRelax();
511 }
512 if (wait == AcqWait::Spin) {
513 return ready();
514 }
515 }
516 while (true) {
517 flag.store(true, std::memory_order_seq_cst);
518 if (counter != nullptr) {
519 counter->fetch_add(1, std::memory_order_seq_cst);
520 }
521 bool woken = false;
522 const bool readyNow = ready();
523 if (!readyNow && Clock::now() < deadline) {
524 woken = semaphore.try_acquire_until(deadline);
525 }
526 if (counter != nullptr) {
527 counter->fetch_sub(1, std::memory_order_seq_cst);
528 }
529 if (!woken && !flag.exchange(false, std::memory_order_acq_rel)) {
530 semaphore.acquire(); // a notifier took the flag; consume its token
531 }
532 if (readyNow || ready()) {
533 return true;
534 }
535 if (Clock::now() >= deadline) {
536 return false;
537 }
538 }
539}
