67void TraceBuffer::reload_channels(bool clear_data)
68{
69 SpecProvider provider;
70 {
71 std::shared_lock lk(m_lock);
72 provider = m_specProvider;
73 }
74 // The provider runs outside the lock: it may read host state that the host guards itself
75 std::vector<ChannelSpec> specs = provider ? provider() : std::vector<ChannelSpec>{};
76 {
77 std::unique_lock lk(m_lock);
78 if (provider)
79 m_specs = std::move(specs);
80 m_data.resize(m_specs.size());
81 if (clear_data) {
82 for (std::vector<float>& channel : m_data)
83 channel.clear();
84 m_firstSample = 0;
85 }
86 resolve();
87 }
88 notify(Change::Channels);
89 if (!clear_data)
90 notify(Change::Data);
91}

103void TraceBuffer::set_data(const Eigen::MatrixXd& data, int first_sample)
104{
105 if (data.rows() == 0 || data.cols() == 0)
106 return;
107 {
108 std::unique_lock lk(m_lock);
109 const auto rows = static_cast<std::size_t>(data.rows());
110 const auto cols = static_cast<std::size_t>(data.cols());
111 m_data.resize(rows);
112 for (std::size_t ch = 0; ch < rows; ++ch) {
113 m_data[ch].resize(cols);
114 for (std::size_t s = 0; s < cols; ++s)
115 m_data[ch][s] = static_cast<float>(data(static_cast<Eigen::Index>(ch), static_cast<Eigen::Index>(s)));
116 }
117 m_firstSample = first_sample;
118 }
119 notify(Change::Data);
120}

124void TraceBuffer::append_data(const Eigen::MatrixXd& data)
125{
126 if (data.rows() == 0 || data.cols() == 0)
127 return;
128 {
129 std::unique_lock lk(m_lock);
130 const auto rows = static_cast<std::size_t>(data.rows());
131 const auto added = static_cast<std::size_t>(data.cols());
132 if (m_data.size() != rows)
133 m_data.resize(rows);
134
135 for (std::size_t ch = 0; ch < rows; ++ch) {
136 std::vector<float>& samples = m_data[ch];
137 const std::size_t old_size = samples.size();
138 samples.resize(old_size + added);
139 for (std::size_t s = 0; s < added; ++s)
140 samples[old_size + s] = static_cast<float>(data(static_cast<Eigen::Index>(ch), static_cast<Eigen::Index>(s)));
141
142 if (m_capacity > 0 && std::ssize(samples) > m_capacity) {
143 const std::ptrdiff_t drop = std::ssize(samples) - m_capacity;
144 samples.erase(samples.begin(), samples.begin() + drop);
145 if (ch == 0)
146 m_firstSample += static_cast<int>(drop);
147 }
148 }
149 }
150 notify(Change::Data);
151}

280float TraceBuffer::rms(int index, int first, int last) const
281{
282 std::shared_lock lk(m_lock);
283 if (index < 0 || index >= std::ssize(m_data))
284 return 0.f;
285 const std::vector<float>& samples = m_data[static_cast<std::size_t>(index)];
286 const int size = static_cast<int>(samples.size());
287 int from = std::clamp(first - m_firstSample, 0, size);
288 const int to = std::clamp(last - m_firstSample, 0, size);
289 if (to <= from)
290 return 0.f;
291 // At most the last 1000 samples, so that the render thread stays fast
292 constexpr int kMax = 1000;
293 from = std::max(from, to - kMax);
294 double sum = 0.0;
295 for (int i = from; i < to; ++i) {
296 const double value = samples[static_cast<std::size_t>(i)];
297 sum += value * value;
298 }
299 return static_cast<float>(std::sqrt(sum / (to - from)));
300}

352TraceVertices TraceBuffer::vertices(int index, int first, int last, int pixel_width) const
353{
354 std::shared_lock lk(m_lock);
355 TraceVertices result{first, {}};
356 if (index < 0 || index >= std::ssize(m_data) || pixel_width <= 0 || first >= last)
357 return result;
358
359 const std::vector<float>& samples = m_data[static_cast<std::size_t>(index)];
360 const int size = static_cast<int>(samples.size());
361 const int from = std::clamp(first - m_firstSample, 0, size);
362 const int to = std::clamp(last - m_firstSample, 0, size);
363
364 // A window starting before the stored samples starts at the first stored one
365 result.first_sample = m_firstSample + from;
366 if (to <= from)
367 return result;
368
369 result.xy = decimate_trace(std::span<const float>(samples).subspan(static_cast<std::size_t>(from), static_cast<std::size_t>(to - from)),
370 pixel_width, m_detrend);
371 return result;
372}

376void TraceBuffer::resolve()
377{
378 const auto count = static_cast<std::size_t>(stored_count());
379 m_channels.assign(count, Channel{});
380 m_virtual.assign(count, false);
381
382 const auto scale_of = [this](const ChannelSpec& spec) {
383 for (const int key : spec.scale_keys) {
384 if (const auto it = m_scales.find(key); it != m_scales.end())
385 return it->second;
386 }
387 return spec.default_scale > 0.f ? spec.default_scale : 1.f;
388 };
389
390 int virtual_number = 0;
391 for (std::size_t ch = 0; ch < count; ++ch) {
392 Channel& channel = m_channels[ch];
393 if (ch >= m_specs.size()) {
394 channel = {std::format("CH {}", ch + 1), "MISC", 1.f, m_signalColor, false};
395 continue;
396 }
397 const ChannelSpec& spec = m_specs[ch];
398 channel = {spec.name, spec.type_label, scale_of(spec), spec.color.value_or(m_signalColor), spec.bad};
399 if (spec.is_virtual) {
400 ++virtual_number;
401 if (channel.name.empty())
402 channel.name = std::format("Virtual {}", virtual_number);
403 if (channel.type_label.empty())
404 channel.type_label = "MISC";
405 m_virtual[ch] = true;
406 }
407 }
408}
