Expected<FiffEvents> FiffEvents::read_fif(const std::filesystem::path& path) { auto stream = FiffStream::open(path); if (!stream) { return std::unexpected(stream.error()); } const std::vector<std::shared_ptr<FiffDirNode>> blocks = stream->dirtree()->dir_tree_find(FIFFB_MNE_EVENTS); if (blocks.empty()) { return fail(ErrorCode::NotFound, "{} holds no events", mne::utf8_from_path(path)); } for (const FiffDirEntry& entry : blocks.front()->dir) { if (entry.kind != FIFF_MNE_EVENT_LIST) { continue; } const std::optional<FiffTag> tag = stream->read_tag(entry.pos); if (!tag || (tag->type != FIFFT_INT && tag->type != FIFFT_UINT)) { break; } Stored row by row (sample, before, after); unsigned lists are read as their bit patterns const Eigen::Index rows = tag->size() / 12; return FiffEvents(Eigen::Map<const Eigen::Matrix<int, Eigen::Dynamic, 3, Eigen::RowMajor>>(reinterpret_cast<const int*>(tag->data()), rows, 3)); } return fail(ErrorCode::NotFound, "{} holds no event list", mne::utf8_from_path(path)); }

Expected<FiffEvents> FiffEvents::read_ascii(const std::filesystem::path& path) { std::ifstream file(path); if (!file) { return fail(ErrorCode::Io, "cannot read {}", mne::utf8_from_path(path)); } std::vector<Eigen::RowVector3i> rows; std::string line; for (int number = 1; std::getline(file, line); ++number) { "sample time before after" as written by write_ascii, or just "sample before after" std::vector<std::string> fields; std::istringstream words(line); for (std::string word; words >> word;) fields.push_back(word); if (fields.empty() || fields.front().starts_with('') || fields.front().starts_with('#')) continue; const std::size_t offset = fields.size() >= 4 ? 1 : 0; const auto field = [&fields](std::size_t i) { return i < fields.size() ? mne::to_int(fields[i]) : std::optional<int>(0); }; const std::optional<int> sample = field(0), before = field(1 + offset), after = field(2 + offset); if (!sample || !before || !after) { return fail(ErrorCode::Format, "{} line {} is not 'sample [time] before after': {}", mne::utf8_from_path(path), number, line); } rows.emplace_back(*sample, *before, *after); } FiffEvents out; out.events.resize(static_cast<Eigen::Index>(rows.size()), 3); for (std::size_t i = 0; i < rows.size(); ++i) { out.events.row(static_cast<Eigen::Index>(i)) = rows[i]; } return out; }

Expected<void> FiffEvents::write_fif(const std::filesystem::path& path) const { if (events.rows() == 0 || events.cols() < 3) { return fail(ErrorCode::Format, "no events to write to {}", mne::utf8_from_path(path)); } auto stream = FiffStream::start_file(path); if (!stream) { return fail(ErrorCode::Io, "cannot write {}", mne::utf8_from_path(path)); } FIFF stores the list row by row (sample, before, after). const Eigen::Matrix<int, Eigen::Dynamic, 3, Eigen::RowMajor> rows = events.leftCols(3); stream->start_block(FIFFB_MNE_EVENTS); stream->write_int(FIFF_MNE_EVENT_LIST, rows.data(), rows.rows() * 3); stream->end_block(FIFFB_MNE_EVENTS); stream->end_file(); stream->close(); return {}; }

Expected<void> FiffEvents::write_ascii(const std::filesystem::path& path, float sfreq) const { std::ofstream out(path, std::ios::binary); if (!out) { return fail(ErrorCode::Io, "cannot write {}", mne::utf8_from_path(path)); } for (Eigen::Index k = 0; k < events.rows(); ++k) { const int sample = events(k, 0); const int before = events.cols() > 1 ? events(k, 1) : 0; const int after = events.cols() > 2 ? events(k, 2) : 0; const float time = sfreq > 0.0f ? static_cast<float>(sample) / sfreq : 0.0f; char digits[64]; const auto result = std::to_chars(digits, digits + sizeof(digits), static_cast<double>(time), std::chars_format::fixed, 3); out << padded(std::to_string(sample), 6) << ' ' << padded(std::string(digits, result.ptr), -10) << ' ' << padded(std::to_string(before), 3) << ' ' << padded(std::to_string(after), 3) << '
'; } if (!out.flush()) { return fail(ErrorCode::Io, "cannot write {}", mne::utf8_from_path(path)); } return {}; }

Expected<FiffEvents> FiffEvents::find_events(const FiffRawData& raw, std::string_view stim_channel, unsigned int mask, bool leading_edge) { const int index = channel_index(raw.info.ch_names, stim_channel); if (index < 0) { return fail(ErrorCode::NotFound, "no stimulus channel {}", stim_channel); }

MatrixXd data; MatrixXd times; if (!raw.read_raw_segment(data, times, raw.first_samp, raw.last_samp, RowVectorXi::Constant(1, index))) { return fail(ErrorCode::Io, "cannot read stimulus channel {}", stim_channel); }

std::vector<Eigen::RowVector3i> rows; const auto value = [&data, mask](Eigen::Index s) { return static_cast<int>(static_cast<unsigned int>(static_cast<int>(data(0, s))) & mask); }; int previous = value(0); for (Eigen::Index s = 1; s < data.cols(); ++s) { const int current = value(s); if (current != previous && (!leading_edge || (previous == 0 && current != 0))) { rows.emplace_back(static_cast<int>(raw.first_samp + s), previous, current); } previous = current; } FiffEvents out; out.events.resize(static_cast<Eigen::Index>(rows.size()), 3); for (std::size_t i = 0; i < rows.size(); ++i) { out.events.row(static_cast<Eigen::Index>(i)) = rows[i]; } return out; }

bool FiffEvents::match_event(const AverageCategory& cat, const MatrixXi& events, int event_idx) { if (event_idx < 0 || event_idx >= events.rows()) return false;

int evFrom = events(event_idx, 1); int evTo = events(event_idx, 2);

Check if any of the category's event codes match bool match = false; for (const auto event : cat.events) { if ((evFrom & ~cat.ignore) == 0 && (evTo & ~cat.ignore) == event) { match = true; break; } } if (!match) return false;

Check previous event constraint if (cat.prevEvent != 0) { bool found = false; for (int j = event_idx - 1; j >= 0; –j) { if ((events(j, 1) & ~cat.prevIgnore) == 0) { found = true; match = match && ((events(j, 2) & ~cat.prevIgnore) == cat.prevEvent); break; } } if (!found) match = false; }

Check next event constraint if (cat.nextEvent != 0) { bool found = false; for (int j = event_idx + 1; j < events.rows(); ++j) { if ((events(j, 1) & ~cat.nextIgnore) == 0) { found = true; match = match && ((events(j, 2) & ~cat.nextIgnore) == cat.nextEvent); break; } } if (!found) match = false; }

return match; }
