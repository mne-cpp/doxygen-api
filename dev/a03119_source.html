63StringList tokenize(std::istream& in)
64{
65 StringList tokens;
66 const auto isSpace = [](char c) {
67 return std::isspace(static_cast<unsigned char>(c)) != 0;
68 };
69 std::string line;
70 while (std::getline(in, line)) {
71 if (mne::trimmed(line).starts_with('#')) {
72 continue;
73 }
74 std::size_t k = 0;
75 while (k < line.size()) {
76 if (isSpace(line[k])) {
77 ++k;
78 continue;
79 }
80 if (line[k] == '"') {
81 const std::size_t end = std::min(line.find('"', k + 1), line.size());
82 tokens.emplace_back(line, k + 1, end - k - 1);
83 k = end + 1;
84 continue;
85 }
86 std::size_t end = k + 1;
87 while (end < line.size() && !isSpace(line[end])) {
88 ++end;
89 }
90 tokens.emplace_back(line, k, end - k);
91 k = end;
92 }
93 }
94 return tokens;
95}

99std::unique_ptr<MNEDeriv> makeDeriv(const std::vector<MNEDerivSet::Definition>& definitions, std::string filename, std::string shortname)
100{
101 StringList inputs;
102 StringList outputs;
103 std::vector<Eigen::Triplet<float>> triplets;
104 for (const MNEDerivSet::Definition& definition : definitions) {
105 std::vector<std::pair<int, double>> row;
106 for (auto it = definition.second.cbegin(); it != definition.second.cend(); ++it) {
107 if (std::fabs(it->second) <= kZero) {
108 continue;
109 }
110 if (!mne::has_channel(inputs, it->first)) {
111 inputs.push_back(it->first);
112 }
113 row.push_back({static_cast<int>(mne::channel_index(inputs, it->first)), it->second});
114 }
115 if (row.empty()) {
116 mne::Log::info("MNEDerivSet - Empty derivation \"{}\" omitted", definition.first);
117 continue;
118 }
119 for (const auto& [column, weight] : row) {
120 triplets.emplace_back(static_cast<int>(outputs.size()), column, static_cast<float>(weight));
121 }
122 outputs.push_back(definition.first);
123 }
124 Eigen::SparseMatrix<float> data(outputs.size(), inputs.size());
125 data.setFromTriplets(triplets.begin(), triplets.end());
126 auto deriv = std::make_unique<MNEDeriv>();
127 deriv->filename = std::move(filename);
128 deriv->shortname = std::move(shortname);
129 deriv->deriv_data = std::make_unique<MNESparseNamedMatrix>();
130 deriv->deriv_data->rowlist = outputs;
131 deriv->deriv_data->collist = inputs;
132 deriv->deriv_data->data = FiffSparseMatrix(std::move(data));
133 return deriv;
134}

142std::optional<MNEDerivSet> MNEDerivSet::read(const std::filesystem::path& path)
143{
144 FiffStream stream(path);
145 if (!stream.open()) {
146 return std::nullopt;
147 }
148 MNEDerivSet set;
149 for (const std::shared_ptr<FiffDirNode>& block : stream.dirtree()->dir_tree_find(FIFFB_MNE_DERIVATIONS)) {
150 for (const std::shared_ptr<FiffDirNode>& matrix : block->dir_tree_find(FIFFB_MNE_NAMED_MATRIX)) {
151 auto data = MNESparseNamedMatrix::read(stream, matrix, FIFF_MNE_DERIVATION_DATA);
152 if (!data) {
153 mne::Log::warning("MNEDerivSet::read - Bad derivation data in {}", mne::utf8_from_path(path));
154 return std::nullopt;
155 }
156 auto deriv = std::make_unique<MNEDeriv>();
157 deriv->filename = utf8_from_path(path);
158 deriv->deriv_data = std::move(data);
159 set.derivs.push_back(std::move(deriv));
160 }
161 }
162 return set;
163}

167std::optional<MNEDerivSet> MNEDerivSet::readText(const std::filesystem::path& path)
168{
169 std::ifstream file(path);
170 if (!file) {
171 mne::Log::warning("MNEDerivSet::readText - Cannot open {}", mne::utf8_from_path(path));
172 return std::nullopt;
173 }
174 const StringList tokens = tokenize(file);
175
176 // A channel token right after another channel token (or first) names a new derivation;
177 // otherwise it is an input whose weight comes from the preceding =, +, - or number.
178 enum class Kind
179 {
180 Channel,
181 Equal,
182 Plus,
183 Minus,
184 Mult,
185 Number
186 };
187 std::vector<Definition> definitions;
188 Kind prev = Kind::Channel;
189 bool havePrev = false;
190 double number = 0.0;
191 for (const std::string& token : tokens) {
192 const std::optional<double> parsed = mne::to_double(token);
193 const bool isNumber = parsed.has_value();
194 const double value = parsed.value_or(0.0);
195 const Kind kind = token == "=" ? Kind::Equal : token == "+" ? Kind::Plus
196 : token == "-" ? Kind::Minus
197 : token == "*" ? Kind::Mult
198 : isNumber ? Kind::Number
199 : Kind::Channel;
200 if (kind == Kind::Mult) {
201 continue;
202 }
203 if (kind == Kind::Channel && (!havePrev || prev == Kind::Channel)) {
204 definitions.push_back({token, {}});
205 } else if (kind == Kind::Equal) {
206 if (definitions.empty() || !definitions.back().second.empty()) {
207 mne::Log::warning("MNEDerivSet::readText - Misplaced equal sign in {}", mne::utf8_from_path(path));
208 return std::nullopt;
209 }
210 } else if (kind == Kind::Channel) {
211 const double weight = prev == Kind::Minus ? -1.0 : prev == Kind::Number ? number
212 : 1.0;
213 if (definitions.empty()) {
214 mne::Log::warning("MNEDerivSet::readText - Misplaced channel name in {}", mne::utf8_from_path(path));
215 return std::nullopt;
216 }
217 definitions.back().second[token] += weight;
218 } else if (kind == Kind::Number) {
219 number = prev == Kind::Minus ? -value : value;
220 }
221 prev = kind;
222 havePrev = true;
223 }
224 if (definitions.empty()) {
225 mne::Log::warning("MNEDerivSet::readText - No derivations in {}", mne::utf8_from_path(path));
226 return std::nullopt;
227 }
228 MNEDerivSet set;
229 set.derivs.push_back(makeDeriv(definitions, mne::utf8_from_path(path), {}));
230 if (set.derivs.front()->deriv_data->nrow() == 0) {
231 return std::nullopt;
232 }
233 return set;
234}

247bool MNEDerivSet::write(const std::filesystem::path& path) const
248{
249 if (derivs.empty()) {
250 mne::Log::warning("MNEDerivSet::write - No derivations to write");
251 return false;
252 }
253 auto stream = FiffStream::start_file(path);
254 if (!stream) {
255 return false;
256 }
257 stream->start_block(FIFFB_MNE_DERIVATIONS);
258 for (const auto& deriv : derivs) {
259 deriv->deriv_data->write(*stream, FIFF_MNE_DERIVATION_DATA);
260 }
261 stream->end_block(FIFFB_MNE_DERIVATIONS);
262 stream->end_file();
263 return true;
264}

268std::vector<MNEDerivSet::Definition> MNEDerivSet::definitions() const
269{
270 std::vector<Definition> result;
271 for (const auto& deriv : derivs) {
272 const MNESparseNamedMatrix& m = *deriv->deriv_data;
273 const Eigen::SparseMatrix<float, Eigen::RowMajor> rows = m.data.eigen();
274 for (int j = 0; j < m.nrow(); ++j) {
275 Definition definition{m.rowlist[j], {}};
276 for (Eigen::SparseMatrix<float, Eigen::RowMajor>::InnerIterator it(rows, j); it; ++it) {
277 definition.second.emplace(m.collist[static_cast<std::size_t>(it.col())], it.value());
278 }
279 result.push_back(std::move(definition));
280 }
281 }
282 return result;
283}

296std::unique_ptr<MNEDeriv> MNEDerivSet::match(const StringList& chNames) const
297{
298 std::vector<Eigen::Triplet<float>> triplets;
299 StringList outputs;
300 int ntot = 0;
301 for (const Definition& definition : definitions()) {
302 ++ntot;
303 bool available = true;
304 for (auto it = definition.second.cbegin(); it != definition.second.cend() && available; ++it) {
305 available = std::fabs(it->second) <= kZero || mne::has_channel(chNames, it->first);
306 }
307 if (!available) {
308 continue;
309 }
310 for (auto it = definition.second.cbegin(); it != definition.second.cend(); ++it) {
311 if (std::fabs(it->second) > kZero) {
312 triplets.emplace_back(static_cast<int>(outputs.size()), static_cast<int>(mne::channel_index(chNames, it->first)), static_cast<float>(it->second));
313 }
314 }
315 outputs.push_back(definition.first);
316 }
317 if (outputs.empty()) {
318 return nullptr;
319 }
320 mne::Log::info("MNEDerivSet::match - {} of {} derivations were matched.", outputs.size(), ntot);
321 Eigen::SparseMatrix<float> data(outputs.size(), chNames.size());
322 data.setFromTriplets(triplets.begin(), triplets.end());
323
324 auto matched = std::make_unique<MNEDeriv>();
325 matched->shortname = "Matched derivations";
326 // in_use counts, per recorded channel, the derived channels it enters (MNE-C convention).
327 matched->in_use = Eigen::VectorXi::Zero(chNames.size());
328 for (int k = 0; k < data.outerSize(); ++k) {
329 for (Eigen::SparseMatrix<float>::InnerIterator it(data, k); it; ++it) {
330 ++matched->in_use[it.col()];
331 }
332 }
333 matched->deriv_data = std::make_unique<MNESparseNamedMatrix>();
334 matched->deriv_data->rowlist = outputs;
335 matched->deriv_data->collist = chNames;
336 matched->deriv_data->data = FiffSparseMatrix(std::move(data));
337 return matched;
338}
