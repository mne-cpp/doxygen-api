89AcqExpected<std::vector<int>> parseStations(std::string_view text)
90{
91 std::array<bool, 5> chosen{};
92 std::size_t begin = 0;
93 while (begin <= text.size()) {
94 const std::size_t end = std::min(text.find(',', begin), text.size());
95 const std::string_view token = text.substr(begin, end - begin);
96 int station = 0;
97 const auto [ptr, ec] = std::from_chars(token.data(), token.data() + token.size(), station);
98 if (ec != std::errc() || ptr != token.data() + token.size() || station < 1 || station > 4 || chosen[static_cast<std::size_t>(station)]) {
99 return invalid(std::format("stations={}: distinct stations 1..4 such as 1,2", text));
100 }
101 chosen[static_cast<std::size_t>(station)] = true;
102 begin = end + 1;
103 }
104 std::vector<int> stations;
105 for (int s = 1; s <= 4; ++s) {
106 if (chosen[static_cast<std::size_t>(s)]) {
107 stations.push_back(s);
108 }
109 }
110 return stations;
111}

113AcqExpected<std::array<double, 3>> parseHemisphere(std::string_view text)
114{
115 std::string spaced(text);
116 std::replace(spaced.begin(), spaced.end(), ',', ' ');
117 std::istringstream stream(spaced);
118 stream.imbue(std::locale::classic());
119 std::array<double, 3> h{};
120 if (!(stream >> h[0] >> h[1] >> h[2]) || !(stream >> std::ws).eof() || (h[0] == 0.0 && h[1] == 0.0 && h[2] == 0.0)) {
121 return invalid(std::format("hemisphere={}: a direction x,y,z such as 0,0,1", text));
122 }
123 return h;
124}

126AcqExpected<Config> parseConfig(const AcqUri& uri)
127{
128 Config c;
129 c.port = uri.target;
130 auto stations = parseStations(uri.param("stations").value_or("1,2"));
131 const auto simulate = paramBool(uri, "simulate", false);
132 const auto baud = paramInt(uri, "baud", 115200, 1200, 230400);
133 const auto rate = paramDouble(uri, "rate", 120.0, 1.0, 1000.0);
134 const auto pause = paramInt(uri, "pause", 300, 10, 10000);
135 const auto seed = paramInt(uri, "seed", 0, 0, INT64_MAX);
136 const auto realtime = paramBool(uri, "realtime", true);
137 const auto duration = paramDouble(uri, "duration", 0.0, 0.0, 1e7);
138 const auto button = paramDouble(uri, "button", 0.0, 0.0, 1e6);
139 auto thread = paramThreadPolicy(uri);
140 for (const AcqError& error : {stations.error(), simulate.error(), baud.error(), rate.error(), pause.error(), seed.error(), realtime.error(),
141 duration.error(), button.error(), thread.error()}) {
142 if (!error.ok()) {
143 return error;
144 }
145 }
146 const std::string_view units = uri.param("units").value_or("in");
147 if (units != "in" && units != "cm") {
148 return invalid(std::format("units={}: in or cm", units));
149 }
150 if (const auto text = uri.param("hemisphere")) {
151 auto hemisphere = parseHemisphere(*text);
152 if (!hemisphere) {
153 return hemisphere.error();
154 }
155 c.hemisphere = *hemisphere;
156 }
157 c.simulate = *simulate;
158 if (c.simulate && !c.port.empty()) {
159 return invalid("target '" + c.port + "': the simulator takes no serial port");
160 }
161 c.baud = static_cast<int>(*baud);
162 c.units = units == "in" ? FastrakUnits::Inches : FastrakUnits::Centimetres;
163 c.stations = std::move(*stations);
164 c.rate = *rate;
165 c.pause = std::chrono::milliseconds(*pause);
166 c.simulation = {c.stations, c.units, c.rate, static_cast<std::uint64_t>(*seed), *realtime, *duration, *button};
167 c.thread = std::move(*thread);
168 return c;
169}

205AcqExpected<std::unique_ptr<PolhemusTransport>> openSerial(const Config& config)
206{
207 std::string name = config.port;
208 if (name.empty()) {
209 const std::vector<std::string> ports = list_serial_ports();
210 if (ports.empty()) {
211 return AcqError(AcqErrorCode::DeviceNotFound, "no serial port; give the digitizer's port as target or use simulate=1");
212 }
213 name = ports.front();
214 }
215 auto port = SerialPort::open(name, config.baud);
216 if (!port) {
217 return transport_error(port.error(), AcqErrorCode::DeviceIo, std::format("cannot open serial port {} at {} baud", name, config.baud));
218 }
219 return std::unique_ptr<PolhemusTransport>(std::make_unique<PolhemusSerial>(std::move(*port), std::move(name)));
220}

223std::string setupCommands(const Config& config)
224{
225 std::string commands;
226 for (int s = 1; s <= 4; ++s) {
227 const bool used = std::find(config.stations.begin(), config.stations.end(), s) != config.stations.end();
228 commands += std::format("l{},{}\r", s, used ? 1 : 0);
229 }
230 for (const int s : config.stations) {
231 commands += std::format("O{},2,4,1\r", s);
232 if (config.hemisphere) {
233 commands += std::format("H{},{},{},{}\r", s, (*config.hemisphere)[0], (*config.hemisphere)[1], (*config.hemisphere)[2]);
234 }
235 }
236 return commands;

244 auto config = parseConfig(uri);

253 if (!serial) {
254 return serial.error();

259 return e;
260 }
261
262 AcqStreamInfo info;
263 info.name = "poses";
264 info.sourceId = transport->description();
265 info.format = AcqSampleFormat::Float32;
266 info.layout = AcqLayout::Interleaved;
267 info.sfreq = config->rate / static_cast<double>(config->stations.size());
268 for (const int s : config->stations) {
269 for (const char* axis : {"X", "Y", "Z"}) {
270 AcqChannelInfo& channel = info.channels.emplace_back();
271 channel.name = std::format("{} {}", s, axis);
272 channel.kind = AcqChannelKind::Position;
273 channel.unit = AcqUnit::Meter;
274 }
275 for (const char* component : {"QW", "QX", "QY", "QZ"}) {
276 AcqChannelInfo& channel = info.channels.emplace_back();
277 channel.name = std::format("{} {}", s, component);
278 channel.kind = AcqChannelKind::Orientation;
279 channel.unit = AcqUnit::None;
280 }
281 }
282
283 std::unique_ptr<PolhemusBackend> backend(new PolhemusBackend(std::move(*config), std::move(transport), uri.toString()));
284 AcqStreamOptions options;
285 options.capacitySamples = 4096;
286 auto writer = backend->addInputStream(std::move(info), options);
287 if (!writer) {
288 return writer.error();

308 {
309 if (AcqError e = m_transport->write("C\r"); !e.ok()) {
310 return e;
311 }
312 const auto now = std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now().time_since_epoch()).count();
313 AcqClockModel clock;
314 clock.sfreq = m_config.rate / static_cast<double>(m_config.stations.size());
315 clock.anchorSample = m_writer->writePosition();
316 clock.anchorHostNs = now;
317 clock.utcOffsetNs =

319 clock.source = AcqClockSource::Nominal;
320 m_writer->setClock(clock);
321 m_stop.store(false, std::memory_order_relaxed);

323 if (!thread) {

350 bool commitFrame()
351 {
352 AcqExpected<AcqWriteView> view = AcqError(AcqErrorCode::Timeout);
353 while (!view && view.error() == AcqErrorCode::Timeout) {
354 if (m_stop.load(std::memory_order_acquire)) {
355 return false;
356 }
357 view = m_writer->beginWrite(1, kPoll); // a gating reader may not have freed the space yet
358 }
359 if (!view) {
360 reportError(view.error());
361 return false;
362 }
363 std::memcpy(view->part(0).at(0, 0), m_frame.data(), m_frame.size() * sizeof(float));
364 m_writer->commit(1);
365 return true;
366 }
367
369 bool decodeLines()
370 {
371 std::size_t begin = 0;
372 for (std::size_t end = m_line.find('\n'); end != std::string::npos; end = m_line.find('\n', begin)) {
373 std::string_view line(m_line.data() + begin, end - begin);
374 begin = end + 1;
375 if (!line.empty() && line.back() == '\r') {
376 line.remove_suffix(1);
377 }
378 const auto record = parse_fastrak_record(line, m_config.units);
379 if (!record) {
380 continue; // blank lines and command echoes
381 }
382 const auto station = std::find(m_config.stations.begin(), m_config.stations.end(), record->station);
383 if (station == m_config.stations.end()) {
384 continue;
385 }
386 float* values = m_frame.data() + static_cast<std::size_t>(station - m_config.stations.begin()) * kValuesPerStation;
387 std::copy(record->position.begin(), record->position.end(), values);
388 std::copy(record->orientation.begin(), record->orientation.end(), values + 3);
389 if (station + 1 == m_config.stations.end() && !commitFrame()) {
390 return false;
391 }
392 }
393 m_line.erase(0, begin);
394 return true;
395 }

397 void run()
398 {
399 std::array<char, 1024> chunk{};
400 auto last = std::chrono::steady_clock::now();
401 bool streaming = false;
402 while (!m_stop.load(std::memory_order_acquire)) {
403 auto got = m_transport->read(chunk, std::min(kWait, m_config.pause));
404 if (!got) {
405 if (got.error() == AcqErrorCode::EndOfStream) {
406 m_writer->finish();
407 return;
408 }
409 if (got.error() != AcqErrorCode::Timeout) {
410 reportError(got.error());
411 return;
412 }
413 if (streaming && std::chrono::steady_clock::now() - last >= m_config.pause) {
414 // The stylus button paused the output: a frame of its own repeats the pose it was pressed at
415 streaming = false;
416 if (AcqError e = m_writer->pushEvent({m_writer->writePosition(), 0, kButtonCode}); !e.ok()) {
417 reportError(e);
418 return;
419 }
420 if (!commitFrame()) {
421 return;
422 }
423 if (AcqError e = m_transport->write("C\r"); !e.ok()) {
424 reportError(e);
425 return;
426 }
427 }
428 continue;
429 }
430 streaming = true;
431 last = std::chrono::steady_clock::now();
432 m_line.append(chunk.data(), *got);
433 if (!decodeLines()) {
434 return;
435 }
436 }
437 }
