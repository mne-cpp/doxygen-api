77std::optional<MNELayout> MNELayout::read(const std::filesystem::path& path)
78{
79 std::ifstream file(path);
80 if (!file) {
81 Log::warning("MNELayout::read - Cannot open {}", utf8_from_path(path));
82 return std::nullopt;
83 }
84
85 MNELayout layout;
86 layout.fileName = utf8_from_path(path);
87 bool haveExtent = false;
88 std::string raw;
89 while (std::getline(file, raw)) {
90 const std::string line(trimmed(raw));
91 if (line.empty() || line.starts_with('#')) {
92 continue;
93 }
94 std::istringstream fields(line);
95 if (!haveExtent) {
96 double v[4];
97 if (!(fields >> v[0] >> v[1] >> v[2] >> v[3])) {
98 Log::warning("MNELayout::read - Illegal extent in {}", utf8_from_path(path));
99 return std::nullopt;
100 }
101 layout.m_extent = Eigen::AlignedBox2d(Eigen::Vector2d(v[0], v[2]), Eigen::Vector2d(v[1], v[3]));
102 haveExtent = true;
103 continue;
104 }
105 MNELayoutPort port;
106 double x0 = 0.0;
107 double y0 = 0.0;
108 double w = 0.0;
109 double h = 0.0;
110 if (!(fields >> port.portno >> x0 >> y0 >> w >> h)) {
111 Log::warning("MNELayout::read - Bad layout item in {}: {}", utf8_from_path(path), line);
112 return std::nullopt;
113 }
114 port.xmin = static_cast<float>(x0);
115 port.xmax = static_cast<float>(x0 + std::fabs(w));
116 port.ymin = static_cast<float>(y0);
117 port.ymax = static_cast<float>(y0 + std::fabs(h));
118 port.invert = h < 0;
119 // The names are the rest of the line and may contain spaces ("MEG 0113:MEG 0112").
120 std::string rest;
121 std::getline(fields, rest);
122 port.label = trimmed(rest);
123 for (const std::string& name : split(port.label, ':', true)) {
124 port.names.push_back(MNELayoutPort::cleanName(name));
125 }
126 layout.ports.push_back(std::move(port));
127 }
128 if (!haveExtent) {
129 Log::warning("MNELayout::read - No extent in {}", utf8_from_path(path));
130 return std::nullopt;
131 }
132 layout.m_visible = layout.m_extent;
133 return layout;
134}

151int MNELayout::matchPorts(std::string_view channel)
152{
153 int nmatch = 0;
154 for (MNELayoutPort& port : ports) {
155 port.matched = !channel.empty() && port.contains(channel) && isVisible(port);
156 nmatch += port.matched ? 1 : 0;
157 }
158 return nmatch;

163int MNELayout::matchPorts(const StringList& channels)
164{
165 m_matches = Eigen::MatrixXi::Zero(static_cast<Eigen::Index>(ports.size()), static_cast<Eigen::Index>(channels.size()));
166 int nmatch = 0;
167 for (std::size_t p = 0; p < ports.size(); ++p) {
168 MNELayoutPort& port = ports[p];
169 port.matched = false;
170 if (!isVisible(port)) {
171 continue;
172 }
173 for (std::size_t c = 0; c < channels.size(); ++c) {
174 if (port.contains(channels[c])) {
175 m_matches(static_cast<Eigen::Index>(p), static_cast<Eigen::Index>(c)) = 1;
176 port.matched = true;
177 ++nmatch;
178 }
179 }
180 }
181 return nmatch;
182}

186int MNELayout::portAt(const Eigen::Vector2d& point) const
187{
188 const auto it = std::find_if(ports.cbegin(), ports.cend(), [&point](const MNELayoutPort& port) {
189 return point.x() > port.xmin && point.x() < port.xmax && point.y() > port.ymin && point.y() < port.ymax;
190 });
191 return it == ports.cend() ? -1 : static_cast<int>(it - ports.cbegin());
192}

196int MNELayout::confine(const Eigen::AlignedBox2d& area)
197{
198 Eigen::AlignedBox2d covered;
199 int nmatch = 0;
200 for (const MNELayoutPort& port : ports) {
201 if (isInside(port, area)) {
202 covered.extend(port.rect());
203 ++nmatch;
204 }
205 }
206 if (nmatch > 0) {
207 const double margin = 0.01 * covered.sizes().x();
208 m_visible = Eigen::AlignedBox2d(covered.min().array() - margin, covered.max().array() + margin);
209 }
210 return nmatch;
211}
