214 //=========================================================================================================
215 // Optical path calibration — determine the microscope optical axis
216 // relative to the tracker sensor.
217 //
218 // Workflow:
219 // 1. Move the OPMI to a position, aim at a point on the head surface,
220 // mark the focus point with the stylus pen → captureOpticalCalibSample().
221 // 2. Repeat at 2–3 different positions / focal distances.
222 // 3. Call solveOpticalCalibration() to fit a 3D line through the focus
223 // points in the tracker's local frame, yielding the optical axis
224 // direction and the optical center offset from the tracker.
225 //=========================================================================================================
226

229 QVector3D trackerPos;
230 QQuaternion trackerOri;
231 QVector3D focusPoint;
232 };
