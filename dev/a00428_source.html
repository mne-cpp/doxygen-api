275template<typename T>
276void Spline::updatePlot(const Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& matClassLimitData,
277 const Eigen::VectorXi& matClassFrequencyData)
278{
279 Eigen::VectorXd resultDisplayValues;
280 int iClassAmount = std::min(static_cast<int>(matClassFrequencyData.rows()),
281 static_cast<int>(matClassLimitData.rows()) - 1);
282 if(iClassAmount < 1) return;
283 this->splitCoefficientAndExponent(matClassLimitData, iClassAmount, resultDisplayValues, m_vecResultExponentValues);
284
285 m_dMinAxisX = resultDisplayValues(0);
286 m_dMaxAxisX = resultDisplayValues(iClassAmount);
287
288 m_seriesData.clear();
289 m_iMaximumFrequency = 0;
290
291 for (int ir = 0; ir < iClassAmount; ++ir)
292 {
293 double classMark = (resultDisplayValues(ir) + resultDisplayValues(ir + 1)) / 2.0;
294 m_seriesData.append(QPointF(classMark, matClassFrequencyData(ir)));
295 if (matClassFrequencyData(ir) > m_iMaximumFrequency)
296 {
297 m_iMaximumFrequency = matClassFrequencyData(ir);
298 }
299 }
300
301 m_bHasData = true;
302 m_dLeftThreshold = m_dMinAxisX;
303 m_dMiddleThreshold = m_dMaxAxisX;
304 m_dRightThreshold = m_dMaxAxisX;
305 m_bHasThresholds = false;
306
307 QWidget::update();
308}

312template <typename T>
313void Spline::splitCoefficientAndExponent(const Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic>& matClassLimitData,
314 int iClassAmount,
315 Eigen::VectorXd& vecCoefficientResults,
316 Eigen::VectorXi& vecExponentValues)
317{
318 vecCoefficientResults.resize(iClassAmount + 1);
319 vecExponentValues.resize(iClassAmount + 1);
320 double originalValue(0.0),
321 limitDisplayValue(0.0),
322 doubleExponentValue(0.0);
323 int limitExponentValue(0);
324 for (int ir = 0; ir <= iClassAmount; ++ir)
325 {
326 originalValue = matClassLimitData(ir);
327 if (originalValue == 0.0)
328 {
329 doubleExponentValue = 0.0;
330 }
331 else
332 {
333 doubleExponentValue = log10(std::fabs(originalValue));
334 }
335
336 limitExponentValue = round(doubleExponentValue);
337 limitDisplayValue = originalValue * (pow(10, -(limitExponentValue)));
338 vecCoefficientResults(ir) = limitDisplayValue;
339 vecExponentValues(ir) = limitExponentValue;
340 }
341
342 int lowestExponentValue{0},
343 highestExponentValue{0};
344
345 for (int ir = 0; ir <= iClassAmount; ++ir)
346 {
347 if (vecExponentValues(ir) < lowestExponentValue)
348 {
349 lowestExponentValue = vecExponentValues(ir);
350 }
351 if (vecExponentValues(ir) > highestExponentValue)
352 {
353 highestExponentValue = vecExponentValues(ir);
354 }
355 }
356
357 if (highestExponentValue > 0)
358 {
359 for (int ir = 0; ir <= iClassAmount; ++ir)
360 {
361 while (vecExponentValues(ir) < highestExponentValue)
362 {
363 vecCoefficientResults(ir) = vecCoefficientResults(ir) / 10;
364 vecExponentValues(ir)++;
365 }
366 }
367 }
368
369 if (lowestExponentValue < 0)
370 {
371 for (int ir = 0; ir <= iClassAmount; ++ir)
372 {
373 while (vecExponentValues(ir) > lowestExponentValue)
374 {
375 vecCoefficientResults(ir) = vecCoefficientResults(ir) * 10;
376 vecExponentValues(ir)--;
377 }
378 }
379 }
380}
