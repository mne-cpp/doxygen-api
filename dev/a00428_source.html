212 //=========================================================================================================
213 // Optical path calibration — determine the microscope optical axis
214 // relative to the tracker sensor.
215 //
216 // Workflow:
217 // 1. Move the OPMI to a position, aim at a point on the head surface,
218 // mark the focus point with the stylus pen → captureOpticalCalibSample().
219 // 2. Repeat at 2–3 different positions / focal distances.
220 // 3. Call solveOpticalCalibration() to fit a 3D line through the focus
221 // points in the tracker's local frame, yielding the optical axis
222 // direction and the optical center offset from the tracker.
223 //=========================================================================================================
224

227 QVector3D trackerPos;
228 QQuaternion trackerOri;
229 QVector3D focusPoint;
230 };
