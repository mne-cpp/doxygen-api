269void ChannelDataModel::setChannelBad(int channelIdx, bool bad)
270{
271 {
272 std::scoped_lock lk(m_infoLock);
273 if (m_pFiffInfo && channelIdx >= 0 && channelIdx < m_pFiffInfo->nchan) {
274 const std::string& channelName = m_pFiffInfo->ch_names[channelIdx];
275 const int badIndex = mne::index_of(m_pFiffInfo->bads, channelName);
276
277 if (bad) {
278 if (badIndex < 0)
279 m_pFiffInfo->bads.push_back(channelName);
280 } else if (badIndex >= 0) {
281 m_pFiffInfo->bads.erase(m_pFiffInfo->bads.begin() + badIndex);
282 }
283 }
284 }
285 m_buffer.set_channel_bad(channelIdx, bad);
286}

359std::vector<viz::TraceBuffer::ChannelSpec> ChannelDataModel::channelSpecs() const
360{
361 std::scoped_lock lk(m_infoLock);
362 std::vector<viz::TraceBuffer::ChannelSpec> specs;
363 const int realChannelCount = m_pFiffInfo ? m_pFiffInfo->nchan : 0;
364 specs.reserve(static_cast<std::size_t>(realChannelCount) + m_virtualDisplayInfo.size());
365
366 for (int ch = 0; ch < realChannelCount; ++ch) {
367 const FiffChInfo& info = m_pFiffInfo->chs[ch];
368 viz::TraceBuffer::ChannelSpec spec;
369 spec.name = m_pFiffInfo->ch_names[ch];
370 spec.type_label = typeLabelFor(info);
371 spec.color = colorForKind(info.kind);
372 // MEG: gradiometer or magnetometer scale by unit, then the common MEG scale
373 if (info.kind == FIFFV_MEG_CH) {
374 const bool grad = info.unit == FIFF_UNIT_T_M;
375 spec.scale_keys = {grad ? kMEGGradKind : kMEGMagKind, FIFFV_MEG_CH};
376 spec.default_scale = grad ? kScaleMEGGrad : kScaleMEGMag;
377 } else {
378 spec.scale_keys = {info.kind};
379 spec.default_scale = kScaleFallback;
380 }
381 spec.bad = std::ranges::contains(m_pFiffInfo->bads, spec.name);
382 specs.push_back(std::move(spec));
383 }
384
385 for (const ChannelDisplayInfo& info : m_virtualDisplayInfo) {
386 viz::TraceBuffer::ChannelSpec spec;
387 spec.name = info.name;
388 spec.type_label = info.typeLabel;
389 if (info.color.isValid())
390 spec.color = toSceneColor(info.color);
391 spec.default_scale = info.amplitudeMax;
392 spec.bad = info.bad;
393 spec.is_virtual = true;
394 specs.push_back(std::move(spec));
395 }
396 return specs;
397}
