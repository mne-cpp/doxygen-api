45uint32_t withAlpha(uint32_t color, uint32_t alpha)
46{
47 return (color & 0x00FFFFFFu) | ((alpha & 0xFFu) << 24);
48}
49
50uint32_t curvatureGray(const std::vector<float>& curvature, int index)
51{
52 return (index >= 0 && index < std::ssize(curvature) && curvature[index] > 0.0f)
53 ? 0x40u
54 : 0xAAu;
55}
56
57// Overlay alpha is blended into RGB here because the shader keeps the curvature grey in the alpha channel.
58uint32_t overCortex(uint32_t overlay, uint32_t gray)
59{
60 const uint32_t a = overlay >> 24;
61 uint32_t rgb = 0;
62 for (int shift = 0; shift < 24; shift += 8) {
63 const uint32_t c = (overlay >> shift) & 0xFFu;
64 rgb |= ((c * a + gray * (255u - a) + 127u) / 255u) << shift;
65 }
66 return withAlpha(rgb, gray);
67}

133void BrainSurface::fromSurface(const mne::FsSurface& surf)
134{
135 const Eigen::MatrixXf& rr = surf.rr();
136 const Eigen::MatrixXf& nn = surf.nn();
137 const Eigen::MatrixXi& tris = surf.tris();
138 const Eigen::VectorXf& curv = surf.curv();
139 m_curvature.resize(curv.size());
140 for (int i = 0; i < curv.size(); ++i)
141 m_curvature[i] = curv[i];
142
143 std::vector<VertexData> vertices(rr.rows());
144 for (int i = 0; i < rr.rows(); ++i)
145 vertices[i] = {QVector3D(rr(i, 0), rr(i, 1), rr(i, 2)), QVector3D(nn(i, 0), nn(i, 1), nn(i, 2))};
146 std::vector<uint32_t> indices(tris.rows() * 3);
147 for (int i = 0; i < tris.rows(); ++i)
148 for (int j = 0; j < 3; ++j)
149 indices[3 * i + j] = tris(i, j);
150 setGeometry(std::move(vertices), std::move(indices), Qt::white);
151}

155void BrainSurface::fromBemSurface(const mne::MNEBemSurface& surf, const QColor& color)
156{
157 m_curvature.clear();
158
159 const int nVerts = surf.rr.rows();
160 Eigen::MatrixX3f nn = surf.nn;
161 if (nn.rows() != nVerts) {
162 nn = mne::FsSurface::compute_normals(Eigen::MatrixX3f(surf.rr), Eigen::MatrixX3i(surf.itris));
163 }
164 std::vector<VertexData> vertices(nVerts);
165 for (int i = 0; i < nVerts; ++i)
166 vertices[i] = {QVector3D(surf.rr(i, 0), surf.rr(i, 1), surf.rr(i, 2)), QVector3D(nn(i, 0), nn(i, 1), nn(i, 2))};
167 std::vector<uint32_t> indices(surf.itris.rows() * 3);
168 for (int i = 0; i < surf.itris.rows(); ++i)
169 for (int j = 0; j < 3; ++j)
170 indices[3 * i + j] = surf.itris(i, j);
171 setGeometry(std::move(vertices), std::move(indices), color);
172}

232void BrainSurface::addAnnotation(const mne::FsAnnotation& annotation)
233{
234 m_annotation = annotation;
235 m_hasAnnotation = true;
236
237 // Per-vertex label ids and colours, so hover lookups and drawing need no search
238 const Eigen::VectorXi& vertices = m_annotation.getVertices();
239 const Eigen::VectorXi& labelIds = m_annotation.getLabelIds();
240 const mne::FsColortable& ct = m_annotation.getColortable();
241 std::unordered_map<int, uint32_t> labelColors;
242 for (int c = 0; c < ct.numEntries; ++c)
243 labelColors.insert_or_assign(ct.table(c, 4), packABGR(ct.table(c, 0), ct.table(c, 1), ct.table(c, 2)));
244 m_vertexLabelIds.assign(m_vertexData.size(), -1);
245 m_annotationColors.assign(m_vertexData.size(), 0u);
246 for (int i = 0; i < labelIds.rows(); ++i) {
247 const int v = vertices(i);
248 if (v < 0 || v >= std::ssize(m_vertexData))
249 continue;
250 m_vertexLabelIds[v] = labelIds(i);
251 const auto color = labelColors.find(labelIds(i));
252 m_annotationColors[v] = color == labelColors.end() ? 0u : color->second;
253 }
254 ++m_annotationGeneration;
255}

328void BrainSurface::updateVertexColors()
329{
330 // Cortex: curvature grey; other surfaces (BEM, sensors, ...): the base colour
331 if (!m_curvature.empty() && m_curvature.size() == m_colors.size()) {
332 for (int i = 0; i < std::ssize(m_colors); ++i) {
333 const uint32_t val = curvatureGray(m_curvature, i);
334 m_colors[i] = packABGR(val, val, val, val);
335 }
336 } else {
337 std::ranges::fill(m_colors, packABGR(m_baseColor.red(), m_baseColor.green(), m_baseColor.blue(), m_baseColor.alpha()));
338 }
339
340 // Source-estimate colours stay in the colour stream: each view selects its overlay by uniform
341 const int n = static_cast<int>(std::min(m_stcColors.size(), m_colors.size()));
342 for (int i = 0; i < n; ++i) {
343 m_colors[i] = overCortex(m_stcColors[i], curvatureGray(m_curvature, i));
344 }
345 ++m_colorGeneration;
346}

378void BrainSurface::transform(const QMatrix4x4& m)
379{
380 // Extract 3x3 normal matrix (inverse transpose of upper-left 3x3)
381 // QMatrix4x4::normalMatrix() returns QMatrix3x3.
382 QMatrix3x3 normalMat = m.normalMatrix();
383
384 for (auto& v : m_vertexData) {
385 // Transform position
386 v.pos = m.map(v.pos);
387
388 // Transform normal
389 // Note: QMatrix3x3 * QVector3D isn't directly supported by some Qt versions conveniently,
390 // but let's assume standard multiplication works or do manually.
391 // Actually QVector3D operator*(QMatrix4x4) exists but is row-vector mul.
392 // QMatrix4x4 operator*(QVector3D) is standard column-vector mul.
393
394 // Use generic map method or just manual multiply if needed.
395 // Easier: mapVector for vectors (ignores translation) but needs to be normal matrix for non-uniform scales.
396 // If scale is uniform, mapVector is fine.
397
398 // Let's do it manually to be safe with QMatrix3x3
399 const float* d = normalMat.constData();
400 float nx = d[0] * v.norm.x() + d[3] * v.norm.y() + d[6] * v.norm.z();
401 float ny = d[1] * v.norm.x() + d[4] * v.norm.y() + d[7] * v.norm.z();
402 float nz = d[2] * v.norm.x() + d[5] * v.norm.y() + d[8] * v.norm.z();
403 v.norm = QVector3D(nx, ny, nz).normalized();
404 }
405 markGeometryDirty();
406}

422std::vector<Eigen::VectorXi> BrainSurface::computeNeighbors() const
423{
424 // Use temporary std::vector<std::set<int>> for dedup during construction
425 std::vector<std::set<int>> tempNeighbors(m_vertexData.size());
426
427 // Triangles are stored in m_indexData as triplets
428 for (int i = 0; i + 2 < std::ssize(m_indexData); i += 3) {
429 int v0 = m_indexData[i];
430 int v1 = m_indexData[i + 1];
431 int v2 = m_indexData[i + 2];
432
433 // Add bidirectional edges (set handles dedup)
434 tempNeighbors[v0].insert(v1);
435 tempNeighbors[v0].insert(v2);
436 tempNeighbors[v1].insert(v0);
437 tempNeighbors[v1].insert(v2);
438 tempNeighbors[v2].insert(v0);
439 tempNeighbors[v2].insert(v1);
440 }
441
442 // Convert to std::vector<VectorXi>
443 std::vector<Eigen::VectorXi> neighbors(tempNeighbors.size());
444 for (size_t k = 0; k < tempNeighbors.size(); ++k) {
445 const auto& s = tempNeighbors[k];
446 neighbors[k].resize(static_cast<Eigen::Index>(s.size()));
447 Eigen::Index idx = 0;
448 for (int val : s) {
449 neighbors[k][idx++] = val;
450 }
451 }
452
453 return neighbors;
454}

472void BrainSurface::boundingBox(QVector3D& min, QVector3D& max) const
473{
474 if (!m_bAABBDirty) {
475 min = m_aabbMin;
476 max = m_aabbMax;
477 return;
478 }
479
480 if (m_vertexData.empty()) {
481 min = QVector3D(0, 0, 0);
482 max = QVector3D(0, 0, 0);
483 return;
484 }
485
486 min = m_vertexData[0].pos;
487 max = m_vertexData[0].pos;
488
489 for (const auto& v : m_vertexData) {
490 min.setX(std::min(min.x(), v.pos.x()));
491 min.setY(std::min(min.y(), v.pos.y()));
492 min.setZ(std::min(min.z(), v.pos.z()));
493
494 max.setX(std::max(max.x(), v.pos.x()));
495 max.setY(std::max(max.y(), v.pos.y()));
496 max.setZ(std::max(max.z(), v.pos.z()));
497 }
498
499 m_aabbMin = min;
500 m_aabbMax = max;
501 m_bAABBDirty = false;
502}

504struct BrainSurface::Bvh
505{
506 struct Node
507 {
508 float min[3];
509 float max[3];
510 int first; // first triangle (leaf) or left child (inner node)
511 int count; // triangles in a leaf, 0 for an inner node
512 int right; // right child of an inner node
513 };
514 std::vector<Node> nodes;
515 std::vector<int> triangles; // triangle indices, grouped by leaf
516 quint64 geometryGeneration = 0;
517 quint64 indexGeneration = 0;
518};

520namespace
521{
522constexpr int kBvhLeafSize = 4;
523// Pads node bounds so a ray grazing a flat, axis-aligned box still enters it
524constexpr float kBvhPad = 1e-6f;
525
526void growBox(float* lo, float* hi, const QVector3D& p)
527{
528 for (int a = 0; a < 3; ++a) {
529 lo[a] = std::min(lo[a], p[a]);
530 hi[a] = std::max(hi[a], p[a]);
531 }
532}
533
534// Ray parameter where the ray enters the box, or +inf when it misses it before maxT
535float enterBox(const float* lo, const float* hi, const float* origin, const float* invDir, float maxT)
536{
537 float tNear = 0.0f;
538 float tFar = maxT;
539 for (int a = 0; a < 3; ++a) {
540 float t0 = (lo[a] - origin[a]) * invDir[a];
541 float t1 = (hi[a] - origin[a]) * invDir[a];
542 if (t0 > t1)
543 std::swap(t0, t1);
544 // NaN (origin on a slab plane of a ray parallel to it) keeps the previous bound
545 tNear = t0 > tNear ? t0 : tNear;
546 tFar = t1 < tFar ? t1 : tFar;
547 if (tNear > tFar)
548 return std::numeric_limits<float>::infinity();
549 }
550 return tNear;
551}
552} // namespace

556const BrainSurface::Bvh& BrainSurface::bvh() const
557{
558 if (m_bvh && m_bvh->geometryGeneration == m_geometryGeneration && m_bvh->indexGeneration == m_indexGeneration)
559 return *m_bvh;
560
561 auto tree = std::make_shared<Bvh>();
562 tree->geometryGeneration = m_geometryGeneration;
563 tree->indexGeneration = m_indexGeneration;
564 const int triangleCount = int(m_indexData.size() / 3);
565
566 // Triangle boxes in one contiguous array that the build partitions in place
567 struct Item
568 {
569 float lo[3];
570 int triangle;
571 float hi[3];
572 float pad;
573 };
574 std::vector<Item> items(static_cast<size_t>(triangleCount));
575 for (int t = 0; t < triangleCount; ++t) {
576 Item& item = items[size_t(t)];
577 item.triangle = t;
578 for (int a = 0; a < 3; ++a) {
579 item.lo[a] = std::numeric_limits<float>::max();
580 item.hi[a] = std::numeric_limits<float>::lowest();
581 }
582 for (int k = 0; k < 3; ++k)
583 growBox(item.lo, item.hi, m_vertexData[m_indexData[3 * t + k]].pos);
584 }
585 tree->nodes.reserve(size_t(std::max(1, 2 * triangleCount / kBvhLeafSize + 1)));
586
587 // Top-down build: split each node at the median box centre along its widest centre axis
588 struct Task
589 {
590 int node;
591 int begin;
592 int end;
593 };
594 std::vector<Task> tasks;
595 tree->nodes.push_back({});
596 tasks.push_back({0, 0, triangleCount});
597 while (!tasks.empty()) {
598 const Task task = tasks.back();
599 tasks.pop_back();
600 Bvh::Node node{};
601 float cLo[3];
602 float cHi[3];
603 for (int a = 0; a < 3; ++a) {
604 node.min[a] = cLo[a] = std::numeric_limits<float>::max();
605 node.max[a] = cHi[a] = std::numeric_limits<float>::lowest();
606 }
607 for (int i = task.begin; i < task.end; ++i) {
608 const Item& item = items[size_t(i)];
609 for (int a = 0; a < 3; ++a) {
610 node.min[a] = std::min(node.min[a], item.lo[a]);
611 node.max[a] = std::max(node.max[a], item.hi[a]);
612 const float c = item.lo[a] + item.hi[a];
613 cLo[a] = std::min(cLo[a], c);
614 cHi[a] = std::max(cHi[a], c);
615 }
616 }
617 for (int a = 0; a < 3; ++a) {
618 node.min[a] -= kBvhPad;
619 node.max[a] += kBvhPad;
620 }
621
622 const int axis = (cHi[1] - cLo[1] > cHi[0] - cLo[0]) ? ((cHi[2] - cLo[2] > cHi[1] - cLo[1]) ? 2 : 1)
623 : ((cHi[2] - cLo[2] > cHi[0] - cLo[0]) ? 2 : 0);
624 if (task.end - task.begin <= kBvhLeafSize || cHi[axis] <= cLo[axis]) {
625 node.first = task.begin;
626 node.count = task.end - task.begin;
627 tree->nodes[size_t(task.node)] = node;
628 continue;
629 }
630 const int mid = (task.begin + task.end) / 2;
631 std::nth_element(items.begin() + task.begin, items.begin() + mid, items.begin() + task.end,
632 [axis](const Item& l, const Item& r) { return l.lo[axis] + l.hi[axis] < r.lo[axis] + r.hi[axis]; });
633 const int left = int(tree->nodes.size());
634 const int right = left + 1;
635 tree->nodes.resize(tree->nodes.size() + 2);
636 node.first = left;
637 node.count = 0;
638 node.right = right;
639 tree->nodes[size_t(task.node)] = node;
640 tasks.push_back({right, mid, task.end});
641 tasks.push_back({left, task.begin, mid});
642 }
643
644 tree->triangles.resize(size_t(triangleCount));
645 for (int i = 0; i < triangleCount; ++i)
646 tree->triangles[size_t(i)] = items[size_t(i)].triangle;
647
648 m_bvh = std::move(tree);
649 return *m_bvh;
650}

654bool BrainSurface::intersects(const QVector3D& rayOrigin, const QVector3D& rayDir, float& dist, int& vertexIdx) const
655{
656 vertexIdx = -1;
657 if (m_vertexData.empty() || m_indexData.size() < 3)
658 return false;
659
660 const Bvh& tree = bvh();
661 const float origin[3] = {rayOrigin.x(), rayOrigin.y(), rayOrigin.z()};
662 const float invDir[3] = {1.0f / rayDir.x(), 1.0f / rayDir.y(), 1.0f / rayDir.z()};
663 const double originX = rayOrigin.x(), originY = rayOrigin.y(), originZ = rayOrigin.z();
664 const double dirX = rayDir.x(), dirY = rayDir.y(), dirZ = rayDir.z();
665
666 double closestDist = std::numeric_limits<double>::max();
667 int closestTriangle = -1;
668
669 // Möller–Trumbore in double precision on the triangles of the leaves the ray enters, nearest
670 // first. Median splits keep the depth near log2(triangles / leaf size), far below the stack size.
671 int stack[64];
672 int top = 0;
673 stack[top++] = 0;
674 while (top > 0) {
675 const Bvh::Node& node = tree.nodes[size_t(stack[--top])];
676 const float limit = closestDist < double(std::numeric_limits<float>::max()) ? float(closestDist) : std::numeric_limits<float>::max();
677 if (enterBox(node.min, node.max, origin, invDir, limit) == std::numeric_limits<float>::infinity())
678 continue;
679 if (node.count == 0) {
680 const Bvh::Node& l = tree.nodes[size_t(node.first)];
681 const Bvh::Node& r = tree.nodes[size_t(node.right)];
682 const float tl = enterBox(l.min, l.max, origin, invDir, limit);
683 const float tr = enterBox(r.min, r.max, origin, invDir, limit);
684 constexpr float miss = std::numeric_limits<float>::infinity();
685 // Push the farther child first so the nearer one is visited next
686 const bool leftFirst = tl <= tr;
687 const int nearChild = leftFirst ? node.first : node.right;
688 const int farChild = leftFirst ? node.right : node.first;
689 if ((leftFirst ? tr : tl) != miss)
690 stack[top++] = farChild;
691 if ((leftFirst ? tl : tr) != miss)
692 stack[top++] = nearChild;
693 continue;
694 }
695
696 for (int i = node.first; i < node.first + node.count; ++i) {
697 const int t = tree.triangles[size_t(i)];
698 const QVector3D& v0q = m_vertexData[m_indexData[3 * t]].pos;
699 const QVector3D& v1q = m_vertexData[m_indexData[3 * t + 1]].pos;
700 const QVector3D& v2q = m_vertexData[m_indexData[3 * t + 2]].pos;
701
702 const double v0x = v0q.x(), v0y = v0q.y(), v0z = v0q.z();
703 const double edge1x = v1q.x() - v0x, edge1y = v1q.y() - v0y, edge1z = v1q.z() - v0z;
704 const double edge2x = v2q.x() - v0x, edge2y = v2q.y() - v0y, edge2z = v2q.z() - v0z;
705
706 const double hx = dirY * edge2z - dirZ * edge2y;
707 const double hy = dirZ * edge2x - dirX * edge2z;
708 const double hz = dirX * edge2y - dirY * edge2x;
709
710 const double a = edge1x * hx + edge1y * hy + edge1z * hz;
711 if (std::abs(a) < 1e-18)
712 continue; // Parallel to the triangle
713
714 const double f = 1.0 / a;
715 const double sx = originX - v0x, sy = originY - v0y, sz = originZ - v0z;
716 const double u = f * (sx * hx + sy * hy + sz * hz);
717 if (u < -1e-7 || u > 1.0000001)
718 continue;
719
720 const double qx = sy * edge1z - sz * edge1y;
721 const double qy = sz * edge1x - sx * edge1z;
722 const double qz = sx * edge1y - sy * edge1x;
723
724 const double v = f * (dirX * qx + dirY * qy + dirZ * qz);
725 if (v < -1e-7 || u + v > 1.0000001)
726 continue;
727
728 const double hitT = f * (edge2x * qx + edge2y * qy + edge2z * qz);
729 if (hitT > 1e-7 && hitT < closestDist) {
730 closestDist = hitT;
731 closestTriangle = t;
732 }
733 }
734 }
735
736 if (closestTriangle < 0)
737 return false;
738
739 // The vertex of the hit triangle nearest to the hit point, used for region lookup
740 const QVector3D hit = rayOrigin + float(closestDist) * rayDir;
741 float best = std::numeric_limits<float>::max();
742 for (int k = 0; k < 3; ++k) {
743 const int v = int(m_indexData[3 * closestTriangle + k]);
744 const float d = (m_vertexData[v].pos - hit).lengthSquared();
745 if (d < best) {
746 best = d;
747 vertexIdx = v;
748 }
749 }
750 dist = float(closestDist);
751 return true;
752}

758 if (!m_hasAnnotation || vertexIdx < 0 || vertexIdx >= std::ssize(m_vertexLabelIds)) {

772 }

774 }
