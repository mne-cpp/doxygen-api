78void MNEMeasData::adjust_baselines(float bmin, float bmax)
79{
80 if (!current)
81 return;
82
83 const float currentSfreq = 1.0f / current->tstep;
84 const float tmin = current->tmin;
85 const float tmax = current->tmin + (current->np - 1) / currentSfreq;
86
87 int b1, b2;
88 if (bmin < tmin)
89 b1 = 0;
90 else if (bmin > tmax)
91 b1 = current->np;
92 else {
93 for (b1 = 0; b1 / currentSfreq + tmin < bmin; b1++)
94 ;
95 b1 = std::clamp(b1, 0, current->np);
96 }
97 if (bmax < tmin)
98 b2 = 0;
99 else if (bmax > tmax)
100 b2 = current->np;
101 else {
102 for (b2 = current->np; b2 / currentSfreq + tmin > bmax; b2--)
103 ;
104 b2 = std::clamp(b2, 0, current->np);
105 }
106
107 Eigen::MatrixXf& bdata = current->data;
108 if (b2 > b1) {
109 for (int c = 0; c < nchan; c++) {
110 float ave = 0.0f;
111 for (int s = b1; s < b2; s++)
112 ave += bdata(s, c);
113 ave /= (b2 - b1);
114 current->baselines[c] += ave;
115 for (int s = 0; s < current->np; s++)
116 bdata(s, c) -= ave;
117 }
118 mne::Log::info("\t{} : using baseline {:7.1f} ... {:7.1f} ms\n", current->comment.c_str() ? current->comment.c_str() : "unknown", 1000 * (tmin + b1 / sfreq), 1000 * (tmin + b2 / sfreq));
119 }
120}

124bool MNEMeasData::read_evoked_set(const std::filesystem::path& name,
125 int set,
126 MNEInverseOperator* op,
127 MNENamedMatrix* fwd,
128 const StringList& namesp,
129 int nnamesp,
130 MNEMeasData& target,
131 bool fresh)
132/*
133 * Read an evoked-response data file
134 */
135{
136 /*
137 * Read evoked data via FiffEvoked (no baseline correction, no projection
138 * — MNEMeasData handles projections separately via MNEProjOp).
139 */
140 auto read = FiffEvoked::read(name, set - 1, {}, false);
141 if (!read) {
142 mne::Log::critical("{}", read.error().what());
143 return false;
144 }
145 const FiffEvoked& evoked = *read;
146
147 /*
148 * Extract fields from the FiffEvoked object
149 */
150 const int nchan_file = evoked.info.nchan;
151 const int nsamp = evoked.last - evoked.first + 1;
152 const float sfreq = evoked.info.sfreq;
153 const float dtmin = static_cast<float>(evoked.first) / sfreq;
154 const float highpass = evoked.info.highpass;
155 const float lowpass = evoked.info.lowpass;
156 const int nave = evoked.nave;
157 const int aspect_kind = evoked.aspect_kind;
158 const std::vector<FiffChInfo>& chs = evoked.info.chs;
159 const FiffId& id = evoked.info.meas_id;
160 const MatrixXf data = evoked.data.cast<float>(); /* nchan × nsamp, already calibrated */
161 const FiffCoordTrans& devHeadT = evoked.info.dev_head_t;
162
163 /*
164 * Desired channels
165 */
166 StringList names;
167 int nchan = 0;
168 /*
169 * Selected channels
170 */
171 Eigen::VectorXi sel;
172 int stim14 = -1;
173 /*
174 * Other stuff
175 */
176 float tmin, tmax;
177 int k, p, c, np, n1, n2;
178
179 const char* trigger = std::getenv(MNE_ENV_TRIGGER_CH);
180 const std::string_view stim14_name = trigger && *trigger ? trigger : MNE_DEFAULT_TRIGGER_CH;
181
182 if (!fresh) {
183 for (int i = 0; i < target.nchan; i++)
184 names.push_back(target.chs[i].ch_name);
185 nchan = target.nchan;
186 } else {
187 if (op) {
188 names = op->eigen_fields.col_names;
189 nchan = op->nchan;
190 } else if (fwd) {
191 names = fwd->collist;
192 nchan = fwd->ncol();
193 } else {
194 names = namesp;
195 nchan = nnamesp;
196 }
197 if (names.empty())
198 nchan = 0;
199 }
200
201 if (!id.empty())
202 mne::Log::info("\tMeasurement file id: {}\n", id.to_string());
203
204 /*
205 * Pick out the necessary channels
206 */
207 if (nchan > 0) {
208 sel = Eigen::VectorXi::Constant(nchan, -1);
209 for (c = 0; c < nchan_file; c++) {
210 for (k = 0; k < nchan; k++) {
211 if (sel[k] == -1 && (chs[c].ch_name == names[k])) {
212 sel[k] = c;
213 break;
214 }
215 }
216 if (mne::same_channel(chs[c].ch_name, stim14_name)) {
217 stim14 = c;
218 }
219 }
220 for (k = 0; k < nchan; k++)
221 if (sel[k] == -1) {
222 mne::Log::critical("All channels needed were not in the MEG/EEG data file (first missing: {}).", names[k]);
223 return false;
224 }
225 } else { /* Load all channels */
226 sel.resize(nchan_file);
227 sel.setZero();
228 for (c = 0, nchan = 0; c < nchan_file; c++) {
229 if (chs[c].kind == FIFFV_MEG_CH || chs[c].kind == FIFFV_EEG_CH) {
230 sel[nchan] = c;
231 nchan++;
232 }
233 if (mne::same_channel(chs[c].ch_name, stim14_name)) {
234 stim14 = c;
235 }
236 }
237 }
238 /*
239 * Cut the data to the analysis time range
240 */
241 n1 = 0;
242 n2 = nsamp;
243 np = n2 - n1;
244 tmin = dtmin;
245 tmax = dtmin + (np - 1) / sfreq;
246 mne::Log::info("\tData time range: {:8.1f} ... {:8.1f} ms\n", 1000 * tmin, 1000 * tmax);
247 /*
248 * Just put it together
249 */
250 if (fresh) {
251 target.filename = utf8_from_path(name);
252 target.meas_id = id;
253 /*
254 * Getting starting time from measurement ID is not too accurate...
255 */
256 {
257 FiffTime md;
258 md.secs = evoked.info.meas_date[0];
259 md.usecs = evoked.info.meas_date[1];
260 if (md.secs != 0)
261 target.meas_date = md;
262 else if (!target.meas_id.empty())
263 target.meas_date = target.meas_id.time;
264 else {
265 target.meas_date.secs = 0;
266 target.meas_date.usecs = 0;
267 }
268 }
269 target.lowpass = lowpass;
270 target.highpass = highpass;
271 target.nchan = nchan;
272 target.sfreq = sfreq;
273
274 if (!devHeadT.empty()) {
275 target.meg_head_t = std::make_unique<FiffCoordTrans>(devHeadT);
276 mne::Log::info("\tUsing MEG <-> head transform from the present data set\n");
277 }
278 if (op != nullptr && !op->mri_head_t.empty()) { /* Copy if available */
279 target.mri_head_t = std::make_unique<FiffCoordTrans>(op->mri_head_t);
280 mne::Log::info("\tPicked MRI <-> head transform from the inverse operator\n");
281 }
282 /*
283 * Channel list
284 */
285 for (k = 0; k < nchan; k++) {
286 target.chs.push_back(FiffChInfo());
287 target.chs[k] = chs[sel[k]];
288 }
289
290 target.op = op; /* Attach inverse operator */
291 target.fwd = fwd; /* ...or a fwd operator */
292 if (op) { /* Attach the projection operator and CTF compensation info to the data, too */
293 target.proj = MNEProjOp::read(name);
294 if (target.proj && !target.proj->items.empty()) {
295 mne::Log::info("\tLoaded projection from {}:\n", mne::utf8_from_path(name));
296 std::ostringstream report;
297 target.proj->report(report, "\t\t");
298 mne::Log::info("{}", report.str());
299 }
300 } else {
301 target.proj = MNEProjOp::read(name);
302 if (target.proj && !target.proj->items.empty()) {
303 mne::Log::info("\tLoaded projection from {}:\n", mne::utf8_from_path(name));
304 std::ostringstream report;
305 target.proj->report(report, "\t\t");
306 mne::Log::info("{}", report.str());
307 }
308 target.comp = MNECTFCompDataSet::read(name);
309 if (!target.comp) {
310 return false;
311 }
312 if (target.comp->ncomp > 0)
313 mne::Log::info("\tRead {} compensation data sets from {}\n", target.comp->ncomp, mne::utf8_from_path(name));
314 }
315 /*
316 * Bad channels — already read by FiffEvoked via FiffStream::read_meas_info()
317 */
318 {
319 target.badlist = evoked.info.bads;
320 target.bad = Eigen::VectorXi::Zero(target.nchan);
321
322 for (const auto& bad_name : target.badlist) {
323 for (k = 0; k < target.nchan; k++) {
324 if (mne::iequals(target.chs[k].ch_name, bad_name)) {
325 target.bad[k] = 1;
326 break;
327 }
328 }
329 }
330 mne::Log::info("\t{} bad channels read from {}{}{}", target.badlist.size(), mne::utf8_from_path(name),
331 target.badlist.empty() ? "" : ": ", mne::join(target.badlist, " "));
332 }
333 }
334 /*
335 * New data set is created anyway
336 */
337 auto dataset = std::make_unique<MNEMeasDataSet>();
338 dataset->tmin = tmin;
339 dataset->tstep = 1.0 / sfreq;
340 dataset->first = n1;
341 dataset->np = np;
342 dataset->nave = nave;
343 dataset->kind = aspect_kind;
344 dataset->data = Eigen::MatrixXf::Zero(np, nchan);
345 dataset->comment = evoked.comment;
346 dataset->baselines = Eigen::VectorXf::Zero(nchan);
347 /*
348 * Pick data from all channels
349 */
350 for (k = 0; k < nchan; k++) {
351 /*
352 * Shift the response
353 */
354 for (p = 0; p < np; p++)
355 dataset->data(p, k) = data(sel[k], p + n1);
356 }
357 /*
358 * Pick the digital trigger channel, too
359 */
360 if (stim14 >= 0) {
361 dataset->stim14 = Eigen::VectorXf(np);
362 for (p = 0; p < np; p++) /* Copy the data and correct for the possible non-unit calibration */
363 dataset->stim14[p] = data(stim14, p + n1) / chs[stim14].cal;
364 }
365 const std::string comment = dataset->comment.empty() ? std::string("unknown") : dataset->comment;
366 target.sets.push_back(std::move(dataset));
367 target.nset++;
368 if (fresh)
369 target.current = target.sets.front().get();
370 mne::Log::info("\t{} dataset {} from {}", fresh ? "Loaded" : "Added", comment, mne::utf8_from_path(name));
371 return true;
372}
