48void SensorLayoutWidget::setLayout(const std::map<std::string, QPointF>& layoutMap, const std::vector<std::string>& badChannels)
49{
50 std::vector<scene::SensorLayout::Sensor> sensors;
51 sensors.reserve(layoutMap.size());
52 for (const auto& [name, position] : layoutMap) {
53 scene::SensorLayout::Sensor sensor;
54 sensor.name = name;
55 sensor.position = {position.x(), position.y()};
56 sensor.kind = name.find("EEG") != std::string::npos ? FIFFV_EEG_CH : FIFFV_MEG_CH;
57 sensor.unit = FIFF_UNIT_T_M;
58 sensor.bad = std::ranges::contains(badChannels, name);
59 sensors.push_back(std::move(sensor));
60 }
61 // New sensors start unselected; as when a QGraphicsScene clears its items, a lost selection is reported
62 const bool hadSelection = !m_layout.selected().empty();
63 m_layout.set_sensors(std::move(sensors));
64 contentsChanged();
65 if (hadSelection) {
66 emit selectionChanged();
67 }
68}

86void SensorLayoutWidget::setSelectable(const std::vector<std::string>& names)
87{
88 // As in a QGraphicsScene, hidden or unselectable sensors lose their selection
89 bool deselected = false;
90 for (scene::SensorLayout::Sensor& sensor : m_layout.sensors()) {
91 sensor.visible = sensor.kind == m_kind;
92 sensor.selectable = sensor.visible && std::ranges::contains(names, sensor.name);
93 deselected |= sensor.selected && !sensor.selectable;
94 sensor.selected = sensor.selected && sensor.selectable;
95 }
96 contentsChanged();
97 if (deselected) {
98 emit selectionChanged();
99 }
100}

147SelectionItem SensorLayoutWidget::selection() const
148{
149 const bool anySelected = !m_layout.selected().empty();
150 SelectionItem item;
151 for (const scene::SensorLayout::Sensor& sensor : m_layout.sensors()) {
152 if (anySelected ? sensor.selected : sensor.visible) {
153 item.m_sChannelName.push_back(sensor.name);
154 item.m_iChannelNumber.push_back(0);
155 item.m_iChannelKind.push_back(sensor.kind);
156 item.m_iChannelUnit.push_back(sensor.unit);
157 item.m_qpChannelPosition.emplace_back(sensor.position.x, sensor.position.y);
158 }
159 }
160 return item;
161}
