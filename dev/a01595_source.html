47constexpr int kRecordsPerLap = 60; // updates of one station per sweep

63 AcqError write(std::string_view command) override
64 {
65 while (!command.empty()) {
66 const std::size_t end = std::min(command.find('\r'), command.size());
67 const std::string_view one = command.substr(0, end);
68 if (one == "C") {
69 m_streaming = true;
70 m_epoch = Clock::now() - std::chrono::duration_cast<Clock::duration>(std::chrono::duration<double>(static_cast<double>(m_record) / m_simulation.rate));
71 } else if (one == "P") {
72 m_streaming = false;
73 }
74 command.remove_prefix(std::min(end + 1, command.size()));
75 }
76 return {};
77 }

79 AcqExpected<std::size_t> read(std::span<char> buffer, std::chrono::milliseconds timeout) override
80 {
81 const auto deadline = Clock::now() + timeout;
82 while (true) {
83 if (!m_pending.empty()) {
84 const std::size_t n = std::min(buffer.size(), m_pending.size());
85 std::memcpy(buffer.data(), m_pending.data(), n);
86 m_pending.erase(0, n);
87 return n;
88 }
89 if (m_simulation.duration > 0.0 && static_cast<double>(m_record) >= m_simulation.duration * m_simulation.rate) {
90 return AcqError(AcqErrorCode::EndOfStream);
91 }
92 if (m_streaming) {
93 const std::uint64_t due = m_simulation.realtime
94 ? static_cast<std::uint64_t>(std::chrono::duration<double>(Clock::now() - m_epoch).count() * m_simulation.rate)
95 : m_record + kBurst;
96 if (due > m_record) {
97 std::uint64_t end = std::min(due, m_record + kBurst);
98 if (m_simulation.duration > 0.0) {
99 end = std::min(end, static_cast<std::uint64_t>(std::ceil(m_simulation.duration * m_simulation.rate)));
100 }
101 generate(end);
102 continue;
103 }
104 const auto next = m_epoch + std::chrono::duration_cast<Clock::duration>(std::chrono::duration<double>(static_cast<double>(m_record + 1) / m_simulation.rate));
105 if (next < deadline) {
106 std::this_thread::sleep_until(next);
107 continue;
108 }
109 }
110 std::this_thread::sleep_until(deadline);
111 return AcqError(AcqErrorCode::Timeout);
112 }
113 }

115private:
117 void generate(std::uint64_t end)
118 {
119 const std::size_t stations = m_simulation.stations.size();
120 const double per_station = m_simulation.rate / static_cast<double>(stations);
121 const double scale = m_simulation.units == FastrakUnits::Inches ? 1.0 / 0.0254 : 100.0;
122 while (m_record < end) {
123 const int station = m_simulation.stations[m_record % stations];
124 const std::uint64_t update = m_record / stations;
125 ++m_record;
126 const double phase = 2.0 * std::numbers::pi * static_cast<double>((update + m_simulation.seed) % kRecordsPerLap) / kRecordsPerLap;
127 const double elevation = 0.3 * std::sin(3.0 * phase);
128 double x = 0.05 * station;
129 double y = 0.0;
130 double z = 0.0;
131 double azimuth = 0.0;
132 if (station == 1) {
133 x = kRadius * std::cos(phase + 0.3);
134 y = kRadius * std::sin(phase + 0.3);
135 z = kRadius * elevation + 0.02;
136 azimuth = phase * 180.0 / std::numbers::pi;
137 if (azimuth > 180.0) {
138 azimuth -= 360.0;
139 }
140 } else if (station == 2) {
141 x = kRadius * std::cos(phase);
142 y = kRadius * std::sin(phase);
143 z = kRadius * elevation;
144 }
145 m_pending += std::format("0{:1d} {:9.4f}{:9.4f}{:9.4f}{:9.3f}{:9.3f}{:9.3f}\r\n", station, x * scale, y * scale, z * scale, azimuth, 0.0, 0.0);
146
147 if (m_simulation.button > 0.0 && m_record % stations == 0) {
148 const auto press = static_cast<std::uint64_t>(m_simulation.button * per_station);
149 if (press > 0 && (update + 1) % press == 0) {
150 m_streaming = false; // the stylus button pauses the output until the next "C"
151 return;
152 }
153 }
154 }
155 }

159 std::uint64_t m_record = 0;
160 Clock::time_point m_epoch = Clock::now();
161 std::string m_pending;
162};
