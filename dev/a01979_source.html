126 settings.beginGroup(m_sSettingsPath + QString("/AverageSelectionView/averageColorMap"));

146{
147 if (m_sSettingsPath.isEmpty()) {
148 return;
149 }
150
151 QSettings settings("MNECPP");
152
153 settings.beginGroup(m_sSettingsPath + QString("/AverageSelectionView/averageColorMap"));
154 for (const QString& key : settings.childKeys()) {
155 m_qMapAverageColor->insert_or_assign(key.toStdString(), settings.value(key).value<QColor>());

160 for (const QString& key : settings.childKeys()) {

162 }

192void AverageSelectionView::redrawGUI()
193{
194 if (m_qMapAverageColor->size() != m_qMapAverageActivation->size()) {
195 mne::Log::debug("AverageSelectionView::update - m_qMapAverageColor and m_qMapAverageActivation do not match in size. Returning.");
196 return;
197 }
198
199 //Delete all widgets in the averages layout
200 QGridLayout* topLayout = static_cast<QGridLayout*>(this->layout());
201 if (!topLayout) {
202 topLayout = new QGridLayout(this);
203 }
204
205 // The widgets go right away: the view is redrawn synchronously and looked up by name
206 detail::clearLayout(topLayout);
207
208 // Create new GUI elements
209 int count = 0;
210 for (const auto& [key, color] : *m_qMapAverageColor) {
211 if (count >= m_iMaxNumAverages) {
212 break;
213 }
214
215 const QString name = QString::fromStdString(key);
216
217 //Create average active checkbox
218 QPointer<QCheckBox> pCheckBox = new QCheckBox(name, this);
219 pCheckBox->setChecked(detail::valueOr(*m_qMapAverageActivation, key));
220 pCheckBox->setObjectName(name);
221 topLayout->addWidget(pCheckBox, count, 0);
222 connect(pCheckBox.data(), &QCheckBox::clicked,
223 this, &AverageSelectionView::onAverageSelectionColorChanged);
224
225 //Create average color pushbutton
226 QPointer<QPushButton> pButton = new QPushButton("Click to change", this);
227 pButton->setObjectName(name);
228 pButton->setStyleSheet(QString("background-color: rgb(%1, %2, %3);").arg(color.red()).arg(color.green()).arg(color.blue()));
229 topLayout->addWidget(pButton, count, 1);
230 connect(pButton.data(), &QPushButton::clicked,
231 this, &AverageSelectionView::onAverageSelectionColorChanged);
232
233 ++count;
234 }
235}

239void AverageSelectionView::onAverageSelectionColorChanged()
240{
241 //Change color for average
242 if (QPointer<QPushButton> button = qobject_cast<QPushButton*>(sender())) {
243 const std::string sObjectName = button->objectName().toStdString();
244
245 QColor color = QColorDialog::getColor(detail::valueOr(*m_qMapAverageColor, sObjectName), this, "Set average color");
246 // Cancelled: getColor() returns an invalid colour
247 if (!color.isValid()) {
248 return;
249 }
250
251 if (button) {
252 QPalette palette(QPalette::Button, color);
253 button->setPalette(palette);
254 button->update();
255
256 //Set color of button new new scene color
257 button->setStyleSheet(QString("background-color: rgb(%1, %2, %3);").arg(color.red()).arg(color.green()).arg(color.blue()));
258 }
259
260 m_qMapAverageColor->insert_or_assign(sObjectName, color);
261
262 emit newAverageColorMap(m_qMapAverageColor);
263 }
264
265 //Change color for average
266 if (QPointer<QCheckBox> checkBox = qobject_cast<QCheckBox*>(sender())) {
267 const std::string sObjectName = checkBox->objectName().toStdString();
268
269 m_qMapAverageActivation->insert_or_assign(sObjectName, checkBox->isChecked());
270
271 emit newAverageActivationMap(m_qMapAverageActivation);
272 }
273
274 saveSettings();
275}
