49void SensorLayoutWidget::setLayout(const std::map<std::string, QPointF>& layoutMap, const std::vector<std::string>& badChannels)
50{
51 std::vector<viz::SensorLayout::Sensor> sensors;
52 sensors.reserve(layoutMap.size());
53 for (const auto& [name, position] : layoutMap) {
54 viz::SensorLayout::Sensor sensor;
55 sensor.name = name;
56 sensor.position = {position.x(), position.y()};
57 sensor.kind = name.find("EEG") != std::string::npos ? FIFFV_EEG_CH : FIFFV_MEG_CH;
58 sensor.unit = FIFF_UNIT_T_M;
59 sensor.bad = std::ranges::contains(badChannels, name);
60 sensors.push_back(std::move(sensor));
61 }
62 // New sensors start unselected; as when a QGraphicsScene clears its items, a lost selection is reported
63 const bool hadSelection = !m_layout.selected().empty();
64 m_layout.set_sensors(std::move(sensors));
65 contentsChanged();
66 if (hadSelection) {
67 emit selectionChanged();
68 }
69}

87void SensorLayoutWidget::setSelectable(const std::vector<std::string>& names)
88{
89 // As in a QGraphicsScene, hidden or unselectable sensors lose their selection
90 bool deselected = false;
91 for (viz::SensorLayout::Sensor& sensor : m_layout.sensors()) {
92 sensor.visible = sensor.kind == m_kind;
93 sensor.selectable = sensor.visible && std::ranges::contains(names, sensor.name);
94 deselected |= sensor.selected && !sensor.selectable;
95 sensor.selected = sensor.selected && sensor.selectable;
96 }
97 contentsChanged();
98 if (deselected) {
99 emit selectionChanged();
100 }
101}

148SelectionItem SensorLayoutWidget::selection() const
149{
150 const bool anySelected = !m_layout.selected().empty();
151 SelectionItem item;
152 for (const viz::SensorLayout::Sensor& sensor : m_layout.sensors()) {
153 if (anySelected ? sensor.selected : sensor.visible) {
154 item.m_sChannelName.push_back(sensor.name);
155 item.m_iChannelNumber.push_back(0);
156 item.m_iChannelKind.push_back(sensor.kind);
157 item.m_iChannelUnit.push_back(sensor.unit);
158 item.m_qpChannelPosition.emplace_back(sensor.position.x, sensor.position.y);
159 }
160 }
161 return item;
162}

187bool SensorLayoutWidget::scenePressed(viz::Point point, Qt::KeyboardModifiers modifiers)
188{
189 const bool extend = modifiers.testFlag(Qt::ControlModifier);
190 const bool hit = m_layout.sensor_at(point) >= 0;
191 // A press beside the sensors clears the selection before a band starts, unless Ctrl is held
192 if ((hit || !extend) && m_layout.click(point, extend)) {
193 update();
