51std::optional<std::array<std::string, 3>> splitTriple(std::string_view text, char separator)
52{
53 std::array<std::string, 3> fields;
54 std::size_t start = 0;
55 for (std::size_t k = 0; k < 3; ++k) {
56 const std::size_t end = k < 2 ? text.find(separator, start) : text.size();
57 if (end == std::string_view::npos) {
58 return std::nullopt;
59 }
60 fields[k] = std::string(mne::trimmed(text.substr(start, end - start)));
61 start = end + 1;
62 }
63 return fields;
64}

81std::optional<std::chrono::sys_time<std::chrono::microseconds>> EDFReader::parse_start_date(std::string_view recordingId,
82 std::string_view startDate,
83 std::string_view startTime)
84{
85 using namespace std::chrono;
86 // EDF+ "Startdate dd-MMM-yyyy ..." carries all four digits of the year and wins over the header date
87 std::optional<sys_days> date;
88 const std::vector<std::string> recording = mne::split(mne::trimmed(recordingId), ' ');
89 if (recording.size() == 5) {
90 if (const auto fields = splitTriple(recording[1], '-')) {
91 static constexpr std::array<std::string_view, 12> months{"JAN", "FEB", "MAR", "APR", "MAY", "JUN",
92 "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"};
93 const auto month = std::ranges::find_if(months, [&](std::string_view name) { return mne::iequals(name, (*fields)[1]); });
94 if (month != months.end()) {
95 date = civilDate(mne::to_int((*fields)[0]), static_cast<int>(month - months.begin()) + 1, mne::to_int((*fields)[2]));
96 }
97 }
98 }
99 // Otherwise the two-digit header year: 85-99 is 1985-1999, 00-84 is 2000-2084
100 if (!date) {
101 if (const auto fields = splitTriple(startDate, '.')) {
102 std::optional<int> year = mne::to_int((*fields)[2]);
103 if (year) {
104 *year += *year < 85 ? 2000 : 1900;
105 }
106 date = civilDate(mne::to_int((*fields)[0]), mne::to_int((*fields)[1]), year);
107 }
108 }
109 if (!date) {
110 return std::nullopt;
111 }
112 sys_time<microseconds> time = *date;
113 if (const auto fields = splitTriple(startTime, '.')) {
114 const auto hour = mne::to_int((*fields)[0]);
115 const auto minute = mne::to_int((*fields)[1]);
116 const auto second = mne::to_int((*fields)[2]);
117 if (hour && minute && second) {
118 time += hours(*hour) + minutes(*minute) + seconds(*second);
119 }
120 }
121 return time;
122}

147FiffChInfo EDFChannelInfo::to_fiff_ch_info() const
148{
149 FiffChInfo info;
150
151 info.scanNo = channelNumber;
152 info.logNo = channelNumber;
153
154 const std::string lower = mne::to_lower(label);
155 const auto has = [&lower](std::string_view part) {
156 return lower.find(part) != std::string::npos;
157 };
158 if (!isMeasurement) {
159 info.kind = has("stim") ? FIFFV_STIM_CH : FIFFV_MISC_CH;
160 } else if (isAutoStimLabel(label)) {
161 info.kind = FIFFV_STIM_CH;
162 } else if (has("ecog")) {
163 info.kind = FIFFV_ECOG_CH;
164 } else if (has("seeg")) {
165 info.kind = FIFFV_SEEG_CH;
166 } else if (has("eeg")) {
167 info.kind = FIFFV_EEG_CH;
168 } else if (has("meg")) {
169 info.kind = FIFFV_MEG_CH;
170 } else if (has("ecg")) {
171 info.kind = FIFFV_ECG_CH;
172 } else if (has("eog")) {
173 info.kind = FIFFV_EOG_CH;
174 } else if (has("emg")) {
175 info.kind = FIFFV_EMG_CH;
176 } else {
177 info.kind = FIFFV_EEG_CH;
178 }
179
180 // Samples are converted to SI by to_si(), so voltages are stored in V
181 info.unit = (info.kind != FIFFV_STIM_CH && physicalDimension.ends_with('V')) ? FIFF_UNIT_V : FIFF_UNIT_NONE;
182 info.unit_mul = FIFF_UNITM_NONE;
183
184 info.cal = 1.0f;
185 info.range = 1.0f;
186 info.ch_name = label;
187
188 return info;
189}

193float EDFChannelInfo::to_si() const
194{
195 // mne read_raw_edf: µV (micro sign, Greek mu, Shift-JIS mu) and uV are 1e-6, mV 1e-3, anything else 1
196 if (isAutoStimLabel(label)) {
197 return 1.0f;
198 }
199 const std::string_view d = physicalDimension;
200 if (d == "\xC2\xB5V" || d == "\xCE\xBCV" || d == "\xC2\x83\xC3\x8AV" || d == "uV") {
201 return 1.0e-6f;
202 }
203 if (d == "mV") {
204 return 1.0e-3f;
205 }
206 return 1.0f;
207}

219bool EDFReader::open(const std::filesystem::path& path)
220{
221 m_file = mne::FileByteStream::open(path, mne::FileByteStream::Mode::Read);
222 if (!m_file) {
223 mne::Log::warning("[EDFReader::open] Could not open file: {}", mne::utf8_from_path(path));
224 return false;
225 }
226
227 // Like mne, the extension decides between EDF (16 bit) and BDF (24 bit)
228 m_bytesPerSample = mne::iequals(mne::utf8_from_path(path.extension()), ".bdf") ? 3 : 2;
229 if (!parse_header()) {
230 mne::Log::warning("[EDFReader::open] Not an EDF/BDF header: {}", mne::utf8_from_path(path));
231 m_file.reset();
232 return false;
233 }
234 return true;
235}

239bool EDFReader::parse_header()
240{
241 // Fixed part: version 8, patient 80, recording 80, start date 8, start time 8, header bytes 8, reserved 44,
242 // record count 8, record duration 8, signal count 4
243 constexpr std::size_t fixedBytes = 256;
244 constexpr std::size_t bytesPerSignal = 256;
245 std::string header(fixedBytes, '\0');
246 if (!m_file->seek(0) || m_file->read(header.data(), fixedBytes) != fixedBytes) {
247 return false;
248 }
249 std::size_t cursor = 0;
250 const auto field = [&header, &cursor](std::size_t width) {
251 const std::string_view value = std::string_view(header).substr(cursor, width);
252 cursor += width;
253 return value;
254 };
255
256 field(8); // version
257 field(80); // patient
258 const std::string_view recordingId = field(80);
259 const std::string_view startDate = field(8);
260 const std::string_view startTime = field(8);
261 m_measDate = parse_start_date(recordingId, startDate, startTime);
262 m_numBytesInHeader = mne::to_int(field(8)).value_or(0);
263 field(44); // reserved
264 m_numDataRecords = mne::to_int(field(8)).value_or(0);
265 const auto recordDuration = static_cast<float>(mne::to_double(field(8)).value_or(0.0));
266 const int numChannels = mne::to_int(field(4)).value_or(0);
267 if (numChannels <= 0 || static_cast<std::uint64_t>(numChannels) * bytesPerSignal + fixedBytes > m_file->size()) {
268 return false;
269 }
270
271 // Per-signal fields, each stored for all signals in turn: label 16, transducer 80, dimension 8, physical min 8,
272 // physical max 8, digital min 8, digital max 8, prefiltering 80, samples per record 8, reserved 32
273 const auto count = static_cast<std::size_t>(numChannels);
274 header.resize(count * bytesPerSignal);
275 if (m_file->read(header.data(), header.size()) != header.size()) {
276 return false;
277 }
278 cursor = 0;
279 m_allChannels.assign(count, EDFChannelInfo{});
280 const auto column = [&](std::size_t width, auto assign) {
281 for (EDFChannelInfo& ch : m_allChannels) {
282 assign(ch, field(width));
283 }
284 };
285 column(16, [](EDFChannelInfo& ch, std::string_view v) { ch.label = headerText(v); });
286 column(80, [](EDFChannelInfo& ch, std::string_view v) { ch.transducerType = headerText(v); });
287 column(8, [](EDFChannelInfo& ch, std::string_view v) { ch.physicalDimension = headerText(v); });
288 column(8, [](EDFChannelInfo& ch, std::string_view v) { ch.physicalMin = static_cast<float>(mne::to_double(v).value_or(0.0)); });
289 column(8, [](EDFChannelInfo& ch, std::string_view v) { ch.physicalMax = static_cast<float>(mne::to_double(v).value_or(0.0)); });
290 column(8, [](EDFChannelInfo& ch, std::string_view v) { ch.digitalMin = mne::to_int(v).value_or(0); });
291 column(8, [](EDFChannelInfo& ch, std::string_view v) { ch.digitalMax = mne::to_int(v).value_or(0); });
292 column(80, [](EDFChannelInfo& ch, std::string_view v) { ch.prefiltering = headerText(v); });
293 column(8, [](EDFChannelInfo& ch, std::string_view v) { ch.samplesPerRecord = mne::to_int(v).value_or(0); });
294
295 if (m_numBytesInHeader != static_cast<long>(fixedBytes + count * bytesPerSignal)) {
296 mne::Log::warning("[EDFReader::parse_header] Header byte count mismatch: read {} expected {}", fixedBytes + count * bytesPerSignal, m_numBytesInHeader);
297 }
298
299 m_numBytesPerDataRecord = 0;
300 long maxSamplesPerRecord = 0;
301 for (std::size_t i = 0; i < count; ++i) {
302 EDFChannelInfo& ch = m_allChannels[i];
303 ch.channelNumber = static_cast<int>(i);
304 ch.sampleCount = ch.samplesPerRecord * m_numDataRecords;
305 ch.frequency = recordDuration > 0.0f ? static_cast<float>(ch.samplesPerRecord) / recordDuration : 0.0f;
306 m_numBytesPerDataRecord += ch.samplesPerRecord * m_bytesPerSample;
307 maxSamplesPerRecord = std::max(maxSamplesPerRecord, ch.samplesPerRecord);
308 }
309
310 // Measurement channels are those with the highest sample rate
311 m_measChannels.clear();
312 for (EDFChannelInfo& ch : m_allChannels) {
313 ch.isMeasurement = ch.samplesPerRecord == maxSamplesPerRecord;
314 if (ch.isMeasurement) {
315 m_measChannels.push_back(ch);
316 }
317 }
318 return true;
319}

342MatrixXf EDFReader::read_raw_segment(long first, long last) const
343{
344 if (!m_file) {
345 mne::Log::warning("[EDFReader::read_raw_segment] File not open");
346 return MatrixXf();
347 }
348
349 const long total = sample_count();
350 if (first < 0 || first >= total || last < 0 || last > total) {
351 mne::Log::warning("[EDFReader::read_raw_segment] Index out of bounds: {} - {}", first, last);
352 return MatrixXf();
353 }
354 const long numSamples = last - first;
355 const long perRecord = m_measChannels.front().samplesPerRecord;
356 if (numSamples <= 0 || perRecord <= 0) {
357 return MatrixXf();
358 }
359
360 // Read the data records that hold the segment in one go
361 const long firstRecord = first / perRecord;
362 const long offsetInRecord = first % perRecord;
363 const long numRecords = (offsetInRecord + numSamples + perRecord - 1) / perRecord;
364 const auto recordBytes = static_cast<std::size_t>(m_numBytesPerDataRecord);
365 std::vector<unsigned char> records(static_cast<std::size_t>(numRecords) * recordBytes);
366 m_file->seek(static_cast<std::uint64_t>(m_numBytesInHeader) + static_cast<std::uint64_t>(firstRecord) * recordBytes);
367 if (m_file->read(records.data(), records.size()) != records.size()) {
368 mne::Log::warning("[EDFReader::read_raw_segment] File ends before sample {}", last);
369 }
370
371 // Channels are interleaved per record; demultiplex and scale the measurement channels
372 MatrixXf result(static_cast<Index>(m_measChannels.size()), numSamples);
373 Index row = 0;
374 long channelOffset = 0;
375 for (const EDFChannelInfo& ch : m_allChannels) {
376 if (ch.isMeasurement) {
377 const double digRange = static_cast<double>(ch.digitalMax - ch.digitalMin);
378 const double physRange = static_cast<double>(ch.physicalMax) - ch.physicalMin;
379 const double si = ch.to_si();
380 const bool stim = isAutoStimLabel(ch.label);
381 for (long s = 0; s < numSamples; ++s) {
382 const long sample = offsetInRecord + s;
383 const unsigned char* bytes = records.data() + static_cast<std::size_t>(sample / perRecord) * recordBytes + static_cast<std::size_t>((channelOffset + sample % perRecord) * m_bytesPerSample);
384 // Little-endian signed integers: 16 bit (EDF) or 24 bit (BDF)
385 int raw = bytes[0] | (bytes[1] << 8);
386 if (m_bytesPerSample == 3) {
387 raw |= bytes[2] << 16;
388 raw = (raw & 0x800000) ? raw - 0x1000000 : raw;
389 } else {
390 raw = static_cast<std::int16_t>(raw);
391 }
392 const double physical = (raw - ch.digitalMin) / digRange * physRange + ch.physicalMin;
393 // mne read_raw_bdf keeps the raw stim value; both readers keep its low 17 bits
394 if (stim) {
395 result(row, s) = static_cast<float>(static_cast<int>(m_bytesPerSample == 3 ? raw : physical) & 0x1FFFF);
396 } else {
397 result(row, s) = static_cast<float>(physical * si);
398 }
399 }
400 ++row;
401 }
402 channelOffset += ch.samplesPerRecord;
403 }
404 return result;
405}
