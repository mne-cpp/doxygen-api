155 //=========================================================================================================
156 // Optical path calibration — determine the microscope optical axis
157 // relative to the tracker sensor.
158 //
159 // Workflow:
160 // 1. Move the OPMI to a position, aim at a point on the head surface,
161 // mark the focus point with the stylus pen → captureOpticalCalibSample().
162 // 2. Repeat at 2–3 different positions / focal distances.
163 // 3. Call solveOpticalCalibration() to fit a 3D line through the focus
164 // points in the tracker's local frame, yielding the optical axis
165 // direction and the optical center offset from the tracker.
166 //=========================================================================================================
167
168 struct OpticalCalibSample {
169 QVector3D trackerPos;
170 QQuaternion trackerOri;
171 QVector3D focusPoint;
172 };
