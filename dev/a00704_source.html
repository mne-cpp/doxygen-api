153 //=========================================================================================================
154 // Optical path calibration — determine the microscope optical axis
155 // relative to the tracker sensor.
156 //
157 // Workflow:
158 // 1. Move the OPMI to a position, aim at a point on the head surface,
159 // mark the focus point with the stylus pen → captureOpticalCalibSample().
160 // 2. Repeat at 2–3 different positions / focal distances.
161 // 3. Call solveOpticalCalibration() to fit a 3D line through the focus
162 // points in the tracker's local frame, yielding the optical axis
163 // direction and the optical center offset from the tracker.
164 //=========================================================================================================
165

168 QQuaternion trackerOri;
