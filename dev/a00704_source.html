174 //=========================================================================================================
175 // Optical path calibration — determine the microscope optical axis
176 // relative to the tracker sensor.
177 //
178 // Workflow:
179 // 1. Move the OPMI to a position, aim at a point on the head surface,
180 // mark the focus point with the stylus pen → captureOpticalCalibSample().
181 // 2. Repeat at 2–3 different positions / focal distances.
182 // 3. Call solveOpticalCalibration() to fit a 3D line through the focus
183 // points in the tracker's local frame, yielding the optical axis
184 // direction and the optical center offset from the tracker.
185 //=========================================================================================================
186
187 struct OpticalCalibSample {
188 QVector3D trackerPos;
189 QQuaternion trackerOri;
190 QVector3D focusPoint;
191 };
