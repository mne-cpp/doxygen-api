116VectorXd InvResolutionMatrix::spatialSpread(const MatrixXd& matResolution,
117 const MatrixX3d& matPositions)
118{
119 const int nSrc = static_cast<int>(matResolution.rows());
120 VectorXd spread(nSrc);
121
122 for (int s = 0; s < nSrc; ++s) {
123 // PSF = column s (or row s for CTF-based; here we use PSF convention)
124 VectorXd psf = matResolution.col(s).cwiseAbs2();
125 double psfSum = psf.sum();
126 if (psfSum < 1e-30) {
127 spread[s] = 0.0;
128 continue;
129 }
130
131 // Normalise to distribution
132 psf /= psfSum;
133
134 // Compute weighted mean distance from source s
135 double meanDist = 0.0;
136 double meanDist2 = 0.0;
137 for (int j = 0; j < nSrc; ++j) {
138 double dist = (matPositions.row(j) - matPositions.row(s)).norm();
139 meanDist += psf[j] * dist;
140 meanDist2 += psf[j] * dist * dist;
141 }
142
143 // Standard deviation of the distance distribution
144 spread[s] = std::sqrt(std::max(0.0, meanDist2 - meanDist * meanDist));
145 }
146
147 return spread;
148}

152VectorXd InvResolutionMatrix::peakLocalisationError(const MatrixXd& matResolution,
153 const MatrixX3d& matPositions)
154{
155 const int nSrc = static_cast<int>(matResolution.rows());
156 VectorXd ple(nSrc);
157
158 for (int s = 0; s < nSrc; ++s) {
159 // Find peak of PSF (column s)
160 VectorXd psf = matResolution.col(s).cwiseAbs();
161 Index peakIdx = 0;
162 psf.maxCoeff(&peakIdx);
163
164 ple[s] = (matPositions.row(peakIdx) - matPositions.row(s)).norm();
165 }
166
167 return ple;
168}
