98QVBoxLayout* QuickControlView::findTabWidgetLayout(const QString& sTabName)
99{
100 QVBoxLayout* pTabWidgetLayout = Q_NULLPTR;
101
102 if(QWidget* pTabWidget = m_pUi->m_pTabWidget->findChild<QWidget *>(sTabName + "TabWidget")) {
103 // Tab widget already exisits. Get the grid layout and return it.
104 pTabWidgetLayout = qobject_cast<QVBoxLayout *>(pTabWidget->layout());
105 } else {
106 // Tab widget does not exist yet. Create it and return grid lyout.
107 pTabWidget = new QWidget();
108 pTabWidget->setObjectName(sTabName + "TabWidget");
109 pTabWidgetLayout = new QVBoxLayout();
110 pTabWidgetLayout->setContentsMargins(4,2,4,4);
111 pTabWidget->setLayout(pTabWidgetLayout);
112 m_pUi->m_pTabWidget->insertTab(0,pTabWidget, sTabName);
113 }
114
115 return pTabWidgetLayout;
116}

120void QuickControlView::addWidgets(const QList<QWidget*>& lWidgets,
121 const QString& sTabName,
122 bool bAddToEnd)
123{
124 for(int i = 0; i < lWidgets.size(); ++i) {
125 QString sObjectName = lWidgets.at(i)->objectName();
126
127 if(sObjectName.contains("widget_", Qt::CaseInsensitive)) {
128 this->addWidget(lWidgets.at(i), sTabName, bAddToEnd);
129 }
130
131 if(sObjectName.contains("group_", Qt::CaseInsensitive)) {
132 if(sObjectName.contains("group_tab_", Qt::CaseInsensitive)) {
133 sObjectName.remove("group_tab_");
134 QStringList sList = sObjectName.split("_");
135
136 if(sList.size() >= 2) {
137 this->addGroupBoxWithTabs(lWidgets.at(i), sList.at(0), sList.at(1), sTabName, bAddToEnd);
138 } else {
139 this->addGroupBoxWithTabs(lWidgets.at(i), "", sObjectName, sTabName, bAddToEnd);
140 }
141 } else {
142 sObjectName.remove("group_");
143 this->addGroupBox(lWidgets.at(i), sObjectName, sTabName, bAddToEnd);
144 }
145 }
146 }
147}

189void QuickControlView::addGroupBoxWithTabs(QWidget* pWidget,
190 const QString& sGroupBoxName,
191 const QString& sTabNameGroupBox,
192 const QString& sTabName,
193 bool bAddToEnd)
194{
195 if(QVBoxLayout* pTabWidgetLayout = findTabWidgetLayout(sTabName)) {
196 QGroupBox* pGroupBox = pTabWidgetLayout->parentWidget()->findChild<QGroupBox *>(sGroupBoxName);
197
198 if(!pGroupBox) {
199 int iPos = bAddToEnd ? -1 : 0;
200 pGroupBox = new QGroupBox(sGroupBoxName);
201 pGroupBox->setObjectName(sGroupBoxName);
202
203 pTabWidgetLayout->insertWidget(iPos,
204 pGroupBox,
205 0);
206
207 QVBoxLayout *pVBox = new QVBoxLayout;
208 QTabWidget* pTabWidget = new QTabWidget;
209 pTabWidget->setTabBarAutoHide(false);
210 pTabWidget->setMovable(true);
211 pTabWidget->setObjectName(sGroupBoxName + "TabWidget");
212
213 pTabWidget->addTab(pWidget, sTabNameGroupBox);
214 pVBox->setContentsMargins(4,2,4,4);
215 pVBox->addWidget(pTabWidget);
216 pGroupBox->setLayout(pVBox);
217 } else {
218 QTabWidget* pTabWidget = pGroupBox->findChild<QTabWidget *>(sGroupBoxName + "TabWidget");
219
220 if(pTabWidget) {
221 pTabWidget->addTab(pWidget, sTabNameGroupBox);
222 }
223 }
224 }
