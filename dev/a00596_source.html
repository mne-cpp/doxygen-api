38RtSourceDataController::RtSourceDataController(QObject* parent)
39: QObject(parent)
40{
41 // Create data worker
42 m_pWorker = new mne::disp::RtSourceDataWorker();
43
44 // Create and configure data thread
45 m_pWorkerThread = new QThread(this);
46 m_pWorker->moveToThread(m_pWorkerThread);
47
48 // Clean up worker when thread finishes
49 connect(m_pWorkerThread, &QThread::finished, m_pWorker, &QObject::deleteLater);
50
51 // Forward worker signals to controller signals
52 connect(m_pWorker, &mne::disp::RtSourceDataWorker::newRtSmoothedData,
53 this, &RtSourceDataController::newSmoothedDataAvailable);
54 connect(m_pWorker, &mne::disp::RtSourceDataWorker::newRtRawData,
55 this, &RtSourceDataController::newRawDataAvailable);
56
57 // Create timer for driving the streaming cadence
58 m_pTimer = new QTimer(this);
59 m_pTimer->setTimerType(Qt::PreciseTimer);
60 connect(m_pTimer, &QTimer::timeout, m_pWorker, &mne::disp::RtSourceDataWorker::streamData);
61
62 // Start the data worker thread
63 m_pWorkerThread->start();
64
65 // Create interpolation matrix worker
66 m_pInterpWorker = new mne::disp::RtSourceInterpolationMatWorker();
67 m_pInterpThread = new QThread(this);
68 m_pInterpWorker->moveToThread(m_pInterpThread);
69
70 connect(m_pInterpThread, &QThread::finished, m_pInterpWorker, &QObject::deleteLater);
71
72 // Forward interpolation results: auto-apply to data worker + re-emit
73 connect(m_pInterpWorker, &mne::disp::RtSourceInterpolationMatWorker::newInterpolationMatrixLeftAvailable,
74 this, &RtSourceDataController::onNewInterpolationMatrixLeft);
75 connect(m_pInterpWorker, &mne::disp::RtSourceInterpolationMatWorker::newInterpolationMatrixRightAvailable,
76 this, &RtSourceDataController::onNewInterpolationMatrixRight);
77
78 m_pInterpThread->start();
79
80 qDebug() << "RtSourceDataController: Initialized with" << m_iTimeInterval << "ms interval";
81}
