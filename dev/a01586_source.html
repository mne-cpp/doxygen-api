76void BarChart::draw(Canvas& canvas, const TextMetrics& metrics, const Font& font) const
77{
78 const double w = canvas.width();
79 const double h = canvas.height();
80 const double marginLeft = 60;
81 const double marginRight = 20;
82 const double marginTop = 40;
83 const double marginBottom = 100;
84 const double plotW = w - marginLeft - marginRight;
85 const double plotH = h - marginTop - marginBottom;
86 if (plotW <= 0 || plotH <= 0) {
87 return;
88 }
89 canvas.save();
90 canvas.set_antialiasing(true);
91 canvas.set_font(font);
92
93 if (!m_title.empty()) {
94 canvas.set_pen(Color::black());
95 Font titleFont = resized_font(font, 2, metrics);
96 titleFont.bold = true;
97 canvas.set_font(titleFont);
98 canvas.draw_text(Rect{0, 0, w, marginTop}, Align::Center, m_title);
99 canvas.set_font(font);
100 }
101
102 if (m_counts.empty()) {
103 canvas.restore();
104 return;
105 }
106
107 const int nBars = static_cast<int>(m_counts.size());
108 const int maxCount = std::max(1, *std::max_element(m_counts.begin(), m_counts.end()));
109
110 // Axes
111 canvas.set_pen(Pen{Color::black(), 1});
112 canvas.draw_line({marginLeft, marginTop}, {marginLeft, marginTop + plotH});
113 canvas.draw_line({marginLeft, marginTop + plotH}, {marginLeft + plotW, marginTop + plotH});
114
115 for (double value : nice_ticks(0.0, maxCount, 5, 1.0)) {
116 const double y = marginTop + plotH - plotH * value / maxCount;
117 canvas.draw_line({marginLeft - 5, y}, {marginLeft, y});
118 canvas.draw_text(Rect{0, y - 10, marginLeft - 8, 20}, Align::Right | Align::VCenter, tick_label(value));
119 }
120
121 const double barWidth = plotW / nBars;
122 const double barPadding = barWidth * 0.1;
123 for (int i = 0; i < nBars; ++i) {
124 const double barH = (static_cast<double>(m_counts[static_cast<std::size_t>(i)]) / maxCount) * plotH;
125 canvas.set_brush(Brush::solid({70, 130, 180}));
126 canvas.set_pen(Pen{{50, 100, 150}, 1});
127 canvas.draw_rect({marginLeft + i * barWidth + barPadding, marginTop + plotH - barH, barWidth - 2.0 * barPadding, barH});
128 }
129
130 // Category labels, rotated when wider than their bar
131 canvas.set_pen(Color::black());
132 const Font small = resized_font(font, -1, metrics);
133 canvas.set_font(small);
134 for (int i = 0; i < nBars && i < static_cast<int>(m_categories.size()); ++i) {
135 const std::string& category = m_categories[static_cast<std::size_t>(i)];
136 const double cx = marginLeft + (i + 0.5) * barWidth;
137 if (metrics.advance(small, category) < barWidth - 4) {
138 canvas.draw_text(Rect{cx - barWidth / 2.0, marginTop + plotH + 6, barWidth, 20}, Align::HCenter | Align::Top, category);
139 continue;
140 }
141 canvas.save();
142 canvas.translate(cx, marginTop + plotH + 5);
143 canvas.rotate(45);
144 canvas.draw_text(Point{0, 0}, category);
145 canvas.restore();
146 }
147
148 canvas.set_font(font);
149 if (!m_xLabel.empty()) {
150 canvas.draw_text(Rect{marginLeft, h - 22, plotW, 20}, Align::Center, m_xLabel);
151 }
152 if (!m_yLabel.empty()) {
153 canvas.save();
154 canvas.translate(4, marginTop + plotH / 2.0);
155 canvas.rotate(-90);
156 canvas.draw_text(Rect{-plotH / 2.0, 0, plotH, 20}, Align::Center, m_yLabel);
157 canvas.restore();
158 }
159 canvas.restore();
160}
