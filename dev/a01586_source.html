142std::array<double, 2> gaussianPair(std::uint64_t h) noexcept
143{
144 const double u1 = (static_cast<double>(h >> 32U) + 0.5) * 0x1.0p-32;
145 const double u2 = (static_cast<double>(h & 0xFFFFFFFFULL) + 0.5) * 0x1.0p-32;
146 const double r = std::sqrt(-2.0 * std::log(u1));
147 return {r * std::cos(2.0 * kPi * u2), r * std::sin(2.0 * kPi * u2)};

151class Hrf
152{
153public:
154 Hrf()
155 {
156 for (int i = 1; i < 3200; ++i) {
157 m_peak = std::max(m_peak, raw(0.01 * i));
158 }
159 }
160
161 [[nodiscard]] double at(double t) const noexcept
162 {
163 return t <= 0.0 ? 0.0 : raw(t) / m_peak;
164 }
165
166private:
167 static double raw(double t) noexcept
168 {
169 constexpr double kGamma6 = 120.0;
170 constexpr double kGamma16 = 1307674368000.0;
171 return std::pow(t, 5.0) * std::exp(-t) / kGamma6 - std::pow(t, 15.0) * std::exp(-t) / (6.0 * kGamma16);
172 }

174 double m_peak = 0.0;

223AcqExpected<Config> parseConfig(const AcqUri& uri)
224{
225 Config c;
226 if (!uri.target.empty()) {
227 return invalid("target '" + uri.target + "': synthetic:// takes no target, only parameters");
228 }
229 const auto eeg = paramInt(uri, "eeg", 8, 0, 4096);
230 const auto meg = paramInt(uri, "meg", 0, 0, 4096);
231 const auto stim = paramBool(uri, "stim", true);
232 const auto sfreq = paramDouble(uri, "sfreq", 1000.0, 0.001, 1e6);
233 const auto block = paramInt(uri, "block", 0, 1, 1 << 20);
234 const auto realtime = paramBool(uri, "realtime", true);
235 const auto speed = paramDouble(uri, "speed", 1.0, 0.01, 1e4);
236 const auto duration = paramDouble(uri, "duration", 0.0, 0.0, 1e7);
237 const auto seed = paramInt(uri, "seed", 0, 0, INT64_MAX);
238 const auto noise = paramDouble(uri, "noise", 1.0, 0.0, 1e3);
239 const auto alpha = paramDouble(uri, "alpha", 10.0, 0.0, 5e5);
240 const auto interval = paramDouble(uri, "interval", 1.0, 0.0, 3600.0);
241 const auto targets = paramDouble(uri, "targets", 0.2, 0.0, 1.0);
242 const auto evoked = paramDouble(uri, "evoked", 1.0, 0.0, 1e3);
243 const auto fnirs = paramInt(uri, "fnirs", 0, 0, 1024);
244 const auto ecog = paramInt(uri, "ecog", 0, 0, 4096);
245 const auto erg = paramInt(uri, "erg", 0, 0, 64);
246 const auto fnirsSfreq = paramDouble(uri, "fnirs_sfreq", 10.0, 3.0, 1e3);
247 const auto ecogSfreq = paramDouble(uri, "ecog_sfreq", 2000.0, 300.0, 1e6);
248 const auto ergSfreq = paramDouble(uri, "erg_sfreq", 2000.0, 300.0, 1e6);
249 auto thread = paramThreadPolicy(uri);
250 for (const AcqError& error :
251 {eeg.error(), meg.error(), stim.error(), sfreq.error(), block.error(), realtime.error(), speed.error(),
252 duration.error(), seed.error(), noise.error(), alpha.error(), interval.error(), targets.error(),
253 evoked.error(), fnirs.error(), ecog.error(), erg.error(), fnirsSfreq.error(), ecogSfreq.error(),
254 ergSfreq.error(), thread.error()}) {
255 if (!error.ok()) {
256 return error;
257 }
258 }
259 c.thread = std::move(*thread);
260 c.fnirs = static_cast<std::size_t>(*fnirs);
261 c.ecog = static_cast<std::size_t>(*ecog);
262 c.erg = static_cast<std::size_t>(*erg);
263 c.fnirsSfreq = *fnirsSfreq;
264 c.ecogSfreq = *ecogSfreq;
265 c.ergSfreq = *ergSfreq;
266 c.eeg = static_cast<std::size_t>(*eeg);
267 c.meg = static_cast<std::size_t>(*meg);
268 c.stim = *stim;
269 c.sfreq = *sfreq;
270 c.block = *block > 0 ? static_cast<std::size_t>(*block) : static_cast<std::size_t>(std::max(1.0, std::round(c.sfreq / 100.0)));
271 c.realtime = *realtime;
272 c.speed = *speed;
273 c.totalSamples = static_cast<std::uint64_t>(std::llround(*duration * c.sfreq));
274 c.seed = static_cast<std::uint64_t>(*seed);
275 c.noise = *noise;
276 c.alpha = *alpha;
277 c.interval = *interval;
278 c.targets = *targets;
279 c.evoked = *evoked;
280 c.pulse = static_cast<std::uint64_t>(std::max(1.0, std::ceil(kPulseSeconds * c.sfreq - 1e-9)));
281
282 if (c.eeg + c.meg == 0 && !c.stim) {
283 return invalid("eeg=0, meg=0 and stim=0 leave synthetic:// without channels");
284 }
285 if (c.alpha >= c.sfreq / 2.0) {
286 return invalid("alpha=" + std::to_string(c.alpha) + " must stay below the Nyquist frequency sfreq/2");
287 }
288 if (c.interval > 0.0 && c.interval * c.sfreq < 2.0 * static_cast<double>(c.pulse)) {
289 return invalid("interval=" + std::to_string(c.interval) + " is shorter than two 5 ms trigger pulses");
290 }
291 // One clock domain: every rate a whole number of Hz, the domain tick their least common multiple.
292 const std::pair<const char*, double> rates[] = {{"sfreq", c.sfreq},
293 {"fnirs_sfreq", c.fnirs > 0 ? c.fnirsSfreq : 0.0},
294 {"ecog_sfreq", c.ecog > 0 ? c.ecogSfreq : 0.0},
295 {"erg_sfreq", c.erg > 0 ? c.ergSfreq : 0.0}};
296 const bool extra = c.fnirs + c.ecog + c.erg > 0;
297 std::uint64_t ticks = 1;
298 for (const auto& [name, rate] : rates) {
299 if (rate == 0.0) {
300 continue;
301 }
302 if (!wholeHz(rate)) {
303 if (extra) {
304 return invalid(std::string(name) + "=" + std::to_string(rate) +
305 " must be a whole number of Hz for the streams to share the device clock");
306 }
307 return c; // a single stream at a fractional rate: no exact clock domain
308 }
309 ticks = std::lcm(ticks, static_cast<std::uint64_t>(rate));
310 }
311 c.ticksPerSecond = ticks;
312 return c;
313}

374 void trials(std::uint64_t first, std::uint64_t last, std::uint64_t window, std::vector<Trial>& out) const
375 {
376 out.clear();
377 if (m_c.interval <= 0.0) {
378 return;
379 }
380 const double perTrial = m_c.interval * m_c.sfreq;
381 const std::uint64_t low = first > window ? first - window : 0;
382 const double kLow = std::floor(static_cast<double>(low) / perTrial) - 2.0;
383 const auto kMin = static_cast<std::uint64_t>(std::max(0.0, kLow));
384 const auto kMax = static_cast<std::uint64_t>(static_cast<double>(last) / perTrial);
385 for (std::uint64_t k = kMin; k <= kMax; ++k) {
386 const Trial t = trial(k);
387 if (t.onset <= last && t.onset + window > first) {
388 out.push_back(t);
389 }
390 }

399 [[nodiscard]] double drift(double t) const noexcept
400 {
401 const double knot = t / kDriftKnotSeconds;
402 const double k = std::floor(knot);
403 const double w = 0.5 - 0.5 * std::cos(kPi * (knot - k));
404 const auto index = static_cast<std::uint64_t>(k);
405 const double a = 2.0 * uniform(hash(m_c.seed, DriftStream, index)) - 1.0;
406 const double b = 2.0 * uniform(hash(m_c.seed, DriftStream, index + 1)) - 1.0;
407 return kDriftRange * (a + (b - a) * w);
408 }

411 void frame(std::uint64_t s, const std::vector<Trial>& trials, float* out) const noexcept
412 {
413 double erp = 0.0;
414 float trigger = 0.0F;
415 for (const Trial& t : trials) {
416 if (s < t.onset || s - t.onset >= m_window) {
417 continue;
418 }
419 const double latency = static_cast<double>(s - t.onset) / m_c.sfreq;
420 erp += kN100.at(latency) + kP200.at(latency) + (t.code == kTarget ? kP300.at(latency) : 0.0);
421 if (s - t.onset < m_c.pulse) {
422 trigger = static_cast<float>(t.code);
423 }
424 }
425 erp *= m_c.evoked;
426
427 const double cycles = m_c.alpha * static_cast<double>(s) / m_c.sfreq;
428 const double alphaPhase = 2.0 * kPi * (cycles - std::floor(cycles)) + drift(static_cast<double>(s) / m_c.sfreq);
429 for (std::size_t c = 0; c < m_analog; c += 2) {
430 double z[2] = {0.0, 0.0};
431 if (m_c.noise > 0.0) {
432 const std::uint64_t h = hash(m_c.seed, NoiseStream, s, c / 2);
433 const double u1 = (static_cast<double>(h >> 32U) + 0.5) * 0x1.0p-32;
434 const double u2 = (static_cast<double>(h & 0xFFFFFFFFULL) + 0.5) * 0x1.0p-32;
435 const double r = std::sqrt(-2.0 * std::log(u1));
436 z[0] = r * std::cos(2.0 * kPi * u2);
437 z[1] = r * std::sin(2.0 * kPi * u2);
438 }
439 for (std::size_t j = 0; j < 2 && c + j < m_analog; ++j) {
440 const std::size_t ch = c + j;
441 double v = m_c.noise * kNoiseRms * z[j] + m_weight[ch] * erp;
442 if (m_c.alpha > 0.0) {
443 v += kAlphaAmplitude * std::sin(alphaPhase + m_phase[ch]);
444 }
445 out[ch] = static_cast<float>(v * m_unit[ch]);
446 }
447 }
448 if (m_c.stim) {
449 out[m_analog] = trigger;
450 }
451 }
452
453private:
454 Config m_c;
455 std::size_t m_analog;
456 std::uint64_t m_window;
457 std::vector<double> m_weight;
458 std::vector<double> m_phase;
459 std::vector<double> m_unit;
460};

543 void frame(std::uint64_t s, const std::vector<Trial>& trials, float* out) const noexcept
544 {
545 const auto tick = static_cast<std::int64_t>(s * m_ticksPerSample);
546 const double t = static_cast<double>(tick) / static_cast<double>(m_c.ticksPerSecond);
547 double slow = 0.0; // ERP (ECoG), haemodynamic response (fNIRS), flash ERG
548 double fast = 0.0; // high-gamma envelope (ECoG)
549 for (const Trial& trial : trials) {
550 const std::int64_t delta = tick - static_cast<std::int64_t>(trial.onset * m_baseTicksPerSample);
551 const double latency = static_cast<double>(delta) / static_cast<double>(m_c.ticksPerSecond);
552 if (latency < 0.0) {
553 continue;
554 }
555 const double gain = trial.code == kTarget ? 1.5 : 1.0;
556 switch (m_modality) {
557 case Modality::Fnirs:
558 slow += latency < kFnirsWindowSeconds ? gain * m_hrf.at(latency) : 0.0;
559 break;
560 case Modality::Ecog:
561 if (latency < kEcogWindowSeconds) {
562 slow += kN100.at(latency) + kP200.at(latency) + (trial.code == kTarget ? kP300.at(latency) : 0.0);
563 fast += gain * kHighGamma.at(latency);
564 }
565 break;
566 case Modality::Erg:
567 if (latency < kErgWindowSeconds) {
568 const double op = (latency - 0.030) / 0.008;
569 slow += kAWave.at(latency) + kBWave.at(latency) +
570 12.0 * std::sin(2.0 * kPi * kOscillatoryPotentialHz * latency) * std::exp(-0.5 * op * op);
571 }
572 break;
573 }
574 }
575 slow *= m_c.evoked;
576 fast *= m_c.evoked;
577
578 if (m_modality == Modality::Fnirs) {
579 const std::size_t g = 4 * m_count;
580 const double mayer = m_c.noise * 0.2 * std::sin(2.0 * kPi * 0.1 * t + m_phase[g]);
581 const double breath = m_c.noise * 0.1 * std::sin(2.0 * kPi * 0.25 * t + m_phase[g + 1]);
582 const double pulse = m_c.noise * 0.15 * std::sin(2.0 * kPi * 1.0 * t + m_phase[g + 2]);
583 for (std::size_t i = 0; i < m_count; ++i) {
584 const auto z = gaussianPair(hash(m_c.seed, FnirsNoiseStream, s, i));
585 const double activation = 0.4 * m_weight[i] * slow;
586 const double hbo = activation + mayer + breath + pulse + m_c.noise * 0.03 * z[0];
587 const double hbr = -0.3 * activation - 0.3 * mayer - 0.1 * breath + 0.1 * pulse + m_c.noise * 0.03 * z[1];
588 out[2 * i] = static_cast<float>(hbo * kFnirsUnit);
589 out[2 * i + 1] = static_cast<float>(hbr * kFnirsUnit);
590 }
591 return;
592 }
593 const Stream noiseStream = m_modality == Modality::Ecog ? EcogNoiseStream : ErgNoiseStream;
594 const double noiseRms = m_modality == Modality::Ecog ? 25.0 : 5.0;
595 for (std::size_t c = 0; c < m_count; c += 2) {
596 const auto z = m_c.noise > 0.0 ? gaussianPair(hash(m_c.seed, noiseStream, s, c / 2)) : std::array<double, 2>{};
597 for (std::size_t j = 0; j < 2 && c + j < m_count; ++j) {
598 const std::size_t ch = c + j;
599 double v = m_c.noise * noiseRms * z[j];
600 if (m_modality == Modality::Ecog) {
601 double burst = 0.0;
602 for (std::size_t f = 0; f < 3; ++f) {
603 burst += std::sin(2.0 * kPi * kHighGammaHz[f] * t + m_phase[3 * ch + f]);
604 }
605 v += 4.0 * m_weight[ch] * slow + m_focal[ch] * fast * burst / 3.0;
606 } else {
607 v += m_weight[ch] * slow;
608 }
609 out[ch] = static_cast<float>(v * kEegUnit);
610 }
611 }
612 }

718 return config.error();
719 }
720 std::unique_ptr<SyntheticBackend> backend(new SyntheticBackend(*config, uri.toString()));
721 AcqStreamOptions options;

723 auto writer = backend->addInputStream(streamInfo(*config), options);
724 if (!writer) {

727 backend->m_writer = *writer;

731 continue;

857 static void plan(ModalityStream& stream, std::uint64_t end)
858 {
859 std::uint64_t until = stream.model.sampleAtOrAfter(end);
860 if (stream.total > 0) {
861 until = std::min(until, stream.total);
862 }
863 stream.pending = until > stream.next ? static_cast<std::size_t>(until - stream.next) : 0;
864 }
865
870 bool waitForSpace(std::size_t n)
871 {
872 constexpr std::chrono::milliseconds kPoll(20);
873 while (!m_stop.load(std::memory_order_acquire)) {
874 AcqError e = m_writer->waitForSpace(n, kPoll);
875 for (ModalityStream& stream : m_streams) {
876 if (e.ok()) {
877 e = stream.writer->waitForSpace(stream.pending, kPoll);
878 }
879 }
880 if (e.ok()) {
881 return true;
882 }
883 if (e.code() != AcqErrorCode::Timeout) {
884 reportError(e);
885 return false;
886 }
887 }
888 return false;
889 }
890
892 bool generate(ModalityStream& stream)
893 {
894 if (stream.pending == 0) {
895 return true;
896 }
897 const std::uint64_t until = stream.next + stream.pending;
898 auto view = stream.writer->beginWrite(stream.pending);
899 if (!view) {
900 reportError(view.error());
901 return false;
902 }
903 const auto [first, last] = stream.model.baseRange(stream.next, until - 1);
904 m_model.trials(first, last, stream.model.window(), stream.trials);
905 std::uint64_t s = stream.next;
906 for (std::size_t p = 0; p < view->partCount(); ++p) {
907 const AcqWritePart& part = view->part(p);
908 for (std::size_t i = 0; i < part.samples; ++i, ++s) {
909 stream.model.frame(s, stream.trials, stream.frame.data());
910 std::memcpy(part.at(0, i), stream.frame.data(), stream.frame.size() * sizeof(float));
911 }
912 }
913 return true;
914 }
915
916 bool commit(ModalityStream& stream)
917 {
918 if (stream.pending == 0) {
919 return true;
920 }
921 const std::uint64_t end = stream.next + stream.pending;
922 for (const Trial& t : stream.trials) {
923 const std::uint64_t sample = stream.model.sampleAtOrAfter(t.onset);
924 if (sample >= stream.next && sample < end) {
925 const AcqError pushed =
926 stream.writer->pushEvent({sample, static_cast<std::uint32_t>(stream.model.pulse()), t.code});
927 if (!pushed.ok()) {
928 reportError(pushed);
929 return false;
930 }
931 }
932 }
933 stream.writer->commit(stream.pending);
934 stream.next = end;
935 return true;
936 }
937
938 void run()
939 {
940 const std::size_t channels = m_model.channels();
941 std::vector<float> frame(channels);
942 std::vector<Trial> trials;
943 trials.reserve(m_model.maxTrials(m_config.block));
944 const Clock::time_point start = Clock::now();
945 const std::uint64_t startSample = m_next;
946 while (true) {
947 std::size_t n = m_config.block;
948 if (m_config.totalSamples > 0) {
949 if (m_next >= m_config.totalSamples) {
950 m_writer->finish();
951 for (ModalityStream& stream : m_streams) {
952 stream.writer->finish();
953 }
954 return;
955 }
956 n = static_cast<std::size_t>(std::min<std::uint64_t>(n, m_config.totalSamples - m_next));
957 }
958 for (ModalityStream& stream : m_streams) {
959 plan(stream, m_next + n);
960 }
961 if (!waitForSpace(n)) {
962 return; // stopped; nothing reserved
963 }
964 auto view = m_writer->beginWrite(n);
965 if (!view) {
966 reportError(view.error());
967 return;
968 }
969 m_model.trials(m_next, m_next + n - 1, trials);
970 std::uint64_t s = m_next;
971 for (std::size_t p = 0; p < view->partCount(); ++p) {
972 const AcqWritePart& part = view->part(p);
973 for (std::size_t i = 0; i < part.samples; ++i, ++s) {
974 m_model.frame(s, trials, frame.data());
975 std::memcpy(part.at(0, i), frame.data(), channels * sizeof(float));
976 }
977 }
978 for (ModalityStream& stream : m_streams) {
979 if (!generate(stream)) {
980 return;
981 }
982 }
983
984 if (m_config.realtime) {
985 const std::chrono::duration<double> due(static_cast<double>(m_next + n - startSample) /
986 (m_config.sfreq * m_config.speed));
987 const auto deadline = start + std::chrono::duration_cast<Clock::duration>(due);
988 std::unique_lock<std::mutex> lock(m_mutex);
989 m_wake.wait_until(lock, deadline, [this] { return m_stop.load(std::memory_order_acquire); });
990 }
991 if (m_stop.load(std::memory_order_acquire)) {
992 return; // the block is dropped in every stream; a restart regenerates it
993 }
994
995 for (const Trial& t : trials) {
996 if (t.onset >= m_next && t.onset < m_next + n) {
997 const AcqError pushed = m_writer->pushEvent({t.onset, static_cast<std::uint32_t>(m_config.pulse), t.code});
998 if (!pushed.ok()) {
999 reportError(pushed);
1000 return;
1001 }
1002 }
1003 }
1004 m_writer->commit(n);
1005 m_next += n;
1006 for (ModalityStream& stream : m_streams) {
1007 if (!commit(stream)) {
1008 return;
1009 }
1010 }
1011 }
1012 }
1013
1014 Config m_config;
1015 Model m_model;
1016 std::string m_uri;
1017 AcqWriter* m_writer = nullptr;
1018 std::uint64_t m_next = 0; // next sample index; owned by the producer thread while it runs
1019 std::vector<ModalityStream> m_streams;
1020 std::thread m_thread;
1021 std::mutex m_mutex;
1022 std::condition_variable m_wake;
1023 std::atomic<bool> m_stop{false};
1024};
