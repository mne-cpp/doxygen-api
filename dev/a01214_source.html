62{
63 InvEcdSet set;

65 mne::Log::info("---- Setting up...\n");
66 std::unique_ptr<FwdEegSphereModel> eeg_model;
67 if (settings->include_eeg) {
68 eeg_model = FwdEegSphereModel::setup_eeg_sphere_model(mne::path_from_utf8(settings->eeg_model_file), settings->eeg_model_name, settings->eeg_sphere_rad);
69 if (!eeg_model)
70 return set;
71 }
72

74 settings->mriname,
75 settings->measname,
76 settings->bemname,
77 &settings->r0,
78 eeg_model.get(),
79 settings->accurate,
80 settings->badname,
81 settings->noisename,
82 settings->grad_std,
83 settings->mag_std,
84 settings->eeg_std,
85 settings->mag_reg,
86 settings->grad_reg,
87 settings->eeg_reg,
88 settings->diagnoise,
89 settings->projnames,
90 settings->include_meg,
91 settings->include_eeg));
92 if (!fit_data)
93 return set;
94
95 eeg_model.release(); // ownership transferred to fit_data->eeg_model
96
97 fit_data->fit_mag_dipoles = settings->fit_mag_dipoles;
98
99 std::unique_ptr<MNERawData> raw;
100 std::unique_ptr<MNEMeasData> data;
101 std::unique_ptr<MNEChSelection> sel;
102
103 if (settings->is_raw) {
104 mne::Log::info("\n---- Opening a raw data file...\n");
105 raw.reset(MNERawData::open_file(mne::path_from_utf8(settings->measname), true, false, settings->filter));
106 if (!raw)
107 return set;
108 /*
109 * Set up a channel selection to pick data from the raw file
110 */
111 sel = std::make_unique<MNEChSelection>(MNEChSelection::these(
112 "fit", std::span(fit_data->ch_names).first(static_cast<std::size_t>(fit_data->nmeg) + static_cast<std::size_t>(fit_data->neeg))));
113 sel->assign_channels(raw->info->chInfo, raw->deriv_matched.get());
114 for (int c = 0; c < sel->nchan(); c++)
115 if (sel->pick[c] < 0) {
116 mne::Log::critical("All desired channels were not available");
117 return set;
118 }
119 mne::Log::info("\tChannel selection created.");
120 /*
121 * Let's be a little generous here
122 */
123 float t1 = raw->first_samp / raw->info->sfreq;
124 float t2 = (raw->first_samp + raw->nsamp - 1) / raw->info->sfreq;
125 if (settings->tmin < t1 + settings->integ)
126 settings->tmin = t1 + settings->integ;
127 if (settings->tmax > t2 - settings->integ)
128 settings->tmax = t2 - settings->integ;
129 if (settings->tstep < 0)
130 settings->tstep = 1.0f / raw->info->sfreq;
131
132 mne::Log::info("\tOpened raw data file {} : {} MEG and {} EEG", settings->measname, fit_data->nmeg, fit_data->neeg);
133 } else {
134 mne::Log::info("\n---- Reading data...\n");
135 data = MNEMeasData::mne_read_meas_data(mne::path_from_utf8(settings->measname),
136 settings->setno,
137 nullptr,
138 nullptr,
139 fit_data->ch_names,
140 fit_data->nmeg + fit_data->neeg);
141 if (!data)
142 return set;
143 if (settings->do_baseline)
144 data->adjust_baselines(settings->bmin, settings->bmax);
145 else
146 mne::Log::info("\tNo baseline setting in effect.");
147 if (settings->tmin < data->current->tmin + settings->integ / 2.0f)
148 settings->tmin = data->current->tmin + settings->integ / 2.0f;
149 if (settings->tmax > data->current->tmin + (data->current->np - 1) * data->current->tstep - settings->integ / 2.0f)
150 settings->tmax = data->current->tmin + (data->current->np - 1) * data->current->tstep - settings->integ / 2.0f;
151 if (settings->tstep < 0)
152 settings->tstep = data->current->tstep;
153
154 mne::Log::info("\tRead data set {} from {} : {} MEG and {} EEG", settings->setno, settings->measname, fit_data->nmeg, fit_data->neeg);
155 if (!settings->noisename.empty()) {
156 mne::Log::info("Scaling the noise covariance...");
157 if (InvDipoleFitData::scale_noise_cov(fit_data.get(), data->current->nave) < 0)
158 return set;
159 }
160 }
161
162 /*
163 * Proceed to computing the fits
164 */
165 mne::Log::info("\n---- Computing the forward solution for the guesses...\n");
166 auto guess = std::make_unique<InvGuessData>(mne::path_from_utf8(settings->guessname),
167 mne::path_from_utf8(settings->guess_surfname),
168 settings->guess_mindist, settings->guess_exclude, settings->guess_grid, fit_data.get(), settings->guess_rad);
169 if (guess->nguess == 0) {
170 mne::Log::critical("Could not create the initial guesses.");
171 return set;
172 }
173
174 mne::Log::info("\n---- Fitting : {:7.1f} ... {:7.1f} ms (step: {:6.1f} ms integ: {:6.1f} ms)\n", 1000 * settings->tmin, 1000 * settings->tmax, 1000 * settings->tstep, 1000 * settings->integ);
175
176 if (raw) {
177 if (!fit_dipoles_raw(settings->measname, raw.get(), sel.get(), fit_data.get(), guess.get(), settings->tmin, settings->tmax, settings->tstep, settings->integ, settings->verbose, set))
178 return set;
179 } else {
180 if (!fit_dipoles(settings->measname, data.get(), fit_data.get(), guess.get(), settings->tmin, settings->tmax, settings->tstep, settings->integ, settings->verbose, set))
181 return set;
182 }
183 mne::Log::info("{} dipoles fitted", set.size());
184
185 return set;
186}

190bool InvDipoleFit::fit_dipoles(const std::string& dataname, MNEMeasData* data, InvDipoleFitData* fit, InvGuessData* guess, float tmin, float tmax, float tstep, float integ, int verbose, InvEcdSet& p_set)
191{
192 Eigen::VectorXf one(data->nchan);
193 InvEcdSet set;
194 InvEcd dip;
195 constexpr int report_interval = 10;
196
197 set.dataname = dataname;
198
199 if (verbose)
200 mne::Log::info("Fitting...");
201 for (int s = 0; tmin + s * tstep < tmax; s++) {
202 float time = tmin + s * tstep;
203 if (data->current->getValuesAtTime(time, integ, false, one) < 0) {
204 mne::Log::warning("Cannot pick time: {:7.1f} ms", 1000.0f * time);
205 continue;
206 }
207
208 if (!InvDipoleFitData::fit_one(fit, guess, time, one, verbose, dip))
209 mne::Log::warning("t = {:7.1f} ms : fit error", 1000.0f * time);
210 else {
211 set.push_back(dip);
212 if (verbose)
213 dip.print();
214 else {
215 if (set.size() % report_interval == 0)
216 mne::Log::info("{}..", set.size());
217 }
218 }
219 }
220 if (!verbose)
221 mne::Log::info("[done]");
222 p_set = set;
223 return true;
224}

228bool InvDipoleFit::fit_dipoles_raw(const std::string& dataname, MNERawData* raw, MNEChSelection* sel, InvDipoleFitData* fit, InvGuessData* guess, float tmin, float tmax, float tstep, float integ, int verbose, InvEcdSet& p_set)
229{
230 const int nchan = sel->nchan();
231 const float sfreq = raw->info->sfreq;
232 const float myinteg = integ > 0.0f ? 2 * integ : 0.1f;
233 const int overlap = static_cast<int>(std::ceil(myinteg * sfreq));
234 const int length = static_cast<int>(SEG_LEN * sfreq);
235 const int step = length - overlap;
236 const int stepo = step + overlap / 2;
237 int start = raw->first_samp;
238 constexpr int report_interval = 10;
239
240 Eigen::VectorXf one(nchan);
241
242 MNERawData::RowMajorMatrixXf segment(nchan, length);
243
244 InvEcd dip;
245 InvEcdSet set;
246 set.dataname = dataname;
247
248 /*
249 * Load the initial data segment
250 */
251 float stime = start / sfreq;
252 if (raw->pick_data_filt(sel, start, segment) < 0)
253 return false;
254 if (verbose)
255 mne::Log::info("Fitting...");
256 for (int s = 0; tmin + s * tstep < tmax; s++) {
257 float time = tmin + s * tstep;
258 int picks = time * sfreq - start;
259 if (picks > stepo) {
260 start = start + step;
261 if (raw->pick_data_filt(sel, start, segment) < 0)
262 return false;
263 picks = time * sfreq - start;
264 stime = start / sfreq;
265 }
266 if (MNEMeasDataSet::getValuesFromChannelData(time, integ, segment, stime, sfreq, false, one) < 0) {
267 mne::Log::warning("Cannot pick time: {:8.3f} s", time);
268 continue;
269 }
270 if (!InvDipoleFitData::fit_one(fit, guess, time, one, verbose, dip))
271 mne::Log::warning("t = {:8.3f} s : fit error", time);
272 else {
273 set.push_back(dip);
274 if (verbose)
275 dip.print();
276 else {
277 if (set.size() % report_interval == 0)
278 mne::Log::info("{}..", set.size());
279 }
280 }
281 }
282 if (!verbose)
283 mne::Log::info("[done]");
284 p_set = set;
285 return true;
286}
