152static QImage tightlyPackedRgba(const QImage &source)
153{
154 if (source.isNull())
155 return {};
156
157 QImage rgba = source.convertToFormat(QImage::Format_RGBA8888);
158 const qsizetype tightStride = qsizetype(rgba.width()) * 4;
159 if (rgba.bytesPerLine() == tightStride)
160 return rgba.copy();
161
162 QImage packed(rgba.size(), QImage::Format_RGBA8888);
163 for (int y = 0; y < rgba.height(); ++y) {
164 memcpy(packed.scanLine(y), rgba.constScanLine(y), size_t(tightStride));
165 }
166 return packed;
167}

215void BrainRenderer::Impl::createResources(QRhi *rhi, QRhiRenderPassDescriptor *rp, int sampleCount)
216{
217 uniformBufferOffsetAlignment = rhi->ubufAlignment();
218
219 // Create Uniform Buffer
220 if (!uniformBuffer) {
221 // Size for 8192 slots with alignment — enough for 4 viewports × ~1000 surfaces
222 uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, kUniformSlotCount * uniformBufferOffsetAlignment));
223 uniformBuffer->create();
224 }
225
226 // Create SRB
227 if (!srb) {
228 srb.reset(rhi->newShaderResourceBindings());
229 srb->setBindings({
230 // Use dynamic offset for the uniform buffer.
231 // The size of one uniform block in the shader is ~104 bytes,
232 // but we use uniformBufferOffsetAlignment for the stride.
233 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage, uniformBuffer.get(),kUniformBlockSize)
234 });
235 srb->create();
236 }
237
238 // Shader Loader
239 auto getShader = [](const QString &name) {
240 QFile f(name);
241 return f.open(QIODevice::ReadOnly) ? QShader::fromSerialized(f.readAll()) : QShader();
242 };
243
244 // List of modes to initialize
245 QList<ShaderMode> modes = {Standard, Holographic, Anatomical, Dipole, XRay, ShowNormals};
246
247 for (ShaderMode mode : modes) {
248 QString vert = (mode == Holographic || mode == XRay) ? ":/holographic.vert.qsb" :
249 (mode == Anatomical) ? ":/anatomical.vert.qsb" :
250 (mode == Dipole) ? ":/dipole.vert.qsb" :
251 (mode == ShowNormals) ? ":/shownormals.vert.qsb" : ":/standard.vert.qsb";
252
253 QString frag = (mode == Holographic || mode == XRay) ? ":/holographic.frag.qsb" :
254 (mode == Anatomical) ? ":/anatomical.frag.qsb" :
255 (mode == Dipole) ? ":/dipole.frag.qsb" :
256 (mode == ShowNormals) ? ":/shownormals.frag.qsb" : ":/standard.frag.qsb";
257
258 QShader vS = getShader(vert);
259 QShader fS = getShader(frag);
260
261 if (!vS.isValid() || !fS.isValid()) {
262 qWarning() << "BrainRenderer: Could not load shaders for mode" << mode << vert << frag;
263 continue;
264 }
265
266 // Setup Pipeline
267 auto pipeline = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
268
269 QRhiGraphicsPipeline::TargetBlend blend;
270 if (mode == Holographic || mode == XRay) {
271 blend.enable = true;
272 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
273 blend.dstColor = QRhiGraphicsPipeline::One;
274 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
275 blend.dstAlpha = QRhiGraphicsPipeline::One;
276 } else if (mode == Dipole) {
277 blend.enable = true;
278 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
279 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
280 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
281 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
282 }
283
284 auto setup = [&](QRhiGraphicsPipeline* p, QRhiGraphicsPipeline::CullMode cull) {
285 p->setShaderStages({{ QRhiShaderStage::Vertex, vS }, { QRhiShaderStage::Fragment, fS }});
286
287 QRhiVertexInputLayout il;
288
289 if (mode == Dipole) {
290 il.setBindings({
291 { 6 * sizeof(float) }, // Binding 0: Vertex Data (Pos + Normal) -> stride 6 floats
292 { 21 * sizeof(float), QRhiVertexInputBinding::PerInstance } // Binding 1: Instance Data (Mat4 + Color + Selected) -> stride 21 floats
293 });
294
295 il.setAttributes({
296 // Vertex Buffer (Binding 0)
297 { 0, 0, QRhiVertexInputAttribute::Float3, 0 }, // Pos
298 { 0, 1, QRhiVertexInputAttribute::Float3, 3 * sizeof(float) }, // Normal
299
300 // Instance Buffer (Binding 1)
301 // Model Matrix (4 x vec4)
302 { 1, 2, QRhiVertexInputAttribute::Float4, 0 },
303 { 1, 3, QRhiVertexInputAttribute::Float4, 4 * sizeof(float) },
304 { 1, 4, QRhiVertexInputAttribute::Float4, 8 * sizeof(float) },
305 { 1, 5, QRhiVertexInputAttribute::Float4, 12 * sizeof(float) },
306 // Color
307 { 1, 6, QRhiVertexInputAttribute::Float4, 16 * sizeof(float) },
308 // isSelected
309 { 1, 7, QRhiVertexInputAttribute::Float, 20 * sizeof(float) }
310 });
311 } else {
312 il.setBindings({{ 36 }}); // sizeof(VertexData) = 36 with surfaceId
313 il.setAttributes({{ 0, 0, QRhiVertexInputAttribute::Float3, 0 },
314 { 0, 1, QRhiVertexInputAttribute::Float3, 12 },
315 { 0, 2, QRhiVertexInputAttribute::UNormByte4, 24 },
316 { 0, 3, QRhiVertexInputAttribute::UNormByte4, 28 },
317 { 0, 4, QRhiVertexInputAttribute::Float, 32 }}); // surfaceId
318 }
319
320 p->setVertexInputLayout(il);
321 p->setShaderResourceBindings(srb.get());
322 p->setRenderPassDescriptor(rp);
323 p->setSampleCount(sampleCount);
324 p->setCullMode(cull);
325 if (mode == Holographic) {
326 p->setTargetBlends({blend});
327 p->setDepthTest(true);
328 p->setDepthWrite(false);
329 } else if (mode == XRay) {
330 p->setTargetBlends({blend});
331 p->setDepthTest(false); // Disable Depth Test to see through head
332 p->setDepthWrite(false);
333 } else if (mode == Dipole) {
334 p->setTargetBlends({blend});
335 p->setCullMode(QRhiGraphicsPipeline::None);
336 p->setDepthTest(true);
337 p->setDepthWrite(false);
338 } else {
339 p->setDepthTest(true);
340 p->setDepthWrite(true);
341 }
342 p->setFlags(QRhiGraphicsPipeline::UsesScissor);
343 p->create();
344 };
345
346 if (mode == Holographic || mode == XRay) { // Handle XRay back-faces same as Holographic
347 auto pipelineBack = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
348 setup(pipelineBack.get(), QRhiGraphicsPipeline::Front);
349 pipelinesBackColor[mode] = std::move(pipelineBack);
350 setup(pipeline.get(), QRhiGraphicsPipeline::Back); // Front faces
351 } else {
352 // Culling: None (Double-sided) to be safe for FreeSurfer meshes
353 setup(pipeline.get(), QRhiGraphicsPipeline::None);
354 }
355 pipelines[mode] = std::move(pipeline);
356 }
357
358 resourcesDirty = false;
359}

365void BrainRenderer::ensureRenderTargets(QRhi *rhi, QRhiTexture *colorTex, const QSize &pixelSize)
366{
367 // Rebuild when size changes OR when the backing texture changes
368 // (QRhiWidget may return a different colorTexture() each frame).
369 if (d->rtClear && d->rtSize == pixelSize && d->rtColorTex == colorTex)
370 return;
371
372 d->rtSize = pixelSize;
373 d->rtColorTex = colorTex;
374
375 // Shared depth-stencil buffer
376 d->dsBuffer.reset(rhi->newRenderBuffer(QRhiRenderBuffer::DepthStencil, pixelSize));
377 d->dsBuffer->create();
378
379 QRhiColorAttachment colorAtt(colorTex);
380 QRhiTextureRenderTargetDescription desc(colorAtt);
381 desc.setDepthStencilBuffer(d->dsBuffer.get());
382
383 // RT 1: Clearing (no preserve flags) — used for the first pass of each frame
384 d->rtClear.reset(rhi->newTextureRenderTarget(desc));
385 d->rpClear.reset(d->rtClear->newCompatibleRenderPassDescriptor());
386 d->rtClear->setRenderPassDescriptor(d->rpClear.get());
387 d->rtClear->create();
388
389 // RT 2: Preserving (load previous contents) — used for passes 2+
390 d->rtPreserve.reset(rhi->newTextureRenderTarget(desc,
391 QRhiTextureRenderTarget::PreserveColorContents
392 | QRhiTextureRenderTarget::PreserveDepthStencilContents));
393 d->rpPreserve.reset(d->rtPreserve->newCompatibleRenderPassDescriptor());
394 d->rtPreserve->setRenderPassDescriptor(d->rpPreserve.get());
395 d->rtPreserve->create();
396}

454void BrainRenderer::prepareVideoOverlay(QRhi *rhi,
455 QRhiResourceUpdateBatch *u,
456 VideoOverlay *overlay)
457{
458 if (!overlay || !overlay->isEnabled()) return;
459 if (!overlay->hasFrame()) return;
460 if (!rhi || !u) return;
461
462 QImage frame = tightlyPackedRgba(overlay->frame());
463 if (frame.isNull()) return;
464
465 auto &k = d->videoOverlay;
466 k.currentUniformOffset = 0;
467
468 // ── Lazy resource creation ──────────────────────────────────────
469 if (!k.initialized) {
470 // Shaders
471 QFile vFile(QStringLiteral(":/video_overlay.vert.qsb"));
472 QFile fFile(QStringLiteral(":/video_overlay.frag.qsb"));
473 if (!vFile.open(QIODevice::ReadOnly) || !fFile.open(QIODevice::ReadOnly)) {
474 qWarning() << "BrainRenderer: failed to open video overlay shaders";
475 return;
476 }
477 QShader vShader = QShader::fromSerialized(vFile.readAll());
478 QShader fShader = QShader::fromSerialized(fFile.readAll());
479 if (!vShader.isValid() || !fShader.isValid()) {
480 qWarning() << "BrainRenderer: invalid video overlay shaders";
481 return;
482 }
483
484 // Uniform buffer (mat4 + vec4 + 4 floats = 96 bytes, pad to 256)
485 constexpr int kUbSize = 256;
486 k.uniformBufferOffsetAlignment = rhi->ubufAlignment();
487 k.uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
488 QRhiBuffer::UniformBuffer,
489 64 * k.uniformBufferOffsetAlignment));
490 k.uniformBuffer->create();
491
492 // Vertex buffer: 4 vertices × (3 pos + 2 uv) floats — refilled each frame
493 constexpr int kVbSize = 4 * 5 * sizeof(float);
494 k.vertexBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
495 QRhiBuffer::VertexBuffer, kVbSize));
496 k.vertexBuffer->create();
497
498 // Index buffer: 2 triangles, 6 indices — uploaded once
499 constexpr int kIbSize = 6 * sizeof(quint32);
500 k.indexBuffer.reset(rhi->newBuffer(QRhiBuffer::Immutable,
501 QRhiBuffer::IndexBuffer, kIbSize));
502 k.indexBuffer->create();
503
504 // Sampler (linear, clamp)
505 k.sampler.reset(rhi->newSampler(QRhiSampler::Linear, QRhiSampler::Linear,
506 QRhiSampler::None,
507 QRhiSampler::ClampToEdge,
508 QRhiSampler::ClampToEdge));
509 k.sampler->create();
510
511 // Initial texture uses the first real frame size.
512 k.texture.reset(rhi->newTexture(QRhiTexture::RGBA8, frame.size()));
513 k.texture->create();
514 k.textureSize = frame.size();
515
516 // SRB binding: uniform @ 0, sampled image @ 1
517 k.srb.reset(rhi->newShaderResourceBindings());
518 k.srb->setBindings({
519 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
520 QRhiShaderResourceBinding::VertexStage |
521 QRhiShaderResourceBinding::FragmentStage,
522 k.uniformBuffer.get(), kUniformBlockSize),
523 QRhiShaderResourceBinding::sampledTexture(1,
524 QRhiShaderResourceBinding::FragmentStage,
525 k.texture.get(), k.sampler.get())
526 });
527 k.srb->create();
528
529 // Pipeline (alpha-blended, depth-test off so it sits on top)
530 QRhiGraphicsPipeline::TargetBlend blend;
531 blend.enable = true;
532 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
533 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
534 blend.srcAlpha = QRhiGraphicsPipeline::One;
535 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
536
537 k.pipeline.reset(rhi->newGraphicsPipeline());
538 k.pipeline->setShaderStages({
539 { QRhiShaderStage::Vertex, vShader },
540 { QRhiShaderStage::Fragment, fShader }
541 });
542 QRhiVertexInputLayout il;
543 il.setBindings({{ 5 * sizeof(float) }});
544 il.setAttributes({
545 { 0, 0, QRhiVertexInputAttribute::Float3, 0 },
546 { 0, 1, QRhiVertexInputAttribute::Float2, 3 * sizeof(float) }
547 });
548 k.pipeline->setVertexInputLayout(il);
549 k.pipeline->setShaderResourceBindings(k.srb.get());
550 k.pipeline->setRenderPassDescriptor(d->rtClear->renderPassDescriptor());
551 k.pipeline->setSampleCount(d->rtClear->sampleCount());
552 k.pipeline->setCullMode(QRhiGraphicsPipeline::None);
553 k.pipeline->setTargetBlends({blend});
554 k.pipeline->setDepthTest(false);
555 k.pipeline->setDepthWrite(false);
556 k.pipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);
557 k.pipeline->create();
558
559 QFile dvFile(QStringLiteral(":/video_decal.vert.qsb"));
560 QFile dfFile(QStringLiteral(":/video_decal.frag.qsb"));
561 if (!dvFile.open(QIODevice::ReadOnly) || !dfFile.open(QIODevice::ReadOnly)) {
562 qWarning() << "BrainRenderer: failed to open video decal shaders";
563 return;
564 }
565 QShader dvShader = QShader::fromSerialized(dvFile.readAll());
566 QShader dfShader = QShader::fromSerialized(dfFile.readAll());
567 if (!dvShader.isValid() || !dfShader.isValid()) {
568 qWarning() << "BrainRenderer: invalid video decal shaders";
569 return;
570 }
571
572 QRhiGraphicsPipeline::TargetBlend decalBlend;
573 decalBlend.enable = true;
574 decalBlend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
575 decalBlend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
576 decalBlend.srcAlpha = QRhiGraphicsPipeline::One;
577 decalBlend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
578
579 k.surfacePipeline.reset(rhi->newGraphicsPipeline());
580 k.surfacePipeline->setShaderStages({
581 { QRhiShaderStage::Vertex, dvShader },
582 { QRhiShaderStage::Fragment, dfShader }
583 });
584 QRhiVertexInputLayout dil;
585 dil.setBindings({{ 36 }});
586 dil.setAttributes({
587 { 0, 0, QRhiVertexInputAttribute::Float3, 0 },
588 { 0, 1, QRhiVertexInputAttribute::Float3, 12 },
589 { 0, 2, QRhiVertexInputAttribute::UNormByte4, 24 },
590 { 0, 3, QRhiVertexInputAttribute::UNormByte4, 28 },
591 { 0, 4, QRhiVertexInputAttribute::Float, 32 }
592 });
593 k.surfacePipeline->setVertexInputLayout(dil);
594 k.surfacePipeline->setShaderResourceBindings(k.srb.get());
595 k.surfacePipeline->setRenderPassDescriptor(d->rtClear->renderPassDescriptor());
596 k.surfacePipeline->setSampleCount(d->rtClear->sampleCount());
597 k.surfacePipeline->setCullMode(QRhiGraphicsPipeline::None);
598 k.surfacePipeline->setTargetBlends({decalBlend});
599 k.surfacePipeline->setDepthTest(true);
600 k.surfacePipeline->setDepthWrite(false);
601 k.surfacePipeline->setFlags(QRhiGraphicsPipeline::UsesScissor);
602 k.surfacePipeline->create();
603
604 k.initialized = true;
605 }
606
607 // ── Index buffer (one-shot upload) ──────────────────────────────
608 if (!k.indexUploaded) {
609 const quint32 idx[6] = { 0, 1, 2, 2, 1, 3 };
610 u->uploadStaticBuffer(k.indexBuffer.get(), idx);
611 k.indexUploaded = true;
612 }
613
614 // ── Texture (re-create on size change, re-upload on new frame) ──
615 if (frame.size() != k.textureSize) {
616 k.texture.reset(rhi->newTexture(QRhiTexture::RGBA8, frame.size()));
617 k.texture->create();
618 k.textureSize = frame.size();
619 // Rebuild SRB to point at the new texture
620 k.srb->setBindings({
621 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0,
622 QRhiShaderResourceBinding::VertexStage |
623 QRhiShaderResourceBinding::FragmentStage,
624 k.uniformBuffer.get(), kUniformBlockSize),
625 QRhiShaderResourceBinding::sampledTexture(1,
626 QRhiShaderResourceBinding::FragmentStage,
627 k.texture.get(), k.sampler.get())
628 });
629 k.srb->create();
630 k.uploadedFrameGen = std::numeric_limits<quint64>::max();
631 }
632 if (overlay->frameGeneration() != k.uploadedFrameGen) {
633 QRhiTextureSubresourceUploadDescription sub(frame);
634 QRhiTextureUploadDescription desc({ 0, 0, sub });
635 u->uploadTexture(k.texture.get(), desc);
636 k.uploadedFrameGen = overlay->frameGeneration();
637 }

640void BrainRenderer::renderVideoOverlay(QRhiCommandBuffer *cb, QRhi *rhi,
641 const SceneData &data,
642 VideoOverlay *overlay)
643{
644 if (!overlay || !overlay->isEnabled()) return;
645 if (!overlay->hasFrame()) return;
646
647 auto &k = d->videoOverlay;
648 if (!k.initialized) return;
649 if (k.uniformBufferOffsetAlignment <= 0) return;
650
651 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
652 const int uniformOffset = k.currentUniformOffset;
653 k.currentUniformOffset += k.uniformBufferOffsetAlignment;
654 if (uniformOffset + kUniformBlockSize > k.uniformBuffer->size()) return;
655
656 // ── Billboard the quad to face the camera ──────────────────────
657 const QVector3D centre = overlay->focusPosition();
658 const float halfSize = 0.5f * overlay->sizeMeters();
659
660 QVector3D viewDir = centre - data.cameraPos;
661 if (viewDir.lengthSquared() < 1e-12f) viewDir = QVector3D(0, 0, -1);
662 viewDir.normalize();
663
664 QVector3D worldUp(0.0f, 0.0f, 1.0f);
665 if (std::abs(QVector3D::dotProduct(viewDir, worldUp)) > 0.95f) {

672 const QVector3D c10 = centre + right * halfSize - up * halfSize;
673 const QVector3D c01 = centre - right * halfSize + up * halfSize;
674 const QVector3D c11 = centre + right * halfSize + up * halfSize;

688 float opacity;
689 float pad0, pad1, pad2;

692 ub.borderColor[0] = 0.0f;
693 ub.borderColor[1] = 0.85f;

695 ub.borderColor[3] = 1.0f;

704 cb->setScissor(toScissor(data));
705 cb->setGraphicsPipeline(k.pipeline.get());

707 cb->setShaderResources(k.srb.get(), 1, &srbOffset);

800 auto *pipeline = d->pipelines[mode].get();

804 // (BrainView::render pre-upload loop). Do not call
805 // surface->updateBuffers() here — it would allocate a redundant
806 // QRhiResourceUpdateBatch per surface.

811 int offset = d->currentUniformOffset;
812 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
813 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
814 qWarning("BrainRenderer: uniform buffer overflow (%d / %d bytes) — too many surfaces. Some draws will be skipped.",
815 d->currentUniformOffset, (int)d->uniformBuffer->size());

817 }

819 // On desktop, when a specific annotation region or vertex range is
820 // selected the CPU vertex-color gold tint (in updateVertexColors)

822 // gold glow so it doesn't drown out the region highlight.

829#endif

852
853 cb->resourceUpdate(u);
854
855 // Re-assert the per-pane viewport and scissor after resourceUpdate.
856 // The scissor provides a hard pixel clip that guarantees no cross-pane
857 // bleeding, regardless of Metal render-encoder restarts.
858 cb->setViewport(toViewport(data));
859 cb->setScissor(toScissor(data));
860
861 auto draw = [&](QRhiGraphicsPipeline *p) {
862 cb->setGraphicsPipeline(p);
863
864 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
865 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
866 const QRhiCommandBuffer::VertexInput vbuf(surface->vertexBuffer(), 0);
867 cb->setVertexInput(0, 1, &vbuf, surface->indexBuffer(), 0, QRhiCommandBuffer::IndexUInt32);
868 cb->drawIndexed(surface->indexCount());
869 };
870
871 if (mode == Holographic && d->pipelinesBackColor[Holographic]) {
872 draw(d->pipelinesBackColor[Holographic].get());
873 }
874
875 draw(pipeline);
876}

1093// WORKAROUND(QRhi-GLES2): Merged single-drawIndexed rendering.
1094// The Qt QRhi GLES2/WebGL backend has a bug where only the first
1095// drawIndexed() per render pass produces visible output. These two
1096// methods merge all surfaces (brain, BEM, sensors, digitizers,

1099//
1100// Remove when upstream Qt fixes the issue.

1102
1103void BrainRenderer::prepareMergedSurfaces(QRhi *rhi, QRhiResourceUpdateBatch * /*u*/,
1104 const QVector<BrainSurface*> &surfaces,
1105 const QString &groupName)
1106{
1107 auto &group = d->mergedGroups[groupName];
1108
1109 // Check if surface list changed (different count or different pointers)
1110 if (!group.dirty) {
1111 if (group.surfaces.size() != surfaces.size()) {
1112 group.dirty = true;
1113 } else {
1114 for (int i = 0; i < surfaces.size(); ++i) {
1115 if (group.surfaces[i] != surfaces[i]) {
1116 group.dirty = true;
1117 break;
1118 }
1119 }
1120 }
1121 }
1122
1123 // If geometry hasn't changed, check if any surface vertex data actually changed
1124 // (STC animation changes vertex colors but not topology)
1125 if (!group.dirty && group.indexCount > 0) {
1126 // Compare per-surface vertex generation counters
1127 bool anyChanged = false;
1128 if (group.surfaceGenerations.size() != surfaces.size()) {
1129 anyChanged = true;
1130 } else {
1131 for (int i = 0; i < surfaces.size(); ++i) {
1132 if (surfaces[i] && surfaces[i]->vertexGeneration() != group.surfaceGenerations[i]) {
1133 anyChanged = true;
1134 break;
1135 }
1136 }
1137 }
1138
1139 if (!anyChanged) {
1140 // Nothing changed — skip vertex rebuild entirely
1141 return;
1142 }
1143
1144 // Re-merge vertex data directly into vertexRaw (no temp allocation)
1145 // Safety: verify vertex count hasn't changed since the full rebuild.
1146 // If it has, fall through to the full rebuild path to update indices.
1147 int totalVerts = 0;
1148 for (int si = 0; si < surfaces.size(); ++si)
1149 if (surfaces[si]) totalVerts += surfaces[si]->vertexDataRef().size();
1150 if (totalVerts != group.totalVertexCount) {
1151 group.dirty = true;
1152 // Fall through to full rebuild below
1153 } else {
1154 float brainId = 0.0f;
1155 float nonBrainId = 100.0f; // offset so shaders can distinguish
1156 group.surfaceGenerations.resize(surfaces.size());
1157 VertexData *dst = reinterpret_cast<VertexData*>(group.vertexRaw.data());
1158 for (int si = 0; si < surfaces.size(); ++si) {
1159 BrainSurface *surf = surfaces[si];
1160 if (!surf) { brainId += 1.0f; nonBrainId += 1.0f; continue; }
1161 const bool isBrain = (surf->tissueType() == BrainSurface::TissueBrain);
1162 const float id = isBrain ? brainId : nonBrainId;
1163 const auto &srcVerts = surf->vertexDataRef();
1164 const int n = srcVerts.size();
1165 memcpy(dst, srcVerts.constData(), n * sizeof(VertexData));
1166 for (int j = 0; j < n; ++j)
1167 dst[j].surfaceId = id;
1168 dst += n;
1169 group.surfaceGenerations[si] = surf->vertexGeneration();
1170 brainId += 1.0f;
1171 nonBrainId += 1.0f;
1172 }
1173 group.gpuVertexDirty = true;
1174 return;
1175 }
1176 }
1177
1178 // Full rebuild: topology or surface list changed
1179 group.surfaces = surfaces;
1180 group.indexCount = 0;
1181 group.totalVertexCount = 0;
1182 group.dirty = false;
1183
1184 // Build merged vertex + index arrays
1185 // Pre-calculate total sizes for single allocation
1186 int totalVerts = 0;
1187 int totalIndices = 0;
1188 for (int si = 0; si < surfaces.size(); ++si) {
1189 if (!surfaces[si]) continue;
1190 totalVerts += surfaces[si]->vertexDataRef().size();
1191 totalIndices += surfaces[si]->indexDataRef().size();
1192 }
1193
1194 group.indexCount = totalIndices;
1195 if (group.indexCount == 0) return;
1196
1197 const quint32 vbufSize = totalVerts * sizeof(VertexData);
1198 const quint32 ibufSize = totalIndices * sizeof(uint32_t);
1199
1200 // (Re-)create Dynamic buffers when they don't exist or are too small
1201 if (!group.vertexBuffer || group.vertexBuffer->size() < vbufSize) {
1202 group.vertexBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
1203 QRhiBuffer::VertexBuffer, vbufSize));
1204 group.vertexBuffer->create();
1205 }
1206 if (!group.indexBuffer || group.indexBuffer->size() < ibufSize) {
1207 group.indexBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
1208 QRhiBuffer::IndexBuffer, ibufSize));
1209 group.indexBuffer->create();
1210 }
1211
1212 // Write directly into QByteArrays — no temp QVector intermediaries
1213 group.vertexRaw.resize(vbufSize);
1214 group.indexRaw.resize(ibufSize);
1215 group.surfaceGenerations.resize(surfaces.size());
1216
1217 VertexData *vDst = reinterpret_cast<VertexData*>(group.vertexRaw.data());
1218 uint32_t *iDst = reinterpret_cast<uint32_t*>(group.indexRaw.data());
1219 float brainId = 0.0f;
1220 float nonBrainId = 100.0f; // offset so shaders can distinguish
1221 uint32_t vertexOffset = 0;
1222 for (int si = 0; si < surfaces.size(); ++si) {
1223 BrainSurface *surf = surfaces[si];
1224 if (!surf) { brainId += 1.0f; nonBrainId += 1.0f; continue; }
1225
1226 const bool isBrain = (surf->tissueType() == BrainSurface::TissueBrain);
1227 const float id = isBrain ? brainId : nonBrainId;
1228 const auto &srcVerts = surf->vertexDataRef();
1229 const auto &srcIdx = surf->indexDataRef();
1230 const int nv = srcVerts.size();
1231 const int ni = srcIdx.size();
1232
1233 // Bulk copy vertices + stamp surfaceId
1234 memcpy(vDst, srcVerts.constData(), nv * sizeof(VertexData));
1235 for (int j = 0; j < nv; ++j)
1236 vDst[j].surfaceId = id;

1275void BrainRenderer::drawMergedSurfaces(QRhiCommandBuffer *cb, QRhi *rhi,
1276 const SceneData &data, ShaderMode mode,
1277 const QString &groupName)
1278{
1279 auto it = d->mergedGroups.find(groupName);
1280 if (it == d->mergedGroups.end()) return;
1281 auto &group = it->second;
1282
1283 if (group.indexCount == 0) return;
1284
1285 auto *pipeline = d->pipelines[mode].get();
1286 if (!pipeline) return;
1287
1288 // Determine which merged surface (if any) is selected.
1289 // surfaceId encoding: brain surfaces get ids 0,1,2...; non-brain get 100,101,102...
1290 float selectedSurfaceId = -1.0f;
1291 for (int i = 0; i < group.surfaces.size(); ++i) {
1292 if (group.surfaces[i] && group.surfaces[i]->isSelected()) {
1293 if (group.surfaces[i]->selectedRegionId() == -1
1294 && group.surfaces[i]->selectedVertexStart() < 0) {
1295 const bool isBrain = (group.surfaces[i]->tissueType() == BrainSurface::TissueBrain);
1296 selectedSurfaceId = static_cast<float>(isBrain ? i : 100 + i);
1297 }
1298 break;
1299 }
1300 }
1301
1302 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
1303
1304 // Re-upload merged geometry only when data actually changed.
1305 // Split VBO / IBO uploads: the fast-update path (STC color changes)
1306 // only modifies vertices; re-uploading the IBO via glBufferSubData on
1307 // WebGL can corrupt the VAO's element-buffer binding.
1308 if (group.gpuVertexDirty) {
1309 u->updateDynamicBuffer(group.vertexBuffer.get(), 0, group.vertexRaw.size(), group.vertexRaw.constData());
1310 group.gpuVertexDirty = false;
1311 }
1312 if (group.gpuIndexDirty) {
1313 u->updateDynamicBuffer(group.indexBuffer.get(), 0, group.indexRaw.size(), group.indexRaw.constData());
1314 group.gpuIndexDirty = false;
1315 }
1316
1317 int offset = d->currentUniformOffset;
1318 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
1319 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
1320 qWarning("BrainRenderer: uniform buffer overflow in drawMergedSurfaces");
1321 return;
1322 }
1323
1324 // Pack all uniforms into a contiguous block for a single upload
1325 // Layout must match the shader's UniformBlock (std140).
1326 struct {
1327 float mvp[16]; // 0..63
1328 float cameraPos[3]; // 64..75
1329 float isSelected; // 76..79
1330 float lightDir[3]; // 80..91
1331 float tissueType; // 92..95
1332 float lightingEnabled; // 96..99
1333 float overlayMode; // 100..103
1334 float selectedSurfaceId;// 104..107
1335 } ub;
1336 memcpy(ub.mvp, data.mvp.constData(), 64);
1337 memcpy(ub.cameraPos, &data.cameraPos, 12);
1338 ub.isSelected = 0.0f;
1339 memcpy(ub.lightDir, &data.lightDir, 12);
1340 ub.tissueType = (!group.surfaces.isEmpty() && group.surfaces.first())
1341 ? static_cast<float>(group.surfaces.first()->tissueType()) : 0.0f;
1342 ub.lightingEnabled = data.lightingEnabled ? 1.0f : 0.0f;
1343 ub.overlayMode = data.overlayMode;
1344 ub.selectedSurfaceId = selectedSurfaceId;
1345
1346 u->updateDynamicBuffer(d->uniformBuffer.get(), offset, sizeof(ub), &ub);
1347
1348 cb->resourceUpdate(u);
1349
1350 cb->setViewport(toViewport(data));
1351 cb->setScissor(toScissor(data));
1352
1353 auto draw = [&](QRhiGraphicsPipeline *p) {
1354 cb->setGraphicsPipeline(p);
1355 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
1356 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
1357 const QRhiCommandBuffer::VertexInput vbuf(group.vertexBuffer.get(), 0);
1358 cb->setVertexInput(0, 1, &vbuf, group.indexBuffer.get(), 0, QRhiCommandBuffer::IndexUInt32);
1359 cb->drawIndexed(group.indexCount);
1360 };
1361
1362 // WORKAROUND(QRhi-GLES2): On WebGL, only one drawIndexed() per pass.
1363 // For Holographic mode, the back-face pass must happen in a separate
1364 // render pass. The caller is responsible for wrapping each call in
1365 // its own beginPreservingPass/endPass on WASM.
1366#ifdef __EMSCRIPTEN__
1367 draw(pipeline);
1368#else
1369 if (mode == Holographic && d->pipelinesBackColor[Holographic]) {
1370 draw(d->pipelinesBackColor[Holographic].get());
1371 }
1372
1373 draw(pipeline);
1374#endif
1375}
