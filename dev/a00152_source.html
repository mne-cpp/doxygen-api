148void BrainRenderer::Impl::createResources(QRhi *rhi, QRhiRenderPassDescriptor *rp, int sampleCount)
149{
150 uniformBufferOffsetAlignment = rhi->ubufAlignment();
151
152 // Create Uniform Buffer
153 if (!uniformBuffer) {
154 // Size for 8192 slots with alignment — enough for 4 viewports × ~1000 surfaces
155 uniformBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic, QRhiBuffer::UniformBuffer, kUniformSlotCount * uniformBufferOffsetAlignment));
156 uniformBuffer->create();
157 }
158
159 // Create SRB
160 if (!srb) {
161 srb.reset(rhi->newShaderResourceBindings());
162 srb->setBindings({
163 // Use dynamic offset for the uniform buffer.
164 // The size of one uniform block in the shader is ~104 bytes,
165 // but we use uniformBufferOffsetAlignment for the stride.
166 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage, uniformBuffer.get(),kUniformBlockSize)
167 });
168 srb->create();
169 }
170
171 // Shader Loader
172 auto getShader = [](const QString &name) {
173 QFile f(name);
174 return f.open(QIODevice::ReadOnly) ? QShader::fromSerialized(f.readAll()) : QShader();
175 };
176
177 // List of modes to initialize
178 QList<ShaderMode> modes = {Standard, Holographic, Anatomical, Dipole, XRay, ShowNormals};
179
180 for (ShaderMode mode : modes) {
181 QString vert = (mode == Holographic || mode == XRay) ? ":/holographic.vert.qsb" :
182 (mode == Anatomical) ? ":/anatomical.vert.qsb" :
183 (mode == Dipole) ? ":/dipole.vert.qsb" :
184 (mode == ShowNormals) ? ":/shownormals.vert.qsb" : ":/standard.vert.qsb";
185
186 QString frag = (mode == Holographic || mode == XRay) ? ":/holographic.frag.qsb" :
187 (mode == Anatomical) ? ":/anatomical.frag.qsb" :
188 (mode == Dipole) ? ":/dipole.frag.qsb" :
189 (mode == ShowNormals) ? ":/shownormals.frag.qsb" : ":/standard.frag.qsb";
190
191 QShader vS = getShader(vert);
192 QShader fS = getShader(frag);
193
194 if (!vS.isValid() || !fS.isValid()) {
195 qWarning() << "BrainRenderer: Could not load shaders for mode" << mode << vert << frag;
196 continue;
197 }
198
199 // Setup Pipeline
200 auto pipeline = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
201
202 QRhiGraphicsPipeline::TargetBlend blend;
203 if (mode == Holographic || mode == XRay) {
204 blend.enable = true;
205 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
206 blend.dstColor = QRhiGraphicsPipeline::One;
207 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
208 blend.dstAlpha = QRhiGraphicsPipeline::One;
209 } else if (mode == Dipole) {
210 blend.enable = true;
211 blend.srcColor = QRhiGraphicsPipeline::SrcAlpha;
212 blend.dstColor = QRhiGraphicsPipeline::OneMinusSrcAlpha;
213 blend.srcAlpha = QRhiGraphicsPipeline::SrcAlpha;
214 blend.dstAlpha = QRhiGraphicsPipeline::OneMinusSrcAlpha;
215 }
216
217 auto setup = [&](QRhiGraphicsPipeline* p, QRhiGraphicsPipeline::CullMode cull) {
218 p->setShaderStages({{ QRhiShaderStage::Vertex, vS }, { QRhiShaderStage::Fragment, fS }});
219
220 QRhiVertexInputLayout il;
221
222 if (mode == Dipole) {
223 il.setBindings({
224 { 6 * sizeof(float) }, // Binding 0: Vertex Data (Pos + Normal) -> stride 6 floats
225 { 21 * sizeof(float), QRhiVertexInputBinding::PerInstance } // Binding 1: Instance Data (Mat4 + Color + Selected) -> stride 21 floats
226 });
227
228 il.setAttributes({
229 // Vertex Buffer (Binding 0)
230 { 0, 0, QRhiVertexInputAttribute::Float3, 0 }, // Pos
231 { 0, 1, QRhiVertexInputAttribute::Float3, 3 * sizeof(float) }, // Normal
232
233 // Instance Buffer (Binding 1)
234 // Model Matrix (4 x vec4)
235 { 1, 2, QRhiVertexInputAttribute::Float4, 0 },
236 { 1, 3, QRhiVertexInputAttribute::Float4, 4 * sizeof(float) },
237 { 1, 4, QRhiVertexInputAttribute::Float4, 8 * sizeof(float) },
238 { 1, 5, QRhiVertexInputAttribute::Float4, 12 * sizeof(float) },
239 // Color
240 { 1, 6, QRhiVertexInputAttribute::Float4, 16 * sizeof(float) },
241 // isSelected
242 { 1, 7, QRhiVertexInputAttribute::Float, 20 * sizeof(float) }
243 });
244 } else {
245 il.setBindings({{ 36 }}); // sizeof(VertexData) = 36 with surfaceId
246 il.setAttributes({{ 0, 0, QRhiVertexInputAttribute::Float3, 0 },
247 { 0, 1, QRhiVertexInputAttribute::Float3, 12 },
248 { 0, 2, QRhiVertexInputAttribute::UNormByte4, 24 },
249 { 0, 3, QRhiVertexInputAttribute::UNormByte4, 28 },
250 { 0, 4, QRhiVertexInputAttribute::Float, 32 }}); // surfaceId
251 }
252
253 p->setVertexInputLayout(il);
254 p->setShaderResourceBindings(srb.get());
255 p->setRenderPassDescriptor(rp);
256 p->setSampleCount(sampleCount);
257 p->setCullMode(cull);
258 if (mode == Holographic) {
259 p->setTargetBlends({blend});
260 p->setDepthTest(true);
261 p->setDepthWrite(false);
262 } else if (mode == XRay) {
263 p->setTargetBlends({blend});
264 p->setDepthTest(false); // Disable Depth Test to see through head
265 p->setDepthWrite(false);
266 } else if (mode == Dipole) {
267 p->setTargetBlends({blend});
268 p->setCullMode(QRhiGraphicsPipeline::None);
269 p->setDepthTest(true);
270 p->setDepthWrite(false);
271 } else {
272 p->setDepthTest(true);
273 p->setDepthWrite(true);
274 }
275 p->setFlags(QRhiGraphicsPipeline::UsesScissor);
276 p->create();
277 };
278
279 if (mode == Holographic || mode == XRay) { // Handle XRay back-faces same as Holographic
280 auto pipelineBack = std::unique_ptr<QRhiGraphicsPipeline>(rhi->newGraphicsPipeline());
281 setup(pipelineBack.get(), QRhiGraphicsPipeline::Front);
282 pipelinesBackColor[mode] = std::move(pipelineBack);
283 setup(pipeline.get(), QRhiGraphicsPipeline::Back); // Front faces
284 } else {
285 // Culling: None (Double-sided) to be safe for FreeSurfer meshes
286 setup(pipeline.get(), QRhiGraphicsPipeline::None);
287 }
288 pipelines[mode] = std::move(pipeline);
289 }
290
291 resourcesDirty = false;
292}

296void BrainRenderer::beginFrame(QRhiCommandBuffer *cb, QRhiRenderTarget *rt)
297{
298 d->currentUniformOffset = 0;
299
300 // WORKAROUND(QRhi-GLES2): Ensure PreserveColorContents is cleared so
301 // the first pass of each frame clears the framebuffer. The multi-pass
302 // WASM path sets this flag via beginAdditionalPass(); if the last
303 // additional pass is still open when endFrame() runs, the flag leaks
304 // into the next frame and the clear never happens — producing ghost
305 // images of previous frames.
306 auto *texRt = dynamic_cast<QRhiTextureRenderTarget*>(rt);
307 if (texRt) {
308 texRt->setFlags(texRt->flags()
309 & ~QRhiTextureRenderTarget::Flags(
310 QRhiTextureRenderTarget::PreserveColorContents
311 | QRhiTextureRenderTarget::PreserveDepthStencilContents));
312 }
313
314 cb->beginPass(rt, QColor(0, 0, 0), { 1.0f, 0 });
315 const int w = rt->pixelSize().width();
316 const int h = rt->pixelSize().height();
317 cb->setViewport(QRhiViewport(0, 0, w, h));
318 cb->setScissor(QRhiScissor(0, 0, w, h));
319}

335//=============================================================================================================
336// ── WORKAROUND(QRhi-GLES2): Multi-pass rendering ───────────────────────
337// The Qt QRhi GLES2/WebGL backend only renders the first drawIndexed()
338// per render pass. To render multiple surface types (brain, BEM,
339// sensors, dipoles, etc.) we use separate render passes — one per
340// surface category — with PreserveColorContents so each pass composites
341// over the previous one. Remove when upstream Qt fixes the bug.
342//=============================================================================================================
343
344void BrainRenderer::beginAdditionalPass(QRhiCommandBuffer *cb, QRhiRenderTarget *rt)
345{
346 // Set PreserveColorContents so this pass does NOT clear the previous
347 // pass's output. The flag is checked by beginPass() to select
348 // GL_LOAD instead of GL_CLEAR.
349 auto *texRt = dynamic_cast<QRhiTextureRenderTarget*>(rt);
350 if (texRt) {
351 texRt->setFlags(texRt->flags()
352 | QRhiTextureRenderTarget::PreserveColorContents
353 | QRhiTextureRenderTarget::PreserveDepthStencilContents);
354 }
355
356 cb->beginPass(rt, QColor(0, 0, 0), { 1.0f, 0 });
357 const int w = rt->pixelSize().width();
358 const int h = rt->pixelSize().height();
359 cb->setViewport(QRhiViewport(0, 0, w, h));
360 cb->setScissor(QRhiScissor(0, 0, w, h));
361}
362
363void BrainRenderer::endAdditionalPass(QRhiCommandBuffer *cb, QRhiRenderTarget *rt)
364{
365 cb->endPass();
366
367 // Restore flags so the NEXT frame's first pass clears normally.
368 auto *texRt = dynamic_cast<QRhiTextureRenderTarget*>(rt);
369 if (texRt) {
370 texRt->setFlags(texRt->flags()
371 & ~QRhiTextureRenderTarget::Flags(
372 QRhiTextureRenderTarget::PreserveColorContents
373 | QRhiTextureRenderTarget::PreserveDepthStencilContents));
374 }
375}

379void BrainRenderer::renderSurface(QRhiCommandBuffer *cb, QRhi *rhi, const SceneData &data, BrainSurface *surface, ShaderMode mode)
380{
381 if (!surface || !surface->isVisible()) return;
382
383 // Check if pipeline for this mode exists
384 if (d->pipelines.find(mode) == d->pipelines.end()) return;
385
386 auto *pipeline = d->pipelines[mode].get();
387 if (!pipeline) return;
388
389 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
390 surface->updateBuffers(rhi, u);
391
392 // Dynamic slot update
393 int offset = d->currentUniformOffset;
394 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
395 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
396 qWarning("BrainRenderer: uniform buffer overflow (%d / %d bytes) — too many surfaces. Some draws will be skipped.",
397 d->currentUniformOffset, (int)d->uniformBuffer->size());
398 return; // Skip this draw rather than silently corrupt earlier viewport data
399 }
400
401 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetMVP, 64, data.mvp.constData());
402 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetCameraPos, 12, &data.cameraPos);
403
404 // On desktop, when a specific annotation region or vertex range is
405 // selected the CPU vertex-color gold tint (in updateVertexColors)
406 // provides per-region feedback. Suppress the shader's whole-surface
407 // gold glow so it doesn't drown out the region highlight.
408 float selected = surface->isSelected() ? 1.0f : 0.0f;
409#ifndef __EMSCRIPTEN__
410 if (surface->isSelected()
411 && (surface->selectedRegionId() != -1 || surface->selectedVertexStart() >= 0)) {
412 selected = 0.0f;
413 }
414#endif
415 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetSelected, 4, &selected);
416
417 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetLightDir, 12, &data.lightDir);
418
419 // Pass tissue type for anatomical shader (offset 92, uses original _pad2 slot)
420 float tissueType = static_cast<float>(surface->tissueType());
421 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetTissueType, 4, &tissueType);
422
423 float lighting = data.lightingEnabled ? 1.0f : 0.0f;
424 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetLighting, 4, &lighting);
425
426 float overlayMode = data.overlayMode;
427 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetOverlayMode, 4, &overlayMode);
428
429 // Per-surface path: disable surfaceId-based selection (handled by isSelected).
430 float noSurfaceSelection = -1.0f;
431 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetSelectedSurfaceId, 4, &noSurfaceSelection);
432
433 cb->resourceUpdate(u);
434
435 // Re-assert the per-pane viewport and scissor after resourceUpdate.
436 // The scissor provides a hard pixel clip that guarantees no cross-pane
437 // bleeding, regardless of Metal render-encoder restarts.
438 cb->setViewport(toViewport(data));
439 cb->setScissor(toScissor(data));
440
441 auto draw = [&](QRhiGraphicsPipeline *p) {
442 cb->setGraphicsPipeline(p);
443
444 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
445 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
446 const QRhiCommandBuffer::VertexInput vbuf(surface->vertexBuffer(), 0);
447 cb->setVertexInput(0, 1, &vbuf, surface->indexBuffer(), 0, QRhiCommandBuffer::IndexUInt32);
448 cb->drawIndexed(surface->indexCount());
449 };
450
451 if (mode == Holographic && d->pipelinesBackColor.count(Holographic) > 0) {
452 draw(d->pipelinesBackColor[Holographic].get());
453 }
454
455 draw(pipeline);
456}

588//=============================================================================================================
589// WORKAROUND(QRhi-GLES2): Merged single-drawIndexed rendering.
590// The Qt QRhi GLES2/WebGL backend has a bug where only the first
591// drawIndexed() per render pass produces visible output. These two
592// methods merge all surfaces (brain, BEM, sensors, digitizers,
593// source-space) into a single VBO/IBO so that all geometry is drawn
594// in one call.
595//
596// Remove when upstream Qt fixes the issue.
597//=============================================================================================================
598
599void BrainRenderer::prepareMergedSurfaces(QRhi *rhi, QRhiResourceUpdateBatch * /*u*/,
600 const QVector<BrainSurface*> &surfaces,
601 const QString &groupName)
602{
603 auto &group = d->mergedGroups[groupName];
604 group.surfaces = surfaces;
605 group.indexCount = 0;
606
607 // Build merged vertex + index arrays
608 QVector<VertexData> mergedVerts;
609 QVector<uint32_t> mergedIndices;
610
611 float surfaceId = 0.0f;
612 for (BrainSurface *surf : surfaces) {
613 if (!surf) continue;
614
615 const uint32_t vertexOffset = mergedVerts.size();
616 const auto &srcVerts = surf->vertexDataRef();
617 const auto &srcIdx = surf->indexDataRef();
618
619 // Copy vertices with surfaceId stamp
620 mergedVerts.reserve(mergedVerts.size() + srcVerts.size());
621 for (const VertexData &v : srcVerts) {
622 VertexData mv = v;
623 mv.surfaceId = surfaceId;
624 mergedVerts.append(mv);
625 }
626
627 // Copy indices with global vertex offset
628 mergedIndices.reserve(mergedIndices.size() + srcIdx.size());
629 for (uint32_t idx : srcIdx) {
630 mergedIndices.append(idx + vertexOffset);
631 }
632
633 surfaceId += 1.0f;
634 }
635
636 group.indexCount = mergedIndices.size();
637 if (group.indexCount == 0) return;
638
639 const quint32 vbufSize = mergedVerts.size() * sizeof(VertexData);
640 const quint32 ibufSize = mergedIndices.size() * sizeof(uint32_t);
641
642 // (Re-)create Dynamic buffers when they don't exist or are too small
643 if (!group.vertexBuffer || group.vertexBuffer->size() < vbufSize) {
644 group.vertexBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
645 QRhiBuffer::VertexBuffer, vbufSize));
646 group.vertexBuffer->create();
647 }
648 if (!group.indexBuffer || group.indexBuffer->size() < ibufSize) {
649 group.indexBuffer.reset(rhi->newBuffer(QRhiBuffer::Dynamic,
650 QRhiBuffer::IndexBuffer, ibufSize));
651 group.indexBuffer->create();
652 }
653
654 // Cache raw data — uploaded in drawMergedSurfaces() inside the render
655 // pass so that GL element-buffer bindings are refreshed right before draw.
656 group.vertexRaw = QByteArray(reinterpret_cast<const char*>(mergedVerts.constData()), vbufSize);
657 group.indexRaw = QByteArray(reinterpret_cast<const char*>(mergedIndices.constData()), ibufSize);
658}

661
662void BrainRenderer::drawMergedSurfaces(QRhiCommandBuffer *cb, QRhi *rhi,
663 const SceneData &data, ShaderMode mode,
664 const QString &groupName)
665{
666 auto it = d->mergedGroups.find(groupName);
667 if (it == d->mergedGroups.end()) return;
668 auto &group = it->second;
669
670 if (group.indexCount == 0) return;
671 if (d->pipelines.find(mode) == d->pipelines.end()) return;
672
673 auto *pipeline = d->pipelines[mode].get();
674 if (!pipeline) return;
675
676 // Determine which merged surface (if any) is selected.
677 float selectedSurfaceId = -1.0f;
678 for (int i = 0; i < group.surfaces.size(); ++i) {
679 if (group.surfaces[i] && group.surfaces[i]->isSelected()) {
680 if (group.surfaces[i]->selectedRegionId() == -1
681 && group.surfaces[i]->selectedVertexStart() < 0) {
682 selectedSurfaceId = static_cast<float>(i);
683 }
684 break;
685 }
686 }
687
688 QRhiResourceUpdateBatch *u = rhi->nextResourceUpdateBatch();
689
690 // Re-upload merged geometry — ensures GL buffer bindings are fresh
691 u->updateDynamicBuffer(group.vertexBuffer.get(), 0, group.vertexRaw.size(), group.vertexRaw.constData());
692 u->updateDynamicBuffer(group.indexBuffer.get(), 0, group.indexRaw.size(), group.indexRaw.constData());
693
694 int offset = d->currentUniformOffset;
695 d->currentUniformOffset += d->uniformBufferOffsetAlignment;
696 if (d->currentUniformOffset >= d->uniformBuffer->size()) {
697 qWarning("BrainRenderer: uniform buffer overflow in drawMergedSurfaces");
698 return;
699 }
700
701 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetMVP, 64, data.mvp.constData());
702 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetCameraPos, 12, &data.cameraPos);
703
704 float selected = 0.0f; // Per-surface isSelected disabled for merged path
705 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetSelected, 4, &selected);
706 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetLightDir, 12, &data.lightDir);
707
708 float tissueType = 0.0f; // Derived from first surface in the group
709 if (!group.surfaces.isEmpty() && group.surfaces.first()) {
710 tissueType = static_cast<float>(group.surfaces.first()->tissueType());
711 }
712 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetTissueType, 4, &tissueType);
713
714 float lighting = data.lightingEnabled ? 1.0f : 0.0f;
715 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetLighting, 4, &lighting);
716
717 float overlayMode = data.overlayMode;
718 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetOverlayMode, 4, &overlayMode);
719
720 u->updateDynamicBuffer(d->uniformBuffer.get(), offset + kOffsetSelectedSurfaceId, 4, &selectedSurfaceId);
721
722 cb->resourceUpdate(u);
723
724 cb->setViewport(toViewport(data));
725 cb->setScissor(toScissor(data));
726
727 auto draw = [&](QRhiGraphicsPipeline *p) {
728 cb->setGraphicsPipeline(p);
729 const QRhiCommandBuffer::DynamicOffset srbOffset = { 0, uint32_t(offset) };
730 cb->setShaderResources(d->srb.get(), 1, &srbOffset);
731 const QRhiCommandBuffer::VertexInput vbuf(group.vertexBuffer.get(), 0);
732 cb->setVertexInput(0, 1, &vbuf, group.indexBuffer.get(), 0, QRhiCommandBuffer::IndexUInt32);
733 cb->drawIndexed(group.indexCount);
734 };
735
736 if (mode == Holographic && d->pipelinesBackColor.count(Holographic) > 0) {
737 draw(d->pipelinesBackColor[Holographic].get());
738 }
739
740 draw(pipeline);
741}
