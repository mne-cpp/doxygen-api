115template<class Scalar>
116ShrunkCovariance<Scalar> shrink(Eigen::Matrix<Scalar, Eigen::Dynamic, Eigen::Dynamic> S, Scalar alpha)
117{
118 const Scalar mu = S.trace() / static_cast<Scalar>(S.rows());
119 S *= Scalar(1) - alpha;
120 S.diagonal().array() += alpha * mu;
121 return {std::move(S), alpha};
122}

136template<class Derived>
137ShrunkCovariance<typename Derived::Scalar> ledoit_wolf_covariance(const Eigen::MatrixBase<Derived>& data)
138{
139 using Scalar = typename Derived::Scalar;
140 const auto p = static_cast<Scalar>(data.rows());
141 const auto n = static_cast<Scalar>(data.cols());
142
143 // Sample covariance (1/n, as in the paper and scikit-learn)
144 const Eigen::Matrix<Scalar, Eigen::Dynamic, Eigen::Dynamic> S = (data * data.transpose()) / n;
145 const Scalar mu = S.trace() / p;
146
147 // delta = ||S - mu I||_F^2 / p
148 Eigen::Matrix<Scalar, Eigen::Dynamic, Eigen::Dynamic> diff = S;
149 diff.diagonal().array() -= mu;
150 const Scalar delta = diff.squaredNorm() / p;
151 if (delta < Scalar(1e-30)) {
152 return {S, Scalar(0)};
153 }
154
155 // beta = sum_k ||x_k x_k^T - S||_F^2 / (n^2 p) = (sum_k |x_k|^4 - n ||S||_F^2) / (n^2 p)
156 const Scalar sumSquaredNorms = data.colwise().squaredNorm().array().square().sum();
157 const Scalar beta = (sumSquaredNorms - n * S.squaredNorm()) / (n * n * p);
158 return detail::shrink<Scalar>(S, std::clamp(beta / delta, Scalar(0), Scalar(1)));
159}

170template<class Derived>
171ShrunkCovariance<typename Derived::Scalar> oas_covariance(const Eigen::MatrixBase<Derived>& data)
172{
173 using Scalar = typename Derived::Scalar;
174 const auto p = static_cast<Scalar>(data.rows());
175 const auto n = static_cast<Scalar>(data.cols());
176
177 const Eigen::Matrix<Scalar, Eigen::Dynamic, Eigen::Dynamic> S = (data * data.transpose()) / n;
178 const Scalar mu = S.trace() / p;
179
180 // With alpha = ||S||_F^2 / p^2: rho = (alpha + mu^2) / ((n + 1) (alpha - mu^2 / p))
181 const Scalar alpha = S.squaredNorm() / (p * p);
182 const Scalar den = (n + Scalar(1)) * (alpha - mu * mu / p);
183 const Scalar rho = std::abs(den) > Scalar(1e-30) ? std::clamp((alpha + mu * mu) / den, Scalar(0), Scalar(1)) : Scalar(1);
184 return detail::shrink<Scalar>(S, rho);
185}

200template<class Derived>
201FactorAnalysisCovariance<typename Derived::Scalar> factor_analysis_covariance(const Eigen::MatrixBase<Derived>& data,
202 int n_factors = 0,
203 int max_iter = 200,
204 typename Derived::Scalar tol = typename Derived::Scalar(1e-6))
205{
206 using Scalar = typename Derived::Scalar;
207 using Matrix = Eigen::Matrix<Scalar, Eigen::Dynamic, Eigen::Dynamic>;
208 using Vector = Eigen::Matrix<Scalar, Eigen::Dynamic, 1>;
209 constexpr Scalar tiny = Scalar(1e-30);
210 const Eigen::Index p = data.rows();
211 const auto n = static_cast<Scalar>(data.cols());
212
213 const Matrix centred = data.colwise() - data.rowwise().mean();
214 const Matrix S = (centred * centred.transpose()) / n;
215
216 int k = n_factors > 0 ? n_factors : std::max(1, static_cast<int>(std::min<Eigen::Index>(data.cols(), p)) / 2);
217 k = std::min(k, static_cast<int>(p) - 1);
218
219 FactorAnalysisCovariance<Scalar> model;
220 Vector psi = S.diagonal();
221 Matrix sigma = psi.asDiagonal();
222 Scalar previous = -std::numeric_limits<Scalar>::infinity();
223 for (int iter = 0; iter < max_iter; ++iter) {
224 const Vector psiInvSqrt = psi.array().max(tiny).inverse().sqrt();
225 const Eigen::SelfAdjointEigenSolver<Matrix> whitened(psiInvSqrt.asDiagonal() * S * psiInvSqrt.asDiagonal());
226 const Vector scale = (whitened.eigenvalues().tail(k).array() - Scalar(1)).max(Scalar(0)).sqrt();
227 model.loadings = psi.array().sqrt().matrix().asDiagonal() * whitened.eigenvectors().rightCols(k) * scale.asDiagonal();
228
229 sigma = model.loadings * model.loadings.transpose();
230 psi = (S - sigma).diagonal().array().max(tiny);
231 sigma.diagonal() += psi;
232
233 const Eigen::SelfAdjointEigenSolver<Matrix> fitted(sigma);
234 const Vector evals = fitted.eigenvalues().array().max(tiny);
235 const Matrix sigmaInv = fitted.eigenvectors() * evals.array().inverse().matrix().asDiagonal() * fitted.eigenvectors().transpose();
236 model.log_likelihood = Scalar(-0.5) * n * (static_cast<Scalar>(p) * std::log(Scalar(2) * std::numbers::pi_v<Scalar>) + evals.array().log().sum() + (sigmaInv * S).trace());
237 model.iterations = iter + 1;
238 if (std::abs(model.log_likelihood - previous) < tol) {
239 break;
240 }
241 previous = model.log_likelihood;
242 }
243 model.covariance = std::move(sigma);
244 model.noise_variance = std::move(psi);
245 return model;
246}
