132InvGuessData::InvGuessData(const QString &guessname, const QString &guess_surfname, float mindist, float exclude, float grid, InvDipoleFitData *f)
133{
134// InvGuessData* res = new InvGuessData();
135 int k,p;
136 float guessrad = 0.080;
137 std::unique_ptr<MNESourceSpace> guesses;
138 dipoleFitFuncs orig;
139
140 if (!guessname.isEmpty()) {
141 /*
142 * Read the guesses and transform to the appropriate coordinate frame
143 */
144 std::vector<std::unique_ptr<MNESourceSpace>> sp;
145 if (MNESourceSpace::read_source_spaces(guessname,sp) == FAIL)
146 goto bad;
147 if (static_cast<int>(sp.size()) != 1) {
148 printf("Incorrect number of source spaces in guess file");
149 goto bad;
150 }
151 printf("Read guesses from %s\n",guessname.toUtf8().constData());
152 guesses = std::move(sp[0]);
153 }
154 else {
155 MNESurface* inner_skull = nullptr;
156 int free_inner_skull = FALSE;
157 Eigen::Vector3f r0 = f->r0;
158
159 Q_ASSERT(f->mri_head_t);
160 FiffCoordTrans::apply_inverse_trans(r0.data(),*f->mri_head_t,TRUE);
161 if (f->bem_model) {
162 printf("Using inner skull surface from the BEM (%s)...\n",f->bemname.toUtf8().constData());
163 if ((inner_skull = f->bem_model->fwd_bem_find_surface(FIFFV_BEM_SURF_ID_BRAIN)) == nullptr)
164 goto bad;
165 }
166 else if (!guess_surfname.isEmpty()) {
167 printf("Reading inner skull surface from %s...\n",guess_surfname.toUtf8().data());
168 if ((inner_skull = MNESurface::read_bem_surface(guess_surfname,FIFFV_BEM_SURF_ID_BRAIN,true)) == nullptr)
169 goto bad;
170 free_inner_skull = TRUE;
171 }
172 guesses.reset((MNESourceSpace*)FwdBemModel::make_guesses(inner_skull,guessrad,r0,grid,exclude,mindist).release());
173 if (!guesses)
174 goto bad;
175 if (free_inner_skull)
176 delete inner_skull;
177 }
178 {
179 std::vector<std::unique_ptr<MNESourceSpace>> guesses_vec;
180 guesses_vec.push_back(std::move(guesses));
181 if (MNESourceSpace::transform_source_spaces_to(f->coord_frame,*f->mri_head_t,guesses_vec) != OK)
182 goto bad;
183 guesses = std::move(guesses_vec[0]);
184 }
185 printf("Guess locations are now in %s coordinates.\n",FiffCoordTrans::frame_name(f->coord_frame).toUtf8().constData());
186
187 this->nguess = guesses->nuse;
188 this->rr.resize(guesses->nuse, 3);
189 for (k = 0, p = 0; k < guesses->np; k++)
190 if (guesses->inuse[k]) {
191 this->rr.row(p) = guesses->rr.row(k);
192 p++;
193 }
194 guesses.reset();
195
196 printf("Go through all guess source locations...");
197 this->guess_fwd.resize(this->nguess);
198 /*
199 * Compute the guesses using the sphere model for speed
200 */
201 orig = f->funcs;
202 if (f->fit_mag_dipoles)
203 f->funcs = f->mag_dipole_funcs.get();
204 else
205 f->funcs = f->sphere_funcs.get();
206
207 for (k = 0; k < this->nguess; k++) {
208 this->guess_fwd[k].reset(InvDipoleFitData::dipole_forward_one(f,Eigen::Vector3f(this->rr.row(k).transpose()),nullptr));
209 if (!this->guess_fwd[k])
210 goto bad;
211#ifdef DEBUG
212 sing = this->guess_fwd[k]->sing;
213 printf("%f %f %f\n",sing[0],sing[1],sing[2]);
214#endif
215 }
216 f->funcs = orig;
217
218 printf("[done %d sources]\n",p);
219
220 return;
221// return res;
222
223bad : {
224 return;
225// return nullptr;
226 }
227}

329bool InvGuessData::compute_guess_fields(InvDipoleFitData* f)
330{
331 dipoleFitFuncs orig = nullptr;
332
333 if (!f) {
334 qCritical("Data missing in compute_guess_fields");
335 return false;
336 }
337 if (!f->noise) {
338 qCritical("Noise covariance missing in compute_guess_fields");
339 return false;
340 }
341 printf("Go through all guess source locations...");
342 orig = f->funcs;
343 if (f->fit_mag_dipoles)
344 f->funcs = f->mag_dipole_funcs.get();
345 else
346 f->funcs = f->sphere_funcs.get();
347 for (int k = 0; k < this->nguess; k++) {
348 this->guess_fwd[k].reset(InvDipoleFitData::dipole_forward_one(f,Eigen::Vector3f(this->rr.row(k).transpose()),this->guess_fwd[k].release()));
349 if (!this->guess_fwd[k]) {
350 if (orig)
351 f->funcs = orig;

353 }

361
362 return true;
