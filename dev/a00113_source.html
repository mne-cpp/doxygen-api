46static std::vector<int> findGoodEegIndices(const FiffInfo& info)
47{
48 std::vector<int> indices;
49 for (int i = 0; i < info.nchan; ++i) {

76void mne::setEegReference(MatrixXd& data,
77 FiffInfo& info,
78 const StringList& refChannels,
79 bool projection)
80{
81 const bool useAverage = refChannels.empty() || (refChannels.size() == 1 && mne::iequals(refChannels.front(), "average"));
82
83 if (projection) {
84 if (!useAverage) {
85 Log::warning("setEegReference: projection is only supported for the average reference.");
86 return;
87 }
88 for (const FiffProj& proj : std::as_const(info.projs)) {
89 if (proj.kind == FIFFV_PROJ_ITEM_EEG_AVREF)
90 return;
91 }
92 StringList names;
93 for (int idx : findGoodEegIndices(info))
94 names.push_back(info.ch_names[idx]);
95 if (names.empty()) {
96 Log::warning("setEegReference: no good EEG channels for an average reference projector.");
97 return;
98 }
99 FiffNamedMatrix vec(1, static_cast<int>(names.size()), StringList(), names,
100 MatrixXd::Constant(1, names.size(), 1.0 / std::sqrt(static_cast<double>(names.size()))));
101 info.projs.push_back(FiffProj(FIFFV_PROJ_ITEM_EEG_AVREF, false, "Average EEG reference", vec));
102 return;
103 }
104
105 if (data.rows() != info.nchan) {
106 Log::warning("setEegReference: data row count ({}) does not match info.nchan ({}).", static_cast<long long>(data.rows()), info.nchan);
107 return;
108 }
109
110 if (data.cols() == 0 || data.rows() == 0) {
111 return;
112 }
113
114 // As mne.set_eeg_reference, bad EEG channels are neither reference nor re-referenced
115 const std::vector<int> goodEegIdx = findGoodEegIndices(info);
116 if (findAllEegIndices(info).empty()) {
117 Log::warning("setEegReference: no EEG channels found in info.");
118 return;
119 }
120
121 // Compute reference signal
122 RowVectorXd refSignal;
123
124 if (useAverage) {
125 // Average reference: mean of all good EEG channels
126 if (goodEegIdx.empty()) {
127 Log::warning("setEegReference: all EEG channels are marked as bad.");
128 return;
129 }
130 refSignal = RowVectorXd::Zero(data.cols());
131 for (int idx : goodEegIdx) {
132 refSignal += data.row(idx);
133 }
134 refSignal /= static_cast<double>(goodEegIdx.size());
135 } else {
136 // Specific channel(s): mean of named channels
137 std::vector<int> refIdx;
138 for (const std::string& name : refChannels) {
139 int idx = mne::channel_index(info.ch_names, name);
140 if (idx < 0) {
141 Log::warning("setEegReference: reference channel '{}' not found.", name);
142 return;
143 }
144 refIdx.push_back(idx);
145 }
146 refSignal = RowVectorXd::Zero(data.cols());
147 for (int idx : refIdx) {
148 refSignal += data.row(idx);
149 }
150 refSignal /= static_cast<double>(refIdx.size());
151 }
152
153 for (int idx : goodEegIdx) {
154 data.row(idx) -= refSignal;
155 }
156}

160void mne::addReferenceChannels(MatrixXd& data,
161 FiffInfo& info,
162 const StringList& chNames)
163{
164 if (chNames.empty()) {
165 return;
166 }
167
168 const Eigen::Index nTimes = data.cols();
169 const Eigen::Index nOldCh = data.rows();
170 const int nNew = static_cast<int>(chNames.size());
171
172 // Expand data matrix with zero rows
173 MatrixXd newData(nOldCh + nNew, nTimes);
174 if (nOldCh > 0 && nTimes > 0) {
175 newData.topRows(nOldCh) = data;
176 }
177 newData.bottomRows(nNew).setZero();
178 data = newData;
179
180 // Find a template EEG channel for defaults
181 FiffChInfo templateCh;
182 bool haveTemplate = false;
183 for (int i = 0; i < info.nchan; ++i) {
184 if (info.chs[i].kind == FIFFV_EEG_CH) {
185 templateCh = info.chs[i];
186 haveTemplate = true;
187 break;
188 }
189 }
190
191 for (int n = 0; n < nNew; ++n) {
192 FiffChInfo ch;
193 if (haveTemplate) {
194 ch = templateCh;
195 }
196 ch.ch_name = chNames[n];
197 ch.kind = FIFFV_EEG_CH;
198 ch.scanNo = info.nchan + n + 1;
199 ch.logNo = ch.scanNo;
200
201 info.chs.push_back(ch);
202 info.ch_names.push_back(chNames[n]);
203 }
204 info.nchan += nNew;
205}

209void mne::setBipolarReference(MatrixXd& data,
210 FiffInfo& info,
211 const StringList& anodes,
212 const StringList& cathodes,
213 bool dropOriginals)
214{
215 if (anodes.size() != cathodes.size()) {
216 Log::warning("setBipolarReference: anodes and cathodes must have the same length ({} vs {}).", static_cast<long long>(anodes.size()), static_cast<long long>(cathodes.size()));
217 return;
218 }
219
220 if (anodes.empty()) {
221 return;
222 }
223
224 if (data.rows() != info.nchan) {
225 Log::warning("setBipolarReference: data row count ({}) does not match info.nchan ({}).", static_cast<long long>(data.rows()), info.nchan);
226 return;
227 }
228
229 const int nPairs = static_cast<int>(anodes.size());
230 const Eigen::Index nTimes = data.cols();
231
232 // Resolve anode and cathode indices
233 std::vector<int> anodeIdx(nPairs), cathodeIdx(nPairs);
234 for (int i = 0; i < nPairs; ++i) {
235 anodeIdx[i] = mne::channel_index(info.ch_names, anodes[i]);
236 if (anodeIdx[i] < 0) {
237 Log::warning("setBipolarReference: anode channel '{}' not found.", anodes[i]);
238 return;
239 }
240 cathodeIdx[i] = mne::channel_index(info.ch_names, cathodes[i]);
241 if (cathodeIdx[i] < 0) {
242 Log::warning("setBipolarReference: cathode channel '{}' not found.", cathodes[i]);
243 return;
244 }
245 }
246
247 // Collect the set of original channels involved
248 std::vector<bool> bipolarOriginals(info.chs.size(), false);
249 for (int i = 0; i < nPairs; ++i) {
250 bipolarOriginals[anodeIdx[i]] = true;
251 bipolarOriginals[cathodeIdx[i]] = true;
252 }
253
254 // Build bipolar data rows
255 MatrixXd bipolarData(nPairs, nTimes);
256 for (int i = 0; i < nPairs; ++i) {
257 bipolarData.row(i) = data.row(anodeIdx[i]) - data.row(cathodeIdx[i]);
258 }
259
260 // Build bipolar channel info
261 std::vector<FiffChInfo> bipolarChs;
262 StringList bipolarNames;
263 for (int i = 0; i < nPairs; ++i) {
264 FiffChInfo ch = info.chs[anodeIdx[i]];
265 ch.ch_name = (anodes[i] + "-" + cathodes[i]);
266 ch.logNo = i + 1;
267 bipolarChs.push_back(ch);
268 bipolarNames.push_back(ch.ch_name);
269 }
270
271 if (dropOriginals) {
272 // Keep only non-bipolar-original rows + append bipolar rows
273 std::vector<int> keepIdx;
274 for (int i = 0; i < static_cast<int>(data.rows()); ++i) {
275 if (!bipolarOriginals[i]) {
276 keepIdx.push_back(i);
277 }
278 }
279
280 const int nKeep = static_cast<int>(keepIdx.size());
281 MatrixXd newData(nKeep + nPairs, nTimes);
282 std::vector<FiffChInfo> newChs;
283 StringList newNames;
284
285 for (int i = 0; i < nKeep; ++i) {
286 newData.row(i) = data.row(keepIdx[i]);
287 newChs.push_back(info.chs[keepIdx[i]]);
288 newNames.push_back(info.ch_names[keepIdx[i]]);
289 }
290 for (int i = 0; i < nPairs; ++i) {
291 newData.row(nKeep + i) = bipolarData.row(i);
292 newChs.push_back(bipolarChs[i]);
293 newNames.push_back(bipolarNames[i]);
294 }
295
296 // Update scan numbers
297 for (int i = 0; i < std::ssize(newChs); ++i) {
298 newChs[i].scanNo = i + 1;
299 }
300
301 data = newData;
302 info.chs = newChs;
303 info.ch_names = newNames;
304 info.nchan = static_cast<int>(newData.rows());
305 } else {
306 // Keep all original rows, append bipolar rows at the end
307 MatrixXd newData(data.rows() + nPairs, nTimes);
308 newData.topRows(data.rows()) = data;
309 newData.bottomRows(nPairs) = bipolarData;
310
311 for (int i = 0; i < nPairs; ++i) {
312 bipolarChs[i].scanNo = info.nchan + i + 1;
313 }
314
315 info.chs.insert(info.chs.end(), bipolarChs.begin(), bipolarChs.end());
316 info.ch_names.insert(info.ch_names.end(), bipolarNames.begin(), bipolarNames.end());
317 info.nchan += nPairs;
318 data = newData;
319 }
320}
