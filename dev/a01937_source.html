56std::optional<std::chrono::sys_time<std::chrono::microseconds>> parseMarkerDate(std::string_view digits)
57{
58 using namespace std::chrono;
59 const auto field = [digits](std::size_t pos, std::size_t count) {
60 return mne::to_int(digits.substr(pos, count)).value_or(-1);
61 };
62 const year_month_day date{year{field(0, 4)}, month(static_cast<unsigned>(field(4, 2))), day(static_cast<unsigned>(field(6, 2)))};
63 const int hour = field(8, 2);
64 const int minute = field(10, 2);
65 const int second = field(12, 2);
66 if (!date.ok() || hour > 23 || minute > 59 || second > 59) {
67 return std::nullopt;
68 }
69 return sys_days{date} + hours(hour) + minutes(minute) + seconds(second) + microseconds(field(14, 6));
70}

76std::string decodeText(std::string bytes)
77{
78 // A line start without std::regex::multiline, which MSVC's library does not have
79 static const std::regex codepagePattern("(?:^|\\n)[ \\t]*Codepage[ \\t]*=[ \\t]*(\\S+)", std::regex::icase);
80 std::smatch codepage;
81 const bool ansi = std::regex_search(bytes, codepage, codepagePattern) && codepage[1] == "ANSI";
82 if (!ansi && mne::is_utf8(bytes)) {
83 return bytes;
84 }
85 if (!ansi) {
86 return mne::latin1_to_utf8(bytes);
87 }
88 // Windows-1252 0x80-0x9F; empty entries are undefined there
89 static constexpr std::array<std::string_view, 32> cp1252{
90 "\xE2\x82\xAC", "", "\xE2\x80\x9A", "\xC6\x92", "\xE2\x80\x9E", "\xE2\x80\xA6", "\xE2\x80\xA0", "\xE2\x80\xA1",
91 "\xCB\x86", "\xE2\x80\xB0", "\xC5\xA0", "\xE2\x80\xB9", "\xC5\x92", "", "\xC5\xBD", "",
92 "", "\xE2\x80\x98", "\xE2\x80\x99", "\xE2\x80\x9C", "\xE2\x80\x9D", "\xE2\x80\xA2", "\xE2\x80\x93", "\xE2\x80\x94",
93 "\xCB\x9C", "\xE2\x84\xA2", "\xC5\xA1", "\xE2\x80\xBA", "\xC5\x93", "", "\xC5\xBE", "\xC5\xB8"};
94 std::string utf8;
95 utf8.reserve(bytes.size());
96 for (const char c : bytes) {
97 const auto byte = static_cast<unsigned char>(c);
98 if (byte >= 0x80 && byte < 0xA0) {
99 if (cp1252[byte - 0x80].empty()) {
100 return mne::latin1_to_utf8(bytes);
101 }
102 utf8 += cp1252[byte - 0x80];
103 } else if (byte >= 0x80) {
104 utf8.push_back(static_cast<char>(0xC0 | (byte >> 6)));
105 utf8.push_back(static_cast<char>(0x80 | (byte & 0x3F)));
106 } else {
107 utf8.push_back(c);
108 }
109 }
110 return utf8;
111}

114std::string decodeCommas(std::string_view text)
115{
116 std::string decoded;
117 decoded.reserve(text.size());
118 for (std::size_t i = 0; i < text.size(); ++i) {
119 if (text[i] == '\\' && i + 1 < text.size() && text[i + 1] == '1') {
120 decoded.push_back(',');
121 ++i;
122 } else {
123 decoded.push_back(text[i]);
124 }
125 }
126 return decoded;
127}

142template<typename Entry>
143void forEachEntry(std::string_view text, Entry entry)
144{
145 std::string_view section;
146 bool versionLine = true;
147 while (!text.empty()) {
148 const std::size_t newline = text.find('\n');
149 const std::string_view line = mne::trimmed(text.substr(0, newline));
150 text = newline == std::string_view::npos ? std::string_view() : text.substr(newline + 1);
151 if (std::exchange(versionLine, false) || line.empty() || line.front() == ';') {
152 continue;
153 }
154 if (line.front() == '[' && line.back() == ']') {
155 section = line.substr(1, line.size() - 2);
156 if (mne::iequals(section, "comment")) {
157 return;
158 }
159 continue;
160 }
161 const std::size_t eq = line.find('=');
162 if (eq != std::string_view::npos && eq > 0 && !section.empty()) {
163 entry(section, mne::trimmed(line.substr(0, eq)), mne::trimmed(line.substr(eq + 1)));
164 }
165 }
166}

179template<mne::Swappable T>
180T fromLittleEndian(const unsigned char* bytes)
181{
182 T value;
183 std::memcpy(&value, bytes, sizeof(T));
184 if constexpr (std::endian::native == std::endian::big) {
185 value = mne::byte_swap(value);
186 }

201FiffChInfo BrainVisionChannelInfo::to_fiff_ch_info() const
202{
203 FiffChInfo info;
204 info.scanNo = channelNumber;
205 info.logNo = channelNumber;
206
207 // Detect channel type from name
208 const std::string lower = mne::to_lower(name);
209 const auto has = [&lower](std::string_view part) {
210 return lower.find(part) != std::string::npos;
211 };
212 if (has("ecog")) {
213 info.kind = FIFFV_ECOG_CH;
214 } else if (has("seeg")) {
215 info.kind = FIFFV_SEEG_CH;
216 } else if (has("eog") || lower == "heogl" || lower == "heogr" || lower == "veogb") {
217 info.kind = FIFFV_EOG_CH;
218 } else if (has("ecg") || has("ekg")) {
219 info.kind = FIFFV_ECG_CH;
220 } else if (has("emg")) {
221 info.kind = FIFFV_EMG_CH;
222 } else if (lower == "sti 014" || has("stim")) {
223 info.kind = FIFFV_STIM_CH;
224 } else {
225 // Like mne read_raw_brainvision: voltage channels are EEG, anything else (ARU, µS, °C, ...) is misc
226 info.kind = BrainVisionReader::is_voltage_unit(unit) ? FIFFV_EEG_CH : FIFFV_MISC_CH;
227 }
228
229 info.unit = FIFF_UNIT_NONE;
230 info.unit_mul = FIFF_UNITM_NONE;
231 if (BrainVisionReader::is_voltage_unit(unit)) {
232 info.unit = FIFF_UNIT_V;
233 if (unit == "nV") {
234 info.unit_mul = FIFF_UNITM_N;
235 } else if (unit == kMicroVolt || unit == "uV") {
236 info.unit_mul = FIFF_UNITM_MU;
237 } else if (unit == "mV") {
238 info.unit_mul = FIFF_UNITM_M;
239 }
240 }
241
242 // Calibration: resolution is the scaling factor from raw → physical units
243 info.cal = resolution;
244 info.range = 1.0f;
245 info.ch_name = name;
246
247 return info;
248}

284bool BrainVisionReader::parse_header(const std::filesystem::path& vhdrPath)
285{
286 mne::Expected<std::string> text = mne::read_text_file(vhdrPath);
287 if (!text) {
288 mne::Log::warning("[BrainVisionReader::parse_header] Could not open header: {}", mne::utf8_from_path(vhdrPath));
289 return false;
290 }
291
292 *text = decodeText(std::move(*text));
293 const std::string_view firstLine = mne::trimmed(std::string_view(*text).substr(0, text->find('\n')));
294 static const std::regex versionPattern("Brain ?Vision( Core| V-Amp)? Data( Exchange)? Header File,? Version [12]\\.0", std::regex::icase);
295 if (!std::regex_search(firstLine.begin(), firstLine.end(), versionPattern)) {
296 mne::Log::warning("[BrainVisionReader::parse_header] Unrecognized header version: {}", firstLine);
297 return false;
298 }
299
300 Sections sections;
301 forEachEntry(*text, [&sections](std::string_view section, std::string_view key, std::string_view entry) {
302 sections[std::string(section)].insert_or_assign(std::string(key), std::string(entry));
303 });
304
305 // "Common infos" is the NeurOne spelling
306 auto common = sections.find("Common Infos");
307 if (common == sections.end()) {
308 common = sections.find("Common infos");
309 }
310 if (common == sections.end() || common->second.empty()) {
311 mne::Log::warning("[BrainVisionReader::parse_header] Missing [Common Infos] section");
312 return false;
313 }
314 const auto& commonInfos = common->second;
315
316 // Data and marker files are relative to the .vhdr
317 const std::string_view dataFile = value(commonInfos, "DataFile");
318 if (dataFile.empty()) {
319 mne::Log::warning("[BrainVisionReader::parse_header] No DataFile specified");
320 return false;
321 }
322 const std::filesystem::path dir = std::filesystem::absolute(vhdrPath).parent_path();
323 m_dataPath = dir / mne::path_from_utf8(dataFile);
324 const std::string_view markerFile = value(commonInfos, "MarkerFile");
325 m_markerPath = markerFile.empty() ? std::filesystem::path() : dir / mne::path_from_utf8(markerFile);
326
327 const std::string_view dataFormat = value(commonInfos, "DataFormat", "BINARY");
328 if (!mne::iequals(dataFormat, "BINARY")) {
329 mne::Log::warning("[BrainVisionReader::parse_header] Only BINARY data is supported, not {}", dataFormat);
330 return false;
331 }
332
333 m_orientation = mne::iequals(value(commonInfos, "DataOrientation", "MULTIPLEXED"), "VECTORIZED") ? BVOrientation::VECTORIZED : BVOrientation::MULTIPLEXED;
334
335 m_numChannels = mne::to_int(value(commonInfos, "NumberOfChannels", "0")).value_or(0);
336 if (m_numChannels <= 0) {
337 mne::Log::warning("[BrainVisionReader::parse_header] Invalid channel count: {}", m_numChannels);
338 return false;
339 }
340
341 // Sampling interval in microseconds
342 const double samplingInterval = mne::to_double(value(commonInfos, "SamplingInterval", "0")).value_or(0.0);
343 m_sfreq = samplingInterval > 0.0 ? static_cast<float>(1.0e6 / samplingInterval) : 0.0f;
344
345 static const std::map<std::string, std::string, std::less<>> none;
346 const auto section = [&sections](std::string_view name) -> const std::map<std::string, std::string, std::less<>>& {
347 const auto it = sections.find(name);
348 return it == sections.end() ? none : it->second;
349 };
350
351 const std::string_view binaryFormat = value(section("Binary Infos"), "BinaryFormat", "INT_16");
352 if (mne::iequals(binaryFormat, "INT_32")) {
353 m_binaryFormat = BVBinaryFormat::INT_32;
354 } else if (mne::iequals(binaryFormat, "IEEE_FLOAT_32")) {
355 m_binaryFormat = BVBinaryFormat::IEEE_FLOAT_32;
356 } else {
357 m_binaryFormat = BVBinaryFormat::INT_16;
358 }
359
360 // Ch<N>=<Name>,<Reference>,<Resolution>,<Unit>
361 const auto& channelInfos = section("Channel Infos");
362 m_channels.clear();
363 m_channels.reserve(static_cast<std::size_t>(m_numChannels));
364 for (int i = 1; i <= m_numChannels; ++i) {
365 BrainVisionChannelInfo ch;
366 ch.channelNumber = i - 1;
367 ch.unit = kMicroVolt;
368 const std::string_view entry = value(channelInfos, "Ch" + std::to_string(i));
369 if (entry.empty()) {
370 ch.name = "Ch" + std::to_string(i);
371 } else {
372 const std::vector<std::string> parts = mne::split(entry, ',');
373 ch.name = decodeCommas(parts[0]);
374 if (parts.size() >= 2) {
375 ch.reference = decodeCommas(parts[1]);
376 }
377 if (parts.size() >= 3 && !parts[2].empty()) {
378 ch.resolution = static_cast<float>(mne::to_double(parts[2]).value_or(0.0));
379 }
380 if (parts.size() >= 4 && !parts[3].empty()) {
381 ch.unit = parts[3];
382 }
383 }
384 m_channels.push_back(std::move(ch));
385 }
386 return true;
387}

391bool BrainVisionReader::parse_markers(const std::filesystem::path& vmrkPath)
392{
393 mne::Expected<std::string> text = mne::read_text_file(vmrkPath);
394 if (!text) {
395 mne::Log::warning("[BrainVisionReader::parse_markers] Could not open marker file: {}", mne::utf8_from_path(vmrkPath));
396 return false;
397 }
398
399 // Mk<N>=<Type>,<Description>,<Position>,<Duration>,<Channel>[,<Date>]
400 *text = decodeText(std::move(*text));
401 bool measDateFound = false;
402 forEachEntry(*text, [this, &measDateFound](std::string_view section, std::string_view, std::string_view entry) {
403 if (section != "Marker Infos") {
404 return;
405 }
406 const std::vector<std::string> parts = mne::split(entry, ',');
407 if (parts.size() < 5) {
408 return;
409 }
410 BrainVisionMarker marker;
411 marker.type = decodeCommas(parts[0]);
412 marker.description = decodeCommas(parts[1]);
413 marker.position = toLong(parts[2], 0) - 1; // 1-based in the file
414 marker.duration = toLong(parts[3], 0);
415 marker.channel = static_cast<int>(toLong(parts[4], 0));
416
417 // Like mne, the recording start is the date of the first New Segment marker that has one
418 const bool hasDate = parts.size() == 6 && parts[5].size() == 20 && std::ranges::all_of(parts[5], [](char c) { return c >= '0' && c <= '9'; });
419 if (hasDate) {
420 marker.date = parseMarkerDate(parts[5]);
421 if (marker.type == "New Segment" && !std::exchange(measDateFound, true)) {
422 m_measDate = marker.date;
423 }
424 }
425 m_markers.push_back(std::move(marker));
426 });
427 return true;
428}

432float BrainVisionReader::unit_scale(std::string_view unit)
433{
434 // mne.io.brainvision _unit_dict; units it does not know are taken as-is (factor 1)
435 if (unit == kMicroVolt || unit == "uV" || unit == "\xC2\xB5S" || unit == "uS") {
436 return 1.0e-6f;
437 }
438 if (unit == "mV") {
439 return 1.0e-3f;
440 }
441 if (unit == "nV") {
442 return 1.0e-9f;
443 }
444 return 1.0f;
445}

475Eigen::MatrixXf BrainVisionReader::read_raw_segment(long first, long last) const
476{
477 if (!m_dataFile) {
478 mne::Log::warning("[BrainVisionReader::read_raw_segment] File not open");
479 return MatrixXf();
480 }
481 if (first < 0 || first >= m_sampleCount || last < 0 || last > m_sampleCount || last <= first) {
482 mne::Log::warning("[BrainVisionReader::read_raw_segment] Invalid range: {} - {}", first, last);
483 return MatrixXf();
484 }
485
486 const long numSamples = last - first;
487 const auto valueBytes = static_cast<std::size_t>(bytesPerValue(m_binaryFormat));
488 const auto channels = static_cast<std::size_t>(m_numChannels);
489 std::vector<float> scales(channels);
490 for (std::size_t ch = 0; ch < channels; ++ch) {
491 scales[ch] = m_channels[ch].resolution * unit_scale(m_channels[ch].unit);
492 }
493 const auto decode = [this](const unsigned char* bytes) -> float {
494 switch (m_binaryFormat) {
495 case BVBinaryFormat::INT_16:
496 return static_cast<float>(fromLittleEndian<std::int16_t>(bytes));
497 case BVBinaryFormat::INT_32:
498 return static_cast<float>(fromLittleEndian<std::int32_t>(bytes));
499 case BVBinaryFormat::IEEE_FLOAT_32:
500 return fromLittleEndian<float>(bytes);
501 }
502 return 0.0f;
503 };
504
505 MatrixXf result(m_numChannels, numSamples);
506 std::vector<unsigned char> buffer;
507 if (m_orientation == BVOrientation::MULTIPLEXED) {
508 // [ch1_t1, ch2_t1, ..., chN_t1, ch1_t2, ...]
509 buffer.resize(static_cast<std::size_t>(numSamples) * channels * valueBytes);
510 m_dataFile->seek(static_cast<std::uint64_t>(first) * channels * valueBytes);
511 m_dataFile->read(buffer.data(), buffer.size());
512 const unsigned char* bytes = buffer.data();
513 for (Index s = 0; s < numSamples; ++s) {
514 for (std::size_t ch = 0; ch < channels; ++ch, bytes += valueBytes) {
515 result(static_cast<Index>(ch), s) = decode(bytes) * scales[ch];
516 }
517 }
518 } else {
519 // [ch1_t1, ch1_t2, ..., ch1_tM, ch2_t1, ...]
520 buffer.resize(static_cast<std::size_t>(numSamples) * valueBytes);
521 for (std::size_t ch = 0; ch < channels; ++ch) {
522 m_dataFile->seek((static_cast<std::uint64_t>(ch) * static_cast<std::uint64_t>(m_sampleCount) + static_cast<std::uint64_t>(first)) * valueBytes);
523 m_dataFile->read(buffer.data(), buffer.size());
524 const unsigned char* bytes = buffer.data();
525 for (Index s = 0; s < numSamples; ++s, bytes += valueBytes) {
526 result(static_cast<Index>(ch), s) = decode(bytes) * scales[ch];
527 }
528 }
529 }
530 return result;
531}
