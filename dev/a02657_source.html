93struct RegionData
94{
95 Eigen::MatrixXd matRoiG;
96 Eigen::MatrixXd matRoiGWhitened;
97 bool bUseWhitened;
98
99 Eigen::MatrixXd matRoiGOrig;
100
101 std::int32_t nClusters;
102
103 Eigen::VectorXi idcs;
104 std::int32_t iLabelIdxIn;
105 std::string sDistMeasure;
106
107 RegionDataOut cluster() const
108 {
109 const std::string t_sDistMeasure = sDistMeasure.empty() ? std::string("cityblock") : sDistMeasure;
110
111 // Kmeans Reduction
112 RegionDataOut p_RegionDataOut;
113
114 mne::KMeans t_kMeans(t_sDistMeasure, "sample", 5);
115
116 if (bUseWhitened) {
117 t_kMeans.calculate(this->matRoiGWhitened, this->nClusters, p_RegionDataOut.roiIdx, p_RegionDataOut.ctrs, p_RegionDataOut.sumd, p_RegionDataOut.D);
118
119 // Centroids of the original gain: all channels, not only those the whitener kept
120 Eigen::MatrixXd newCtrs = Eigen::MatrixXd::Zero(p_RegionDataOut.ctrs.rows(), this->matRoiG.cols());
121 for (std::int32_t c = 0; c < p_RegionDataOut.ctrs.rows(); ++c) {
122 std::int32_t num = 0;
123
124 for (std::int32_t idx = 0; idx < p_RegionDataOut.roiIdx.size(); ++idx) {
125 if (c == p_RegionDataOut.roiIdx[idx]) {
126 newCtrs.row(c) += this->matRoiG.row(idx); //just take whitened to get indeces calculate centroids using the original matrix
127 ++num;
128 }
129 }
130
131 if (num > 0)
132 newCtrs.row(c) /= num;
133 }
134 p_RegionDataOut.ctrs = newCtrs; //Replace whitened with original
135 } else
136 t_kMeans.calculate(this->matRoiG, this->nClusters, p_RegionDataOut.roiIdx, p_RegionDataOut.ctrs, p_RegionDataOut.sumd, p_RegionDataOut.D);
137
138 p_RegionDataOut.iLabelIdxOut = this->iLabelIdxIn;
139
140 return p_RegionDataOut;
141 }
142};
