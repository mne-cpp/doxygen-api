967 // Reads one value in the stream's byte order; a short read yields 0 and marks the stream failed
968 template<mne::Swappable T>
969 void read(T& value)
970 {
971 if (m_bytes->read(&value, sizeof(T)) != sizeof(T)) {
972 value = T{};
973 m_readFailed = true;
974 return;
975 }
976 if (m_littleEndian != (std::endian::native == std::endian::little))
977 value = mne::byte_swap(value);

988 // Writes @p count values in the stream's byte order with as few writes as possible
989 template<mne::Swappable T>
990 void write_values(const T* values, std::size_t count)
991 {
992 if (m_littleEndian == (std::endian::native == std::endian::little)) {
993 m_bytes->write(values, count * sizeof(T));
994 return;
995 }
996 constexpr std::size_t kChunk = 4096;
997 std::array<T, kChunk> swapped;
998 for (std::size_t done = 0; done < count;) {
999 const std::size_t n = std::min(kChunk, count - done);
1000 for (std::size_t i = 0; i < n; ++i)
1001 swapped[i] = mne::byte_swap(values[done + i]);
1002 m_bytes->write(swapped.data(), n * sizeof(T));
1003 done += n;
1004 }
1005 }
