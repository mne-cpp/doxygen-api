966 // Reads one value in the stream's byte order; a short read yields 0 and marks the stream failed
967 template<mne::Swappable T>
968 void read(T& value)
969 {
970 if (m_bytes->read(&value, sizeof(T)) != sizeof(T)) {
971 value = T{};
972 m_readFailed = true;
973 return;
974 }
975 if (m_littleEndian != (std::endian::native == std::endian::little))
976 value = mne::byte_swap(value);

987 // Writes @p count values in the stream's byte order with as few writes as possible
988 template<mne::Swappable T>
989 void write_values(const T* values, std::size_t count)
990 {
991 if (m_littleEndian == (std::endian::native == std::endian::little)) {
992 m_bytes->write(values, count * sizeof(T));
993 return;
994 }
995 constexpr std::size_t kChunk = 4096;
996 std::array<T, kChunk> swapped;
997 for (std::size_t done = 0; done < count;) {
998 const std::size_t n = std::min(kChunk, count - done);
999 for (std::size_t i = 0; i < n; ++i)
1000 swapped[i] = mne::byte_swap(values[done + i]);
1001 m_bytes->write(swapped.data(), n * sizeof(T));
1002 done += n;
1003 }
1004 }
