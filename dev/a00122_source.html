49double supportedRate(double requested)
50{
51 double best = kBaseRate;
52 for (int divider = 1; divider <= 16; divider *= 2) {
53 const double rate = kBaseRate / divider;
54 if (std::abs(std::log2(rate / requested)) < std::abs(std::log2(best / requested))) {
55 best = rate;
56 }
57 }
58 return best;
59}

61std::vector<TmsiChannel> refaChannels(std::size_t exg)
62{
63 std::vector<TmsiChannel> channels;
64 channels.reserve(exg + kBip + kAux + 2);
65 // 71.5 nV per bit on the inputs, as a Refa's 22-bit converters
66 for (std::size_t i = 0; i < exg; ++i) {
67 channels.push_back({std::format("ExG{}", i + 1), TmsiChannelType::Exg, TmsiUnit::Volt, 0.0715256, 0.0, -6});
68 }
69 for (std::size_t i = 0; i < kBip; ++i) {
70 channels.push_back({std::format("Bip{}", i + 1), TmsiChannelType::Bip, TmsiUnit::Volt, 0.0715256, 0.0, -6});
71 }
72 for (std::size_t i = 0; i < kAux; ++i) {
73 channels.push_back({std::format("Aux{}", i + 1), TmsiChannelType::Aux, TmsiUnit::Volt, 1.0, 100.0, -6});
74 }
75 channels.push_back({"Digi", TmsiChannelType::Dig, TmsiUnit::Bit, 1.0, 0.0, 0});
76 channels.push_back({"Saw", TmsiChannelType::Saw, TmsiUnit::Unknown, 1.0, 0.0, 0});
77 return channels;
78}

110 AcqError start() override
111 {
112 // The session restarts at sample 0, as the amplifier's buffer does
113 const auto block = std::max<std::size_t>(1, static_cast<std::size_t>(m_rate / 128.0));
114 std::string uri = std::format("synthetic://?sfreq={}&stim=1&eeg={}&block={}&seed={}&realtime={}", m_rate,
115 m_simulation.exg + kBip + kAux, block, m_simulation.seed, m_simulation.realtime ? 1 : 0);
116 if (m_simulation.duration > 0.0) {
117 uri += std::format("&duration={}", m_simulation.duration);
118 }
119 auto session = AcqDevice::open(uri);
120 if (!session) {
121 return session.error();
122 }
123 AcqReaderOptions gating;
124 gating.gating = true; // the session waits for this reader instead of overwriting unread samples
125 auto reader = session->stream().reader(AcqReader::ReadMode::Lossless, gating);
126 if (!reader) {
127 return reader.error();
128 }
129 if (AcqError e = session->start(); !e.ok()) {
130 return e;
131 }
132 m_session = std::move(*session);
133 m_reader = std::move(*reader);
134 m_saw = 0;
135 return {};
136 }
137
138 void stop() override
139 {
140 m_reader = AcqReader();
141 m_session.close();
142 }

144 AcqExpected<std::size_t> read(std::span<std::int32_t> frames) override
145 {
146 const std::size_t width = m_channels.size();
147 auto view = m_reader.acquire(1, std::chrono::nanoseconds(0), frames.size() / width);
148 if (!view) {
149 if (view.error() == AcqErrorCode::Timeout) {
150 return std::size_t(0);
151 }
152 return view.error();
153 }
154 // The session's channels are the analog inputs, then the trigger code
155 std::size_t samples = 0;
156 for (std::size_t p = 0; p < view->partCount(); ++p) {
157 const AcqBlockPart& part = view->part(p);
158 for (std::size_t s = 0; s < part.samples; ++s, ++samples) {
159 std::int32_t* frame = frames.data() + samples * width;
160 for (std::size_t c = 0; c + 2 < width; ++c) {
161 float v = 0.0F;
162 std::memcpy(&v, part.at(c, s), sizeof(v));
163 frame[c] = encode(m_simulation.gain * v, m_channels[c]);
164 }
165 float code = 0.0F;
166 std::memcpy(&code, part.at(width - 2, s), sizeof(code));
167 frame[width - 2] = static_cast<std::int32_t>(code);
168 frame[width - 1] = static_cast<std::int32_t>(m_saw++ & 0xFFFF);
169 }
170 }
171 if (AcqError e = m_reader.release(*view); !e.ok()) {
172 return e;
173 }
174 return samples;
175 }

177 AcqExpected<std::vector<double>> impedances(std::chrono::milliseconds /*timeout*/) override
178 {
179 // Electrodes of a fresh gel cap: a few kilo-ohms, the same for a seed
180 std::mt19937_64 random(m_simulation.seed);
181 std::uniform_real_distribution<double> ohms(1e3, 20e3);
182 std::vector<double> values(m_simulation.exg);
183 for (double& value : values) {
184 value = ohms(random);
185 }
186 return values;
187 }

189private:
190 static std::int32_t encode(double volts, const TmsiChannel& channel) noexcept
191 {
192 const double raw = (volts * std::pow(10.0, -channel.exponent) - channel.offset) / channel.gain;
193 constexpr double kLow = std::numeric_limits<std::int32_t>::min() + 1.0; // the minimum flags an overflow
194 constexpr double kHigh = std::numeric_limits<std::int32_t>::max();
195 return static_cast<std::int32_t>(std::clamp(std::nearbyint(raw), kLow, kHigh));
196 }
197
198 TmsiSimulation m_simulation;

201 AcqDevice m_session;
202 AcqReader m_reader;
203 std::uint32_t m_saw = 0;
204};
