99MatrixXd Correlation::compute(const ConnectivitySettings::IntermediateTrialData& inputData)
100{
101 const MatrixXd centered = inputData.matData.colwise() - inputData.matData.rowwise().mean();
102 const VectorXd norms = centered.rowwise().norm();
103 MatrixXd matDist = centered * centered.transpose();
104
105 for (int i = 0; i < matDist.rows(); ++i) {
106 for (int j = 0; j < matDist.cols(); ++j) {
107 const double denom = norms(i) * norms(j);
108 matDist(i, j) = denom > 0.0 ? matDist(i, j) / denom : 0.0;
109 }
110 }
111
112 return matDist;
113}
