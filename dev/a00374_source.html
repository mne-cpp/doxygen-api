78scene::PointerEvent mne::disp::toScenePointerEvent(const QMouseEvent& event, scene::PointerEvent::Type type)
79{
80 scene::PointerEvent pointer;
81 pointer.type = type;
82 pointer.x = event.position().x();
83 pointer.y = event.position().y();
84 for (const auto& [qtButton, button] : kButtons) {
85 if (event.button() == qtButton) {
86 pointer.button = button;
87 }
88 if (event.buttons() & qtButton) {
89 pointer.buttons = pointer.buttons | scene::button_flag(button);
90 }
91 }
92 pointer.modifiers = toSceneModifiers(event.modifiers());
93 pointer.timestamp_ms = sceneTimestampMs();
94 return pointer;
95}
