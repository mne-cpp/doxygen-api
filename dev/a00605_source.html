112void SceneViewport::zoomBy(double factor, const QPointF& anchor)
113{
114 if (!(factor > 0.0)) {
115 return;
116 }
117 const scene::Point fixed = mapToScene(anchor);
118 m_zoom *= factor;
119 m_offset = anchor - QPointF(fixed.x, fixed.y) * m_zoom;
120 m_fitted = false;
121 update();
122}

149scene::DisplayList SceneViewport::figure(const QSize& size) const
150{
151 scene::Canvas canvas(size.width(), size.height(), toSceneColor(m_background));
152 const scene::Rect bounds = sceneBounds();
153 if (!bounds.empty()) {
154 const double zoom = std::min((size.width() - 2 * kFitMargin) / bounds.width,
155 (size.height() - 2 * kFitMargin) / bounds.height);
156 canvas.translate(size.width() / 2.0 - bounds.center().x * zoom, size.height() / 2.0 - bounds.center().y * zoom);
157 canvas.scale(zoom, zoom);
158 drawScene(canvas);
159 }
160 return canvas.take();
161}

242void SceneViewport::mousePressEvent(QMouseEvent* event)
243{
244 if (event->button() == Qt::RightButton) {
245 m_drag = Drag::Pan;
246 m_lastPosition = event->position();
247 } else if (event->button() == Qt::LeftButton) {
248 if (scenePressed(mapToScene(event->position()), event->modifiers())) {
249 m_drag = Drag::Item;
250 } else if (bandSelectionEnabled()) {
251 m_drag = Drag::Band;
252 m_bandStart = m_bandEnd = event->position();
253 m_bandModifiers = event->modifiers();
254 }
255 }
256 event->accept();
257}

275void SceneViewport::mouseReleaseEvent(QMouseEvent* event)
276{
277 const Drag drag = std::exchange(m_drag, Drag::None);
278 if (drag == Drag::Item) {
279 sceneReleased(mapToScene(event->position()));
280 } else if (drag == Drag::Band) {
281 m_bandEnd = event->position();
282 const scene::Point a = mapToScene(m_bandStart);
283 const scene::Point b = mapToScene(m_bandEnd);
284 sceneBandSelected(scene::Rect{std::min(a.x, b.x), std::min(a.y, b.y), std::abs(b.x - a.x), std::abs(b.y - a.y)},
285 m_bandModifiers);
286 update();
287 }
288 event->accept();
289}

303bool SceneViewport::event(QEvent* event)
304{
305 if (event->type() == QEvent::Gesture) {
306 auto* gestures = static_cast<QGestureEvent*>(event);
307 if (auto* pinch = static_cast<QPinchGesture*>(gestures->gesture(Qt::PinchGesture))) {
308 zoomBy(pinch->scaleFactor(), mapFromGlobal(pinch->centerPoint()));
309 }
310 if (auto* pan = static_cast<QPanGesture*>(gestures->gesture(Qt::PanGesture))) {
311 panBy(-pan->delta());
312 }
313 return true;
314 }
315 return QWidget::event(event);
316}

320void SceneViewport::applyFit()
321{
322 const scene::Rect bounds = sceneBounds();
323 if (bounds.empty() || width() <= 2 * kFitMargin || height() <= 2 * kFitMargin) {
324 return;
325 }
326 m_zoom = std::min((width() - 2 * kFitMargin) / bounds.width, (height() - 2 * kFitMargin) / bounds.height);
327 m_offset = QPointF(width() / 2.0 - bounds.center().x * m_zoom, height() / 2.0 - bounds.center().y * m_zoom);
328}
