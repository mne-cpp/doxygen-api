82Expected<MnaProject> MnaIO::read(const std::filesystem::path& path)
83{
84 const std::string extension = to_lower(utf8_from_path(path.extension()));
85 if (extension == ".mna") {
86 const Expected<std::string> text = read_text_file(path);
87 if (!text) {
88 return std::unexpected(text.error());
89 }
90 JsonParseError error;
91 const JsonDocument doc = JsonDocument::from_json(*text, &error);
92 if (error.error != JsonParseError::NoError) {
93 return fail(ErrorCode::Format, "{} is not valid JSON: {} at byte {}", utf8_from_path(path), error.error_string(), error.offset);
94 }
95 return checkVersion(MnaProject::from_json(doc.object()), path);
96 }
97 if (extension == ".mnx") {
98 const Expected<std::vector<std::byte>> bytes = read_file_bytes(path);
99 if (!bytes) {
100 return std::unexpected(bytes.error());
101 }
102 const std::span<const std::byte> data(*bytes);
103 if (data.size() < kMagic.size() || !std::ranges::equal(data.first(kMagic.size()), kMagic)) {
104 return fail(ErrorCode::Format, "{} does not start with the MNX1 magic bytes", utf8_from_path(path));
105 }
106 CborParseError error;
107 const CborValue root = CborValue::from_cbor(data.subspan(kMagic.size()), &error);
108 if (error.error != CborParseError::NoError || !root.is_map()) {
109 return fail(ErrorCode::Format, "{} does not hold a CBOR map: {}", utf8_from_path(path), error.error_string());
110 }
111 return checkVersion(MnaProject::from_cbor(root.to_map()), path);
112 }
113 return fail(ErrorCode::Format, "{} is neither a .mna nor a .mnx file", utf8_from_path(path));
114}

118Expected<void> MnaIO::write(const MnaProject& project, const std::filesystem::path& path)
119{
120 const std::string extension = to_lower(utf8_from_path(path.extension()));
121 if (extension == ".mna") {
122 return write_text_file(path, JsonDocument(project.to_json()).to_json(JsonDocument::Indented));
123 }
124 if (extension != ".mnx") {
125 return fail(ErrorCode::Format, "{} is neither a .mna nor a .mnx file", utf8_from_path(path));
126 }
127 return write_file_bytes(path, to_mnx(project));
128}
