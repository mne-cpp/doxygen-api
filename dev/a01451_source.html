98AcqExpected<AcqWriter*> AcqBackend::addInputStream(AcqStreamInfo info, const AcqStreamOptions& options)
99{
100 if (m_pImpl->sealed) {
101 return AcqError(AcqErrorCode::InvalidState, "streams can only be added while opening");
102 }
103 if (info.direction != AcqDirection::Input) {
104 return AcqError(AcqErrorCode::WrongDirection, "addInputStream() needs an input stream");
105 }
106 auto writer = AcqWriter::create(std::move(info), options);
107 if (!writer) {
108 return writer.error();
109 }
110 m_pImpl->writers.push_back(std::make_unique<AcqWriter>(std::move(*writer)));
111 m_pImpl->streams.push_back(m_pImpl->writers.back()->stream());
112 return m_pImpl->writers.back().get();
113}

117AcqExpected<AcqReader*> AcqBackend::addOutputStream(AcqStreamInfo info, const AcqStreamOptions& options, AcqWait wait)
118{
119 if (m_pImpl->sealed) {
120 return AcqError(AcqErrorCode::InvalidState, "streams can only be added while opening");
121 }
122 if (info.direction != AcqDirection::Output) {
123 return AcqError(AcqErrorCode::WrongDirection, "addOutputStream() needs an output stream");
124 }
125 // The application takes the writer later through AcqStream::writer(), so the core is created without one.
126 auto core = detail::AcqStreamCore::create(std::move(info), options);
127 if (!core) {
128 return core.error();
129 }
130 AcqStream stream(std::move(*core));
131 AcqReaderOptions gating;
132 gating.wait = wait;
133 gating.gating = true;
134 auto reader = stream.reader(AcqReader::ReadMode::Lossless, gating);
135 if (!reader) {
136 return reader.error();
137 }
138 m_pImpl->deviceReaders.push_back(std::make_unique<AcqReader>(std::move(*reader)));
139 m_pImpl->streams.push_back(std::move(stream));
140 return m_pImpl->deviceReaders.back().get();
141}

145void AcqBackend::reportError(AcqError error) noexcept
146{
147 int expected = 0;
148 if (m_pImpl->errorState.compare_exchange_strong(expected, 1, std::memory_order_acq_rel)) {
149 m_pImpl->error = std::move(error);
150 m_pImpl->errorState.store(2, std::memory_order_release);
151 }
