229bool SubView::shouldRenderSurface(std::string_view key) const
230{
231 if (key.starts_with("lh_"))
232 return visibility.lh;
233 if (key.starts_with("rh_"))
234 return visibility.rh;
235
236 if (key == "bem_head")
237 return visibility.bemHead;
238 if (key == "bem_outer_skull")
239 return visibility.bemOuterSkull;
240 if (key == "bem_inner_skull")
241 return visibility.bemInnerSkull;
242
243 if (key.starts_with("sens_contour_meg"))
244 return visibility.megFieldMap && visibility.megFieldContours;
245 if (key.starts_with("sens_contour_eeg"))
246 return visibility.eegFieldMap && visibility.eegFieldContours;
247 if (key.starts_with("sens_surface_meg"))
248 return visibility.sensMegHelmet;
249 if (key.starts_with("sens_meg_grad_"))
250 return visibility.sensMeg && visibility.sensMegGrad;
251 if (key.starts_with("sens_meg_mag_"))
252 return visibility.sensMeg && visibility.sensMegMag;
253 if (key.starts_with("sens_meg_"))
254 return visibility.sensMeg;
255 if (key.starts_with("sens_eeg_"))
256 return visibility.sensEeg;
257
258 if (key.starts_with("dig_live_"))
259 return true; // always show live tracker
260 if (key.starts_with("dig_static_"))
261 return true; // always show static markers
262 if (key.starts_with("dig_ray_"))
263 return true; // always show optical ray
264 if (key.starts_with("dig_probe_"))
265 return true; // always show catheter probe
266 if (key.starts_with("dig_cardinal"))
267 return visibility.dig && visibility.digCardinal;
268 if (key.starts_with("dig_hpi"))
269 return visibility.dig && visibility.digHpi;
270 if (key.starts_with("dig_eeg"))
271 return visibility.dig && visibility.digEeg;
272 if (key.starts_with("dig_extra"))
273 return visibility.dig && visibility.digExtra;
274 if (key.starts_with("dig_"))
275 return visibility.dig;
276
277 if (key.starts_with("srcsp_"))
278 return visibility.sourceSpace;
279
280 return true;
281}

409Color mneAnalyzeColor(double v)
410{
411 double x = 2.0 * v - 1.0;
412 x = std::clamp(x, -1.0, 1.0);
413
414 static constexpr int N = 7;
415 static const double pos[N] = {-1.0, -0.90, -0.30, 0.0, 0.30, 0.90, 1.0};
416 static const double rr[N] = {0.0, 0.0, 0.5, 0.5, 0.5, 1.0, 1.0};
417 static const double gg[N] = {1.0, 0.0, 0.5, 0.5, 0.5, 0.0, 1.0};
418 static const double bb[N] = {1.0, 1.0, 0.5, 0.5, 0.5, 0.0, 0.0};
419
420 int seg = 0;
421 for (int i = 0; i < N - 1; ++i) {
422 if (x >= pos[i] && x <= pos[i + 1]) {
423 seg = i;
424 break;
425 }
426 }
427 if (x > pos[N - 1])
428 seg = N - 2;
429
430 double t = (pos[seg + 1] != pos[seg])
431 ? (x - pos[seg]) / (pos[seg + 1] - pos[seg])
432 : 0.0;
433 t = std::clamp(t, 0.0, 1.0);
434
435 int r = static_cast<int>(std::round((rr[seg] + t * (rr[seg + 1] - rr[seg])) * 255.0));
436 int g = static_cast<int>(std::round((gg[seg] + t * (gg[seg + 1] - gg[seg])) * 255.0));
437 int b = static_cast<int>(std::round((bb[seg] + t * (bb[seg + 1] - bb[seg])) * 255.0));
438
439 return {static_cast<std::uint8_t>(std::clamp(r, 0, 255)), static_cast<std::uint8_t>(std::clamp(g, 0, 255)), static_cast<std::uint8_t>(std::clamp(b, 0, 255)), 255};
440}

446void SubView::load(const SettingsStore& settings, const std::string& prefix,
447 const Eigen::Quaternionf& fallbackRotation)
448{
449 surfaceType = settings.read_string(prefix + "surfaceType", surfaceType);
450 brainShader = shaderModeFromName(settings.read_string(prefix + "shader", shaderModeName(brainShader)));
451 bemShader = shaderModeFromName(settings.read_string(prefix + "bemShader", shaderModeName(bemShader)));
452 overlayMode = visualizationModeFromName(settings.read_string(prefix + "overlay", visualizationModeName(overlayMode)));
453
454 visibility.load(settings, prefix + "vis_");
455
456 zoom = static_cast<float>(settings.read_double(prefix + "zoom", zoom));
457 pan = Eigen::Vector2f(static_cast<float>(settings.read_double(prefix + "panX", pan.x())), static_cast<float>(settings.read_double(prefix + "panY", pan.y())));
458 preset = viz::clamp_view_preset(settings.read_int(prefix + "preset", preset));
459
460 const bool hasQuat = settings.contains(prefix + "perspRotW") && settings.contains(prefix + "perspRotX") && settings.contains(prefix + "perspRotY") && settings.contains(prefix + "perspRotZ");
461 if (hasQuat) {
462 const auto component = [&](const char* name, double fallback) {
463 return static_cast<float>(settings.read_double(prefix + name, fallback));
464 };
465 rotation = Eigen::Quaternionf(component("perspRotW", 1.0), component("perspRotX", 0.0), component("perspRotY", 0.0), component("perspRotZ", 0.0));
466 if (rotation.squaredNorm() <= std::numeric_limits<float>::epsilon()) {
467 rotation = Eigen::Quaternionf::Identity();
468 } else {
469 rotation.normalize();
470 }
471 } else {
472 rotation = fallbackRotation;
473 }
474}
