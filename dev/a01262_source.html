77MatrixXd SurfaceLaplacian::evaluateLegendre(const MatrixXd& matX, int iMaxOrder)
78{
79 // Returns matrix of shape (iMaxOrder+1, matX.size()) where row n = P_n(x)
80 // using flattened matX
81 const Index nElements = matX.size();
82 MatrixXd result(iMaxOrder + 1, nElements);
83
84 // P_0(x) = 1
85 result.row(0).setOnes();
86
87 if (iMaxOrder >= 1) {
88 // P_1(x) = x
89 for (Index i = 0; i < nElements; ++i)
90 result(1, i) = matX.data()[i];
91 }
92
93 // Bonnet's recurrence: (n+1) P_{n+1}(x) = (2n+1) x P_n(x) - n P_{n-1}(x)
94 for (int n = 1; n < iMaxOrder; ++n) {
95 const double a = static_cast<double>(2 * n + 1) / static_cast<double>(n + 1);
96 const double b = static_cast<double>(n) / static_cast<double>(n + 1);
97 for (Index i = 0; i < nElements; ++i)
98 result(n + 1, i) = a * matX.data()[i] * result(n, i) - b * result(n - 1, i);
99 }
100
101 return result;
102}

106MatrixXd SurfaceLaplacian::computeG(const MatrixXd& matCosAng,
107 int iStiffness,
108 int iNLegendreTerms)
109{
110 // G(cos θ) = Σ_{n=1}^{N} (2n+1) / (n^m (n+1)^m 4π) · P_n(cos θ)
111 const Index nRows = matCosAng.rows();
112 const Index nCols = matCosAng.cols();
113
114 // Evaluate Legendre polynomials
115 MatrixXd legP = evaluateLegendre(matCosAng, iNLegendreTerms);
116
117 // Accumulate weighted sum
118 MatrixXd G = MatrixXd::Zero(nRows, nCols);
119
120 for (int n = 1; n <= iNLegendreTerms; ++n) {
121 const double factor = static_cast<double>(2 * n + 1)
122 / (std::pow(static_cast<double>(n), iStiffness)
123 * std::pow(static_cast<double>(n + 1), iStiffness)
124 * 4.0 * CSD_PI);
125
126 for (Index i = 0; i < nRows * nCols; ++i)
127 G.data()[i] += factor * legP(n, i);
128 }
129
130 return G;
131}

135MatrixXd SurfaceLaplacian::computeH(const MatrixXd& matCosAng,
136 int iStiffness,
137 int iNLegendreTerms)
138{
139 // H(cos θ) = Σ_{n=1}^{N} (2n+1) / (n^(m-1) (n+1)^(m-1) 4π) · P_n(cos θ)
140 const Index nRows = matCosAng.rows();
141 const Index nCols = matCosAng.cols();
142
143 MatrixXd legP = evaluateLegendre(matCosAng, iNLegendreTerms);
144
145 MatrixXd H = MatrixXd::Zero(nRows, nCols);
146
147 for (int n = 1; n <= iNLegendreTerms; ++n) {
148 const double factor = static_cast<double>(2 * n + 1)
149 / (std::pow(static_cast<double>(n), iStiffness - 1)
150 * std::pow(static_cast<double>(n + 1), iStiffness - 1)
151 * 4.0 * CSD_PI);
152
153 for (Index i = 0; i < nRows * nCols; ++i)
154 H.data()[i] += factor * legP(n, i);
155 }
156
157 return H;
158}

162MatrixXd SurfaceLaplacian::computeTransform(const MatrixX3d& matPositions,
163 double dLambda2,
164 int iStiffness,
165 int iNLegendreTerms,
166 double dSphereRadius)
167{
168 const int nCh = static_cast<int>(matPositions.rows());
169 if (nCh < 2) {
170 qWarning() << "[SurfaceLaplacian::computeTransform] Need at least 2 channels.";
171 return MatrixXd();
172 }
173
174 // 1. Centre positions and normalise to unit sphere
175 Vector3d centre = matPositions.colwise().mean();
176 MatrixX3d posCentered = matPositions.rowwise() - centre.transpose();
177
178 // Fit sphere radius if not provided
179 if (dSphereRadius <= 0.0) {
180 dSphereRadius = 0.0;
181 for (int i = 0; i < nCh; ++i)
182 dSphereRadius += posCentered.row(i).norm();
183 dSphereRadius /= static_cast<double>(nCh);
184 }
185
186 // Normalise to unit sphere
187 MatrixX3d posNorm(nCh, 3);
188 for (int i = 0; i < nCh; ++i) {
189 double norm = posCentered.row(i).norm();
190 if (norm > 0.0)
191 posNorm.row(i) = posCentered.row(i) / norm;
192 else
193 posNorm.row(i).setZero();
194 }
195
196 // 2. Cosine angle matrix: cos(θ_ij) = pos_i · pos_j (on unit sphere)
197 MatrixXd cosAng = posNorm * posNorm.transpose();
198
199 // Clamp to [-1, 1]
200 cosAng = cosAng.cwiseMax(-1.0).cwiseMin(1.0);
201
202 // 3. Compute G and H matrices
203 MatrixXd G = computeG(cosAng, iStiffness, iNLegendreTerms);
204 MatrixXd H = computeH(cosAng, iStiffness, iNLegendreTerms);
205
206 // 4. Regularise G: G(i,i) += lambda²
207 for (int i = 0; i < nCh; ++i)
208 G(i, i) += dLambda2;
209
210 // 5. Invert G
211 MatrixXd Gi = G.inverse();
212
213 // Column sums of Gi
214 VectorXd TC = Gi.colwise().sum();
215 double sgi = TC.sum();
216
217 // 6. Build CSD transform
218 // Z = I - (1/n) * ones: average-reference the identity
219 MatrixXd Z = MatrixXd::Identity(nCh, nCh);
220 Z.array() -= 1.0 / static_cast<double>(nCh);
221
222 // Cp2 = Gi * Z
223 MatrixXd Cp2 = Gi * Z;
224
225 // c02 = (colwise sum of Cp2) / sgi
226 RowVectorXd c02 = Cp2.colwise().sum() / sgi;
227
228 // C2 = Cp2 - TC * c02
229 MatrixXd C2 = Cp2 - TC * c02;
230
231 // Transform = (C2^T * H)^T / R^2 = H^T * C2 / R^2
232 MatrixXd X = (H.transpose() * C2) / (dSphereRadius * dSphereRadius);
233
234 return X;
235}

239SurfaceLaplacianResult SurfaceLaplacian::compute(const MatrixXd& matData,
240 const MatrixX3d& matPositions,
241 double dLambda2,
242 int iStiffness,
243 int iNLegendreTerms,
244 double dSphereRadius)
245{
246 SurfaceLaplacianResult result;
247
248 if (matData.rows() != matPositions.rows()) {
249 qWarning() << "[SurfaceLaplacian::compute] Data rows" << matData.rows()
250 << "!= position rows" << matPositions.rows();
251 return result;
252 }
253
254 result.matTransform = computeTransform(matPositions, dLambda2, iStiffness,
255 iNLegendreTerms, dSphereRadius);
256
257 if (result.matTransform.size() == 0)
258 return result;
259
260 // Apply transform: CSD_data = X * data
261 result.matData = result.matTransform * matData;
262
263 return result;
264}
