54InvBeamformer InvLCMV::makeLCMV([[maybe_unused]] const FiffInfo& info,
55 const MNEForwardSolution& forward,
56 const FiffCov& dataCov,
57 double reg,
58 const FiffCov& noiseCov,
59 BeamformerPickOri pickOri,
60 BeamformerWeightNorm weightNorm,
61 bool reduceRank,
62 BeamformerInversion invMethod)
63{
64 InvBeamformer result;
65 result.kind = "LCMV";
66
67 // -----------------------------------------------------------------------
68 // Extract leadfield G from forward solution
69 // -----------------------------------------------------------------------
70 if (forward.sol.data.size() == 0) {
71 Log::warning("InvLCMV::makeLCMV - Forward solution has no gain matrix!");
72 return result;
73 }
74
75 MatrixXd G = forward.sol.data; // (n_channels, n_sources * n_orient)
76 const int nChannels = static_cast<int>(G.rows());
77 const int nOrient = (forward.source_ori == FIFFV_MNE_FREE_ORI) ? 3 : 1;
78 const int nSources = static_cast<int>(G.cols()) / nOrient;
79
80 Log::info("InvLCMV::makeLCMV - Leadfield: {} channels x {} sources (n_orient={})", nChannels, nSources, nOrient);
81
82 // -----------------------------------------------------------------------
83 // Build whitening matrix from noise covariance
84 // -----------------------------------------------------------------------
85 MatrixXd whitener;
86 if (noiseCov.data.size() > 0) {
87 // Compute whitener from noise covariance eigendecomposition
88 // whitener = diag(1/sqrt(eig)) @ eigvec (rows of eigvec are the eigenvectors)
89 if (noiseCov.eig.size() > 0 && noiseCov.eigvec.size() > 0) {
90 VectorXd invSqrtEig(noiseCov.eig.size());
91 for (int i = 0; i < noiseCov.eig.size(); ++i) {
92 invSqrtEig(i) = (noiseCov.eig(i) > 1e-30)
93 ? 1.0 / std::sqrt(noiseCov.eig(i))
94 : 0.0;
95 }
96 whitener = invSqrtEig.asDiagonal() * noiseCov.eigvec;
97 } else {
98 // Fallback: identity whitening
99 whitener = MatrixXd::Identity(nChannels, nChannels);
100 }
101 } else {
102 whitener = MatrixXd::Identity(nChannels, nChannels);
103 }
104
105 // -----------------------------------------------------------------------
106 // Build SSP projection matrix
107 // -----------------------------------------------------------------------
108 MatrixXd projMat = MatrixXd::Identity(nChannels, nChannels);
109 // Note: SSP projections from info.projs are typically pre-applied to
110 // the forward solution. If not, they should be applied here.
111
112 // -----------------------------------------------------------------------
113 // Whiten leadfield and data covariance
114 // G_w = whitener @ G
115 // Cm_w = whitener @ Cm @ whitener^T
116 // -----------------------------------------------------------------------
117 MatrixXd Gw = whitener * G;
118
119 MatrixXd CmData = dataCov.data;
120 if (CmData.rows() != nChannels || CmData.cols() != nChannels) {
121 Log::warning("InvLCMV::makeLCMV - Data covariance dimension ({} x {}) does not match leadfield channels ({})!", CmData.rows(), CmData.cols(), nChannels);
122 return result;
123 }
124
125 MatrixXd CmW = whitener * CmData * whitener.transpose();
126 // Ensure Hermitian (for numerical stability)
127 CmW = (CmW + CmW.transpose()) * 0.5;
128
129 // -----------------------------------------------------------------------
130 // Source normals for orientation picking
131 // -----------------------------------------------------------------------
132 MatrixX3d nn = forward.source_nn.cast<double>();
133 if (nOrient == 3 && nn.rows() == 3 * nSources) {
134 // Free orientation stores three rows per source; mne-python uses nn[2::3].
135 MatrixX3d perSource(nSources, 3);
136 for (int s = 0; s < nSources; ++s)
137 perSource.row(s) = nn.row(3 * s + 2);
138 nn = perSource;
139 }
140
141 // -----------------------------------------------------------------------
142 // Compute spatial filter
143 // -----------------------------------------------------------------------
144 MatrixXd W;
145 MatrixX3d mpOri;
146
147 bool ok = InvBeamformerCompute::computeBeamformer(
148 Gw, CmW, reg, nOrient,
149 weightNorm, pickOri, reduceRank, invMethod,
150 nn, W, mpOri);
151
152 if (!ok) {
153 Log::warning("InvLCMV::makeLCMV - Beamformer computation failed!");
154 return result;
155 }
156
157 // -----------------------------------------------------------------------
158 // Populate result
159 // -----------------------------------------------------------------------
160 result.weights.push_back(W);
161 result.whitener = whitener;
162 result.proj = projMat;
163 result.chNames = forward.sol.row_names;
164 result.isFreOri = (nOrient == 3 && pickOri != BeamformerPickOri::Normal && pickOri != BeamformerPickOri::MaxPower);
165 result.nSourcesTotal = nSources;
166 result.srcType = "surface";
167 result.weightNorm = weightNorm;
168 result.pickOri = pickOri;
169 result.inversion = invMethod;
170 result.reg = reg;
171 result.rank = static_cast<int>(CmW.rows());
172 result.maxPowerOri = mpOri;
173 result.sourceNn = forward.source_nn;
174
175 // Vertex indices
176 VectorXi verts(0);
177 if (forward.src.size() >= 2) {
178 verts.resize(forward.src[0].vertno.size() + forward.src[1].vertno.size());
179 verts << forward.src[0].vertno, forward.src[1].vertno;
180 // Keep where the left hemisphere ends. The concatenation above is the
181 // only place that knows it, and without it the result cannot be
182 // written as the -lh/-rh pair mne-python expects.
183 result.nVerticesLh = static_cast<int>(forward.src[0].vertno.size());
184 } else if (forward.src.size() == 1) {
185 verts = forward.src[0].vertno;
186 }
187 result.vertices = verts;
188
189 Log::info("InvLCMV::makeLCMV - Done. Filter: {} x {} (sources={}, orient={})", W.rows(), W.cols(), nSources, result.nOrient());
190
191 return result;
192}

196MatrixXd InvLCMV::applyFilter(const MatrixXd& data, const InvBeamformer& filters)
197{
198 // Apply projection + whitening + spatial filter
199 MatrixXd processed = data;
200
201 // Project
202 if (filters.proj.size() > 0 && filters.proj.rows() == data.rows()) {
203 processed = filters.proj * processed;
204 }
205
206 // Whiten
207 if (filters.whitener.size() > 0 && filters.whitener.cols() == processed.rows()) {
208 processed = filters.whitener * processed;
209 }
210
211 // Apply spatial filter: sol = W @ processed
212 return filters.weights[0] * processed;
213}

217InvSourceEstimate InvLCMV::applyLCMV(const FiffEvoked& evoked, const InvBeamformer& filters)
218{
219 if (!filters.isValid() || filters.kind != "LCMV") {
220 Log::warning("InvLCMV::applyLCMV - Invalid or non-LCMV filters!");
221 return InvSourceEstimate();
222 }
223
224 // Pick the filter's channels from evoked by name, as mne's apply_lcmv (the order may differ)
225 MatrixXd data;
226 if (filters.chNames.size() > 0) {
227 const int nFilterCh = static_cast<int>(filters.chNames.size());
228 const int nTimes = static_cast<int>(evoked.data.cols());
229 data.resize(nFilterCh, nTimes);
230 for (int i = 0; i < nFilterCh; ++i) {
231 int idx = mne::channel_index(evoked.info.ch_names, filters.chNames[i]);
232 if (idx < 0) {
233 Log::warning("InvLCMV::applyLCMV - Channel {} not found in evoked!", filters.chNames[i]);
234 return InvSourceEstimate();
235 }
236 data.row(i) = evoked.data.row(idx);
237 }
238 } else {
239 data = evoked.data;
240 }
241
242 MatrixXd sol = applyFilter(data, filters);
243
244 // Combine XYZ for free orientation if needed
245 const int nOrient = filters.nOrient();
246 if (nOrient == 3 && filters.pickOri != BeamformerPickOri::Vector) {
247 // Combine: sqrt(x^2 + y^2 + z^2) per source per time
248 const int nSources = static_cast<int>(sol.rows()) / 3;
249 const int nTimes = static_cast<int>(sol.cols());
250 MatrixXd combined(nSources, nTimes);
251 for (int s = 0; s < nSources; ++s) {
252 combined.row(s) = sol.middleRows(s * 3, 3).colwise().norm();
253 }
254 sol = combined;
255 }
256
257 float tmin = evoked.times.size() > 0 ? evoked.times[0] : 0.0f;
258 float tstep = (evoked.info.sfreq > 0) ? 1.0f / evoked.info.sfreq : 1.0f;
259
260 InvSourceEstimate stc(sol, filters.vertices, tmin, tstep);
261 stc.method = InvEstimateMethod::LCMV;
262 stc.sourceSpaceType = InvSourceSpaceType::Surface;
263 stc.nVerticesLh = filters.nVerticesLh;
264 stc.orientationType = filters.isFreOri ? InvOrientationType::Free : InvOrientationType::Fixed;
265
266 return stc;
267}

305InvSourceEstimate InvLCMV::applyLCMVCov(const FiffCov& dataCov,
306 const InvBeamformer& filters)
307{
308 if (!filters.isValid() || filters.kind != "LCMV") {
309 Log::warning("InvLCMV::applyLCMVCov - Invalid or non-LCMV filters!");
310 return InvSourceEstimate();
311 }
312
313 // Whiten data covariance
314 MatrixXd CmW = dataCov.data;
315 if (filters.whitener.size() > 0) {
316 CmW = filters.whitener * CmW * filters.whitener.transpose();
317 }
318
319 const int nOrient = filters.nOrient();
320 VectorXd power = InvBeamformerCompute::computePower(CmW, filters.weights[0], nOrient);
321
322 // Return as 1-column source estimate
323 MatrixXd powerMat = power; // (nSources, 1) implicitly via VectorXd
324
325 InvSourceEstimate stc(powerMat, filters.vertices, 0.0f, 1.0f);
326 stc.method = InvEstimateMethod::LCMV;
327 stc.sourceSpaceType = InvSourceSpaceType::Surface;
328 stc.nVerticesLh = filters.nVerticesLh;
329 stc.orientationType = filters.isFreOri ? InvOrientationType::Free : InvOrientationType::Fixed;
330
331 return stc;
332}

336std::vector<InvSourceEstimate> InvLCMV::applyLCMVEpochs(const std::vector<MatrixXd>& epochs,
337 float tmin,
338 float tstep,
339 const InvBeamformer& filters)
340{
341 std::vector<InvSourceEstimate> results;
342
343 if (epochs.empty()) {
344 Log::warning("InvLCMV::applyLCMVEpochs - No epochs provided.");
345 return results;
346 }
347 if (!filters.isValid() || filters.kind != "LCMV") {
348 Log::warning("InvLCMV::applyLCMVEpochs - Invalid or non-LCMV filters!");
349 return results;
350 }
351
352 results.reserve(epochs.size());
353 for (const MatrixXd& epoch : epochs) {
354 results.push_back(applyLCMVRaw(epoch, tmin, tstep, filters));
355 if (results.back().empty()) {
356 Log::warning("InvLCMV::applyLCMVEpochs - Epoch {} produced empty source estimate.", results.size() - 1);
357 }
358 }
359
360 return results;
361}

365MatrixXd InvLCMV::makeLCMVResolutionMatrix(
366 const MNEForwardSolution& forward,
367 const FiffInfo& info,
368 const FiffCov& dataCov,
369 double reg,
370 const FiffCov& noiseCov)
371{
372 // Build the LCMV filter
373 InvBeamformer filters = makeLCMV(info, forward, dataCov, reg, noiseCov);
374
375 if (!filters.isValid() || filters.weights.empty()) {
376 Log::warning("InvLCMV::makeLCMVResolutionMatrix - Could not compute LCMV filter.");
377 return MatrixXd();
378 }
379
380 // Extract leadfield: G (n_channels x n_dipoles)
381 MatrixXd G = forward.sol.data;
382
383 // Apply whitening to leadfield
384 MatrixXd Gw = G;
385 if (filters.proj.size() > 0 && filters.proj.rows() == G.rows()) {
386 Gw = filters.proj * Gw;
387 }
388 if (filters.whitener.size() > 0 && filters.whitener.cols() == Gw.rows()) {
389 Gw = filters.whitener * Gw;
390 }
391
392 // Resolution matrix: R = W @ G_whitened
393 MatrixXd R = filters.weights[0] * Gw;
394
395 Log::info("InvLCMV::makeLCMVResolutionMatrix - Resolution matrix: {} x {}", R.rows(), R.cols());
396
397 return R;
