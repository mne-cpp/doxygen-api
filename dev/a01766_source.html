247void RawViewPanel::changed(const std::string& key)
248{
249 if (key == kZoom && m_callbacks.zoom_changed) {
250 m_callbacks.zoom_changed(zoom());
251 } else if (key == kWindow && m_callbacks.window_size_changed) {
252 m_callbacks.window_size_changed(window_size());
253 } else if (key == kSpacer && m_callbacks.time_spacer_changed) {
254 m_callbacks.time_spacer_changed(time_spacer());
255 } else if (key == kSignal && m_callbacks.signal_color_changed) {
256 m_callbacks.signal_color_changed(signal_color());
257 } else if (key == kBackground && m_callbacks.background_color_changed) {
258 m_callbacks.background_color_changed(background_color());
259 } else if (key == kScreenshot && m_callbacks.screenshot_wanted) {
260 m_callbacks.screenshot_wanted(image_type());
261 }
