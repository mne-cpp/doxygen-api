53template<typename Fn>
54void for_wrapped_span(int start, int length, int size, Fn fn)
55{
56 start = ((start % size) + size) % size;
57 for (int done = 0; done < length;) {
58 const int piece = std::min(length - done, size - start);
59 fn(start, done, piece);
60 done += piece;
61 start = 0;
62 }
63}
