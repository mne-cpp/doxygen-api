67InvGuessData::InvGuessData(const std::filesystem::path& guessname, const std::filesystem::path& guess_surfname, float mindist, float exclude, float grid, InvDipoleFitData* f, float guessrad)
68{
69 int k, p;
70 std::unique_ptr<MNESourceSpace> guesses;
71
72 if (!guessname.empty()) {
73 /*
74 * Read the guesses and transform to the appropriate coordinate frame
75 */
76 std::vector<std::unique_ptr<MNESourceSpace>> sp;
77 if (MNESourceSpace::read_source_spaces(guessname, sp) == FIFF_FAIL)
78 return;
79 if (static_cast<int>(sp.size()) != 1) {
80 mne::Log::critical("Incorrect number of source spaces in guess file");
81 return;
82 }
83 mne::Log::info("Read guesses from {}\n", mne::utf8_from_path(guessname));
84 guesses = std::move(sp[0]);
85 } else {
86 MNESurface* inner_skull = nullptr;
87 std::unique_ptr<MNESurface> inner_skull_owned;
88 Eigen::Vector3f r0 = f->r0;
89
90 assert(f->mri_head_t);
91 FiffCoordTrans::apply_inverse_trans(r0.data(), *f->mri_head_t, true);
92 if (f->bem_model) {
93 mne::Log::info("Using inner skull surface from the BEM ({})...\n", f->bemname);
94 if ((inner_skull = f->bem_model->fwd_bem_find_surface(FIFFV_BEM_SURF_ID_BRAIN)) == nullptr)
95 return;
96 } else if (!guess_surfname.empty()) {
97 mne::Log::info("Reading inner skull surface from {}...\n", mne::utf8_from_path(guess_surfname));
98 inner_skull_owned = MNESurface::read_bem_surface(guess_surfname, FIFFV_BEM_SURF_ID_BRAIN, true);
99 if (!inner_skull_owned)
100 return;
101 inner_skull = inner_skull_owned.get();
102 }
103 guesses = FwdBemModel::make_guesses(inner_skull, guessrad, r0, grid, exclude, mindist);
104 if (!guesses)
105 return;
106 }
107 if (guesses->nuse == 0) {
108 mne::Log::critical("No active guess locations remaining.");
109 return;
110 }
111 /*
112 * Transform the guess locations to the appropriate coordinate frame
113 */
114 {
115 std::vector<std::unique_ptr<MNESourceSpace>> guesses_vec;
116 guesses_vec.push_back(std::move(guesses));
117 if (MNESourceSpace::transform_source_spaces_to(f->coord_frame, *f->mri_head_t, guesses_vec) != OK)
118 return;
119 guesses = std::move(guesses_vec[0]);
120 }
121 mne::Log::info("Guess locations are now in {} coordinates.\n", std::string(FiffCoordTrans::frame_name(f->coord_frame)));
122
123 this->nguess = guesses->nuse;
124 this->rr.resize(guesses->nuse, 3);
125 for (k = 0, p = 0; k < guesses->np; k++)
126 if (guesses->inuse[k]) {
127 this->rr.row(p) = guesses->rr.row(k);
128 p++;
129 }
130 guesses.reset();
131
132 this->guess_fwd.resize(this->nguess);
133 /*
134 * Compute the guesses using the sphere model for speed
135 */
136 if (!this->compute_guess_fields(f))
137 return;
138
139 return;
140}

148bool InvGuessData::compute_guess_fields(InvDipoleFitData* f)
149{
150 dipoleFitFuncs orig = nullptr;
151
152 if (!f) {
153 mne::Log::critical("Data missing in compute_guess_fields");
154 return false;
155 }
156 if (!f->noise) {
157 mne::Log::critical("Noise covariance missing in compute_guess_fields");
158 return false;
159 }
160 mne::Log::info("Go through all guess source locations...");
161 orig = f->funcs;
162 if (f->fit_mag_dipoles)
163 f->funcs = f->mag_dipole_funcs.get();
164 else
165 f->funcs = f->sphere_funcs.get();
166 for (int k = 0; k < this->nguess; k++) {
167 this->guess_fwd[k] = InvDipoleFitData::dipole_forward_one(f, Eigen::Vector3f(this->rr.row(k).transpose()), std::move(this->guess_fwd[k]));
168 if (!this->guess_fwd[k]) {
169 if (orig)
170 f->funcs = orig;
171 return false;
172 }
173#ifdef DEBUG
174 sing = this->guess_fwd[k]->sing;
175 mne::Log::info("{:f} {:f} {:f}\n", sing[0], sing[1], sing[2]);
176#endif
177 }
178 f->funcs = orig;
179 mne::Log::info("[done {} sources]\n", this->nguess);
180
181 return true;
182}
