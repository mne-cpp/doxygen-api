121void ArtifactSettingsView::saveSettings()
122{
123 if (m_sSettingsPath.isEmpty()) {
124 return;
125 }
126
127 // Save Settings
128 QSettings settings("MNECPP");
129
130 settings.setValue(m_sSettingsPath + QString("/ArtifactSettingsView/doArtifactThresholdReduction"), m_bDoArtifactThresholdReduction);
131
132 settings.beginGroup(m_sSettingsPath + QString("/ArtifactSettingsView/artifactThresholdsFirst"));
133 std::map<std::string, double>::const_iterator itrFirst = m_mapThresholdsFirst.cbegin();
134 while (itrFirst != m_mapThresholdsFirst.cend()) {
135 settings.setValue(QString::fromStdString(itrFirst->first), itrFirst->second);
136 ++itrFirst;
137 }
138 settings.endGroup();
139
140 settings.beginGroup(m_sSettingsPath + QString("/ArtifactSettingsView/artifactThresholdsSecond"));
141 std::map<std::string, int>::const_iterator itrSecond = m_mapThresholdsSecond.cbegin();
142 while (itrSecond != m_mapThresholdsSecond.cend()) {
143 settings.setValue(QString::fromStdString(itrSecond->first), itrSecond->second);
144 ++itrSecond;
145 }
146 settings.endGroup();
147}

151void ArtifactSettingsView::loadSettings()
152{
153 m_mapThresholdsFirst["grad"] = 1.0;
154 m_mapThresholdsFirst["mag"] = 1.0;
155 m_mapThresholdsFirst["eeg"] = 1.0;
156 m_mapThresholdsFirst["ecg"] = 1.0;
157 m_mapThresholdsFirst["emg"] = 1.0;
158 m_mapThresholdsFirst["eog"] = 1.0;
159
160 m_mapThresholdsSecond["grad"] = -1;
161 m_mapThresholdsSecond["mag"] = -1;
162 m_mapThresholdsSecond["eeg"] = -1;
163 m_mapThresholdsSecond["ecg"] = -1;
164 m_mapThresholdsSecond["emg"] = -1;
165 m_mapThresholdsSecond["eog"] = -1;
166
167 if (m_sSettingsPath.isEmpty()) {
168 m_mapThresholds["Active"] = m_bDoArtifactThresholdReduction ? 1.0 : 0.0;
169 return;
170 }
171
172 // Load Settings
173 QSettings settings("MNECPP");
174
175 m_bDoArtifactThresholdReduction = settings.value(m_sSettingsPath + QString("/ArtifactSettingsView/doArtifactThresholdReduction"), false).toBool();
176 m_mapThresholds["Active"] = m_bDoArtifactThresholdReduction ? 1.0 : 0.0;
177
178 settings.beginGroup(m_sSettingsPath + QString("/ArtifactSettingsView/artifactThresholdsFirst"));
179 for (const QString& key : settings.childKeys()) {
180 m_mapThresholdsFirst.insert_or_assign(key.toStdString(), settings.value(key, 1.0).toDouble());
181 }
182 settings.endGroup();
183
184 settings.beginGroup(m_sSettingsPath + QString("/ArtifactSettingsView/artifactThresholdsSecond"));
185 for (const QString& key : settings.childKeys()) {
186 m_mapThresholdsSecond.insert_or_assign(key.toStdString(), settings.value(key, -1).toInt());
187 }
188 settings.endGroup();
189}

217void ArtifactSettingsView::redrawGUI()
218{
219 detail::resetLayout(this);
220 m_mapChThresholdsDoubleSpinBoxes.clear();
221 m_mapChThresholdsSpinBoxes.clear();
222
223 auto pGroupBoxArtifactRejection = std::make_unique<QGridLayout>();
224
225 //Artifact rejection
226 if (!m_fiffChInfoList.empty()) {
227 std::vector<std::string> channelTypes;
228 const auto addType = [&channelTypes](const char* type) {
229 if (!std::ranges::contains(channelTypes, type)) {
230 channelTypes.push_back(type);
231 }
232 };
233 int kind, unit;
234
235 for (int i = 0; i < std::ssize(m_fiffChInfoList); ++i) {
236 kind = m_fiffChInfoList.at(i).kind;
237 unit = m_fiffChInfoList.at(i).unit;
238
239 if (kind == FIFFV_MEG_CH && unit == FIFF_UNIT_T_M) {
240 addType("grad");
241 }
242 if (kind == FIFFV_MEG_CH && unit == FIFF_UNIT_T) {
243 addType("mag");
244 }
245 if (kind == FIFFV_EEG_CH) {
246 addType("eeg");
247 }
248 if (kind == FIFFV_EOG_CH) {
249 addType("eog");
250 }
251 if (kind == FIFFV_EMG_CH) {
252 addType("emg");
253 }
254 if (kind == FIFFV_ECG_CH) {
255 addType("ecg");
256 }
257 }
258
259 if (!channelTypes.empty()) {
260 m_pArtifactRejectionCheckBox = new QCheckBox("Activate artifact rejection", this);
261 pGroupBoxArtifactRejection->addWidget(m_pArtifactRejectionCheckBox, 0, 0, 1, 2);
262 m_pArtifactRejectionCheckBox->setChecked(m_bDoArtifactThresholdReduction);
263 connect(m_pArtifactRejectionCheckBox.data(), &QCheckBox::clicked,
264 this, &ArtifactSettingsView::onChangeArtifactThreshold);
265
266 for (int i = 0; i < std::ssize(channelTypes); ++i) {
267 QLabel* pLabel = new QLabel(QString::fromStdString(channelTypes.at(i)), this);
268 pGroupBoxArtifactRejection->addWidget(pLabel, i + 1, 0);
269
270 QDoubleSpinBox* pDoubleSpinBox = new QDoubleSpinBox(this);
271 pDoubleSpinBox->setPrefix("+/-");
272 pDoubleSpinBox->setMinimum(0.0);
273 pDoubleSpinBox->setMaximum(100000.0);
274 pDoubleSpinBox->setValue(m_mapThresholdsFirst[channelTypes.at(i)]);
275 pGroupBoxArtifactRejection->addWidget(pDoubleSpinBox, i + 1, 1);
276 connect(pDoubleSpinBox, static_cast<void (QDoubleSpinBox::*)(double)>(&QDoubleSpinBox::valueChanged),
277 this, &ArtifactSettingsView::onChangeArtifactThreshold);
278 m_mapChThresholdsDoubleSpinBoxes[channelTypes.at(i)] = pDoubleSpinBox;
279
280 QSpinBox* pSpinBox = new QSpinBox(this);
281 pSpinBox->setPrefix("e");
282 pSpinBox->setMaximum(0);
283 pSpinBox->setMinimum(-10000);
284 pSpinBox->setValue(m_mapThresholdsSecond[channelTypes.at(i)]);
285 pGroupBoxArtifactRejection->addWidget(pSpinBox, i + 1, 2);
286 connect(pSpinBox, static_cast<void (QSpinBox::*)(int)>(&QSpinBox::valueChanged),
287 this, &ArtifactSettingsView::onChangeArtifactThreshold);
288 m_mapChThresholdsSpinBoxes[channelTypes.at(i)] = pSpinBox;
289 }
290 }
291 }
292 setLayout(pGroupBoxArtifactRejection.release());
293}

297void ArtifactSettingsView::onChangeArtifactThreshold()
298{
299 m_mapThresholds.clear();
300 m_mapThresholdsFirst.clear();
301 m_mapThresholdsSecond.clear();
302
303 if (m_pArtifactRejectionCheckBox) {
304 if (m_pArtifactRejectionCheckBox->isChecked()) {
305 m_mapThresholds["Active"] = 1.0;
306 m_bDoArtifactThresholdReduction = true;
307 } else {
308 m_mapThresholds["Active"] = 0.0;
309 m_bDoArtifactThresholdReduction = false;
310 }
311 }
312
313 for (const auto& [key, pDoubleSpinBox] : m_mapChThresholdsDoubleSpinBoxes) {
314 if (pDoubleSpinBox) {
315 const int exponent = m_mapChThresholdsSpinBoxes[key]->value();
316 m_mapThresholdsFirst[key] = pDoubleSpinBox->value();
317 m_mapThresholdsSecond[key] = exponent;
318
319 if (mne::to_lower(key).contains("grad")) {
320 m_mapThresholds["grad"] = pDoubleSpinBox->value() * pow(10, exponent);
321 }
322 if (mne::to_lower(key).contains("mag")) {
323 m_mapThresholds["mag"] = pDoubleSpinBox->value() * pow(10, exponent);
324 }
325 if (mne::to_lower(key).contains("eeg")) {
326 m_mapThresholds["eeg"] = pDoubleSpinBox->value() * pow(10, exponent);
327 }
328 if (mne::to_lower(key).contains("ecg")) {
329 m_mapThresholds["ecg"] = pDoubleSpinBox->value() * pow(10, exponent);
330 }
331 if (mne::to_lower(key).contains("eog")) {
332 m_mapThresholds["eog"] = pDoubleSpinBox->value() * pow(10, exponent);
333 }
334 if (mne::to_lower(key).contains("emg")) {
335 m_mapThresholds["emg"] = pDoubleSpinBox->value() * pow(10, exponent);
336 }
337 }
338 }
339
340 emit changeArtifactThreshold(m_mapThresholds);
341
342 saveSettings();
343}
