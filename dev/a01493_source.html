40PixelRect PaneLayout::slot_rect(int slot, int pane_count, PixelSize size) const
41{
42 const int w = size.width;
43 const int h = size.height;
44 if (pane_count <= 1) {
45 return {0, 0, w, h};
46 }
47
48 const int left_w = split_pos(m_split_x, w);
49 const int right_w = std::max(1, w - left_w);
50
51 if (pane_count == 2) {
52 return (slot == 0) ? PixelRect{0, 0, left_w, h} : PixelRect{left_w, 0, right_w, h};
53 }
54
55 const int top_h = split_pos(m_split_y, h);
56 const int bottom_h = std::max(1, h - top_h);
57
58 if (pane_count == 3) {
59 if (slot == 0) {
60 return {0, 0, w, top_h};
61 }
62 return (slot == 1) ? PixelRect{0, top_h, left_w, bottom_h} : PixelRect{left_w, top_h, right_w, bottom_h};
63 }
64
65 const int col = slot % 2;
66 const int row = slot / 2;
67 return {(col == 0) ? 0 : left_w, (row == 0) ? 0 : top_h, (col == 0) ? left_w : right_w, (row == 0) ? top_h : bottom_h};
68}

72SplitterHit PaneLayout::hit_test_splitter(Point pos, int pane_count, PixelSize size) const
73{
74 if (pane_count <= 1) {
75 return SplitterHit::None;
76 }
77
78 const int split_x = split_pos(m_split_x, size.width);
79 const bool near_vertical = std::abs(pos.x - split_x) <= m_hit_tolerance_px;
80 if (pane_count == 2) {
81 return near_vertical ? SplitterHit::Vertical : SplitterHit::None;
82 }
83
84 const int split_y = split_pos(m_split_y, size.height);
85 const bool near_horizontal = std::abs(pos.y - split_y) <= m_hit_tolerance_px;
86 // With three panes the vertical splitter only divides the bottom row
87 const bool vertical_active = near_vertical && (pane_count != 3 || pos.y > split_y);
88
89 if (vertical_active && near_horizontal) {
90 return SplitterHit::Both;
91 }
92 if (vertical_active) {
93 return SplitterHit::Vertical;
94 }
95 if (near_horizontal) {
96 return SplitterHit::Horizontal;
97 }
98 return SplitterHit::None;
99}

133PixelRect PaneLayout::inset_for_separator(const PixelRect& pane, int slot, int pane_count) const
134{
135 PixelRect r = pane;
136 bool right_neighbour = false;
137 bool bottom_neighbour = false;
138 if (pane_count == 2) {
139 right_neighbour = slot == 0;
140 } else if (pane_count == 3) {
141 bottom_neighbour = slot == 0;
142 right_neighbour = slot == 1;
143 } else if (pane_count >= 4) {
144 right_neighbour = (slot % 2 == 0) && (slot + 1 < pane_count);
145 bottom_neighbour = (slot / 2 == 0) && (slot + 2 < pane_count);
146 }
147 if (right_neighbour) {
148 r.width = std::max(1, r.width - m_separator_line_px);
149 }
150 if (bottom_neighbour) {
151 r.height = std::max(1, r.height - m_separator_line_px);
152 }
153 return r;
154}

158PaneLayout::Separators PaneLayout::separator_geometries(int pane_count, PixelSize size) const
159{
160 Separators separators;
161 if (pane_count <= 1) {
162 return separators;
163 }
164
165 const int w = std::max(1, size.width);
166 const int h = std::max(1, size.height);
167 const int split_x = split_pos(m_split_x, w);
168 const int half_line = m_separator_line_px / 2;
169
170 if (pane_count == 2) {
171 separators.vertical = {split_x - half_line, 0, m_separator_line_px, h};
172 return separators;
173 }
174
175 const int split_y = split_pos(m_split_y, h);
176 separators.horizontal = {0, split_y - half_line, w, m_separator_line_px};
177 // With three panes the vertical separator divides the bottom row only
178 separators.vertical = (pane_count == 3) ? PixelRect{split_x - half_line, split_y, m_separator_line_px, h - split_y}
179 : PixelRect{split_x - half_line, 0, m_separator_line_px, h};
180 return separators;
181}

185void PaneLayout::drag_splitter(Point pos, SplitterHit splitter, PixelSize size)
186{
187 const int w = std::max(1, size.width);
188 const int h = std::max(1, size.height);
189
190 if (splitter == SplitterHit::Vertical || splitter == SplitterHit::Both) {
191 const int x = std::clamp(static_cast<int>(std::lround(pos.x)), m_min_pane_px, std::max(m_min_pane_px, w - m_min_pane_px));
192 m_split_x = clamp_split(static_cast<float>(x) / static_cast<float>(w));
193 }
194 if (splitter == SplitterHit::Horizontal || splitter == SplitterHit::Both) {
195 const int y = std::clamp(static_cast<int>(std::lround(pos.y)), m_min_pane_px, std::max(m_min_pane_px, h - m_min_pane_px));
196 m_split_y = clamp_split(static_cast<float>(y) / static_cast<float>(h));
197 }
198}
