99bool SensorLayout::click(Point point, bool extend)
100{
101 const int hit = sensor_at(point);
102 bool changed = false;
103 for (std::size_t i = 0; i < m_sensors.size(); ++i) {
104 const bool isHit = static_cast<int>(i) == hit;
105 bool select = m_sensors[i].selected;
106 if (extend) {
107 select = isHit ? !select : select;
108 } else {
109 select = isHit;
110 }
111 changed |= select != m_sensors[i].selected;
112 m_sensors[i].selected = select;
113 }
114 return changed;
115}

160void SensorLayout::draw(Canvas& canvas, const TextMetrics& metrics, const Font& font) const
161{
162 canvas.save();
163 canvas.set_antialiasing(true);
164 canvas.set_font(font);
165 const double line = metrics.height(font);
166 for (std::size_t i = 0; i < m_sensors.size(); ++i) {
167 const Sensor& sensor = m_sensors[i];
168 if (!sensor.visible) {
169 continue;
170 }
171 const Point p = scene_position(i);
172 canvas.save();
173 canvas.set_opacity(sensor.selectable ? 1.0 : 0.25);
174 // Shadow, then the dot
175 canvas.set_pen(Pen::none());
176 canvas.set_brush(Brush::solid({128, 128, 128}));
177 canvas.draw_ellipse({p.x + 3, p.y + 3}, kRadius, kRadius);
178 canvas.set_brush(Brush::solid(sensor.bad ? Color{255, 0, 0} : sensor.color));
179 canvas.draw_ellipse(p, kRadius, kRadius);
180 canvas.set_brush(Brush{});
181 if (sensor.highlighted) {
182 canvas.set_pen(Pen{{255, 84, 22}, 4});
183 canvas.draw_ellipse(p, kRadius, kRadius);
184 }
185 if (sensor.selected) {
186 canvas.set_pen(Pen{{255, 0, 0}, 5});
187 canvas.draw_ellipse(p, kRadius, kRadius);
188 }
189 canvas.set_pen(Color::black());
190 const double width = metrics.advance(font, sensor.name);
191 canvas.draw_text(Rect{p.x - width / 2.0, p.y - 32, width, line}, Align::Left | Align::Top, sensor.name);
192 canvas.restore();
193 }
194 canvas.restore();
195}
