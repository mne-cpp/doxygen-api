60[[nodiscard]] inline AcqExpected<DLManagedTensorVersioned*> toDLPack(const AcqBlockView& view, std::size_t part = 0)
61{
62 if (part >= view.partCount()) {
63 return AcqErrorCode::InvalidParameter;
64 }
65 DLDataType dtype{};
66 dtype.lanes = 1;
67 switch (view.format()) {
68 case AcqSampleFormat::Int16:
69 dtype.code = kDLInt;
70 dtype.bits = 16;
71 break;
72 case AcqSampleFormat::Int32:
73 dtype.code = kDLInt;
74 dtype.bits = 32;
75 break;
76 case AcqSampleFormat::Float32:
77 dtype.code = kDLFloat;
78 dtype.bits = 32;
79 break;
80 case AcqSampleFormat::Float64:
81 dtype.code = kDLFloat;
82 dtype.bits = 64;
83 break;
84 case AcqSampleFormat::ComplexFloat32:
85 dtype.code = kDLComplex;
86 dtype.bits = 64;
87 break;
88 default:
89 return AcqErrorCode::UnsupportedFormat;
90 }
91 const AcqBlockPart& run = view.part(part);
92 const auto bytes = static_cast<std::ptrdiff_t>(bytesPerSample(view.format()));
93 if (run.channelStride % bytes != 0 || run.sampleStride % bytes != 0) {
94 return AcqErrorCode::UnsupportedFormat; // DLPack strides count elements
95 }
96 DLDevice device{kDLCPU, 0};
97 const AcqMemoryLocation memory = view.memory();
98 const std::int32_t ordinal = memory.device < 0 ? 0 : memory.device;
99 switch (memory.kind) {
100 case AcqMemoryKind::Host:
101 break;
102 case AcqMemoryKind::HostPinned:
103 device = {kDLCUDAHost, ordinal};
104 break;
105 case AcqMemoryKind::Device:
106 device = {kDLCUDA, ordinal};
107 break;
108 case AcqMemoryKind::Managed:
109 device = {kDLCUDAManaged, ordinal};
110 break;
111 }

113 struct Holder
114 {
115 DLManagedTensorVersioned managed; // every field is set below
116 std::int64_t shape[2]{};
117 std::int64_t strides[2]{};
118 };
119 auto* holder = new (std::nothrow) Holder;
120 if (holder == nullptr) {
121 return AcqErrorCode::OutOfMemory;
122 }
123 holder->shape[0] = static_cast<std::int64_t>(view.channelCount());
124 holder->shape[1] = static_cast<std::int64_t>(run.samples);
125 holder->strides[0] = run.channelStride / bytes;
126 holder->strides[1] = run.sampleStride / bytes;
127 DLManagedTensorVersioned& m = holder->managed;
128 m.version = {DLPACK_MAJOR_VERSION, DLPACK_MINOR_VERSION};
129 m.manager_ctx = holder;
130 m.deleter = [](DLManagedTensorVersioned* self) {
131 std::unique_ptr<Holder>(static_cast<Holder*>(self->manager_ctx)).reset();
132 };
133 m.flags = DLPACK_FLAG_BITMASK_READ_ONLY;
134 m.dl_tensor.data = const_cast<std::byte*>(run.data); // read-only by the flag above
135 m.dl_tensor.device = device;
136 m.dl_tensor.ndim = 2;
137 m.dl_tensor.dtype = dtype;
138 m.dl_tensor.shape = holder->shape;
139 m.dl_tensor.strides = holder->strides;
140 m.dl_tensor.byte_offset = 0;
141 return &m;
142}
