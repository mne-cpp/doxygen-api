279void EvokedSetModel::updateData()
280{
281 if (!m_pEvokedSet) {
282 return;
283 }
284
285 m_matData.clear();
286 m_lAvrTypes.clear();
287
288 for (const FiffEvoked& evoked : m_pEvokedSet->evoked) {
289 m_matData.push_back(m_correction.apply(evoked.data));
290 m_pairBaseline = evoked.baseline;
291 m_lAvrTypes.push_back(evoked.comment);
292 }
293
294 // Update average selection information map. Use old colors if existing.
295 std::vector<std::string> slCurrentAvrComments;
296 for (const FiffEvoked& evoked : m_pEvokedSet->evoked) {
297 slCurrentAvrComments.push_back(evoked.comment);
298 }
299
300 const bool bActivationChanged = viz::reconcile_conditions(*m_qMapAverageActivation, *m_qMapAverageActivationOld, slCurrentAvrComments, true);
301 const bool bColorChanged = viz::reconcile_conditions(*m_qMapAverageColor, *m_qMapAverageColorOld, slCurrentAvrComments, QColor(Qt::yellow));
302
303 // Only emit new colors and activations if evoked types were added or deleted
304 if (bColorChanged) {
305 emit newAverageColorMap(m_qMapAverageColor);
306 }
307
308 if (bActivationChanged) {
309 emit newAverageActivationMap(m_qMapAverageActivation);
310 }
311
312 //Update data content
313 QModelIndex topLeft = this->index(0, 1);
314 QModelIndex bottomRight = this->index(m_pEvokedSet->info.nchan - 1, 1);
315 emit dataChanged(topLeft, bottomRight, {Qt::DisplayRole});
316}

507{
508 if (!m_pEvokedSet) {

510 }
511
512 beginResetModel();
513
514 m_qMapIdxRowSelection.clear();

519 m_qMapIdxRowSelection.insert_or_assign(count, selection[i]);
520 ++count;
521 }
522 }
523
524 emit newSelection(selection);
525
526 endResetModel();
