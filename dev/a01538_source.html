142bool ButterflyFigure::shown(int row) const
143{
144 if (row < 0 || row >= static_cast<int>(m_channels.size()) || std::ranges::find(m_selection, row) == m_selection.end()) {
145 return false;
146 }
147 const Channel& channel = m_channels[static_cast<std::size_t>(row)];
148 switch (channel.kind) {
149 case FIFFV_MEG_CH:
150 return (channel.unit == FIFF_UNIT_T_M && value_or_false(m_modalities, "GRAD")) ||
151 (channel.unit == FIFF_UNIT_T && value_or_false(m_modalities, "MAG"));
152 case FIFFV_EEG_CH:
153 return value_or_false(m_modalities, "EEG");
154 case FIFFV_EOG_CH:
155 return value_or_false(m_modalities, "EOG");
156 case FIFFV_MISC_CH:
157 return value_or_false(m_modalities, "MISC");
158 default:
159 return false;
160 }
161}

165ButterflyPlot ButterflyFigure::plot() const
166{
167 ButterflyPlot plot;
168 plot.set_background(m_background);
169 plot.set_foreground(m_foreground);
170 plot.set_timing(m_timing.samples, m_timing.pre_stim_samples, m_timing.sfreq);
171 plot.set_baseline(m_timing.baseline_from, m_timing.baseline_to);
172 plot.set_time_spacers(m_timing.time_spacers);
173 plot.set_frozen(m_timing.frozen);
174
175 std::vector<ButterflyPlot::Channel> channels;
176 channels.reserve(m_selection.size());
177 for (int row = 0; row < static_cast<int>(m_channels.size()); ++row) {
178 if (!shown(row)) {
179 continue;
180 }
181 const Channel& channel = m_channels[static_cast<std::size_t>(row)];
182 ButterflyPlot::Channel drawn;
183 drawn.max_value = channel_scale(m_scales, channel.kind, channel.unit);
184 drawn.bad = channel.bad;
185 drawn.traces.reserve(channel.averages.size());
186 for (const Average& average : channel.averages) {
187 if (!average.data || !value_or_false(m_activation, average.comment)) {
188 continue;
189 }
190 ButterflyPlot::Trace trace;
191 trace.samples.reserve(static_cast<std::size_t>(average.samples));
192 for (int sample = 0; sample < average.samples; ++sample) {
193 trace.samples.push_back(average.data[static_cast<std::ptrdiff_t>(sample) * average.stride]);
194 }
195 const auto color = m_colors.find(average.comment);
196 trace.color = color != m_colors.end() ? color->second : m_signalColor;
197 drawn.traces.push_back(std::move(trace));
198 }
199 channels.push_back(std::move(drawn));
200 }
201 plot.set_channels(std::move(channels));
202 return plot;
203}
