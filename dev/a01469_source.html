78#if defined(__linux__)
80bool parseCpuList(const std::string& text, std::vector<unsigned>& out)
81{
82 const char* p = text.data();
83 const char* end = p + text.size();
84 while (p < end && *p != '\n') {
85 unsigned first = 0;
86 auto r = std::from_chars(p, end, first);
87 if (r.ec != std::errc()) {
88 return false;
89 }
90 unsigned last = first;
91 p = r.ptr;
92 if (p < end && *p == '-') {
93 r = std::from_chars(p + 1, end, last);
94 if (r.ec != std::errc() || last < first) {
95 return false;
96 }
97 p = r.ptr;
98 }
99 for (unsigned c = first; c <= last; ++c) {
100 out.push_back(c);
101 }
102 if (p < end && *p == ',') {
103 ++p;
104 }
105 }
106 return true;
107}
108#endif

115AcqError mne::acqPinThisThread(std::span<const unsigned> cpus) noexcept
116{
117#if defined(__linux__)
118 if (cpus.empty()) {
119 return AcqError(AcqErrorCode::InvalidParameter, "no CPU given");
120 }
121 cpu_set_t set;
122 CPU_ZERO(&set);
123 for (const unsigned c : cpus) {
124 if (c >= static_cast<unsigned>(CPU_SETSIZE)) {
125 return AcqError(AcqErrorCode::InvalidParameter, "CPU " + std::to_string(c) + " does not exist");
126 }
127 CPU_SET(c, &set);
128 }
129 const int rc = pthread_setaffinity_np(pthread_self(), sizeof(set), &set);
130 if (rc == EINVAL) {
131 return AcqError(AcqErrorCode::InvalidParameter, "none of the CPUs is available to this process");
132 }
133 return rc == 0 ? AcqError() : systemError("pthread_setaffinity_np", rc);
134#elif defined(_WIN32)
135 if (cpus.empty()) {
136 return AcqError(AcqErrorCode::InvalidParameter, "no CPU given");
137 }
138 GROUP_AFFINITY affinity{};
139 affinity.Group = static_cast<WORD>(cpus.front() / 64);
140 for (const unsigned c : cpus) {
141 if (c / 64 != affinity.Group || c / 64 >= GetActiveProcessorGroupCount() ||
142 c % 64 >= GetActiveProcessorCount(static_cast<WORD>(c / 64))) {
143 return AcqError(AcqErrorCode::InvalidParameter,
144 "CPU " + std::to_string(c) + " does not exist or is in another processor group");
145 }
146 affinity.Mask |= KAFFINITY(1) << (c % 64);
147 }
148 if (SetThreadGroupAffinity(GetCurrentThread(), &affinity, nullptr) == 0) {
149 return AcqError(AcqErrorCode::DeviceIo, "SetThreadGroupAffinity failed: " + std::to_string(GetLastError()));
150 }
151 return {};
152#else
153 static_cast<void>(cpus);
154 return AcqError(AcqErrorCode::PlatformUnsupported, "this platform does not let threads choose their CPUs");
155#endif
156}

160AcqError mne::acqSetThisThreadPriority(AcqThreadPriority priority) noexcept
161{
162#if defined(__linux__)
163 const auto tid = static_cast<id_t>(syscall(SYS_gettid));
164 if (priority == AcqThreadPriority::Realtime) {
165 sched_param param{};
166 param.sched_priority = std::min(80, sched_get_priority_max(SCHED_FIFO));
167 const int rc = pthread_setschedparam(pthread_self(), SCHED_FIFO, &param);
168 return rc == 0 ? AcqError() : systemError("SCHED_FIFO", rc);
169 }
170 sched_param param{};
171 if (const int rc = pthread_setschedparam(pthread_self(), SCHED_OTHER, &param); rc != 0) {
172 return systemError("SCHED_OTHER", rc);
173 }
174 if (setpriority(PRIO_PROCESS, tid, priority == AcqThreadPriority::High ? -10 : 0) != 0) {
175 return systemError("setpriority", errno);
176 }
177 return {};
178#elif defined(_WIN32)
179 if (g_mmcss != nullptr) {
180 AvRevertMmThreadCharacteristics(g_mmcss);
181 g_mmcss = nullptr;
182 }
183 int level = THREAD_PRIORITY_NORMAL;
184 if (priority == AcqThreadPriority::Realtime) {
185 DWORD task = 0;
186 g_mmcss = AvSetMmThreadCharacteristicsW(L"Pro Audio", &task);
187 if (g_mmcss == nullptr) {
188 return AcqError(AcqErrorCode::DeviceIo, "MMCSS registration failed: " + std::to_string(GetLastError()));
189 }
190 level = THREAD_PRIORITY_TIME_CRITICAL;
191 } else if (priority == AcqThreadPriority::High) {
192 level = THREAD_PRIORITY_HIGHEST;
193 }
194 if (SetThreadPriority(GetCurrentThread(), level) == 0) {
195 return AcqError(AcqErrorCode::DeviceIo, "SetThreadPriority failed: " + std::to_string(GetLastError()));
196 }
197 return {};
198#elif defined(__APPLE__) && defined(__clang__)
199 const thread_port_t self = pthread_mach_thread_np(pthread_self());
200 if (priority == AcqThreadPriority::Realtime) {
201 mach_timebase_info_data_t timebase{};
202 mach_timebase_info(&timebase);
203 const auto ticks = [&](double ns) {
204 return static_cast<std::uint32_t>(ns * timebase.denom / timebase.numer);
205 };
206 thread_time_constraint_policy_data_t policy{};
207 policy.period = ticks(1e6);
208 policy.computation = ticks(5e5);
209 policy.constraint = ticks(1e6);
210 policy.preemptible = 1;
211 const kern_return_t rc = thread_policy_set(self, THREAD_TIME_CONSTRAINT_POLICY,
212 reinterpret_cast<thread_policy_t>(&policy),
213 THREAD_TIME_CONSTRAINT_POLICY_COUNT);
214 if (rc != KERN_SUCCESS) {
215 return AcqError(AcqErrorCode::DeviceIo, "time-constraint policy: " + std::to_string(rc));
216 }
217 g_optedOutOfQos = true;
218 return {};
219 }
220 if (g_optedOutOfQos) {
221 // The time-constraint policy opted the thread out of QoS for good (so does the standard policy, hence only
222 // here): the standard policy is all that is left.
223 thread_standard_policy_data_t standard{};
224 thread_policy_set(self, THREAD_STANDARD_POLICY, reinterpret_cast<thread_policy_t>(&standard),
225 THREAD_STANDARD_POLICY_COUNT);
226 return priority == AcqThreadPriority::Normal
227 ? AcqError()
228 : AcqError(AcqErrorCode::InvalidState, "a macOS thread that was realtime cannot become high, only normal");
229 }
230 const int rc =
231 pthread_set_qos_class_self_np(priority == AcqThreadPriority::High ? QOS_CLASS_USER_INTERACTIVE : QOS_CLASS_DEFAULT, 0);
232 return rc == 0 ? AcqError() : systemError("pthread_set_qos_class_self_np", rc);
233#else
234 return priority == AcqThreadPriority::Normal
235 ? AcqError()
236 : AcqError(AcqErrorCode::PlatformUnsupported, "this platform has no thread priorities");
237#endif
238}

242AcqError mne::acqApplyThreadPolicy(const AcqThreadPolicy& policy) noexcept
243{
244 if (!policy.cpus.empty()) {
245 if (AcqError e = acqPinThisThread(policy.cpus); !e.ok()) {
246 return e;
247 }
248 }
249 return policy.priority == AcqThreadPriority::Normal ? AcqError() : acqSetThisThreadPriority(policy.priority);
250}

269AcqExpected<std::vector<unsigned>> mne::acqNumaNodeCpus(int node)
270{
271 if (node < 0) {
272 return AcqError(AcqErrorCode::InvalidParameter, "NUMA node " + std::to_string(node) + " does not exist");
273 }
274 std::vector<unsigned> cpus;
275#if defined(__linux__)
276 std::ifstream file("/sys/devices/system/node/node" + std::to_string(node) + "/cpulist");
277 std::string text;
278 if (file && std::getline(file, text)) {
279 if (!parseCpuList(text, cpus)) {
280 return AcqError(AcqErrorCode::DeviceIo, "cannot parse the CPU list of NUMA node " + std::to_string(node));
281 }
282 return cpus;
283 }
284#elif defined(_WIN32)
285 ULONG highest = 0;
286 GROUP_AFFINITY affinity{};
287 if (GetNumaHighestNodeNumber(&highest) != 0 && static_cast<ULONG>(node) <= highest &&
288 GetNumaNodeProcessorMaskEx(static_cast<USHORT>(node), &affinity) != 0) {
289 for (unsigned bit = 0; bit < 64; ++bit) {
290 if ((affinity.Mask >> bit) & 1U) {
291 cpus.push_back(static_cast<unsigned>(affinity.Group) * 64 + bit);
292 }
293 }
294 return cpus;
295 }
296#endif
297 // Without NUMA information every CPU belongs to node 0.
298 if (node != 0) {
299 return AcqError(AcqErrorCode::InvalidParameter, "NUMA node " + std::to_string(node) + " does not exist");
300 }
301 for (unsigned c = 0; c < std::max(1U, std::thread::hardware_concurrency()); ++c) {
302 cpus.push_back(c);
303 }
304 return cpus;
305}

309AcqExpected<std::thread> mne::start_thread(const AcqThreadPolicy& policy, std::function<void()> body)
310{
311 // The promise's shared state outlives both ends, unlike a flag the thread could still notify after we return
312 std::promise<AcqError> applied;
313 std::future<AcqError> result = applied.get_future();
314 std::thread thread([policy, body = std::move(body), applied = std::move(applied)]() mutable {
315 AcqError error = acqApplyThreadPolicy(policy);
316 const bool ok = error.ok();
317 applied.set_value(std::move(error));
318 if (ok) {
319 body();
320 }
321 });
322 if (AcqError error = result.get(); !error.ok()) {
323 thread.join();
324 return error;
325 }
326 return thread;
327}
