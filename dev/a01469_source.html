51// Moves a helmet into head coordinates; normals turn with the inverse transpose
52void transformHelmet(mne::MNEBemSurface& helmet, const Eigen::Matrix4f& devHead)
53{
54 const Eigen::Matrix3f normalMat = devHead.topLeftCorner<3, 3>().inverse().transpose();
55 for (int i = 0; i < helmet.rr.rows(); ++i) {
56 const Eigen::Vector4f p = devHead * Eigen::Vector3f(helmet.rr.row(i).transpose()).homogeneous();
57 helmet.rr.row(i) = (p.head<3>() / p.w()).transpose();
58 helmet.nn.row(i) = (normalMat * Eigen::Vector3f(helmet.nn.row(i).transpose())).normalized().transpose();
59 }
60}

71DataLoader::SensorLoadResult DataLoader::loadSensors(const std::filesystem::path& fifPath,
72 const std::filesystem::path& megHelmetOverridePath)
73{
74 SensorLoadResult result;
75
76 std::error_code error;
77 if (!std::filesystem::exists(fifPath, error))
78 return result;
79
80 FiffInfo info;
81 FiffDigPointSet digSet;
82 FiffStream stream(fifPath);
83
84 if (stream.open()) {
85 std::shared_ptr<FiffDirNode> tree = stream.dirtree();
86 std::shared_ptr<FiffDirNode> nodeInfo;
87 if (stream.read_meas_info(tree, info, nodeInfo)) {
88 result.hasInfo = !info.empty();
89 }
90 }
91
92 if (result.hasInfo) {
93 result.info = info;
94
95 // Prepare Device->Head transformation
96 Eigen::Matrix4f devHead = Eigen::Matrix4f::Identity();
97 bool hasDevHead = false;
98 if (!info.dev_head_t.empty() &&
99 info.dev_head_t.from == FIFFV_COORD_DEVICE &&
100 info.dev_head_t.to == FIFFV_COORD_HEAD &&
101 !info.dev_head_t.trans.isIdentity()) {
102 hasDevHead = true;
103 devHead = info.dev_head_t.trans;
104 }
105
106 result.devHeadTrans = devHead;
107 result.hasDevHead = hasDevHead;
108
109 for (const auto& ch : info.chs) {
110 if (ch.kind == FIFFV_MEG_CH) {
111 DataLoader::SensorPlacement sensor;
112 sensor.name = ch.ch_name;
113 sensor.position = ch.chpos.r0;
114 sensor.orientation.topLeftCorner<3, 3>() = ch.coil_trans.topLeftCorner<3, 3>();
115 if (hasDevHead) {
116 const Eigen::Vector4f p = devHead * ch.chpos.r0.homogeneous();
117 sensor.position = p.head<3>() / p.w();
118 sensor.orientation.topLeftCorner<3, 3>() = devHead.topLeftCorner<3, 3>() * sensor.orientation.topLeftCorner<3, 3>();
119 }
120
121 if (ch.unit == FIFF_UNIT_T_M) {
122 result.megGrad.push_back(std::move(sensor));
123 } else {
124 result.megMag.push_back(std::move(sensor));
125 }
126 } else if (ch.kind == FIFFV_EEG_CH) {
127 result.eeg.push_back({ch.ch_name, ch.chpos.r0, Eigen::Matrix4f::Identity()});
128 }
129 }
130
131 // ── MEG helmet surface ──────────────────────────────────────
132 if (!result.megGrad.empty() || !result.megMag.empty()) {
133 auto pickHelmetFile = [&info]() -> std::filesystem::path {
134 int coilType = -1;
135 int nMeg = 0;
136 for (const auto& ch : info.chs) {
137 if (ch.kind == FIFFV_MEG_CH) {
138 coilType = ch.chpos.coil_type & 0xFFFF;
139 ++nMeg;
140 }
141 }
142
143 std::string fileName = "306m.fif";
144 if (coilType == FIFFV_COIL_BABY_GRAD) {
145 fileName = "BabySQUID.fif";
146 } else if (coilType == FIFFV_COIL_NM_122) {
147 fileName = "122m.fif";
148 } else if (coilType == FIFFV_COIL_CTF_GRAD) {
149 fileName = "CTF_275.fif";
150 } else if (coilType == FIFFV_COIL_KIT_GRAD) {
151 fileName = "KIT.fif";
152 } else if (coilType == FIFFV_COIL_MAGNES_MAG || coilType == FIFFV_COIL_MAGNES_GRAD) {
153 fileName = (nMeg > 150) ? "Magnes_3600wh.fif" : "Magnes_2500wh.fif";
154 } else if (coilType / 1000 == 3) {
155 fileName = "306m.fif";
156 }
157
158 return mne::executable_dir() / ".." / "resources" / "general" / "sensorSurfaces" / fileName;
159 };
160
161 const auto fileExists = [](const std::filesystem::path& path) {
162 std::error_code existsError;
163 return !path.empty() && std::filesystem::exists(path, existsError);
164 };
165
166 std::filesystem::path helmetPath;
167 if (!megHelmetOverridePath.empty()) {
168 helmetPath = megHelmetOverridePath;
169 if (!fileExists(helmetPath)) {
170 mne::Log::warning("MEG helmet override file not found: {} - falling back to auto selection.", mne::utf8_from_path(helmetPath));
171 helmetPath.clear();
172 }
173 }
174
175 if (helmetPath.empty()) {
176 helmetPath = pickHelmetFile();
177#ifdef __EMSCRIPTEN__
178 // WASM: the executable directory may not resolve correctly.
179 // Try the preloaded virtual FS path directly.
180 if (!fileExists(helmetPath)) {
181 std::filesystem::path fn = helmetPath.filename();
182 if (fn.empty())
183 fn = "306m.fif";
184 helmetPath = std::filesystem::path("/resources/general/sensorSurfaces") / fn;
185 }
186#endif
187 }
188
189 if (!fileExists(helmetPath)) {
190 const std::filesystem::path fallback = mne::executable_dir() / ".." / "resources" / "general" / "sensorSurfaces" / "306m.fif";
191 if (fileExists(fallback)) {
192 helmetPath = fallback;
193 }
194 }
195
196#ifdef __EMSCRIPTEN__
197 // WASM fallback: sensor surfaces are preloaded into the
198 // Emscripten virtual FS at /resources/general/sensorSurfaces/.
199 // The executable directory may not resolve the /../ correctly.
200 if (!fileExists(helmetPath)) {
201 std::filesystem::path fn = helmetPath.filename();
202 if (fn.empty())
203 fn = "306m.fif";
204 const std::filesystem::path wasmFallback = std::filesystem::path("/resources/general/sensorSurfaces") / fn;
205 if (fileExists(wasmFallback)) {
206 helmetPath = wasmFallback;
207 }
208 }
209#endif
210
211 if (!fileExists(helmetPath)) {
212 mne::Log::warning("MEG helmet surface file not found. Checked: {}", mne::utf8_from_path(helmetPath));
213 } else {
214 MNEBem helmetBem(helmetPath);
215 if (helmetBem.size() > 0) {
216 MNEBemSurface helmetSurf = helmetBem[0];
217 if (helmetSurf.nn.rows() != helmetSurf.rr.rows()) {
218 helmetSurf.nn = mne::FsSurface::compute_normals(Eigen::MatrixX3f(helmetSurf.rr), Eigen::MatrixX3i(helmetSurf.itris));
219 }
220
221 if (hasDevHead) {
222 transformHelmet(helmetSurf, devHead);
223 }
224
225 auto helmetSurface = std::make_shared<BrainSurface>();
226 helmetSurface->fromBemSurface(helmetSurf, Color{0, 0, 77, 200});
227 helmetSurface->setVisible(true);
228 result.helmetSurface = helmetSurface;
229 } else {
230 mne::Log::warning("DataLoader::loadSensors: helmetBem[0] has 0 verts/tris!");
231 }
232 }
233 }
234 }
235
236 // ── Digitizer points ────────────────────────────────────────────
237 if (result.hasInfo && info.dig.size() > 0) {
238 result.digitizerPoints = info.dig;
239 result.hasDigitizer = true;
240 } else if (!result.hasInfo) {
241 if (auto dig = FiffDigPointSet::read(fifPath); dig && !dig->empty()) {
242 result.digitizerPoints = std::move(*dig);
243 result.hasDigitizer = true;
244 }
245 }
246
247 return result;
248}

252std::shared_ptr<BrainSurface> DataLoader::loadHelmetSurface(
253 const std::filesystem::path& helmetFilePath,
254 const Eigen::Matrix4f& devHeadTrans,
255 bool applyTrans)
256{
257 std::error_code error;
258 if (!std::filesystem::exists(helmetFilePath, error)) {
259 mne::Log::warning("DataLoader::loadHelmetSurface: file not found: {}", mne::utf8_from_path(helmetFilePath));
260 return nullptr;
261 }
262
263 MNEBem helmetBem(helmetFilePath);
264 if (helmetBem.size() == 0) {
265 mne::Log::warning("DataLoader::loadHelmetSurface: no BEM surfaces in {}", mne::utf8_from_path(helmetFilePath));
266 return nullptr;
267 }
268
269 MNEBemSurface helmetSurf = helmetBem[0];
270
271 if (helmetSurf.nn.rows() != helmetSurf.rr.rows()) {
272 helmetSurf.nn = mne::FsSurface::compute_normals(Eigen::MatrixX3f(helmetSurf.rr), Eigen::MatrixX3i(helmetSurf.itris));
273 }
274
275 if (applyTrans) {
276 transformHelmet(helmetSurf, devHeadTrans);
277 }
278
279 auto surface = std::make_shared<BrainSurface>();
280 surface->fromBemSurface(helmetSurf, Color{0, 0, 77, 200});
281 surface->setVisible(true);
282
283 return surface;
284}

343std::vector<std::string> DataLoader::probeEvokedSets(const std::filesystem::path& evokedPath)
344{
345 std::vector<std::string> result;
346 std::error_code error;
347 if (!std::filesystem::exists(evokedPath, error)) {
348 return result;
349 }
350
351 const auto read = FiffEvokedSet::read(evokedPath);
352 if (!read) {
353 return result;
354 }
355 const FiffEvokedSet& evokedSet = *read;
356 for (int i = 0; i < std::ssize(evokedSet.evoked); ++i) {
357 const auto& ev = evokedSet.evoked.at(i);
358 result.push_back(std::format("{}: {} ({}, nave={})",
359 i,
360 ev.comment.empty() ? std::format("Set {}", i) : ev.comment,
361 ev.aspectKindToString(),
362 ev.nave));
363 }
364 return result;
365}
