92TriggerDetectionView::TriggerDetectionView(const QString& sSettingsPath,
93 QWidget* parent,
94 Qt::WindowFlags f)
95: AbstractView(parent, f)
96, m_pForm(std::make_shared<viz::SettingsForm>("Trigger Detection Settings"))
97{
98 m_sSettingsPath = sSettingsPath;
99
100 m_pForm->add_flag("realTime", true);
101
102 SettingField activate = realTimeField("activateTriggerDetection", SettingField::Kind::Bool, "Activate trigger detection");
103 activate.default_value = false;
104 m_pForm->add(activate);
105
106 m_pForm->add_choice("triggerChannel", "Trigger channel:", {}, {});
107
108 m_pForm->add(realTimeField("triggerColorType", SettingField::Kind::Choice, "Type color:"));
109 m_pForm->add(realTimeField("triggerColor", SettingField::Kind::Custom, {}));
110
111 m_pForm->add_double("detectionThresholdFirst", "Threshold:", 0.1, -1000.0, 1000.0, 0.1, 2);
112 SettingField exponent;
113 exponent.key = "detectionThresholdSecond";
114 exponent.kind = SettingField::Kind::Int;
115 exponent.default_value = -1;
116 exponent.minimum = -100;
117 exponent.maximum = 100;
118 exponent.decimals = 0;
119 exponent.prefix = "e^";
120 m_pForm->add(exponent);
121
122 m_pForm->add(realTimeField("numberDetectedTriggers", SettingField::Kind::Custom, "Detected trigger:"));
123 m_pForm->add(realTimeField("resetNumberTriggers", SettingField::Kind::Action, "Reset"));
124
125 SettingField detect;
126 detect.key = "detectTriggers";
127 detect.kind = SettingField::Kind::Action;
128 detect.label = "Detect Triggers";
129 detect.visible_if = "!realTime";
130 m_pForm->add(detect);
131
132 // The controls react once init() connected the view to measurement info
133 m_listener = m_pForm->add_listener([this](const std::string& key, const std::optional<viz::SettingValue>&) {
134 if (!m_bInitialized || m_bSilent) {
135 return;
136 }
137 if (key == "activateTriggerDetection" || key == "triggerChannel" || key.starts_with("detectionThreshold")) {
138 onTriggerInfoChanged();
139 } else if (key == "triggerColorType") {
140 onRealTimeTriggerColorTypeChanged(QString::fromStdString(m_pForm->get<std::string>("triggerColorType")));
141 } else if (key == "resetNumberTriggers") {
142 onResetTriggerNumbers();
143 } else if (key == "detectTriggers") {
144 onDetectTriggers();
145 }
146 });
147
148 auto* layout = new QVBoxLayout(this);
149 layout->setContentsMargins(0, 0, 0, 0);
150 auto* pPanel = new SettingsPanel(m_pForm, this);
151 layout->addWidget(pPanel);
152
153 if (auto* pColor = pPanel->customWidget("triggerColor")) {
154 m_pTriggerColorButton = new QPushButton(QStringLiteral("Click to change"), pColor);
155 m_pTriggerColorButton->setObjectName(QStringLiteral("triggerColorButton"));
156 pColor->layout()->addWidget(m_pTriggerColorButton);
157 connect(m_pTriggerColorButton, &QPushButton::clicked, this, [this](bool state) {
158 if (m_bInitialized) {
159 onRealTimeTriggerColorChanged(state);
160 }
161 });
162 }
163 if (auto* pCount = pPanel->customWidget("numberDetectedTriggers")) {
164 m_pNumberDetectedTriggers = new QLabel(QStringLiteral("0"), pCount);
165 m_pNumberDetectedTriggers->setObjectName(QStringLiteral("numberDetectedTriggersLabel"));
166 pCount->layout()->addWidget(m_pNumberDetectedTriggers);
167 }
168
169 this->setWindowTitle("Trigger Detection Settings");
170 this->setMinimumWidth(330);
171 this->setMaximumWidth(330);
172
173 updateProcessingMode(RealTime);
174
175 loadSettings();
176}

188void TriggerDetectionView::init(const std::shared_ptr<FiffInfo>& pFiffInfo)
189{
190 if (pFiffInfo) {
191 m_pFiffInfo = pFiffInfo;
192
193 std::vector<SettingChoice> channels;
194 for (int i = 0; i < std::ssize(m_pFiffInfo->chs); i++) {
195 if (m_pFiffInfo->chs[i].kind == FIFFV_STIM_CH) {
196 channels.push_back({m_pFiffInfo->chs[i].ch_name, m_pFiffInfo->chs[i].ch_name});
197 }
198 }
199 m_pForm->set_choices("triggerChannel", std::move(channels));
200
201 m_bInitialized = true;
202
203 if (m_pTriggerColorButton) {
204 m_pTriggerColorButton->setAutoFillBackground(true);
205 m_pTriggerColorButton->setFlat(true);
206 }
207 showColor(m_pTriggerColorButton, QColor(177, 0, 0));
208
209 loadSettings();
210 }

215void TriggerDetectionView::setNumberDetectedTriggersAndTypes(int numberDetections, const std::map<int, std::vector<std::pair<int, double>>>& mapDetectedTriggers)
216{
217 if (m_pNumberDetectedTriggers) {
218 m_pNumberDetectedTriggers->setText(QString("%1").arg(numberDetections));
219 }
220
221 //Set trigger types
222 std::vector<SettingChoice> types = m_pForm->field("triggerColorType")->choices;
223 const std::size_t count = types.size();
224 for (const auto& [channel, triggers] : mapDetectedTriggers) {
225 for (const auto& [sample, value] : triggers) {
226 const std::string type = QString::number(value).toStdString();
227 if (std::ranges::none_of(types, [&type](const SettingChoice& choice) { return choice.value == type; })) {
228 types.push_back({type, type});
229 }
230 }
231 }
232 if (types.size() != count) {
233 m_pForm->set_choices("triggerColorType", std::move(types));
234 }
235}

239void TriggerDetectionView::saveSettings()
240{
241 if (m_sSettingsPath.isEmpty()) {
242 return;
243 }
244
245 QSettings settings("MNECPP");
246
247 const std::vector<SettingChoice>& channels = m_pForm->field("triggerChannel")->choices;
248 const std::string channel = m_pForm->get<std::string>("triggerChannel");
249 const auto current = std::ranges::find(channels, channel, &SettingChoice::value);
250 const int channelIndex = current == channels.end() ? -1 : static_cast<int>(std::distance(channels.begin(), current));
251
252 settings.setValue(m_sSettingsPath + QString("/TriggerDetectionView/Activated"), m_pForm->get<bool>("activateTriggerDetection"));
253 settings.setValue(m_sSettingsPath + QString("/TriggerDetectionView/ChannelIndex"), channelIndex);
254 settings.setValue(m_sSettingsPath + QString("/TriggerDetectionView/FirstThresholdValue"), m_pForm->get<double>("detectionThresholdFirst"));
255 settings.setValue(m_sSettingsPath + QString("/TriggerDetectionView/SecondThresholdValue"), m_pForm->get<int>("detectionThresholdSecond"));
256
257 settings.beginGroup(m_sSettingsPath + QString("/TriggerDetectionView/Colors"));
258 std::map<double, QColor>::const_iterator i = m_qMapTriggerColor.cbegin();
259 while (i != m_qMapTriggerColor.cend()) {
260 settings.setValue(QString::number(i->first), i->second);
261 ++i;
262 }
263 settings.endGroup();
264}

268void TriggerDetectionView::loadSettings()
269{
270 if (m_sSettingsPath.isEmpty()) {
271 return;
272 }
273
274 QSettings settings("MNECPP");
275
276 {
277 const detail::ScopedFlag silent(m_bSilent);
278 m_pForm->set_value("activateTriggerDetection", settings.value(m_sSettingsPath + QString("/TriggerDetectionView/Activated"), false).toBool());
279 const std::vector<SettingChoice>& channels = m_pForm->field("triggerChannel")->choices;
280 const int channelIndex = settings.value(m_sSettingsPath + QString("/TriggerDetectionView/ChannelIndex"), 0).toInt();
281 if (channelIndex >= 0 && channelIndex < std::ssize(channels)) {
282 m_pForm->set_value("triggerChannel", channels[static_cast<std::size_t>(channelIndex)].value);
283 }
284 m_pForm->set_value("detectionThresholdFirst", settings.value(m_sSettingsPath + QString("/TriggerDetectionView/FirstThresholdValue"), 0.1).toDouble());
285 m_pForm->set_value("detectionThresholdSecond", settings.value(m_sSettingsPath + QString("/TriggerDetectionView/SecondThresholdValue"), -1).toInt());
286 }
287
288 settings.beginGroup(m_sSettingsPath + QString("/TriggerDetectionView/Colors"));
289 for (const QString& key : settings.childKeys()) {
290 double dKey = key.toDouble();
291 m_qMapTriggerColor[dKey] = settings.value(key).value<QColor>();
292 }
293 settings.endGroup();
294
295 onTriggerInfoChanged();
296}
