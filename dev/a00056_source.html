61bool MriCorIO::read(const std::string& dir,
62 std::vector<MriSlice>& slices,
63 bool verbose)
64{
65 //
66 // COR files contain 256 coronal slices of 256×256 unsigned chars (1mm isotropic).
67 //
68 // The coordinate system is:
69 // - Coronal orientation
70 // - 1mm pixel spacing
71 // - Origin offset: (0.128, -0.128, 0.128) meters
72 // - Rotation: x -> -x, y -> z, z -> y (coronal to MRI surface RAS)
73 //
74 // Ported from make_cor_set() in mne_make_cor_set/make_cor_set.c
75 //
76
77 const std::size_t nPixels = static_cast<std::size_t>(COR_WIDTH) * static_cast<std::size_t>(COR_HEIGHT);
78 slices.resize(COR_NSLICE);
79
80 for (int k = 0; k < COR_NSLICE; ++k) {
81 char name[16];
82 std::snprintf(name, sizeof(name), "/COR-%03d", k + 1);
83 const std::string fileName = dir + name;
84 const auto data = mne::read_file_bytes(mne::path_from_utf8(fileName));
85 if (!data) {
86 mne::Log::critical("MriCorIO::read - Could not open COR file {}", fileName);
87 return false;
88 }
89
90 if (data->size() < nPixels) {
91 mne::Log::critical("MriCorIO::read - COR file {} is too small: {} bytes (expected {})", fileName, data->size(), nPixels);
92 return false;
93 }
94
95 MriSlice& slice = slices[static_cast<std::size_t>(k)];
96 slice.fileName = fileName;
97 slice.width = COR_WIDTH;
98 slice.height = COR_HEIGHT;
99 slice.dimx = COR_PIXEL_SIZE;
100 slice.dimy = COR_PIXEL_SIZE;
101 slice.scale = 1.0f;
102 slice.pixelFormat = FIFFV_MRI_PIXEL_BYTE;
103
104 // Copy pixel data
105 slice.pixels.resize(nPixels);
106 std::memcpy(slice.pixels.data(), data->data(), nPixels);
107
108 // Build the coordinate transformation: slice -> MRI (surface RAS)
109 // Ported from the original C code in make_cor_set.c:
110 // move[X] = 0.128; move[Y] = -0.128 + k/1000.0; move[Z] = 0.128;
111 // rot = {{-1,0,0},{0,0,1},{0,1,0}};
112 Matrix3f rot;
113 rot << -1.0f, 0.0f, 0.0f,
114 0.0f, 0.0f, 1.0f,
115 0.0f, 1.0f, 0.0f;
116
117 Eigen::Vector3f move;
118 move << 0.128f, -0.128f + static_cast<float>(k) / 1000.0f, 0.128f;
119
120 slice.trans = FiffCoordTrans(FIFFV_COORD_MRI_SLICE, FIFFV_COORD_MRI, rot, move);
121 }
122
123 if (verbose) {
124 mne::Log::info("Read {} COR slices from {}", COR_NSLICE, dir);
125 }
126
127 return true;
128}
