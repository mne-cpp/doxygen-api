81bool SourceEstimateManager::load(const QString &lhPath, const QString &rhPath,
82 const QMap<QString, std::shared_ptr<BrainSurface>> &surfaces,
83 const QString &activeSurfaceType)
84{
85 if (m_isLoading) {
86 qWarning() << "SourceEstimateManager: STC loading already in progress";
87 return false;
88 }
89
90 // Find surfaces for the active surface type
91 BrainSurface *lhSurface = nullptr;
92 BrainSurface *rhSurface = nullptr;
93
94 const QString lhKey = "lh_" + activeSurfaceType;
95 const QString rhKey = "rh_" + activeSurfaceType;
96
97 if (surfaces.contains(lhKey))
98 lhSurface = surfaces[lhKey].get();
99 if (surfaces.contains(rhKey))
100 rhSurface = surfaces[rhKey].get();
101
102 // Fallback: search for any lh_*/rh_* brain surface
103 if (!lhSurface || !rhSurface) {
104 for (auto it = surfaces.begin(); it != surfaces.end(); ++it) {
105 if (it.value() && it.value()->tissueType() == BrainSurface::TissueBrain) {
106 if (!lhSurface && it.key().startsWith("lh_")) {
107 lhSurface = it.value().get();
108 qDebug() << "SourceEstimateManager: Using fallback LH surface:" << it.key();
109 } else if (!rhSurface && it.key().startsWith("rh_")) {
110 rhSurface = it.value().get();
111 qDebug() << "SourceEstimateManager: Using fallback RH surface:" << it.key();
112 }
113 }
114 }
115 }
116
117 if (!lhSurface && !rhSurface) {
118 qWarning() << "SourceEstimateManager: No surfaces available for STC loading."
119 << "Active surface type:" << activeSurfaceType
120 << "Available keys:" << surfaces.keys();
121 return false;
122 }
123
124 // Clean up any previous loading thread
125 if (m_loadingThread) {
126 m_loadingThread->quit();
127 m_loadingThread->wait();
128 delete m_loadingThread;
129 m_loadingThread = nullptr;
130 }
131
132 // Create overlay for results
133 m_overlay = std::make_unique<SourceEstimateOverlay>();
134
135 // Create worker and thread
136 m_loadingThread = new QThread(this);
137 m_stcWorker = new StcLoadingWorker(lhPath, rhPath, lhSurface, rhSurface);
138 m_stcWorker->moveToThread(m_loadingThread);
139
140 connect(m_loadingThread, &QThread::started, m_stcWorker, &StcLoadingWorker::process);
141 connect(m_stcWorker, &StcLoadingWorker::progress, this, &SourceEstimateManager::loadingProgress);
142 connect(m_stcWorker, &StcLoadingWorker::finished, this, &SourceEstimateManager::onStcLoadingFinished);
143 connect(m_stcWorker, &StcLoadingWorker::finished, m_loadingThread, &QThread::quit);
144 connect(m_loadingThread, &QThread::finished, m_stcWorker, &QObject::deleteLater);
145
146 m_isLoading = true;
147 m_loadingThread->start();
148
149 return true;
150}

161void SourceEstimateManager::onStcLoadingFinished(bool success)
162{
163 m_isLoading = false;
164
165 if (!success || !m_stcWorker) {
166 qWarning() << "SourceEstimateManager: Async STC loading failed";
167 m_overlay.reset();
168 return;
169 }
170
171 // Transfer data from worker to overlay
172 if (m_stcWorker->hasLh()) {
173 m_overlay->setStcData(m_stcWorker->stcLh(), 0);
174 if (m_stcWorker->interpolationMatLh())
175 m_overlay->setInterpolationMatrix(m_stcWorker->interpolationMatLh(), 0);
176 }
177
178 if (m_stcWorker->hasRh()) {
179 m_overlay->setStcData(m_stcWorker->stcRh(), 1);
180 if (m_stcWorker->interpolationMatRh())
181 m_overlay->setInterpolationMatrix(m_stcWorker->interpolationMatRh(), 1);
182 }
183
184 m_overlay->updateThresholdsFromData();
185 emit thresholdsUpdated(m_overlay->thresholdMin(),
186 m_overlay->thresholdMid(),
187 m_overlay->thresholdMax());
188
189 if (m_overlay->isLoaded()) {
190 emit loaded(m_overlay->numTimePoints());
191 } else {
192 m_overlay.reset();
193 }
194}

198void SourceEstimateManager::setTimePoint(int index,
199 const QMap<QString, std::shared_ptr<BrainSurface>> &surfaces,
200 const SubView &singleView,
201 const QVector<SubView> &subViews)
202{
203 if (!m_overlay || !m_overlay->isLoaded()) return;
204
205 m_currentTimePoint = qBound(0, index, m_overlay->numTimePoints() - 1);
206
207 // Collect all distinct surface types used across single + multi views
208 QSet<QString> activeTypes;
209 activeTypes.insert(singleView.surfaceType);
210 for (int i = 0; i < subViews.size(); ++i)
211 activeTypes.insert(subViews[i].surfaceType);
212
213 // Apply source estimate to surfaces matching ANY active type
214 for (auto it = surfaces.begin(); it != surfaces.end(); ++it) {
215 for (const QString &type : activeTypes) {
216 if (it.key().endsWith(type)) {
217 m_overlay->applyToSurface(it.value().get(), m_currentTimePoint);
218 break;
219 }
220 }
221 }
222
223 emit timePointChanged(m_currentTimePoint, m_overlay->timeAtIndex(m_currentTimePoint));
224}

249int SourceEstimateManager::closestIndex(float timeSec) const
250{
251 if (!m_overlay || !m_overlay->isLoaded()) return -1;
252
253 const float t0 = m_overlay->tmin();
254 const float dt = m_overlay->tstep();
255 const int numPts = m_overlay->numTimePoints();
256 if (numPts <= 0 || dt <= 0.0f) return -1;
257
258 const int idx = qRound((timeSec - t0) / dt);
259 return qBound(0, idx, numPts - 1);
260}

283void SourceEstimateManager::startStreaming(const QMap<QString, std::shared_ptr<BrainSurface>> &surfaces,
284 const SubView &singleView,
285 const QVector<SubView> &subViews)
286{
287 Q_UNUSED(surfaces)
288 Q_UNUSED(singleView)
289 Q_UNUSED(subViews)
290
291 if (m_isStreaming) {
292 qDebug() << "SourceEstimateManager: Real-time streaming already active";
293 return;
294 }
295
296 if (!m_overlay || !m_overlay->isLoaded()) {
297 qWarning() << "SourceEstimateManager: Cannot start streaming — no source estimate loaded";
298 return;
299 }
300
301 // Create controller on first use
302 if (!m_rtController) {
303 m_rtController = std::make_unique<RtSourceDataController>(this);
304 connect(m_rtController.get(), &RtSourceDataController::newSmoothedDataAvailable,
305 this, &SourceEstimateManager::realtimeColorsAvailable);
306 }
307
308 // Propagate interpolation matrices from the overlay
309 m_rtController->setInterpolationMatrixLeft(m_overlay->interpolationMatLh());
310 m_rtController->setInterpolationMatrixRight(m_overlay->interpolationMatRh());
311
312 // Propagate current visualization parameters
313 m_rtController->setColormapType(m_overlay->colormap());
314 m_rtController->setThresholds(m_overlay->thresholdMin(),
315 m_overlay->thresholdMid(),
316 m_overlay->thresholdMax());
317 m_rtController->setSFreq(1.0 / m_overlay->tstep());
318
319 // Feed all STC time-points into the queue
320 const int nTimePoints = m_overlay->numTimePoints();
321 qDebug() << "SourceEstimateManager: Feeding" << nTimePoints << "time points into real-time queue";
322 m_rtController->clearData();
323
324 for (int t = 0; t < nTimePoints; ++t) {
325 Eigen::VectorXd col = m_overlay->sourceDataColumn(t);
326 if (col.size() > 0)
327 m_rtController->addData(col);
328 }
329
330 m_rtController->setStreamingState(true);
331 m_isStreaming = true;
332
333 qDebug() << "SourceEstimateManager: Real-time streaming started";
334}
