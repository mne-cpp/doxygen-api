96 void openStream()
97 {
98 if (m_socket.is_open()) {
99 return;
100 }
101 const std::string& host = m_info.data_host();
102 const int port = m_info.data_port();
103 if (host.empty() || port <= 0 || port > 65535) {
104 throw failure("Invalid data host or port in stream_info");
105 }
106 if (m_info.channel_format() == ChannelFormat::Undefined || m_info.channel_count() <= 0) {
107 throw failure("stream_info has no channel format or channels");
108 }
109 auto connection = TcpSocket::connect(host, static_cast<std::uint16_t>(port), kConnect);
110 if (!connection) {
111 throw failure("Failed to connect to outlet: " + connection.error().message());
112 }
113 m_socket = std::move(*connection);
114 // The outlet buffers this many samples for us (liblsl: 360 s, or 36000 samples for irregular streams).
115 const auto bufferSamples = static_cast<long long>(m_info.nominal_srate() > 0.0 ? 360.0 * m_info.nominal_srate() : 36000.0);
116 const std::string request = "LSL:streamfeed/110 " + m_info.uid() + "\r\nNative-Byte-Order: " + byteOrderName() +
117 "\r\nEndian-Performance: 0\r\nHas-IEEE754-Floats: 1\r\nSupports-Subnormals: 1\r\nValue-Size: " +
118 std::to_string(valueSize(m_info.channel_format())) + "\r\nData-Protocol-Version: 110\r\nMax-Buffer-Length: " +
119 std::to_string(bufferSamples) + "\r\nMax-Chunk-Length: 0\r\nHostname: " + host_name() + "\r\nSource-Id: " +
120 m_info.source_id() + "\r\nSession-Id: default\r\n\r\n";
121 try {
122 if (const std::error_code error = m_socket.write(bytes(request))) {
123 throw failure("Failed to send the request: " + error.message());
124 }
125 readHeaders();
126 checkTestPatterns();
127 } catch (...) {
128 closeStream();
129 throw;
130 }
131 }

152 template<typename T>
153 std::vector<std::vector<T>> pullChunk(std::vector<double>* timestamps)
154 {
155 std::vector<std::vector<T>> chunk;
156 if (timestamps != nullptr) {
157 timestamps->clear();
158 }
159 if (!m_socket.is_open()) {
160 return chunk;
161 }
162 receive(std::chrono::milliseconds(0));
163 std::uint8_t tag = 0;
164 double timestamp = 0.0;
165 int result = 0;
166 while ((result = m_reader.read(m_buffer, m_pos, tag, timestamp, m_values)) == 1) {
167 m_lastTimestamp = stamp(tag, timestamp);
168 chunk.emplace_back(m_values.begin(), m_values.end());
169 if (timestamps != nullptr) {
170 timestamps->push_back(m_lastTimestamp);
171 }
172 }
173 if (result < 0) {
174 mne::Log::warning("[lsl::stream_inlet] Malformed sample from the outlet; closing the stream");
175 closeStream();
176 return chunk;
177 }
178 if (m_pos > (1 << 20)) {
179 m_buffer.erase(0, m_pos);
180 m_pos = 0;
181 }
182 return chunk;
183 }

185 stream_info fullInfo(double timeout)
186 {
187 if (m_hasFullInfo) {
188 return m_info;
189 }
190 const std::string& host = m_info.data_host();
191 const int port = m_info.data_port();
192 if (host.empty() || port <= 0 || port > 65535) {
193 throw failure("Invalid data host or port in stream_info");
194 }
195 const Clock::time_point deadline = Clock::now() + std::chrono::duration_cast<Clock::duration>(std::chrono::duration<double>(timeout));
196 auto connection = TcpSocket::connect(host, static_cast<std::uint16_t>(port), left(deadline));
197 if (!connection) {
198 throw failure("Failed to connect to outlet: " + connection.error().message());
199 }
200 if (const std::error_code error = connection->write(bytes("LSL:fullinfo\r\n"))) {
201 throw failure("Failed to send the fullinfo request: " + error.message());
202 }
203 // The outlet sends the XML and closes the connection.
204 std::string xml;
205 std::array<std::byte, 65536> chunk{};
206 while (xml.find("</info>") == std::string::npos) {
207 const auto got = connection->read_some(chunk, left(deadline));
208 if (!got) {
209 break;
210 }
211 xml.append(reinterpret_cast<const char*>(chunk.data()), *got);
212 }
213 const stream_info full = stream_info::from_string(xml);
214 if (full.name().empty()) {
215 throw failure("The outlet sent no valid stream info");
216 }
217 m_info.set_channels(full.channels());
218 m_hasFullInfo = true;
219 return m_info;
220 }

222 double timeCorrection(double timeout)
223 {
224 const int port = m_info.service_port();
225 if (port <= 0 || port > 65535 || m_info.data_host().empty()) {
226 return std::numeric_limits<double>::quiet_NaN();
227 }
228 auto socket = detail::bindQuerySocket(0);
229 if (!socket) {
230 return std::numeric_limits<double>::quiet_NaN();
231 }
232 // The estimate is off by at most half the round trip of the probe it uses. A loaded host can delay
233 // every reply of a short burst, so probing goes on past the first few until one returns promptly.
234 constexpr int kMinProbes = 8;
235 constexpr int kMaxProbes = 64;
236 constexpr double kPromptRtt = 1e-3;
237 double bestRtt = std::numeric_limits<double>::infinity();
238 double best = std::numeric_limits<double>::quiet_NaN();
239 const Clock::time_point deadline = Clock::now() + std::chrono::duration_cast<Clock::duration>(std::chrono::duration<double>(timeout));
240 std::array<std::byte, 512> buffer{};
241 for (int wave = 0; wave < kMaxProbes && Clock::now() < deadline && (wave < kMinProbes || bestRtt > kPromptRtt); ++wave) {
242 const double t0 = local_clock();
243 const std::string probe = std::format("LSL:timedata\r\n{} {:.17g}\r\n", wave, t0);
244 if (socket->send_to(bytes(probe), m_info.data_host(), static_cast<std::uint16_t>(port))) {
245 break;
246 }
247 const Clock::time_point waitUntil = std::min(deadline, Clock::now() + std::chrono::milliseconds(100));
248 bool answered = false;
249 while (!answered && Clock::now() < waitUntil) {
250 const auto datagram = socket->receive_from(buffer, left(waitUntil));
251 if (!datagram) {
252 continue;
253 }
254 const double t3 = local_clock();
255 const std::vector<std::string> reply = split(trimmed(std::string_view(reinterpret_cast<const char*>(buffer.data()), datagram->size)), ' ');
256 if (reply.size() != 4 || to_int(reply[0]) != wave) {
257 continue;
258 }
259 const double t1 = to_double(reply[2]).value_or(std::numeric_limits<double>::quiet_NaN());
260 const double t2 = to_double(reply[3]).value_or(std::numeric_limits<double>::quiet_NaN());
261 const double rtt = (t3 - t0) - (t2 - t1);
262 if (rtt < bestRtt) {
263 bestRtt = rtt;
264 best = -((t1 - t0) + (t2 - t3)) / 2.0; // remote clock minus local clock, negated
265 }
266 answered = true;
267 }
268 }
269 return best;
270 }

274 bool receive(std::chrono::milliseconds timeout)
275 {
276 std::array<std::byte, 65536> chunk{};
277 bool any = false;
278 while (true) {
279 const auto got = m_socket.read_some(chunk, any ? std::chrono::milliseconds(0) : timeout);
280 if (!got) {
281 if (got.error() != std::errc::timed_out) {
282 m_socket.close();
283 }
284 return any;
285 }
286 m_buffer.append(reinterpret_cast<const char*>(chunk.data()), *got);
287 any = true;
288 }
289 }

292 void readHeaders()
293 {
294 const Clock::time_point deadline = Clock::now() + kConnect;
295 std::size_t end = std::string::npos;
296 while ((end = m_buffer.find("\r\n\r\n")) == std::string::npos) {
297 if (Clock::now() >= deadline || m_buffer.size() > 65536 || !m_socket.is_open() || !receive(left(deadline))) {
298 throw failure("Timeout waiting for the outlet's response");
299 }
300 }
301 const std::vector<std::string> lines = split(std::string_view(m_buffer).substr(0, end), '\n');
302 const std::vector<std::string> status = split(trimmed(lines.front()), ' ');
303 if (status.size() < 2 || !status[0].starts_with("LSL/") || status[1] != "200") {
304 throw failure("Outlet refused the stream: " + std::string(trimmed(lines.front())));
305 }
306 bool reverse = false;
307 for (const std::string& line : lines) {
308 const std::size_t colon = line.find(':');
309 if (colon == std::string::npos) {
310 continue;
311 }
312 const std::string key(trimmed(std::string_view(line).substr(0, colon)));
313 const std::string value(trimmed(std::string_view(line).substr(colon + 1)));
314 if (key == "Byte-Order") {
315 reverse = value != byteOrderName();
316 } else if (key == "Data-Protocol-Version" && to_int(value) != kProtocolVersion) {
317 throw failure("Outlet speaks data protocol " + value + "; 110 is supported");
318 }
319 }
320 m_reader = SampleReader(m_info.channel_format(), m_info.channel_count(), reverse);
321 m_buffer.erase(0, end + 4);
322 m_pos = 0;
323 }

326 void checkTestPatterns()
327 {
328 const Clock::time_point deadline = Clock::now() + kConnect;
329 for (const int offset : {4, 2}) {
330 std::uint8_t tag = 0;
331 double timestamp = 0.0;
332 int result = 0;
333 while ((result = m_reader.read(m_buffer, m_pos, tag, timestamp, m_values)) == 0) {
334 if (Clock::now() >= deadline || !m_socket.is_open() || !receive(left(deadline))) {
335 throw failure("Timeout waiting for the test-pattern samples");
336 }
337 }
338 bool match = result == 1 && tag == kTagTransmitted && timestamp == kTestPatternTime;
339 for (std::size_t k = 0; match && k < m_values.size(); ++k) {
340 const double expected = testPatternValue(m_info.channel_format(), static_cast<int>(k), offset);
341 match = m_info.channel_format() == ChannelFormat::Float32 ? static_cast<float>(m_values[k]) == static_cast<float>(expected)
342 : m_values[k] == expected;
343 }
344 if (!match) {
345 throw failure("The outlet's test-pattern samples do not match; the protocol formats are incompatible");
346 }
347 }
348 }

350 double stamp(std::uint8_t tag, double timestamp) const noexcept
351 {
352 if (tag == kTagTransmitted) {
353 return timestamp;
354 }
355 return m_info.nominal_srate() > 0.0 ? m_lastTimestamp + 1.0 / m_info.nominal_srate() : m_lastTimestamp;
356 }
